混合检索

稠密检索的局限

纯稠密检索（Dense Retrieval）依赖嵌入模型的语义理解能力，但在某些场景下表现不佳：精确关键词匹配、专业术语查询、罕见词汇等。混合检索（Hybrid Search）结合稠密检索和稀疏检索的优势，提供更鲁棒的检索效果。

BM25 稀疏检索
BM25 是经典的基于词频的检索算法，它考虑了词频（TF）、逆文档频率（IDF）和文档长度归一化：


BM25(q,d)=∑t∈qIDF(t)⋅f(t,d)⋅(k1+1)f(t,d)+k1⋅(1−b+b⋅∣d∣avgdl)\text{BM25}(q, d) = \sum_{t \in q} \text{IDF}(t) \cdot \frac{f(t, d) \cdot (k_1 + 1)}{f(t, d) + k_1 \cdot (1 - b + b \cdot \frac{|d|}{avgdl})}BM25(q,d)=t∈q∑​IDF(t)⋅f(t,d)+k1​⋅(1−b+b⋅avgdl∣d∣​)f(t,d)⋅(k1​+1)​
import math
from collections import Counter

class BM25:
    def __init__(self, documents, k1=1.5, b=0.75):
        self.k1 = k1
        self.b = b
        self.docs = [doc.split() for doc in documents]
        self.n = len(self.docs)
        self.avgdl = sum(len(d) for d in self.docs) / self.n
        self.df = {}
        for doc in self.docs:
            for term in set(doc):
                self.df[term] = self.df.get(term, 0) + 1

    def idf(self, term):
        df = self.df.get(term, 0)
        return math.log((self.n - df + 0.5) / (df + 0.5) + 1)

    def score(self, query, doc_idx):
        doc = self.docs[doc_idx]
        tf = Counter(doc)
        dl = len(doc)
        score = 0
        for term in query.split():
            f = tf.get(term, 0)
            idf = self.idf(term)
            numerator = f * (self.k1 + 1)
            denominator = f + self.k1 * (1 - self.b + self.b * dl / self.avgdl)
            score += idf * numerator / denominator
        return score

    def search(self, query, top_k=3):
        scores = [(i, self.score(query, i)) for i in range(self.n)]
        scores.sort(key=lambda x: x[1], reverse=True)
        return scores[:top_k]

documents = [
    "Transformer 架构 使用 自注意力 机制",
    "BM25 是 经典 的 文本 检索 算法",
    "向量 数据库 存储 嵌入 向量",
    "BERT 使用 掩码 语言 建模 预训练",
    "RAG 结合 检索 和 生成 技术",
]

bm25 = BM25(documents)
results = bm25.search("检索 算法")
for idx, score in results:
    print(f"  [{score:.3f}] {documents[idx]}")


融合策略
线性加权融合
def linear_fusion(dense_scores, sparse_scores, alpha=0.5):
    all_docs = set(dense_scores.keys()) | set(sparse_scores.keys())

    dense_max = max(dense_scores.values()) if dense_scores else 1
    sparse_max = max(sparse_scores.values()) if sparse_scores else 1

    fused = {}
    for doc_id in all_docs:
        d_score = dense_scores.get(doc_id, 0) / dense_max
        s_score = sparse_scores.get(doc_id, 0) / sparse_max
        fused[doc_id] = alpha * d_score + (1 - alpha) * s_score

    return sorted(fused.items(), key=lambda x: x[1], reverse=True)

dense = {"doc1": 0.9, "doc2": 0.7, "doc3": 0.3}
sparse = {"doc2": 0.8, "doc3": 0.6, "doc4": 0.9}

for alpha in [0.3, 0.5, 0.7]:
    results = linear_fusion(dense, sparse, alpha=alpha)
    top = results[0]
    print(f"alpha={alpha}: 最佳 = {top[0]} (分数: {top[1]:.3f})")
倒数排名融合（RRF）
RRF（Reciprocal Rank Fusion）是一种不依赖分数归一化的融合方法，只使用排名信息：


RRF(d)=∑r∈R1k+r(d)\text{RRF}(d) = \sum_{r \in R} \frac{1}{k + r(d)}RRF(d)=r∈R∑​k+r(d)1​
def reciprocal_rank_fusion(rankings, k=60):
    fused_scores = {}
    for ranking in rankings:
        for rank, (doc_id, _) in enumerate(ranking):
            if doc_id not in fused_scores:
                fused_scores[doc_id] = 0
            fused_scores[doc_id] += 1 / (k + rank + 1)

    return sorted(fused_scores.items(), key=lambda x: x[1], reverse=True)

dense_ranking = [("doc1", 0.9), ("doc2", 0.7), ("doc3", 0.3)]
sparse_ranking = [("doc2", 0.8), ("doc4", 0.6), ("doc1", 0.4)]

results = reciprocal_rank_fusion([dense_ranking, sparse_ranking])
for doc_id, score in results:
    print(f"  {doc_id}: RRF 分数 = {score:.4f}")


提示： RRF 是实践中最常用的融合方法，因为它不需要对不同检索器的分数进行归一化，且对超参数 kkk 不敏感（默认 60 即可）。

def rerank_pipeline(query, candidates, reranker, top_k=5):
    scored = []
    for doc in candidates:
        score = reranker.predict([(query, doc["content"])])[0]
        scored.append({**doc, "rerank_score": score})

    scored.sort(key=lambda x: x["rerank_score"], reverse=True)
    return scored[:top_k]

完整混合检索 Pipeline
class HybridSearchPipeline:
    def __init__(self, dense_retriever, sparse_retriever, reranker, fusion="rrf"):
        self.dense = dense_retriever
        self.sparse = sparse_retriever
        self.reranker = reranker
        self.fusion = fusion

    def search(self, query, retrieve_k=20, final_k=5):
        dense_results = self.dense.search(query, top_k=retrieve_k)
        sparse_results = self.sparse.search(query, top_k=retrieve_k)

        if self.fusion == "rrf":
            fused = reciprocal_rank_fusion([dense_results, sparse_results])
        else:
            fused = linear_fusion(
                dict(dense_results), dict(sparse_results)
            )

        candidates = fused[:retrieve_k]
        reranked = self.reranker.rerank(query, candidates, top_k=final_k)
        return reranked

print("完整 Pipeline: 稠密检索 + 稀疏检索 → RRF 融合 → 重排序 → Top-K")


小结
    

  • 混合检索结合稠密（语义）和稀疏（关键词）检索的优势，比单一方法更鲁棒。
    • 

  • BM25 是最经典的稀疏检索算法，在精确匹配和专业术语查询上表现优异。
    • 

  • RRF 是最实用的融合策略，无需分数归一化且对超参数不敏感。
    • 

  • 重排序（Cross-Encoder）能显著提升最终检索精度，是生产环境的必备环节。
    • 

  • 完整的混合检索 Pipeline：稠密 + 稀疏 → 融合 → 重排序 → Top-K。
