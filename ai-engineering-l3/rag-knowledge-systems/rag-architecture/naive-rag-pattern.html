Naive RAG 模式

核心概念

Naive RAG（朴素 RAG）是最基础的检索增强生成架构，由"索引-检索-生成"三个阶段组成。它是所有高级 RAG 变体（Advanced RAG、Modular RAG、Agentic RAG）的起点和基准线。理解 Naive RAG 的工作原理、能力边界和局限性，是掌握整个 RAG 技术体系的第一步。

为什么从 Naive RAG 开始

特性 说明
概念清晰 三阶段流程简洁易懂，适合理解 RAG 核心思想
快速原型 可以在数小时内搭建一个可运行的系统
基准参照 所有优化的效果都以 Naive RAG 为基准衡量
组件解耦 每个组件可以独立替换和优化
成本可控 最少的计算资源和开发成本

TypeScript 实现
import OpenAI from 'openai';
import { QdrantClient } from '@qdrant/js-client-rest';

interface RAGConfig {
  collectionName: string;
  embeddingModel: string;
  llmModel: string;
  topK: number;
  chunkSize: number;
  chunkOverlap: number;
}

interface RAGResponse {
  answer: string;
  sources: Array<{
    text: string;
    score: number;
    source: string;
  }>;
  tokensUsed: number;
}

class NaiveRAG {
  private openai: OpenAI;
  private qdrant: QdrantClient;
  private config: RAGConfig;

  constructor(config: RAGConfig) {
    this.openai = new OpenAI();
    this.qdrant = new QdrantClient({ url: 'http://localhost:6333' });
    this.config = config;
  }

  async indexDocuments(documents: Array<{ text: string; source: string }>) {
    const chunks = documents.flatMap((doc) => this.chunkText(doc.text, doc.source));

    const batchSize = 50;
    for (let i = 0; i < chunks.length; i += batchSize) {
      const batch = chunks.slice(i, i + batchSize);
      const texts = batch.map((c) => c.text);

      const embeddingResponse = await this.openai.embeddings.create({
        model: this.config.embeddingModel,
        input: texts,
      });

      const points = batch.map((chunk, j) => ({
        id: crypto.randomUUID(),
        vector: embeddingResponse.data[j].embedding,
        payload: {
          text: chunk.text,
          source: chunk.source,
          chunkIndex: chunk.index,
        },
      }));

      await this.qdrant.upsert(this.config.collectionName, {
        wait: true,
        points,
      });
    }
  }

  async query(question: string): Promise<RAGResponse> {
    const queryEmbedding = await this.openai.embeddings.create({
      model: this.config.embeddingModel,
      input: question,
    });

    const searchResults = await this.qdrant.search(this.config.collectionName, {
      vector: queryEmbedding.data[0].embedding,
      limit: this.config.topK,
      with_payload: true,
    });

    const context = searchResults.map((r, i) => `[${i + 1}] ${(r.payload as any).text}`).join('\n\n');

    const prompt = this.buildPrompt(question, context);

    const completion = await this.openai.chat.completions.create({
      model: this.config.llmModel,
      messages: [
        {
          role: 'system',
          content:
            '你是一个知识问答助手。根据提供的参考资料回答问题。' + '如果资料中没有答案，请明确说明。不要编造信息。',
        },
        { role: 'user', content: prompt },
      ],
      temperature: 0.1,
    });

    return {
      answer: completion.choices[0].message.content || '',
      sources: searchResults.map((r) => ({
        text: (r.payload as any).text,
        score: r.score,
        source: (r.payload as any).source,
      })),
      tokensUsed: completion.usage?.total_tokens || 0,
    };
  }

  private chunkText(text: string, source: string): Array<{ text: string; source: string; index: number }> {
    const chunks: Array<{ text: string; source: string; index: number }> = [];
    const step = this.config.chunkSize - this.config.chunkOverlap;
    let index = 0;

    for (let i = 0; i < text.length; i += step) {
      const chunk = text.slice(i, i + this.config.chunkSize);
      if (chunk.trim().length > 0) {
        chunks.push({ text: chunk, source, index });
        index++;
      }
    }

    return chunks;
  }

  private buildPrompt(question: string, context: string): string {
    return `请根据以下参考资料回答问题。\n\n参考资料：\n${context}\n\n问题：${question}`;
  }
}

const rag = new NaiveRAG({
  collectionName: 'knowledge_base',
  embeddingModel: 'text-embedding-3-small',
  llmModel: 'gpt-4o-mini',
  topK: 5,
  chunkSize: 512,
  chunkOverlap: 100,
});
Python 实现
from openai import OpenAI
from qdrant_client import QdrantClient
from qdrant_client.models import PointStruct, VectorParams, Distance
from dataclasses import dataclass
import uuid

@dataclass
class RAGResponse:
    answer: str
    sources: list[dict]
    tokens_used: int

class NaiveRAG:
    def __init__(
        self,
        collection_name: str = "knowledge_base",
        embedding_model: str = "text-embedding-3-small",
        llm_model: str = "gpt-4o-mini",
        top_k: int = 5,
        chunk_size: int = 512,
        chunk_overlap: int = 100,
    ):
        self.openai = OpenAI()
        self.qdrant = QdrantClient(url="http://localhost:6333")
        self.collection_name = collection_name
        self.embedding_model = embedding_model
        self.llm_model = llm_model
        self.top_k = top_k
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap

    def setup_collection(self, dimension: int = 1536):
        self.qdrant.recreate_collection(
            collection_name=self.collection_name,
            vectors_config=VectorParams(
                size=dimension, distance=Distance.COSINE
            ),
        )

    def index_documents(self, documents: list[dict]):
        all_chunks = []
        for doc in documents:
            chunks = self._chunk_text(doc["text"], doc["source"])
            all_chunks.extend(chunks)

        batch_size = 50
        for i in range(0, len(all_chunks), batch_size):
            batch = all_chunks[i : i + batch_size]
            texts = [c["text"] for c in batch]

            response = self.openai.embeddings.create(
                model=self.embedding_model, input=texts
            )

            points = [
                PointStruct(
                    id=str(uuid.uuid4()),
                    vector=response.data[j].embedding,
                    payload=batch[j],
                )
                for j in range(len(batch))
            ]

            self.qdrant.upsert(
                collection_name=self.collection_name, points=points
            )

    def query(self, question: str) -> RAGResponse:
        query_response = self.openai.embeddings.create(
            model=self.embedding_model, input=question
        )
        query_vector = query_response.data[0].embedding

        results = self.qdrant.search(
            collection_name=self.collection_name,
            query_vector=query_vector,
            limit=self.top_k,
            with_payload=True,
        )

        context_parts = []
        for i, r in enumerate(results, 1):
            context_parts.append(f"[{i}] {r.payload['text']}")
        context = "\n\n".join(context_parts)

        completion = self.openai.chat.completions.create(
            model=self.llm_model,
            messages=[
                {
                    "role": "system",
                    "content": (
                        "你是一个知识问答助手。根据提供的参考资料回答问题。"
                        "如果资料中没有答案，请明确说明。不要编造信息。"
                    ),
                },
                {
                    "role": "user",
                    "content": f"参考资料：\n{context}\n\n问题：{question}",
                },
            ],
            temperature=0.1,
        )

        return RAGResponse(
            answer=completion.choices[0].message.content or "",
            sources=[
                {"text": r.payload["text"], "score": r.score, "source": r.payload["source"]}
                for r in results
            ],
            tokens_used=completion.usage.total_tokens if completion.usage else 0,
        )

    def _chunk_text(self, text: str, source: str) -> list[dict]:
        chunks = []
        step = self.chunk_size - self.chunk_overlap
        idx = 0

        i = 0
        while i < len(text):
            chunk = text[i : i + self.chunk_size]
            if chunk.strip():
                chunks.append({"text": chunk, "source": source, "chunk_index": idx})
                idx += 1
            i += step

        return chunks

常见误区
❌ 误区 1：Naive RAG 效果不好就放弃
Naive RAG 效果不好通常是因为配置不当，而非架构本身的问题：


    

  • 分块太大/太小 → 调整分块参数
    • 

  • 嵌入模型不合适 → 换更好的模型
    • 

  • 提示词不好 → 优化系统提示词
    • 

  • Top-K 太少 → 增加检索数量
    • 



先把 Naive RAG 调到最优，再考虑升级。


❌ 误区 2：一步到位上 Agentic RAG
复杂架构带来更多的故障点和调试难度：


    

  • 先从 Naive RAG 开始，建立效果基线
    • 

  • 逐步添加优化，每步衡量效果增量
    • 

  • 只有当简单方案无法满足需求时才升级
    • 



❌ 误区 3：忽视提示词的重要性
在 Naive RAG 中，提示词对最终效果的影响巨大（可能占 20-30%）：


    

  • 明确告诉 LLM 只根据提供的上下文回答
    • 

  • 要求不知道时说"不知道"
    • 

  • 提供输出格式约束
    • 

  • 适当的 few-shot 示例
    • 



❌ 误区 4：不做评估就优化
没有评估基线，无法判断优化是否有效：


    

  • 先准备 50-100 条评估数据（查询 + 标准答案）
    • 

  • 建立 Naive RAG 的基线指标
    • 

  • 每次优化后重新评估
    • 



小结
Naive RAG 是 RAG 技术体系的基础架构，由索引、检索、生成三个阶段组成。它概念简洁、实现快速，是学习 RAG 和快速验证想法的最佳起点。


核心要点回顾：


    

  • Naive RAG 的三阶段：离线索引（加载→分块→向量化→存储）→ 在线检索（向量化查询→ANN 搜索→Top-K）→ 生成（组装提示词→LLM 回答）
    • 

  • 适合学习、快速原型、简单场景；不适合高质量要求的生产系统
    • 

  • 主要局限在于：检索不精确、缺乏查询改写、无重排序、不支持多步推理
    • 

  • 提示词设计对 Naive RAG 效果影响巨大，必须包含"不知道就说不知道"的约束
    • 

  • 先把 Naive RAG 调到最优（分块、模型、Top-K、提示词），再考虑升级到 Advanced RAG
    • 

  • 每步升级都要有评估数据支撑，避免盲目增加复杂度
    •
