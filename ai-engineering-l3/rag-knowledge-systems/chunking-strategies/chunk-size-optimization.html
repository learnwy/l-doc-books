分块大小优化

核心概念

分块大小优化（Chunk Size Optimization）是 RAG 系统中最影响检索质量的超参数调优之一。它的本质是在两个相互矛盾的目标之间找到最佳平衡点：语义完整性（需要更大的块）和检索精确性（需要更小的块）。

TypeScript 重叠分块实现
interface ChunkConfig {
  chunkSize: number;
  chunkOverlap: number;
  separators: string[];
}

interface TextChunk {
  text: string;
  startOffset: number;
  endOffset: number;
  tokenCount: number;
  overlapTokens: number;
}

function chunkWithOverlap(text: string, config: ChunkConfig): TextChunk[] {
  const chunks: TextChunk[] = [];
  const tokens = tokenize(text);
  const step = config.chunkSize - config.chunkOverlap;

  for (let i = 0; i < tokens.length; i += step) {
    const end = Math.min(i + config.chunkSize, tokens.length);
    const chunkTokens = tokens.slice(i, end);
    const chunkText = detokenize(chunkTokens);

    const adjustedText = adjustToBoundary(chunkText, config.separators);

    chunks.push({
      text: adjustedText,
      startOffset: i,
      endOffset: end,
      tokenCount: chunkTokens.length,
      overlapTokens: i > 0 ? config.chunkOverlap : 0,
    });

    if (end >= tokens.length) break;
  }

  return chunks;
}

function adjustToBoundary(text: string, separators: string[]): string {
  for (const sep of separators) {
    const lastIdx = text.lastIndexOf(sep);
    if (lastIdx > text.length * 0.7) {
      return text.slice(0, lastIdx + sep.length);
    }
  }
  return text;
}

系统化的分块大小评估
Python 自动化评估框架
import numpy as np
from dataclasses import dataclass
from sentence_transformers import SentenceTransformer

@dataclass
class ChunkSizeConfig:
    chunk_size: int
    chunk_overlap: int
    name: str

@dataclass
class EvalMetrics:
    config_name: str
    hit_rate_at_5: float
    mrr_at_10: float
    avg_chunk_count: float
    avg_relevant_rank: float
    context_utilization: float

class ChunkSizeOptimizer:
    def __init__(self, model_name: str = "BAAI/bge-base-zh-v1.5"):
        self.model = SentenceTransformer(model_name)

    def evaluate_config(
        self,
        config: ChunkSizeConfig,
        documents: list[str],
        eval_queries: list[str],
        ground_truth: list[list[str]],
    ) -> EvalMetrics:
        chunks = []
        for doc in documents:
            doc_chunks = self._chunk_document(doc, config)
            chunks.extend(doc_chunks)

        chunk_embeddings = self.model.encode(
            chunks, normalize_embeddings=True, batch_size=64
        )
        query_embeddings = self.model.encode(
            eval_queries, normalize_embeddings=True, batch_size=64
        )

        similarities = np.dot(query_embeddings, chunk_embeddings.T)

        hits = 0
        mrr_sum = 0.0
        rank_sum = 0.0

        for i, query in enumerate(eval_queries):
            top_indices = np.argsort(similarities[i])[::-1][:10]
            top_chunks = [chunks[idx] for idx in top_indices]

            for rank, chunk in enumerate(top_chunks[:5], 1):
                if any(gt in chunk for gt in ground_truth[i]):
                    hits += 1
                    break

            for rank, chunk in enumerate(top_chunks, 1):
                if any(gt in chunk for gt in ground_truth[i]):
                    mrr_sum += 1.0 / rank
                    rank_sum += rank
                    break

        n = len(eval_queries)
        return EvalMetrics(
            config_name=config.name,
            hit_rate_at_5=hits / n,
            mrr_at_10=mrr_sum / n,
            avg_chunk_count=len(chunks) / len(documents),
            avg_relevant_rank=rank_sum / n if rank_sum > 0 else float("inf"),
            context_utilization=config.chunk_size * 5 / 4096,
        )

    def _chunk_document(self, doc: str, config: ChunkSizeConfig) -> list[str]:
        words = doc.split()
        chunks = []
        step = config.chunk_size - config.chunk_overlap

        i = 0
        while i < len(words):
            end = min(i + config.chunk_size, len(words))
            chunk = " ".join(words[i:end])
            chunks.append(chunk)
            i += step

        return chunks

    def compare_configs(
        self,
        configs: list[ChunkSizeConfig],
        documents: list[str],
        eval_queries: list[str],
        ground_truth: list[list[str]],
    ) -> list[EvalMetrics]:
        results = []
        for config in configs:
            metrics = self.evaluate_config(
                config, documents, eval_queries, ground_truth
            )
            results.append(metrics)

        results.sort(key=lambda x: x.hit_rate_at_5, reverse=True)
        return results

configs = [
    ChunkSizeConfig(256, 50, "small-256"),
    ChunkSizeConfig(512, 100, "medium-512"),
    ChunkSizeConfig(1024, 200, "large-1024"),
    ChunkSizeConfig(2048, 400, "xlarge-2048"),
]

optimizer = ChunkSizeOptimizer()
results = optimizer.compare_configs(configs, documents, queries, ground_truth)

for r in results:
    print(f"{r.config_name}: HR@5={r.hit_rate_at_5:.3f} "
          f"MRR@10={r.mrr_at_10:.3f} "
          f"Chunks/Doc={r.avg_chunk_count:.1f}")
多级分块（Multi-Granularity Chunking）
多级分块同时用多种粒度索引同一文档，检索时融合不同粒度的结果。这是目前最先进的分块策略之一。


Reciprocal Rank Fusion 融合算法
interface RankedResult {
  id: string;
  rank: number;
  source: string;
}

function reciprocalRankFusion(
  resultSets: RankedResult[][],
  k: number = 60,
  topN: number = 10,
): Array<{ id: string; score: number }> {
  const scores = new Map<string, number>();

  for (const results of resultSets) {
    for (const result of results) {
      const current = scores.get(result.id) || 0;
      scores.set(result.id, current + 1 / (k + result.rank));
    }
  }

  return Array.from(scores.entries())
    .map(([id, score]) => ({ id, score }))
    .sort((a, b) => b.score - a.score)
    .slice(0, topN);
}

自适应分块
根据文档内容特征自动调整块大小，而非使用固定值：


from dataclasses import dataclass

@dataclass
class AdaptiveConfig:
    min_size: int = 128
    max_size: int = 2048
    target_density: float = 0.7

def adaptive_chunk_size(
    text: str,
    config: AdaptiveConfig,
) -> int:
    sentence_count = text.count("。") + text.count(".")
    word_count = len(text)

    if sentence_count == 0:
        return config.max_size

    avg_sentence_len = word_count / sentence_count

    if avg_sentence_len > 200:
        return min(config.max_size, 2048)

    if avg_sentence_len < 50:
        return max(config.min_size, 256)

    has_code = "```" in text or "def " in text or "function " in text
    if has_code:
        return min(config.max_size, 1500)

    has_list = text.count("\n- ") > 3 or text.count("\n1.") > 3
    if has_list:
        return 512

    return 768

渐进式优化流程
常见误区
❌ 误区 1：追求"一个最优值"
不存在全局最优的块大小，最优值取决于：


    

  • 你的文档类型（技术 vs 叙述）
    • 

  • 你的嵌入模型（窗口大小）
    • 

  • 你的查询模式（精确 vs 宽泛）
    • 

  • 你的 LLM（上下文窗口大小）
    • 



正确做法：针对你的具体场景做 A/B 测试。


❌ 误区 2：忽视重叠的重要性
不设重叠会导致分割点附近的信息被切断，这些恰好是语义转换的关键位置。15-25% 的重叠几乎没有额外成本，但能显著提升检索效果。


❌ 误区 3：只看检索指标不看端到端效果
分块大小优化的最终目标是提升 LLM 回答质量，而非仅仅提升检索指标：


    

  • 检索命中率高但答案不好 → 可能块太大，LLM 无法从长上下文中抽取关键信息
    • 

  • 检索命中率低但答案还行 → 可能块太小，但 LLM 自身知识弥补了不足
    • 

  • 必须做端到端评估
    • 



❌ 误区 4：设定后从不调整
随着知识库内容变化和用户查询模式演变，最优块大小也会变化。建议每季度用评估集重新测试，确保配置仍然最优。


小结
分块大小优化是一个需要持续迭代的工程过程，没有放之四海而皆准的最优值。核心原则是：从合理的起点出发，用数据驱动决策。


核心要点回顾：


    

  • 256-1024 token 是大多数 RAG 场景的甜区，从 512 token 开始是合理起点
    • 

  • 块大小不能超过嵌入模型窗口的 50%
    • 

  • 重叠大小建议为块大小的 15-25%
    • 

  • 不同文档类型需要不同的块大小配置
    • 

  • 多级分块是提升召回率的高级策略，代价是存储和复杂度
    • 

  • 必须做 A/B 测试，用自己的数据和场景评估
    • 

  • 用端到端指标（答案质量）而非仅用检索指标来评估
    • 

  • 持续监控，季度复审配置是否仍然最优
    •
