自定义框架设计

为什么要自定义

虽然 LangChain、LlamaIndex 等框架功能强大，但在某些场景下自定义框架更有优势：减少依赖、完全控制执行流程、针对特定领域优化、降低抽象开销。理解如何从零构建 Agent 框架，也能加深对 Agent 系统本质的理解。

最小化 Agent 循环
一个最小化的 Agent 框架只需要三个核心组件：LLM 客户端、工具注册表和执行循环：


from openai import OpenAI
import json
from typing import Any, Callable

class MinimalAgent:
    def __init__(self, model: str = "gpt-4o", system_prompt: str = ""):
        self.client = OpenAI()
        self.model = model
        self.system_prompt = system_prompt
        self.tools: dict[str, dict] = {}
        self.handlers: dict[str, Callable] = {}

    def tool(self, name: str, description: str, parameters: dict):
        def decorator(func: Callable):
            self.tools[name] = {
                "type": "function",
                "function": {
                    "name": name,
                    "description": description,
                    "parameters": parameters
                }
            }
            self.handlers[name] = func
            return func
        return decorator

    def run(self, user_input: str, max_steps: int = 10) -> str:
        messages = []
        if self.system_prompt:
            messages.append({"role": "system", "content": self.system_prompt})
        messages.append({"role": "user", "content": user_input})

        tool_list = list(self.tools.values()) if self.tools else None

        for _ in range(max_steps):
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                tools=tool_list
            )

            msg = response.choices[0].message
            messages.append(msg)

            if response.choices[0].finish_reason == "stop":
                return msg.content

            if msg.tool_calls:
                for tc in msg.tool_calls:
                    result = self._call_tool(tc)
                    messages.append({
                        "role": "tool",
                        "tool_call_id": tc.id,
                        "content": json.dumps(result, ensure_ascii=False)
                    })

        return "达到最大步数"

    def _call_tool(self, tool_call) -> Any:
        name = tool_call.function.name
        args = json.loads(tool_call.function.arguments)
        handler = self.handlers.get(name)
        if not handler:
            return {"error": f"工具 {name} 未注册"}
        try:
            return handler(**args)
        except Exception as e:
            return {"error": str(e)}
使用方式简洁直观：


agent = MinimalAgent(system_prompt="你是一个有用的助手。")

@agent.tool("search", "搜索信息", {
    "type": "object",
    "properties": {"query": {"type": "string"}},
    "required": ["query"]
})
def search(query: str):
    return {"results": f"关于 {query} 的搜索结果..."}

@agent.tool("calculate", "数学计算", {
    "type": "object",
    "properties": {"expression": {"type": "string"}},
    "required": ["expression"]
})
def calculate(expression: str):
    return {"result": eval(expression)}

print(agent.run("搜索 Python 最新版本号，然后计算它的主版本号乘以 10"))


插件架构
可扩展的 Agent 框架需要插件架构，支持动态加载工具、中间件和策略：


interface Plugin {
  name: string;
  version: string;
  init(agent: AgentCore): void;
}

interface ToolPlugin extends Plugin {
  tools: ToolDefinition[];
  handlers: Record<string, ToolHandler>;
}

interface MiddlewarePlugin extends Plugin {
  beforeLLMCall?(messages: Message[]): Message[];
  afterLLMCall?(response: LLMResponse): LLMResponse;
  beforeToolCall?(name: string, args: unknown): unknown;
  afterToolCall?(name: string, result: unknown): unknown;
}

class AgentCore {
  private plugins: Plugin[] = [];
  private middleware: MiddlewarePlugin[] = [];

  use(plugin: Plugin): this {
    plugin.init(this);
    this.plugins.push(plugin);
    if ('beforeLLMCall' in plugin || 'afterLLMCall' in plugin) {
      this.middleware.push(plugin as MiddlewarePlugin);
    }
    return this;
  }

  async callLLM(messages: Message[]): Promise<LLMResponse> {
    let processed = messages;
    for (const mw of this.middleware) {
      if (mw.beforeLLMCall) {
        processed = mw.beforeLLMCall(processed);
      }
    }

    let response = await this.llmClient.chat(processed);

    for (const mw of this.middleware) {
      if (mw.afterLLMCall) {
        response = mw.afterLLMCall(response);
      }
    }

    return response;
  }
}


可扩展性设计
良好的框架设计应该允许替换或扩展任何核心组件：


from abc import ABC, abstractmethod

class LLMProvider(ABC):
    @abstractmethod
    async def chat(self, messages: list[dict], **kwargs) -> dict: ...

class ToolExecutor(ABC):
    @abstractmethod
    async def execute(self, name: str, args: dict) -> Any: ...

class MemoryBackend(ABC):
    @abstractmethod
    async def store(self, key: str, value: Any) -> None: ...
    @abstractmethod
    async def retrieve(self, query: str, top_k: int) -> list: ...

class AgentConfig:
    llm_provider: LLMProvider
    tool_executor: ToolExecutor
    memory_backend: MemoryBackend
    max_steps: int = 10
    system_prompt: str = ""

class ConfigurableAgent:
    def __init__(self, config: AgentConfig):
        self.config = config

    async def run(self, input_text: str) -> str:
        memories = await self.config.memory_backend.retrieve(input_text, top_k=3)
        context = self._build_context(input_text, memories)

        messages = [
            {"role": "system", "content": self.config.system_prompt},
            {"role": "user", "content": context}
        ]

        for _ in range(self.config.max_steps):
            response = await self.config.llm_provider.chat(messages)
            if response.get("finish"):
                await self.config.memory_backend.store(input_text, response["content"])
                return response["content"]
            if response.get("tool_calls"):
                for tc in response["tool_calls"]:
                    result = await self.config.tool_executor.execute(tc["name"], tc["args"])
                    messages.append({"role": "tool", "content": str(result)})

        return "达到最大步数"


提示： 自定义框架的最佳起点是先实现一个最小化的 Agent 循环（~100 行代码），确保核心流程跑通，然后根据实际需求逐步添加插件支持、中间件、记忆系统等扩展。避免过度设计——先让它工作，再让它优雅。






小结
    

  • 自定义框架在精简依赖、完全控制、领域优化方面有优势，适合生产环境和特定场景。
    • 

  • 最小化 Agent 循环只需要 LLM 客户端、工具注册表和执行循环三个核心组件。
    • 

  • 插件架构通过工具插件、中间件插件和策略插件实现框架的可扩展性。
    • 

  • 依赖注入和抽象接口（LLMProvider、ToolExecutor、MemoryBackend）保证组件可替换。
    • 

  • 建议从最小实现开始，按需扩展，避免过度设计。
