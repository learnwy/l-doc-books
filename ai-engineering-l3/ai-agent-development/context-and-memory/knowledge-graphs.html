知识图谱

从向量到图：结构化知识表示

向量数据库擅长语义相似性搜索，但无法表达实体之间的关系。知识图谱（Knowledge Graph）以「实体-关系-实体」三元组的形式组织知识，使 Agent 能够进行结构化推理和多跳查询。



实体-关系提取
从非结构化文本中提取实体和关系是构建知识图谱的第一步。LLM 可以作为强大的信息提取工具：


from pydantic import BaseModel

class Entity(BaseModel):
    name: str
    entity_type: str
    description: str

class Relation(BaseModel):
    source: str
    target: str
    relation_type: str
    description: str

class KnowledgeTriples(BaseModel):
    entities: list[Entity]
    relations: list[Relation]

def extract_knowledge(text: str) -> KnowledgeTriples:
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[
            {
                "role": "system",
                "content": (
                    "从给定文本中提取实体和关系。"
                    "实体类型包括: 人物、组织、技术、概念、产品。"
                    "关系类型包括: 创建、属于、使用、包含、竞争、依赖。"
                )
            },
            {"role": "user", "content": text}
        ],
        response_format=KnowledgeTriples
    )
    return response.choices[0].message.parsed

text = """
LangChain 是由 Harrison Chase 创建的开源 LLM 框架。
它包含 LCEL 表达式语言和 AgentExecutor 组件。
LlamaIndex 是另一个流行的 LLM 框架，专注于数据索引。
"""

triples = extract_knowledge(text)
for entity in triples.entities:
    print(f"实体: {entity.name} ({entity.entity_type})")
for relation in triples.relations:
    print(f"关系: {relation.source} --[{relation.relation_type}]--> {relation.target}")

import networkx as nx

class SimpleKnowledgeGraph:
    def __init__(self):
        self.graph = nx.DiGraph()

    def add_entity(self, name: str, entity_type: str, **attrs):
        self.graph.add_node(name, entity_type=entity_type, **attrs)

    def add_relation(self, source: str, target: str, relation_type: str):
        self.graph.add_edge(source, target, relation=relation_type)

    def query_neighbors(self, entity: str, depth: int = 1) -> dict:
        if entity not in self.graph:
            return {}
        subgraph = nx.ego_graph(self.graph, entity, radius=depth)
        return {
            "entities": list(subgraph.nodes(data=True)),
            "relations": list(subgraph.edges(data=True))
        }

    def find_path(self, source: str, target: str) -> list:
        try:
            return nx.shortest_path(self.graph, source, target)
        except nx.NetworkXNoPath:
            return []

    def to_context_string(self, entity: str, depth: int = 2) -> str:
        neighbors = self.query_neighbors(entity, depth)
        lines = []
        for src, tgt, data in neighbors["relations"]:
            lines.append(f"{src} --[{data['relation']}]--> {tgt}")
        return "\n".join(lines)

kg = SimpleKnowledgeGraph()
kg.add_entity("LangChain", "框架")
kg.add_entity("Harrison Chase", "人物")
kg.add_entity("LCEL", "组件")
kg.add_relation("Harrison Chase", "LangChain", "创建")
kg.add_relation("LangChain", "LCEL", "包含")


Graph RAG
Graph RAG 将知识图谱与检索增强生成（RAG）结合，通过图遍历获取结构化上下文，然后交给 LLM 生成回答：


class GraphRAG:
    def __init__(self, kg: SimpleKnowledgeGraph, vector_store):
        self.kg = kg
        self.vector_store = vector_store

    def query(self, question: str) -> str:
        entities = self._extract_entities(question)

        graph_context = ""
        for entity in entities:
            context = self.kg.to_context_string(entity, depth=2)
            if context:
                graph_context += f"\n[{entity}的关系]:\n{context}\n"

        text_chunks = self.vector_store.search(question, top_k=3)
        text_context = "\n".join(chunk.content for chunk in text_chunks)

        prompt = (
            f"基于以下信息回答问题。\n\n"
            f"知识图谱信息:\n{graph_context}\n\n"
            f"文档信息:\n{text_context}\n\n"
            f"问题: {question}"
        )

        return call_llm(prompt)

    def _extract_entities(self, text: str) -> list[str]:
        all_entities = set(self.kg.graph.nodes())
        found = [e for e in all_entities if e.lower() in text.lower()]
        return found


图上推理
知识图谱的独特价值在于支持多跳推理——通过遍历多层关系得出结论：


class GraphReasoner:
    def __init__(self, kg: SimpleKnowledgeGraph):
        self.kg = kg

    def multi_hop_query(self, start: str, hops: list[str]) -> list[str]:
        current_nodes = {start}
        for relation_type in hops:
            next_nodes = set()
            for node in current_nodes:
                for _, target, data in self.kg.graph.edges(node, data=True):
                    if data["relation"] == relation_type:
                        next_nodes.add(target)
            current_nodes = next_nodes
        return list(current_nodes)

    def find_common_connections(self, entity_a: str, entity_b: str) -> list[str]:
        neighbors_a = set(self.kg.graph.neighbors(entity_a))
        neighbors_b = set(self.kg.graph.neighbors(entity_b))
        return list(neighbors_a & neighbors_b)

reasoner = GraphReasoner(kg)
results = reasoner.multi_hop_query("Harrison Chase", ["创建", "包含"])
print(f"Harrison Chase 创建的框架包含的组件: {results}")


提示： 知识图谱在 Agent 系统中最大的价值是提供结构化上下文。当用户问「X 和 Y 有什么关系」时，向量搜索可能找不到直接提及二者关系的文本，但图遍历可以通过中间节点发现间接关联。






小结
    

  • 知识图谱以实体-关系三元组组织知识，弥补向量搜索无法表达关系的不足。
    • 

  • LLM 可以作为实体-关系提取器，从非结构化文本自动构建知识图谱。
    • 

  • Graph RAG 结合图遍历和向量检索，提供结构化+语义化的混合上下文。
    • 

  • 多跳推理是知识图谱的独特能力，支持通过关系链发现间接关联。
    • 

  • 知识图谱适合实体关系密集的领域，如代码库分析、企业知识管理、学术文献网络。
