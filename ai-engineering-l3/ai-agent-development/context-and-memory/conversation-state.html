对话状态管理

为什么需要状态管理

Agent 的执行不只是简单的请求-响应。在多步骤任务中，Agent 需要维护当前进度、中间结果、分支条件等状态信息。对话状态管理让 Agent 能够在复杂交互中保持一致性，支持暂停恢复、分支回溯等高级能力。



状态机模型
使用有限状态机（FSM）来管理 Agent 的对话状态是一种结构化的方法。每个状态对应特定的行为模式，转换条件明确：


from enum import Enum
from dataclasses import dataclass, field
from typing import Any

class AgentState(Enum):
    WAITING = "waiting"
    ANALYZING = "analyzing"
    CLARIFYING = "clarifying"
    PLANNING = "planning"
    EXECUTING = "executing"
    VALIDATING = "validating"
    ERROR_RECOVERY = "error_recovery"
    COMPLETED = "completed"

@dataclass
class ConversationState:
    current_state: AgentState = AgentState.WAITING
    task: str = ""
    plan: list[str] = field(default_factory=list)
    current_step: int = 0
    results: dict[int, Any] = field(default_factory=dict)
    errors: list[str] = field(default_factory=list)
    metadata: dict = field(default_factory=dict)

class StateMachine:
    def __init__(self):
        self.state = ConversationState()
        self.transitions = {
            AgentState.WAITING: [AgentState.ANALYZING],
            AgentState.ANALYZING: [AgentState.PLANNING, AgentState.CLARIFYING],
            AgentState.CLARIFYING: [AgentState.ANALYZING],
            AgentState.PLANNING: [AgentState.EXECUTING],
            AgentState.EXECUTING: [AgentState.VALIDATING, AgentState.ERROR_RECOVERY],
            AgentState.VALIDATING: [AgentState.EXECUTING, AgentState.COMPLETED],
            AgentState.ERROR_RECOVERY: [AgentState.EXECUTING, AgentState.COMPLETED],
        }

    def transition(self, new_state: AgentState):
        allowed = self.transitions.get(self.state.current_state, [])
        if new_state not in allowed:
            raise ValueError(
                f"不允许从 {self.state.current_state} 转换到 {new_state}"
            )
        self.state.current_state = new_state


检查点
检查点（Checkpointing）允许 Agent 保存当前状态快照，支持后续恢复、回溯或审计：


import json
import hashlib
from datetime import datetime
from pathlib import Path

class CheckpointManager:
    def __init__(self, storage_dir: str = "./checkpoints"):
        self.storage_dir = Path(storage_dir)
        self.storage_dir.mkdir(parents=True, exist_ok=True)

    def save(self, state: ConversationState, label: str = "") -> str:
        checkpoint_id = hashlib.md5(
            f"{datetime.now().isoformat()}{label}".encode()
        ).hexdigest()[:12]

        data = {
            "id": checkpoint_id,
            "timestamp": datetime.now().isoformat(),
            "label": label,
            "state": {
                "current_state": state.current_state.value,
                "task": state.task,
                "plan": state.plan,
                "current_step": state.current_step,
                "results": state.results,
                "errors": state.errors,
                "metadata": state.metadata
            }
        }

        filepath = self.storage_dir / f"{checkpoint_id}.json"
        filepath.write_text(json.dumps(data, ensure_ascii=False, indent=2))
        return checkpoint_id

    def restore(self, checkpoint_id: str) -> ConversationState:
        filepath = self.storage_dir / f"{checkpoint_id}.json"
        data = json.loads(filepath.read_text())

        state = ConversationState()
        state.current_state = AgentState(data["state"]["current_state"])
        state.task = data["state"]["task"]
        state.plan = data["state"]["plan"]
        state.current_step = data["state"]["current_step"]
        state.results = data["state"]["results"]
        state.errors = data["state"]["errors"]
        return state

    def list_checkpoints(self) -> list[dict]:
        checkpoints = []
        for f in sorted(self.storage_dir.glob("*.json")):
            data = json.loads(f.read_text())
            checkpoints.append({
                "id": data["id"],
                "label": data["label"],
                "timestamp": data["timestamp"],
                "state": data["state"]["current_state"]
            })
        return checkpoints


分支与回溯
高级的状态管理支持分支（Branching）和回溯（Backtracking），让 Agent 能够探索多个执行路径：


interface StateNode {
  id: string;
  state: ConversationState;
  parentId: string | null;
  children: string[];
  branchLabel: string;
}

class BranchingStateManager {
  private nodes: Map<string, StateNode> = new Map();
  private currentNodeId: string;

  branch(label: string): string {
    const currentNode = this.nodes.get(this.currentNodeId);
    const newNode: StateNode = {
      id: generateId(),
      state: deepClone(currentNode.state),
      parentId: this.currentNodeId,
      children: [],
      branchLabel: label,
    };
    this.nodes.set(newNode.id, newNode);
    currentNode.children.push(newNode.id);
    return newNode.id;
  }

  switchBranch(nodeId: string): void {
    if (!this.nodes.has(nodeId)) {
      throw new Error(`节点 ${nodeId} 不存在`);
    }
    this.currentNodeId = nodeId;
  }

  backtrack(): void {
    const current = this.nodes.get(this.currentNodeId);
    if (current?.parentId) {
      this.currentNodeId = current.parentId;
    }
  }

  getHistory(): StateNode[] {
    const history: StateNode[] = [];
    let nodeId: string | null = this.currentNodeId;
    while (nodeId) {
      const node = this.nodes.get(nodeId);
      if (node) {
        history.unshift(node);
        nodeId = node.parentId;
      } else {
        break;
      }
    }
    return history;
  }
}

from abc import ABC, abstractmethod

class StateStore(ABC):
    @abstractmethod
    async def save(self, session_id: str, state: dict) -> None: ...

    @abstractmethod
    async def load(self, session_id: str) -> dict | None: ...

    @abstractmethod
    async def delete(self, session_id: str) -> None: ...

class RedisStateStore(StateStore):
    def __init__(self, redis_client, ttl: int = 86400):
        self.redis = redis_client
        self.ttl = ttl

    async def save(self, session_id: str, state: dict):
        await self.redis.setex(
            f"agent:state:{session_id}",
            self.ttl,
            json.dumps(state, ensure_ascii=False)
        )

    async def load(self, session_id: str) -> dict | None:
        data = await self.redis.get(f"agent:state:{session_id}")
        return json.loads(data) if data else None

    async def delete(self, session_id: str):
        await self.redis.delete(f"agent:state:{session_id}")


提示： LangGraph 通过内置的 StateGraph 和 Checkpointer 提供了开箱即用的状态管理。如果你使用 LangChain 生态，优先考虑 LangGraph 而非自建状态管理系统。






小结
    

  • 对话状态管理让 Agent 在复杂多步骤任务中保持一致性和可恢复性。
    • 

  • 有限状态机模型为 Agent 的行为转换提供了结构化的约束。
    • 

  • 检查点机制支持状态快照的保存和恢复，用于暂停恢复和审计。
    • 

  • 分支与回溯让 Agent 能够探索多个执行路径，在失败时回退到之前的状态。
    • 

  • 持久化策略的选择取决于速度需求、存储周期和部署架构。
