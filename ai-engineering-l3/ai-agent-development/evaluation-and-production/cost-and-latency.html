成本与延迟优化

Agent 的成本结构

Agent 系统的成本主要来自 LLM 调用。由于 Agent 在一次任务中可能进行多轮 LLM 调用（推理 + 工具选择 + 结果综合），Token 消耗和成本会快速累积。理解成本结构是优化的第一步。

模型路由
模型路由（Model Routing）是最有效的成本优化策略——根据任务复杂度动态选择模型。简单任务用便宜模型，复杂任务才用高端模型：


class ModelRouter:
    def __init__(self):
        self.models = {
            "simple": {"name": "gpt-4o-mini", "cost_per_1k_input": 0.00015},
            "medium": {"name": "gpt-4o", "cost_per_1k_input": 0.0025},
            "complex": {"name": "claude-3-5-sonnet-20241022", "cost_per_1k_input": 0.003}
        }

    def route(self, task: str, context_length: int) -> str:
        complexity = self._estimate_complexity(task, context_length)
        model_config = self.models[complexity]
        return model_config["name"]

    def _estimate_complexity(self, task: str, context_length: int) -> str:
        complexity_indicators = {
            "complex": ["分析", "比较", "设计", "架构", "优化", "debug"],
            "medium": ["实现", "编写", "解释", "总结"],
            "simple": ["查询", "查找", "转换", "格式化"]
        }

        task_lower = task.lower()
        for level, keywords in complexity_indicators.items():
            if any(kw in task_lower for kw in keywords):
                return level

        if context_length > 10000:
            return "complex"
        elif context_length > 3000:
            return "medium"
        return "simple"

router = ModelRouter()
model = router.route("帮我格式化这段 JSON", context_length=500)
print(f"选择模型: {model}")


缓存策略
对于重复或相似的请求，缓存可以避免不必要的 LLM 调用：


import hashlib
import json
from datetime import datetime, timedelta

class SemanticCache:
    def __init__(self, similarity_threshold: float = 0.95, ttl_hours: int = 24):
        self.cache: dict[str, dict] = {}
        self.embeddings: dict[str, list[float]] = {}
        self.threshold = similarity_threshold
        self.ttl = timedelta(hours=ttl_hours)

    def get(self, query: str) -> str | None:
        query_hash = self._hash(query)
        if query_hash in self.cache:
            entry = self.cache[query_hash]
            if datetime.now() - entry["timestamp"] < self.ttl:
                return entry["response"]
            del self.cache[query_hash]

        query_emb = get_embedding(query)
        for key, emb in self.embeddings.items():
            if self._cosine_sim(query_emb, emb) > self.threshold:
                entry = self.cache.get(key)
                if entry and datetime.now() - entry["timestamp"] < self.ttl:
                    return entry["response"]

        return None

    def set(self, query: str, response: str):
        query_hash = self._hash(query)
        self.cache[query_hash] = {
            "response": response,
            "timestamp": datetime.now()
        }
        self.embeddings[query_hash] = get_embedding(query)

    def _hash(self, text: str) -> str:
        return hashlib.md5(text.encode()).hexdigest()

    def _cosine_sim(self, a, b) -> float:
        import numpy as np
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

并行执行
当 Agent 需要调用多个独立工具时，并行执行可以显著降低总延迟：


import asyncio
import time

class ParallelToolExecutor:
    def __init__(self, timeout: float = 30.0):
        self.timeout = timeout

    async def execute_parallel(self, tool_calls: list[dict]) -> list[dict]:
        tasks = [
            self._execute_with_timeout(tc) for tc in tool_calls
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)

        return [
            {"tool": tc["name"], "result": r if not isinstance(r, Exception) else {"error": str(r)}}
            for tc, r in zip(tool_calls, results)
        ]

    async def _execute_with_timeout(self, tool_call: dict) -> dict:
        return await asyncio.wait_for(
            self._execute(tool_call),
            timeout=self.timeout
        )

    async def _execute(self, tool_call: dict) -> dict:
        handler = tool_registry.get(tool_call["name"])
        return await handler(**tool_call["args"])


Token 优化
减少每次 LLM 调用的 Token 数量是降低成本的直接手段：


class TokenOptimizer:
    def optimize_system_prompt(self, prompt: str) -> str:
        lines = prompt.strip().split("\n")
        optimized = [line.strip() for line in lines if line.strip()]
        return "\n".join(optimized)

    def compress_tool_results(self, results: list[dict], max_tokens: int = 500) -> list[dict]:
        compressed = []
        for result in results:
            content = result.get("content", "")
            if self._count_tokens(content) > max_tokens:
                content = self._truncate(content, max_tokens)
            compressed.append({**result, "content": content})
        return compressed

    def prune_message_history(self, messages: list[dict]) -> list[dict]:
        pruned = []
        for msg in messages:
            if msg["role"] == "tool":
                content = msg.get("content", "")
                if self._count_tokens(content) > 200:
                    msg = {**msg, "content": self._summarize_tool_result(content)}
            pruned.append(msg)
        return pruned

    def _count_tokens(self, text: str) -> int:
        return len(text) // 4

    def _truncate(self, text: str, max_tokens: int) -> str:
        return text[:max_tokens * 4] + "\n... (已截断)"

    def _summarize_tool_result(self, content: str) -> str:
        return content[:800] + "\n... (已压缩)"


提示： 成本优化最重要的原则是「分层」：90% 的请求用便宜模型处理，只有需要深度推理的 10% 才使用高端模型。结合语义缓存，可以进一步将实际 LLM 调用量减少 30-50%。

小结
    

  • Agent 的成本主要来自多轮 LLM 调用，输入 Token 随上下文积累而增长。
    • 

  • 模型路由根据任务复杂度动态选择模型，是最有效的成本优化手段。
    • 

  • 语义缓存通过向量相似性匹配避免重复请求，显著降低成本和延迟。
    • 

  • 并行执行独立的工具调用可以将总延迟降低到最慢单次调用的水平。
    • 

  • Token 优化通过压缩系统提示、裁剪工具结果和清理历史消息来减少每次调用的 Token 数。
