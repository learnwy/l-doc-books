生产部署

从原型到生产的挑战

Agent 在开发环境中运行良好不意味着它能在生产环境中可靠工作。生产部署面临的核心挑战包括：并发处理、执行隔离、安全防护和可靠性保障。



扩展架构
Agent 系统的扩展需要将请求接收、任务执行和结果交付解耦：


from fastapi import FastAPI, BackgroundTasks
from pydantic import BaseModel
import uuid

app = FastAPI()

class AgentRequest(BaseModel):
    task: str
    tools: list[str] = []
    max_steps: int = 10

class AgentResponse(BaseModel):
    task_id: str
    status: str

tasks_store: dict = {}

@app.post("/agent/run", response_model=AgentResponse)
async def create_task(request: AgentRequest, background: BackgroundTasks):
    task_id = str(uuid.uuid4())
    tasks_store[task_id] = {"status": "queued", "result": None}

    background.add_task(execute_agent_task, task_id, request)

    return AgentResponse(task_id=task_id, status="queued")

@app.get("/agent/status/{task_id}")
async def get_status(task_id: str):
    task = tasks_store.get(task_id)
    if not task:
        return {"error": "任务不存在"}
    return task

async def execute_agent_task(task_id: str, request: AgentRequest):
    tasks_store[task_id]["status"] = "running"
    try:
        result = await run_agent(request.task, request.tools, request.max_steps)
        tasks_store[task_id] = {"status": "completed", "result": result}
    except Exception as e:
        tasks_store[task_id] = {"status": "failed", "error": str(e)}

import { Queue, Worker } from 'bullmq';

const agentQueue = new Queue('agent-tasks', {
  connection: { host: 'localhost', port: 6379 },
});

const worker = new Worker(
  'agent-tasks',
  async (job) => {
    const { task, tools, maxSteps } = job.data;

    await job.updateProgress(0);

    const agent = createAgent(tools);
    const result = await agent.run(task, {
      maxSteps,
      onStep: async (step: number, total: number) => {
        await job.updateProgress(Math.floor((step / total) * 100));
      },
    });

    return result;
  },
  {
    connection: { host: 'localhost', port: 6379 },
    concurrency: 5,
    limiter: { max: 10, duration: 60000 },
  },
);

async function submitTask(task: string): Promise<string> {
  const job = await agentQueue.add(
    'run-agent',
    {
      task,
      tools: ['search', 'calculate'],
      maxSteps: 15,
    },
    {
      attempts: 3,
      backoff: { type: 'exponential', delay: 5000 },
      timeout: 300000,
    },
  );

  return job.id;
}


沙箱执行
Agent 的工具调用——特别是代码执行——需要在隔离的沙箱环境中运行，防止恶意操作影响宿主系统：


import docker
import tempfile
from pathlib import Path

class CodeSandbox:
    def __init__(self, timeout: int = 30, memory_limit: str = "256m"):
        self.client = docker.from_env()
        self.timeout = timeout
        self.memory_limit = memory_limit

    def execute_python(self, code: str) -> dict:
        with tempfile.TemporaryDirectory() as tmpdir:
            code_path = Path(tmpdir) / "script.py"
            code_path.write_text(code)

            try:
                container = self.client.containers.run(
                    "python:3.12-slim",
                    f"python /code/script.py",
                    volumes={tmpdir: {"bind": "/code", "mode": "ro"}},
                    mem_limit=self.memory_limit,
                    network_disabled=True,
                    remove=True,
                    timeout=self.timeout,
                    stdout=True,
                    stderr=True
                )
                return {"success": True, "output": container.decode("utf-8")}
            except docker.errors.ContainerError as e:
                return {"success": False, "error": e.stderr.decode("utf-8")}
            except Exception as e:
                return {"success": False, "error": str(e)}

class SecurityGuard:
    def __init__(self):
        self.blocked_patterns = [
            r"ignore.*previous.*instructions",
            r"system.*prompt",
            r"reveal.*secret",
        ]
        self.pii_patterns = [
            r"\b\d{3}-\d{2}-\d{4}\b",
            r"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b",
        ]

    def check_input(self, user_input: str) -> dict:
        import re
        for pattern in self.blocked_patterns:
            if re.search(pattern, user_input, re.IGNORECASE):
                return {"safe": False, "reason": "检测到可疑的注入模式"}
        return {"safe": True}

    def check_output(self, agent_output: str) -> str:
        import re
        sanitized = agent_output
        for pattern in self.pii_patterns:
            sanitized = re.sub(pattern, "[已脱敏]", sanitized)
        return sanitized

    def check_tool_permission(self, tool_name: str, user_role: str) -> bool:
        permissions = {
            "admin": ["search", "code_exec", "file_write", "db_query"],
            "user": ["search", "calculate"],
            "guest": ["search"]
        }
        allowed = permissions.get(user_role, [])
        return tool_name in allowed


提示： Prompt 注入是 Agent 系统面临的最大安全风险之一。攻击者可能通过工具返回的数据注入恶意指令。建议在每次工具返回后对结果进行清洗，或使用专门的 Prompt 注入检测模型。






小结
    

  • 从原型到生产需要解决可扩展性、异步执行、执行隔离和安全防护四大挑战。
    • 

  • 请求接收、任务执行、结果交付的解耦是扩展架构的核心。
    • 

  • 消息队列实现异步执行，支持重试、超时和并发控制。
    • 

  • 代码执行必须在 Docker 沙箱中进行，限制内存、网络和执行时间。
    • 

  • 多层安全防护覆盖输入过滤、工具权限、输出审查、执行隔离和速率限制。
