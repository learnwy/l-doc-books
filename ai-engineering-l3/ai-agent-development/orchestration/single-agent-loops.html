单 Agent 循环

基本循环结构

单 Agent 循环是最基础的 Agent 执行模式——一个 Agent 在循环中反复进行推理、选择工具、执行行动，直到任务完成或达到终止条件。理解这一循环的细节是构建可靠 Agent 系统的基础。



工具选择机制
Agent 在每一步都需要决定：是否需要调用工具？调用哪个工具？传什么参数？这个决策完全由 LLM 基于当前上下文做出。


from openai import OpenAI
import json

client = OpenAI()

class SingleAgentLoop:
    def __init__(self, system_prompt: str, tools: list[dict], max_iterations: int = 15):
        self.system_prompt = system_prompt
        self.tools = tools
        self.max_iterations = max_iterations
        self.tool_handlers: dict[str, callable] = {}

    def register_handler(self, name: str, handler: callable):
        self.tool_handlers[name] = handler

    def run(self, user_input: str) -> str:
        messages = [
            {"role": "system", "content": self.system_prompt},
            {"role": "user", "content": user_input}
        ]

        for iteration in range(self.max_iterations):
            response = client.chat.completions.create(
                model="gpt-4o",
                messages=messages,
                tools=self.tools if self.tools else None
            )

            choice = response.choices[0]
            messages.append(choice.message)

            if choice.finish_reason == "stop":
                return choice.message.content

            if choice.message.tool_calls:
                for tool_call in choice.message.tool_calls:
                    result = self._execute_tool(tool_call)
                    messages.append({
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "content": result
                    })

        return "已达到最大迭代次数"

    def _execute_tool(self, tool_call) -> str:
        name = tool_call.function.name
        args = json.loads(tool_call.function.arguments)
        handler = self.tool_handlers.get(name)
        if not handler:
            return json.dumps({"error": f"未注册的工具: {name}"})
        try:
            result = handler(**args)
            return json.dumps(result, ensure_ascii=False)
        except Exception as e:
            return json.dumps({"error": str(e)})

interface StoppingCriteria {
  maxIterations: number;
  maxTimeMs: number;
  maxTokens: number;
  detectLoops: boolean;
  loopThreshold: number;
}

class AgentRunner {
  private iterationCount = 0;
  private startTime = 0;
  private totalTokens = 0;
  private recentActions: string[] = [];

  constructor(private criteria: StoppingCriteria) {}

  shouldStop(): { stop: boolean; reason: string } {
    if (this.iterationCount >= this.criteria.maxIterations) {
      return { stop: true, reason: '达到最大迭代次数' };
    }

    const elapsed = Date.now() - this.startTime;
    if (elapsed > this.criteria.maxTimeMs) {
      return { stop: true, reason: '执行超时' };
    }

    if (this.totalTokens > this.criteria.maxTokens) {
      return { stop: true, reason: 'Token 预算耗尽' };
    }

    if (this.criteria.detectLoops && this.hasLoop()) {
      return { stop: true, reason: '检测到执行循环' };
    }

    return { stop: false, reason: '' };
  }

  private hasLoop(): boolean {
    if (this.recentActions.length < this.criteria.loopThreshold) return false;
    const recent = this.recentActions.slice(-this.criteria.loopThreshold);
    return new Set(recent).size === 1;
  }
}


上下文管理
随着 Agent 循环的推进，消息列表不断增长，可能超过 LLM 的上下文窗口限制。有效的上下文管理策略至关重要：


import tiktoken

class ContextManager:
    def __init__(self, max_tokens: int = 100000, model: str = "gpt-4o"):
        self.max_tokens = max_tokens
        self.encoder = tiktoken.encoding_for_model(model)

    def count_tokens(self, messages: list[dict]) -> int:
        total = 0
        for msg in messages:
            content = msg.get("content", "") or ""
            total += len(self.encoder.encode(content)) + 4
        return total

    def trim_messages(self, messages: list[dict]) -> list[dict]:
        if self.count_tokens(messages) <= self.max_tokens:
            return messages

        system = messages[0] if messages[0]["role"] == "system" else None
        user_messages = [m for m in messages if m["role"] == "user"]
        first_user = user_messages[0] if user_messages else None

        recent = messages[-10:]
        result = []
        if system:
            result.append(system)
        if first_user and first_user not in recent:
            result.append(first_user)
        result.extend(recent)

        return result


提示： 上下文管理的核心原则是「保留系统提示 + 用户原始请求 + 最近的交互」。中间的工具调用细节往往可以被安全地移除或压缩，因为 Agent 已经利用这些信息做出了后续决策。

agent = SingleAgentLoop(
    system_prompt="你是一个能够搜索信息和执行计算的智能助手。",
    tools=[
        {
            "type": "function",
            "function": {
                "name": "web_search",
                "description": "搜索互联网获取最新信息",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "query": {"type": "string", "description": "搜索关键词"}
                    },
                    "required": ["query"]
                }
            }
        },
        {
            "type": "function",
            "function": {
                "name": "calculate",
                "description": "执行数学计算",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "expression": {"type": "string", "description": "数学表达式"}
                    },
                    "required": ["expression"]
                }
            }
        }
    ]
)

agent.register_handler("web_search", lambda query: {"results": "模拟搜索结果..."})
agent.register_handler("calculate", lambda expression: {"result": eval(expression)})

result = agent.run("帮我查一下 Python 最新版本号，然后计算它的主版本号乘以 100")
print(result)


小结
    

  • 单 Agent 循环是 Agent 执行的最基本模式，由 LLM 推理、工具调用和结果观察三个环节构成。
    • 

  • 工具选择完全由 LLM 基于上下文自主决定，工具描述的质量直接影响选择准确率。
    • 

  • 终止条件必须多维设置：最大步数、超时、Token 预算、循环检测。
    • 

  • 上下文管理需要在保留关键信息和控制 Token 消耗之间取得平衡。
    • 

  • 完整的 Agent 循环需要将工具注册、错误处理、终止检测和上下文管理整合在一起。
