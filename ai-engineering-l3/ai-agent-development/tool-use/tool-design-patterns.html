工具设计模式

工具抽象层

在 Agent 系统中，工具不仅仅是函数——它需要被抽象为统一的接口，包含名称、描述、输入输出契约和错误处理。良好的工具抽象让 Agent 能够以一致的方式发现、选择和调用任意工具。

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any

@dataclass
class ToolResult:
    success: bool
    data: Any
    error: str | None = None

class BaseTool(ABC):
    @property
    @abstractmethod
    def name(self) -> str: ...

    @property
    @abstractmethod
    def description(self) -> str: ...

    @property
    @abstractmethod
    def parameters_schema(self) -> dict: ...

    @abstractmethod
    def execute(self, **kwargs) -> ToolResult: ...

    def to_openai_tool(self) -> dict:
        return {
            "type": "function",
            "function": {
                "name": self.name,
                "description": self.description,
                "parameters": self.parameters_schema
            }
        }


输入输出契约
每个工具都应该有清晰的输入输出契约（Contract）。输入契约定义参数类型和约束，输出契约定义返回值结构。这种契约设计确保 Agent 能够可靠地解析和使用工具结果。

错误处理
工具调用必然会遇到各种错误——网络超时、权限不足、参数无效等。Agent 需要能够理解错误类型并采取适当的应对策略。


import time
from enum import Enum

class ErrorType(Enum):
    RETRYABLE = "retryable"
    NON_RETRYABLE = "non_retryable"
    RATE_LIMITED = "rate_limited"

class ToolError(Exception):
    def __init__(self, message: str, error_type: ErrorType, retry_after: float = 0):
        super().__init__(message)
        self.error_type = error_type
        self.retry_after = retry_after

class ResilientTool(BaseTool):
    def __init__(self, max_retries: int = 3, base_delay: float = 1.0):
        self.max_retries = max_retries
        self.base_delay = base_delay

    def safe_execute(self, **kwargs) -> ToolResult:
        for attempt in range(self.max_retries + 1):
            try:
                return self.execute(**kwargs)
            except ToolError as e:
                if e.error_type == ErrorType.NON_RETRYABLE:
                    return ToolResult(success=False, data=None, error=str(e))

                if attempt == self.max_retries:
                    return ToolResult(success=False, data=None, error=f"重试 {self.max_retries} 次后仍失败: {e}")

                delay = self.base_delay * (2 ** attempt)
                if e.error_type == ErrorType.RATE_LIMITED:
                    delay = max(delay, e.retry_after)

                time.sleep(delay)

        return ToolResult(success=False, data=None, error="未知错误")

import random

def exponential_backoff_with_jitter(
    attempt: int,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    jitter: bool = True
) -> float:
    delay = min(base_delay * (2 ** attempt), max_delay)
    if jitter:
        delay = random.uniform(0, delay)
    return delay


提示： 在 Agent 系统中，工具的错误消息应该对 LLM 友好——清晰描述出了什么错、是否可重试、建议的替代方案。这样 LLM 才能做出正确的后续决策。






工具注册与发现
当工具数量增多时，需要一个工具注册表来管理工具的注册、查找和动态加载：


class ToolRegistry:
    def __init__(self):
        self._tools: dict[str, BaseTool] = {}

    def register(self, tool: BaseTool):
        self._tools[tool.name] = tool

    def get(self, name: str) -> BaseTool | None:
        return self._tools.get(name)

    def list_tools(self) -> list[dict]:
        return [tool.to_openai_tool() for tool in self._tools.values()]

    def search(self, query: str) -> list[BaseTool]:
        results = []
        query_lower = query.lower()
        for tool in self._tools.values():
            if (query_lower in tool.name.lower() or
                query_lower in tool.description.lower()):
                results.append(tool)
        return results

registry = ToolRegistry()
registry.register(SearchTool())
registry.register(CalculatorTool())
registry.register(CodeRunnerTool())

openai_tools = registry.list_tools()


小结
    

  • 工具抽象层为 Agent 提供统一的工具发现、选择和调用接口。
    • 

  • 输入输出契约通过类型定义和验证确保工具调用的可靠性。
    • 

  • 错误处理需要区分可重试和不可重试错误，并采用相应的重试策略。
    • 

  • 指数退避+抖动是分布式系统中最推荐的重试策略。
    • 

  • 工具注册表管理工具的生命周期，支持动态注册和语义搜索。
