API 集成

Agent 与外部服务的连接

Agent 需要通过 API 与外部服务交互来获取数据、执行操作。API 集成是 Agent 工具系统的重要组成部分，涉及 REST/GraphQL 接口调用、身份认证、限流处理和数据转换等核心挑战。



REST API 集成
REST API 是最常见的外部服务接口。将 REST API 封装为 Agent 工具时，需要关注请求构造、响应解析和错误处理：


import httpx
from pydantic import BaseModel

class RestApiTool:
    def __init__(self, base_url: str, headers: dict | None = None):
        self.client = httpx.AsyncClient(
            base_url=base_url,
            headers=headers or {},
            timeout=30.0
        )

    async def get(self, path: str, params: dict | None = None) -> dict:
        response = await self.client.get(path, params=params)
        response.raise_for_status()
        return response.json()

    async def post(self, path: str, data: dict) -> dict:
        response = await self.client.post(path, json=data)
        response.raise_for_status()
        return response.json()

class GithubTool(RestApiTool):
    def __init__(self, token: str):
        super().__init__(
            base_url="https://api.github.com",
            headers={
                "Authorization": f"Bearer {token}",
                "Accept": "application/vnd.github.v3+json"
            }
        )

    async def search_repos(self, query: str, limit: int = 5) -> list[dict]:
        result = await self.get("/search/repositories", {
            "q": query,
            "per_page": limit,
            "sort": "stars"
        })
        return [
            {"name": r["full_name"], "stars": r["stargazers_count"], "description": r["description"]}
            for r in result["items"]
        ]

interface GraphQLToolConfig {
  endpoint: string;
  headers: Record<string, string>;
  maxDepth: number;
}

class GraphQLTool {
  constructor(private config: GraphQLToolConfig) {}

  async query<T>(query: string, variables?: Record<string, unknown>): Promise<T> {
    const response = await fetch(this.config.endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        ...this.config.headers,
      },
      body: JSON.stringify({ query, variables }),
    });

    const result = await response.json();
    if (result.errors) {
      throw new Error(result.errors.map((e: any) => e.message).join(', '));
    }
    return result.data;
  }
}

const github = new GraphQLTool({
  endpoint: 'https://api.github.com/graphql',
  headers: { Authorization: `Bearer ${process.env.GITHUB_TOKEN}` },
  maxDepth: 3,
});

const data = await github.query(

    }
  }
`,
  { owner: 'langchain-ai', name: 'langchain' },
);


身份认证
API 集成中的身份认证需要安全地管理凭证，同时支持多种认证方式：


import time
from abc import ABC, abstractmethod

class AuthProvider(ABC):
    @abstractmethod
    async def get_headers(self) -> dict[str, str]: ...

class ApiKeyAuth(AuthProvider):
    def __init__(self, key: str, header_name: str = "Authorization", prefix: str = "Bearer"):
        self._key = key
        self._header_name = header_name
        self._prefix = prefix

    async def get_headers(self) -> dict[str, str]:
        return {self._header_name: f"{self._prefix} {self._key}"}

class OAuth2Auth(AuthProvider):
    def __init__(self, client_id: str, client_secret: str, token_url: str):
        self._client_id = client_id
        self._client_secret = client_secret
        self._token_url = token_url
        self._token: str | None = None
        self._expires_at: float = 0

    async def get_headers(self) -> dict[str, str]:
        if time.time() >= self._expires_at:
            await self._refresh_token()
        return {"Authorization": f"Bearer {self._token}"}

    async def _refresh_token(self):
        async with httpx.AsyncClient() as client:
            response = await client.post(self._token_url, data={
                "grant_type": "client_credentials",
                "client_id": self._client_id,
                "client_secret": self._client_secret
            })
            data = response.json()
            self._token = data["access_token"]
            self._expires_at = time.time() + data["expires_in"] - 60


限流处理
外部 API 通常有调用频率限制。Agent 在高频调用场景下必须实现限流控制，避免被服务方封禁：

import asyncio
import time

class RateLimiter:
    def __init__(self, max_calls: int, period: float):
        self.max_calls = max_calls
        self.period = period
        self.calls: list[float] = []
        self._lock = asyncio.Lock()

    async def acquire(self):
        async with self._lock:
            now = time.time()
            self.calls = [t for t in self.calls if now - t < self.period]

            if len(self.calls) >= self.max_calls:
                sleep_time = self.period - (now - self.calls[0])
                await asyncio.sleep(sleep_time)

            self.calls.append(time.time())

class RateLimitedApiTool:
    def __init__(self, base_url: str, rate_limiter: RateLimiter):
        self.client = httpx.AsyncClient(base_url=base_url)
        self.limiter = rate_limiter

    async def call(self, path: str, **kwargs) -> dict:
        await self.limiter.acquire()
        response = await self.client.get(path, **kwargs)
        response.raise_for_status()
        return response.json()


数据转换
API 返回的原始数据往往包含大量对 Agent 无用的字段。数据转换层负责将原始响应裁剪为 LLM 友好的格式，减少 Token 消耗：


class DataTransformer:
    @staticmethod
    def transform_github_repo(raw: dict) -> str:
        return (
            f"仓库: {raw['full_name']}\n"
            f"描述: {raw.get('description', '无')}\n"
            f"Stars: {raw['stargazers_count']}\n"
            f"语言: {raw.get('language', '未知')}\n"
            f"最近更新: {raw['updated_at'][:10]}"
        )

    @staticmethod
    def transform_for_llm(data: dict, max_length: int = 2000) -> str:
        text = json.dumps(data, ensure_ascii=False, indent=2)
        if len(text) > max_length:
            text = text[:max_length] + "\n... (已截断)"
        return text


提示： 数据转换是控制 Agent Token 消耗的重要手段。API 返回的 JSON 可能包含数十个字段，但 Agent 可能只需要其中 3-5 个关键字段。始终在工具输出端做裁剪，而不是将原始数据全部注入上下文。






小结
    

  • API 集成是 Agent 连接外部服务的核心能力，需要处理 REST 和 GraphQL 两种主流接口。
    • 

  • GraphQL 的精确查询能力天然适合 Agent 的 Token 预算控制需求。
    • 

  • 身份认证需要支持 API Key、OAuth 2.0 等多种方式，并安全管理凭证。
    • 

  • 限流处理通过令牌桶或滑动窗口机制避免被外部服务封禁。
    • 

  • 数据转换层将原始 API 响应裁剪为 LLM 友好格式，是控制 Token 消耗的关键。
