Prompt 注入

提示词组合架构

Prompt 注入是技能系统的最后一环。当 SkillRegistry 匹配到相关技能后，Prompt Composer 负责将基础系统提示词、匹配到的技能指令和上下文信息组合成最终发送给 LLM 的系统提示词。这个过程需要精确控制注入顺序和 token 预算分配。



提示词分层模型
最终系统提示词由多个层级按固定顺序拼接而成。每一层有独立的 token 预算和优先级：

Token 预算管理
每次请求的系统提示词受到上下文窗口大小的限制。Token Budget Manager 负责在各层之间分配预算，并在超出时对可裁剪层进行截断。


interface TokenBudget {
  total: number;
  layers: Record<string, LayerBudget>;
}

interface LayerBudget {
  maxTokens: number;
  priority: number;
  trimmable: boolean;
}

function createBudget(contextWindowSize: number): TokenBudget {
  const systemBudget = Math.floor(contextWindowSize * 0.3);

  return {
    total: systemBudget,
    layers: {
      identity: {
        maxTokens: Math.floor(systemBudget * 0.2),
        priority: 0,
        trimmable: false,
      },
      skillPurpose: {
        maxTokens: Math.floor(systemBudget * 0.3),
        priority: 1,
        trimmable: false,
      },
      skillImplementation: {
        maxTokens: Math.floor(systemBudget * 0.25),
        priority: 2,
        trimmable: true,
      },
      safety: {
        maxTokens: Math.floor(systemBudget * 0.1),
        priority: 3,
        trimmable: false,
      },
      projectContext: {
        maxTokens: Math.floor(systemBudget * 0.15),
        priority: 4,
        trimmable: true,
      },
    },
  };
}


PromptComposer 实现
import { encode } from 'gpt-tokenizer';

interface ComposerInput {
  baseSystemPrompt: string;
  matchedSkills: MatchResult[];
  projectRules: string[];
  contextWindowSize: number;
}

interface ComposedPrompt {
  content: string;
  tokenCount: number;
  includedSkills: string[];
  trimmedLayers: string[];
}

class PromptComposer {
  compose(input: ComposerInput): ComposedPrompt {
    const budget = createBudget(input.contextWindowSize);
    const layers = this.buildLayers(input);
    const trimmedLayers: string[] = [];
    const parts: string[] = [];
    let totalTokens = 0;

    for (const layer of layers) {
      const tokens = encode(layer.content).length;
      const layerBudget = budget.layers[layer.name];

      if (totalTokens + tokens <= budget.total) {
        parts.push(layer.content);
        totalTokens += tokens;
      } else if (layerBudget.trimmable) {
        const remaining = budget.total - totalTokens;
        if (remaining > 50) {
          const trimmed = this.trimToTokens(layer.content, remaining);
          parts.push(trimmed);
          totalTokens += remaining;
        }
        trimmedLayers.push(layer.name);
      } else {
        parts.push(layer.content);
        totalTokens += tokens;
      }
    }

    return {
      content: parts.join('\n\n'),
      tokenCount: totalTokens,
      includedSkills: input.matchedSkills.map((r) => r.skill.name),
      trimmedLayers,
    };
  }

  private buildLayers(input: ComposerInput) {
    const layers: { name: string; content: string }[] = [];

    layers.push({
      name: 'identity',
      content: input.baseSystemPrompt,
    });

    for (const result of input.matchedSkills) {
      const skill = result.skill;
      const purpose = skill.sections['Purpose'] ?? '';
      const usage = skill.sections['Usage'] ?? '';

      layers.push({
        name: 'skillPurpose',
        content: [`<skill name="${skill.name}">`, `## Purpose\n${purpose}`, `## Usage\n${usage}`].join('\n'),
      });
    }

    for (const result of input.matchedSkills) {
      const impl = result.skill.sections['Implementation'] ?? '';
      if (impl) {
        layers.push({
          name: 'skillImplementation',
          content: `## Implementation (${result.skill.name})\n${impl}`,
        });
      }
    }

    const safetyRules = input.matchedSkills.map((r) => r.skill.sections['Safety'] ?? '').filter(Boolean);

    if (safetyRules.length > 0) {
      layers.push({
        name: 'safety',
        content: `## Safety Rules\n${safetyRules.join('\n')}`,
      });
    }

    if (input.projectRules.length > 0) {
      layers.push({
        name: 'projectContext',
        content: `## Project Rules\n${input.projectRules.join('\n')}`,
      });
    }

    for (const result of input.matchedSkills) {
      const idx = layers.findIndex((l) => l.name === 'skillPurpose' && l.content.includes(result.skill.name));
      if (idx !== -1) {
        layers[idx].content += '\n</skill>';
      }
    }

    return layers;
  }

  private trimToTokens(content: string, maxTokens: number): string {
    const tokens = encode(content);
    if (tokens.length <= maxTokens) return content;

    const trimmedTokens = tokens.slice(0, maxTokens - 3);
    return new TextDecoder().decode(new Uint8Array(trimmedTokens.flatMap((t) => [...encode(String(t))]))) + '...';
  }
}


与 AI SDK 集成
将 PromptComposer 接入 AI SDK 的 generateText 调用：


import { generateText } from 'ai';
import { openai } from '@ai-sdk/openai';

async function handleChat(userMessage: string, registry: SkillRegistry, composer: PromptComposer) {
  const matched = registry.match(userMessage);

  const composed = composer.compose({
    baseSystemPrompt: '你是一个专业的编程助手。',
    matchedSkills: matched,
    projectRules: ['使用 TypeScript 和 Hono 框架', '代码风格遵循 ESLint 规则'],
    contextWindowSize: 128000,
  });

  const result = await generateText({
    model: openai('gpt-4o'),
    system: composed.content,
    prompt: userMessage,
    tools: extractTools(matched),
  });

  return result;
}

function extractTools(matched: MatchResult[]) {
  const toolNames = new Set<string>();
  for (const r of matched) {
    for (const tool of r.skill.requiredTools) {
      toolNames.add(tool);
    }
  }
  return Object.fromEntries(Array.from(toolNames).map((name) => [name, getToolDefinition(name)]));
}

小结
    

  • Prompt Composer 负责将基础提示词、技能指令和项目上下文按层级组合成最终系统提示词。
    • 

  • 分层模型（Layer 0-4）确保关键信息始终存在，可裁剪层在 token 超限时被截断。
    • 

  • Token 预算按比例分配给各层，系统提示词占上下文窗口的 30%。
    • 

  • 技能的 Purpose/Usage 和 Implementation 分开注入，便于独立控制预算。
    • 

  • 与 AI SDK 的 generateText 集成时，同时从匹配技能中提取所需工具定义。
    •
