A/B 测试

为什么需要 A/B 测试

选择哪个模型、哪套 prompt、哪种工具配置来部署 Agent，不能靠直觉判断。A/B 测试提供了系统化的模型对比方法：在相同测试套件上运行不同的 Agent 配置，收集指标，通过统计分析判断哪个配置更优。

ABTestRunner 实现
ABTestRunner 对每个候选配置运行完整的测试套件，收集各配置的评估结果：


interface TestCase {
  id: string;
  input: string;
  expectedOutput: string;
  tools: string[];
  maxSteps: number;
  timeout: number;
}

interface RunResult {
  candidateId: string;
  testCaseId: string;
  passed: boolean;
  actualOutput: string;
  latencyMs: number;
  totalTokens: number;
  toolCallCount: number;
}

interface ABTestReport {
  timestamp: string;
  totalCases: number;
  candidates: CandidateResult[];
  winner: string;
  significanceTest: SignificanceResult;
}

interface CandidateResult {
  candidateId: string;
  name: string;
  passRate: number;
  meanLatencyMs: number;
  totalTokens: number;
  totalCost: number;
  results: RunResult[];
}

class ABTestRunner {
  private tools: Record<string, CoreTool>;
  private cases: TestCase[] = [];

  constructor(tools: Record<string, CoreTool>) {
    this.tools = tools;
  }

  loadCases(cases: TestCase[]): void {
    this.cases = cases;
  }

  async run(candidates: CandidateConfig[]): Promise<ABTestReport> {
    const candidateResults: CandidateResult[] = [];

    for (const candidate of candidates) {
      const results = await this.runCandidate(candidate);
      const passed = results.filter((r) => r.passed).length;

      candidateResults.push({
        candidateId: candidate.id,
        name: candidate.name,
        passRate: this.cases.length > 0 ? passed / this.cases.length : 0,
        meanLatencyMs: this.mean(results.map((r) => r.latencyMs)),
        totalTokens: results.reduce((s, r) => s + r.totalTokens, 0),
        totalCost: results.reduce((s, r) => s + r.totalTokens, 0) * 0.00001,
        results,
      });
    }

    const sorted = [...candidateResults].sort((a, b) => b.passRate - a.passRate);
    const significance = this.computeSignificance(sorted[0], sorted[1]);

    return {
      timestamp: new Date().toISOString(),
      totalCases: this.cases.length,
      candidates: candidateResults,
      winner: sorted[0].candidateId,
      significanceTest: significance,
    };
  }

  private async runCandidate(candidate: CandidateConfig): Promise<RunResult[]> {
    const results: RunResult[] = [];

    for (const tc of this.cases) {
      const availableTools: Record<string, CoreTool> = {};
      for (const name of tc.tools) {
        if (this.tools[name]) availableTools[name] = this.tools[name];
      }

      const startTime = Date.now();
      try {
        const result = await generateText({
          model: candidate.model,
          system: candidate.systemPrompt,
          prompt: tc.input,
          tools: availableTools,
          maxSteps: candidate.maxSteps,
          temperature: candidate.temperature,
          abortSignal: AbortSignal.timeout(tc.timeout),
        });

        const actual = result.text.trim().toLowerCase();
        const expected = tc.expectedOutput.trim().toLowerCase();

        results.push({
          candidateId: candidate.id,
          testCaseId: tc.id,
          passed: actual.includes(expected),
          actualOutput: result.text,
          latencyMs: Date.now() - startTime,
          totalTokens: result.usage.totalTokens,
          toolCallCount: result.steps.reduce((s, step) => s + step.toolCalls.length, 0),
        });
      } catch {
        results.push({
          candidateId: candidate.id,
          testCaseId: tc.id,
          passed: false,
          actualOutput: '',
          latencyMs: Date.now() - startTime,
          totalTokens: 0,
          toolCallCount: 0,
        });
      }
    }

    return results;
  }

  private mean(values: number[]): number {
    if (values.length === 0) return 0;
    return values.reduce((a, b) => a + b, 0) / values.length;
  }

  private computeSignificance(a: CandidateResult, b: CandidateResult): SignificanceResult {
    if (!a || !b) return { zScore: 0, pValue: 1, significant: false };

    const n = this.cases.length;
    const p1 = a.passRate;
    const p2 = b.passRate;
    const pPooled = (p1 * n + p2 * n) / (2 * n);
    const se = Math.sqrt(pPooled * (1 - pPooled) * (2 / n));
    const zScore = se > 0 ? (p1 - p2) / se : 0;
    const pValue = 2 * (1 - this.normalCdf(Math.abs(zScore)));

    return { zScore, pValue, significant: pValue < 0.05 };
  }

  private normalCdf(x: number): number {
    const a1 = 0.254829592;
    const a2 = -0.284496736;
    const a3 = 1.421413741;
    const a4 = -1.453152027;
    const a5 = 1.061405429;
    const p = 0.3275911;
    const t = 1 / (1 + p * Math.abs(x));
    const y = 1 - ((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t * Math.exp((-x * x) / 2);
    return x >= 0 ? y : 1 - y;
  }
}

interface SignificanceResult {
  zScore: number;
  pValue: number;
  significant: boolean;
}


使用示例
import { tool } from 'ai';
import { z } from 'zod';

const tools = {
  weather: tool({
    description: '获取城市天气',
    parameters: z.object({ city: z.string() }),
    execute: async ({ city }) => {
      const db: Record<string, number> = { 北京: 28, 东京: 32, 纽约: 22 };
      return { city, temp: db[city] ?? 20 };
    },
  }),
  calculator: tool({
    description: '数学计算',
    parameters: z.object({ expression: z.string() }),
    execute: async ({ expression }) => ({
      result: new Function(`return ${expression}`)(),
    }),
  }),
};

const runner = new ABTestRunner(tools);

runner.loadCases([
  { id: 't1', input: '北京多少度？', expectedOutput: '28', tools: ['weather'], maxSteps: 5, timeout: 30_000 },
  { id: 't2', input: '15 + 27 等于多少？', expectedOutput: '42', tools: ['calculator'], maxSteps: 5, timeout: 30_000 },
  {
    id: 't3',
    input: '北京和东京温差？',
    expectedOutput: '4',
    tools: ['weather', 'calculator'],
    maxSteps: 10,
    timeout: 60_000,
  },
]);

const report = await runner.run(candidates);


结果分析与对比
function printABTestReport(report: ABTestReport): void {
  console.log(`\nA/B 测试报告 — ${report.timestamp}`);
  console.log(`测试用例数: ${report.totalCases}\n`);

  const header = ['指标', ...report.candidates.map((c) => c.name)];
  const rows = [
    ['通过率', ...report.candidates.map((c) => `${(c.passRate * 100).toFixed(1)}%`)],
    ['平均延迟', ...report.candidates.map((c) => `${c.meanLatencyMs.toFixed(0)}ms`)],
    ['总 Token', ...report.candidates.map((c) => `${c.totalTokens}`)],
    ['估算成本', ...report.candidates.map((c) => `$${c.totalCost.toFixed(4)}`)],
  ];

  console.log(formatTable(header, rows));

  const sig = report.significanceTest;
  console.log(`\n胜出: ${report.winner}`);
  console.log(`Z-Score: ${sig.zScore.toFixed(3)}`);
  console.log(`P-Value: ${sig.pValue.toFixed(4)}`);
  console.log(`统计显著: ${sig.significant ? '是 ✓' : '否 ✗'}`);
}

function formatTable(header: string[], rows: string[][]): string {
  const widths = header.map((h, i) => Math.max(h.length, ...rows.map((r) => r[i]?.length ?? 0)));
  const sep = widths.map((w) => '-'.repeat(w + 2)).join('|');
  const fmt = (row: string[]) => row.map((cell, i) => ` ${cell.padEnd(widths[i])} `).join('|');

  return [fmt(header), sep, ...rows.map(fmt)].join('\n');
}

盲评与偏差控制
为消除评估中的确认偏差，A/B 测试应遵循盲评原则：






策略
说明





随机化执行顺序
打乱候选配置的运行顺序，避免顺序偏差




固定随机种子
设置 temperature: 0，减少输出的随机性




独立评估
每个候选配置的结果独立收集，不互相影响




多次运行取均值
对非确定性输出多次运行，取统计平均值




统计显著性检验
使用 Z-Test 验证差异是否显著（P < 0.05）





function shuffleArray<T>(array: T[]): T[] {
  const shuffled = [...array];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled;
}

async function runBlindABTest(
  runner: ABTestRunner,
  candidates: CandidateConfig[],
  iterations: number,
): Promise<ABTestReport[]> {
  const reports: ABTestReport[] = [];
  for (let i = 0; i < iterations; i++) {
    const shuffled = shuffleArray(candidates);
    const report = await runner.run(shuffled);
    reports.push(report);
  }
  return reports;
}


小结
A/B 测试是 Agent 模型选型和配置优化的核心手段。ABTestRunner 在相同测试套件上运行多个候选配置，收集通过率、延迟、Token 消耗和成本指标，通过 Z-Test 统计检验判断差异是否显著。盲评原则——随机化执行顺序、固定温度、独立评估、多次运行取均值——消除了评估中的系统性偏差。当 P-Value 小于 0.05 时，可以有信心地选择胜出配置部署；否则需要增加测试用例数量提高统计功效。A/B 测试与 Eval Harness、测试用例、指标体系共同构成了完整的 Agent 质量保障闭环。
