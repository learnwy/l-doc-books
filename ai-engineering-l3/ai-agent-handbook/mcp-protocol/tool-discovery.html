工具发现与注册

动态发现的必要性

当 Agent 需要连接多个 MCP Server 时，手动维护工具列表变得不可行。MCP Server 可能随时添加或移除工具，Agent 需要一套自动化的工具发现机制来应对这种动态性。



Server 注册表配置
通过配置文件管理多个 MCP Server 的连接信息：


{
  "servers": [
    {
      "name": "fs-server",
      "transport": "stdio",
      "command": "node",
      "args": ["./mcp-servers/fs-server/dist/index.js"],
      "env": { "MCP_BASE_DIR": "./workspace" },
      "enabled": true
    },
    {
      "name": "db-server",
      "transport": "sse",
      "url": "http://localhost:3002/sse",
      "enabled": true
    }
  ],
  "cacheDir": "./.cache/mcp",
  "refreshInterval": 300000
}
对应的 TypeScript 类型定义：


interface MCPServerConfig {
  name: string;
  transport: 'stdio' | 'sse';
  command?: string;
  args?: string[];
  url?: string;
  env?: Record<string, string>;
  enabled: boolean;
}

interface MCPRegistryConfig {
  servers: MCPServerConfig[];
  cacheDir: string;
  refreshInterval: number;
}


MCPToolRegistry 实现
核心注册中心类负责管理多个 Server 连接，合并工具集，并提供统一的工具访问接口：


import { Client } from '@modelcontextprotocol/sdk/client/index.js';
import { StdioClientTransport } from '@modelcontextprotocol/sdk/client/stdio.js';
import { SSEClientTransport } from '@modelcontextprotocol/sdk/client/sse.js';
import { tool, jsonSchema } from 'ai';
import { writeFile, mkdir } from 'fs/promises';
import { join } from 'path';

interface ToolEntry {
  serverName: string;
  name: string;
  description: string;
  inputSchema: Record<string, unknown>;
}

class MCPToolRegistry {
  private connections = new Map<string, { client: Client; tools: ToolEntry[] }>();
  private config: MCPRegistryConfig;
  private refreshTimer: ReturnType<typeof setInterval> | null = null;

  constructor(config: MCPRegistryConfig) {
    this.config = config;
  }

  async initialize(): Promise<void> {
    await mkdir(this.config.cacheDir, { recursive: true });

    const enabled = this.config.servers.filter((s) => s.enabled);
    const results = await Promise.allSettled(enabled.map((s) => this.connectServer(s)));
    results.forEach((r, i) => {
      if (r.status === 'rejected') console.error(`[MCPRegistry] ${enabled[i].name} 连接失败:`, r.reason);
    });

    if (this.config.refreshInterval > 0) {
      this.refreshTimer = setInterval(() => this.refreshAll(), this.config.refreshInterval);
    }
  }

  private async connectServer(config: MCPServerConfig): Promise<void> {
    const client = new Client({ name: 'hono-agent', version: '1.0.0' });
    const transport =
      config.transport === 'stdio'
        ? new StdioClientTransport({ command: config.command!, args: config.args ?? [] })
        : new SSEClientTransport(new URL(config.url!));
    await client.connect(transport);

    const { tools } = await client.listTools();
    const entries: ToolEntry[] = tools.map((t) => ({
      serverName: config.name,
      name: t.name,
      description: t.description ?? '',
      inputSchema: t.inputSchema as Record<string, unknown>,
    }));

    await this.cacheTools(config.name, entries);
    this.connections.set(config.name, { client, tools: entries });
    console.log(`[MCPRegistry] ${config.name}: ${entries.length} 个工具已注册`);
  }

  private async cacheTools(name: string, tools: ToolEntry[]): Promise<void> {
    const path = join(this.config.cacheDir, `${name}.json`);
    await writeFile(path, JSON.stringify(tools, null, 2));
  }

  async refreshAll(): Promise<void> {
    for (const [name, conn] of this.connections) {
      try {
        const { tools } = await conn.client.listTools();
        conn.tools = tools.map((t) => ({
          serverName: name,
          name: t.name,
          description: t.description ?? '',
          inputSchema: t.inputSchema as Record<string, unknown>,
        }));
        await this.cacheTools(name, conn.tools);
      } catch (error) {
        console.error(`[MCPRegistry] ${name} 刷新失败:`, error);
      }
    }
  }

  getAISDKTools(): Record<string, ReturnType<typeof tool>> {
    const merged: Record<string, ReturnType<typeof tool>> = {};
    for (const [serverName, conn] of this.connections) {
      for (const entry of conn.tools) {
        merged[`${serverName}__${entry.name}`] = tool({
          description: `[${serverName}] ${entry.description}`,
          parameters: jsonSchema(entry.inputSchema),
          execute: async (args) => {
            const result = await conn.client.callTool({
              name: entry.name,
              arguments: args as Record<string, unknown>,
            });
            const text = result.content
              .filter((c): c is { type: 'text'; text: string } => c.type === 'text')
              .map((c) => c.text)
              .join('\n');
            if (result.isError) throw new Error(text);
            return text;
          },
        });
      }
    }
    return merged;
  }

  getStatus() {
    return Array.from(this.connections.entries()).map(([name, conn]) => ({
      name,
      toolCount: conn.tools.length,
      tools: conn.tools.map((t) => t.name),
    }));
  }

  async shutdown(): Promise<void> {
    if (this.refreshTimer) clearInterval(this.refreshTimer);
    for (const [, conn] of this.connections) await conn.client.close();
    this.connections.clear();
  }
}


工具名称冲突处理
多个 Server 可能暴露同名工具。Registry 使用 serverName__toolName 的命名空间策略避免冲突：


全局唯一时可使用短名称以提升 LLM 选择准确性：


function mergeWithShortNames(registry: MCPToolRegistry): Record<string, ReturnType<typeof tool>> {
  const allTools = registry.getAISDKTools();
  const nameCount = new Map<string, number>();
  for (const key of Object.keys(allTools)) {
    const short = key.split('__')[1];
    nameCount.set(short, (nameCount.get(short) ?? 0) + 1);
  }
  const result: Record<string, ReturnType<typeof tool>> = {};
  for (const [qualified, t] of Object.entries(allTools)) {
    const short = qualified.split('__')[1];
    result[nameCount.get(short)! > 1 ? qualified : short] = t;
  }
  return result;
}


集成到 Hono Agent
import { Hono } from 'hono';
import { streamText } from 'ai';
import { openai } from '@ai-sdk/openai';

const app = new Hono();
const registry = new MCPToolRegistry({
  servers: [
    {
      name: 'fs-server',
      transport: 'stdio',
      command: 'node',
      args: ['./mcp-servers/fs-server/dist/index.js'],
      enabled: true,
    },
    { name: 'db-server', transport: 'sse', url: 'http://localhost:3002/sse', enabled: true },
  ],
  cacheDir: './.cache/mcp',
  refreshInterval: 300_000,
});
await registry.initialize();

app.post('/api/chat', async (c) => {
  const { messages } = await c.req.json();
  const result = streamText({
    model: openai('gpt-4o'),
    messages,
    tools: mergeWithShortNames(registry),
    maxSteps: 15,
  });
  return result.toDataStreamResponse();
});

app.get('/api/mcp/status', (c) => c.json(registry.getStatus()));
app.post('/api/mcp/refresh', async (c) => {
  await registry.refreshAll();
  return c.json({ ok: true, status: registry.getStatus() });
});

process.on('SIGINT', async () => {
  await registry.shutdown();
  process.exit(0);
});
export default app;


小结
    

  • MCPToolRegistry 统一管理多个 MCP Server 的连接、工具发现和生命周期。
    • 

  • 使用配置文件声明 Server 列表，支持 stdio 和 SSE 两种传输方式混合使用。
    • 

  • 工具发现结果缓存到本地文件，Server 不可用时可从缓存恢复工具 Schema。
    • 

  • 通过 serverName__toolName 命名空间解决多 Server 同名工具冲突，全局唯一时自动使用短名称。
    • 

  • 定时刷新机制检测 Server 工具变更，getAISDKTools() 返回合并后的 AI SDK 格式工具集。
