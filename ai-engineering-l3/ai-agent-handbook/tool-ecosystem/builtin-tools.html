文件/Web/数据库工具

三类核心工具

Agent 最常用的工具分为三大类：文件操作、Web 交互和数据库查询。它们覆盖了绝大多数实际任务场景——读写本地文件、获取网络信息、查询结构化数据。

类别 工具 职责 超时
文件 readFile 读取文件内容 5s
文件 writeFile 写入文件内容 5s
文件 listFiles 列出目录文件 3s
文件 globFiles 按模式匹配文件 5s
Web httpFetch 发起 HTTP 请求 30s
Web webSearch 搜索引擎查询 15s
数据库 sqlQuery 执行 SQL 查询 10s
数据库 kvGet / kvSet 键值存储读写 3s

文件工具：readFile

import { tool } from 'ai';
import { z } from 'zod';
import { readFile as fsReadFile } from 'node:fs/promises';

const readFile = tool({
  description: '读取指定路径文件的文本内容',
  parameters: z.object({
    path: z.string().describe('文件路径，相对于项目根目录'),
    encoding: z.enum(['utf-8', 'ascii', 'latin1']).default('utf-8').describe('文件编码'),
    maxBytes: z.number().max(1_000_000).default(100_000).describe('最大读取字节数'),
  }),
  execute: async ({ path, encoding, maxBytes }) => {
    try {
      const buffer = await fsReadFile(path);
      if (buffer.length > maxBytes) {
        return { success: true, content: buffer.subarray(0, maxBytes).toString(encoding), truncated: true };
      }
      return { success: true, content: buffer.toString(encoding), truncated: false };
    } catch (err) {
      return { success: false, error: err instanceof Error ? err.message : '读取失败' };
    }
  },
});

文件工具：writeFile

import { writeFile as fsWriteFile, mkdir } from 'node:fs/promises';
import { dirname } from 'node:path';

const writeFile = tool({
  description: '将文本内容写入指定路径的文件，自动创建不存在的目录',
  parameters: z.object({
    path: z.string().describe('目标文件路径'),
    content: z.string().describe('要写入的文本内容'),
  }),
  execute: async ({ path, content }) => {
    try {
      await mkdir(dirname(path), { recursive: true });
      await fsWriteFile(path, content, 'utf-8');
      return { success: true, path, bytes: Buffer.byteLength(content) };
    } catch (err) {
      return { success: false, error: err instanceof Error ? err.message : '写入失败' };
    }
  },
});

文件工具：listFiles / globFiles

import { readdir } from 'node:fs/promises';
import { glob } from 'node:fs/promises';

const listFiles = tool({
  description: '列出目录下的文件和子目录名称',
  parameters: z.object({
    dir: z.string().default('.').describe('目录路径'),
    recursive: z.boolean().default(false).describe('是否递归列出子目录'),
  }),
  execute: async ({ dir, recursive }) => {
    try {
      const entries = await readdir(dir, { withFileTypes: true, recursive });
      const items = entries.map((e) => ({
        name: e.name,
        type: e.isDirectory() ? 'directory' : 'file',
      }));
      return { success: true, items, count: items.length };
    } catch (err) {
      return { success: false, error: err instanceof Error ? err.message : '目录读取失败' };
    }
  },
});

const globFiles = tool({
  description: '按 glob 模式匹配文件路径',
  parameters: z.object({
    pattern: z.string().describe('glob 模式，例如 src/**/*.ts'),
    cwd: z.string().default('.').describe('搜索起始目录'),
  }),
  execute: async ({ pattern, cwd }) => {
    try {
      const matches: string[] = [];
      for await (const entry of glob(pattern, { cwd })) {
        matches.push(entry);
        if (matches.length >= 500) break;
      }
      return { success: true, files: matches, count: matches.length };
    } catch (err) {
      return { success: false, error: err instanceof Error ? err.message : '匹配失败' };
    }
  },
});

Web 工具：httpFetch

const httpFetch = tool({
  description: '发起 HTTP 请求获取 URL 内容，支持 GET 和 POST 方法',
  parameters: z.object({
    url: z.string().url().describe('目标 URL'),
    method: z.enum(['GET', 'POST']).default('GET').describe('HTTP 方法'),
    headers: z.record(z.string()).optional().describe('请求头键值对'),
    body: z.string().optional().describe('POST 请求体'),
    maxBytes: z.number().max(5_000_000).default(500_000).describe('最大响应字节数'),
  }),
  execute: async ({ url, method, headers, body, maxBytes }) => {
    try {
      const controller = new AbortController();
      const timeout = setTimeout(() => controller.abort(), 30_000);
      const response = await fetch(url, {
        method,
        headers,
        body: method === 'POST' ? body : undefined,
        signal: controller.signal,
      });
      clearTimeout(timeout);
      const text = await response.text();
      return {
        success: true,
        status: response.status,
        content: text.slice(0, maxBytes),
        truncated: text.length > maxBytes,
      };
    } catch (err) {
      return { success: false, error: err instanceof Error ? err.message : '请求失败' };
    }
  },
});

Web 工具：webSearch

interface SearchResult {
  title: string;
  url: string;
  snippet: string;
}

const webSearch = tool({
  description: '使用搜索引擎搜索关键词，返回标题、链接和摘要',
  parameters: z.object({
    query: z.string().min(1).describe('搜索关键词'),
    limit: z.number().min(1).max(10).default(5).describe('最大结果数量'),
  }),
  execute: async ({ query, limit }): Promise<{ success: boolean; results?: SearchResult[]; error?: string }> => {
    try {
      const params = new URLSearchParams({ q: query, limit: String(limit) });
      const response = await fetch(`https://api.search.example/v1?${params}`);
      if (!response.ok) {
        return { success: false, error: `搜索失败: HTTP ${response.status}` };
      }
      const data = await response.json();
      return { success: true, results: data.items };
    } catch (err) {
      return { success: false, error: err instanceof Error ? err.message : '搜索请求失败' };
    }
  },
});

数据库工具：sqlQuery

import type { Pool } from 'pg';

function createSqlQueryTool(pool: Pool) {
  return tool({
    description: '执行只读 SQL 查询，返回结果行',
    parameters: z.object({
      sql: z.string().describe('SELECT 查询语句，禁止 INSERT/UPDATE/DELETE'),
      params: z.array(z.unknown()).default([]).describe('查询参数，防止 SQL 注入'),
    }),
    execute: async ({ sql, params }) => {
      const forbidden = /^\s*(INSERT|UPDATE|DELETE|DROP|ALTER|TRUNCATE)/i;
      if (forbidden.test(sql)) {
        return { success: false, error: '仅允许 SELECT 查询' };
      }
      try {
        const result = await pool.query(sql, params);
        return {
          success: true,
          rows: result.rows.slice(0, 100),
          rowCount: result.rowCount,
          truncated: (result.rowCount ?? 0) > 100,
        };
      } catch (err) {
        return { success: false, error: err instanceof Error ? err.message : '查询失败' };
      }
    },
  });
}

数据库工具：键值存储

function createKvTools(store: Map<string, string>) {
  const kvGet = tool({
    description: '从键值存储中读取指定 key 的值',
    parameters: z.object({
      key: z.string().min(1).describe('存储键名'),
    }),
    execute: async ({ key }) => {
      const value = store.get(key);
      return value !== undefined ? { success: true, key, value } : { success: false, error: `键 "${key}" 不存在` };
    },
  });

  const kvSet = tool({
    description: '向键值存储中写入键值对',
    parameters: z.object({
      key: z.string().min(1).describe('存储键名'),
      value: z.string().describe('存储值'),
      ttl: z.number().min(0).optional().describe('过期时间，单位秒'),
    }),
    execute: async ({ key, value, ttl }) => {
      store.set(key, value);
      if (ttl) {
        setTimeout(() => store.delete(key), ttl * 1000);
      }
      return { success: true, key };
    },
  });

  return { kvGet, kvSet };
}

统一注册

将所有内置工具注册到 generateText 的 tools 对象中：

import { generateText } from 'ai';
import { openai } from '@ai-sdk/openai';

const result = await generateText({
  model: openai('gpt-4o'),
  prompt: '读取 src/index.ts 的内容，然后搜索 Hono 框架的最新版本',
  tools: {
    readFile,
    writeFile,
    listFiles,
    globFiles,
    httpFetch,
    webSearch,
    sqlQuery: createSqlQueryTool(pool),
    ...createKvTools(new Map()),
  },
  maxSteps: 10,
});

小结

文件工具（readFile / writeFile / listFiles / globFiles）让 Agent 操作本地文件系统，Web 工具（httpFetch / webSearch）打通网络信息获取，数据库工具（sqlQuery / kvGet / kvSet）支持结构化数据查询和临时状态存储。所有工具遵循统一模式：Zod 校验参数、try-catch 捕获错误、返回 { success, data?, error? } 结构。通过 tools 对象统一注册，Agent 可以根据任务需求自由组合这些基础能力。
