Agent 安全工程

核心概念

Agent 安全是 AI 系统工程中最关键的横切关注点。Agent 拥有调用工具、访问数据和执行操作的能力，一旦被恶意利用或出现非预期行为，可能造成数据泄露、资源滥用甚至系统破坏。Agent 安全工程涵盖输入防护、输出过滤、权限控制和行为约束四个层面。

提示注入防护
提示注入是 Agent 系统面临的最普遍威胁。攻击者试图在用户输入中嵌入指令来覆盖系统提示，改变 Agent 的预期行为：


interface InjectionCheckResult {
  safe: boolean;
  riskLevel: 'none' | 'low' | 'medium' | 'high';
  detectedPatterns: string[];
}

const createInjectionDetector = () => {
  const patterns = [
    { regex: /ignore\s+(previous|above|all)\s+(instructions?|prompts?)/i, name: '指令覆盖' },
    { regex: /you\s+are\s+now\s+/i, name: '角色重定义' },
    { regex: /system\s*:\s*/i, name: '系统消息伪造' },
    { regex: /\[INST\]|\[\/INST\]|<\|im_start\|>/i, name: '特殊标记注入' },
    { regex: /forget\s+(everything|what)/i, name: '记忆清除' },
    { regex: /do\s+not\s+follow\s+(the|any)\s+(rules|instructions)/i, name: '规则绕过' },
    { regex: /pretend\s+(you\s+are|to\s+be)/i, name: '角色伪装' },
    { regex: /output\s+(the|your)\s+(system|initial)\s+(prompt|message)/i, name: '提示泄露' },
  ];

  return (input: string): InjectionCheckResult => {
    const detected: string[] = [];
    for (const { regex, name } of patterns) {
      if (regex.test(input)) detected.push(name);
    }

    const riskLevel =
      detected.length === 0 ? 'none' : detected.length === 1 ? 'low' : detected.length <= 3 ? 'medium' : 'high';

    return { safe: detected.length === 0, riskLevel, detectedPatterns: detected };
  };
};

const detector = createInjectionDetector();

const processUserInput = (input: string) => {
  const check = detector(input);
  if (check.riskLevel === 'high') {
    return { blocked: true, reason: `检测到高风险注入模式: ${check.detectedPatterns.join('、')}` };
  }
  if (check.riskLevel === 'medium') {
    return { blocked: false, warning: true, sanitizedInput: sanitizeInput(input) };
  }
  return { blocked: false, input };
};

const sanitizeInput = (input: string): string => {
  return input
    .replace(/\[INST\]|\[\/INST\]|<\|im_start\|>|<\|im_end\|>/gi, '')
    .replace(/system\s*:/gi, '系统:')
    .trim();
};
提示注入防护应该采用纵深防御策略——不仅在输入层检测，还需要在 Agent 的推理循环中监控行为异常。单靠正则匹配无法防御所有攻击，还需结合 LLM 辅助检测和行为分析。


权限与访问控制
Agent 的工具访问权限应该遵循最小权限原则——每个 Agent 只能访问完成其任务所需的最小资源集：


interface Permission {
  resource: string;
  actions: Array<'read' | 'write' | 'execute' | 'delete'>;
  conditions?: Record<string, unknown>;
}

interface AgentRole {
  name: string;
  permissions: Permission[];
  maxToolCallsPerMinute: number;
  maxCostPerRequest: number;
}

const createAccessController = () => {
  const roles = new Map<string, AgentRole>();

  roles.set('reader', {
    name: 'reader',
    permissions: [
      { resource: 'database', actions: ['read'], conditions: { tables: ['products', 'categories'] } },
      { resource: 'filesystem', actions: ['read'], conditions: { extensions: ['.md', '.txt'] } },
    ],
    maxToolCallsPerMinute: 30,
    maxCostPerRequest: 0.1,
  });

  roles.set('writer', {
    name: 'writer',
    permissions: [
      { resource: 'database', actions: ['read', 'write'], conditions: { tables: ['drafts', 'reviews'] } },
      { resource: 'filesystem', actions: ['read', 'write'] },
      { resource: 'api', actions: ['read', 'execute'] },
    ],
    maxToolCallsPerMinute: 50,
    maxCostPerRequest: 0.5,
  });

  const checkPermission = (
    role: string,
    resource: string,
    action: 'read' | 'write' | 'execute' | 'delete',
  ): { allowed: boolean; reason?: string } => {
    const agentRole = roles.get(role);
    if (!agentRole) return { allowed: false, reason: '未知角色' };

    const permission = agentRole.permissions.find((p) => p.resource === resource);
    if (!permission) return { allowed: false, reason: `角色 ${role} 无权访问 ${resource}` };

    if (!permission.actions.includes(action)) {
      return { allowed: false, reason: `角色 ${role} 无 ${action} 权限` };
    }

    return { allowed: true };
  };

  return { checkPermission, getRole: (name: string) => roles.get(name) };
};

输出安全与内容过滤
Agent 的输出同样需要安全检查，防止泄露敏感信息或生成有害内容：


interface ContentFilter {
  name: string;
  check: (content: string) => { safe: boolean; reason?: string; sanitized?: string };
}

const createOutputSanitizer = () => {
  const filters: ContentFilter[] = [
    {
      name: '个人信息脱敏',
      check: (content) => {
        const patterns = [
          { regex: /\b\d{3}[-.]?\d{4}[-.]?\d{4}\b/g, replacement: '***-****-****', type: '手机号' },
          { regex: /\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b/g, replacement: '***@***.com', type: '邮箱' },
          {
            regex: /\b\d{6}(19|20)\d{2}(0[1-9]|1[0-2])(0[1-9]|[12]\d|3[01])\d{3}[\dXx]\b/g,
            replacement: '****',
            type: '身份证',
          },
        ];

        let sanitized = content;
        const detected: string[] = [];
        for (const { regex, replacement, type } of patterns) {
          if (regex.test(sanitized)) {
            detected.push(type);
            sanitized = sanitized.replace(regex, replacement);
          }
        }

        return detected.length > 0
          ? { safe: false, reason: `检测到 ${detected.join('、')}`, sanitized }
          : { safe: true };
      },
    },
    {
      name: '系统信息泄露',
      check: (content) => {
        const sensitive = [
          /api[_-]?key\s*[:=]\s*\S+/gi,
          /password\s*[:=]\s*\S+/gi,
          /secret\s*[:=]\s*\S+/gi,
          /token\s*[:=]\s*\S+/gi,
          /-----BEGIN\s+(RSA\s+)?PRIVATE\s+KEY-----/i,
        ];

        for (const pattern of sensitive) {
          if (pattern.test(content)) {
            return { safe: false, reason: '检测到潜在的凭证信息泄露' };
          }
        }
        return { safe: true };
      },
    },
  ];

  return (content: string): { safe: boolean; content: string; warnings: string[] } => {
    let result = content;
    const warnings: string[] = [];

    for (const filter of filters) {
      const check = filter.check(result);
      if (!check.safe) {
        warnings.push(`[${filter.name}] ${check.reason}`);
        if (check.sanitized) result = check.sanitized;
        else return { safe: false, content: '内容已被安全策略拦截', warnings };
      }
    }

    return { safe: warnings.length === 0, content: result, warnings };
  };
};
行为审计与合规
完整的行为审计日志是安全事件调查和合规证明的基础：


interface AuditEntry {
  timestamp: string;
  sessionId: string;
  userId: string;
  action: string;
  resource: string;
  input: string;
  output: string;
  toolCalls: Array<{ tool: string; args: Record<string, unknown>; result: string }>;
  riskFlags: string[];
  duration: number;
}

const createAuditLogger = (config: {
  storage: { append: (entry: AuditEntry) => Promise<void> };
  sensitiveFields: string[];
}) => {
  const redactSensitive = (obj: Record<string, unknown>): Record<string, unknown> => {
    const redacted = { ...obj };
    for (const field of config.sensitiveFields) {
      if (field in redacted) redacted[field] = '[REDACTED]';
    }
    return redacted;
  };

  return {
    log: async (entry: Omit<AuditEntry, 'timestamp'>) => {
      const sanitizedEntry: AuditEntry = {
        ...entry,
        timestamp: new Date().toISOString(),
        toolCalls: entry.toolCalls.map((tc) => ({
          ...tc,
          args: redactSensitive(tc.args),
        })),
      };

      await config.storage.append(sanitizedEntry);
    },
  };
};
小结
    

  • Agent 安全工程涵盖输入防护、输出过滤、权限控制和行为约束四个层面
    • 

  • 提示注入防护需要纵深防御——正则检测 + LLM 辅助分析 + 行为监控
    • 

  • 权限控制遵循最小权限原则，每个 Agent 只能访问完成任务所需的最小资源集
    • 

  • 输出安全包括个人信息脱敏、凭证检测和有害内容过滤
    • 

  • 行为审计日志是安全事件调查和合规证明的基础，需要记录完整的工具调用链路
    • 

  • 安全是持续的过程，需要定期进行红队测试和安全评估
