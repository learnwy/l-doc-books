状态管理

34 行自定义 Store

state/store.ts 用仅 34 行代码实现了一个高性能的状态管理库，基于闭包封装状态、Object.is 相等性检查和 Set<Listener> 订阅管理。

type Listener = () => void;

interface Store<T> {
  getState: () => T;
  setState: (updater: (state: T) => T) => void;
  subscribe: (listener: Listener) => () => void;
}

function createStore<T>(initialState: T): Store<T> {
  let state = initialState;
  const listeners = new Set<Listener>();

  function getState(): T {
    return state;
  }

  function setState(updater: (state: T) => T): void {
    const nextState = updater(state);
    if (Object.is(state, nextState)) return;
    state = nextState;
    for (const listener of listeners) {
      listener();
    }
  }

  function subscribe(listener: Listener): () => void {
    listeners.add(listener);
    return () => listeners.delete(listener);
  }

  return { getState, setState, subscribe };
}

提示： 这个极简 Store 的性能超越了大多数状态管理库。没有中间件、没有 reducer、没有 action — 只有最核心的状态读写和订阅通知。






DeepImmutable 类型包装
DeepImmutable<T> 在类型层面强制状态不可变性，覆盖 AppState 的全部 100+ 字段。


type DeepImmutable<T> =
  T extends Map<infer K, infer V>
    ? ReadonlyMap<DeepImmutable<K>, DeepImmutable<V>>
    : T extends Set<infer S>
      ? ReadonlySet<DeepImmutable<S>>
      : T extends Array<infer E>
        ? ReadonlyArray<DeepImmutable<E>>
        : T extends object
          ? { readonly [K in keyof T]: DeepImmutable<T[K]> }
          : T;

type ImmutableAppState = DeepImmutable<AppState>;

React 集成
通过 useSyncExternalStore 将自定义 Store 与 React 渲染周期零开销集成。


function useAppState<S>(selector: (state: ImmutableAppState) => S): S {
  return useSyncExternalStore(
    appStore.subscribe,
    () => selector(appStore.getState()),
    () => selector(appStore.getState())
  );
}

function MessageCount(): React.ReactElement {
  const count = useAppState((s) => s.conversation.messages.length);
  return <Text>消息数: {count}</Text>;
}

function ModelName(): React.ReactElement {
  const model = useAppState((s) => s.model.currentModel);
  return <Text>模型: {model}</Text>;
}

function TokenStats(): React.ReactElement {
  const stats = useAppState((s) => ({
    input: s.stats.inputTokens,
    output: s.stats.outputTokens,
  }));
  return <Text>Tokens: {stats.input}/{stats.output}</Text>;
}

onChange 同步副作用
特定状态切片的变更会触发同步持久化副作用。


function setupOnChangeEffects(store: Store<ImmutableAppState>): void {
  let prevState = store.getState();

  store.subscribe(() => {
    const nextState = store.getState();

    if (prevState.settings !== nextState.settings) {
      persistSettings(nextState.settings);
    }

    if (prevState.model !== nextState.model) {
      persistModel(nextState.model);
    }

    if (prevState.ui.theme !== nextState.ui.theme) {
      applyTheme(nextState.ui.theme);
    }

    prevState = nextState;
  });
}

React Compiler 运行时
Claude Code 代码库使用 React Compiler（原 React Forget），通过 react/compiler-runtime 提供的 _c() 函数实现自动 memoization。


import { c as _c } from "react/compiler-runtime";

function MessageItem({ message }: { message: Message }): React.ReactElement {
  const $ = _c(4);

  let formattedTime;
  if ($[0] !== message.timestamp) {
    formattedTime = formatTime(message.timestamp);
    $[0] = message.timestamp;
    $[1] = formattedTime;
  } else {
    formattedTime = $[1];
  }

  let element;
  if ($[2] !== message || $[3] !== formattedTime) {
    element = (
      <Box>
        <Text dimColor>{formattedTime}</Text>
        <Text>{message.content}</Text>
      </Box>
    );
    $[2] = message;
    $[3] = formattedTime;
  } else {
    element = $[3];
  }

  return element;
}

Teammate 内存管理
多代理场景下的 transcript 内存管理，使用保留/驱逐策略和 30 秒宽限期。


interface TeammateTranscript {
  id: string;
  messages: Message[];
  lastAccessedAt: number;
  retained: boolean;
}

const GRACE_PERIOD = 30_000;

function evictStaleTranscripts(transcripts: Map<string, TeammateTranscript>): void {
  const now = Date.now();

  for (const [id, transcript] of transcripts) {
    if (transcript.retained) continue;

    if (now - transcript.lastAccessedAt > GRACE_PERIOD) {
      transcripts.delete(id);
    }
  }
}

function retainTranscript(id: string): void {
  const transcript = transcripts.get(id);
  if (transcript) {
    transcript.retained = true;
  }
}

function releaseTranscript(id: string): void {
  const transcript = transcripts.get(id);
  if (transcript) {
    transcript.retained = false;
    transcript.lastAccessedAt = Date.now();
  }
}




操作
行为
说明





retain
标记为保留
不会被驱逐




release
取消保留，重置时间戳
进入 30s 宽限期




evict
删除超过宽限期的未保留 transcript
释放内存







提示： 30 秒宽限期的设计是为了避免频繁的 retain/release 导致 transcript 过早被驱逐。在快速切换 teammate 时，刚释放的 transcript 不会立即被清理。
