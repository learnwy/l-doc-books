启动优化

导入时并行副作用

main.tsx 在文件顶部利用 ESM 模块的导入执行语义，在模块加载阶段就启动关键的并行副作用，而不是等到 main 函数执行。

import { prefetchMDMRaw } from './services/mdm';
import { prefetchKeychain } from './services/keychain';

const mdmPromise = prefetchMDMRaw();

const keychainPromise = process.platform === 'darwin' ? prefetchKeychain() : Promise.resolve(null);

async function main(): Promise<void> {
  const [mdmConfig, keychainData] = await Promise.all([mdmPromise, keychainPromise]);

  await startApp(mdmConfig, keychainData);
}

延迟预取
startDeferredPrefetches() 在应用启动后异步预取不紧急但后续会用到的数据。


function startDeferredPrefetches(): void {
  const prefetches = [
    () => initializeUser(),
    () => prefetchContext(),
    () => initAnalytics(),
    () => fetchModelCapabilities(),
  ];

  Promise.allSettled(prefetches.map((fn) => fn())).catch(() => {});
}

async function initializeUser(): Promise<void> {
  const user = await fetchCurrentUser();
  updateAppState((state) => ({ ...state, user }));
}

async function prefetchContext(): Promise<void> {
  const contextFiles = await findContextFiles(getWorkspaceRoot());
  updateAppState((state) => ({
    ...state,
    prefetchedContext: contextFiles,
  }));
}

async function fetchModelCapabilities(): Promise<void> {
  const capabilities = await getModelCapabilities(getCurrentModel());
  updateAppState((state) => ({
    ...state,
    modelCapabilities: capabilities,
  }));
}

工具懒加载
工具系统采用 import() 动态导入，避免在启动时加载所有工具模块。


const TOOL_REGISTRY: Record<string, () => Promise<ToolConstructor>> = {
  Read: () => import('./tools/Read').then((m) => m.ReadTool),
  Write: () => import('./tools/Write').then((m) => m.WriteTool),
  Bash: () => import('./tools/Bash').then((m) => m.BashTool),
  Glob: () => import('./tools/Glob').then((m) => m.GlobTool),
  Grep: () => import('./tools/Grep').then((m) => m.GrepTool),
  LS: () => import('./tools/LS').then((m) => m.LSTool),
};

const loadedTools = new Map<string, ToolConstructor>();

async function loadTool(name: string): Promise<ToolConstructor> {
  const cached = loadedTools.get(name);
  if (cached) return cached;

  const loader = TOOL_REGISTRY[name];
  if (!loader) throw new Error(`Unknown tool: ${name}`);

  const constructor = await loader();
  loadedTools.set(name, constructor);
  return constructor;
}


ToolSearch 延迟激活
ToolSearch 机制将 10+ 个工具推迟到关键词匹配时才激活。


interface DeferredTool {
  name: string;
  keywords: string[];
  loader: () => Promise<ToolConstructor>;
  loaded: boolean;
}

const deferredTools: DeferredTool[] = [
  {
    name: 'LSPTool',
    keywords: ['definition', 'reference', 'hover', 'symbol'],
    loader: () => import('./tools/LSP'),
    loaded: false,
  },
  {
    name: 'NotebookTool',
    keywords: ['notebook', 'jupyter', 'cell'],
    loader: () => import('./tools/Notebook'),
    loaded: false,
  },
  { name: 'MCPTool', keywords: ['mcp', 'server', 'resource'], loader: () => import('./tools/MCP'), loaded: false },
];

async function searchAndActivateTools(query: string): Promise<Tool[]> {
  const activated: Tool[] = [];
  const lowerQuery = query.toLowerCase();

  for (const deferred of deferredTools) {
    if (deferred.loaded) continue;

    const matches = deferred.keywords.some((kw) => lowerQuery.includes(kw));
    if (matches) {
      const module = await deferred.loader();
      deferred.loaded = true;
      activated.push(new module.default());
    }
  }

  return activated;
}

setup.ts 后台任务
setup.ts 在初始化阶段编排多个后台任务并行执行。


async function setupBackground(): Promise<void> {
  const backgroundJobs = [
    initSessionMemory(),
    initContextCollapseEngine(),
    lockVersion(),
    prefetchCommands(getWorkspaceRoot()),
    startPluginWatcher(),
  ];

  await Promise.allSettled(backgroundJobs);
}

async function prefetchCommands(cwd: string): Promise<void> {
  const commands = await getCommands(cwd);
  commandCache.set(cwd, commands);
}

async function startPluginWatcher(): Promise<void> {
  const watcher = createFileWatcher(getPluginDir(), {
    onChange: () => reinitializeLspServerManager(),
    interval: 5000,
  });

  registerCleanup(() => watcher.close());
}

命令预取
getCommands(cwd) 在 setup 阶段预加载工作目录的可用命令，加速后续的命令补全。


const commandCache = new Map<string, string[]>();

async function getCommands(cwd: string): Promise<string[]> {
  const cached = commandCache.get(cwd);
  if (cached) return cached;

  const pathDirs = (process.env.PATH ?? '').split(':');
  const commands: Set<string> = new Set();

  await Promise.all(
    pathDirs.map(async (dir) => {
      try {
        const entries = await fs.readdir(dir);
        for (const entry of entries) {
          commands.add(entry);
        }
      } catch {
        // 忽略不可访问的目录
      }
    }),
  );

  const result = Array.from(commands).sort();
  commandCache.set(cwd, result);
  return result;
}


插件热重载
后台轮询监测插件目录变化，变更时触发 LSP 服务器管理器重新初始化。


function createFileWatcher(dir: string, options: { onChange: () => void; interval: number }): FileWatcher {
  let lastSnapshot = '';

  const timer = setInterval(async () => {
    try {
      const snapshot = await computeDirectoryHash(dir);
      if (snapshot !== lastSnapshot) {
        lastSnapshot = snapshot;
        options.onChange();
      }
    } catch {
      // 目录不存在时忽略
    }
  }, options.interval);

  return {
    close: () => clearInterval(timer),
  };
}


提示： 插件热重载使用轮询而非 fs.watch，因为 fs.watch 在跨平台场景（特别是 Docker 和网络文件系统）中行为不一致。5 秒轮询间隔在可靠性和性能之间取得了平衡。






小结
    

  • 导入时副作用利用 ESM 模块加载语义，在模块顶层启动 MDM 和 Keychain 的并行预取。
    • 

  • 延迟预取使用 Promise.allSettled 异步加载用户信息、上下文文件和模型能力，不阻塞 UI 渲染。
    • 

  • 工具懒加载通过 import() 动态导入和 ToolSearch 关键词匹配，将 10+ 个工具的加载推迟到使用时。
    • 

  • setup.ts 编排 5 个后台任务并行执行，包括会话记忆初始化、上下文折叠引擎和命令预取。
    • 

  • 插件热重载使用 5 秒轮询监测目录变化，变更时触发 LSP 管理器重新初始化。
