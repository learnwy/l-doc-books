LSP 集成

闭包工厂客户端

services/lsp/LSPClient.ts 采用闭包工厂模式创建 LSP 客户端，每个客户端封装独立的连接状态和生命周期。

function createLSPClient(serverName: string, onCrash: (serverName: string) => void): LSPClient {
  let connection: MessageConnection | null = null;
  let childProcess: ChildProcess | null = null;
  let ready = false;
  const pendingHandlers: Array<(conn: MessageConnection) => void> = [];

  async function start(command: string, args: string[]): Promise<void> {
    childProcess = spawn(command, args, { stdio: 'pipe' });
    const reader = new StreamMessageReader(childProcess.stdout!);
    const writer = new StreamMessageWriter(childProcess.stdin!);
    connection = createMessageConnection(reader, writer);

    childProcess.on('exit', (code) => {
      if (code !== 0 && code !== null) {
        onCrash(serverName);
      }
      connection = null;
      ready = false;
    });

    connection.listen();
    await initialize(connection);
    ready = true;
    pendingHandlers.forEach((handler) => handler(connection!));
    pendingHandlers.length = 0;
  }

  function onReady(handler: (conn: MessageConnection) => void): void {
    if (ready && connection) {
      handler(connection);
    } else {
      pendingHandlers.push(handler);
    }
  }

  return { start, onReady, shutdown, isReady: () => ready };
}

提示： 闭包模式的优势在于完全隐藏内部状态，外部无法直接访问 connection 或 childProcess，只能通过公开 API 操作。






JSON-RPC 通信
LSP 基于 JSON-RPC 2.0 协议，通过子进程的 stdio 管道传输消息。


async function initialize(connection: MessageConnection): Promise<ServerCapabilities> {
  const result = await connection.sendRequest<InitializeResult>('initialize', {
    processId: process.pid,
    capabilities: {
      textDocument: {
        definition: { dynamicRegistration: true },
        references: { dynamicRegistration: true },
        hover: { contentFormat: ['markdown', 'plaintext'] },
        documentSymbol: { hierarchicalDocumentSymbolSupport: true },
      },
      workspace: {
        workspaceFolders: true,
        symbol: { dynamicRegistration: true },
      },
    },
    rootUri: workspaceRootUri,
    workspaceFolders: [{ uri: workspaceRootUri, name: 'workspace' }],
  });

  connection.sendNotification('initialized', {});
  return result.capabilities;
}

排队处理器注册
在连接就绪之前注册的处理器会被排队，就绪后按顺序执行。


function registerHandlers(client: LSPClient): void {
  client.onReady((connection) => {
    connection.onNotification('textDocument/publishDiagnostics', (params) => {
      handleDiagnostics(params);
    });

    connection.onNotification('window/logMessage', (params) => {
      logServerMessage(params);
    });

    connection.onRequest('window/showMessageRequest', (params) => {
      return handleMessageRequest(params);
    });
  });
}


崩溃检测与恢复
当 LSP 服务器进程意外退出时，通过回调通知管理器进行恢复。


function createLSPClient(serverName: string, onCrash: (serverName: string) => void): LSPClient {
  let crashCount = 0;
  const MAX_CRASH_RETRIES = 3;

  function handleProcessExit(code: number | null): void {
    if (code !== 0 && code !== null) {
      crashCount++;
      if (crashCount <= MAX_CRASH_RETRIES) {
        onCrash(serverName);
      }
    }
  }

  return {
    /* ... */
  };
}


优雅关闭
LSP 规范要求关闭时按序执行 4 个步骤，确保服务器正确释放资源。


async function shutdown(client: LSPClient): Promise<void> {
  const connection = client.getConnection();
  if (!connection) return;

  try {
    await connection.sendRequest('shutdown');
  } catch {
    // 服务器可能已经退出
  }

  try {
    connection.sendNotification('exit');
  } catch {
    // 忽略通知发送失败
  }

  connection.dispose();

  const proc = client.getProcess();
  if (proc && !proc.killed) {
    proc.kill('SIGTERM');
    setTimeout(() => {
      if (!proc.killed) proc.kill('SIGKILL');
    }, 5000);
  }
}

LSP 管理器
管理器使用单例模式管理所有 LSP 客户端，通过生代计数器（generation counter）防止过期的异步操作。


let manager: LSPManager | null = null;
let generation = 0;

function getLspServerManager(): LSPManager {
  if (!manager) {
    manager = createLSPManager();
  }
  return manager;
}

async function reinitializeLspServerManager(): Promise<void> {
  generation++;
  const currentGeneration = generation;

  if (manager) {
    await manager.shutdownAll();
    manager = null;
  }

  manager = createLSPManager();
  await manager.initializeAll();

  if (generation !== currentGeneration) {
    return;
  }
}


提示： 生代计数器是防止并发初始化竞争的经典模式。如果在初始化过程中又触发了 reinitialize，先前的操作会因 generation 不匹配而被自动丢弃。






LSPTool 九种操作
LSPTool 暴露 9 种代码导航操作，作为工具供 AI 代理使用。


interface LSPTool {
  operations: LSPOperation[];
  isDeferred: boolean;
  isEnabled: () => boolean;
}

type LSPOperation =
  | 'goToDefinition'
  | 'findReferences'
  | 'hover'
  | 'documentSymbol'
  | 'workspaceSymbol'
  | 'goToImplementation'
  | 'prepareCallHierarchy'
  | 'incomingCalls'
  | 'outgoingCalls';

async function executeLSPOperation(operation: LSPOperation, params: LSPOperationParams): Promise<LSPOperationResult> {
  const client = getLspServerManager().getClient(params.languageId);
  if (!client || !client.isReady()) {
    throw new Error('LSP server not connected');
  }

  switch (operation) {
    case 'goToDefinition':
      return client.sendRequest('textDocument/definition', params);
    case 'findReferences':
      return client.sendRequest('textDocument/references', params);
    case 'hover':
      return client.sendRequest('textDocument/hover', params);
    case 'documentSymbol':
      return client.sendRequest('textDocument/documentSymbol', params);
    case 'workspaceSymbol':
      return client.sendRequest('workspace/symbol', params);
    case 'goToImplementation':
      return client.sendRequest('textDocument/implementation', params);
    case 'prepareCallHierarchy':
      return client.sendRequest('textDocument/prepareCallHierarchy', params);
    case 'incomingCalls':
      return client.sendRequest('callHierarchy/incomingCalls', params);
    case 'outgoingCalls':
      return client.sendRequest('callHierarchy/outgoingCalls', params);
  }
}

小结
    

  • LSP 客户端使用闭包工厂模式，每个 createLSPClient() 调用创建封装独立状态的实例。
    • 

  • 基于 JSON-RPC 2.0 协议通过子进程 stdio 通信，支持请求、响应和通知三种消息类型。
    • 

  • 排队处理器机制允许在连接就绪前注册回调，就绪后自动执行。
    • 

  • 管理器通过生代计数器防止并发 reinitialize 导致的竞态条件。
    • 

  • LSPTool 暴露 9 种代码导航操作，采用延迟加载策略，仅在 LSP 连接后启用。
