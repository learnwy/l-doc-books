流式处理与重试

withRetry 架构

src/services/withRetry.ts（822 行）实现了 Claude Code 的核心重试机制。它基于 AsyncGenerator 模式设计——在重试过程中可以 yield 错误消息到 UI，让用户实时看到重试进度，而不是阻塞等待。

// src/services/withRetry.ts — 核心结构（简化）
async function* withRetry<T>(
  operation: () => AsyncGenerator<T>,
  config: RetryConfig,
): AsyncGenerator<T | ErrorMessage> {
  let attempt = 0;
  let lastError: Error | null = null;

  while (attempt < config.maxRetries) {
    try {
      yield* operation();
      return;
    } catch (error) {
      attempt++;
      lastError = error as Error;

      const strategy = classifyError(error);
      const shouldRetry = strategy.retryable && attempt < config.maxRetries;

      yield {
        type: 'system',
        content: formatRetryMessage(error, attempt, shouldRetry),
        timestamp: Date.now(),
      } as ErrorMessage;

      if (!shouldRetry) throw error;

      await sleep(strategy.getDelay(attempt));

      if (strategy.refreshCredentials) {
        await strategy.refreshCredentials();
      }
    }
  }

  throw lastError;
}

四提供商客户端工厂

// src/services/claude.ts — 客户端工厂（简化）
type ClientProvider = 'direct' | 'bedrock' | 'vertex' | 'foundry';

interface ClientConfig {
  provider: ClientProvider;
  apiKey?: string;
  baseUrl?: string;
  region?: string;
  projectId?: string;
  modelMapping?: Record<string, string>;
}

function createAPIClient(config: ClientConfig): AnthropicClient {
  switch (config.provider) {
    case 'direct':
      return new Anthropic({
        apiKey: config.apiKey,
        baseURL: config.baseUrl,
        defaultHeaders: { 'anthropic-beta': 'prompt-caching-2024-07-31' },
      });

    case 'bedrock':
      return new AnthropicBedrock({
        awsRegion: config.region,
        awsAccessKey: process.env.AWS_ACCESS_KEY_ID,
        awsSecretKey: process.env.AWS_SECRET_ACCESS_KEY,
        awsSessionToken: process.env.AWS_SESSION_TOKEN,
      });

    case 'vertex':
      return new AnthropicVertex({
        region: config.region,
        projectId: config.projectId,
      });

    case 'foundry':
      return createFoundryAdapter(config);
  }
}

错误分类与重试策略
不同类型的 API 错误有差异化的重试策略：


// src/services/withRetry.ts — 错误分类（简化）
interface RetryStrategy {
  retryable: boolean;
  maxRetries: number;
  getDelay: (attempt: number) => number;
  refreshCredentials?: () => Promise<void>;
  fallbackModel?: string;
}

function classifyError(error: unknown): RetryStrategy {
  if (isHTTPError(error)) {
    switch (error.status) {
      case 429:
        return {
          retryable: true,
          maxRetries: 5,
          getDelay: (attempt) => {
            const retryAfter = parseRetryAfter(error.headers);
            return retryAfter ?? exponentialBackoff(attempt, 1000, 60000);
          },
        };

      case 529:
        return {
          retryable: true,
          maxRetries: 3,
          getDelay: (attempt) => exponentialBackoff(attempt, 2000, 30000),
          fallbackModel: getFallbackModel(),
        };

      case 401:
        return {
          retryable: true,
          maxRetries: 1,
          getDelay: () => 0,
          refreshCredentials: refreshOAuthToken,
        };

      case 413:
        return {
          retryable: false,
          maxRetries: 0,
          getDelay: () => 0,
        };

      default:
        return {
          retryable: error.status >= 500,
          maxRetries: 3,
          getDelay: (attempt) => exponentialBackoff(attempt, 1000, 30000),
        };
    }
  }

  if (isNetworkError(error)) {
    return {
      retryable: true,
      maxRetries: 3,
      getDelay: (attempt) => exponentialBackoff(attempt, 500, 15000),
    };
  }

  return { retryable: false, maxRetries: 0, getDelay: () => 0 };
}

指数退避实现
// src/services/withRetry.ts — 退避算法（简化）
function exponentialBackoff(attempt: number, baseDelay: number, maxDelay: number): number {
  const delay = baseDelay * Math.pow(2, attempt - 1);
  const jitter = delay * 0.1 * Math.random();
  return Math.min(delay + jitter, maxDelay);
}


提示： 退避算法添加了 10% 的随机抖动（jitter），避免多个客户端在相同时间点同时重试造成的"惊群效应"。






Fallback 模型切换
当 529 overloaded 错误达到最大重试次数后，系统会自动切换到 fallback 模型：


// src/services/withRetry.ts — Fallback 切换（简化）
class FallbackTriggeredError extends Error {
  constructor(
    public originalModel: string,
    public fallbackModel: string,
    public reason: string,
  ) {
    super(`Switching from ${originalModel} to ${fallbackModel}: ${reason}`);
  }
}

async function* withFallback<T>(
  operation: (model: string) => AsyncGenerator<T>,
  config: FallbackConfig,
): AsyncGenerator<T | SDKMessage> {
  try {
    yield* withRetry(() => operation(config.primaryModel), config.retryConfig);
  } catch (error) {
    if (error instanceof FallbackTriggeredError) {
      yield {
        type: 'system',
        content: `模型 ${config.primaryModel} 不可用，正在切换到 ${config.fallbackModel}...`,
        timestamp: Date.now(),
      };

      yield* withRetry(() => operation(config.fallbackModel), config.fallbackRetryConfig);
    } else {
      throw error;
    }
  }
}


持久重试与凭证刷新
在无人值守场景（CI/CD、--print 模式），Claude Code 启用持久重试模式。与标准重试不同，持久重试会无限循环直到成功或超时（默认 30 分钟），并通过 yield 心跳消息让 CI 系统知道任务仍在运行。

401 错误触发凭证自动刷新，支持三种认证方式：Direct（OAuth refresh_token 刷新）、Bedrock（AWS STS 重新获取）、Vertex（GCP OAuth 重新获取）。刷新后的令牌通过 updateGlobalConfig() 持久化到 ~/.claude.json。




提示： 持久重试模式特别适合 GitHub Actions 等 CI 环境。当 API 出现短暂过载时，任务会持续等待而不是立即失败，显著提高了自动化流水线的稳定性。
