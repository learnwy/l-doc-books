App 组合架构

Claude Code 的终端应用通过 Screen 路由和 Context Provider 嵌套组合而成，screens/REPL.tsx 是主交互界面，screens/Doctor.tsx 是诊断界面。

function App(): React.ReactElement {
  const [screen, setScreen] = useState<"repl" | "doctor">("repl");

  return (
    <FPSMetricsProvider>
      <MailboxProvider>
        <ModalProvider>
          <NotificationsProvider>
            <StatsProvider>
              <VoiceProvider>
                {screen === "repl" ? (
                  <REPL onNavigate={setScreen} />
                ) : (
                  <Doctor onBack={() => setScreen("repl")} />
                )}
              </VoiceProvider>
            </StatsProvider>
          </NotificationsProvider>
        </ModalProvider>
      </MailboxProvider>
    </FPSMetricsProvider>
  );
}

FPS 指标监控
context/fpsMetrics.tsx 实现帧率监控，跟踪渲染性能。


interface FPSMetrics {
  currentFPS: number;
  averageFPS: number;
  minFPS: number;
  maxFPS: number;
  frameCount: number;
  droppedFrames: number;
}

function FPSMetricsProvider({ children }: { children: React.ReactNode }): React.ReactElement {
  const [metrics, setMetrics] = useState<FPSMetrics>({
    currentFPS: 0, averageFPS: 0, minFPS: Infinity, maxFPS: 0,
    frameCount: 0, droppedFrames: 0,
  });

  const lastFrameTime = useRef(performance.now());
  const frameTimes = useRef<number[]>([]);

  const recordFrame = useCallback(() => {
    const now = performance.now();
    const delta = now - lastFrameTime.current;
    lastFrameTime.current = now;

    frameTimes.current.push(delta);
    if (frameTimes.current.length > 60) frameTimes.current.shift();

    const fps = 1000 / delta;
    const avg = 1000 / (frameTimes.current.reduce((a, b) => a + b, 0) / frameTimes.current.length);

    setMetrics((prev) => ({
      currentFPS: Math.round(fps),
      averageFPS: Math.round(avg),
      minFPS: Math.min(prev.minFPS, fps),
      maxFPS: Math.max(prev.maxFPS, fps),
      frameCount: prev.frameCount + 1,
      droppedFrames: prev.droppedFrames + (delta > 50 ? 1 : 0),
    }));
  }, []);

  return (
    <FPSMetricsContext.Provider value={{ metrics, recordFrame }}>
      {children}
    </FPSMetricsContext.Provider>
  );
}




指标
计算方式
用途





currentFPS
1000 / 当前帧间隔
实时帧率




averageFPS
滑动窗口（60 帧）平均
平均性能




droppedFrames
帧间隔 > 50ms 计为丢帧
卡顿检测







Mailbox 消息队列
context/mailbox.tsx 实现了组件间的异步消息传递机制。


interface MailboxMessage {
  id: string;
  type: string;
  payload: unknown;
  timestamp: number;
}

interface MailboxContext {
  send: (type: string, payload: unknown) => void;
  subscribe: (type: string, handler: (msg: MailboxMessage) => void) => () => void;
  peek: (type: string) => MailboxMessage | undefined;
}

function MailboxProvider({ children }: { children: React.ReactNode }): React.ReactElement {
  const queue = useRef<MailboxMessage[]>([]);
  const subscribers = useRef<Map<string, Set<(msg: MailboxMessage) => void>>>(new Map());

  const send = useCallback((type: string, payload: unknown) => {
    const msg: MailboxMessage = {
      id: crypto.randomUUID(),
      type,
      payload,
      timestamp: Date.now(),
    };

    queue.current.push(msg);

    const handlers = subscribers.current.get(type);
    if (handlers) {
      for (const handler of handlers) {
        handler(msg);
      }
    }
  }, []);

  const subscribe = useCallback((type: string, handler: (msg: MailboxMessage) => void) => {
    const set = subscribers.current.get(type) ?? new Set();
    set.add(handler);
    subscribers.current.set(type, set);
    return () => set.delete(handler);
  }, []);

  return (
    <MailboxContext.Provider value={{ send, subscribe, peek: /* ... */ }}>
      {children}
    </MailboxContext.Provider>
  );
}


Modal 覆盖层
context/modal.tsx 管理模态对话框，支持堆栈式的多层覆盖。


interface ModalState {
  stack: ModalEntry[];
  push: (content: React.ReactElement, options?: ModalOptions) => string;
  pop: (id?: string) => void;
  current: ModalEntry | null;
}

interface ModalEntry {
  id: string;
  content: React.ReactElement;
  dismissable: boolean;
  onDismiss?: () => void;
}

function ModalProvider({ children }: { children: React.ReactNode }): React.ReactElement {
  const [stack, setStack] = useState<ModalEntry[]>([]);

  const push = useCallback((content: React.ReactElement, options?: ModalOptions) => {
    const id = crypto.randomUUID();
    setStack((prev) => [...prev, {
      id,
      content,
      dismissable: options?.dismissable ?? true,
      onDismiss: options?.onDismiss,
    }]);
    return id;
  }, []);

  const pop = useCallback((id?: string) => {
    setStack((prev) => {
      if (id) return prev.filter((m) => m.id !== id);
      return prev.slice(0, -1);
    });
  }, []);

  const current = stack[stack.length - 1] ?? null;

  return (
    <ModalContext.Provider value={{ stack, push, pop, current }}>
      {children}
      {current && (
        <Box style={{ position: "absolute", top: 0, left: 0, width: "100%", height: "100%" }}>
          {current.content}
        </Box>
      )}
    </ModalContext.Provider>
  );
}

interface BridgeConfig {

interface BuddyContext {
  isActive: boolean;
  buddyId: string | null;
  sendToBuddy: (message: string) => Promise<void>;
  onBuddyMessage: (handler: (message: string) => void) => () => void;
}

function BuddyProvider({ children }: { children: React.ReactNode }): React.ReactElement {
  const [isActive, setIsActive] = useState(false);
  const [buddyId, setBuddyId] = useState<string | null>(null);

  return (

REPL 主界面
screens/REPL.tsx 是用户的主要交互界面，组合了输入区域、输出区域和工具栏。


function REPL({ onNavigate }: { onNavigate: (screen: string) => void }): React.ReactElement {
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState("");
  const appState = useAppState((s) => s.conversation);

  return (
    <Box style={{ flexDirection: "column", height: "100%" }}>
      <Box style={{ flexGrow: 1, overflow: "scroll" }}>
        <MessageList messages={messages} />
      </Box>

      <Box style={{ borderTop: true, padding: 1 }}>
        <InputArea
          value={input}
          onChange={setInput}
          onSubmit={handleSubmit}
        />
      </Box>

      <Box style={{ height: 1 }}>
        <StatusBar stats={appState.stats} />
      </Box>
    </Box>
  );
}


提示： REPL 界面使用 flexbox 的 column 布局，消息列表区域使用 flexGrow: 1 占据所有剩余空间，输入区和状态栏固定高度。
