事件系统

事件类型体系

events/ 目录定义了终端 UI 的事件类型体系，将原始终端输入抽象为结构化的 DispatcherEvent。

interface ClickEvent {
  type: 'click';
  x: number;
  y: number;
  button: 'left' | 'right' | 'middle';
}

interface FocusEvent {
  type: 'focus' | 'blur';
  target: DOMElement;
}

interface InputEvent {
  type: 'input';
  key: string;
  raw: Buffer;
  ctrl: boolean;
  alt: boolean;
  shift: boolean;
  meta: boolean;
}

type DispatcherEvent = ClickEvent | FocusEvent | InputEvent;

EventDispatcher 路由
dispatcher.ts 实现事件路由，将事件分发到当前获得焦点的组件或全局处理器。


class EventDispatcher {
  private focusedElement: DOMElement | null = null;
  private globalHandlers: Map<string, EventHandler[]> = new Map();
  private elementHandlers: Map<DOMElement, Map<string, EventHandler>> = new Map();

  dispatch(event: DispatcherEvent): boolean {
    if (this.dispatchToGlobalHandlers(event)) {
      return true;
    }

    if (this.focusedElement) {
      return this.dispatchToElement(this.focusedElement, event);
    }

    return false;
  }

  private dispatchToGlobalHandlers(event: DispatcherEvent): boolean {
    const handlers = this.globalHandlers.get(event.type) ?? [];
    for (const handler of handlers) {
      if (handler(event)) return true;
    }
    return false;
  }

  private dispatchToElement(element: DOMElement, event: DispatcherEvent): boolean {
    const handlers = this.elementHandlers.get(element);
    if (!handlers) return false;

    const handler = handlers.get(event.type);
    if (handler) return handler(event);

    if (element.parent) {
      return this.dispatchToElement(element.parent, event);
    }

    return false;
  }

  setFocus(element: DOMElement | null): void {
    if (this.focusedElement === element) return;

    if (this.focusedElement) {
      this.dispatch({ type: 'blur', target: this.focusedElement });
    }

    this.focusedElement = element;

    if (element) {
      this.dispatch({ type: 'focus', target: element });
    }
  }

  onGlobal(eventType: string, handler: EventHandler): () => void {
    const handlers = this.globalHandlers.get(eventType) ?? [];
    handlers.push(handler);
    this.globalHandlers.set(eventType, handlers);
    return () => {
      const idx = handlers.indexOf(handler);
      if (idx !== -1) handlers.splice(idx, 1);
    };
  }
}

提示： 事件分发采用了与浏览器 DOM 类似的冒泡机制。全局处理器类似于浏览器的 window 事件监听，优先级最高。






EventEmitter 类型化发布订阅
emitter.ts 实现了类型安全的发布-订阅模式。


type EventMap = Record<string, unknown>;

class EventEmitter<T extends EventMap> {
  private listeners = new Map<keyof T, Set<(data: unknown) => void>>();

  on<K extends keyof T>(event: K, listener: (data: T[K]) => void): () => void {
    const set = this.listeners.get(event) ?? new Set();
    set.add(listener as (data: unknown) => void);
    this.listeners.set(event, set);

    return () => set.delete(listener as (data: unknown) => void);
  }

  emit<K extends keyof T>(event: K, data: T[K]): void {
    const set = this.listeners.get(event);
    if (set) {
      for (const listener of set) {
        listener(data);
      }
    }
  }

  removeAllListeners(event?: keyof T): void {
    if (event) {
      this.listeners.delete(event);
    } else {
      this.listeners.clear();
    }
  }
}

interface UIEvents {
  render: void;
  resize: { rows: number; cols: number };
  keypress: InputEvent;
  exit: { code: number };
}

const uiEmitter = new EventEmitter<UIEvents>();


Hooks 体系
hooks/ 目录提供一组 React hooks，封装终端交互的常见模式。


function useInput(handler: (input: string, key: KeyInfo) => void): void {
  const { stdin } = useApp();

  useEffect(() => {
    const onData = (data: Buffer) => {
      const parsed = parseKeyInput(data);
      handler(parsed.input, parsed.key);
    };

    stdin.on('data', onData);
    return () => stdin.off('data', onData);
  }, [handler, stdin]);
}

function useStdin(): { stdin: NodeJS.ReadStream; isRawMode: boolean; setRawMode: (mode: boolean) => void } {
  const { stdin } = useApp();
  const [isRawMode, setIsRawMode] = useState(false);

  const setRawMode = useCallback(
    (mode: boolean) => {
      stdin.setRawMode(mode);
      setIsRawMode(mode);
    },
    [stdin],
  );

  return { stdin, isRawMode, setRawMode };
}

function useApp(): AppContext {
  return useContext(AppContext);
}

function useInterval(callback: () => void, delay: number | null): void {
  const savedCallback = useRef(callback);

  useEffect(() => {
    savedCallback.current = callback;
  }, [callback]);

  useEffect(() => {
    if (delay === null) return;
    const id = setInterval(() => savedCallback.current(), delay);
    return () => clearInterval(id);
  }, [delay]);
}

键绑定系统
keybindings/ 目录将按键组合映射为命名动作。


interface KeyBinding {
  key: string;
  ctrl?: boolean;
  alt?: boolean;
  shift?: boolean;
  action: string;
}

const defaultKeyBindings: KeyBinding[] = [
  { key: 'c', ctrl: true, action: 'interrupt' },
  { key: 'd', ctrl: true, action: 'exit' },
  { key: 'l', ctrl: true, action: 'clear' },
  { key: 'z', ctrl: true, action: 'suspend' },
  { key: 'escape', action: 'cancel' },
  { key: 'tab', action: 'autocomplete' },
  { key: 'return', action: 'submit' },
  { key: 'up', action: 'history-prev' },
  { key: 'down', action: 'history-next' },
];

function matchKeyBinding(event: InputEvent, bindings: KeyBinding[]): string | null {
  for (const binding of bindings) {
    if (
      event.key === binding.key &&
      (binding.ctrl ?? false) === event.ctrl &&
      (binding.alt ?? false) === event.alt &&
      (binding.shift ?? false) === event.shift
    ) {
      return binding.action;
    }
  }
  return null;
}

输入处理全链路
从终端接收原始字节到组件响应，经过 5 个处理阶段。


function setupInputPipeline(stdin: NodeJS.ReadStream, dispatcher: EventDispatcher, keyBindings: KeyBinding[]): void {
  stdin.setRawMode(true);
  stdin.resume();

  stdin.on('data', (data: Buffer) => {
    const tokens = tokenize(data);

    for (const token of tokens) {
      const parsed = parseToken(token);
      const event: InputEvent = {
        type: 'input',
        key: parsed.key,
        raw: data,
        ctrl: parsed.ctrl,
        alt: parsed.alt,
        shift: parsed.shift,
        meta: parsed.meta,
      };

      const action = matchKeyBinding(event, keyBindings);
      if (action) {
        dispatcher.dispatch({ type: 'action', action });
        return;
      }

      dispatcher.dispatch(event);
    }
  });
}


提示： Raw mode 是终端输入处理的关键。在 raw mode 下，终端不会进行行缓冲和特殊字符处理，所有按键都直接传递给程序，包括 Ctrl+C。
