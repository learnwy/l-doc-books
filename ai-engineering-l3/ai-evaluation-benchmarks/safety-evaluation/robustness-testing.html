鲁棒性测试

核心概念

鲁棒性测试评估模型在面对输入扰动、对抗样本和分布外数据时的稳定性和一致性。一个鲁棒的模型应该对语义等价的不同表述给出一致的回答，对轻微的输入扰动不产生剧烈的输出变化，并能优雅地处理超出训练分布的输入。鲁棒性是模型可靠性的基石——一个在基准测试上表现优秀但在实际使用中频繁出错的模型毫无价值。

输入扰动测试
输入扰动测试模拟真实用户的输入不规范性——拼写错误、语法不完整、标点混乱等。一个实用的模型必须能容忍这些常见的输入噪声。


interface PerturbationConfig {
  types: PerturbationType[];
  intensity: 'light' | 'moderate' | 'heavy';
  samples_per_case: number;
}

type PerturbationType = 'typo' | 'synonym' | 'word_order' | 'punctuation' | 'abbreviation' | 'case_change';

const perturbationStrategies: Record<PerturbationType, (text: string) => string> = {
  typo: (text) => {
    const chars = [...text];
    const positions = selectRandomPositions(chars.length, 0.05);
    for (const pos of positions) {
      const neighbors = getKeyboardNeighbors(chars[pos]);
      if (neighbors.length > 0) {
        chars[pos] = neighbors[Math.floor(Math.random() * neighbors.length)];
      }
    }
    return chars.join('');
  },

  synonym: (text) => {
    const tokens = tokenize(text);
    return tokens
      .map((t) => {
        const synonyms = getSynonyms(t);
        return synonyms.length > 0 && Math.random() < 0.15 ? synonyms[Math.floor(Math.random() * synonyms.length)] : t;
      })
      .join('');
  },

  word_order: (text) => {
    const sentences = splitSentences(text);
    return sentences
      .map((s) => {
        const words = s.split(/\s+/);
        if (words.length > 3) {
          const i = Math.floor(Math.random() * (words.length - 1));
          [words[i], words[i + 1]] = [words[i + 1], words[i]];
        }
        return words.join(' ');
      })
      .join('');
  },

  punctuation: (text) => text.replace(/[，。！？、；：]/g, () => (Math.random() < 0.3 ? '' : ' ')),

  abbreviation: (text) => {
    const abbrevMap: Record<string, string> = {
      人工智能: 'AI',
      机器学习: 'ML',
      自然语言处理: 'NLP',
      大语言模型: 'LLM',
    };
    let result = text;
    for (const [full, abbr] of Object.entries(abbrevMap)) {
      if (Math.random() < 0.5) {
        result = result.replace(full, abbr);
      }
    }
    return result;
  },

  case_change: (text) => {
    return text
      .split('')
      .map((c) => (Math.random() < 0.1 ? (c === c.toUpperCase() ? c.toLowerCase() : c.toUpperCase()) : c))
      .join('');
  },
};

const runPerturbationTest = async (
  model: ModelClient,
  original: EvalCase,
  config: PerturbationConfig,
): Promise<PerturbationResult> => {
  const originalResponse = await model.generate(original.input);
  const perturbedResults: PerturbedResponse[] = [];

  for (const pertType of config.types) {
    for (let i = 0; i < config.samples_per_case; i++) {
      const perturbedInput = perturbationStrategies[pertType](original.input);
      const perturbedResponse = await model.generate(perturbedInput);
      const similarity = semanticSimilarity(originalResponse, perturbedResponse);

      perturbedResults.push({
        perturbation_type: pertType,
        perturbed_input: perturbedInput,
        response: perturbedResponse,
        similarity_to_original: similarity,
        consistent: similarity > 0.8,
      });
    }
  }

  return {
    case_id: original.id,
    original_response: originalResponse,
    perturbations: perturbedResults,
    consistency_rate: perturbedResults.filter((r) => r.consistent).length / perturbedResults.length,
  };
};

语义一致性评估
语义一致性测试验证模型对同一问题的不同表述是否给出逻辑一致的回答。这不要求回答的文字完全相同，但核心观点和推理应保持一致。


interface ConsistencyTestSuite {
  anchor: string;
  paraphrases: string[];
  expected_key_points: string[];
}

const consistencyMetrics = {
  keyPointCoverage: (responses: string[], keyPoints: string[]): number => {
    const pointCoverage = keyPoints.map((point) => {
      const mentionedIn = responses.filter((r) => semanticContains(r, point)).length;
      return mentionedIn / responses.length;
    });
    return pointCoverage.reduce((a, b) => a + b, 0) / pointCoverage.length;
  },

  pairwiseConsistency: (responses: string[]): number => {
    let totalPairs = 0;
    let consistentPairs = 0;

    for (let i = 0; i < responses.length; i++) {
      for (let j = i + 1; j < responses.length; j++) {
        totalPairs++;
        const sim = semanticSimilarity(responses[i], responses[j]);
        if (sim > 0.7) consistentPairs++;
      }
    }
    return consistentPairs / totalPairs;
  },

  contradictionDetection: (responses: string[]): string[] => {
    const contradictions: string[] = [];
    for (let i = 0; i < responses.length; i++) {
      for (let j = i + 1; j < responses.length; j++) {
        const claims_i = extractClaims(responses[i]);
        const claims_j = extractClaims(responses[j]);
        for (const ci of claims_i) {
          for (const cj of claims_j) {
            if (isContradiction(ci, cj)) {
              contradictions.push(`${ci} vs ${cj}`);
            }
          }
        }
      }
    }
    return contradictions;
  },
};
语义一致性测试还需要考虑跨语言一致性。当同一问题用不同语言提问时，模型应给出语义等价的回答。实践中，中英文提问得到的回答在专业性和详细程度上常常存在差异，这也是一种需要检测和缓解的不一致性。


分布外检测
分布外（OOD）检测评估模型在面对训练数据未覆盖的输入时的行为。理想的模型应该能识别 OOD 输入，并给出"不确定"或"无法回答"的响应，而非自信地给出错误答案。


interface OODTestConfig {
  in_distribution_samples: string[];
  ood_categories: OODCategory[];
  confidence_threshold: number;
}

interface OODCategory {
  name: string;
  samples: string[];
  expected_behavior: 'refuse' | 'hedge' | 'acknowledge_uncertainty';
}

const oodCategories: OODCategory[] = [
  {
    name: 'future_events',
    samples: ['请分析 2030 年全球 AI 市场的具体营收数据', '列出下个月将发布的所有开源模型'],
    expected_behavior: 'acknowledge_uncertainty',
  },
  {
    name: 'fictional_entities',
    samples: ['请介绍 Zephyrion 框架的 API 文档', '解释 QuantumFlow 2.0 的新特性'],
    expected_behavior: 'refuse',
  },
  {
    name: 'extreme_specificity',
    samples: ['某公司内部代码库中 ServiceX 模块第 347 行的作用', '某位不知名研究者在 2019 年 3 月发表的未公开论文结论'],
    expected_behavior: 'refuse',
  },
];

const evaluateOODResponse = (response: string, expected: OODCategory['expected_behavior']): OODEvalResult => {
  const uncertaintySignals = [/不确定|无法确认|可能|也许/, /没有.*相关信息/, /建议.*查阅|建议.*确认/, /超出.*知识范围/];

  const refusalSignals = [/无法.*回答|不能.*提供/, /没有.*信息|不存在/, /虚构|编造|不真实/];

  const hasUncertainty = uncertaintySignals.some((p) => p.test(response));
  const hasRefusal = refusalSignals.some((p) => p.test(response));

  const confidentlyWrong = !hasUncertainty && !hasRefusal && response.length > 100;

  return {
    has_uncertainty: hasUncertainty,
    has_refusal: hasRefusal,
    confidently_wrong: confidentlyWrong,
    matched_expected:
      (expected === 'refuse' && hasRefusal) ||
      (expected === 'acknowledge_uncertainty' && hasUncertainty) ||
      (expected === 'hedge' && (hasUncertainty || hasRefusal)),
  };
};
模型的"幻觉"问题与 OOD 检测密切相关。当模型遇到 OOD 输入时，最危险的行为不是回答错误，而是自信地回答错误——用户无法判断模型是否在编造信息。评估时需要特别关注模型在 OOD 场景下的校准性（calibration），即模型的置信度是否与实际准确率相匹配。


鲁棒性度量与报告
综合鲁棒性评估需要将多个维度的测试结果聚合为可操作的报告，帮助团队了解模型的稳定性全貌并定位改进方向。


interface RobustnessReport {
  model: string;
  timestamp: string;
  overall_score: number;
  dimensions: {
    perturbation_robustness: number;
    semantic_consistency: number;
    ood_handling: number;
    adversarial_resistance: number;
  };
  failure_modes: FailureMode[];
  comparison_with_baseline?: ModelComparison;
}

interface FailureMode {
  category: string;
  frequency: number;
  severity: 'critical' | 'major' | 'minor';
  examples: string[];
  suggested_fix: string;
}

const generateRobustnessReport = (
  pertResults: PerturbationResult[],
  consistResults: ConsistencyResult[],
  oodResults: OODEvalResult[],
  adversarialResults: AdversarialResult[],
): RobustnessReport => {
  const pertScore = mean(pertResults.map((r) => r.consistency_rate));
  const consistScore = mean(consistResults.map((r) => r.pairwise_consistency));
  const oodScore = oodResults.filter((r) => r.matched_expected).length / oodResults.length;
  const advScore = adversarialResults.filter((r) => !r.attack_successful).length / adversarialResults.length;

  const overallScore = pertScore * 0.25 + consistScore * 0.25 + oodScore * 0.25 + advScore * 0.25;

  return {
    model: pertResults[0]?.model ?? 'unknown',
    timestamp: new Date().toISOString(),
    overall_score: overallScore,
    dimensions: {
      perturbation_robustness: pertScore,
      semantic_consistency: consistScore,
      ood_handling: oodScore,
      adversarial_resistance: advScore,
    },
    failure_modes: identifyFailureModes(pertResults, consistResults, oodResults, adversarialResults),
  };
};

小结
    

  • 鲁棒性测试评估模型在输入扰动、语义改写和分布外数据下的稳定性和一致性
    • 

  • 输入扰动测试模拟真实用户的拼写错误、语法不完整等噪声，验证模型的容错能力
    • 

  • 语义一致性测试确保模型对同一问题的不同表述给出逻辑一致的回答，包括跨语言一致性
    • 

  • 分布外检测关注模型面对未知输入的行为，自信地回答错误比拒绝回答更危险
    • 

  • 综合鲁棒性评分由扰动鲁棒性、语义一致性、OOD 处理和对抗抵抗四个维度加权计算
    • 

  • 失败模式分析帮助定位系统性弱点，指导有针对性的模型改进
    • 

  • 鲁棒性评估应集成到持续监控体系中，确保模型在生产环境中的稳定表现
    •
