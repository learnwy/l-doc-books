事件溯源与 CQRS 模式

概述

事件溯源（Event Sourcing）和 CQRS（Command Query Responsibility Segregation，命令查询职责分离）是事件驱动架构中两个互补的高级设计模式。事件溯源将系统状态的变更记录为不可变的事件序列，而非直接存储当前状态；CQRS 将读模型和写模型分离，允许各自独立优化。两者结合构成了高度可扩展、可审计、可演化的数据架构。

本文深入分析事件溯源和 CQRS 的核心机制、实现模式和工程权衡，帮助工程师在合适的场景中应用这些高级模式。

核心概念

事件溯源基本原理

传统系统直接存储和修改当前状态；事件溯源系统存储导致状态变化的所有事件：

传统 CRUD:
  Account Balance = 1000 (直接覆写)
  UPDATE accounts SET balance = 1000 WHERE id = 1;

事件溯源:
  Event 1: AccountCreated { id: 1, balance: 0 }
  Event 2: MoneyDeposited { id: 1, amount: 500 }
  Event 3: MoneyDeposited { id: 1, amount: 800 }
  Event 4: MoneyWithdrawn { id: 1, amount: 300 }
  → 当前状态 = replay(events) = 0 + 500 + 800 - 300 = 1000

事件溯源的核心特性：

特性 描述 价值
不可变性 事件一旦写入不可修改 完整审计追踪
完整历史 保留所有状态变更过程 时间旅行、回溯分析
事件回放 从事件序列重建任意时间点的状态 灵活的物化视图
松耦合 生产者不知道消费者如何使用事件 系统可演化性

事件存储设计
class EventStore:
    """事件存储的核心接口"""

    async def append(self, stream_id: str, events: list, expected_version: int):
        """追加事件到流（乐观并发控制）"""
        current_version = await self._get_version(stream_id)
        if current_version != expected_version:
            raise ConcurrencyConflict(
                f"Expected version {expected_version}, got {current_version}"
            )
        for event in events:
            await self._persist_event(stream_id, event, current_version + 1)
            current_version += 1
        # 发布事件到订阅者
        await self._publish(stream_id, events)

    async def read_stream(self, stream_id: str, from_version: int = 0):
        """读取某个聚合根的完整事件流"""
        return await self._load_events(stream_id, from_version)

    async def subscribe(self, category: str, handler, from_position: int = 0):
        """订阅某类事件（用于构建投影）"""
        position = from_position
        while True:
            events = await self._poll(category, position, batch_size=100)
            for event in events:
                await handler(event)
                position = event.global_position + 1
聚合根与命令处理
class BankAccount:
    """事件溯源风格的聚合根"""

    def __init__(self, account_id):
        self.id = account_id
        self.balance = 0
        self.is_active = False
        self.version = 0
        self._pending_events = []

    # --- 命令处理 (决策) ---
    def open(self, initial_deposit):
        if self.is_active:
            raise InvalidOperation("Account already active")
        if initial_deposit < 0:
            raise InvalidOperation("Initial deposit must be positive")
        self._apply(AccountOpened(self.id, initial_deposit))

    def deposit(self, amount):
        if not self.is_active:
            raise InvalidOperation("Account not active")
        if amount <= 0:
            raise InvalidOperation("Amount must be positive")
        self._apply(MoneyDeposited(self.id, amount))

    def withdraw(self, amount):
        if not self.is_active:
            raise InvalidOperation("Account not active")
        if amount > self.balance:
            raise InsufficientFunds(self.balance, amount)
        self._apply(MoneyWithdrawn(self.id, amount))

    # --- 事件应用 (状态变更) ---
    def _apply(self, event):
        self._handle_event(event)
        self._pending_events.append(event)

    def _handle_event(self, event):
        match event:
            case AccountOpened(_, deposit):
                self.is_active = True
                self.balance = deposit
            case MoneyDeposited(_, amount):
                self.balance += amount
            case MoneyWithdrawn(_, amount):
                self.balance -= amount
        self.version += 1

    @classmethod
    def from_events(cls, account_id, events):
        """从事件流重建聚合根状态"""
        account = cls(account_id)
        for event in events:
            account._handle_event(event)
        return account
投影（Projection）
投影是将事件流转化为查询友好的读模型的过程：


class AccountBalanceProjection:
    """构建账户余额查询视图"""

    def __init__(self, read_db):
        self.db = read_db

    async def handle(self, event):
        match event:
            case AccountOpened(account_id, deposit):
                await self.db.execute(
                    "INSERT INTO account_balances (id, balance, status) VALUES (?, ?, ?)",
                    [account_id, deposit, 'active']
                )
            case MoneyDeposited(account_id, amount):
                await self.db.execute(
                    "UPDATE account_balances SET balance = balance + ? WHERE id = ?",
                    [amount, account_id]
                )
            case MoneyWithdrawn(account_id, amount):
                await self.db.execute(
                    "UPDATE account_balances SET balance = balance - ? WHERE id = ?",
                    [amount, account_id]
                )

class TransactionHistoryProjection:
    """构建交易历史查询视图（不同的读模型）"""

    async def handle(self, event):
        if isinstance(event, (MoneyDeposited, MoneyWithdrawn)):
            await self.db.execute(
                "INSERT INTO transactions (account_id, type, amount, timestamp) VALUES (?, ?, ?, ?)",
                [event.account_id, event.__class__.__name__, event.amount, event.timestamp]
            )
快照策略
当事件流过长时，每次重建聚合根需要回放大量事件。快照定期保存聚合根的当前状态：


class SnapshotStore:
    """快照管理"""

    SNAPSHOT_INTERVAL = 100  # 每 100 个事件创建快照

    async def load_aggregate(self, aggregate_id):
        # 1. 尝试加载最近的快照
        snapshot = await self._load_latest_snapshot(aggregate_id)

        if snapshot:
            aggregate = self._restore_from_snapshot(snapshot)
            from_version = snapshot.version + 1
        else:
            aggregate = BankAccount(aggregate_id)
            from_version = 0

        # 2. 回放快照之后的事件
        events = await self.event_store.read_stream(aggregate_id, from_version)
        for event in events:
            aggregate._handle_event(event)

        # 3. 如果回放事件过多，创建新快照
        if aggregate.version - (snapshot.version if snapshot else 0) > self.SNAPSHOT_INTERVAL:
            await self._save_snapshot(aggregate)

        return aggregate

from confluent_kafka import Producer, Consumer

# 生产者: 发布事件
producer = Producer({'bootstrap.servers': 'kafka:9092'})

def publish_event(stream_id, event):
    producer.produce(
        topic='account-events',
        key=stream_id.encode(),         # 分区键 = 聚合根 ID
        value=json.dumps(event).encode(),
        headers=[('event-type', event['type'].encode())]
    )
    producer.flush()

# 消费者: 构建投影
consumer = Consumer({
    'bootstrap.servers': 'kafka:9092',
    'group.id': 'balance-projection',
    'auto.offset.reset': 'earliest',
    'enable.auto.commit': False
})
consumer.subscribe(['account-events'])

while True:

事件版本演化
# 事件 Schema 会随时间演化，需要向后兼容
class EventUpgrader:
    """事件版本升级器"""

    def upgrade(self, event):
        event_type = event['type']
        version = event.get('version', 1)

        # V1 → V2: MoneyDeposited 新增 currency 字段
        if event_type == 'MoneyDeposited' and version < 2:
            event['currency'] = 'USD'  # 默认值
            event['version'] = 2

        # V2 → V3: 重命名字段
        if event_type == 'MoneyDeposited' and version < 3:
            event['deposit_amount'] = event.pop('amount')
            event['version'] = 3

        return event

总结
事件溯源和 CQRS 是强大但复杂的架构模式。事件溯源通过保存完整的状态变更历史，实现了审计追踪、时间旅行和灵活的物化视图；CQRS 通过读写分离，允许各自独立优化和扩展。两者结合是事件驱动架构的终极形态。


然而，这些模式的复杂度不容忽视——最终一致性带来的 UI 设计挑战、事件版本演化的长期维护成本、投影故障的排查难度——都是工程师需要权衡的因素。2026 年的最佳实践是"选择性应用"：对核心领域（如金融账务、订单状态）使用事件溯源获得审计和追溯能力，对辅助域使用传统 CRUD 保持简单。Kafka 作为事件骨干 + 专用投影数据库的组合，已成为事件驱动架构的主流技术栈。
