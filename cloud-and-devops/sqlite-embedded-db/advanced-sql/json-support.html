JSON 支持

JSON 函数概述

SQLite 从 3.9.0 版本开始内置 JSON 函数支持，3.38.0 起默认编译启用（之前需要 -DSQLITE_ENABLE_JSON1）。这些函数让你可以直接在 SQL 中存储、查询和操作 JSON 数据，无需在应用层进行解析。

SQLite 中 JSON 以 TEXT 类型存储。从 3.45.0 起新增了 JSONB 二进制格式，存储为 BLOB，解析和序列化更高效。

提取与查询
json_extract() 使用 JSON Path 语法从 JSON 数据中提取值：


CREATE TABLE products (
    id   INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    meta TEXT NOT NULL
);

INSERT INTO products VALUES
(1, '笔记本电脑', '{"brand":"Apple","specs":{"ram":16,"storage":512},"tags":["pro","laptop"]}'),
(2, '智能手机', '{"brand":"Samsung","specs":{"ram":8,"storage":256},"tags":["phone","5g"]}'),
(3, '平板电脑', '{"brand":"Apple","specs":{"ram":8,"storage":128},"tags":["tablet","education"]}');

-- 提取顶层字段
SELECT name, json_extract(meta, '$.brand') AS brand FROM products;

-- 提取嵌套字段
SELECT name, json_extract(meta, '$.specs.ram') AS ram_gb FROM products;

-- 提取数组元素
SELECT name, json_extract(meta, '$.tags[0]') AS first_tag FROM products;
-> 和 ->> 操作符（3.38+）提供了更简洁的语法：


-- -> 返回 JSON 值（保留引号）
SELECT name, meta -> '$.brand' AS brand_json FROM products;

-- ->> 返回 SQL 值（去除引号）
SELECT name, meta ->> '$.brand' AS brand_text FROM products;

-- 在 WHERE 中使用
SELECT name FROM products WHERE meta ->> '$.brand' = 'Apple';

-- 嵌套路径
SELECT name FROM products WHERE meta ->> '$.specs.ram' >= 16;


提示： ->> 操作符返回的是 SQL 原生类型（TEXT/INTEGER/REAL），而 -> 返回的是 JSON 文本。在 WHERE 条件和索引中，通常使用 ->> 更合适。






修改 JSON
-- json_set：设置值（存在则更新，不存在则插入）
SELECT json_set('{"a":1,"b":2}', '$.c', 3);
-- {"a":1,"b":2,"c":3}

-- json_replace：仅更新已存在的键
SELECT json_replace('{"a":1,"b":2}', '$.a', 10, '$.c', 3);
-- {"a":10,"b":2}（$.c 不存在，被忽略）

-- json_insert：仅插入不存在的键
SELECT json_insert('{"a":1,"b":2}', '$.a', 10, '$.c', 3);
-- {"a":1,"b":2,"c":3}（$.a 已存在，被忽略）

-- json_remove：删除键
SELECT json_remove('{"a":1,"b":2,"c":3}', '$.b');
-- {"a":1,"c":3}

-- json_patch：合并两个 JSON 对象（RFC 7396）
SELECT json_patch('{"a":1,"b":2}', '{"b":null,"c":3}');
-- {"a":1,"c":3}

UPDATE products

表值函数
json_each() 和 json_tree() 将 JSON 数据展开为虚拟表行：


-- json_each：展开数组或对象的直接子元素
SELECT
    p.name,
    je.value AS tag
FROM products p, json_each(p.meta, '$.tags') je;

-- json_tree：递归展开所有层级
SELECT
    key, value, type, path
FROM json_tree('{"a":{"b":[1,2],"c":"x"}}');

-- 查找包含特定 tag 的产品
SELECT DISTINCT p.name
FROM products p, json_each(p.meta, '$.tags') je
WHERE je.value = 'pro';

-- 统计每个品牌的产品数量和平均 RAM
SELECT
    meta ->> '$.brand' AS brand,
    COUNT(*) AS count,
    AVG(CAST(meta ->> '$.specs.ram' AS INTEGER)) AS avg_ram
FROM products
GROUP BY 1;


聚合为 JSON
-- json_group_array：将多行聚合为 JSON 数组
SELECT json_group_array(name) AS all_products FROM products;
-- ["笔记本电脑","智能手机","平板电脑"]

-- json_group_object：将多行聚合为 JSON 对象
SELECT json_group_object(name, meta ->> '$.brand') AS product_brands
FROM products;
-- {"笔记本电脑":"Apple","智能手机":"Samsung","平板电脑":"Apple"}

-- 构建嵌套 JSON 结构
SELECT json_group_object(
    brand,
    json(product_list)
) AS by_brand
FROM (
    SELECT
        meta ->> '$.brand' AS brand,
        json_group_array(name) AS product_list
    FROM products
    GROUP BY 1
);


JSON 索引
在 JSON 字段上创建索引可以显著加速查询：


-- 在 JSON 路径上创建索引
CREATE INDEX idx_products_brand ON products(json_extract(meta, '$.brand'));

-- 使用 ->> 操作符创建索引（3.38+）
CREATE INDEX idx_products_ram ON products(meta ->> '$.specs.ram');

-- 查询将自动使用索引
SELECT name FROM products WHERE meta ->> '$.brand' = 'Apple';


提示： JSON 路径索引是 表达式索引（Expression Index）的一种。确保 WHERE 子句中的表达式与索引定义完全一致，否则索引不会被使用。






小结
    

  • SQLite 的 JSON 函数（3.9+）支持在 SQL 中直接操作 JSON 数据。
    • 

  • ->> 操作符（3.38+）返回 SQL 原生类型，适合在 WHERE 和索引中使用。
    • 

  • json_each 和 json_tree 将 JSON 展开为虚拟表行，实现关系化查询。
    • 

  • 通过表达式索引可以加速 JSON 路径查询。
    • 

  • JSONB（3.45+）以二进制格式存储 JSON，解析性能更高。
