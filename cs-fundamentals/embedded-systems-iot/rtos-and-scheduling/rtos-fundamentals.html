RTOS 基础原理

核心概念

实时操作系统（Real-Time Operating System, RTOS）是专为嵌入式系统设计的轻量级操作系统内核，核心职责是任务调度、资源管理和时间确定性保证。与通用操作系统（Linux/Windows）追求吞吐量不同，RTOS 追求的是可预测的响应时间——在硬实时系统中，错过截止时间等同于系统失败。理解 RTOS 的任务模型、调度机制和同步原语，是构建可靠嵌入式系统的关键。

任务状态模型
RTOS 中的任务（Task/Thread）有明确的状态转换模型。调度器根据任务的当前状态和优先级决定哪个任务获得 CPU 时间：


// FreeRTOS 任务创建
TaskHandle_t sensorTask;
xTaskCreate(
    sensorTaskFunc,      // 任务函数
    "Sensor",            // 任务名（调试用）
    256,                 // 栈大小（words）
    NULL,                // 参数
    2,                   // 优先级（数字越大越高）
    &sensorTask          // 任务句柄
);

void sensorTaskFunc(void *pvParameters) {
    while (1) {
        readSensor();
        processData();
        vTaskDelay(pdMS_TO_TICKS(100));  // 释放 CPU 100ms
    }
}

同步与通信原语
信号量与互斥量
// 二值信号量：ISR → 任务 的事件通知
SemaphoreHandle_t uartSem = xSemaphoreCreateBinary();

void USART1_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    xSemaphoreGiveFromISR(uartSem, &xHigherPriorityTaskWoken);
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void uartTask(void *p) {
    while (1) {
        xSemaphoreTake(uartSem, portMAX_DELAY);
        processUartData();
    }
}

// 互斥量：保护共享资源（支持优先级继承）
SemaphoreHandle_t spiMutex = xSemaphoreCreateMutex();

void spiTransfer(uint8_t *data, size_t len) {
    xSemaphoreTake(spiMutex, portMAX_DELAY);
    HAL_SPI_Transmit(&hspi1, data, len, 1000);
    xSemaphoreGive(spiMutex);
}
消息队列
// 消息队列：任务间类型安全的数据传递
typedef struct {
    uint16_t sensorId;
    float value;
    uint32_t timestamp;
} SensorReading;

QueueHandle_t sensorQueue = xQueueCreate(10, sizeof(SensorReading));

void sensorTask(void *p) {
    while (1) {
        SensorReading reading = {
            .sensorId = 1,
            .value = readTemperature(),
            .timestamp = xTaskGetTickCount()
        };
        xQueueSend(sensorQueue, &reading, pdMS_TO_TICKS(100));
        vTaskDelay(pdMS_TO_TICKS(500));
    }
}

void processingTask(void *p) {
    SensorReading reading;
    while (1) {
        if (xQueueReceive(sensorQueue, &reading, portMAX_DELAY) == pdTRUE) {
            logReading(&reading);
        }
    }
}

优先级反转与继承
优先级反转是 RTOS 中的经典问题——低优先级任务持有高优先级任务需要的互斥量，而中优先级任务抢占了低优先级任务，导致高优先级任务间接被中优先级任务阻塞：


// 优先级反转场景：
// 高优先级任务 H 等待互斥量 M
// 低优先级任务 L 持有 M
// 中优先级任务 M 抢占 L
// → H 被 M 间接阻塞（反转）

// 解决方案：优先级继承
// FreeRTOS 的 xSemaphoreCreateMutex() 自动支持
// 当 H 等待 L 持有的互斥量时，L 临时提升到 H 的优先级

小结
    

  • RTOS 的核心价值是确定性的任务调度和微秒级的响应延迟
    • 

  • 任务有 Running、Ready、Blocked、Suspended 四种状态，调度器始终运行最高优先级的就绪任务
    • 

  • 二值信号量用于 ISR→任务的事件通知，互斥量用于共享资源保护（支持优先级继承）
    • 

  • 消息队列是任务间传递数据的标准方式，比全局变量更安全
    • 

  • 优先级反转通过优先级继承或天花板协议解决
    • 

  • FreeRTOS 的 heap_4 是通用推荐的内存分配策略，兼顾灵活性和碎片控制
    •
