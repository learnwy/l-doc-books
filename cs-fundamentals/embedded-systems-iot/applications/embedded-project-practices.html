嵌入式项目工程实践

核心概念

嵌入式项目从原型到量产的过程中，工程实践的质量直接决定了产品的可靠性和可维护性。与纯软件项目不同，嵌入式工程需要同时管理硬件依赖、实时约束、资源限制和现场调试等特殊挑战。本文覆盖固件架构设计、版本管理、持续集成、调试技术和生产部署的全流程最佳实践。

固件分层架构
良好的固件架构将硬件依赖隔离在底层，使业务逻辑可以独立于硬件开发和测试：


// 分层架构
// Layer 4: 应用层（Application）— 业务逻辑
// Layer 3: 服务层（Service）— 协议处理、状态管理
// Layer 2: 驱动层（Driver）— 外设驱动封装
// Layer 1: HAL 层（Hardware Abstraction）— 芯片厂商 HAL
// Layer 0: 硬件（Hardware）— MCU + 外设

// 项目目录结构
// firmware/
// ├── CMakeLists.txt
// ├── src/
// │   ├── app/           # 应用层
// │   │   ├── main.c
// │   │   └── task_sensor.c
// │   ├── service/       # 服务层
// │   │   ├── mqtt_client.c
// │   │   └── data_logger.c
// │   ├── driver/        # 驱动层
// │   │   ├── bme280.c
// │   │   └── ssd1306.c
// │   └── hal/           # HAL 抽象接口
// │       ├── hal_i2c.h
// │       ├── hal_gpio.h
// │       └── hal_uart.h
// ├── test/              # 单元测试（桌面运行）
// │   ├── test_bme280.c
// │   └── mock_hal_i2c.c
// └── platform/          # 平台特定实现
//     ├── stm32f4/
//     └── native/        # 桌面模拟
HAL 抽象接口：


// hal/hal_i2c.h — 硬件无关接口
typedef struct {
    int (*init)(uint32_t speed);
    int (*write)(uint8_t addr, const uint8_t *data, size_t len);
    int (*read)(uint8_t addr, uint8_t *data, size_t len);
    int (*mem_read)(uint8_t addr, uint8_t reg, uint8_t *data, size_t len);
} hal_i2c_t;

// platform/stm32f4/hal_i2c_stm32.c — STM32 实现
static int stm32_i2c_mem_read(uint8_t addr, uint8_t reg, uint8_t *data, size_t len) {
    return HAL_I2C_Mem_Read(&hi2c1, addr << 1, reg,
                            I2C_MEMADD_SIZE_8BIT, data, len, 100) == HAL_OK ? 0 : -1;
}

// platform/native/hal_i2c_mock.c — 桌面模拟实现
static int mock_i2c_mem_read(uint8_t addr, uint8_t reg, uint8_t *data, size_t len) {
    memcpy(data, &mock_registers[reg], len);
    return 0;
}
CMake 构建系统
CMake 是嵌入式项目的标准构建系统，支持交叉编译和多平台构建：


cmake_minimum_required(VERSION 3.20)

set(CMAKE_TOOLCHAIN_FILE ${CMAKE_SOURCE_DIR}/cmake/arm-gcc-toolchain.cmake)

project(iot-sensor LANGUAGES C)

add_executable(firmware
    src/app/main.c
    src/app/task_sensor.c
    src/service/mqtt_client.c
    src/driver/bme280.c
)

target_include_directories(firmware PRIVATE
    src/hal
    src/driver
    src/service
)

target_compile_options(firmware PRIVATE
    -mcpu=cortex-m4
    -mthumb
    -mfloat-abi=hard
    -mfpu=fpv4-sp-d16
    -Os
    -Wall -Werror
    -ffunction-sections
    -fdata-sections
)

target_link_options(firmware PRIVATE
    -T${CMAKE_SOURCE_DIR}/linker/STM32F411CE.ld
    -Wl,--gc-sections
    -Wl,--print-memory-usage
)
交叉编译工具链文件：

// SWO 调试输出（比 UART 更快，不占用串口）
#include <stdio.h>
int _write(int fd, char *ptr, int len) {
    for (int i = 0; i < len; i++) {
        ITM_SendChar(ptr[i]);
    }
    return len;
}

// 断言与故障记录
void HardFault_Handler(void) {
    __asm volatile(
        "tst lr, #4          \n"
        "ite eq              \n"
        "mrseq r0, msp       \n"
        "mrsne r0, psp       \n"
        "b hard_fault_handler \n"
    );
}

void hard_fault_handler(uint32_t *stack_frame) {
    volatile uint32_t pc = stack_frame[6];
    volatile uint32_t lr = stack_frame[5];
    // 将故障信息写入备份寄存器，重启后可读取
    RTC->BKP0R = pc;
    RTC->BKP1R = lr;
    NVIC_SystemReset();
}

嵌入式测试策略
// 单元测试（桌面运行，使用 Unity 框架）
#include "unity.h"
#include "bme280.h"
#include "mock_hal_i2c.h"

void test_bme280_read_temperature(void) {
    mock_i2c_set_register(0xFA, (uint8_t[]){0x80, 0x00, 0x00}, 3);
    mock_i2c_set_calibration(dig_T1, dig_T2, dig_T3);

    float temp = bme280_read_temperature();
    TEST_ASSERT_FLOAT_WITHIN(0.1, 25.0, temp);
}

void test_bme280_init_failure(void) {
    mock_i2c_set_failure(true);
    int result = bme280_init();
    TEST_ASSERT_EQUAL(-1, result);
}

小结
    

  • 固件分层架构（应用→服务→驱动→HAL）将硬件依赖隔离在底层，使业务逻辑可独立测试
    • 

  • HAL 抽象接口允许同一驱动代码在真实硬件和桌面模拟环境下运行
    • 

  • CMake 是嵌入式标准构建系统，通过工具链文件实现交叉编译
    • 

  • 嵌入式调试需要 JTAG/SWD + 日志 + 逻辑分析仪多层工具协同
    • 

  • HardFault 处理应记录故障现场（PC、LR）到备份寄存器，重启后可分析
    • 

  • 测试金字塔：大量桌面单元测试 + 模拟器集成测试 + 少量 HIL 硬件测试
    •
