传感器与执行器接口

核心概念

传感器和执行器是嵌入式系统与物理世界交互的桥梁。传感器将温度、湿度、加速度、光照等物理量转换为电信号，执行器则将电信号转换为机械运动、热量、光照等物理输出。理解传感器的信号特性、接口协议和校准方法，以及执行器的驱动方式，是构建可靠嵌入式应用的关键环节。

I2C 传感器读取
// BME280 温湿度气压传感器 I2C 读取
#define BME280_ADDR  0x76

uint8_t bme280_read_reg(uint8_t reg) {
    uint8_t data;
    HAL_I2C_Mem_Read(&hi2c1, BME280_ADDR << 1, reg,
                     I2C_MEMADD_SIZE_8BIT, &data, 1, 100);
    return data;
}

void bme280_read_raw(int32_t *temp_raw, int32_t *hum_raw, int32_t *press_raw) {
    uint8_t buf[8];
    HAL_I2C_Mem_Read(&hi2c1, BME280_ADDR << 1, 0xF7,
                     I2C_MEMADD_SIZE_8BIT, buf, 8, 100);

    *press_raw = ((int32_t)buf[0] << 12) | ((int32_t)buf[1] << 4) | (buf[2] >> 4);
    *temp_raw  = ((int32_t)buf[3] << 12) | ((int32_t)buf[4] << 4) | (buf[5] >> 4);
    *hum_raw   = ((int32_t)buf[6] << 8) | buf[7];
}

// BME280 温度补偿算法（数据手册公式）
int32_t bme280_compensate_temp(int32_t raw, int32_t *t_fine) {
    int32_t var1 = ((((raw >> 3) - ((int32_t)dig_T1 << 1))) * (int32_t)dig_T2) >> 11;
    int32_t var2 = (((((raw >> 4) - (int32_t)dig_T1) *
                     ((raw >> 4) - (int32_t)dig_T1)) >> 12) * (int32_t)dig_T3) >> 14;
    *t_fine = var1 + var2;
    return (*t_fine * 5 + 128) >> 8;  // 返回 0.01°C 精度
}
SPI 传感器读取
// LSM6DSO 六轴 IMU SPI 读取
#define LSM6DSO_READ   0x80

void lsm6dso_read_accel(float *ax, float *ay, float *az) {
    uint8_t tx[7] = {LSM6DSO_READ | 0x28, 0, 0, 0, 0, 0, 0};
    uint8_t rx[7];

    HAL_GPIO_WritePin(CS_GPIO_Port, CS_Pin, GPIO_PIN_RESET);
    HAL_SPI_TransmitReceive(&hspi1, tx, rx, 7, 100);
    HAL_GPIO_WritePin(CS_GPIO_Port, CS_Pin, GPIO_PIN_SET);

    int16_t raw_x = (int16_t)(rx[2] << 8 | rx[1]);
    int16_t raw_y = (int16_t)(rx[4] << 8 | rx[3]);
    int16_t raw_z = (int16_t)(rx[6] << 8 | rx[5]);

    float sensitivity = 0.061f;  // mg/LSB (±2g 量程)
    *ax = raw_x * sensitivity / 1000.0f;
    *ay = raw_y * sensitivity / 1000.0f;
    *az = raw_z * sensitivity / 1000.0f;
}
模拟信号采集
ADC（模数转换器）将模拟电压转换为数字值。关键参数包括分辨率、采样率和参考电压：


// STM32 ADC 配置（12 位分辨率）
HAL_ADC_Start(&hadc1);
HAL_ADC_PollForConversion(&hadc1, 100);
uint32_t raw = HAL_ADC_GetValue(&hadc1);

// 电压计算
float vref = 3.3f;
float voltage = raw * vref / 4095.0f;

// NTC 热敏电阻温度计算
float resistance = 10000.0f * raw / (4095.0f - raw);
float temperature = 1.0f / (1.0f / 298.15f +
    log(resistance / 10000.0f) / 3950.0f) - 273.15f;

// 滤波：移动平均
#define FILTER_SIZE 8
uint32_t filter_buf[FILTER_SIZE];
uint8_t filter_idx = 0;

uint32_t adc_filtered(uint32_t new_val) {
    filter_buf[filter_idx++ % FILTER_SIZE] = new_val;
    uint32_t sum = 0;
    for (int i = 0; i < FILTER_SIZE; i++) sum += filter_buf[i];
    return sum / FILTER_SIZE;
}

// 直流电机 PWM 调速 + H 桥方向控制
void motor_set(int16_t speed) {
    if (speed >= 0) {
        HAL_GPIO_WritePin(DIR_A_GPIO, DIR_A_PIN, GPIO_PIN_SET);
        HAL_GPIO_WritePin(DIR_B_GPIO, DIR_B_PIN, GPIO_PIN_RESET);
    } else {
        HAL_GPIO_WritePin(DIR_A_GPIO, DIR_A_PIN, GPIO_PIN_RESET);
        HAL_GPIO_WritePin(DIR_B_GPIO, DIR_B_PIN, GPIO_PIN_SET);
        speed = -speed;
    }
    __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_1, speed);
}

// 舵机控制（50Hz PWM，脉宽 500-2500μs）
void servo_set_angle(uint16_t angle) {
    uint32_t pulse = 500 + (uint32_t)angle * 2000 / 180;
    __HAL_TIM_SET_COMPARE(&htim2, TIM_CHANNEL_1, pulse);
}

传感器校准
传感器原始读数通常需要校准才能获得准确的物理量值：


// 两点线性校准
typedef struct {
    float offset;
    float scale;
} CalibData;

CalibData calibrate(float raw_low, float ref_low, float raw_high, float ref_high) {
    CalibData cal;
    cal.scale = (ref_high - ref_low) / (raw_high - raw_low);
    cal.offset = ref_low - cal.scale * raw_low;
    return cal;
}

float apply_calibration(CalibData *cal, float raw) {
    return cal->scale * raw + cal->offset;
}

小结
    

  • 传感器将物理量转换为电信号，数字传感器通过 I2C/SPI 接口输出数据，模拟传感器通过 ADC 采样
    • 

  • I2C 适合低速多设备场景（传感器、EEPROM），SPI 适合高速场景（IMU、Flash）
    • 

  • ADC 的有效精度取决于分辨率、参考电压稳定性和信噪比，软件滤波可提升实际精度
    • 

  • 电机驱动使用 H 桥 + PWM 调速，舵机使用脉宽调制（500-2500μs）控制角度
    • 

  • 传感器校准是获得准确物理量的必要步骤，数字传感器通常内置出厂校准参数
    • 

  • 信号调理（放大、滤波、电平转换）是连接传感器和 MCU 的关键中间环节
    •
