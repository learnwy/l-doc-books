传感器数据管线

核心概念

传感器数据管线（Sensor Data Pipeline）是 IoT 系统的数据骨架，负责将传感器原始信号经过采集、校准、过滤、传输、存储和分析的完整流程。一条设计良好的数据管线需要在数据质量、延迟、吞吐量和资源消耗之间取得平衡。从嵌入式固件中的 ADC 采样到云端的时序数据库存储，每个环节都有其特定的技术挑战。

ADC 采样与 DMA
高效多通道采样
使用 DMA（Direct Memory Access）实现 ADC 多通道连续采样，不占用 CPU 时间：


// STM32 HAL ADC + DMA 多通道采样
#define ADC_CHANNELS 4
#define BUFFER_SIZE 256

static uint16_t adc_buffer[ADC_CHANNELS * BUFFER_SIZE];

void adc_dma_init(void) {
    HAL_ADC_Start_DMA(&hadc1, (uint32_t *)adc_buffer,
                      ADC_CHANNELS * BUFFER_SIZE);
}

void HAL_ADC_ConvCpltCallback(ADC_HandleTypeDef *hadc) {
    process_adc_batch(&adc_buffer[ADC_CHANNELS * BUFFER_SIZE / 2],
                      BUFFER_SIZE / 2);
}

void HAL_ADC_ConvHalfCpltCallback(ADC_HandleTypeDef *hadc) {
    process_adc_batch(&adc_buffer[0], BUFFER_SIZE / 2);
}

数字滤波算法
常用滤波器
// 指数移动平均滤波（EMA）
typedef struct {
    float alpha;
    float output;
    bool initialized;
} ema_filter_t;

float ema_update(ema_filter_t *f, float input) {
    if (!f->initialized) {
        f->output = input;
        f->initialized = true;
    } else {
        f->output = f->alpha * input + (1.0f - f->alpha) * f->output;
    }
    return f->output;
}

// 中值滤波（去除脉冲噪声）
#define MEDIAN_WINDOW 5
float median_filter(float *buffer, int size) {
    float sorted[MEDIAN_WINDOW];
    memcpy(sorted, buffer, size * sizeof(float));
    for (int i = 0; i < size - 1; i++) {
        for (int j = i + 1; j < size; j++) {
            if (sorted[j] < sorted[i]) {
                float tmp = sorted[i];
                sorted[i] = sorted[j];
                sorted[j] = tmp;
            }
        }
    }
    return sorted[size / 2];
}

// 简化卡尔曼滤波（一维）
typedef struct {
    float q;  // 过程噪声
    float r;  // 测量噪声
    float x;  // 估计值
    float p;  // 估计协方差
} kalman_1d_t;

float kalman_update(kalman_1d_t *kf, float measurement) {
    kf->p += kf->q;
    float k = kf->p / (kf->p + kf->r);  // 卡尔曼增益
    kf->x += k * (measurement - kf->x);
    kf->p *= (1.0f - k);
    return kf->x;
}

数据编码与传输优化
紧凑编码
IoT 设备的带宽和电量极其有限，高效的数据编码直接影响电池寿命和网络容量：


// CBOR 编码（比 JSON 紧凑 30-50%）
#include "tinycbor/cbor.h"

int encode_sensor_data(uint8_t *buffer, size_t buf_size,
                       uint16_t device_id, float temp, float humidity) {
    CborEncoder encoder, map;
    cbor_encoder_init(&encoder, buffer, buf_size, 0);
    cbor_encoder_create_map(&encoder, &map, 3);

    cbor_encode_text_stringz(&map, "id");
    cbor_encode_uint(&map, device_id);

    cbor_encode_text_stringz(&map, "t");
    cbor_encode_float(&map, temp);

    cbor_encode_text_stringz(&map, "h");
    cbor_encode_float(&map, humidity);

    cbor_encoder_close_container(&encoder, &map);
    return cbor_encoder_get_buffer_size(&encoder, buffer);
}

小结
    

  • 传感器数据管线涵盖从 ADC 采样到云端存储的完整链路，每个环节都影响数据质量和系统性能
    • 

  • DMA 是高效多通道 ADC 采样的关键技术，释放 CPU 用于数据处理
    • 

  • 数字滤波在嵌入式端完成，卡尔曼滤波和 EMA 是实时场景的首选
    • 

  • CBOR 和 Protobuf 比 JSON 节省 40-60% 带宽，对电池供电设备至关重要
    • 

  • 时序数据库针对 IoT 写入模式优化，TDengine 和 InfluxDB 是 IoT 场景的主流选择
    • 

  • 管线设计需要在数据质量、延迟、带宽和功耗之间找到适合业务需求的平衡点
    •
