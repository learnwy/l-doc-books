内存与寄存器管理

核心概念

嵌入式系统的内存资源极度受限——典型 MCU 只有几十 KB 的 SRAM 和几百 KB 的 Flash。在这种约束下，理解内存布局、链接脚本、栈与堆的管理方式，以及寄存器级编程技巧，是写出可靠嵌入式代码的基础。与通用计算不同，嵌入式开发中的内存错误（栈溢出、内存碎片、未初始化访问）往往导致系统崩溃或不可预测的行为，且极难调试。

链接脚本与内存映射
链接脚本（Linker Script）定义了代码和数据在 MCU 内存中的布局。理解链接脚本是控制内存分配的关键：


/* STM32F411 链接脚本片段 */
MEMORY
{
    FLASH (rx)  : ORIGIN = 0x08000000, LENGTH = 512K
    SRAM  (rwx) : ORIGIN = 0x20000000, LENGTH = 128K
}

SECTIONS
{
    .isr_vector : {
        KEEP(*(.isr_vector))
    } > FLASH

    .text : {
        *(.text*)
        *(.rodata*)
    } > FLASH

    .data : {
        _sdata = .;
        *(.data*)
        _edata = .;
    } > SRAM AT > FLASH

    .bss : {
        _sbss = .;
        *(.bss*)
        *(COMMON)
        _ebss = .;
    } > SRAM

    /* 栈放在 SRAM 顶部 */
    _estack = ORIGIN(SRAM) + LENGTH(SRAM);
}
启动代码负责在 main() 之前完成内存初始化：


void Reset_Handler(void) {
    /* 从 Flash 复制 .data 到 SRAM */
    uint32_t *src = &_sidata;
    uint32_t *dst = &_sdata;
    while (dst < &_edata) {
        *dst++ = *src++;
    }

    /* 清零 .bss */
    dst = &_sbss;
    while (dst < &_ebss) {
        *dst++ = 0;
    }

    /* 调用 C++ 静态构造函数（如果有）*/
    __libc_init_array();

    main();
}
栈管理与溢出检测
栈溢出是嵌入式系统最危险的内存问题之一——栈向下增长覆盖 .bss 或 .data 区域，导致不可预测的行为：


/* 静态栈大小分析：计算最深调用链 */
// main() → processData() → parsePacket() → checkCRC()
//  每个函数栈帧：局部变量 + 保存的寄存器 + 对齐
//  加上最深中断嵌套的栈消耗

/* 运行时栈检测方法1：填充模式 */
#define STACK_CANARY 0xDEADBEEF
void stack_init(void) {
    extern uint32_t _sstack;
    extern uint32_t _estack;
    uint32_t *p = &_sstack;
    while (p < &_estack - 256) {
        *p++ = STACK_CANARY;
    }
}

uint32_t stack_usage(void) {
    extern uint32_t _sstack;
    extern uint32_t _estack;
    uint32_t *p = &_sstack;
    while (*p == STACK_CANARY) p++;
    return (uint32_t)(&_estack) - (uint32_t)p;
}

/* 运行时栈检测方法2：MPU 保护 */
// 在栈底部设置一个不可访问的 MPU 区域
// 栈溢出时触发 MemManage Fault

嵌入式动态内存管理
嵌入式系统中应尽量避免动态内存分配，但某些场景（如网络协议栈、RTOS 任务创建）无法完全避免。嵌入式内存分配器需要解决碎片化问题：


/* 方案1：静态内存池（推荐） */
typedef struct {
    uint8_t data[BLOCK_SIZE];
    uint8_t used;
} MemBlock;

static MemBlock pool[POOL_SIZE];

void *pool_alloc(void) {
    for (int i = 0; i < POOL_SIZE; i++) {
        if (!pool[i].used) {
            pool[i].used = 1;
            return pool[i].data;
        }
    }
    return NULL;
}

void pool_free(void *ptr) {
    MemBlock *blk = (MemBlock *)((uint8_t *)ptr - offsetof(MemBlock, data));
    blk->used = 0;
}

/* 方案2：TLSF（Two-Level Segregated Fit）分配器 */
// O(1) 复杂度的分配和释放
// 适合实时系统，碎片化可控
// 常用于 RTOS 的堆管理

DMA 控制器
DMA（Direct Memory Access）允许外设在不经过 CPU 的情况下直接与内存交换数据，释放 CPU 处理能力：


/* DMA 传输：ADC → SRAM（循环模式）*/
uint16_t adc_buffer[256];

DMA_HandleTypeDef hdma_adc;
hdma_adc.Instance = DMA2_Stream0;
hdma_adc.Init.Channel = DMA_CHANNEL_0;
hdma_adc.Init.Direction = DMA_PERIPH_TO_MEMORY;
hdma_adc.Init.PeriphInc = DMA_PINC_DISABLE;
hdma_adc.Init.MemInc = DMA_MINC_ENABLE;
hdma_adc.Init.PeriphDataAlignment = DMA_PDATAALIGN_HALFWORD;
hdma_adc.Init.MemDataAlignment = DMA_MDATAALIGN_HALFWORD;
hdma_adc.Init.Mode = DMA_CIRCULAR;
hdma_adc.Init.Priority = DMA_PRIORITY_HIGH;
HAL_DMA_Init(&hdma_adc);

HAL_ADC_Start_DMA(&hadc1, (uint32_t *)adc_buffer, 256);

/* 双缓冲 DMA：前台处理 buffer A，后台 DMA 填充 buffer B */
void HAL_ADC_ConvHalfCpltCallback(ADC_HandleTypeDef *hadc) {
    process_data(adc_buffer, 128);
}
void HAL_ADC_ConvCpltCallback(ADC_HandleTypeDef *hadc) {
    process_data(adc_buffer + 128, 128);
}

小结
    

  • 嵌入式内存布局分为 Flash（.text/.rodata）和 SRAM（.data/.bss/Stack/Heap），由链接脚本精确控制
    • 

  • 启动代码负责从 Flash 复制 .data 到 SRAM 并清零 .bss，在 main() 之前完成
    • 

  • 栈溢出是嵌入式最危险的问题，通过静态分析、填充模式检测和 MPU 硬件保护三级防御
    • 

  • 嵌入式动态内存优先使用静态内存池，需要通用分配时选择 TLSF 等实时友好的分配器
    • 

  • DMA 控制器实现外设与内存间的零 CPU 数据传输，双缓冲模式支持实时流处理
    • 

  • 内存资源极度受限是嵌入式开发的核心约束，一切设计决策都围绕这一约束展开
    •
