组合数学

核心概念

组合数学是研究离散对象的计数、排列和组合的数学分支。在算法中，组合数学为"有多少种方案"这类计数问题提供了精确的数学工具。从基本的排列组合公式到帕斯卡三角形的递推计算，再到卡特兰数等特殊数列，组合数学的知识体系贯穿了动态规划、概率算法和竞赛编程的方方面面。

排列与组合的计算
直接用阶乘公式计算组合数在 nnn 较大时会溢出。实际编程中有两种常用方法：递推法和乘除法。


帕斯卡三角形（杨辉三角）利用递推关系 C(n,k)=C(n−1,k−1)+C(n−1,k)C(n,k) = C(n-1,k-1) + C(n-1,k)C(n,k)=C(n−1,k−1)+C(n−1,k) 来计算：


function buildPascalTriangle(n: number): number[][] {
  const triangle: number[][] = Array.from({ length: n + 1 }, () => []);

  for (let i = 0; i <= n; i++) {
    triangle[i][0] = 1;
    triangle[i][i] = 1;
    for (let j = 1; j < i; j++) {
      triangle[i][j] = triangle[i - 1][j - 1] + triangle[i - 1][j];
    }
  }

  return triangle;
}

function combination(n: number, k: number, triangle: number[][]): number {
  return triangle[n][k];
}
乘除法避免阶乘溢出，利用 C(n,k)=n×(n−1)×⋯×(n−k+1)k!C(n,k) = \frac{n \times (n-1) \times \cdots \times (n-k+1)}{k!}C(n,k)=k!n×(n−1)×⋯×(n−k+1)​：


function comb(n: number, k: number): number {
  if (k > n - k) k = n - k;
  let result = 1;
  for (let i = 0; i < k; i++) {
    result = (result * (n - i)) / (i + 1);
  }
  return result;
}
取模场景下的组合数计算（需要模逆元）：


function combMod(n: number, k: number, mod: number): number {
  if (k > n) return 0;
  const fact = new Array(n + 1);
  fact[0] = 1;
  for (let i = 1; i <= n; i++) {
    fact[i] = (fact[i - 1] * i) % mod;
  }

  const invFact = new Array(n + 1);
  invFact[n] = modPow(fact[n], mod - 2, mod);
  for (let i = n - 1; i >= 0; i--) {
    invFact[i] = (invFact[i + 1] * (i + 1)) % mod;
  }

  return (((fact[n] * invFact[k]) % mod) * invFact[n - k]) % mod;
}

function modPow(base: number, exp: number, mod: number): number {
  let result = 1;
  base %= mod;
  while (exp > 0) {
    if (exp % 2 === 1) result = (result * base) % mod;
    exp = Math.floor(exp / 2);
    base = (base * base) % mod;
  }
  return result;
}
排列生成算法
生成全排列是回溯法的经典应用，但也可以用迭代法实现"下一个排列"：


function nextPermutation(nums: number[]): boolean {
  const n = nums.length;
  let i = n - 2;

  while (i >= 0 && nums[i] >= nums[i + 1]) i--;

  if (i < 0) return false;

  let j = n - 1;
  while (nums[j] <= nums[i]) j--;

  [nums[i], nums[j]] = [nums[j], nums[i]];

  let left = i + 1;
  let right = n - 1;
  while (left < right) {
    [nums[left], nums[right]] = [nums[right], nums[left]];
    left++;
    right--;
  }

  return true;
}
第 kkk 个排列（康托展开）：


function getPermutation(n: number, k: number): string {
  const factorial = [1];
  for (let i = 1; i <= n; i++) {
    factorial[i] = factorial[i - 1] * i;
  }

  k--;
  const available = Array.from({ length: n }, (_, i) => i + 1);
  let result = '';

  for (let i = n; i >= 1; i--) {
    const idx = Math.floor(k / factorial[i - 1]);
    result += available[idx];
    available.splice(idx, 1);
    k %= factorial[i - 1];
  }

  return result;
}

function catalanNumbers(n: number): number[] {
  const catalan = new Array(n + 1).fill(0);
  catalan[0] = 1;
  catalan[1] = 1;

  for (let i = 2; i <= n; i++) {
    for (let j = 0; j < i; j++) {
      catalan[i] += catalan[j] * catalan[i - 1 - j];
    }
  }

  return catalan;
}

容斥原理
容斥原理是组合计数中处理"至少满足一个条件"问题的核心工具。对于 nnn 个集合：


∣A1∪A2∪⋯∪An∣=∑∣Ai∣−∑∣Ai∩Aj∣+∑∣Ai∩Aj∩Ak∣−⋯|A_1 \cup A_2 \cup \cdots \cup A_n| = \sum|A_i| - \sum|A_i \cap A_j| + \sum|A_i \cap A_j \cap A_k| - \cdots∣A1​∪A2​∪⋯∪An​∣=∑∣Ai​∣−∑∣Ai​∩Aj​∣+∑∣Ai​∩Aj​∩Ak​∣−⋯
经典应用——求 [1,n][1, n][1,n] 中与 nnn 互质的数的个数（欧拉函数）：


function eulerTotient(n: number): number {
  let result = n;
  let temp = n;

  for (let i = 2; i * i <= temp; i++) {
    if (temp % i === 0) {
      while (temp % i === 0) temp /= i;
      result = (result / i) * (i - 1);
    }
  }

  if (temp > 1) {
    result = (result / temp) * (temp - 1);
  }

  return result;
}
容斥原理求不被任何给定质数整除的数的个数：


function countCoprime(n: number, primes: number[]): number {
  let count = 0;
  const m = primes.length;

  for (let mask = 1; mask < 1 << m; mask++) {
    let product = 1;
    let bits = 0;

    for (let i = 0; i < m; i++) {
      if ((mask >> i) & 1) {
        product *= primes[i];
        bits++;
      }
    }

    if (bits % 2 === 1) {
      count += Math.floor(n / product);
    } else {
      count -= Math.floor(n / product);
    }
  }

  return n - count;
}
