矩阵快速幂
矩阵快速幂是加速线性递推的通用工具。其核心思想：将递推关系表示为矩阵乘法，然后利用快速幂在 O(k3log⁡n)O(k^3 \log n)O(k3logn) 时间内（kkk 为矩阵维度）计算第 nnn 项。


以斐波那契数列为例：F(n)=F(n−1)+F(n−2)F(n) = F(n-1) + F(n-2)F(n)=F(n−1)+F(n−2) 可以表示为：


[F(n)F(n−1)]=[1110][F(n−1)F(n−2)]\begin{bmatrix} F(n) \\ F(n-1) \end{bmatrix} = \begin{bmatrix} 1 & 1 \\ 1 & 0 \end{bmatrix} \begin{bmatrix} F(n-1) \\ F(n-2) \end{bmatrix}[F(n)F(n−1)​]=[11​10​][F(n−1)F(n−2)​]
因此 [F(n)F(n−1)]=[1110]n−1[F(1)F(0)]\begin{bmatrix} F(n) \\ F(n-1) \end{bmatrix} = \begin{bmatrix} 1 & 1 \\ 1 & 0 \end{bmatrix}^{n-1} \begin{bmatrix} F(1) \\ F(0) \end{bmatrix}[F(n)F(n−1)​]=[11​10​]n−1[F(1)F(0)​]


type Matrix = number[][];

function matMul(a: Matrix, b: Matrix, mod: number): Matrix {
  const n = a.length;
  const m = b[0].length;
  const k = b.length;
  const result: Matrix = Array.from({ length: n }, () => new Array(m).fill(0));

  for (let i = 0; i < n; i++) {
    for (let j = 0; j < m; j++) {
      for (let p = 0; p < k; p++) {
        result[i][j] = (result[i][j] + a[i][p] * b[p][j]) % mod;
      }
    }
  }

  return result;
}

function matPow(mat: Matrix, exp: number, mod: number): Matrix {
  const n = mat.length;
  let result: Matrix = Array.from({ length: n }, (_, i) => Array.from({ length: n }, (_, j) => (i === j ? 1 : 0)));

  while (exp > 0) {
    if (exp % 2 === 1) {
      result = matMul(result, mat, mod);
    }
    exp = Math.floor(exp / 2);
    mat = matMul(mat, mat, mod);
  }

  return result;
}

function fibonacciN(n: number, mod: number): number {
  if (n <= 1) return n;
  const base: Matrix = [
    [1, 1],
    [1, 0],
  ];
  const result = matPow(base, n - 1, mod);
  return result[0][0];
}

function tribonacci(n: number, mod: number): number {
  if (n === 0) return 0;
  if (n <= 2) return 1;

  const base: Matrix = [
    [1, 1, 1],
    [1, 0, 0],
    [0, 1, 0],
  ];

  const result = matPow(base, n - 2, mod);
  return (result[0][0] + result[0][1]) % mod;
}
数位动态规划
数位 DP 是解决"在 [L,R][L, R][L,R] 范围内满足某条件的整数有多少个"这类问题的标准方法。核心思路是逐位构造数字，利用"是否还受上界限制"（tight constraint）来递推。


统计 [1,n][1, n][1,n] 中数字 1 出现的次数：


function countDigitOne(n: number): number {
  const digits = String(n).split('').map(Number);
  const len = digits.length;
  const memo = new Map<string, number>();

  function dp(pos: number, count: number, tight: boolean, started: boolean): number {
    if (pos === len) return count;

    const key = `${pos},${count},${tight},${started}`;
    if (memo.has(key)) return memo.get(key)!;

    const limit = tight ? digits[pos] : 9;
    let result = 0;

    for (let d = 0; d <= limit; d++) {
      result += dp(pos + 1, count + (d === 1 ? 1 : 0), tight && d === limit, started || d > 0);
    }

    memo.set(key, result);
    return result;
  }

  return dp(0, 0, true, false);
}

约瑟夫问题
约瑟夫问题：nnn 个人围成一圈，从第 1 个人开始报数，每报到 kkk 就淘汰该人，求最后幸存者的编号。


递推公式：J(n,k)=(J(n−1,k)+k) mod nJ(n, k) = (J(n-1, k) + k) \bmod nJ(n,k)=(J(n−1,k)+k)modn，其中 J(1,k)=0J(1, k) = 0J(1,k)=0。


function josephus(n: number, k: number): number {
  let survivor = 0;
  for (let i = 2; i <= n; i++) {
    survivor = (survivor + k) % i;
  }
  return survivor + 1;
}

function josephusK2(n: number): number {
  const highestBit = 1 << (31 - Math.clz32(n));
  return 2 * (n - highestBit) + 1;
}

曼哈顿距离与切比雪夫距离的转换：


曼哈顿距离 ∣x1−x2∣+∣y1−y2∣|x_1-x_2| + |y_1-y_2|∣x1​−x2​∣+∣y1​−y2​∣ 可以通过坐标变换 (x,y)→(x+y,x−y)(x, y) \rightarrow (x+y, x-y)(x,y)→(x+y,x−y) 转化为切比雪夫距离 max⁡(∣x1′−x2′∣,∣y1′−y2′∣)\max(|x_1'-x_2'|, |y_1'-y_2'|)max(∣x1′​−x2′​∣,∣y1′​−y2′​∣)。这个转换在需要求"最远曼哈顿距离"等问题时非常有用。


function maxManhattanDistance(points: [number, number][]): number {
  let maxSum = -Infinity,
    minSum = Infinity;
  let maxDiff = -Infinity,
    minDiff = Infinity;

  for (const [x, y] of points) {
    maxSum = Math.max(maxSum, x + y);
    minSum = Math.min(minSum, x + y);
    maxDiff = Math.max(maxDiff, x - y);
    minDiff = Math.min(minDiff, x - y);
  }

  return Math.max(maxSum - minSum, maxDiff - minDiff);
}
求众数（Boyer-Moore 投票算法）：


function majorityElement(nums: number[]): number {
  let candidate = nums[0];
  let count = 0;

  for (const num of nums) {
    if (count === 0) candidate = num;
    count += num === candidate ? 1 : -1;
  }

  return candidate;
}
