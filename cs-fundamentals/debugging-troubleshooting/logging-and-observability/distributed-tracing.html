分布式追踪

概述

分布式追踪（Distributed Tracing）是在微服务架构中追踪一个请求从进入系统到最终响应的完整调用路径的技术。当一个用户请求可能经过 10+ 个服务、数十次 RPC 调用时，单个服务的日志无法还原全貌。分布式追踪通过 Trace ID 将所有相关操作串联为一棵调用树，让跨服务问题的排查从"大海捞针"变为"按图索骥"。

OpenTelemetry 实现
// Go: OpenTelemetry 手动创建 Span
package main

import (
    "context"
    "go.opentelemetry.io/otel"
    "go.opentelemetry.io/otel/attribute"
    "go.opentelemetry.io/otel/trace"
)

var tracer = otel.Tracer("order-service")

func ProcessOrder(ctx context.Context, orderID string) error {
    ctx, span := tracer.Start(ctx, "ProcessOrder",
        trace.WithAttributes(
            attribute.String("order.id", orderID),
            attribute.String("order.type", "standard"),
        ),
    )
    defer span.End()

    // 子操作自动成为子 Span
    if err := validateOrder(ctx, orderID); err != nil {
        span.RecordError(err)
        span.SetStatus(codes.Error, err.Error())
        return err
    }

    if err := chargePayment(ctx, orderID); err != nil {
        span.RecordError(err)
        return err
    }

    span.AddEvent("order_completed", trace.WithAttributes(
        attribute.Int("items_count", 5),
    ))
    return nil
}
# Python: OpenTelemetry 自动和手动混合追踪
from opentelemetry import trace
from opentelemetry.instrumentation.requests import RequestsInstrumentor
from opentelemetry.instrumentation.sqlalchemy import SQLAlchemyInstrumentor

# 自动追踪 HTTP 客户端和数据库查询
RequestsInstrumentor().instrument()
SQLAlchemyInstrumentor().instrument()

# 手动追踪业务逻辑
tracer = trace.get_tracer("payment-service")

def process_payment(order_id: str, amount: float):
    with tracer.start_as_current_span("process_payment") as span:
        span.set_attribute("payment.order_id", order_id)
        span.set_attribute("payment.amount", amount)

        # 风控检查（自动成为子 Span）
        risk_score = check_risk(order_id, amount)
        span.set_attribute("payment.risk_score", risk_score)

        if risk_score > 0.8:
            span.add_event("payment_blocked", {
                "reason": "high_risk_score"
            })
            raise PaymentBlockedError(f"Risk score too high: {risk_score}")

        # 调用银行 API（requests 自动被追踪）
        result = requests.post(BANK_API, json={"amount": amount})
        span.set_attribute("payment.bank_status", result.status_code)
        return result.json()
采样策略
全量追踪在高流量系统中成本过高，需要智能采样：


# OpenTelemetry Collector 配置示例
processors:
  # 尾采样：基于完整 trace 特征决定是否保留
  tail_sampling:
    policies:
      # 所有错误 trace 全量保留
      - name: errors
        type: status_code
        status_code: { status_codes: [ERROR] }

      # 慢请求全量保留
      - name: slow-requests
        type: latency
        latency: { threshold_ms: 1000 }

      # 正常请求按 10% 采样
      - name: normal-traffic
        type: probabilistic
        probabilistic: { sampling_percentage: 10 }

追踪数据的分析价值
除了单次问题排查，追踪数据还可以用于：


-- 从追踪数据提取服务依赖图
SELECT
    parent_service,
    child_service,
    COUNT(*) as call_count,
    AVG(duration_ms) as avg_latency,
    PERCENTILE_CONT(0.99) WITHIN GROUP (ORDER BY duration_ms) as p99
FROM spans
WHERE timestamp > NOW() - INTERVAL '1 hour'
GROUP BY parent_service, child_service
ORDER BY avg_latency DESC;

-- 识别性能退化
SELECT
    operation_name,
    DATE_TRUNC('hour', timestamp) as hour,
    PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY duration_ms) as p95
FROM spans
WHERE service = 'order-service'
GROUP BY operation_name, hour
ORDER BY hour DESC;

总结
分布式追踪是微服务架构中排查跨服务问题的核心技术。采用 OpenTelemetry 作为标准、实施分阶段引入策略、配合合理的采样和存储方案，可以在可控成本下获得对系统行为的深度可见性。关键原则：先自动后手动、错误全量采样、与日志和指标关联使用。
