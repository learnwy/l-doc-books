结构化日志设计

概述

结构化日志是将日志从"给人读的文本行"升级为"给程序解析的数据记录"。在传统的非结构化日志中，每条消息是一段自由格式文本；而结构化日志使用固定的键值对格式（通常是 JSON），让日志分析工具能够高效地索引、过滤和聚合。对于任何需要在生产环境中排查问题的系统来说，结构化日志是可观测性的基石。

核心概念

非结构化 vs 结构化日志

# 非结构化日志 — 人可以读，但机器难以解析
"2024-03-15 10:23:45 ERROR [OrderService] Failed to process order #12345 for user john@example.com: payment timeout after 30s"

# 结构化日志 — 人和机器都能处理
{
  "timestamp": "2024-03-15T10:23:45.123Z",
  "level": "error",
  "service": "order-service",
  "module": "OrderService",
  "event": "order_processing_failed",
  "order_id": "12345",
  "user_email": "j***@example.com",
  "error_type": "payment_timeout",
  "timeout_ms": 30000,
  "trace_id": "abc123def456",
  "span_id": "789ghi"
}

上下文传播
在分布式系统中，日志的价值很大程度取决于能否将相关日志串联起来：


// Go 示例：使用 context 传播日志上下文
package logging

import (
    "context"
    "log/slog"
)

type contextKey string

const (
    traceIDKey   contextKey = "trace_id"
    userIDKey    contextKey = "user_id"
    requestIDKey contextKey = "request_id"
)

func WithTraceID(ctx context.Context, traceID string) context.Context {
    return context.WithValue(ctx, traceIDKey, traceID)
}

func Logger(ctx context.Context) *slog.Logger {
    attrs := []slog.Attr{}
    if traceID, ok := ctx.Value(traceIDKey).(string); ok {
        attrs = append(attrs, slog.String("trace_id", traceID))
    }
    if userID, ok := ctx.Value(userIDKey).(string); ok {
        attrs = append(attrs, slog.String("user_id", userID))
    }
    return slog.Default().With(attrsToArgs(attrs)...)
}
# Python 示例：使用 structlog 自动注入上下文
import structlog
import uuid

structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.processors.JSONRenderer()
    ]
)

def create_request_logger(request):
    """为每个请求创建带上下文的 logger"""
    return structlog.get_logger().bind(
        trace_id=request.headers.get("X-Trace-ID", str(uuid.uuid4())),
        user_id=request.user.id if request.user else None,
        path=request.path,
        method=request.method,
    )

# 使用
log = create_request_logger(request)
log.info("order_created", order_id="12345", amount=99.99)
# 输出: {"timestamp":"2024-03-15T10:23:45Z","level":"info","trace_id":"abc-123","user_id":"u-456","path":"/api/orders","method":"POST","event":"order_created","order_id":"12345","amount":99.99}
敏感信息处理
# 日志脱敏策略
import re
from functools import wraps

SENSITIVE_PATTERNS = {
    "email": (r'[\w.-]+@[\w.-]+\.\w+', lambda m: m.group()[:3] + '***@***'),
    "phone": (r'\d{11}', lambda m: m.group()[:3] + '****' + m.group()[-4:]),
    "card": (r'\d{16}', lambda m: '****' + m.group()[-4:]),
    "token": (r'(Bearer\s+)\S+', lambda m: m.group(1) + '***REDACTED***'),
}

def sanitize_log_value(key, value):
    """对敏感字段进行脱敏"""
    sensitive_keys = {"password", "secret", "token", "credit_card", "ssn"}
    if key.lower() in sensitive_keys:
        return "***REDACTED***"
    if isinstance(value, str):
        for pattern_name, (pattern, replacer) in SENSITIVE_PATTERNS.items():
            value = re.sub(pattern, replacer, value)
    return value

日志采样策略
在高 QPS 系统中，全量日志既昂贵又无必要：


// 基于速率的采样
type RateSampler struct {
    rate     float64  // 0.0 - 1.0
    counter  atomic.Int64
    interval int64    // 每 N 条采样 1 条
}

func (s *RateSampler) ShouldLog(level slog.Level) bool {
    // ERROR 和 FATAL 永不采样
    if level >= slog.LevelError {
        return true
    }
    // INFO 按比率采样
    if level == slog.LevelInfo {
        count := s.counter.Add(1)
        return count%s.interval == 0
    }
    // DEBUG 默认关闭，除非动态开启
    return false
}

// 基于错误的自适应采样：出错时提高采样率
type AdaptiveSampler struct {
    normalRate  float64
    errorRate   float64
    errorWindow time.Duration
    lastError   atomic.Int64
}

func (s *AdaptiveSampler) ShouldLog() bool {
    lastErr := time.Unix(0, s.lastError.Load())
    if time.Since(lastErr) < s.errorWindow {
        return rand.Float64() < s.errorRate  // 错误后提高采样率
    }
    return rand.Float64() < s.normalRate
}
动态日志级别调整
// Spring Boot: 运行时动态调整日志级别（无需重启）
// POST /actuator/loggers/com.example.order
// {"configuredLevel": "DEBUG"}

// 自定义实现：基于 header 的请求级别调试
@Component
public class DebugHeaderFilter implements Filter {
    @Override
    public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain) {
        HttpServletRequest request = (HttpServletRequest) req;
        String debugLevel = request.getHeader("X-Debug-Level");
        if ("verbose".equals(debugLevel)) {
            MDC.put("dynamic_level", "DEBUG");
        }
        try {
            chain.doFilter(req, res);
        } finally {
            MDC.remove("dynamic_level");
        }
    }
}

总结
结构化日志是所有可观测性实践的基础。关键设计原则：使用 JSON 格式、建立统一 Schema、传播上下文标识（trace_id）、脱敏敏感信息、实施合理的采样策略。好的日志设计应该让你在收到告警的 30 秒内就能通过 trace_id 找到完整的请求路径和错误上下文。
