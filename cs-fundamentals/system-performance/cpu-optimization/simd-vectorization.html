SIMD 向量化

核心概念

SIMD（Single Instruction, Multiple Data，单指令多数据）是一种数据并行计算模型，用一条指令同时处理多个数据元素。例如，一条 256 位的 AVX2 加法指令可以同时完成 8 个 32 位整数的加法，理论上比标量代码快 8 倍。SIMD 是 CPU 级数据并行的核心技术，广泛应用于多媒体处理、科学计算、机器学习推理和字符串处理等场景。

自动向量化
现代编译器（GCC、Clang、MSVC）具备自动向量化（Auto-Vectorization）能力，可以自动将标量循环转换为 SIMD 指令。但自动向量化有严格的前提条件——编译器必须能够证明循环迭代之间没有数据依赖。


// 编译器可以自动向量化的循环
void add_arrays(float* a, float* b, float* c, int n) {
    for (int i = 0; i < n; i++) {
        c[i] = a[i] + b[i];  // 每次迭代独立，无数据依赖
    }
}
// gcc -O2 -march=native 会自动生成 SIMD 指令

// 编译器无法自动向量化的循环
void prefix_sum(float* data, int n) {
    for (int i = 1; i < n; i++) {
        data[i] += data[i - 1];  // 循环携带依赖: data[i] 依赖 data[i-1]
    }
}
检查编译器是否成功向量化的方法：


# GCC: 输出向量化报告
gcc -O2 -march=native -ftree-vectorize -fopt-info-vec-optimized -c mycode.c

# Clang: 输出向量化失败的原因
clang -O2 -march=native -Rpass=loop-vectorize -Rpass-missed=loop-vectorize -c mycode.c
帮助编译器自动向量化的技巧：


// 使用 restrict 关键字告诉编译器指针不会别名
void add_arrays_safe(float* restrict a, float* restrict b, float* restrict c, int n) {
    for (int i = 0; i < n; i++) {
        c[i] = a[i] + b[i];
    }
}

// 对齐内存分配，匹配 SIMD 寄存器宽度
#include <stdlib.h>
float* aligned_alloc_floats(int n) {
    float* ptr;
    posix_memalign((void**)&ptr, 32, n * sizeof(float));  // 32 字节对齐 (AVX)
    return ptr;
}

// 使用 OpenMP SIMD 指导
#pragma omp simd
for (int i = 0; i < n; i++) {
    c[i] = a[i] * b[i] + d[i];
}
手动向量化 (Intrinsics)
当自动向量化无法满足需求时（编译器未能向量化，或需要使用特定的 SIMD 指令），可以使用Intrinsics（编译器内建函数）进行手动向量化。Intrinsics 是 SIMD 汇编指令的 C/C++ 封装，提供了接近汇编的控制力和 C 语言的可读性。


#include <immintrin.h>

void add_arrays_avx2(float* a, float* b, float* c, int n) {
    int i = 0;
    for (; i + 8 <= n; i += 8) {
        __m256 va = _mm256_loadu_ps(&a[i]);   // 加载 8 个 float
        __m256 vb = _mm256_loadu_ps(&b[i]);
        __m256 vc = _mm256_add_ps(va, vb);    // 8 路并行加法
        _mm256_storeu_ps(&c[i], vc);          // 存储 8 个 float
    }
    for (; i < n; i++) {                      // 处理尾部不足 8 个的元素
        c[i] = a[i] + b[i];
    }
}

float dot_product_avx2(float* a, float* b, int n) {
    __m256 sum = _mm256_setzero_ps();
    int i = 0;
    for (; i + 8 <= n; i += 8) {
        __m256 va = _mm256_loadu_ps(&a[i]);
        __m256 vb = _mm256_loadu_ps(&b[i]);
        sum = _mm256_fmadd_ps(va, vb, sum);   // FMA: sum += a * b
    }
    float result[8];
    _mm256_storeu_ps(result, sum);
    float total = 0;
    for (int j = 0; j < 8; j++) total += result[j];
    for (; i < n; i++) total += a[i] * b[i];
    return total;
}
实际应用场景
SIMD 不仅用于数值计算，在字符串处理、数据库查询和网络包处理等领域也有广泛应用。


字符串搜索：使用 SIMD 一次比较 16/32 个字节，比逐字符比较快一个数量级：


#include <immintrin.h>
#include <string.h>

int find_char_simd(const char* str, char target, int len) {
    __m256i vtarget = _mm256_set1_epi8(target);
    int i = 0;
    for (; i + 32 <= len; i += 32) {
        __m256i vdata = _mm256_loadu_si256((__m256i*)&str[i]);
        __m256i vcmp = _mm256_cmpeq_epi8(vdata, vtarget);
        int mask = _mm256_movemask_epi8(vcmp);
        if (mask != 0) {
            return i + __builtin_ctz(mask);
        }
    }
    for (; i < len; i++) {
        if (str[i] == target) return i;
    }
    return -1;
}
数据库列式过滤：SIMD 在列式数据库（如 ClickHouse、DuckDB）中用于高速过滤和聚合：


int count_greater_than_simd(int32_t* column, int n, int32_t threshold) {
    __m256i vthreshold = _mm256_set1_epi32(threshold);
    __m256i vcount = _mm256_setzero_si256();
    __m256i vone = _mm256_set1_epi32(1);

    for (int i = 0; i + 8 <= n; i += 8) {
        __m256i vdata = _mm256_loadu_si256((__m256i*)&column[i]);
        __m256i vcmp = _mm256_cmpgt_epi32(vdata, vthreshold);
        vcount = _mm256_sub_epi32(vcount, vcmp); // cmpgt 结果是 -1，减法等于加 1
    }

    int32_t counts[8];
    _mm256_storeu_si256((__m256i*)counts, vcount);
    int total = 0;
    for (int j = 0; j < 8; j++) total += counts[j];
    return total;
}

跨平台向量化策略
不同 CPU 架构支持不同的 SIMD 指令集（x86: SSE/AVX, ARM: NEON/SVE），直接使用 Intrinsics 意味着代码不可移植。解决方案是使用跨平台抽象层或高级语言的向量化支持。


// Rust 的 portable SIMD (std::simd, nightly)
#![feature(portable_simd)]
use std::simd::f32x8;

fn add_arrays_rust(a: &[f32], b: &[f32], c: &mut [f32]) {
    let chunks = a.len() / 8;
    for i in 0..chunks {
        let va = f32x8::from_slice(&a[i*8..]);
        let vb = f32x8::from_slice(&b[i*8..]);
        let vc = va + vb;
        vc.copy_to_slice(&mut c[i*8..]);
    }
}
在 C++ 中，Highway 和 xsimd 是两个优秀的跨平台 SIMD 库：


// 使用 Google Highway 库
#include "hwy/highway.h"
namespace hn = hwy::HWY_NAMESPACE;

void AddArrays(const float* a, const float* b, float* c, size_t n) {
    const hn::ScalableTag<float> d;
    for (size_t i = 0; i < n; i += hn::Lanes(d)) {
        auto va = hn::LoadU(d, a + i);
        auto vb = hn::LoadU(d, b + i);
        hn::StoreU(hn::Add(va, vb), d, c + i);
    }
}
小结
    

  • SIMD 通过数据并行实现单指令处理多个数据元素，AVX2 可以一次处理 8 个 float32
    • 

  • 自动向量化是首选方案，使用 restrict、内存对齐和编译器报告来帮助编译器优化
    • 

  • 当自动向量化不足时，使用 Intrinsics 手动编写 SIMD 代码，注意处理循环尾部
    • 

  • SIMD 不仅用于数值计算，在字符串处理、数据库引擎、JSON 解析等领域同样有巨大收益
    • 

  • 跨平台可移植性是 SIMD 编程的主要挑战，使用 Highway 或 Rust portable SIMD 等抽象层
    • 

  • SIMD 优化的前提是数据连续排列——先优化数据布局（SoA 模式），再考虑向量化
