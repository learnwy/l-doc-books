DDD 测试策略

核心概念

DDD 的分层架构为测试策略提供了清晰的指导：领域层用单元测试验证业务规则，应用层用集成测试验证工作流编排，基础设施层用适配器测试验证技术正确性。好的测试策略应该让领域层的测试快速、独立且有表达力——它们是业务规则的可执行文档。

测试金字塔在 DDD 项目中有了更精确的含义：底层大量的聚合单元测试保护业务规则，中层的应用服务集成测试验证流程编排，顶层少量的端到端测试验证系统整体行为。

聚合单元测试
聚合的单元测试是 DDD 测试中最有价值的部分。它们验证业务不变量，速度极快，且不需要任何外部依赖。


describe('Order Aggregate', () => {
  function createDraftOrder(): Order {
    return new Order(
      createOrderId('ord-1'),
      createCustomerId('cust-1'),
      Address.create({ province: '浙江', city: '杭州', district: '西湖区', street: '文三路', postalCode: '310000' }),
    );
  }

  describe('addItem', () => {
    it('should add item to draft order', () => {
      const order = createDraftOrder();
      order.addItem(createProductId('prod-1'), 2, Money.of(100, 'CNY'));

      expect(order.itemCount).toBe(1);
      expect(order.totalAmount.equals(Money.of(200, 'CNY'))).toBe(true);
    });

    it('should merge quantity when adding same product', () => {
      const order = createDraftOrder();
      const productId = createProductId('prod-1');
      order.addItem(productId, 2, Money.of(100, 'CNY'));
      order.addItem(productId, 3, Money.of(100, 'CNY'));

      expect(order.itemCount).toBe(1);
      expect(order.totalAmount.equals(Money.of(500, 'CNY'))).toBe(true);
    });

    it('should reject adding item to submitted order', () => {
      const order = createDraftOrder();
      order.addItem(createProductId('prod-1'), 1, Money.of(100, 'CNY'));
      order.submit();

      expect(() => {
        order.addItem(createProductId('prod-2'), 1, Money.of(50, 'CNY'));
      }).toThrow('Cannot add items to a non-draft order');
    });

    it('should reject zero or negative quantity', () => {
      const order = createDraftOrder();
      expect(() => {
        order.addItem(createProductId('prod-1'), 0, Money.of(100, 'CNY'));
      }).toThrow('Quantity must be positive');
    });
  });

  describe('submit', () => {
    it('should transition to submitted status', () => {
      const order = createDraftOrder();
      order.addItem(createProductId('prod-1'), 1, Money.of(100, 'CNY'));
      order.submit();

      expect(order.status).toBe('submitted');
    });

    it('should reject submitting empty order', () => {
      const order = createDraftOrder();
      expect(() => order.submit()).toThrow('Cannot submit an empty order');
    });
  });
});
领域事件测试
验证聚合在状态变更时是否正确发布了领域事件。


describe('Order Domain Events', () => {
  it('should emit OrderPlaced event when submitted', () => {
    const order = createDraftOrder();
    order.addItem(createProductId('prod-1'), 2, Money.of(100, 'CNY'));
    order.submit();

    const events = order.pullDomainEvents();
    expect(events).toHaveLength(1);
    expect(events[0].type).toBe('OrderPlaced');

    const event = events[0] as OrderPlacedEvent;
    expect(event.orderId).toBe('ord-1');
    expect(event.items).toHaveLength(1);
    expect(event.totalAmount).toBe(200);
  });

  it('should emit OrderCancelled event when cancelled', () => {
    const order = createSubmittedOrder();
    order.cancel('用户主动取消', 'user-1');

    const events = order.pullDomainEvents();
    expect(events).toHaveLength(1);
    expect(events[0].type).toBe('OrderCancelled');

    const event = events[0] as OrderCancelledEvent;
    expect(event.reason).toBe('用户主动取消');
  });

  it('should clear events after pulling', () => {
    const order = createDraftOrder();
    order.addItem(createProductId('prod-1'), 1, Money.of(100, 'CNY'));
    order.submit();

    order.pullDomainEvents();
    const secondPull = order.pullDomainEvents();
    expect(secondPull).toHaveLength(0);
  });
});
应用服务集成测试
应用服务测试使用内存仓储替代真实数据库，验证命令处理的完整流程。


describe('PlaceOrderUseCase', () => {
  let orderRepo: InMemoryOrderRepository;
  let catalogStub: StubProductCatalog;
  let notificationSpy: SpyNotificationService;
  let useCase: PlaceOrderUseCase;

  beforeEach(() => {
    orderRepo = new InMemoryOrderRepository();
    catalogStub = new StubProductCatalog();
    notificationSpy = new SpyNotificationService();
    useCase = new PlaceOrderUseCase(orderRepo, catalogStub, notificationSpy);
  });

  it('should create and persist order', async () => {
    catalogStub.setPrice('prod-1', Money.of(100, 'CNY'));
    catalogStub.setAvailability('prod-1', true);

    const orderId = await useCase.execute({
      customerId: 'cust-1',
      items: [{ productId: 'prod-1', quantity: 2 }],
      shippingAddress: { province: '浙江', city: '杭州', street: '文三路' },
    });

    const savedOrder = await orderRepo.findById(orderId);
    expect(savedOrder).not.toBeNull();
    expect(savedOrder!.status).toBe('submitted');
  });

  it('should send notification after order placed', async () => {
    catalogStub.setPrice('prod-1', Money.of(100, 'CNY'));
    catalogStub.setAvailability('prod-1', true);

    await useCase.execute({
      customerId: 'cust-1',
      items: [{ productId: 'prod-1', quantity: 1 }],
      shippingAddress: { province: '浙江', city: '杭州', street: '文三路' },
    });

    expect(notificationSpy.sentNotifications).toHaveLength(1);
    expect(notificationSpy.sentNotifications[0].customerId).toBe('cust-1');
  });

  it('should reject order when product out of stock', async () => {
    catalogStub.setAvailability('prod-1', false);

    await expect(
      useCase.execute({
        customerId: 'cust-1',
        items: [{ productId: 'prod-1', quantity: 1 }],
        shippingAddress: { province: '浙江', city: '杭州', street: '文三路' },
      }),
    ).rejects.toThrow('out of stock');
  });
});

class StubProductCatalog implements ProductCatalogPort {
  private prices = new Map<string, Money>();
  private availability = new Map<string, boolean>();

  setPrice(productId: string, price: Money): void {
    this.prices.set(productId, price);
  }

  setAvailability(productId: string, available: boolean): void {
    this.availability.set(productId, available);
  }

  async getProductPrice(productId: string): Promise<Money> {
    return this.prices.get(productId) ?? Money.of(0, 'CNY');
  }

  async checkAvailability(productId: string): Promise<boolean> {
    return this.availability.get(productId) ?? false;
  }
}

class SpyNotificationService implements NotificationPort {
  sentNotifications: Array<{ customerId: string; orderId: string }> = [];

  async sendOrderConfirmation(customerId: string, orderId: string): Promise<void> {
    this.sentNotifications.push({ customerId, orderId });
  }
}
小结
    

  • 聚合单元测试是 DDD 测试的核心，验证业务不变量和状态转换
    • 

  • 领域事件测试确保聚合在正确时机发布携带正确数据的事件
    • 

  • 应用服务测试使用内存仓储和测试替身，验证命令处理全流程
    • 

  • 优先使用 Fake 和 Stub，谨慎使用 Mock——过度 Mock 导致测试脆弱
    • 

  • 领域层测试不应依赖任何框架或外部系统，保持毫秒级执行速度
    • 

  • 测试是业务规则的可执行文档，测试命名应使用业务语言
    •
