熔断器模式

核心概念

熔断器模式（Circuit Breaker）是分布式系统中防止级联故障的关键弹性模式。当下游服务出现故障时，熔断器自动切断对该服务的调用，避免大量请求堆积导致调用方资源耗尽。故障恢复后，熔断器逐步恢复流量，实现系统的自愈能力。

熔断器的设计灵感来自电气领域的断路器——当电路过载时自动断开，防止整个系统烧毁。在微服务架构中，一个服务的故障不应导致整个调用链的崩溃。

状态机实现
熔断器的核心是一个三状态有限状态机，配合滑动窗口统计失败率来触发状态转换。


type CircuitState = 'closed' | 'open' | 'half-open';

interface CircuitBreakerOptions {
  failureThreshold: number;
  successThreshold: number;
  timeout: number;
  windowSize: number;
}

interface CircuitBreaker {
  execute<T>(fn: () => Promise<T>): Promise<T>;
  getState(): CircuitState;
  getStats(): { failures: number; successes: number; state: CircuitState };
  reset(): void;
}

const createCircuitBreaker = (options: CircuitBreakerOptions): CircuitBreaker => {
  let state: CircuitState = 'closed';
  let failures = 0;
  let successes = 0;
  let lastFailureTime = 0;
  let halfOpenSuccesses = 0;
  const recentResults: boolean[] = [];

  const recordResult = (success: boolean) => {
    recentResults.push(success);
    if (recentResults.length > options.windowSize) {
      recentResults.shift();
    }
  };

  const getFailureRate = () => {
    if (recentResults.length < options.windowSize) return 0;
    const failureCount = recentResults.filter((r) => !r).length;
    return failureCount / recentResults.length;
  };

  const transition = (newState: CircuitState) => {
    if (state !== newState) {
      console.log(`Circuit breaker: ${state} → ${newState}`);
      state = newState;
      if (newState === 'half-open') {
        halfOpenSuccesses = 0;
      }
    }
  };

  const execute = async <T>(fn: () => Promise<T>): Promise<T> => {
    switch (state) {
      case 'open': {
        if (Date.now() - lastFailureTime >= options.timeout) {
          transition('half-open');
        } else {
          throw new CircuitOpenError('Circuit breaker is open');
        }
        break;
      }
      case 'half-open':
        break;
      case 'closed':
        break;
    }

    try {
      const result = await fn();
      onSuccess();
      return result;
    } catch (error) {
      onFailure();
      throw error;
    }
  };

  const onSuccess = () => {
    successes++;
    recordResult(true);

    if (state === 'half-open') {
      halfOpenSuccesses++;
      if (halfOpenSuccesses >= options.successThreshold) {
        transition('closed');
        recentResults.length = 0;
        failures = 0;
      }
    }
  };

  const onFailure = () => {
    failures++;
    lastFailureTime = Date.now();
    recordResult(false);

    if (state === 'half-open') {
      transition('open');
    } else if (state === 'closed' && getFailureRate() >= options.failureThreshold) {
      transition('open');
    }
  };

  return {
    execute,
    getState: () => state,
    getStats: () => ({ failures, successes, state }),
    reset: () => {
      state = 'closed';
      failures = 0;
      successes = 0;
      recentResults.length = 0;
    },
  };
};

class CircuitOpenError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'CircuitOpenError';
  }
}
滑动窗口统计确保熔断决策基于近期的失败率而非历史累计。窗口大小决定了熔断器的敏感度——窗口越小反应越快但可能误判，窗口越大越稳定但反应较慢。


降级策略
熔断器打开时，服务应提供降级响应而非直接报错。降级策略决定了在故障期间用户获得的体验。


interface FallbackStrategy<T> {
  execute(): Promise<T>;
}

const createCachedFallback = <T>(cacheKey: string, cache: Map<string, T>): FallbackStrategy<T> => ({
  execute: async () => {
    const cached = cache.get(cacheKey);
    if (cached) return cached;
    throw new Error('No cached fallback available');
  },
});

const createDefaultFallback = <T>(defaultValue: T): FallbackStrategy<T> => ({
  execute: async () => defaultValue,
});

const createCircuitBreakerWithFallback = <T>(options: CircuitBreakerOptions, fallback: FallbackStrategy<T>) => {
  const breaker = createCircuitBreaker(options);

  return {
    async execute(fn: () => Promise<T>): Promise<T> {
      try {
        return await breaker.execute(fn);
      } catch (error) {
        if (error instanceof CircuitOpenError) {
          console.warn('Circuit open, using fallback');
          return fallback.execute();
        }
        throw error;
      }
    },
    getState: breaker.getState,
    getStats: breaker.getStats,
  };
};

const cache = new Map<string, unknown>();

const productService = createCircuitBreakerWithFallback(
  { failureThreshold: 0.5, successThreshold: 3, timeout: 30_000, windowSize: 10 },
  createCachedFallback('products', cache),
);

服务调用弹性封装
在实际项目中，熔断器通常与重试、超时组合使用，形成完整的服务调用弹性层。


interface ResilientClientOptions {
  circuitBreaker: CircuitBreakerOptions;
  retry: { maxRetries: number; delayMs: number; backoff: number };
  timeout: number;
}

const createResilientClient = (options: ResilientClientOptions) => {
  const breaker = createCircuitBreaker(options.circuitBreaker);

  const withTimeout = <T>(promise: Promise<T>): Promise<T> => {
    return Promise.race([
      promise,
      new Promise<never>((_, reject) => setTimeout(() => reject(new Error('Request timeout')), options.timeout)),
    ]);
  };

  const withRetry = async <T>(fn: () => Promise<T>): Promise<T> => {
    let lastError: Error | undefined;
    let delay = options.retry.delayMs;

    for (let i = 0; i <= options.retry.maxRetries; i++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error as Error;
        if (i < options.retry.maxRetries) {
          await new Promise((r) => setTimeout(r, delay));
          delay *= options.retry.backoff;
        }
      }
    }
    throw lastError;
  };

  return {
    execute: <T>(fn: () => Promise<T>): Promise<T> => breaker.execute(() => withRetry(() => withTimeout(fn()))),
    getState: breaker.getState,
    getStats: breaker.getStats,
  };
};

const userServiceClient = createResilientClient({
  circuitBreaker: {
    failureThreshold: 0.5,
    successThreshold: 3,
    timeout: 30_000,
    windowSize: 20,
  },
  retry: { maxRetries: 2, delayMs: 500, backoff: 2 },
  timeout: 5000,
});

const fetchUser = (id: string) =>
  userServiceClient.execute(() =>
    fetch(`http://user-service/api/users/${id}`).then((r) => {
      if (!r.ok) throw new Error(`HTTP ${r.status}`);
      return r.json();
    }),
  );
弹性封装的执行顺序是：熔断器 → 重试 → 超时。熔断器在最外层决定是否尝试调用，重试在失败时自动重试，超时在单次调用层面限制等待时间。这三者的组合提供了多层次的故障防护。


小结
    

  • 熔断器通过三态状态机（Closed → Open → Half-Open）实现自动故障隔离和恢复
    • 

  • 滑动窗口统计提供基于近期表现的熔断决策，避免历史数据干扰
    • 

  • 降级策略在熔断期间提供替代响应，缓存降级适合读操作，队列降级适合写操作
    • 

  • 弹性封装将熔断器、重试和超时组合为完整的服务调用防护层
    • 

  • Half-Open 状态通过有限探测请求验证下游恢复，避免一次性放量
    • 

  • 熔断器参数（阈值、窗口、超时）需要根据服务特性和 SLA 调整
