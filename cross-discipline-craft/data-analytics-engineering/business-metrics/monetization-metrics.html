变现指标：LTV/CAC/ARPU/ARPPU、付费率、ROI 分析

变现指标体系

变现指标衡量产品的商业可持续性。理解和监控这些指标是评估商业模型健康度和优化收入策略的基础。



核心指标定义




指标
全称
公式
含义





LTV
Lifetime Value
用户全生命周期的总贡献收入
一个用户值多少钱




CAC
Customer Acquisition Cost
总获客花费 / 新增付费用户数
获取一个用户花多少钱




ARPU
Average Revenue Per User
总收入 / 总活跃用户数
平均每用户贡献收入




ARPPU
Average Revenue Per Paying User
总收入 / 付费用户数
平均每付费用户贡献收入




付费率
Paying Rate
付费用户数 / 总活跃用户数
用户的付费转化比例





指标关系
核心公式推导：






公式
含义





ARPU = 付费率 × ARPPU
总人均 = 付费比例 × 付费人均




LTV = ARPU × 平均生命周期月数
总价值 = 月均贡献 × 存活月数




LTV = ARPPU × 付费率 × 生命周期
拆解到付费维度




ROI = (LTV - CAC) / CAC
投资回报率




回收期 = CAC / 月 ARPU
多少个月收回获客成本







LTV 计算方法
方法一：历史平均法
-- 基于历史数据计算 LTV
WITH user_revenue AS (
    SELECT
        u.user_id,
        u.created_at AS registration_date,
        EXTRACT(EPOCH FROM (MAX(o.order_date) - u.created_at)) / 86400 / 30 AS lifetime_months,
        SUM(o.amount) AS total_revenue
    FROM users u
    LEFT JOIN orders o ON u.user_id = o.user_id
    WHERE u.created_at <= CURRENT_DATE - INTERVAL '12 months'
    GROUP BY u.user_id, u.created_at
)
SELECT
    ROUND(AVG(total_revenue), 2) AS avg_ltv,
    ROUND(PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY total_revenue), 2) AS median_ltv,
    ROUND(AVG(lifetime_months), 1) AS avg_lifetime_months,
    ROUND(AVG(total_revenue) / NULLIF(AVG(lifetime_months), 0), 2) AS monthly_arpu
FROM user_revenue;
方法二：队列累积法
-- 按注册月份的队列 LTV 曲线
WITH cohort_revenue AS (
    SELECT
        DATE_TRUNC('month', u.created_at) AS cohort_month,
        EXTRACT(YEAR FROM AGE(DATE_TRUNC('month', o.order_date), DATE_TRUNC('month', u.created_at))) * 12
            + EXTRACT(MONTH FROM AGE(DATE_TRUNC('month', o.order_date), DATE_TRUNC('month', u.created_at))) AS month_number,
        COUNT(DISTINCT u.user_id) AS paying_users,
        SUM(o.amount) AS revenue
    FROM users u
    INNER JOIN orders o ON u.user_id = o.user_id
    GROUP BY cohort_month, month_number
),
cohort_sizes AS (
    SELECT DATE_TRUNC('month', created_at) AS cohort_month, COUNT(*) AS cohort_size
    FROM users
    GROUP BY DATE_TRUNC('month', created_at)
)
SELECT
    cr.cohort_month,
    cs.cohort_size,
    cr.month_number,
    cr.revenue,
    SUM(cr.revenue) OVER (
        PARTITION BY cr.cohort_month ORDER BY cr.month_number
    ) AS cumulative_revenue,
    ROUND(
        SUM(cr.revenue) OVER (
            PARTITION BY cr.cohort_month ORDER BY cr.month_number
        ) / cs.cohort_size, 2
    ) AS ltv_per_user
FROM cohort_revenue cr
JOIN cohort_sizes cs ON cr.cohort_month = cs.cohort_month
WHERE cr.month_number <= 12
ORDER BY cr.cohort_month, cr.month_number;


CAC 计算
渠道级别 CAC
-- 按渠道计算 CAC
WITH channel_costs AS (
    SELECT
        DATE_TRUNC('month', spend_date) AS month,
        channel,
        SUM(spend_amount) AS total_spend
    FROM marketing_spend
    GROUP BY DATE_TRUNC('month', spend_date), channel
),
channel_acquisitions AS (
    SELECT
        DATE_TRUNC('month', u.created_at) AS month,
        u.registration_source AS channel,
        COUNT(DISTINCT u.user_id) AS new_users,
        COUNT(DISTINCT o.user_id) AS new_paying_users
    FROM users u
    LEFT JOIN orders o ON u.user_id = o.user_id
        AND o.order_date BETWEEN u.created_at AND u.created_at + INTERVAL '30 days'
    GROUP BY DATE_TRUNC('month', u.created_at), u.registration_source
)
SELECT
    a.month,
    a.channel,
    a.new_users,
    a.new_paying_users,
    c.total_spend,
    ROUND(c.total_spend / NULLIF(a.new_users, 0), 2) AS cac_per_user,
    ROUND(c.total_spend / NULLIF(a.new_paying_users, 0), 2) AS cac_per_paying_user
FROM channel_acquisitions a
LEFT JOIN channel_costs c ON a.month = c.month AND a.channel = c.channel
ORDER BY a.month, a.channel;


ARPU 与 ARPPU
-- ARPU 和 ARPPU 月度趋势
SELECT
    DATE_TRUNC('month', e.event_time)::DATE AS month,
    COUNT(DISTINCT e.user_id) AS mau,
    COUNT(DISTINCT o.user_id) AS paying_users,
    COALESCE(SUM(o.amount), 0) AS total_revenue,
    ROUND(COALESCE(SUM(o.amount), 0) / NULLIF(COUNT(DISTINCT e.user_id), 0), 2) AS arpu,
    ROUND(COALESCE(SUM(o.amount), 0) / NULLIF(COUNT(DISTINCT o.user_id), 0), 2) AS arppu,
    ROUND(COUNT(DISTINCT o.user_id)::NUMERIC / NULLIF(COUNT(DISTINCT e.user_id), 0) * 100, 2) AS paying_rate
FROM events e
LEFT JOIN orders o ON e.user_id = o.user_id
    AND DATE_TRUNC('month', o.order_date) = DATE_TRUNC('month', e.event_time)
WHERE e.event_time >= CURRENT_DATE - INTERVAL '12 months'
GROUP BY DATE_TRUNC('month', e.event_time)
ORDER BY month;

ROI 分析
LTV:CAC 比率




LTV:CAC 比率
含义
建议行动





< 1:1
亏损
立即优化获客和变现




1:1 - 3:1
微利或风险
优化变现效率




3:1
健康
这是 SaaS 行业的黄金比




> 5:1
可能增长不足
加大获客投入





-- LTV:CAC 比率按渠道分析
WITH channel_ltv AS (
    SELECT
        u.registration_source AS channel,
        COUNT(DISTINCT u.user_id) AS users,
        ROUND(AVG(COALESCE(rev.total_revenue, 0)), 2) AS avg_ltv
    FROM users u
    LEFT JOIN (
        SELECT user_id, SUM(amount) AS total_revenue
        FROM orders
        GROUP BY user_id
    ) rev ON u.user_id = rev.user_id
    WHERE u.created_at >= CURRENT_DATE - INTERVAL '12 months'
    GROUP BY u.registration_source
),
channel_cac AS (
    SELECT
        channel,
        SUM(spend_amount) AS total_spend,
        COUNT(DISTINCT user_id) AS acquired_users,
        ROUND(SUM(spend_amount) / NULLIF(COUNT(DISTINCT user_id), 0), 2) AS cac
    FROM marketing_attribution
    WHERE attribution_date >= CURRENT_DATE - INTERVAL '12 months'
    GROUP BY channel
)
SELECT
    l.channel,
    l.users,
    l.avg_ltv,
    c.cac,
    ROUND(l.avg_ltv / NULLIF(c.cac, 0), 2) AS ltv_cac_ratio,
    CASE
        WHEN l.avg_ltv / NULLIF(c.cac, 0) >= 3 THEN '健康'
        WHEN l.avg_ltv / NULLIF(c.cac, 0) >= 1 THEN '关注'
        ELSE '亏损'
    END AS health_status
FROM channel_ltv l
LEFT JOIN channel_cac c ON l.channel = c.channel
ORDER BY ltv_cac_ratio DESC;
回收期分析
-- CAC 回收期计算
WITH monthly_arpu AS (
    SELECT
        u.registration_source AS channel,
        EXTRACT(YEAR FROM AGE(DATE_TRUNC('month', o.order_date), DATE_TRUNC('month', u.created_at))) * 12
            + EXTRACT(MONTH FROM AGE(DATE_TRUNC('month', o.order_date), DATE_TRUNC('month', u.created_at))) AS month_number,
        SUM(o.amount) / COUNT(DISTINCT u.user_id) AS monthly_arpu
    FROM users u
    INNER JOIN orders o ON u.user_id = o.user_id
    WHERE u.created_at >= CURRENT_DATE - INTERVAL '12 months'
    GROUP BY u.registration_source, month_number
),
cumulative_arpu AS (
    SELECT
        channel,
        month_number,
        SUM(monthly_arpu) OVER (PARTITION BY channel ORDER BY month_number) AS cumulative_arpu
    FROM monthly_arpu
)
SELECT
    ca.channel,
    MIN(ca.month_number) FILTER (
        WHERE ca.cumulative_arpu >= mc.cac
    ) AS payback_months
FROM cumulative_arpu ca
CROSS JOIN (
    SELECT channel, SUM(spend_amount) / NULLIF(COUNT(DISTINCT user_id), 0) AS cac
    FROM marketing_attribution
    GROUP BY channel
) mc
WHERE ca.channel = mc.channel
GROUP BY ca.channel, mc.cac;

小结
    

  • LTV、CAC、ARPU 和 ARPPU 是变现指标体系的四大支柱
    • 

  • ARPU = 付费率 × ARPPU，提升 ARPU 可以从付费率和 ARPPU 两个方向入手
    • 

  • LTV:CAC >= 3:1 是健康商业模型的黄金比率
    • 

  • 队列累积法是计算 LTV 最可靠的方法，可以观察不同队列的 LTV 曲线
    • 

  • CAC 回收期衡量投资回报的时间维度，与 LTV:CAC 比率共同评估渠道效率
    •
