留存分析：DAU/MAU/留存率计算、留存曲线、提升策略

活跃用户指标

活跃用户指标是衡量产品健康度的最基础指标，DAU（日活跃用户）和 MAU（月活跃用户）是最常用的两个度量。

活跃用户定义

指标 全称 定义 常见口径
DAU Daily Active Users 一天内活跃的独立用户数 触发任意事件
WAU Weekly Active Users 一周内活跃的独立用户数 7 天窗口内触发任意事件
MAU Monthly Active Users 一月内活跃的独立用户数 30 天窗口内触发任意事件 
-- DAU 计算
SELECT
    DATE(event_time) AS date,
    COUNT(DISTINCT user_id) AS dau
FROM events
WHERE event_time >= CURRENT_DATE - INTERVAL '90 days'
GROUP BY DATE(event_time)
ORDER BY date;

-- MAU 计算（滚动 30 天窗口）
SELECT
    d.date,
    COUNT(DISTINCT e.user_id) AS mau
FROM generate_series(
    CURRENT_DATE - INTERVAL '90 days',
    CURRENT_DATE,
    '1 day'
) AS d(date)
LEFT JOIN events e ON DATE(e.event_time) BETWEEN d.date - INTERVAL '29 days' AND d.date
GROUP BY d.date
ORDER BY d.date;

DAU/MAU 比率

DAU/MAU 比率衡量用户的使用频率，反映产品的粘性。

DAU/MAU 比率 含义 产品类型示例
> 50% 极高粘性 即时通讯、社交
25% - 50% 高粘性 新闻、短视频
10% - 25% 中等粘性 电商、工具
< 10% 低粘性 旅行预订、房产 
-- DAU/MAU 比率
WITH daily_active AS (
    SELECT DATE(event_time) AS date, COUNT(DISTINCT user_id) AS dau
    FROM events
    WHERE event_time >= CURRENT_DATE - INTERVAL '60 days'
    GROUP BY DATE(event_time)
),
monthly_active AS (
    SELECT
        d.date,
        COUNT(DISTINCT e.user_id) AS mau
    FROM generate_series(CURRENT_DATE - INTERVAL '60 days', CURRENT_DATE, '1 day') AS d(date)
    LEFT JOIN events e ON DATE(e.event_time) BETWEEN d.date - INTERVAL '29 days' AND d.date
    GROUP BY d.date
)
SELECT
    da.date,
    da.dau,
    ma.mau,
    ROUND(da.dau::NUMERIC / NULLIF(ma.mau, 0), 4) AS stickiness_ratio
FROM daily_active da
JOIN monthly_active ma ON da.date = ma.date
ORDER BY da.date;

-- 经典 N 日留存率
WITH cohort AS (
    SELECT
        user_id,
        MIN(DATE(event_time)) AS first_date
    FROM events
    GROUP BY user_id
)
SELECT
    c.first_date AS cohort_date,
    COUNT(DISTINCT c.user_id) AS cohort_size,
    COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 1 THEN e.user_id END) AS day_1,
    COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 3 THEN e.user_id END) AS day_3,
    COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 7 THEN e.user_id END) AS day_7,
    COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 14 THEN e.user_id END) AS day_14,
    COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 30 THEN e.user_id END) AS day_30,
    ROUND(COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 1 THEN e.user_id END)::NUMERIC
        / COUNT(DISTINCT c.user_id) * 100, 2) AS day_1_rate,
    ROUND(COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 7 THEN e.user_id END)::NUMERIC
        / COUNT(DISTINCT c.user_id) * 100, 2) AS day_7_rate,
    ROUND(COUNT(DISTINCT CASE WHEN DATE(e.event_time) = c.first_date + 30 THEN e.user_id END)::NUMERIC
        / COUNT(DISTINCT c.user_id) * 100, 2) AS day_30_rate
FROM cohort c
LEFT JOIN events e ON c.user_id = e.user_id
WHERE c.first_date >= CURRENT_DATE - INTERVAL '60 days'
  AND c.first_date <= CURRENT_DATE - INTERVAL '31 days'
GROUP BY c.first_date
ORDER BY c.first_date;

-- 流失预警：识别活跃度下降的用户
WITH user_activity AS (
    SELECT
        user_id,
        COUNT(DISTINCT DATE(event_time)) FILTER (
            WHERE event_time BETWEEN CURRENT_DATE - INTERVAL '7 days' AND CURRENT_DATE
        ) AS active_days_recent,
        COUNT(DISTINCT DATE(event_time)) FILTER (
            WHERE event_time BETWEEN CURRENT_DATE - INTERVAL '14 days' AND CURRENT_DATE - INTERVAL '7 days'
        ) AS active_days_previous
    FROM events
    WHERE event_time >= CURRENT_DATE - INTERVAL '14 days'
    GROUP BY user_id
)
SELECT
    user_id,
    active_days_recent,
    active_days_previous,
    CASE
        WHEN active_days_recent = 0 AND active_days_previous > 0 THEN '已流失'
        WHEN active_days_recent < active_days_previous * 0.5 THEN '高流失风险'
        WHEN active_days_recent < active_days_previous THEN '活跃度下降'
        ELSE '正常'
    END AS risk_level
FROM user_activity
WHERE active_days_previous > 0
ORDER BY risk_level;


用户生命周期状态
-- 用户生命周期状态划分
WITH user_monthly AS (
    SELECT
        user_id,
        DATE_TRUNC('month', event_time)::DATE AS month,
        1 AS active
    FROM events
    GROUP BY user_id, DATE_TRUNC('month', event_time)
),
user_lifecycle AS (
    SELECT
        user_id,
        month,
        LAG(month) OVER (PARTITION BY user_id ORDER BY month) AS prev_active_month,
        LEAD(month) OVER (PARTITION BY user_id ORDER BY month) AS next_active_month,
        MIN(month) OVER (PARTITION BY user_id) AS first_month
    FROM user_monthly
)
SELECT
    month,
    COUNT(DISTINCT CASE WHEN month = first_month THEN user_id END) AS new_users,
    COUNT(DISTINCT CASE WHEN month != first_month AND prev_active_month = month - INTERVAL '1 month' THEN user_id END) AS retained_users,
    COUNT(DISTINCT CASE WHEN month != first_month AND prev_active_month < month - INTERVAL '1 month' THEN user_id END) AS resurrected_users
FROM user_lifecycle
GROUP BY month
ORDER BY month;

    • 

  • 留存率的计算有经典 N 日留存、范围留存、滚动留存等多种口径
