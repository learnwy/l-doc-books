增长指标：病毒系数、K-Factor、增长飞轮、增长实验框架

增长的度量

产品增长不是简单的用户数增加，而是一个涵盖获客效率、用户激活、传播扩散和价值积累的系统性过程。增长指标帮助团队量化增长的各个环节，识别增长杠杆。

-- K-Factor 计算
WITH invite_metrics AS (
    SELECT
        DATE_TRUNC('week', e.event_time)::DATE AS week,
        COUNT(DISTINCT e.user_id) AS inviters,
        COUNT(*) AS total_invites,
        COUNT(DISTINCT CASE WHEN u.user_id IS NOT NULL THEN e.properties->>'invited_user_id' END) AS converted_invites
    FROM events e
    LEFT JOIN users u ON (e.properties->>'invited_user_id') = u.user_id::TEXT
        AND u.created_at BETWEEN e.event_time AND e.event_time + INTERVAL '7 days'
    WHERE e.event_name = 'invite_sent'
      AND e.event_time >= CURRENT_DATE - INTERVAL '12 weeks'
    GROUP BY DATE_TRUNC('week', e.event_time)
),
active_users AS (
    SELECT
        DATE_TRUNC('week', event_time)::DATE AS week,
        COUNT(DISTINCT user_id) AS wau
    FROM events
    WHERE event_time >= CURRENT_DATE - INTERVAL '12 weeks'
    GROUP BY DATE_TRUNC('week', event_time)
)
SELECT
    im.week,
    au.wau,
    im.inviters,
    im.total_invites,
    im.converted_invites,
    ROUND(im.total_invites::NUMERIC / NULLIF(au.wau, 0), 2) AS invites_per_user_i,
    ROUND(im.converted_invites::NUMERIC / NULLIF(im.total_invites, 0), 4) AS conversion_rate_c,
    ROUND(
        (im.total_invites::NUMERIC / NULLIF(au.wau, 0))
        * (im.converted_invites::NUMERIC / NULLIF(im.total_invites, 0)),
    4) AS k_factor
FROM invite_metrics im
JOIN active_users au ON im.week = au.week
ORDER BY im.week;

-- 飞轮关键指标监控
SELECT
    DATE_TRUNC('month', event_time)::DATE AS month,
    -- 用户侧
    COUNT(DISTINCT user_id) AS mau,
    COUNT(DISTINCT CASE WHEN event_name = 'content_create' THEN user_id END) AS content_creators,
    -- 内容侧
    COUNT(CASE WHEN event_name = 'content_create' THEN 1 END) AS new_content_count,
    -- 消费侧
    COUNT(CASE WHEN event_name = 'content_view' THEN 1 END) AS content_views,
    -- 传播侧
    COUNT(CASE WHEN event_name = 'share' THEN 1 END) AS shares,
    COUNT(DISTINCT CASE WHEN event_name = 'organic_signup' THEN user_id END) AS organic_signups,
    -- 飞轮效率
    ROUND(
        COUNT(DISTINCT CASE WHEN event_name = 'organic_signup' THEN user_id END)::NUMERIC
        / NULLIF(COUNT(DISTINCT user_id), 0), 4
    ) AS organic_growth_ratio
FROM events
WHERE event_time >= CURRENT_DATE - INTERVAL '12 months'
GROUP BY DATE_TRUNC('month', event_time)
ORDER BY month;

-- A/B 测试结果分析
WITH experiment_data AS (
    SELECT
        properties->>'experiment_group' AS test_group,
        COUNT(DISTINCT user_id) AS users,
        COUNT(DISTINCT CASE WHEN event_name = 'purchase' THEN user_id END) AS converters,
        SUM(CASE WHEN event_name = 'purchase' THEN (properties->>'amount')::DECIMAL ELSE 0 END) AS revenue
    FROM events
    WHERE properties->>'experiment_id' = 'exp_checkout_v2'
      AND event_time BETWEEN '2025-03-01' AND '2025-03-15'
    GROUP BY properties->>'experiment_group'
)
SELECT
    test_group,
    users,
    converters,
    ROUND(converters::NUMERIC / users * 100, 4) AS conversion_rate,
    revenue,
    ROUND(revenue / users, 2) AS revenue_per_user
FROM experiment_data
ORDER BY test_group;

增长核算
增长分解
-- 月度增长分解
WITH monthly_users AS (
    SELECT
        DATE_TRUNC('month', event_time)::DATE AS month,
        user_id
    FROM events
    GROUP BY DATE_TRUNC('month', event_time), user_id
),
user_status AS (
    SELECT
        m.month,
        m.user_id,
        CASE
            WHEN first_active.first_month = m.month THEN '新增'
            WHEN prev_month.user_id IS NOT NULL THEN '留存'
            ELSE '回流'
        END AS user_type
    FROM monthly_users m
    LEFT JOIN (SELECT user_id, MIN(month) AS first_month FROM monthly_users GROUP BY user_id) first_active
        ON m.user_id = first_active.user_id
    LEFT JOIN monthly_users prev_month
        ON m.user_id = prev_month.user_id AND prev_month.month = m.month - INTERVAL '1 month'
),
churned AS (
    SELECT
        prev.month + INTERVAL '1 month' AS month,
        COUNT(DISTINCT prev.user_id) AS churned_users
    FROM monthly_users prev
    LEFT JOIN monthly_users curr ON prev.user_id = curr.user_id
        AND curr.month = prev.month + INTERVAL '1 month'
    WHERE curr.user_id IS NULL
    GROUP BY prev.month
)
SELECT
    us.month,
    COUNT(DISTINCT CASE WHEN us.user_type = '新增' THEN us.user_id END) AS new_users,
    COUNT(DISTINCT CASE WHEN us.user_type = '留存' THEN us.user_id END) AS retained_users,
    COUNT(DISTINCT CASE WHEN us.user_type = '回流' THEN us.user_id END) AS resurrected_users,
    COALESCE(c.churned_users, 0) AS churned_users,
    COUNT(DISTINCT us.user_id) AS total_mau,
    COUNT(DISTINCT us.user_id) - COALESCE(
        LAG(COUNT(DISTINCT us.user_id)) OVER (ORDER BY us.month), 0
    ) AS net_growth
FROM user_status us
LEFT JOIN churned c ON us.month = c.month
GROUP BY us.month, c.churned_users
ORDER BY us.month;


小结
    

  • K-Factor = 邀请数/用户 × 邀请转化率，K > 1 意味着病毒式增长
    • 

  • 增长飞轮是自我强化的增长循环，识别和加速飞轮是增长策略的核心
    • 

  • A/B 测试是增长实验的标准方法，需要明确假设、核心指标、护栏指标和统计功效
    • 

  • ICE 评分（Impact × Confidence × Ease）帮助团队为增长实验排优先级
    • 

  • 增长分解将 MAU 变化拆解为新增、留存、回流和流失四个组成部分
