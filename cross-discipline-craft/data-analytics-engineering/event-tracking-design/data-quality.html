数据质量的重要性

数据质量是数据分析可信度的基础。低质量的数据不仅会导致错误的分析结论，还会侵蚀团队对数据的信任，最终使数据驱动决策沦为空谈。

入库后校验 SQL
-- 完整性检查：各事件的空值率
SELECT
    event_name,
    COUNT(*) AS total_events,
    COUNT(*) FILTER (WHERE properties->>'product_id' IS NULL) AS null_product_id,
    ROUND(
        COUNT(*) FILTER (WHERE properties->>'product_id' IS NULL)::NUMERIC
        / COUNT(*) * 100, 2
    ) AS null_rate_pct
FROM events
WHERE event_time >= CURRENT_DATE - INTERVAL '1 day'
  AND event_name IN ('view_product', 'add_to_cart', 'submit_order')
GROUP BY event_name
ORDER BY null_rate_pct DESC;
-- 唯一性检查：重复事件检测
WITH duplicate_check AS (
    SELECT
        event_id,
        event_name,
        user_id,
        event_time,
        ROW_NUMBER() OVER (
            PARTITION BY event_id ORDER BY event_time
        ) AS rn
    FROM events
    WHERE event_time >= CURRENT_DATE - INTERVAL '1 day'
)
SELECT
    event_name,
    COUNT(*) FILTER (WHERE rn > 1) AS duplicate_count,
    COUNT(DISTINCT event_id) AS unique_events,
    ROUND(
        COUNT(*) FILTER (WHERE rn > 1)::NUMERIC
        / COUNT(*) * 100, 2
    ) AS duplicate_rate_pct
FROM duplicate_check
GROUP BY event_name
HAVING COUNT(*) FILTER (WHERE rn > 1) > 0;
-- 有效性检查：业务规则违反
SELECT
    '负数金额' AS rule,
    COUNT(*) AS violation_count
FROM events
WHERE event_name = 'purchase'
  AND (properties->>'amount')::DECIMAL < 0
  AND event_time >= CURRENT_DATE - INTERVAL '1 day'

UNION ALL

SELECT
    '未来时间戳' AS rule,
    COUNT(*) AS violation_count
FROM events
WHERE event_time > NOW() + INTERVAL '1 hour'
  AND event_time >= CURRENT_DATE - INTERVAL '1 day'

UNION ALL

SELECT
    '异常长会话' AS rule,
    COUNT(*) AS violation_count
FROM (
    SELECT session_id, MAX(event_time) - MIN(event_time) AS duration
    FROM events
    WHERE event_time >= CURRENT_DATE - INTERVAL '1 day'
    GROUP BY session_id
    HAVING MAX(event_time) - MIN(event_time) > INTERVAL '24 hours'
) sub;


异常检测
数据量异常检测
-- 数据量同比环比异常检测
WITH daily_counts AS (
    SELECT
        DATE(event_time) AS date,
        event_name,
        COUNT(*) AS event_count
    FROM events
    WHERE event_time >= CURRENT_DATE - INTERVAL '30 days'
    GROUP BY DATE(event_time), event_name
),
with_comparisons AS (
    SELECT
        date,
        event_name,
        event_count,
        LAG(event_count, 1) OVER (PARTITION BY event_name ORDER BY date) AS prev_day,
        LAG(event_count, 7) OVER (PARTITION BY event_name ORDER BY date) AS same_day_last_week,
        AVG(event_count) OVER (
            PARTITION BY event_name
            ORDER BY date
            ROWS BETWEEN 7 PRECEDING AND 1 PRECEDING
        ) AS avg_7d,
        STDDEV(event_count) OVER (
            PARTITION BY event_name
            ORDER BY date
            ROWS BETWEEN 7 PRECEDING AND 1 PRECEDING
        ) AS stddev_7d
    FROM daily_counts
)
SELECT
    date,
    event_name,
    event_count,
    prev_day,
    ROUND((event_count - prev_day)::NUMERIC / NULLIF(prev_day, 0) * 100, 2) AS dod_change_pct,
    ROUND((event_count - same_day_last_week)::NUMERIC / NULLIF(same_day_last_week, 0) * 100, 2) AS wow_change_pct,
    CASE
        WHEN ABS(event_count - avg_7d) > 3 * NULLIF(stddev_7d, 0) THEN '3-sigma 异常'
        WHEN ABS(event_count - avg_7d) > 2 * NULLIF(stddev_7d, 0) THEN '2-sigma 警告'
        ELSE '正常'
    END AS anomaly_status
FROM with_comparisons
WHERE date = CURRENT_DATE - INTERVAL '1 day'
ORDER BY anomaly_status DESC, event_name;

-- 数据修复示例：标记删除重复事件
UPDATE events
SET is_deleted = TRUE
WHERE event_id IN (
    SELECT event_id
    FROM (
        SELECT
            event_id,
            ROW_NUMBER() OVER (
                PARTITION BY user_id, event_name, event_time
                ORDER BY event_id
            ) AS rn
        FROM events
        WHERE event_time >= '2025-03-01' AND event_time < '2025-03-02'
    ) sub
    WHERE rn > 1
);
-- 数据修复示例：属性值纠正
UPDATE events
SET properties = jsonb_set(
    properties,
    '{currency}',
    '"CNY"'
)
WHERE event_name = 'purchase'
  AND properties->>'currency' IS NULL
  AND event_time >= '2025-03-01'
  AND event_time < '2025-03-15';

-- 数据质量日报 SQL
WITH quality_metrics AS (
    SELECT
        event_name,
        COUNT(*) AS total_count,
        COUNT(*) FILTER (WHERE user_id IS NULL) AS null_user_count,
        COUNT(*) FILTER (WHERE properties = '{}' OR properties IS NULL) AS empty_props_count,
        MAX(event_time) AS latest_event,
        NOW() - MAX(event_time) AS data_freshness
    FROM events
    WHERE event_time >= CURRENT_DATE - INTERVAL '1 day'
      AND event_time < CURRENT_DATE
    GROUP BY event_name
)
SELECT
    event_name,
    total_count,
    ROUND(null_user_count::NUMERIC / total_count * 100, 2) AS null_user_rate,
    ROUND(empty_props_count::NUMERIC / total_count * 100, 2) AS empty_props_rate,
    data_freshness,
    CASE
        WHEN null_user_count::NUMERIC / total_count > 0.05 THEN '异常'
        WHEN empty_props_count::NUMERIC / total_count > 0.1 THEN '警告'
        ELSE '正常'
    END AS quality_status
FROM quality_metrics
ORDER BY quality_status DESC, total_count DESC;


小结
    

  • 数据质量的六个核心维度是完整性、准确性、一致性、及时性、唯一性和有效性
    • 

  • 数据校验应在采集端、传输层、入库前和入库后四个层级分别实施
    • 

  • 异常检测结合同比环比分析和 3-Sigma 统计方法，覆盖数据量、空值率和重复率
    • 

  • 数据修复策略包括补发、回填、标记删除和脱敏替换，选择取决于问题类型和严重程度
    • 

  • 持续的数据质量监控体系是保障数据信任度的基础设施
