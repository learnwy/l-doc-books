数据建模：范式设计、星型/雪花模型、维度建模

关系数据库范式

数据建模的基础是关系范式理论。范式是衡量数据库表结构规范程度的标准，遵循范式可以减少数据冗余、避免更新异常，但过度规范化会牺牲查询性能。

范式化示例
-- 违反 2NF 的表（假设主键为 order_id + product_id）
-- product_name 只依赖于 product_id，不依赖于 order_id
CREATE TABLE order_details_bad (
    order_id    INT,
    product_id  INT,
    product_name VARCHAR(100),
    quantity    INT,
    unit_price  DECIMAL(10,2),
    PRIMARY KEY (order_id, product_id)
);

-- 规范化为 2NF
CREATE TABLE products (
    product_id   INT PRIMARY KEY,
    product_name VARCHAR(100)
);

CREATE TABLE order_details (
    order_id   INT,
    product_id INT REFERENCES products(product_id),
    quantity   INT,
    unit_price DECIMAL(10,2),
    PRIMARY KEY (order_id, product_id)
);
-- 违反 3NF 的表
-- city 依赖于 zip_code，zip_code 依赖于主键，形成传递依赖
CREATE TABLE customers_bad (
    customer_id INT PRIMARY KEY,
    name        VARCHAR(100),
    zip_code    VARCHAR(10),
    city        VARCHAR(50)
);

-- 规范化为 3NF
CREATE TABLE zip_codes (
    zip_code VARCHAR(10) PRIMARY KEY,
    city     VARCHAR(50)
);

CREATE TABLE customers (
    customer_id INT PRIMARY KEY,
    name        VARCHAR(100),
    zip_code    VARCHAR(10) REFERENCES zip_codes(zip_code)
);

维度建模
维度建模是面向分析场景的数据建模方法论，由 Ralph Kimball 提出。其核心思想是将数据组织为事实表和维度表的结构，以优化查询性能和用户理解。


事实表与维度表

维度表

事实表类型
-- 事务事实表：记录每一笔业务事件
CREATE TABLE fact_orders (
    order_key       BIGINT PRIMARY KEY,
    date_key        INT REFERENCES dim_date(date_key),
    product_key     INT REFERENCES dim_product(product_key),
    customer_key    INT REFERENCES dim_customer(customer_key),
    order_id        VARCHAR(32),
    quantity        INT,
    unit_price      DECIMAL(10,2),
    discount_amount DECIMAL(10,2),
    total_amount    DECIMAL(10,2)
);

-- 周期快照事实表：记录固定时间间隔的状态
CREATE TABLE fact_daily_inventory (
    date_key        INT REFERENCES dim_date(date_key),
    product_key     INT REFERENCES dim_product(product_key),
    warehouse_key   INT REFERENCES dim_warehouse(warehouse_key),
    quantity_on_hand INT,
    quantity_on_order INT,
    PRIMARY KEY (date_key, product_key, warehouse_key)
);

-- 累积快照事实表：跟踪业务过程的生命周期
CREATE TABLE fact_order_lifecycle (
    order_key       BIGINT PRIMARY KEY,
    order_date_key  INT,
    pay_date_key    INT,
    ship_date_key   INT,
    deliver_date_key INT,
    customer_key    INT,
    total_amount    DECIMAL(10,2),
    order_to_pay_days INT,
    pay_to_ship_days  INT,
    ship_to_deliver_days INT

星型模型与雪花模型
星型模型
星型模型的特点是维度表直接连接到事实表，维度表之间没有关联。这种简洁的结构使得查询只需要一层 JOIN。


SELECT
    d.year,
    d.quarter,
    p.category,
    s.region,
    SUM(f.amount) AS total_sales,
    COUNT(DISTINCT f.customer_key) AS unique_customers
FROM fact_sales f
JOIN dim_date d ON f.date_key = d.date_key
JOIN dim_product p ON f.product_key = p.product_key
JOIN dim_store s ON f.store_key = s.store_key
WHERE d.year = 2025
GROUP BY d.year, d.quarter, p.category, s.region
ORDER BY total_sales DESC;

-- SCD Type 2 实现
CREATE TABLE dim_customer (
    customer_key    SERIAL PRIMARY KEY,
    customer_id     VARCHAR(32),
    customer_name   VARCHAR(100),
    city            VARCHAR(50),
    segment         VARCHAR(20),
    effective_date  DATE NOT NULL,
    expiry_date     DATE NOT NULL DEFAULT '9999-12-31',
    is_current      BOOLEAN NOT NULL DEFAULT TRUE
);

-- 查询当前有效记录
SELECT * FROM dim_customer WHERE is_current = TRUE;

-- 查询某个历史时间点的记录
SELECT * FROM dim_customer
WHERE effective_date <= '2025-03-15'
  AND expiry_date > '2025-03-15';

-- 处理维度变更的 SQL
UPDATE dim_customer
SET expiry_date = CURRENT_DATE,
    is_current = FALSE
WHERE customer_id = 'C001' AND is_current = TRUE;

时间维度设计
时间维度是所有数据仓库中最通用的维度，值得特别设计。


CREATE TABLE dim_date (
    date_key        INT PRIMARY KEY,
    full_date       DATE NOT NULL UNIQUE,
    year            SMALLINT NOT NULL,
    quarter         SMALLINT NOT NULL,
    month           SMALLINT NOT NULL,
    week_of_year    SMALLINT NOT NULL,
    day_of_month    SMALLINT NOT NULL,
    day_of_week     SMALLINT NOT NULL,
    day_name        VARCHAR(10) NOT NULL,
    month_name      VARCHAR(10) NOT NULL,
    is_weekend      BOOLEAN NOT NULL,
    is_holiday      BOOLEAN NOT NULL DEFAULT FALSE,
    fiscal_year     SMALLINT,
    fiscal_quarter  SMALLINT
);

-- 生成时间维度数据
INSERT INTO dim_date
SELECT
    TO_CHAR(d, 'YYYYMMDD')::INT AS date_key,
    d AS full_date,
    EXTRACT(YEAR FROM d) AS year,
    EXTRACT(QUARTER FROM d) AS quarter,
    EXTRACT(MONTH FROM d) AS month,
    EXTRACT(WEEK FROM d) AS week_of_year,
    EXTRACT(DAY FROM d) AS day_of_month,
    EXTRACT(DOW FROM d) AS day_of_week,
    TO_CHAR(d, 'Day') AS day_name,
    TO_CHAR(d, 'Month') AS month_name,
    EXTRACT(DOW FROM d) IN (0, 6) AS is_weekend,
    FALSE AS is_holiday,
    CASE WHEN EXTRACT(MONTH FROM d) >= 4
         THEN EXTRACT(YEAR FROM d)
         ELSE EXTRACT(YEAR FROM d) - 1
    END AS fiscal_year,
    CASE
        WHEN EXTRACT(MONTH FROM d) BETWEEN 4 AND 6 THEN 1
        WHEN EXTRACT(MONTH FROM d) BETWEEN 7 AND 9 THEN 2
        WHEN EXTRACT(MONTH FROM d) BETWEEN 10 AND 12 THEN 3
        ELSE 4
    END AS fiscal_quarter
FROM generate_series('2020-01-01'::DATE, '2030-12-31'::DATE, '1 day') AS d;

实践

审计字段

小结
    

  • 关系范式（1NF → 3NF → BCNF）通过消除函数依赖减少冗余，适用于 OLTP 场景
    • 

  • 维度建模将数据组织为事实表和维度表，以业务过程为核心，适用于 OLAP 场景
    • 

  • 星型模型以简洁性和查询性能见长，是数据仓库的首选模型
    • 

  • 雪花模型在维度属性需要独立管理时使用，但增加了查询复杂度
    • 

  • SCD Type 2 通过新增行的方式保留维度的完整历史变更记录
    • 

  • 时间维度是最通用的维度，应预生成并包含丰富的日历属性
    •
