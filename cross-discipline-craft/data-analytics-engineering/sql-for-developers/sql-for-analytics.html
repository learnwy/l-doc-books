分析 SQL：留存分析、漏斗查询、队列分析

留存分析

留存分析是衡量用户持续使用产品的核心方法。通过 SQL 实现留存分析，你可以精确地计算不同时间粒度下的留存率，并按用户属性进行多维下钻。

留存分析模型

N 日留存率计算
WITH user_first_active AS (
    SELECT
        user_id,
        MIN(DATE(event_time)) AS first_active_date
    FROM user_events
    GROUP BY user_id
),
user_retention AS (
    SELECT
        f.first_active_date,
        f.user_id,
        DATE(e.event_time) AS active_date,
        DATE(e.event_time) - f.first_active_date AS days_since_first
    FROM user_first_active f
    INNER JOIN user_events e ON f.user_id = e.user_id
)
SELECT
    first_active_date,
    COUNT(DISTINCT user_id) AS cohort_size,
    COUNT(DISTINCT CASE WHEN days_since_first = 1 THEN user_id END) AS day_1_retained,
    COUNT(DISTINCT CASE WHEN days_since_first = 7 THEN user_id END) AS day_7_retained,
    COUNT(DISTINCT CASE WHEN days_since_first = 30 THEN user_id END) AS day_30_retained,
    ROUND(COUNT(DISTINCT CASE WHEN days_since_first = 1 THEN user_id END)::NUMERIC
        / COUNT(DISTINCT user_id) * 100, 2) AS day_1_retention_rate,
    ROUND(COUNT(DISTINCT CASE WHEN days_since_first = 7 THEN user_id END)::NUMERIC
        / COUNT(DISTINCT user_id) * 100, 2) AS day_7_retention_rate,
    ROUND(COUNT(DISTINCT CASE WHEN days_since_first = 30 THEN user_id END)::NUMERIC
        / COUNT(DISTINCT user_id) * 100, 2) AS day_30_retention_rate
FROM user_retention
GROUP BY first_active_date
ORDER BY first_active_date;
留存矩阵
留存矩阵以注册日为行、留存天数为列，形成一个完整的留存全景图。


WITH user_first_active AS (
    SELECT
        user_id,
        MIN(DATE(event_time)) AS cohort_date
    FROM user_events
    GROUP BY user_id
),
retention_data AS (
    SELECT
        f.cohort_date,
        DATE(e.event_time) - f.cohort_date AS day_n,
        COUNT(DISTINCT e.user_id) AS retained_users
    FROM user_first_active f
    INNER JOIN user_events e ON f.user_id = e.user_id
    WHERE DATE(e.event_time) - f.cohort_date BETWEEN 0 AND 30
    GROUP BY f.cohort_date, DATE(e.event_time) - f.cohort_date
),
cohort_sizes AS (
    SELECT cohort_date, retained_users AS cohort_size
    FROM retention_data
    WHERE day_n = 0
)
SELECT
    r.cohort_date,
    c.cohort_size,
    r.day_n,
    r.retained_users,
    ROUND(r.retained_users::NUMERIC / c.cohort_size * 100, 2) AS retention_rate
FROM retention_data r
JOIN cohort_sizes c ON r.cohort_date = c.cohort_date
ORDER BY r.cohort_date, r.day_n;

WITH user_first_active AS (
    SELECT user_id, MIN(DATE(event_time)) AS cohort_date
    FROM user_events
    GROUP BY user_id
)
SELECT
    f.cohort_date,
    COUNT(DISTINCT f.user_id) AS cohort_size,
    COUNT(DISTINCT CASE
        WHEN DATE(e.event_time) - f.cohort_date BETWEEN 1 AND 7
        THEN e.user_id END) AS week_1_retained,
    COUNT(DISTINCT CASE
        WHEN DATE(e.event_time) - f.cohort_date BETWEEN 8 AND 14
        THEN e.user_id END) AS week_2_retained,
    COUNT(DISTINCT CASE
        WHEN DATE(e.event_time) - f.cohort_date BETWEEN 15 AND 21
        THEN e.user_id END) AS week_3_retained
FROM user_first_active f
LEFT JOIN user_events e ON f.user_id = e.user_id
GROUP BY f.cohort_date
ORDER BY f.cohort_date;


漏斗分析
漏斗分析追踪用户在预定义步骤序列中的转化情况，是产品分析中最重要的分析范式之一。


漏斗模型
严格有序漏斗
严格有序漏斗要求用户按照预定义的步骤顺序依次完成，中间不能跳步。


WITH step_1 AS (
    SELECT DISTINCT
        user_id,
        MIN(event_time) AS step_1_time
    FROM user_events
    WHERE event_name = 'view_product'
      AND DATE(event_time) = '2025-03-01'
    GROUP BY user_id
),
step_2 AS (
    SELECT
        s1.user_id,
        s1.step_1_time,
        MIN(e.event_time) AS step_2_time
    FROM step_1 s1
    INNER JOIN user_events e ON s1.user_id = e.user_id
    WHERE e.event_name = 'add_to_cart'
      AND e.event_time > s1.step_1_time
      AND e.event_time <= s1.step_1_time + INTERVAL '1 hour'
    GROUP BY s1.user_id, s1.step_1_time
),
step_3 AS (
    SELECT
        s2.user_id,
        s2.step_2_time,
        MIN(e.event_time) AS step_3_time
    FROM step_2 s2
    INNER JOIN user_events e ON s2.user_id = e.user_id
    WHERE e.event_name = 'submit_order'
      AND e.event_time > s2.step_2_time
      AND e.event_time <= s2.step_2_time + INTERVAL '30 minutes'
    GROUP BY s2.user_id, s2.step_2_time
),
step_4 AS (
    SELECT
        s3.user_id,
        s3.step_3_time,
        MIN(e.event_time) AS step_4_time
    FROM step_3 s3
    INNER JOIN user_events e ON s3.user_id = e.user_id
    WHERE e.event_name = 'payment_success'
      AND e.event_time > s3.step_3_time
      AND e.event_time <= s3.step_3_time + INTERVAL '30 minutes'
    GROUP BY s3.user_id, s3.step_3_time
)
SELECT
    '浏览商品' AS step_name,
    1 AS step_order,
    (SELECT COUNT(*) FROM step_1) AS user_count,
    100.0 AS conversion_rate
UNION ALL
SELECT '加入购物车', 2,
    (SELECT COUNT(*) FROM step_2),
    ROUND((SELECT COUNT(*) FROM step_2)::NUMERIC / (SELECT COUNT(*) FROM step_1) * 100, 2)
UNION ALL
SELECT '提交订单', 3,
    (SELECT COUNT(*) FROM step_3),
    ROUND((SELECT COUNT(*) FROM step_3)::NUMERIC / (SELECT COUNT(*) FROM step_1) * 100, 2)
UNION ALL
SELECT '完成支付', 4,
    (SELECT COUNT(*) FROM step_4),
    ROUND((SELECT COUNT(*) FROM step_4)::NUMERIC / (SELECT COUNT(*) FROM step_1) * 100, 2);
窗口函数漏斗
使用窗口函数可以更优雅地实现漏斗分析。


WITH ordered_events AS (
    SELECT
        user_id,
        event_name,
        event_time,
        CASE event_name
            WHEN 'view_product'    THEN 1
            WHEN 'add_to_cart'     THEN 2
            WHEN 'submit_order'    THEN 3
            WHEN 'payment_success' THEN 4
        END AS step_order
    FROM user_events
    WHERE event_name IN ('view_product', 'add_to_cart', 'submit_order', 'payment_success')
      AND DATE(event_time) = '2025-03-01'
),
user_max_step AS (
    SELECT
        user_id,
        MAX(step_order) AS max_step_reached
    FROM (
        SELECT
            user_id,
            step_order,
            step_order - ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY event_time) AS grp
        FROM ordered_events
    ) sub
    WHERE grp = 0
    GROUP BY user_id
)
SELECT
    step_order,
    CASE step_order
        WHEN 1 THEN '浏览商品'
        WHEN 2 THEN '加入购物车'
        WHEN 3 THEN '提交订单'
        WHEN 4 THEN '完成支付'
    END AS step_name,
    COUNT(*) AS user_count,
    ROUND(COUNT(*)::NUMERIC / MAX(COUNT(*)) OVER () * 100, 2) AS overall_conversion
FROM user_max_step
CROSS JOIN generate_series(1, 4) AS step_order
WHERE max_step_reached >= step_order
GROUP BY step_order
ORDER BY step_order;

队列分析（Cohort Analysis）
队列分析将用户按某个共同特征（通常是注册时间）分组，比较不同队列在相同生命阶段的行为差异。


队列分析模型
收入队列分析
WITH user_cohort AS (
    SELECT
        user_id,
        DATE_TRUNC('month', MIN(created_at)) AS cohort_month
    FROM users
    GROUP BY user_id
),
monthly_revenue AS (
    SELECT
        uc.cohort_month,
        DATE_TRUNC('month', o.order_date) AS revenue_month,
        EXTRACT(YEAR FROM AGE(DATE_TRUNC('month', o.order_date), uc.cohort_month)) * 12
            + EXTRACT(MONTH FROM AGE(DATE_TRUNC('month', o.order_date), uc.cohort_month)) AS months_since_cohort,
        SUM(o.amount) AS revenue,
        COUNT(DISTINCT uc.user_id) AS paying_users
    FROM user_cohort uc
    INNER JOIN orders o ON uc.user_id = o.user_id
    GROUP BY uc.cohort_month, DATE_TRUNC('month', o.order_date)
),
cohort_sizes AS (
    SELECT cohort_month, COUNT(*) AS cohort_size
    FROM user_cohort
    GROUP BY cohort_month
)
SELECT
    mr.cohort_month,
    cs.cohort_size,
    mr.months_since_cohort,
    mr.revenue,
    mr.paying_users,
    ROUND(mr.revenue / cs.cohort_size, 2) AS revenue_per_user,
    ROUND(mr.paying_users::NUMERIC / cs.cohort_size * 100, 2) AS paying_rate
FROM monthly_revenue mr
JOIN cohort_sizes cs ON mr.cohort_month = cs.cohort_month
WHERE mr.months_since_cohort BETWEEN 0 AND 11
ORDER BY mr.cohort_month, mr.months_since_cohort;
行为队列分析
WITH user_first_action AS (
    SELECT
        user_id,
        CASE
            WHEN first_event = 'search' THEN '搜索型'
            WHEN first_event = 'browse_category' THEN '浏览型'
            WHEN first_event = 'direct_product' THEN '目标型'
            ELSE '其他'
        END AS behavior_cohort
    FROM (
        SELECT
            user_id,
            FIRST_VALUE(event_name) OVER (
                PARTITION BY user_id ORDER BY event_time
            ) AS first_event
        FROM user_events
    ) sub
    GROUP BY user_id, first_event
),
cohort_metrics AS (
    SELECT
        ufa.behavior_cohort,
        COUNT(DISTINCT ufa.user_id) AS cohort_size,
        COUNT(DISTINCT o.user_id) AS converters,
        SUM(o.amount) AS total_revenue,
        AVG(o.amount) AS avg_order_value
    FROM user_first_action ufa
    LEFT JOIN orders o ON ufa.user_id = o.user_id
        AND o.order_date <= ufa.first_date + INTERVAL '30 days'
    GROUP BY ufa.behavior_cohort
)
SELECT
    behavior_cohort,
    cohort_size,
    converters,
    ROUND(converters::NUMERIC / cohort_size * 100, 2) AS conversion_rate,
    ROUND(total_revenue / cohort_size, 2) AS revenue_per_user,
    ROUND(avg_order_value, 2) AS avg_order_value
FROM cohort_metrics
ORDER BY conversion_rate DESC;


高级分析模式
RFM 分析
RFM（Recency-Frequency-Monetary）模型是用户价值分层的经典方法。


WITH rfm_raw AS (
    SELECT
        user_id,
        CURRENT_DATE - MAX(DATE(order_date)) AS recency_days,
        COUNT(DISTINCT order_id) AS frequency,
        SUM(amount) AS monetary
    FROM orders
    WHERE order_date >= CURRENT_DATE - INTERVAL '1 year'
    GROUP BY user_id
),
rfm_scores AS (
    SELECT
        user_id,
        recency_days,
        frequency,
        monetary,
        NTILE(5) OVER (ORDER BY recency_days ASC) AS r_score,
        NTILE(5) OVER (ORDER BY frequency DESC) AS f_score,
        NTILE(5) OVER (ORDER BY monetary DESC) AS m_score
    FROM rfm_raw
)
SELECT
    user_id,
    r_score, f_score, m_score,
    CASE
        WHEN r_score >= 4 AND f_score >= 4 AND m_score >= 4 THEN '高价值活跃'
        WHEN r_score >= 4 AND f_score >= 4 THEN '忠实用户'
        WHEN r_score >= 4 AND m_score >= 4 THEN '高消费新客'
        WHEN r_score <= 2 AND f_score >= 4 THEN '流失风险'
        WHEN r_score <= 2 AND f_score <= 2 THEN '已流失'
        ELSE '一般用户'
    END AS user_segment
FROM rfm_scores
ORDER BY r_score DESC, f_score DESC, m_score DESC;
用户生命周期价值（LTV）估算
WITH monthly_cohort_revenue AS (
    SELECT
        DATE_TRUNC('month', u.created_at) AS cohort_month,
        EXTRACT(YEAR FROM AGE(DATE_TRUNC('month', o.order_date), DATE_TRUNC('month', u.created_at))) * 12
            + EXTRACT(MONTH FROM AGE(DATE_TRUNC('month', o.order_date), DATE_TRUNC('month', u.created_at))) AS month_number,
        SUM(o.amount) AS revenue,
        COUNT(DISTINCT u.user_id) AS active_users
    FROM users u
    LEFT JOIN orders o ON u.user_id = o.user_id
    GROUP BY cohort_month, month_number
),
cumulative_revenue AS (
    SELECT
        cohort_month,
        month_number,
        revenue,
        SUM(revenue) OVER (
            PARTITION BY cohort_month ORDER BY month_number
        ) AS cumulative_revenue,
        (SELECT COUNT(*) FROM users
         WHERE DATE_TRUNC('month', created_at) = mcr.cohort_month) AS cohort_size
    FROM monthly_cohort_revenue mcr
)
SELECT
    cohort_month,
    month_number,
    cumulative_revenue,
    cohort_size,
    ROUND(cumulative_revenue / cohort_size, 2) AS ltv_per_user
FROM cumulative_revenue
WHERE month_number <= 12
ORDER BY cohort_month, month_number;
活跃用户趋势分析
WITH daily_active AS (
    SELECT
        DATE(event_time) AS date,
        COUNT(DISTINCT user_id) AS dau
    FROM user_events
    WHERE event_time >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY DATE(event_time)
),
weekly_active AS (
    SELECT
        DATE_TRUNC('week', event_time)::DATE AS week_start,
        COUNT(DISTINCT user_id) AS wau
    FROM user_events
    WHERE event_time >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY DATE_TRUNC('week', event_time)
),
monthly_active AS (
    SELECT
        DATE_TRUNC('month', event_time)::DATE AS month,
        COUNT(DISTINCT user_id) AS mau
    FROM user_events
    WHERE event_time >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY DATE_TRUNC('month', event_time)
)
SELECT
    da.date,
    da.dau,
    wa.wau,
    ma.mau,
    ROUND(da.dau::NUMERIC / NULLIF(ma.mau, 0), 4) AS dau_mau_ratio
FROM daily_active da
LEFT JOIN weekly_active wa ON da.date = wa.week_start
LEFT JOIN monthly_active ma ON DATE_TRUNC('month', da.date)::DATE = ma.month
ORDER BY da.date;


分析 SQL 编写最佳实践




实践
说明





CTE 分层构建
每个 CTE 完成一个逻辑步骤，便于调试和复用




明确时间窗口
所有分析查询都应有明确的时间范围限制




DISTINCT 谨慎使用
只在确实需要去重时使用，避免隐藏数据问题




NULL 处理
使用 COALESCE 处理 NULL，避免计算偏差




数据类型一致
JOIN 和 WHERE 条件中确保类型匹配




增量计算
大数据量场景优先使用增量计算而非全量重算







小结
    

  • 留存分析通过比较首次活跃日期与后续活跃日期的差值计算留存率，留存矩阵提供完整的留存全景图
    • 

  • 漏斗分析追踪预定义步骤序列中的转化情况，严格有序漏斗和窗口函数漏斗各有优劣
    • 

  • 队列分析按共同特征将用户分组，比较不同群组在相同生命阶段的行为差异
    • 

  • RFM 模型通过最近消费时间、消费频率和消费金额三个维度对用户进行价值分层
    • 

  • 所有分析 SQL 都应以 CTE 分层构建、明确时间窗口、谨慎处理 NULL 值
    •
