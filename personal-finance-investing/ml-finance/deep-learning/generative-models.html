生成模型与金融数据合成

概述

金融领域的数据稀缺性是一个结构性难题：极端市场事件（如 2008 年金融危机、2020 年 COVID 崩盘）样本稀少但至关重要；合规和隐私约束限制了真实数据的共享和使用；新兴市场和另类资产的历史数据有限。生成模型（Generative Models）通过学习真实数据的分布特征，能够合成具有统计真实性的金融数据，为模型训练、风险压力测试和隐私保护提供全新工具。

本文系统梳理变分自编码器（VAE）、生成对抗网络（GAN）、扩散模型（Diffusion Model）在金融数据合成中的原理、架构与实践。

变分自编码器（VAE）
基本原理
VAE 将高维金融数据映射到低维潜空间，学习数据的隐含表示：


L=Eq(z∣x)[log⁡p(x∣z)]−DKL(q(z∣x)∥p(z))\mathcal{L} = \mathbb{E}_{q(z|x)}[\log p(x|z)] - D_{KL}(q(z|x) \| p(z))L=Eq(z∣x)​[logp(x∣z)]−DKL​(q(z∣x)∥p(z))
第一项是重构损失（生成质量），第二项是 KL 散度（正则化潜空间）。


金融时序 VAE
import torch
import torch.nn as nn

class FinancialTimeSeriesVAE(nn.Module):
    def __init__(self, seq_len, input_dim, latent_dim=32, hidden_dim=128):
        super().__init__()
        self.seq_len = seq_len
        self.input_dim = input_dim
        self.latent_dim = latent_dim

        self.encoder = nn.LSTM(
            input_dim, hidden_dim, num_layers=2, batch_first=True
        )
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)

        self.decoder_input = nn.Linear(latent_dim, hidden_dim)
        self.decoder = nn.LSTM(
            hidden_dim, hidden_dim, num_layers=2, batch_first=True
        )
        self.output_layer = nn.Linear(hidden_dim, input_dim)

    def encode(self, x):
        _, (h, _) = self.encoder(x)
        h = h[-1]
        return self.fc_mu(h), self.fc_logvar(h)

    def reparameterize(self, mu, logvar):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + eps * std

    def decode(self, z):
        h = self.decoder_input(z)
        h = h.unsqueeze(1).repeat(1, self.seq_len, 1)
        output, _ = self.decoder(h)
        return self.output_layer(output)

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        recon = self.decode(z)
        return recon, mu, logvar

    def generate(self, n_samples):
        z = torch.randn(n_samples, self.latent_dim)
        return self.decode(z)

    def loss_function(self, recon, x, mu, logvar, beta=1.0):
        recon_loss = nn.functional.mse_loss(recon, x, reduction="sum")
        kl_loss = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp())
        return recon_loss + beta * kl_loss

class ConditionalFinanceVAE(nn.Module):
    def __init__(self, seq_len, input_dim, condition_dim, latent_dim=32,
                 hidden_dim=128):
        super().__init__()

        self.encoder = nn.LSTM(
            input_dim + condition_dim, hidden_dim, num_layers=2,
            batch_first=True
        )
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)

        self.decoder_input = nn.Linear(latent_dim + condition_dim, hidden_dim)
        self.decoder = nn.LSTM(
            hidden_dim, hidden_dim, num_layers=2, batch_first=True
        )
        self.output_layer = nn.Linear(hidden_dim, input_dim)
        self.seq_len = seq_len

    def encode(self, x, c):
        c_expanded = c.unsqueeze(1).expand(-1, x.size(1), -1)
        xc = torch.cat([x, c_expanded], dim=-1)
        _, (h, _) = self.encoder(xc)
        h = h[-1]
        return self.fc_mu(h), self.fc_logvar(h)

    def decode(self, z, c):
        zc = torch.cat([z, c], dim=-1)
        h = self.decoder_input(zc)
        h = h.unsqueeze(1).repeat(1, self.seq_len, 1)
        output, _ = self.decoder(h)
        return self.output_layer(output)

    def generate(self, n_samples, condition):
        z = torch.randn(n_samples, self.fc_mu.out_features)
        return self.decode(z, condition)

class TimeGANGenerator(nn.Module):
    def __init__(self, noise_dim, hidden_dim, seq_len):
        super().__init__()
        self.rnn = nn.GRU(noise_dim, hidden_dim, num_layers=3,
                          batch_first=True)
        self.fc = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.Sigmoid()
        )

    def forward(self, z):
        h, _ = self.rnn(z)
        return self.fc(h)

class TimeGANDiscriminator(nn.Module):
    def __init__(self, hidden_dim):
        super().__init__()
        self.rnn = nn.GRU(hidden_dim, hidden_dim, num_layers=3,
                          batch_first=True)
        self.fc = nn.Linear(hidden_dim, 1)

    def forward(self, h):
        output, _ = self.rnn(h)
        return self.fc(output[:, -1, :])

def financial_gan_loss(real_data, fake_data, d_loss, lambda_moment=1.0,
                       lambda_autocorr=0.5, lambda_corr=0.5):
    moment_loss = compute_moment_loss(real_data, fake_data)

    autocorr_loss = compute_autocorrelation_loss(real_data, fake_data, lags=10)

    corr_loss = compute_correlation_loss(real_data, fake_data)

    total = d_loss + lambda_moment * moment_loss + \
            lambda_autocorr * autocorr_loss + lambda_corr * corr_loss

    return total

def compute_moment_loss(real, fake):
    loss = 0
    for k in range(1, 5):
        real_moment = real.pow(k).mean(dim=0)
        fake_moment = fake.pow(k).mean(dim=0)
        loss += (real_moment - fake_moment).pow(2).mean()
    return loss

def compute_autocorrelation_loss(real, fake, lags=10):
    loss = 0
    for lag in range(1, lags + 1):
        real_ac = torch.corrcoef(
            torch.stack([real[:, :-lag].flatten(), real[:, lag:].flatten()])
        )[0, 1]
        fake_ac = torch.corrcoef(
            torch.stack([fake[:, :-lag].flatten(), fake[:, lag:].flatten()])
        )[0, 1]
        loss += (real_ac - fake_ac).pow(2)
    return loss

def compute_correlation_loss(real, fake):
    real_corr = torch.corrcoef(real.T)
    fake_corr = torch.corrcoef(fake.T)
    return (real_corr - fake_corr).pow(2).mean()
扩散模型
金融扩散模型
扩散模型（Denoising Diffusion Probabilistic Model, DDPM）通过逐步加噪和去噪过程生成数据，在图像生成领域取得了惊人成果，近年来也被引入金融数据合成。


前向过程（加噪）：


q(xt∣xt−1)=N(xt;1−βtxt−1,βtI)q(x_t | x_{t-1}) = \mathcal{N}(x_t; \sqrt{1-\beta_t}x_{t-1}, \beta_t I)q(xt​∣xt−1​)=N(xt​;1−βt​​xt−1​,βt​I)
反向过程（去噪/生成）：


pθ(xt−1∣xt)=N(xt−1;μθ(xt,t),σt2I)p_\theta(x_{t-1} | x_t) = \mathcal{N}(x_{t-1}; \mu_\theta(x_t, t), \sigma_t^2 I)pθ​(xt−1​∣xt​)=N(xt−1​;μθ​(xt​,t),σt2​I)
class FinancialDiffusionModel(nn.Module):
    def __init__(self, input_dim, hidden_dim=256, n_steps=1000):
        super().__init__()
        self.n_steps = n_steps

        betas = torch.linspace(1e-4, 0.02, n_steps)
        alphas = 1.0 - betas
        alphas_cumprod = torch.cumprod(alphas, dim=0)
        self.register_buffer("betas", betas)
        self.register_buffer("alphas_cumprod", alphas_cumprod)
        self.register_buffer("sqrt_alphas_cumprod", torch.sqrt(alphas_cumprod))
        self.register_buffer("sqrt_one_minus_alphas_cumprod",
                             torch.sqrt(1.0 - alphas_cumprod))

        self.noise_predictor = nn.Sequential(
            nn.Linear(input_dim + 1, hidden_dim),
            nn.SiLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.SiLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.SiLU(),
            nn.Linear(hidden_dim, input_dim)
        )

    def forward_diffusion(self, x_0, t):
        noise = torch.randn_like(x_0)
        sqrt_alpha = self.sqrt_alphas_cumprod[t].unsqueeze(-1)
        sqrt_one_minus_alpha = self.sqrt_one_minus_alphas_cumprod[t].unsqueeze(-1)
        x_t = sqrt_alpha * x_0 + sqrt_one_minus_alpha * noise
        return x_t, noise

    def predict_noise(self, x_t, t):
        t_embed = t.float().unsqueeze(-1) / self.n_steps
        x_input = torch.cat([x_t, t_embed], dim=-1)
        return self.noise_predictor(x_input)

    @torch.no_grad()
    def sample(self, n_samples, input_dim):
        x = torch.randn(n_samples, input_dim)
        for t in reversed(range(self.n_steps)):
            t_batch = torch.full((n_samples,), t, dtype=torch.long)
            predicted_noise = self.predict_noise(x, t_batch)

            alpha = 1.0 - self.betas[t]
            alpha_cumprod = self.alphas_cumprod[t]
            beta = self.betas[t]

            mean = (1 / torch.sqrt(alpha)) * (
                x - (beta / torch.sqrt(1 - alpha_cumprod)) * predicted_noise
            )

            if t > 0:
                noise = torch.randn_like(x)
                x = mean + torch.sqrt(beta) * noise
            else:
                x = mean

        return x

from scipy.stats import ks_2samp
from sklearn.metrics import mean_squared_error

def evaluate_synthetic_quality(real_data, synthetic_data):
    results = {}

    ks_stats = []
    for col in range(real_data.shape[1]):
        stat, p = ks_2samp(real_data[:, col], synthetic_data[:, col])
        ks_stats.append({"column": col, "ks_stat": stat, "p_value": p})
    results["marginal_ks"] = ks_stats

    for k in range(1, 5):
        real_moment = np.mean(real_data**k, axis=0)
        synth_moment = np.mean(synthetic_data**k, axis=0)
        results[f"moment_{k}_mse"] = mean_squared_error(
            real_moment, synth_moment
        )

    real_corr = np.corrcoef(real_data.T)
    synth_corr = np.corrcoef(synthetic_data.T)
    results["correlation_frobenius"] = np.linalg.norm(
        real_corr - synth_corr, "fro"
    )

    return results

总结
生成模型为金融领域提供了"数据创造"的能力——从数据增强到压力测试，从隐私保护到市场模拟。VAE 提供了结构化的潜空间表示，适合数据填补和条件生成；GAN 在分布匹配上表现最强，TimeGAN 等变体针对时序特性做了专门优化；扩散模型在生成质量上已超越 GAN，且训练更稳定，但采样速度是其在实时应用中的瓶颈。核心挑战在于金融约束的嵌入——合成的金融数据不仅需要统计真实性（分布匹配），还需要经济合理性（无套利、因果一致、尾部行为），这要求生成模型的损失函数深度融合金融领域知识。
