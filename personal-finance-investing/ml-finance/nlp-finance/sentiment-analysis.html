市场情绪分析

概述

市场情绪（Market Sentiment）是驱动资产价格短期波动的核心力量之一。行为金融学表明，投资者情绪偏差（过度乐观、恐慌抛售、羊群效应）系统性地影响市场定价。情绪分析（Sentiment Analysis）利用 NLP 技术从新闻、社交媒体、研报等文本数据中量化提取情绪信号，将"市场感觉"转化为可计算、可回测的交易因子。

本文系统梳理金融情绪分析的数据源、模型方法、因子构建与应用实践。

情绪分析模型
词典方法
基于预定义情感词典的规则方法，简单快速：


class FinancialSentimentLexicon:
    def __init__(self):
        self.positive_words = {
            "surge", "rally", "growth", "profit", "upgrade",
            "outperform", "breakthrough", "record", "bullish",
            "上涨", "利好", "增长", "突破", "看多", "强劲"
        }
        self.negative_words = {
            "plunge", "crash", "loss", "downgrade", "risk",
            "default", "recession", "bearish", "warning",
            "下跌", "利空", "亏损", "风险", "暴跌", "看空"
        }
        self.intensifiers = {
            "very", "extremely", "significantly", "sharply",
            "大幅", "急剧", "显著"
        }
        self.negations = {
            "not", "no", "never", "neither", "不", "没有", "未"
        }

    def score(self, tokens):
        score = 0.0
        negate = False

        for i, token in enumerate(tokens):
            if token.lower() in self.negations:
                negate = True
                continue

            multiplier = 1.0
            if i > 0 and tokens[i-1].lower() in self.intensifiers:
                multiplier = 1.5

            if token.lower() in self.positive_words:
                s = 1.0 * multiplier
            elif token.lower() in self.negative_words:
                s = -1.0 * multiplier
            else:
                negate = False
                continue

            if negate:
                s = -s
                negate = False

            score += s

        return score / max(len(tokens), 1)
Loughran-McDonald 金融情感词典是金融领域最权威的词典资源，包含约 2,700 个正面词和 354 个负面词，专门针对 10-K 报告等金融文本设计。


深度学习方法
FinBERT
FinBERT 是在金融语料上微调的 BERT 模型，是当前金融情绪分析的标准选择：


from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch

class FinBERTSentimentAnalyzer:
    def __init__(self, model_name="ProsusAI/finbert"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForSequenceClassification.from_pretrained(
            model_name
        )
        self.model.eval()
        self.labels = ["positive", "negative", "neutral"]

    def analyze(self, texts):
        inputs = self.tokenizer(
            texts, padding=True, truncation=True,
            max_length=512, return_tensors="pt"
        )

        with torch.no_grad():
            outputs = self.model(**inputs)
            probs = torch.softmax(outputs.logits, dim=-1)

        results = []
        for i, text in enumerate(texts):
            label_idx = probs[i].argmax().item()
            results.append({
                "text": text[:100],
                "label": self.labels[label_idx],
                "confidence": probs[i][label_idx].item(),
                "positive_prob": probs[i][0].item(),
                "negative_prob": probs[i][1].item(),
                "neutral_prob": probs[i][2].item(),
                "sentiment_score": (
                    probs[i][0].item() - probs[i][1].item()
                )
            })

        return results

    def batch_analyze(self, texts, batch_size=32):
        all_results = []
        for i in range(0, len(texts), batch_size):
            batch = texts[i:i + batch_size]
            results = self.analyze(batch)
            all_results.extend(results)
        return all_results

情绪因子构建
个股情绪因子
import pandas as pd
import numpy as np

def build_stock_sentiment_factor(news_sentiments, stock_mapping,
                                  decay_halflife=3):
    factor = pd.DataFrame()

    for stock_id, stock_news in news_sentiments.groupby("stock_id"):
        daily_sentiment = stock_news.groupby(
            stock_news["publish_time"].dt.date
        ).agg({
            "sentiment_score": ["mean", "std", "count"],
            "positive_prob": "mean",
            "negative_prob": "mean"
        })

        daily_sentiment.columns = [
            "sent_mean", "sent_std", "news_count",
            "pos_ratio", "neg_ratio"
        ]

        weights = np.exp(
            -np.log(2) * np.arange(len(daily_sentiment)) / decay_halflife
        )[::-1]

        for col in ["sent_mean", "pos_ratio", "neg_ratio"]:
            daily_sentiment[f"{col}_ewm"] = (
                daily_sentiment[col]
                .ewm(halflife=decay_halflife)
                .mean()
            )

        daily_sentiment["sentiment_momentum"] = (
            daily_sentiment["sent_mean_ewm"]
            - daily_sentiment["sent_mean_ewm"].shift(5)
        )

        daily_sentiment["attention_surprise"] = (
            daily_sentiment["news_count"]
            / daily_sentiment["news_count"].rolling(20).mean()
        )

        daily_sentiment["stock_id"] = stock_id
        factor = pd.concat([factor, daily_sentiment])

    return factor
市场情绪指数
将个股情绪聚合为市场级别的情绪指标：


def build_market_sentiment_index(stock_sentiments, market_caps):
    daily_index = pd.DataFrame()

    for date in stock_sentiments.index.get_level_values("date").unique():
        day_data = stock_sentiments.loc[date]
        caps = market_caps.loc[date]

        common = day_data.index.intersection(caps.index)
        if len(common) < 50:
            continue

        weights = caps[common] / caps[common].sum()

        weighted_sentiment = (
            day_data.loc[common, "sent_mean"] * weights
        ).sum()

        breadth_positive = (
            day_data.loc[common, "sent_mean"] > 0
        ).mean()

        daily_index.loc[date, "market_sentiment"] = weighted_sentiment
        daily_index.loc[date, "sentiment_breadth"] = breadth_positive
        daily_index.loc[date, "extreme_negative_ratio"] = (
            day_data.loc[common, "sent_mean"] < -0.5
        ).mean()

    return daily_index

与传统因子的互补性
情绪因子与动量、价值等传统因子的相关性通常较低（< 0.2），具有增量信息价值：


def analyze_factor_complementarity(sentiment_factor, traditional_factors,
                                    returns):
    from sklearn.linear_model import LinearRegression

    residual_ic_list = []
    for date in returns.index:
        if date not in sentiment_factor.index:
            continue

        trad = traditional_factors.loc[date].dropna()
        sent = sentiment_factor.loc[date].dropna()
        ret = returns.loc[date].dropna()

        common = trad.index.intersection(sent.index).intersection(ret.index)
        if len(common) < 50:
            continue

        reg = LinearRegression()
        reg.fit(trad.loc[common].values.reshape(-1, 1), sent.loc[common])
        residual = sent.loc[common] - reg.predict(
            trad.loc[common].values.reshape(-1, 1)
        )

        from scipy.stats import spearmanr
        ic, _ = spearmanr(residual, ret.loc[common])
        residual_ic_list.append(ic)

    return np.mean(residual_ic_list)
实时情绪监控系统
异常检测与交易信号
情绪的异常变化（突然恶化或改善）比绝对水平更具预测价值：


def detect_sentiment_anomaly(sentiment_series, window=20, threshold=2.5):
    rolling_mean = sentiment_series.rolling(window).mean()
    rolling_std = sentiment_series.rolling(window).std()

    z_score = (sentiment_series - rolling_mean) / rolling_std

    anomalies = pd.DataFrame({
        "sentiment": sentiment_series,
        "z_score": z_score,
        "is_anomaly": z_score.abs() > threshold,
        "direction": np.where(z_score > threshold, "positive_shock",
                    np.where(z_score < -threshold, "negative_shock", "normal"))
    })

    return anomalies

总结
市场情绪分析是将非结构化文本信息转化为量化交易信号的桥梁。从词典方法到 FinBERT，技术手段不断进化，但核心逻辑不变——情绪的变化比绝对水平更重要，异常比常态更有价值。情绪因子作为量化策略中的"另类 alpha"，与传统因子具有良好的互补性，尤其在短期预测和事件驱动场景中表现突出。构建有效的情绪因子需要在数据质量（去噪、去重、实体链接）和因子设计（衰减权重、截面标准化、动量构造）两端同时下功夫。
