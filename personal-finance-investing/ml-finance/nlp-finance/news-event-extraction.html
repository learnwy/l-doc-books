新闻事件提取与因果推断

概述

金融市场对事件高度敏感——一则央行加息消息可以在数秒内引发全球资产价格重定价，一起企业丑闻可以瞬间蒸发数十亿市值。新闻事件提取（Event Extraction）从海量非结构化文本中自动识别、分类和结构化金融事件，而因果推断（Causal Inference）则进一步分析事件之间的因果链条，构建从事件到价格影响的传导模型。

本文系统梳理金融事件提取的技术管线——从命名实体识别、关系抽取到事件检测与因果图构建，以及在事件驱动投资策略中的实际应用。

from transformers import pipeline

class FinancialNER:
    def __init__(self):
        self.ner_pipeline = pipeline(
            "ner",
            model="Jean-Baptiste/camembert-ner-with-dates",
            aggregation_strategy="simple"
        )
        self.financial_patterns = self._build_patterns()

    def _build_patterns(self):
        import re
        return {
            "MONEY": re.compile(
                r"[\$€¥£]\s*[\d,.]+\s*[万亿百千]?[美元|欧元|日元|英镑|人民币]?"
            ),
            "PERCENTAGE": re.compile(r"[\d.]+\s*[%‰个百分点基点]"),
            "STOCK_CODE": re.compile(r"[A-Z]{1,5}\.\w{1,3}|\d{6}\.[A-Z]{2}"),
            "DATE_REL": re.compile(r"[上下本]?[周月季年][一二三四末初]?")
        }

    def extract(self, text):
        ner_results = self.ner_pipeline(text)

        for pattern_name, pattern in self.financial_patterns.items():
            for match in pattern.finditer(text):
                ner_results.append({
                    "entity_group": pattern_name,
                    "word": match.group(),
                    "start": match.start(),
                    "end": match.end(),
                    "score": 0.95
                })

        return sorted(ner_results, key=lambda x: x["start"])
实体链接与消歧
金融文本中"苹果"可能指苹果公司也可能指苹果水果，"中国银行"和"中行"需要映射到同一实体：


class EntityLinker:
    def __init__(self, entity_kb):
        self.kb = entity_kb
        self.alias_map = self._build_alias_map()

    def _build_alias_map(self):
        alias_map = {}
        for entity_id, entity in self.kb.items():
            for alias in entity.get("aliases", []):
                alias_map[alias.lower()] = entity_id
            alias_map[entity["name"].lower()] = entity_id
        return alias_map

    def link(self, entity_mention, context):
        mention_lower = entity_mention.lower()

        if mention_lower in self.alias_map:
            candidates = [self.alias_map[mention_lower]]
        else:
            candidates = self._fuzzy_match(mention_lower)

        if len(candidates) == 1:
            return candidates[0]

        return self._disambiguate(entity_mention, candidates, context)

    def _disambiguate(self, mention, candidates, context):
        best_score = -1
        best_entity = None

        for cand_id in candidates:
            entity = self.kb[cand_id]
            score = self._context_similarity(
                context, entity.get("description", "")
            )
            if score > best_score:
                best_score = score
                best_entity = cand_id

        return best_entity

    def _fuzzy_match(self, mention):
        from difflib import get_close_matches
        matches = get_close_matches(mention, self.alias_map.keys(), n=5)
        return [self.alias_map[m] for m in matches]

    def _context_similarity(self, context, description):
        context_words = set(context.lower().split())
        desc_words = set(description.lower().split())
        if not desc_words:
            return 0
        return len(context_words & desc_words) / len(desc_words)

from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch

class FinancialEventClassifier:
    EVENT_TYPES = [
        "earnings_release", "merger_acquisition", "management_change",
        "regulatory_action", "product_launch", "dividend_announcement",
        "share_buyback", "debt_issuance", "rating_change",
        "macro_data_release", "policy_change", "litigation",
        "partnership", "restructuring", "ipo_listing"
    ]

    def __init__(self, model_path):
        self.tokenizer = AutoTokenizer.from_pretrained(model_path)
        self.model = AutoModelForSequenceClassification.from_pretrained(
            model_path
        )
        self.model.eval()

    def classify(self, texts):
        inputs = self.tokenizer(
            texts, padding=True, truncation=True,
            max_length=256, return_tensors="pt"
        )

        with torch.no_grad():
            outputs = self.model(**inputs)
            probs = torch.sigmoid(outputs.logits)

        results = []
        for i, text in enumerate(texts):
            events = []
            for j, event_type in enumerate(self.EVENT_TYPES):
                if probs[i][j] > 0.5:
                    events.append({
                        "type": event_type,
                        "confidence": probs[i][j].item()
                    })
            results.append({
                "text": text[:200],
                "events": sorted(events, key=lambda x: -x["confidence"])
            })

        return results

class CausalRelationExtractor:
    def __init__(self, llm_client):
        self.llm = llm_client

    def extract_causal_relations(self, news_text):
        prompt = f"""分析以下金融新闻，提取所有因果关系对。

新闻：{news_text}

请以JSON格式输出，每个因果关系包含：
- cause: 原因事件
- effect: 结果事件
- confidence: 置信度(0-1)
- mechanism: 传导机制简述

仅输出JSON数组，不要其他文字。"""

        response = self.llm.generate(prompt)
        return self._parse_response(response)

    def build_causal_graph(self, causal_pairs):
        import networkx as nx
        G = nx.DiGraph()

        for pair in causal_pairs:
            G.add_edge(
                pair["cause"],
                pair["effect"],
                weight=pair["confidence"],
                mechanism=pair.get("mechanism", "")
            )

        return G

    def find_impact_path(self, graph, source_event, target_asset):
        import networkx as nx
        try:
            paths = list(nx.all_simple_paths(
                graph, source_event, target_asset, cutoff=5
            ))
            return paths
        except nx.NetworkXError:
            return []

    def _parse_response(self, response):
        import json
        try:
            return json.loads(response)
        except json.JSONDecodeError:
            return []

def quantify_event_impact(events_df, returns_df, windows=[1, 3, 5, 10, 20]):
    impact_stats = {}

    for event_type in events_df["event_type"].unique():
        type_events = events_df[events_df["event_type"] == event_type]
        impacts = []

        for _, event in type_events.iterrows():
            stock = event["stock_id"]
            date = event["event_date"]

            if stock not in returns_df.columns:
                continue
            if date not in returns_df.index:
                continue

            date_idx = returns_df.index.get_loc(date)

            impact = {}
            for w in windows:
                end_idx = min(date_idx + w, len(returns_df) - 1)
                cum_return = returns_df.iloc[date_idx:end_idx + 1][stock].sum()
                impact[f"car_{w}d"] = cum_return

            pre_vol = returns_df.iloc[max(0, date_idx-20):date_idx][stock].std()
            impact["pre_vol"] = pre_vol
            impacts.append(impact)

        if impacts:
            impact_df = pd.DataFrame(impacts)
            impact_stats[event_type] = {
                "count": len(impacts),
                "mean_car_1d": impact_df["car_1d"].mean(),
                "mean_car_5d": impact_df["car_5d"].mean(),
                "mean_car_20d": impact_df["car_20d"].mean(),
                "positive_ratio_1d": (impact_df["car_1d"] > 0).mean(),
                "t_stat_1d": (
                    impact_df["car_1d"].mean()
                    / (impact_df["car_1d"].std() / np.sqrt(len(impacts)))
                )
            }

    return pd.DataFrame(impact_stats).T

事件衰减模型
事件对价格的影响随时间衰减，衰减速度因事件类型而异：


def event_decay_signal(event_date, current_date, event_type,
                       decay_params=None):
    if decay_params is None:
        decay_params = {
            "earnings_release": {"halflife": 5, "max_impact": 0.8},
            "merger_acquisition": {"halflife": 20, "max_impact": 1.0},
            "management_change": {"halflife": 10, "max_impact": 0.6},
            "regulatory_action": {"halflife": 15, "max_impact": 0.9},
            "rating_change": {"halflife": 7, "max_impact": 0.7},
        }

    params = decay_params.get(
        event_type, {"halflife": 10, "max_impact": 0.5}
    )

    days_elapsed = (current_date - event_date).days
    decay_factor = np.exp(-np.log(2) * days_elapsed / params["halflife"])

    return params["max_impact"] * decay_factor

总结
新闻事件提取与因果推断是金融 NLP 中最具"alpha 生成"潜力的方向。通过结构化地理解"发生了什么"（事件检测）、"为什么重要"（影响量化）和"会怎样传导"（因果推断），我们可以在信息发布后的极短时间内生成交易信号。关键挑战在于事件提取的准确性和实时性——假阳性事件会导致错误交易，而延迟数分钟就可能错过最佳交易窗口。未来方向是将 LLM 的理解能力与知识图谱的结构化推理相结合，构建"理解→推理→行动"的端到端事件驱动系统。
