金融特征工程

概述

特征工程（Feature Engineering）是金融机器学习中投入产出比最高的环节。在信噪比极低的金融数据中，模型性能的 80% 取决于特征质量而非算法选择。一个优秀的特征能将分散在原始数据中的微弱信号聚合放大为可预测的 alpha 因子，而一个泄露未来信息的特征则会制造虚幻的回测收益。

本文系统梳理金融特征工程的全流程：从原始数据源的特征提取，经因子构造的方法论框架，到特征筛选与组合的工程实践，最终建立从数据到信号的完整管线。

动量类因子
动量效应（Momentum）是金融市场中最持久的异象之一——过去表现好的资产倾向于继续表现好（中期），过去表现差的资产倾向于反转（短期/长期）。


import pandas as pd
import numpy as np

def compute_momentum_factors(df):
    factors = pd.DataFrame(index=df.index)

    for period in [5, 10, 20, 60, 120, 250]:
        factors[f"mom_{period}d"] = df["close"].pct_change(period)

    factors["mom_12m_1m"] = (
        df["close"].pct_change(250) - df["close"].pct_change(20)
    )

    for fast, slow in [(5, 20), (10, 60), (20, 120)]:
        ma_fast = df["close"].rolling(fast).mean()
        ma_slow = df["close"].rolling(slow).mean()
        factors[f"ma_cross_{fast}_{slow}"] = (ma_fast - ma_slow) / ma_slow

    for w in [14, 28]:
        delta = df["close"].diff()
        gain = delta.where(delta > 0, 0).rolling(w).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(w).mean()
        rs = gain / loss
        factors[f"rsi_{w}"] = 100 - (100 / (1 + rs))

    for w in [12, 26]:
        ema = df["close"].ewm(span=w, adjust=False).mean()
        factors[f"ema_{w}"] = (df["close"] - ema) / ema

    macd_fast = df["close"].ewm(span=12, adjust=False).mean()
    macd_slow = df["close"].ewm(span=26, adjust=False).mean()
    macd_line = macd_fast - macd_slow
    signal_line = macd_line.ewm(span=9, adjust=False).mean()
    factors["macd_hist"] = macd_line - signal_line

    return factors

def compute_volatility_factors(df, windows=[5, 10, 20, 60]):
    factors = pd.DataFrame(index=df.index)

    for w in windows:
        returns = df["close"].pct_change()
        factors[f"realized_vol_{w}d"] = returns.rolling(w).std() * np.sqrt(252)

        log_hl = np.log(df["high"] / df["low"])
        factors[f"parkinson_vol_{w}d"] = (
            np.sqrt(log_hl.pow(2).rolling(w).mean() / (4 * np.log(2))) * np.sqrt(252)
        )

        up_returns = returns.where(returns > 0, 0)
        down_returns = returns.where(returns < 0, 0)
        factors[f"vol_skew_{w}d"] = (
            up_returns.rolling(w).std() / down_returns.abs().rolling(w).std()
        )

    factors["vol_term_structure"] = (
        factors["realized_vol_5d"] / factors["realized_vol_60d"]
    )

    return factors

def compute_liquidity_factors(df, windows=[5, 20, 60]):
    factors = pd.DataFrame(index=df.index)

    for w in windows:
        abs_ret = df["close"].pct_change().abs()
        dollar_volume = df["volume"] * df["close"]
        amihud = abs_ret / dollar_volume
        factors[f"amihud_{w}d"] = amihud.rolling(w).mean()

        factors[f"turnover_{w}d"] = df["volume"].rolling(w).mean()
        factors[f"volume_cv_{w}d"] = (
            df["volume"].rolling(w).std() / df["volume"].rolling(w).mean()
        )

    factors["kyle_lambda"] = estimate_kyle_lambda(df)

    return factors

def estimate_kyle_lambda(df, window=20):
    returns = df["close"].pct_change()
    signed_volume = df["volume"] * np.sign(returns)
    lambda_series = pd.Series(index=df.index, dtype=float)

    for i in range(window, len(df)):
        r = returns.iloc[i - window:i]
        sv = signed_volume.iloc[i - window:i]
        valid = r.notna() & sv.notna()
        if valid.sum() < 10:
            continue
        slope = np.polyfit(sv[valid], r[valid], 1)[0]
        lambda_series.iloc[i] = slope

    return lambda_series

质量因子
质量因子衡量企业经营的内在健康度：


def compute_quality_factors(fundamentals):
    factors = pd.DataFrame(index=fundamentals.index)

    factors["roe"] = (
        fundamentals["net_income_ttm"] / fundamentals["book_value"]
    )

    factors["roa"] = (
        fundamentals["net_income_ttm"] / fundamentals["total_assets"]
    )

    factors["gross_margin"] = (
        fundamentals["gross_profit_ttm"] / fundamentals["revenue_ttm"]
    )

    factors["accruals"] = (
        (fundamentals["net_income_ttm"] - fundamentals["operating_cf_ttm"])
        / fundamentals["total_assets"]
    )

    factors["asset_turnover"] = (
        fundamentals["revenue_ttm"] / fundamentals["total_assets"]
    )

    factors["debt_to_equity"] = (
        fundamentals["total_debt"] / fundamentals["book_value"]
    )

    factors["earnings_stability"] = (
        fundamentals["net_income_ttm"].rolling(8).std()
        / fundamentals["net_income_ttm"].rolling(8).mean().abs()
    )

    return factors

def compute_news_sentiment_features(news_df, stock_id, windows=[1, 3, 7, 14]):
    stock_news = news_df[news_df["stock_id"] == stock_id].copy()
    stock_news = stock_news.set_index("publish_date")

    features = {}
    for w in windows:
        rolling = stock_news["sentiment_score"].rolling(f"{w}D")
        features[f"sentiment_mean_{w}d"] = rolling.mean()
        features[f"sentiment_std_{w}d"] = rolling.std()
        features[f"news_count_{w}d"] = rolling.count()

        pos_count = (stock_news["sentiment_score"] > 0.3).rolling(f"{w}D").sum()
        neg_count = (stock_news["sentiment_score"] < -0.3).rolling(f"{w}D").sum()
        total = pos_count + neg_count
        features[f"sentiment_ratio_{w}d"] = np.where(
            total > 0, (pos_count - neg_count) / total, 0
        )

    return pd.DataFrame(features)

去极值
金融数据中的极端值会严重扭曲模型训练。常用方法：


def winsorize_features(df, method="mad", n_sigma=5):
    result = df.copy()

    if method == "mad":
        for col in result.columns:
            median = result[col].median()
            mad = (result[col] - median).abs().median()
            upper = median + n_sigma * 1.4826 * mad
            lower = median - n_sigma * 1.4826 * mad
            result[col] = result[col].clip(lower, upper)

    elif method == "percentile":
        for col in result.columns:
            lower = result[col].quantile(0.01)
            upper = result[col].quantile(0.99)
            result[col] = result[col].clip(lower, upper)

    return result
标准化与中性化
截面标准化将因子转换为可比较的 z-score；行业/市值中性化消除结构性偏差：


def cross_sectional_zscore(factor_series):
    return (factor_series - factor_series.mean()) / factor_series.std()

def neutralize_factor(factor_df, industry_df, market_cap_df):
    neutralized = pd.DataFrame(index=factor_df.index, columns=factor_df.columns)

    for date in factor_df.index:
        f = factor_df.loc[date].dropna()
        ind = pd.get_dummies(industry_df.loc[date][f.index])
        cap = np.log(market_cap_df.loc[date][f.index])

        X = pd.concat([ind, cap], axis=1).dropna()
        common = f.index.intersection(X.index)
        f = f[common]
        X = X.loc[common]

        from sklearn.linear_model import LinearRegression
        reg = LinearRegression().fit(X, f)
        residual = f - reg.predict(X)
        neutralized.loc[date, common] = residual

    return neutralized

单因子检验
from scipy.stats import spearmanr

def single_factor_test(factor_df, return_df, top_n=10):
    results = {}

    for col in factor_df.columns:
        ic_list = []
        for date in factor_df.index:
            f = factor_df[col].loc[date]
            r = return_df.loc[date] if date in return_df.index else None
            if r is None:
                continue
            valid = f.notna() & r.notna()
            if valid.sum() < 30:
                continue
            ic, _ = spearmanr(f[valid], r[valid])
            ic_list.append(ic)

        if len(ic_list) < 20:
            continue

        ic_array = np.array(ic_list)
        results[col] = {
            "mean_ic": ic_array.mean(),
            "ic_std": ic_array.std(),
            "ir": ic_array.mean() / ic_array.std(),
            "ic_positive_ratio": (ic_array > 0).mean(),
            "abs_ic_mean": np.abs(ic_array).mean(),
            "t_stat": ic_array.mean() / (ic_array.std() / np.sqrt(len(ic_array)))
        }

    return pd.DataFrame(results).T.sort_values("ir", ascending=False)
多因子共线性与冗余检测
高度相关的因子会导致模型不稳定，需要去冗余：


    

  • 相关性过滤：删除相关系数 > 0.7 的因子对中 IC 更低的一个
    • 

  • VIF 检验：方差膨胀因子 > 10 表示严重共线性
    • 

  • PCA/因子分析：将高度相关的因子压缩为正交主成分
    • 



def filter_correlated_factors(factor_df, ic_dict, threshold=0.7):
    corr = factor_df.corr().abs()
    upper = corr.where(np.triu(np.ones(corr.shape), k=1).astype(bool))

    to_drop = set()
    for col in upper.columns:
        high_corr = upper.index[upper[col] > threshold].tolist()
        for h in high_corr:
            ic_col = abs(ic_dict.get(col, 0))
            ic_h = abs(ic_dict.get(h, 0))
            if ic_col >= ic_h:
                to_drop.add(h)
            else:
                to_drop.add(col)

    return factor_df.drop(columns=list(to_drop))
特征工程的时间一致性
金融特征工程中最致命的错误是前视偏差（Look-Ahead Bias）。


Point-in-Time 数据管理是特征工程的基础设施。每条财务数据必须标记两个日期：报告期（reporting period）和发布日期（release date）。在构造特征时，只能使用在预测时点之前已经公开的数据。


总结
金融特征工程的核心原则可以凝练为三个关键词：信号、纪律、简洁。信号指从多源异构数据中提取具有经济含义的预测信号；纪律指严格遵守时间一致性、避免前视偏差和幸存者偏差；简洁指在因子数量和复杂度之间保持克制——少数稳健因子的组合通常优于大量脆弱因子的堆砌。特征工程不是"越多越好"的数据搬运工作，而是对金融市场微观结构和经济逻辑的深刻理解在数据层面的表达。
