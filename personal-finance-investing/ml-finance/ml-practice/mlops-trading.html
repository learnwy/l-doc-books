MLOps 与交易模型生命周期

概述

将机器学习模型从 Jupyter Notebook 中的实验推向生产交易系统，是量化金融中最被低估的工程挑战。据统计，超过 87% 的 ML 项目从未进入生产环境——在金融领域，这个比例更高，因为交易系统对延迟、可靠性和合规性有极端严苛的要求。MLOps（Machine Learning Operations）是将 DevOps 理念应用于 ML 系统的工程实践，涵盖模型版本管理、自动化训练管线、部署监控、漂移检测和持续改进的完整生命周期。

本文系统梳理交易模型 MLOps 的架构设计与工程实践，从模型开发到退役的全流程管理。

import mlflow
import json
from datetime import datetime

class ExperimentTracker:
    def __init__(self, experiment_name, tracking_uri="sqlite:///mlflow.db"):
        mlflow.set_tracking_uri(tracking_uri)
        mlflow.set_experiment(experiment_name)

    def log_experiment(self, config, metrics, model, artifacts=None):
        with mlflow.start_run():
            mlflow.log_params({
                "model_type": config["model_type"],
                "features": json.dumps(config["features"][:20]),
                "n_features": len(config["features"]),
                "train_start": config["train_start"],
                "train_end": config["train_end"],
                "test_start": config["test_start"],
                "test_end": config["test_end"],
                "lookback_window": config.get("lookback_window", "N/A"),
                "label_horizon": config.get("label_horizon", "N/A")
            })

            mlflow.log_metrics({
                "train_ic": metrics["train_ic"],
                "test_ic": metrics["test_ic"],
                "train_ir": metrics["train_ir"],
                "test_ir": metrics["test_ir"],
                "ic_decay": metrics["train_ic"] - metrics["test_ic"],
                "feature_stability": metrics.get("feature_stability", 0),
                "turnover": metrics.get("turnover", 0)
            })

            mlflow.log_param(
                "timestamp", datetime.now().isoformat()
            )

            if artifacts:
                for name, path in artifacts.items():
                    mlflow.log_artifact(path)

            mlflow.sklearn.log_model(model, "model")

class ModelRegistry:
    def __init__(self):
        self.models = {}

    def register(self, model_name, model_version, metadata):
        key = f"{model_name}:{model_version}"
        self.models[key] = {
            "model_name": model_name,
            "version": model_version,
            "stage": "staging",
            "registered_at": datetime.now().isoformat(),
            "metadata": metadata,
            "performance_history": []
        }
        return key

    def promote(self, model_key, target_stage):
        valid_transitions = {
            "staging": ["canary", "archived"],
            "canary": ["production", "staging", "archived"],
            "production": ["archived"]
        }

        current_stage = self.models[model_key]["stage"]
        if target_stage not in valid_transitions.get(current_stage, []):
            raise ValueError(
                f"Invalid transition: {current_stage} -> {target_stage}"
            )

        self.models[model_key]["stage"] = target_stage
        self.models[model_key][f"{target_stage}_at"] = (
            datetime.now().isoformat()
        )

    def get_production_model(self, model_name):
        for key, info in self.models.items():
            if (info["model_name"] == model_name
                    and info["stage"] == "production"):
                return key, info
        return None, None
特征存储（Feature Store）
架构设计
class FeatureStore:
    def __init__(self, storage_backend):
        self.storage = storage_backend
        self.registry = {}

    def register_feature(self, feature_name, config):
        self.registry[feature_name] = {
            "name": feature_name,
            "description": config["description"],
            "dtype": config["dtype"],
            "compute_fn": config["compute_fn"],
            "dependencies": config.get("dependencies", []),
            "freshness_sla": config.get("freshness_sla", "1d"),
            "owner": config.get("owner", "unknown"),
            "version": config.get("version", "1.0"),
            "created_at": datetime.now().isoformat()
        }

    def get_features(self, feature_names, entity_ids, timestamp):
        features = {}
        for name in feature_names:
            if name not in self.registry:
                raise ValueError(f"Unknown feature: {name}")

            data = self.storage.read(
                feature_name=name,
                entity_ids=entity_ids,
                as_of=timestamp
            )
            features[name] = data

        return pd.DataFrame(features, index=entity_ids)

    def get_training_dataset(self, feature_names, entity_ids,
                              start_date, end_date):
        datasets = []
        for date in pd.date_range(start_date, end_date, freq="B"):
            try:
                snapshot = self.get_features(
                    feature_names, entity_ids, date
                )
                snapshot["date"] = date
                datasets.append(snapshot)
            except Exception:
                continue

        return pd.concat(datasets)

    def validate_feature_freshness(self):
        stale_features = []
        for name, config in self.registry.items():
            last_updated = self.storage.get_last_update(name)
            sla = pd.Timedelta(config["freshness_sla"])
            if datetime.now() - last_updated > sla:
                stale_features.append({
                    "feature": name,
                    "last_updated": last_updated,
                    "sla": config["freshness_sla"]
                })
        return stale_features

漂移检测
模型性能衰退的根本原因通常是数据分布漂移（Data Drift）：


from scipy.stats import ks_2samp, wasserstein_distance

class DriftDetector:
    def __init__(self, reference_data, feature_names):
        self.reference = reference_data
        self.feature_names = feature_names
        self.thresholds = self._compute_thresholds()

    def _compute_thresholds(self):
        thresholds = {}
        for col in self.feature_names:
            ref_std = self.reference[col].std()
            thresholds[col] = {
                "ks_threshold": 0.1,
                "psi_threshold": 0.2,
                "mean_shift_threshold": 0.5 * ref_std
            }
        return thresholds

    def detect_drift(self, current_data):
        drift_report = {}

        for col in self.feature_names:
            ref = self.reference[col].dropna()
            cur = current_data[col].dropna()

            if len(cur) < 30:
                continue

            ks_stat, ks_pvalue = ks_2samp(ref, cur)
            w_distance = wasserstein_distance(ref, cur)
            psi = self._compute_psi(ref, cur)
            mean_shift = abs(cur.mean() - ref.mean())

            drift_report[col] = {
                "ks_statistic": ks_stat,
                "ks_pvalue": ks_pvalue,
                "wasserstein_distance": w_distance,
                "psi": psi,
                "mean_shift": mean_shift,
                "is_drifted": (
                    ks_stat > self.thresholds[col]["ks_threshold"]
                    or psi > self.thresholds[col]["psi_threshold"]
                )
            }

        return drift_report

    def _compute_psi(self, reference, current, n_bins=10):
        bins = np.histogram_bin_edges(reference, bins=n_bins)
        ref_hist = np.histogram(reference, bins=bins)[0] / len(reference)
        cur_hist = np.histogram(current, bins=bins)[0] / len(current)

        ref_hist = np.clip(ref_hist, 1e-6, None)
        cur_hist = np.clip(cur_hist, 1e-6, None)

        psi = np.sum((cur_hist - ref_hist) * np.log(cur_hist / ref_hist))
        return psi
性能监控看板
class ModelPerformanceDashboard:
    def __init__(self, model_name):
        self.model_name = model_name
        self.metrics_history = []

    def record_daily_metrics(self, date, predictions, actuals):
        from scipy.stats import spearmanr

        valid = predictions.notna() & actuals.notna()
        if valid.sum() < 30:
            return

        ic, _ = spearmanr(predictions[valid], actuals[valid])

        self.metrics_history.append({
            "date": date,
            "ic": ic,
            "prediction_mean": predictions.mean(),
            "prediction_std": predictions.std(),
            "coverage": valid.mean(),
            "n_stocks": valid.sum()
        })

    def generate_report(self, lookback_days=60):
        if len(self.metrics_history) < lookback_days:
            lookback_days = len(self.metrics_history)

        recent = self.metrics_history[-lookback_days:]
        ic_series = [m["ic"] for m in recent]

        report = {
            "model_name": self.model_name,
            "period": f"last {lookback_days} days",
            "mean_ic": np.mean(ic_series),
            "ic_std": np.std(ic_series),
            "ir": np.mean(ic_series) / max(np.std(ic_series), 1e-8),
            "ic_positive_ratio": np.mean(np.array(ic_series) > 0),
            "worst_ic": np.min(ic_series),
            "best_ic": np.max(ic_series),
            "avg_coverage": np.mean([m["coverage"] for m in recent])
        }

        all_ic = [m["ic"] for m in self.metrics_history]
        if len(all_ic) > lookback_days * 2:
            prev_ic = all_ic[-lookback_days*2:-lookback_days]
            report["ic_trend"] = np.mean(ic_series) - np.mean(prev_ic)
            report["degradation_alert"] = report["ic_trend"] < -0.005

        return report
自动化训练管线
调度与编排
class TrainingPipeline:
    def __init__(self, feature_store, model_registry, config):
        self.fs = feature_store
        self.registry = model_registry
        self.config = config

    def run(self, trigger_reason="scheduled"):
        run_id = f"run_{datetime.now().strftime('%Y%m%d_%H%M%S')}"

        data = self._extract_data()
        validation = self._validate_data(data)
        if not validation["passed"]:
            self._alert(f"Data validation failed: {validation['issues']}")
            return {"status": "failed", "reason": "data_validation"}

        X_train, y_train, X_val, y_val, X_test, y_test = (
            self._split_data(data)
        )

        model = self._train_model(X_train, y_train, X_val, y_val)

        metrics = self._evaluate(model, X_test, y_test)

        if not self._meets_threshold(metrics):
            return {
                "status": "below_threshold",
                "metrics": metrics,
                "run_id": run_id
            }

        model_key = self.registry.register(
            self.config["model_name"],
            run_id,
            {"metrics": metrics, "trigger": trigger_reason}
        )

        return {
            "status": "success",
            "model_key": model_key,
            "metrics": metrics,
            "run_id": run_id
        }

    def _extract_data(self):
        return self.fs.get_training_dataset(
            self.config["features"],
            self.config["entity_ids"],
            self.config["train_start"],
            self.config["train_end"]
        )

    def _validate_data(self, data):
        issues = []
        missing_rate = data.isnull().mean()
        high_missing = missing_rate[missing_rate > 0.3]
        if len(high_missing) > 0:
            issues.append(f"High missing rate: {high_missing.to_dict()}")

        for col in data.select_dtypes(include=[np.number]).columns:
            if data[col].std() < 1e-10:
                issues.append(f"Zero variance: {col}")

        return {"passed": len(issues) == 0, "issues": issues}

    def _split_data(self, data):
        dates = data["date"].unique()
        n = len(dates)
        train_end = dates[int(n * 0.6)]
        val_end = dates[int(n * 0.8)]

        train = data[data["date"] <= train_end]
        val = data[(data["date"] > train_end) & (data["date"] <= val_end)]
        test = data[data["date"] > val_end]

        feature_cols = self.config["features"]
        label_col = self.config["label"]

        return (
            train[feature_cols], train[label_col],
            val[feature_cols], val[label_col],
            test[feature_cols], test[label_col]
        )

    def _train_model(self, X_train, y_train, X_val, y_val):
        import lightgbm as lgb
        train_data = lgb.Dataset(X_train, label=y_train)
        val_data = lgb.Dataset(X_val, label=y_val, reference=train_data)

        model = lgb.train(
            self.config["model_params"],
            train_data,
            num_boost_round=1000,
            valid_sets=[val_data],
            callbacks=[lgb.early_stopping(50)]
        )
        return model

    def _evaluate(self, model, X_test, y_test):
        from scipy.stats import spearmanr
        predictions = model.predict(X_test)
        ic, _ = spearmanr(predictions, y_test)
        return {"test_ic": ic}

    def _meets_threshold(self, metrics):
        return metrics["test_ic"] > self.config.get("min_ic_threshold", 0.02)

    def _alert(self, message):
        print(f"[ALERT] {message}")

灰度发布策略
class CanaryDeployment:
    def __init__(self, old_model, new_model, initial_ratio=0.1):
        self.old_model = old_model
        self.new_model = new_model
        self.ratio = initial_ratio
        self.metrics = {"old": [], "new": []}

    def predict(self, X, request_id):
        use_new = hash(request_id) % 100 < self.ratio * 100

        if use_new:
            pred = self.new_model.predict(X)
            self.metrics["new"].append(pred)
        else:
            pred = self.old_model.predict(X)
            self.metrics["old"].append(pred)

        return pred, "new" if use_new else "old"

    def evaluate_and_adjust(self):
        if len(self.metrics["new"]) < 100:
            return "insufficient_data"

        new_quality = self._compute_quality(self.metrics["new"])
        old_quality = self._compute_quality(self.metrics["old"])

        if new_quality > old_quality * 1.05:
            self.ratio = min(self.ratio * 2, 1.0)
            return "expanding"
        elif new_quality < old_quality * 0.9:
            self.ratio = 0
            return "rollback"
        else:
            return "holding"

    def _compute_quality(self, predictions):
        return np.mean([abs(p) for p in predictions])
总结
MLOps 是交易模型从"能跑"到"能赚"的关键桥梁。一个完整的交易模型 MLOps 体系包括四个核心组件：实验管理（追踪每次实验的完整上下文，确保可复现）；特征存储（统一管理特征的计算、存储和服务，确保训练-推理一致性）；自动化管线（定时或事件触发的训练-评估-部署流水线，减少人工干预）；持续监控（实时追踪模型性能和数据漂移，及时发现衰退并触发再训练）。在金融场景中，MLOps 还需要特别关注合规审计（模型版本可追溯）、灾难恢复（快速回滚到前一版本）和成本控制（计算资源与策略收益的 ROI 分析）。最终目标是建立一个闭环系统——模型自动感知环境变化、自动触发更新、自动验证质量——让量化研究员把精力集中在信号发现而非系统运维上。
