模型可解释性与监管要求

概述

金融行业对模型可解释性（Interpretability / Explainability）的要求远超其他领域。当一个信贷模型拒绝了某人的贷款申请时，监管法规要求金融机构必须给出明确的理由——"模型说不行"不是一个合法的解释。从美国的 ECOA（平等信贷机会法）到欧盟的 GDPR（通用数据保护条例）中的"算法解释权"，再到中国人民银行的征信业务管理办法，全球监管体系对金融 AI 模型的透明度要求日益严格。

本文系统梳理模型可解释性的理论框架、技术方法和监管合规实践，帮助量化从业者在模型性能与可解释性之间找到最优平衡。

本质可解释模型
线性模型与评分卡
信用评分卡是金融领域最经典的可解释模型——每个特征对应一个分数，总分直接映射到违约概率：


import numpy as np
from sklearn.linear_model import LogisticRegression

class ScorecardModel:
    def __init__(self, base_score=600, pdo=20, base_odds=50):
        self.base_score = base_score
        self.pdo = pdo
        self.base_odds = base_odds
        self.factor = pdo / np.log(2)
        self.offset = base_score - self.factor * np.log(base_odds)
        self.model = None
        self.scorecard = None

    def fit(self, X_woe, y, feature_names):
        self.model = LogisticRegression(C=1.0, penalty="l2", max_iter=1000)
        self.model.fit(X_woe, y)

        self.scorecard = {}
        intercept_score = self.offset - self.factor * self.model.intercept_[0]

        for i, name in enumerate(feature_names):
            coef = self.model.coef_[0][i]
            self.scorecard[name] = {
                "coefficient": coef,
                "score_contribution": -self.factor * coef
            }

        self.scorecard["_intercept"] = {
            "score": intercept_score / len(feature_names)
        }

        return self

    def predict_score(self, X_woe, feature_names):
        scores = np.full(X_woe.shape[0], self.offset)
        scores -= self.factor * self.model.intercept_[0]

        for i, name in enumerate(feature_names):
            coef = self.model.coef_[0][i]
            scores -= self.factor * coef * X_woe[:, i]

        return scores

    def explain_decision(self, x_single, feature_names, top_k=5):
        contributions = []
        for i, name in enumerate(feature_names):
            coef = self.model.coef_[0][i]
            contrib = -self.factor * coef * x_single[i]
            contributions.append({
                "feature": name,
                "contribution": contrib,
                "direction": "positive" if contrib > 0 else "negative"
            })

        contributions.sort(key=lambda x: abs(x["contribution"]), reverse=True)
        return contributions[:top_k]
广义可加模型（GAM）
GAM 允许每个特征有非线性效应，同时保持可解释性——每个特征的效应可以单独可视化：


g(E[y])=β0+f1(x1)+f2(x2)+⋯+fp(xp)g(\mathbb{E}[y]) = \beta_0 + f_1(x_1) + f_2(x_2) + \cdots + f_p(x_p)g(E[y])=β0​+f1​(x1​)+f2​(x2​)+⋯+fp​(xp​)
from pygam import LogisticGAM, s

def train_interpretable_gam(X, y, feature_names):
    terms = s(0)
    for i in range(1, X.shape[1]):
        terms += s(i)

    gam = LogisticGAM(terms)
    gam.gridsearch(X, y)

    explanations = {}
    for i, name in enumerate(feature_names):
        XX = gam.generate_X_grid(term=i)
        pdep, confi = gam.partial_dependence(term=i, X=XX, width=0.95)
        explanations[name] = {
            "grid": XX[:, i],
            "partial_dependence": pdep,
            "confidence_lower": confi[:, 0],
            "confidence_upper": confi[:, 1]
        }

    return gam, explanations

事后解释方法
SHAP（SHapley Additive exPlanations）
SHAP 基于 Shapley 值理论，为每个特征分配对预测结果的公平贡献：


ϕi=∑S⊆F∖{i}∣S∣!(∣F∣−∣S∣−1)!∣F∣![f(S∪{i})−f(S)]\phi_i = \sum_{S \subseteq F \setminus \{i\}} \frac{|S|!(|F|-|S|-1)!}{|F|!} [f(S \cup \{i\}) - f(S)]ϕi​=S⊆F∖{i}∑​∣F∣!∣S∣!(∣F∣−∣S∣−1)!​[f(S∪{i})−f(S)]
import shap

class SHAPExplainer:
    def __init__(self, model, X_background, feature_names=None):
        self.model = model
        self.feature_names = feature_names

        if hasattr(model, "predict_proba"):
            self.explainer = shap.TreeExplainer(model)
        else:
            self.explainer = shap.KernelExplainer(
                model.predict, shap.sample(X_background, 100)
            )

    def explain_instance(self, x_single, top_k=5):
        shap_values = self.explainer.shap_values(x_single.reshape(1, -1))

        if isinstance(shap_values, list):
            shap_values = shap_values[1]

        contributions = []
        for i in range(len(shap_values[0])):
            name = self.feature_names[i] if self.feature_names else f"feature_{i}"
            contributions.append({
                "feature": name,
                "shap_value": shap_values[0][i],
                "feature_value": x_single[i],
                "direction": "increases" if shap_values[0][i] > 0 else "decreases"
            })

        contributions.sort(key=lambda x: abs(x["shap_value"]), reverse=True)
        return contributions[:top_k]

    def explain_global(self, X_explain):
        shap_values = self.explainer.shap_values(X_explain)

        if isinstance(shap_values, list):
            shap_values = shap_values[1]

        mean_abs_shap = np.abs(shap_values).mean(axis=0)
        ranking = np.argsort(-mean_abs_shap)

        global_importance = []
        for idx in ranking:
            name = self.feature_names[idx] if self.feature_names else f"feature_{idx}"
            global_importance.append({
                "feature": name,
                "mean_abs_shap": mean_abs_shap[idx],
                "rank": len(global_importance) + 1
            })

        return global_importance

    def generate_adverse_action_reasons(self, x_single, n_reasons=4):
        contributions = self.explain_instance(x_single, top_k=n_reasons)

        reasons = []
        for c in contributions:
            if c["direction"] == "increases":
                reasons.append({
                    "factor": c["feature"],
                    "impact": c["shap_value"],
                    "message": self._generate_reason_message(
                        c["feature"], c["feature_value"], c["direction"]
                    )
                })

        return reasons[:n_reasons]

    def _generate_reason_message(self, feature, value, direction):
        reason_templates = {
            "debt_to_income": f"债务收入比过高（当前值: {value:.2f}）",
            "credit_score": f"信用评分不足（当前值: {value:.0f}）",
            "delinquency_count": f"历史逾期次数较多（{value:.0f}次）",
            "years_employed": f"工作年限较短（{value:.1f}年）"
        }
        return reason_templates.get(feature, f"{feature}: {value}")
LIME（Local Interpretable Model-agnostic Explanations）
LIME 在单个预测的局部邻域拟合一个简单的可解释模型：


import lime
import lime.lime_tabular

def lime_explain(model, X_train, instance, feature_names, class_names=None):
    explainer = lime.lime_tabular.LimeTabularExplainer(
        X_train,
        feature_names=feature_names,
        class_names=class_names or ["reject", "approve"],
        mode="classification"
    )

    explanation = explainer.explain_instance(
        instance,
        model.predict_proba,
        num_features=10,
        num_samples=5000
    )

    return {
        "prediction": model.predict_proba(instance.reshape(1, -1))[0],
        "top_features": explanation.as_list(),
        "local_prediction": explanation.local_pred[0],
        "intercept": explanation.intercept[1]
    }

前沿方向
Concept Bottleneck Models
将预测分解为先预测人类可理解的"概念"，再从概念预测最终结果——天然可解释且允许人类干预：


    

  • 概念层：高 PE 比、低现金流、行业下行 → 人类可读
    • 

  • 预测层：概念 → 违约概率
    • 



因果解释
从"相关性解释"走向"因果解释"——不仅告诉你"哪个特征重要"，还告诉你"如果改变这个特征会怎样"：


    

  • 反事实解释："如果您的收入增加 20%，贷款将被批准"
    • 

  • 因果公平性：确保保护属性（种族、性别）不通过代理变量影响决策
    • 



总结
模型可解释性在金融领域不是"锦上添花"，而是"合规底线"。在信贷、保险等强监管场景中，可解释性是模型上线的必要条件；在量化投资场景中，可解释性是防止过拟合和建立模型信任的重要手段。SHAP 已成为金融模型解释的事实标准，但它本身也有局限（如对特征相关性的处理）。最佳实践是根据应用场景的监管要求选择适当的可解释性水平——在需要强制解释的场景使用本质可解释模型，在允许"解释作为补充"的场景使用高性能模型 + 事后解释方法。
