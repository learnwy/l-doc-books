回测框架与偏差控制

概述

回测（Backtesting）是将交易策略应用于历史数据以评估其潜在表现的过程。一个严格的回测框架是量化交易研究的基石——它既是策略开发的实验平台，也是策略上线前的最后一道质量关卡。然而，回测环境与实盘交易之间存在系统性差异，如果不加以控制，回测结果可能严重高估策略的真实表现。本文系统梳理回测框架的设计原则和各类偏差的识别与控制方法。

回测框架设计
事件驱动回测引擎
import numpy as np
import pandas as pd
from dataclasses import dataclass, field
from abc import ABC, abstractmethod
from typing import Optional
from collections import defaultdict

@dataclass
class BacktestConfig:
    start_date: str
    end_date: str
    initial_capital: float = 1_000_000
    commission_rate: float = 0.0003
    slippage_bps: float = 1.0
    margin_requirement: float = 1.0

@dataclass
class BacktestTrade:
    timestamp: pd.Timestamp
    symbol: str
    side: str
    quantity: int
    price: float
    commission: float
    slippage: float

class BacktestEngine:
    """事件驱动回测引擎"""

    def __init__(self, config: BacktestConfig):
        self.config = config
        self.cash = config.initial_capital
        self.positions: dict[str, int] = defaultdict(int)
        self.position_costs: dict[str, float] = defaultdict(float)
        self.trades: list[BacktestTrade] = []
        self.equity_curve: list[dict] = []
        self._current_prices: dict[str, float] = {}

    def run(self, strategy, data: dict[str, pd.DataFrame]):
        all_dates = sorted(set().union(*[df.index for df in data.values()]))
        filtered_dates = [d for d in all_dates
                          if self.config.start_date <= str(d) <= self.config.end_date]

        for date in filtered_dates:
            bars = {}
            for symbol, df in data.items():
                if date in df.index:
                    bars[symbol] = df.loc[date]
                    self._current_prices[symbol] = df.loc[date]["close"]

            if bars:
                signals = strategy.on_bars(bars, self._get_portfolio_state())
                self._execute_signals(signals, date)
                self._record_equity(date)

    def _execute_signals(self, signals: list[dict], date: pd.Timestamp):
        for signal in signals:
            symbol = signal["symbol"]
            target_qty = signal["quantity"]
            current_qty = self.positions[symbol]
            trade_qty = target_qty - current_qty

            if trade_qty == 0:
                continue

            price = self._current_prices[symbol]
            side = "buy" if trade_qty > 0 else "sell"
            slippage = price * self.config.slippage_bps / 10000
            if side == "buy":
                exec_price = price + slippage
            else:
                exec_price = price - slippage

            commission = abs(trade_qty) * exec_price * self.config.commission_rate
            cost = trade_qty * exec_price + commission

            if self.cash - cost < 0 and side == "buy":
                continue

            self.cash -= cost
            self.positions[symbol] += trade_qty
            self.trades.append(BacktestTrade(
                timestamp=date, symbol=symbol, side=side,
                quantity=abs(trade_qty), price=exec_price,
                commission=commission, slippage=slippage * abs(trade_qty),
            ))

    def _get_portfolio_state(self) -> dict:
        portfolio_value = self.cash + sum(
            qty * self._current_prices.get(sym, 0)
            for sym, qty in self.positions.items()
        )
        return {
            "cash": self.cash,
            "positions": dict(self.positions),
            "portfolio_value": portfolio_value,
        }

    def _record_equity(self, date: pd.Timestamp):
        state = self._get_portfolio_state()
        self.equity_curve.append({
            "date": date,
            "portfolio_value": state["portfolio_value"],
            "cash": state["cash"],
        })
回测结果分析
class BacktestAnalyzer:
    """回测结果分析器"""

    def __init__(self, equity_curve: pd.DataFrame, trades: list[BacktestTrade],
                 risk_free_rate: float = 0.02):
        self.equity = equity_curve
        self.trades = trades
        self.rf = risk_free_rate

    def compute_metrics(self) -> dict:
        returns = self.equity["portfolio_value"].pct_change().dropna()
        total_return = (self.equity["portfolio_value"].iloc[-1] /
                        self.equity["portfolio_value"].iloc[0] - 1)
        n_years = len(returns) / 252
        annual_return = (1 + total_return) ** (1 / n_years) - 1
        annual_vol = returns.std() * np.sqrt(252)
        sharpe = (annual_return - self.rf) / annual_vol if annual_vol > 0 else 0

        rolling_max = self.equity["portfolio_value"].expanding().max()
        drawdown = self.equity["portfolio_value"] / rolling_max - 1
        max_drawdown = drawdown.min()

        calmar = annual_return / abs(max_drawdown) if max_drawdown != 0 else 0
        sortino_vol = returns[returns < 0].std() * np.sqrt(252)
        sortino = (annual_return - self.rf) / sortino_vol if sortino_vol > 0 else 0

        winning_trades = [t for t in self.trades if self._trade_pnl(t) > 0]
        win_rate = len(winning_trades) / len(self.trades) if self.trades else 0

        return {
            "total_return": total_return,
            "annual_return": annual_return,
            "annual_volatility": annual_vol,
            "sharpe_ratio": sharpe,
            "sortino_ratio": sortino,
            "calmar_ratio": calmar,
            "max_drawdown": max_drawdown,
            "win_rate": win_rate,
            "total_trades": len(self.trades),
            "total_commission": sum(t.commission for t in self.trades),
            "total_slippage": sum(t.slippage for t in self.trades),
        }

    def _trade_pnl(self, trade: BacktestTrade) -> float:
        return 0.0

    def monthly_returns(self) -> pd.DataFrame:
        returns = self.equity.set_index("date")["portfolio_value"].pct_change()
        monthly = returns.resample("M").apply(lambda x: (1 + x).prod() - 1)
        return monthly

Point-in-Time 数据管理
class PointInTimeData:
    """Point-in-Time 数据管理"""

    def __init__(self):
        self._data: dict[str, list[dict]] = defaultdict(list)

    def add_record(self, symbol: str, report_date: str, available_date: str, data: dict):
        self._data[symbol].append({
            "report_date": report_date,
            "available_date": available_date,
            **data,
        })

    def get_as_of(self, symbol: str, as_of_date: str) -> Optional[dict]:
        records = self._data.get(symbol, [])
        available = [r for r in records if r["available_date"] <= as_of_date]
        if not available:
            return None
        return max(available, key=lambda r: r["available_date"])

    def expanding_zscore(self, series: pd.Series) -> pd.Series:
        result = pd.Series(np.nan, index=series.index)
        for i in range(30, len(series)):
            window = series.iloc[:i + 1]
            result.iloc[i] = (series.iloc[i] - window.mean()) / window.std()
        return result
幸存者偏差（Survivorship Bias）
影响量化
幸存者偏差指的是只使用当前仍然存在的证券进行回测，忽略了期间退市、被收购或破产的证券：


class SurvivorshipBiasCheck:
    """幸存者偏差检查"""

    def __init__(self, universe_snapshots: dict[str, list[str]]):
        self.snapshots = universe_snapshots

    def get_universe_at(self, date: str) -> list[str]:
        available_dates = sorted([d for d in self.snapshots if d <= date])
        if not available_dates:
            return []
        return self.snapshots[available_dates[-1]]

    def delisted_stocks(self, start_date: str, end_date: str) -> list[str]:
        start_universe = set(self.get_universe_at(start_date))
        end_universe = set(self.get_universe_at(end_date))
        return list(start_universe - end_universe)

    def bias_estimate(self, strategy_returns: pd.Series,
                      survivorship_free_returns: pd.Series) -> dict:
        bias = strategy_returns.mean() - survivorship_free_returns.mean()
        return {
            "annual_bias_bps": bias * 252 * 10000,
            "biased_sharpe": strategy_returns.mean() / strategy_returns.std() * np.sqrt(252),
            "corrected_sharpe": survivorship_free_returns.mean() / survivorship_free_returns.std() * np.sqrt(252),
        }

Walk-Forward 分析
class WalkForwardAnalysis:
    """Walk-Forward 分析"""

    def __init__(self, train_window: int = 252 * 3, test_window: int = 252,
                 step: int = 63):
        self.train_window = train_window
        self.test_window = test_window
        self.step = step

    def run(self, data: pd.DataFrame, strategy_factory: callable) -> pd.DataFrame:
        results = []
        total_length = len(data)

        for start in range(0, total_length - self.train_window - self.test_window, self.step):
            train_end = start + self.train_window
            test_end = min(train_end + self.test_window, total_length)

            train_data = data.iloc[start:train_end]
            test_data = data.iloc[train_end:test_end]

            strategy = strategy_factory()
            strategy.fit(train_data)

            in_sample_returns = strategy.predict(train_data)
            out_sample_returns = strategy.predict(test_data)

            is_sharpe = in_sample_returns.mean() / in_sample_returns.std() * np.sqrt(252)
            oos_sharpe = out_sample_returns.mean() / out_sample_returns.std() * np.sqrt(252)

            results.append({
                "train_start": data.index[start],
                "train_end": data.index[train_end],
                "test_start": data.index[train_end],
                "test_end": data.index[test_end - 1],
                "is_sharpe": is_sharpe,
                "oos_sharpe": oos_sharpe,
                "degradation": 1 - oos_sharpe / is_sharpe if is_sharpe > 0 else float("inf"),
            })

        return pd.DataFrame(results)
Deflated Sharpe Ratio
Harvey 和 Liu（2015）提出的 Deflated Sharpe Ratio 修正了多重检验造成的过拟合风险：


class DeflatedSharpeRatio:
    """Deflated Sharpe Ratio"""

    def compute(self, observed_sharpe: float, n_trials: int, n_observations: int,
                skewness: float = 0.0, kurtosis: float = 3.0) -> dict:
        from scipy.stats import norm

        expected_max_sharpe = self._expected_max_sharpe(n_trials, n_observations)
        se_sharpe = np.sqrt(
            (1 - skewness * observed_sharpe + (kurtosis - 1) / 4 * observed_sharpe ** 2)
            / n_observations
        )
        dsr_statistic = (observed_sharpe - expected_max_sharpe) / se_sharpe
        p_value = 1 - norm.cdf(dsr_statistic)

        return {
            "observed_sharpe": observed_sharpe,
            "expected_max_sharpe": expected_max_sharpe,
            "dsr_statistic": dsr_statistic,
            "p_value": p_value,
            "is_significant": p_value < 0.05,
            "n_trials": n_trials,
        }

    def _expected_max_sharpe(self, n_trials: int, n_observations: int) -> float:
        from scipy.stats import norm
        gamma = 0.5772
        e_max = (1 - gamma) * norm.ppf(1 - 1 / n_trials) + gamma * norm.ppf(1 - 1 / (n_trials * np.e))
        return e_max / np.sqrt(n_observations)
参数稳健性检验
class ParameterRobustness:
    """参数稳健性检验"""

    def sensitivity_analysis(self, base_params: dict, param_ranges: dict,
                             evaluate_fn: callable) -> pd.DataFrame:
        results = []
        for param_name, values in param_ranges.items():
            for value in values:
                test_params = base_params.copy()
                test_params[param_name] = value
                metric = evaluate_fn(test_params)
                results.append({
                    "parameter": param_name,
                    "value": value,
                    "sharpe": metric["sharpe_ratio"],
                    "return": metric["annual_return"],
                    "drawdown": metric["max_drawdown"],
                })
        return pd.DataFrame(results)

    def cliff_detection(self, param_name: str, results: pd.DataFrame,
                        threshold: float = 0.3) -> bool:
        param_results = results[results["parameter"] == param_name].sort_values("value")
        sharpes = param_results["sharpe"].values
        for i in range(1, len(sharpes)):
            if abs(sharpes[i] - sharpes[i - 1]) > threshold:
                return True
        return False

class RealisticCostModel:
    """真实交易成本模型"""

    def __init__(self):
        self.commission_rate = 0.0003
        self.min_commission = 5.0

    def estimate_cost(self, symbol: str, quantity: int, price: float,
                      adv: float, spread: float, volatility: float) -> dict:
        commission = max(quantity * price * self.commission_rate, self.min_commission)
        spread_cost = spread / 2 * quantity
        participation = quantity / adv
        impact = 0.1 * volatility * price * quantity * participation ** 0.5
        slippage = price * quantity * 0.0001
        total = commission + spread_cost + impact + slippage

        return {
            "commission": commission,
            "spread_cost": spread_cost,
            "market_impact": impact,
            "slippage": slippage,
            "total_cost": total,
            "total_bps": total / (price * quantity) * 10000,
        }

总结
回测框架是量化策略开发的实验基础，其可靠性直接决定了研究产出的质量。事件驱动架构为回测提供了与实盘一致的执行逻辑；Point-in-Time 数据管理消除了前视偏差的主要来源；Walk-Forward 分析和 Deflated Sharpe Ratio 提供了系统性的过拟合检测手段；真实的交易成本模型确保回测结果具有实盘可参考性。最关键的原则是保持对回测结果的怀疑态度——回测中看起来太好的结果，在实盘中往往不会如预期般表现，严格的偏差控制是连接回测与实盘的必经之路。
