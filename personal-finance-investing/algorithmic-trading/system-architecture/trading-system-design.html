交易系统架构设计

概述

一个生产级算法交易系统是高度复杂的分布式实时系统。它需要在微秒到毫秒级的时间窗口内完成市场数据接收、策略信号计算、风控检查和订单发送的全链路处理。系统架构的设计直接决定了策略的执行质量和运营可靠性。本文梳理交易系统的核心架构模式，从整体分层设计到关键子系统的实现细节。

事件驱动架构
核心设计模式
现代交易系统几乎普遍采用事件驱动架构（Event-Driven Architecture, EDA）。相比传统的轮询模式，事件驱动架构在延迟、吞吐量和可扩展性方面都有显著优势。


from dataclasses import dataclass, field
from enum import Enum
from typing import Callable
from collections import defaultdict
import time

class EventType(Enum):
    TICK = "tick"
    BAR = "bar"
    ORDER_UPDATE = "order_update"
    FILL = "fill"
    SIGNAL = "signal"
    RISK_ALERT = "risk_alert"
    TIMER = "timer"
    SYSTEM = "system"

@dataclass
class Event:
    event_type: EventType
    timestamp: float
    data: dict
    source: str = ""
    priority: int = 0

class EventBus:
    """事件总线"""

    def __init__(self):
        self._handlers: dict[EventType, list[Callable]] = defaultdict(list)
        self._event_queue: list[Event] = []
        self._metrics = {"total_events": 0, "total_latency_us": 0}

    def subscribe(self, event_type: EventType, handler: Callable):
        self._handlers[event_type].append(handler)

    def publish(self, event: Event):
        self._event_queue.append(event)
        self._metrics["total_events"] += 1

    def process(self):
        while self._event_queue:
            event = self._event_queue.pop(0)
            start = time.perf_counter_ns()
            for handler in self._handlers.get(event.event_type, []):
                handler(event)
            elapsed_us = (time.perf_counter_ns() - start) / 1000
            self._metrics["total_latency_us"] += elapsed_us

from collections import deque

@dataclass
class MarketTick:
    symbol: str
    timestamp: float
    bid: float
    ask: float
    bid_size: int
    ask_size: int
    last_price: float
    last_size: int
    exchange: str

class MarketDataHandler:
    """市场数据处理器"""

    def __init__(self, max_symbols: int = 10000):
        self._latest: dict[str, MarketTick] = {}
        self._tick_buffers: dict[str, deque] = defaultdict(lambda: deque(maxlen=10000))
        self._bar_builders: dict[str, dict] = {}
        self._subscribers: list[Callable] = []

    def on_tick(self, tick: MarketTick):
        self._latest[tick.symbol] = tick
        self._tick_buffers[tick.symbol].append(tick)
        self._update_bar(tick)
        for subscriber in self._subscribers:
            subscriber(tick)

    def _update_bar(self, tick: MarketTick):
        symbol = tick.symbol
        if symbol not in self._bar_builders:
            self._bar_builders[symbol] = {
                "open": tick.last_price,
                "high": tick.last_price,
                "low": tick.last_price,
                "close": tick.last_price,
                "volume": tick.last_size,
                "start_time": tick.timestamp,
            }
        else:
            bar = self._bar_builders[symbol]
            bar["high"] = max(bar["high"], tick.last_price)
            bar["low"] = min(bar["low"], tick.last_price)
            bar["close"] = tick.last_price
            bar["volume"] += tick.last_size

    def get_snapshot(self, symbol: str) -> dict:
        tick = self._latest.get(symbol)
        if not tick:
            return None
        return {
            "symbol": symbol,
            "bid": tick.bid,
            "ask": tick.ask,
            "mid": (tick.bid + tick.ask) / 2,
            "spread": tick.ask - tick.bid,
            "last": tick.last_price,
        }

class DataQualityChecker:
    """数据质量检查"""

    def __init__(self, max_spread_bps: float = 500, max_price_change_pct: float = 10):
        self.max_spread_bps = max_spread_bps
        self.max_price_change = max_price_change_pct / 100
        self._last_prices: dict[str, float] = {}

    def validate_tick(self, tick: MarketTick) -> tuple[bool, str]:
        if tick.bid > tick.ask:
            return False, "crossed_market"

        spread_bps = (tick.ask - tick.bid) / tick.mid_price * 10000 if tick.bid > 0 else 0
        if spread_bps > self.max_spread_bps:
            return False, "excessive_spread"

        if tick.symbol in self._last_prices:
            last = self._last_prices[tick.symbol]
            change = abs(tick.last_price - last) / last
            if change > self.max_price_change:
                return False, "excessive_price_change"

        self._last_prices[tick.symbol] = tick.last_price
        return True, "ok"

OMS 核心设计
from enum import Enum
import uuid

class OrderStatus(Enum):
    PENDING = "pending"
    SUBMITTED = "submitted"
    PARTIALLY_FILLED = "partially_filled"
    FILLED = "filled"
    CANCELLED = "cancelled"
    REJECTED = "rejected"

@dataclass
class Order:
    order_id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    symbol: str = ""
    side: str = ""
    quantity: int = 0
    order_type: str = "limit"
    price: float = 0.0
    status: OrderStatus = OrderStatus.PENDING
    filled_quantity: int = 0
    avg_fill_price: float = 0.0
    created_at: float = 0.0
    updated_at: float = 0.0
    strategy_id: str = ""
    parent_order_id: str = ""

class OrderManagementSystem:
    """订单管理系统"""

    def __init__(self):
        self._orders: dict[str, Order] = {}
        self._active_orders: dict[str, Order] = {}
        self._positions: dict[str, dict] = defaultdict(lambda: {"quantity": 0, "avg_price": 0.0})

    def submit_order(self, order: Order) -> str:
        order.created_at = time.time()
        order.status = OrderStatus.SUBMITTED
        self._orders[order.order_id] = order
        self._active_orders[order.order_id] = order
        return order.order_id

    def on_fill(self, order_id: str, fill_qty: int, fill_price: float):
        order = self._orders.get(order_id)
        if not order:
            return

        total_filled = order.filled_quantity + fill_qty
        order.avg_fill_price = (
            (order.avg_fill_price * order.filled_quantity + fill_price * fill_qty) / total_filled
        )
        order.filled_quantity = total_filled
        order.updated_at = time.time()

        if total_filled >= order.quantity:
            order.status = OrderStatus.FILLED
            self._active_orders.pop(order_id, None)
        else:
            order.status = OrderStatus.PARTIALLY_FILLED

        self._update_position(order.symbol, order.side, fill_qty, fill_price)

    def _update_position(self, symbol: str, side: str, qty: int, price: float):
        pos = self._positions[symbol]
        direction = 1 if side == "buy" else -1
        new_qty = pos["quantity"] + direction * qty

        if pos["quantity"] == 0:
            pos["avg_price"] = price
        elif (pos["quantity"] > 0 and side == "buy") or (pos["quantity"] < 0 and side == "sell"):
            total_cost = pos["avg_price"] * abs(pos["quantity"]) + price * qty
            pos["avg_price"] = total_cost / (abs(pos["quantity"]) + qty)

        pos["quantity"] = new_qty

    def cancel_order(self, order_id: str) -> bool:
        order = self._active_orders.get(order_id)
        if not order:
            return False
        order.status = OrderStatus.CANCELLED
        order.updated_at = time.time()
        self._active_orders.pop(order_id, None)
        return True

    def get_active_orders(self, symbol: str = None) -> list[Order]:
        if symbol:
            return [o for o in self._active_orders.values() if o.symbol == symbol]
        return list(self._active_orders.values())

from abc import ABC, abstractmethod

class StrategyBase(ABC):
    """策略基类"""

    def __init__(self, strategy_id: str, universe: list[str]):
        self.strategy_id = strategy_id
        self.universe = universe
        self._oms: OrderManagementSystem = None
        self._data: MarketDataHandler = None

    def attach(self, oms: OrderManagementSystem, data: MarketDataHandler):
        self._oms = oms
        self._data = data

    @abstractmethod
    def on_tick(self, tick: MarketTick):
        pass

    @abstractmethod
    def on_bar(self, symbol: str, bar: dict):
        pass

    @abstractmethod
    def on_fill(self, order: Order, fill_qty: int, fill_price: float):
        pass

    def place_order(self, symbol: str, side: str, quantity: int,
                    order_type: str = "limit", price: float = 0.0) -> str:
        order = Order(
            symbol=symbol, side=side, quantity=quantity,
            order_type=order_type, price=price, strategy_id=self.strategy_id,
        )
        return self._oms.submit_order(order)

class StrategyEngine:
    """策略引擎"""

    def __init__(self, event_bus: EventBus, oms: OrderManagementSystem,
                 data: MarketDataHandler):
        self.event_bus = event_bus
        self.oms = oms
        self.data = data
        self._strategies: dict[str, StrategyBase] = {}

    def register_strategy(self, strategy: StrategyBase):
        strategy.attach(self.oms, self.data)
        self._strategies[strategy.strategy_id] = strategy
        self.event_bus.subscribe(EventType.TICK, lambda e: strategy.on_tick(e.data))
        self.event_bus.subscribe(EventType.BAR, lambda e: strategy.on_bar(e.data["symbol"], e.data))

class PreallocatedPool:
    """预分配对象池"""

    def __init__(self, factory: Callable, size: int = 10000):
        self._pool = [factory() for _ in range(size)]
        self._available = list(range(size))
        self._in_use = set()

    def acquire(self):
        if not self._available:
            raise RuntimeError("对象池耗尽")
        idx = self._available.pop()
        self._in_use.add(idx)
        return self._pool[idx]

    def release(self, obj):
        for idx, pooled in enumerate(self._pool):
            if pooled is obj and idx in self._in_use:
                self._in_use.remove(idx)
                self._available.append(idx)
                return

class HealthMonitor:
    """系统健康监控"""

    def __init__(self):
        self._component_status: dict[str, dict] = {}
        self._heartbeat_intervals: dict[str, float] = {}
        self._alert_handlers: list[Callable] = []

    def register_component(self, name: str, heartbeat_interval: float = 5.0):
        self._component_status[name] = {
            "status": "healthy",
            "last_heartbeat": time.time(),
            "error_count": 0,
        }
        self._heartbeat_intervals[name] = heartbeat_interval

    def heartbeat(self, name: str):
        if name in self._component_status:
            self._component_status[name]["last_heartbeat"] = time.time()
            self._component_status[name]["status"] = "healthy"
            self._component_status[name]["error_count"] = 0

    def check_health(self) -> dict:
        now = time.time()
        report = {}
        for name, status in self._component_status.items():
            interval = self._heartbeat_intervals[name]
            elapsed = now - status["last_heartbeat"]
            if elapsed > interval * 3:
                status["status"] = "critical"
                self._trigger_alert(name, "critical")
            elif elapsed > interval * 2:
                status["status"] = "warning"
            report[name] = status.copy()
        return report

    def _trigger_alert(self, component: str, severity: str):
        for handler in self._alert_handlers:
            handler(component, severity)

总结
交易系统架构设计是将量化策略从研究推向生产的工程核心。事件驱动架构提供了高效、可扩展的消息处理模式；市场数据子系统确保策略获得高质量、低延迟的输入；订单管理系统维护从订单创建到成交的全生命周期状态；策略引擎提供标准化的策略开发框架。在性能方面，延迟优化需要从算法、语言、系统和硬件四个层次综合考虑。在可靠性方面，健康监控、故障检测和自动恢复机制保障系统的持续运行。一个好的交易系统架构应该让策略开发者专注于策略逻辑，而将延迟优化、可靠性保障等基础设施关注点封装在框架层。
