实时风控与熔断机制

概述

实时风控是交易系统的"最后防线"。无论策略逻辑多么精密，都可能因为代码 Bug、数据异常、市场极端波动或人为操作失误而产生灾难性亏损。2012 年 Knight Capital 因软件故障在 45 分钟内亏损 4.4 亿美元的事件，充分说明了实时风控系统的重要性。本文系统梳理多层次风控架构设计，从单笔订单级别的基础检查，到策略级别的风险监控，再到系统级别的熔断机制。

订单级风控
单笔订单检查
订单级风控是最细粒度的检查层，对每一笔即将发出的订单进行合法性验证：


import numpy as np
import time
from dataclasses import dataclass
from enum import Enum
from typing import Optional

class RiskCheckResult(Enum):
    PASS = "pass"
    REJECT = "reject"
    WARNING = "warning"

@dataclass
class RiskCheckResponse:
    result: RiskCheckResult
    reason: str = ""
    details: dict = None

class OrderLevelRiskCheck:
    """订单级风控检查"""

    def __init__(self, config: dict):
        self.max_order_value = config.get("max_order_value", 1_000_000)
        self.max_order_quantity = config.get("max_order_quantity", 100_000)
        self.max_price_deviation = config.get("max_price_deviation", 0.05)
        self.allowed_symbols = config.get("allowed_symbols", None)

    def check(self, order: dict, market_data: dict) -> RiskCheckResponse:
        checks = [
            self._check_symbol(order),
            self._check_quantity(order),
            self._check_value(order),
            self._check_price(order, market_data),
            self._check_side(order),
        ]

        for check in checks:
            if check.result == RiskCheckResult.REJECT:
                return check

        return RiskCheckResponse(result=RiskCheckResult.PASS)

    def _check_symbol(self, order: dict) -> RiskCheckResponse:
        if self.allowed_symbols and order["symbol"] not in self.allowed_symbols:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"symbol_not_allowed: {order['symbol']}",
            )
        return RiskCheckResponse(result=RiskCheckResult.PASS)

    def _check_quantity(self, order: dict) -> RiskCheckResponse:
        if order["quantity"] > self.max_order_quantity:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"quantity_exceeded: {order['quantity']} > {self.max_order_quantity}",
            )
        if order["quantity"] <= 0:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason="invalid_quantity",
            )
        return RiskCheckResponse(result=RiskCheckResult.PASS)

    def _check_value(self, order: dict) -> RiskCheckResponse:
        value = order["quantity"] * order.get("price", 0)
        if value > self.max_order_value:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"value_exceeded: {value} > {self.max_order_value}",
            )
        return RiskCheckResponse(result=RiskCheckResult.PASS)

    def _check_price(self, order: dict, market_data: dict) -> RiskCheckResponse:
        if order.get("order_type") == "market":
            return RiskCheckResponse(result=RiskCheckResult.PASS)

        mid_price = market_data.get("mid_price", 0)
        if mid_price > 0:
            deviation = abs(order["price"] - mid_price) / mid_price
            if deviation > self.max_price_deviation:
                return RiskCheckResponse(
                    result=RiskCheckResult.REJECT,
                    reason=f"price_deviation: {deviation:.2%} > {self.max_price_deviation:.2%}",
                )
        return RiskCheckResponse(result=RiskCheckResult.PASS)

    def _check_side(self, order: dict) -> RiskCheckResponse:
        if order["side"] not in ["buy", "sell"]:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"invalid_side: {order['side']}",
            )
        return RiskCheckResponse(result=RiskCheckResult.PASS)
频率控制
class RateLimit:
    """下单频率控制"""

    def __init__(self, max_orders_per_second: int = 50, max_orders_per_minute: int = 500):
        self.max_per_second = max_orders_per_second
        self.max_per_minute = max_orders_per_minute
        self._second_counter = 0
        self._minute_counter = 0
        self._last_second = 0
        self._last_minute = 0

    def check(self) -> RiskCheckResponse:
        now = time.time()
        current_second = int(now)
        current_minute = int(now / 60)

        if current_second != self._last_second:
            self._second_counter = 0
            self._last_second = current_second

        if current_minute != self._last_minute:
            self._minute_counter = 0
            self._last_minute = current_minute

        self._second_counter += 1
        self._minute_counter += 1

        if self._second_counter > self.max_per_second:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"rate_limit_second: {self._second_counter}/{self.max_per_second}",
            )

        if self._minute_counter > self.max_per_minute:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"rate_limit_minute: {self._minute_counter}/{self.max_per_minute}",
            )

        return RiskCheckResponse(result=RiskCheckResult.PASS)

class StrategyRiskMonitor:
    """策略级风控监控"""

    def __init__(self, config: dict):
        self.max_position_value = config.get("max_position_value", 5_000_000)
        self.max_gross_exposure = config.get("max_gross_exposure", 10_000_000)
        self.max_net_exposure = config.get("max_net_exposure", 2_000_000)
        self.max_concentration = config.get("max_concentration", 0.20)
        self.max_drawdown = config.get("max_drawdown", 0.05)
        self._peak_value = 0
        self._strategy_pnl = 0

    def check_position(self, positions: dict[str, dict], prices: dict[str, float]) -> RiskCheckResponse:
        gross_exposure = 0
        net_exposure = 0
        max_single = 0

        for symbol, pos in positions.items():
            value = abs(pos["quantity"]) * prices.get(symbol, 0)
            signed_value = pos["quantity"] * prices.get(symbol, 0)
            gross_exposure += value
            net_exposure += signed_value
            max_single = max(max_single, value)

        if gross_exposure > self.max_gross_exposure:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"gross_exposure: {gross_exposure:,.0f} > {self.max_gross_exposure:,.0f}",
            )

        if abs(net_exposure) > self.max_net_exposure:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"net_exposure: {abs(net_exposure):,.0f} > {self.max_net_exposure:,.0f}",
            )

        if gross_exposure > 0 and max_single / gross_exposure > self.max_concentration:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"concentration: {max_single / gross_exposure:.1%} > {self.max_concentration:.1%}",
            )

        return RiskCheckResponse(result=RiskCheckResult.PASS)

    def check_drawdown(self, current_value: float) -> RiskCheckResponse:
        self._peak_value = max(self._peak_value, current_value)
        drawdown = (self._peak_value - current_value) / self._peak_value

        if drawdown > self.max_drawdown:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"max_drawdown_breached: {drawdown:.2%} > {self.max_drawdown:.2%}",
            )

        if drawdown > self.max_drawdown * 0.8:
            return RiskCheckResponse(
                result=RiskCheckResult.WARNING,
                reason=f"drawdown_warning: {drawdown:.2%} approaching limit",
            )

        return RiskCheckResponse(result=RiskCheckResult.PASS)
损益监控
class PnLMonitor:
    """损益实时监控"""

    def __init__(self, daily_loss_limit: float = 100_000, trailing_loss_limit: float = 50_000):
        self.daily_loss_limit = daily_loss_limit
        self.trailing_loss_limit = trailing_loss_limit
        self._daily_pnl = 0.0
        self._peak_daily_pnl = 0.0
        self._last_reset_date = None

    def update(self, pnl_change: float, current_date: str):
        if current_date != self._last_reset_date:
            self._daily_pnl = 0.0
            self._peak_daily_pnl = 0.0
            self._last_reset_date = current_date

        self._daily_pnl += pnl_change
        self._peak_daily_pnl = max(self._peak_daily_pnl, self._daily_pnl)

    def check(self) -> RiskCheckResponse:
        if self._daily_pnl < -self.daily_loss_limit:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"daily_loss_limit: {self._daily_pnl:,.0f} < -{self.daily_loss_limit:,.0f}",
            )

        trailing_loss = self._peak_daily_pnl - self._daily_pnl
        if trailing_loss > self.trailing_loss_limit:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"trailing_loss: {trailing_loss:,.0f} > {self.trailing_loss_limit:,.0f}",
            )

        return RiskCheckResponse(result=RiskCheckResult.PASS)
系统级风控与熔断
多级熔断机制
class CircuitBreaker:
    """系统级熔断机制"""

    def __init__(self, config: dict):
        self.system_loss_limit = config.get("system_loss_limit", 500_000)
        self.max_open_orders = config.get("max_open_orders", 1000)
        self.max_message_rate = config.get("max_message_rate", 10000)
        self.heartbeat_timeout = config.get("heartbeat_timeout", 5.0)
        self._is_tripped = False
        self._trip_reason = ""
        self._trip_time = None
        self._level = 0

    def evaluate(self, system_state: dict) -> dict:
        checks = [
            self._check_system_pnl(system_state.get("total_pnl", 0)),
            self._check_open_orders(system_state.get("open_orders", 0)),
            self._check_connectivity(system_state.get("feed_latency_ms", 0)),
            self._check_data_quality(system_state.get("stale_symbols", 0),
                                     system_state.get("total_symbols", 1)),
        ]

        max_level = 0
        reasons = []
        for check in checks:
            if check["level"] > max_level:
                max_level = check["level"]
            if check["level"] > 0:
                reasons.append(check["reason"])

        if max_level >= 4 and not self._is_tripped:
            self._is_tripped = True
            self._trip_reason = "; ".join(reasons)
            self._trip_time = time.time()

        self._level = max_level

        return {
            "level": max_level,
            "is_tripped": self._is_tripped,
            "reasons": reasons,
            "action": self._level_to_action(max_level),
        }

    def _check_system_pnl(self, total_pnl: float) -> dict:
        if total_pnl < -self.system_loss_limit:
            return {"level": 4, "reason": f"system_loss: {total_pnl:,.0f}"}
        elif total_pnl < -self.system_loss_limit * 0.8:
            return {"level": 3, "reason": f"system_loss_warning: {total_pnl:,.0f}"}
        elif total_pnl < -self.system_loss_limit * 0.5:
            return {"level": 2, "reason": f"system_loss_caution: {total_pnl:,.0f}"}
        return {"level": 0, "reason": ""}

    def _check_open_orders(self, count: int) -> dict:
        if count > self.max_open_orders:
            return {"level": 3, "reason": f"open_orders_exceeded: {count}"}
        return {"level": 0, "reason": ""}

    def _check_connectivity(self, feed_latency_ms: float) -> dict:
        if feed_latency_ms > self.heartbeat_timeout * 1000:
            return {"level": 4, "reason": f"feed_disconnected: {feed_latency_ms}ms"}
        elif feed_latency_ms > 1000:
            return {"level": 2, "reason": f"feed_delayed: {feed_latency_ms}ms"}
        return {"level": 0, "reason": ""}

    def _check_data_quality(self, stale_count: int, total_count: int) -> dict:
        stale_ratio = stale_count / max(total_count, 1)
        if stale_ratio > 0.5:
            return {"level": 3, "reason": f"data_quality: {stale_ratio:.0%} stale"}
        elif stale_ratio > 0.2:
            return {"level": 1, "reason": f"data_quality_warning: {stale_ratio:.0%} stale"}
        return {"level": 0, "reason": ""}

    def _level_to_action(self, level: int) -> str:
        actions = {
            0: "normal",
            1: "log_and_alert",
            2: "reduce_limits",
            3: "stop_new_orders_and_flatten",
            4: "emergency_flatten_and_disconnect",
        }
        return actions.get(level, "unknown")

    def reset(self, authorized_by: str):
        self._is_tripped = False
        self._trip_reason = ""
        self._level = 0

异常场景处理
Fat Finger 防护
"胖手指"错误（Fat Finger）是指交易员因操作失误下达了远超预期的订单：


class FatFingerProtection:
    """胖手指防护"""

    def __init__(self, max_notional_pct_of_adv: float = 0.05,
                 max_price_deviation_pct: float = 0.03):
        self.max_adv_pct = max_notional_pct_of_adv
        self.max_price_dev = max_price_deviation_pct

    def check(self, order: dict, market_info: dict) -> RiskCheckResponse:
        notional = order["quantity"] * order.get("price", market_info.get("last_price", 0))
        adv_notional = market_info.get("adv", 1) * market_info.get("last_price", 1)

        if adv_notional > 0 and notional / adv_notional > self.max_adv_pct:
            return RiskCheckResponse(
                result=RiskCheckResult.REJECT,
                reason=f"fat_finger_size: {notional / adv_notional:.1%} of ADV",
            )

        if order.get("price") and market_info.get("last_price"):
            dev = abs(order["price"] - market_info["last_price"]) / market_info["last_price"]
            if dev > self.max_price_dev:
                return RiskCheckResponse(
                    result=RiskCheckResult.REJECT,
                    reason=f"fat_finger_price: {dev:.1%} from market",
                )

        return RiskCheckResponse(result=RiskCheckResult.PASS)

风控审计与回顾
class RiskAuditLog:
    """风控审计日志"""

    def __init__(self):
        self._logs: list[dict] = []

    def record(self, event_type: str, details: dict, response: RiskCheckResponse):
        self._logs.append({
            "timestamp": time.time(),
            "event_type": event_type,
            "result": response.result.value,
            "reason": response.reason,
            "details": details,
        })

    def daily_summary(self, date: str) -> dict:
        day_logs = [l for l in self._logs
                    if time.strftime("%Y-%m-%d", time.localtime(l["timestamp"])) == date]

        total_checks = len(day_logs)
        rejections = [l for l in day_logs if l["result"] == "reject"]
        warnings = [l for l in day_logs if l["result"] == "warning"]

        rejection_reasons = {}
        for r in rejections:
            reason = r["reason"].split(":")[0]
            rejection_reasons[reason] = rejection_reasons.get(reason, 0) + 1

        return {
            "date": date,
            "total_checks": total_checks,
            "rejections": len(rejections),
            "warnings": len(warnings),
            "rejection_rate": len(rejections) / max(total_checks, 1),
            "top_rejection_reasons": sorted(rejection_reasons.items(), key=lambda x: -x[1])[:5],
        }
总结
实时风控是交易系统抵御灾难性亏损的最后防线。多层风控架构从订单级（单笔合法性检查）、策略级（持仓敞口和回撤监控）到系统级（全局熔断机制）逐层递进，每一层都有明确的检查项和处理策略。关键设计原则包括：风控检查必须在订单发出之前完成、熔断机制必须能在毫秒级时间内切断所有交易活动、所有风控事件必须有完整的审计日志、以及熔断后的恢复必须经过人工授权。Knight Capital 的教训告诉我们：在交易系统中，风控不是锦上添花的功能，而是关乎生存的基础设施。
