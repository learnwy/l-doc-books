统计套利与配对交易

概述

统计套利（Statistical Arbitrage，简称 StatArb）是一类基于统计关系进行多空组合交易的策略。不同于传统套利利用确定性的价格关系，统计套利利用的是资产之间的统计关联——这种关联在大部分时间成立，但并非始终如此。配对交易（Pairs Trading）是统计套利的最经典形式，通过同时做多一只相对低估的股票和做空一只相对高估的股票来获取价差回归的收益。

统计套利最早由 Morgan Stanley 的量化交易团队在 20 世纪 80 年代开发，后来被 D.E. Shaw、Renaissance Technologies 等量化对冲基金发扬光大。时至今日，统计套利仍然是全球量化对冲基金的核心策略之一。

配对交易的理论基础
协整（Cointegration）
配对交易的统计基础是协整理论。两个非平稳时间序列 XtX_tXt​ 和 YtY_tYt​ 是协整的，如果存在常数 β\betaβ，使得 Zt=Yt−βXtZ_t = Y_t - \beta X_tZt​=Yt​−βXt​ 是平稳的。换句话说，虽然两个价格序列各自是随机游走的，但它们之间的线性组合却表现出均值回归特性。

Engle-Granger 两步法
import numpy as np
import pandas as pd
from statsmodels.tsa.stattools import adfuller, coint
from statsmodels.regression.linear_model import OLS

class PairSelector:
    """配对选择器"""

    def engle_granger_test(self, y: pd.Series, x: pd.Series) -> dict:
        model = OLS(y, x).fit()
        residuals = model.resid
        hedge_ratio = model.params[0]
        adf_stat, adf_pvalue, _, _, critical_values, _ = adfuller(residuals)
        return {
            "hedge_ratio": hedge_ratio,
            "adf_statistic": adf_stat,
            "p_value": adf_pvalue,
            "critical_values": critical_values,
            "is_cointegrated": adf_pvalue < 0.05,
            "residuals": residuals,
        }

    def find_pairs(self, prices: pd.DataFrame, significance: float = 0.05) -> list:
        n = prices.shape[1]
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                score, pvalue, _ = coint(prices.iloc[:, i], prices.iloc[:, j])
                if pvalue < significance:
                    pairs.append({
                        "asset_1": prices.columns[i],
                        "asset_2": prices.columns[j],
                        "coint_score": score,
                        "p_value": pvalue,
                    })
        pairs.sort(key=lambda x: x["p_value"])
        return pairs
Johansen 检验
当需要处理多资产协整关系时，Johansen 检验比 Engle-Granger 方法更为强大：


from statsmodels.tsa.vector_ar.vecm import coint_johansen

class JohansenAnalyzer:
    """Johansen 协整分析"""

    def test(self, data: pd.DataFrame, det_order: int = 0) -> dict:
        result = coint_johansen(data, det_order=det_order, k_ar_diff=1)
        trace_stats = result.lr1
        critical_values_95 = result.cvt[:, 1]
        n_coint = sum(trace_stats > critical_values_95)
        eigenvectors = result.evec
        return {
            "n_cointegrating_vectors": n_coint,
            "trace_statistics": trace_stats,
            "critical_values_95": critical_values_95,
            "eigenvectors": eigenvectors,
        }

    def portfolio_weights(self, data: pd.DataFrame) -> np.ndarray:
        result = coint_johansen(data, det_order=0, k_ar_diff=1)
        weights = result.evec[:, 0]
        weights = weights / np.abs(weights).sum()
        return weights
配对交易的实现方法
距离法（Distance Approach）
距离法是最早的配对交易方法，由 Gatev、Goetzmann 和 Rouwenhorst（2006）系统化：


class DistanceMethod:
    """距离法配对交易"""

    def __init__(self, formation_period: int = 252, trading_period: int = 126):
        self.formation_period = formation_period
        self.trading_period = trading_period

    def normalize_prices(self, prices: pd.DataFrame) -> pd.DataFrame:
        return prices / prices.iloc[0]

    def find_closest_pairs(self, prices: pd.DataFrame, n_pairs: int = 20) -> list:
        normalized = self.normalize_prices(prices)
        n = normalized.shape[1]
        distances = []

        for i in range(n):
            for j in range(i + 1, n):
                ssd = ((normalized.iloc[:, i] - normalized.iloc[:, j]) ** 2).sum()
                distances.append({
                    "asset_1": normalized.columns[i],
                    "asset_2": normalized.columns[j],
                    "ssd": ssd,
                })

        distances.sort(key=lambda x: x["ssd"])
        return distances[:n_pairs]

    def trade_pair(self, p1: pd.Series, p2: pd.Series, threshold: float = 2.0) -> pd.DataFrame:
        spread = self.normalize_prices(pd.DataFrame({"a": p1, "b": p2}))
        diff = spread["a"] - spread["b"]
        mean = diff.rolling(60).mean()
        std = diff.rolling(60).std()
        zscore = (diff - mean) / std

        positions = pd.DataFrame(0.0, index=p1.index, columns=["asset_1", "asset_2"])
        positions.loc[zscore > threshold, "asset_1"] = -1.0
        positions.loc[zscore > threshold, "asset_2"] = 1.0
        positions.loc[zscore < -threshold, "asset_1"] = 1.0
        positions.loc[zscore < -threshold, "asset_2"] = -1.0

        return positions
协整法（Cointegration Approach）
class CointegrationPairTrader:
    """协整配对交易"""

    def __init__(self, lookback: int = 252, entry_z: float = 2.0, exit_z: float = 0.5,
                 stop_z: float = 4.0):
        self.lookback = lookback
        self.entry_z = entry_z
        self.exit_z = exit_z
        self.stop_z = stop_z

    def dynamic_hedge_ratio(self, y: pd.Series, x: pd.Series) -> pd.Series:
        hedge_ratios = pd.Series(np.nan, index=y.index)
        for i in range(self.lookback, len(y)):
            window_y = y.iloc[i - self.lookback:i]
            window_x = x.iloc[i - self.lookback:i]
            model = OLS(window_y, window_x).fit()
            hedge_ratios.iloc[i] = model.params[0]
        return hedge_ratios

    def compute_spread(self, y: pd.Series, x: pd.Series) -> pd.DataFrame:
        hedge_ratios = self.dynamic_hedge_ratio(y, x)
        spread = y - hedge_ratios * x
        spread_mean = spread.rolling(self.lookback).mean()
        spread_std = spread.rolling(self.lookback).std()
        zscore = (spread - spread_mean) / spread_std
        return pd.DataFrame({
            "spread": spread,
            "zscore": zscore,
            "hedge_ratio": hedge_ratios,
        })

    def generate_signals(self, y: pd.Series, x: pd.Series) -> pd.DataFrame:
        data = self.compute_spread(y, x)
        zscore = data["zscore"]
        hedge_ratio = data["hedge_ratio"]

        pos_y = pd.Series(0.0, index=y.index)
        pos_x = pd.Series(0.0, index=x.index)
        current_pos = 0.0

        for i in range(self.lookback, len(y)):
            z = zscore.iloc[i]
            hr = hedge_ratio.iloc[i]

            if abs(z) > self.stop_z:
                current_pos = 0.0
            elif current_pos == 0:
                if z < -self.entry_z:
                    current_pos = 1.0
                elif z > self.entry_z:
                    current_pos = -1.0
            elif abs(z) < self.exit_z:
                current_pos = 0.0

            pos_y.iloc[i] = current_pos
            pos_x.iloc[i] = -current_pos * hr

        return pd.DataFrame({"pos_y": pos_y, "pos_x": pos_x})
PCA 多因子统计套利
更高级的统计套利不局限于两两配对，而是使用主成分分析（PCA）提取一组股票的共同因子，然后交易偏离因子模型的残差。


from sklearn.decomposition import PCA

class PCAStatArb:
    """PCA 多因子统计套利"""

    def __init__(self, n_components: int = 5, lookback: int = 252, entry_z: float = 1.5):
        self.n_components = n_components
        self.lookback = lookback
        self.entry_z = entry_z

    def fit_factor_model(self, returns: pd.DataFrame) -> dict:
        pca = PCA(n_components=self.n_components)
        factors = pca.fit_transform(returns)
        loadings = pca.components_.T
        explained_variance = pca.explained_variance_ratio_
        fitted = returns.values @ loadings @ np.linalg.pinv(loadings)
        residuals = returns.values - fitted
        return {
            "factors": factors,
            "loadings": loadings,
            "explained_variance": explained_variance,
            "residuals": pd.DataFrame(residuals, index=returns.index, columns=returns.columns),
        }

    def generate_signals(self, returns: pd.DataFrame) -> pd.DataFrame:
        result = self.fit_factor_model(returns.iloc[-self.lookback:])
        residuals = result["residuals"]
        cum_residuals = residuals.cumsum()
        zscore = (cum_residuals.iloc[-1] - cum_residuals.mean()) / cum_residuals.std()

        signals = pd.Series(0.0, index=returns.columns)
        signals[zscore < -self.entry_z] = 1.0
        signals[zscore > self.entry_z] = -1.0
        signals = signals / signals.abs().sum()
        return signals
ETF 套利
ETF 套利利用 ETF 价格与其底层资产净值之间的偏离：


class ETFArbitrage:
    """ETF 套利策略"""

    def __init__(self, creation_cost: float = 0.001, redemption_cost: float = 0.001):
        self.creation_cost = creation_cost
        self.redemption_cost = redemption_cost

    def compute_premium_discount(self, etf_price: pd.Series, nav: pd.Series) -> pd.Series:
        return (etf_price - nav) / nav

    def generate_signals(self, etf_price: pd.Series, nav: pd.Series) -> pd.Series:
        premium = self.compute_premium_discount(etf_price, nav)
        signal = pd.Series(0.0, index=etf_price.index)
        signal[premium > self.creation_cost * 2] = -1.0
        signal[premium < -self.redemption_cost * 2] = 1.0
        return signal

class RobustPairSelection:
    """稳健的配对选择"""

    def __init__(self, n_windows: int = 4, min_pass_rate: float = 0.75):
        self.n_windows = n_windows
        self.min_pass_rate = min_pass_rate

    def stability_test(self, y: pd.Series, x: pd.Series, window_size: int = 252) -> dict:
        total_length = len(y)
        step = (total_length - window_size) // self.n_windows
        pass_count = 0

        for i in range(self.n_windows):
            start = i * step
            end = start + window_size
            _, pvalue, _ = coint(y.iloc[start:end], x.iloc[start:end])
            if pvalue < 0.05:
                pass_count += 1

        pass_rate = pass_count / self.n_windows
        return {
            "pass_rate": pass_rate,
            "is_stable": pass_rate >= self.min_pass_rate,
        }

    def bonferroni_correction(self, p_values: list, alpha: float = 0.05) -> list:
        n_tests = len(p_values)
        adjusted_alpha = alpha / n_tests
        return [p < adjusted_alpha for p in p_values]
统计套利的风险管理
头寸规模与风险控制
class StatArbRiskManager:
    """统计套利风险管理"""

    def __init__(self, max_pair_risk: float = 0.02, max_total_risk: float = 0.10):
        self.max_pair_risk = max_pair_risk
        self.max_total_risk = max_total_risk

    def size_position(self, spread_vol: float, capital: float) -> float:
        target_risk = self.max_pair_risk * capital
        position_size = target_risk / (spread_vol * np.sqrt(252))
        return position_size

    def portfolio_risk_check(self, pair_positions: list, correlation_matrix: np.ndarray) -> dict:
        weights = np.array([p["weight"] for p in pair_positions])
        vols = np.array([p["vol"] for p in pair_positions])
        portfolio_vol = np.sqrt(weights @ np.diag(vols ** 2) @ weights.T +
                                2 * weights @ (correlation_matrix * np.outer(vols, vols)) @ weights.T)
        return {
            "portfolio_vol": portfolio_vol,
            "within_limit": portfolio_vol < self.max_total_risk,
        }

总结
统计套利与配对交易通过利用资产间的统计关系构建市场中性组合。从最简单的距离法到基于协整的动态对冲比率策略，再到 PCA 多因子模型和 ETF 套利，策略的复杂度与潜在收益能力不断提升。成功的统计套利需要：严格的协整检验与多重检验修正以避免数据挖掘、具有经济逻辑的配对选择、自适应的对冲比率估计、以及针对配对关系断裂和借券风险的完善风控体系。作为天然的市场中性策略，统计套利是量化组合中不可或缺的配置。
