TWAP/VWAP 算法与变体

概述

TWAP（Time-Weighted Average Price）和 VWAP（Volume-Weighted Average Price）是两种最基础、最广泛使用的执行算法。它们的核心思想是将大额订单拆分为小单，在一段时间内均匀或按成交量分布执行，以降低市场冲击。据估计，全球股票市场约 40-50% 的机构订单通过 TWAP/VWAP 算法执行。

TWAP 算法
基本原理
TWAP 算法将订单在指定时间段内均匀分布执行，不考虑市场成交量的变化。其目标是使平均执行价格尽可能接近时间加权平均价格。


标准实现
import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Optional

@dataclass
class Order:
    symbol: str
    side: str
    total_quantity: int
    start_time: pd.Timestamp
    end_time: pd.Timestamp

class TWAPAlgorithm:
    """标准 TWAP 执行算法"""

    def __init__(self, num_slices: int = 100, randomize: bool = True):
        self.num_slices = num_slices
        self.randomize = randomize

    def create_schedule(self, order: Order) -> pd.DataFrame:
        total_duration = (order.end_time - order.start_time).total_seconds()
        slice_duration = total_duration / self.num_slices
        base_quantity = order.total_quantity // self.num_slices
        remainder = order.total_quantity % self.num_slices

        schedule = []
        for i in range(self.num_slices):
            slice_time = order.start_time + pd.Timedelta(seconds=i * slice_duration)
            qty = base_quantity + (1 if i < remainder else 0)

            if self.randomize:
                time_jitter = np.random.uniform(-slice_duration * 0.3, slice_duration * 0.3)
                slice_time += pd.Timedelta(seconds=time_jitter)
                qty_jitter = int(np.random.uniform(-qty * 0.2, qty * 0.2))
                qty = max(1, qty + qty_jitter)

            schedule.append({
                "time": slice_time,
                "quantity": qty,
                "slice_id": i,
            })

        df = pd.DataFrame(schedule)
        scale_factor = order.total_quantity / df["quantity"].sum()
        df["quantity"] = (df["quantity"] * scale_factor).round().astype(int)
        diff = order.total_quantity - df["quantity"].sum()
        df.loc[df.index[-1], "quantity"] += diff
        return df.sort_values("time").reset_index(drop=True)

VWAP 算法
基本原理
VWAP 算法按照市场的历史或预测成交量分布来分配执行节奏。在成交量大的时段多执行，在成交量小的时段少执行，使平均执行价格接近全天成交量加权平均价。


成交量分布预测
class VolumeProfilePredictor:
    """日内成交量分布预测"""

    def __init__(self, lookback_days: int = 20, n_buckets: int = 78):
        self.lookback_days = lookback_days
        self.n_buckets = n_buckets

    def historical_profile(self, intraday_volumes: pd.DataFrame) -> np.ndarray:
        daily_totals = intraday_volumes.sum(axis=1)
        normalized = intraday_volumes.div(daily_totals, axis=0)
        profile = normalized.iloc[-self.lookback_days:].mean().values
        profile = profile / profile.sum()
        return profile

    def adjusted_profile(self, base_profile: np.ndarray, current_volumes: np.ndarray,
                         elapsed_buckets: int) -> np.ndarray:
        if elapsed_buckets == 0:
            return base_profile

        actual_pct = current_volumes[:elapsed_buckets].sum() / current_volumes[:elapsed_buckets].sum()
        expected_pct = base_profile[:elapsed_buckets].sum()

        remaining_profile = base_profile[elapsed_buckets:]
        if remaining_profile.sum() > 0:
            remaining_profile = remaining_profile / remaining_profile.sum()
        return remaining_profile
VWAP 执行引擎
class VWAPAlgorithm:
    """VWAP 执行算法"""

    def __init__(self, volume_predictor: VolumeProfilePredictor, participation_limit: float = 0.15):
        self.volume_predictor = volume_predictor
        self.participation_limit = participation_limit

    def create_schedule(self, order: Order, volume_profile: np.ndarray) -> pd.DataFrame:
        n_buckets = len(volume_profile)
        target_quantities = (volume_profile * order.total_quantity).round().astype(int)
        diff = order.total_quantity - target_quantities.sum()
        max_idx = np.argmax(volume_profile)
        target_quantities[max_idx] += diff

        total_duration = (order.end_time - order.start_time).total_seconds()
        bucket_duration = total_duration / n_buckets

        schedule = []
        for i in range(n_buckets):
            bucket_time = order.start_time + pd.Timedelta(seconds=i * bucket_duration)
            schedule.append({
                "time": bucket_time,
                "quantity": int(target_quantities[i]),
                "volume_weight": volume_profile[i],
                "bucket_id": i,
            })

        return pd.DataFrame(schedule)

    def real_time_adjustment(self, target_qty: int, actual_market_volume: int) -> int:
        max_qty = int(actual_market_volume * self.participation_limit)
        return min(target_qty, max_qty)

TWAP/VWAP 的高级变体
参与率算法（Percentage of Volume）
参与率算法不预设执行时间表，而是实时跟踪市场成交量，保持固定的参与率：


class POVAlgorithm:
    """参与率算法"""

    def __init__(self, target_rate: float = 0.10, max_rate: float = 0.20):
        self.target_rate = target_rate
        self.max_rate = max_rate

    def calculate_slice(self, market_volume_since_last: int, remaining_qty: int) -> int:
        target_qty = int(market_volume_since_last * self.target_rate)
        max_qty = int(market_volume_since_last * self.max_rate)
        slice_qty = min(target_qty, remaining_qty)
        slice_qty = min(slice_qty, max_qty)
        return max(0, slice_qty)

    def should_accelerate(self, executed_qty: int, total_qty: int,
                          elapsed_pct: float) -> bool:
        completion_pct = executed_qty / total_qty
        return completion_pct < elapsed_pct * 0.8
自适应 VWAP
自适应 VWAP 根据实时市场状况调整执行节奏：


class AdaptiveVWAP:
    """自适应 VWAP"""

    def __init__(self, urgency: float = 0.5):
        self.urgency = urgency

    def price_adaptive_quantity(self, base_qty: int, current_price: float,
                                arrival_price: float, side: str) -> int:
        price_diff = (current_price - arrival_price) / arrival_price

        if side == "buy":
            if price_diff < -0.005:
                multiplier = 1.5
            elif price_diff > 0.005:
                multiplier = 0.5
            else:
                multiplier = 1.0
        else:
            if price_diff > 0.005:
                multiplier = 1.5
            elif price_diff < -0.005:
                multiplier = 0.5
            else:
                multiplier = 1.0

        urgency_factor = 0.5 + self.urgency
        adjusted_qty = int(base_qty * multiplier * urgency_factor)
        return max(1, adjusted_qty)

    def volume_adaptive_quantity(self, base_qty: int, actual_volume: int,
                                  expected_volume: int) -> int:
        volume_ratio = actual_volume / max(expected_volume, 1)
        if volume_ratio > 1.5:
            return int(base_qty * 1.3)
        elif volume_ratio < 0.5:
            return int(base_qty * 0.7)
        return base_qty
执行质量评估
VWAP 偏差分析
class ExecutionAnalyzer:
    """执行质量分析"""

    def vwap_slippage(self, executed_prices: pd.Series, executed_volumes: pd.Series,
                      market_vwap: float) -> dict:
        exec_vwap = (executed_prices * executed_volumes).sum() / executed_volumes.sum()
        slippage_bps = (exec_vwap - market_vwap) / market_vwap * 10000
        return {
            "exec_vwap": exec_vwap,
            "market_vwap": market_vwap,
            "slippage_bps": slippage_bps,
        }

    def participation_rate(self, executed_volumes: pd.Series,
                           market_volumes: pd.Series) -> pd.Series:
        return executed_volumes / market_volumes

    def timing_analysis(self, slice_times: pd.Series, slice_prices: pd.Series,
                        market_prices: pd.DataFrame) -> dict:
        early_half = slice_prices.iloc[:len(slice_prices) // 2].mean()
        late_half = slice_prices.iloc[len(slice_prices) // 2:].mean()
        full_day_avg = market_prices.mean()
        return {
            "early_avg": early_half,
            "late_avg": late_half,
            "full_day_avg": full_day_avg,
            "front_loading_impact_bps": (early_half - full_day_avg) / full_day_avg * 10000,
        }

反检测增强
class AntiGamingEnhancement:
    """反博弈增强"""

    def __init__(self, randomness_level: float = 0.3):
        self.randomness_level = randomness_level

    def randomize_timing(self, base_times: pd.Series, avg_interval: float) -> pd.Series:
        jitter = np.random.exponential(avg_interval * self.randomness_level, len(base_times))
        signs = np.random.choice([-1, 1], len(base_times))
        return base_times + pd.to_timedelta(jitter * signs, unit="s")

    def randomize_quantity(self, base_quantities: np.ndarray) -> np.ndarray:
        noise = np.random.normal(1.0, self.randomness_level, len(base_quantities))
        randomized = (base_quantities * noise).round().astype(int)
        randomized = np.maximum(randomized, 1)
        scale = base_quantities.sum() / randomized.sum()
        return (randomized * scale).round().astype(int)

    def iceberg_split(self, visible_qty: int, total_qty: int) -> list:
        visible_ratio = visible_qty / total_qty
        slices = []
        remaining = total_qty
        while remaining > 0:
            visible = min(int(visible_qty * np.random.uniform(0.5, 1.5)), remaining)
            slices.append(visible)
            remaining -= visible
        return slices
特殊场景处理
收盘竞价
许多指数基金需要在收盘竞价中执行，这是流动性最集中的时段：


class ClosingAuction:
    """收盘竞价执行"""

    def __init__(self, auction_participation: float = 0.05):
        self.auction_participation = auction_participation

    def estimate_auction_volume(self, historical_auction_volumes: pd.Series) -> float:
        return historical_auction_volumes.rolling(20).mean().iloc[-1]

    def auction_order_size(self, total_remaining: int, estimated_volume: float) -> int:
        max_auction = int(estimated_volume * self.auction_participation)
        return min(total_remaining, max_auction)
开盘策略
class OpeningStrategy:
    """开盘策略"""

    def opening_allocation(self, total_qty: int, open_volume_pct: float = 0.08) -> dict:
        opening_qty = int(total_qty * min(open_volume_pct, 0.15))
        continuous_qty = total_qty - opening_qty
        return {
            "opening_auction": opening_qty,
            "continuous_trading": continuous_qty,
        }
总结
TWAP 和 VWAP 是执行算法的两大基石。TWAP 追求时间维度的均匀执行，适合流动性稳定的市场；VWAP 根据成交量分布调整节奏，在大多数股票市场中表现更优。参与率算法和自适应 VWAP 在此基础上引入实时市场反馈，进一步优化执行效果。在实践中，关键考量包括：根据订单紧迫度和市场流动性选择合适的算法类型、通过随机化技术防止执行模式被高频交易者识别、以及建立完善的执行质量评估体系来持续改进算法性能。
