智能订单路由

概述

智能订单路由（Smart Order Routing，SOR）是在多交易场所环境下，自动选择最优场所执行订单的技术。随着全球金融市场的碎片化——美国有十多个股票交易所和数十个暗池、欧洲在 MiFID II 后也形成了多交易场所竞争格局——SOR 已成为机构交易的标配能力。一个优秀的 SOR 系统需要在毫秒级时间内综合考虑价格、流动性、费用、延迟等因素，做出最优路由决策。

NBBO 与最优执行义务
National Best Bid and Offer（NBBO）是所有受保护报价的最优买卖价格。Reg NMS 的 Order Protection Rule 要求路由必须尊重各场所的受保护报价：


from dataclasses import dataclass

@dataclass
class Quote:
    venue: str
    bid: float
    ask: float
    bid_size: int
    ask_size: int
    latency_us: int
    fee_per_share: float
    rebate_per_share: float

class NBBOCalculator:
    """NBBO 计算器"""

    def compute_nbbo(self, quotes: list[Quote]) -> dict:
        valid_quotes = [q for q in quotes if q.bid_size > 0 and q.ask_size > 0]
        if not valid_quotes:
            return None

        best_bid = max(valid_quotes, key=lambda q: q.bid)
        best_ask = min(valid_quotes, key=lambda q: q.ask)

        total_bid_size = sum(q.bid_size for q in valid_quotes if q.bid == best_bid.bid)
        total_ask_size = sum(q.ask_size for q in valid_quotes if q.ask == best_ask.ask)

        return {
            "nbbo_bid": best_bid.bid,
            "nbbo_ask": best_ask.ask,
            "nbbo_spread": best_ask.ask - best_bid.bid,
            "nbbo_bid_size": total_bid_size,
            "nbbo_ask_size": total_ask_size,
            "best_bid_venue": best_bid.venue,
            "best_ask_venue": best_ask.venue,
        }
SOR 路由策略
基于价格的路由
最基本的 SOR 策略是价格优先——将订单发送到提供最优价格的场所：


import numpy as np
from typing import Optional

class PriceBasedRouter:
    """基于价格的路由策略"""

    def __init__(self, max_venues: int = 5):
        self.max_venues = max_venues

    def route_order(self, side: str, quantity: int, quotes: list[Quote]) -> list[dict]:
        if side == "buy":
            sorted_quotes = sorted(quotes, key=lambda q: q.ask)
            price_field = "ask"
            size_field = "ask_size"
        else:
            sorted_quotes = sorted(quotes, key=lambda q: -q.bid)
            price_field = "bid"
            size_field = "bid_size"

        routes = []
        remaining = quantity

        for q in sorted_quotes[:self.max_venues]:
            if remaining <= 0:
                break
            available = getattr(q, size_field)
            fill_qty = min(remaining, available)
            routes.append({
                "venue": q.venue,
                "quantity": fill_qty,
                "price": getattr(q, price_field),
                "expected_fee": fill_qty * q.fee_per_share,
            })
            remaining -= fill_qty

        return routes
净价路由
净价路由不仅考虑报价，还考虑交易费用和回扣：


class NetPriceRouter:
    """净价路由策略"""

    def effective_price(self, quote: Quote, side: str) -> float:
        if side == "buy":
            return quote.ask + quote.fee_per_share - quote.rebate_per_share
        else:
            return quote.bid - quote.fee_per_share + quote.rebate_per_share

    def route_order(self, side: str, quantity: int, quotes: list[Quote]) -> list[dict]:
        if side == "buy":
            scored = sorted(quotes, key=lambda q: self.effective_price(q, "buy"))
        else:
            scored = sorted(quotes, key=lambda q: -self.effective_price(q, "sell"))

        routes = []
        remaining = quantity

        for q in scored:
            if remaining <= 0:
                break
            size = q.ask_size if side == "buy" else q.bid_size
            fill_qty = min(remaining, size)
            eff_price = self.effective_price(q, side)
            routes.append({
                "venue": q.venue,
                "quantity": fill_qty,
                "gross_price": q.ask if side == "buy" else q.bid,
                "net_price": eff_price,
                "fee": fill_qty * q.fee_per_share,
                "rebate": fill_qty * q.rebate_per_share,
            })
            remaining -= fill_qty

        return routes
流动性感知路由
class LiquidityAwareRouter:
    """流动性感知路由"""

    def __init__(self, fill_probability_model: Optional[dict] = None):
        self.fill_prob = fill_probability_model or {}

    def estimate_fill_probability(self, venue: str, order_type: str,
                                   size_vs_displayed: float) -> float:
        base_prob = self.fill_prob.get(venue, 0.5)
        if order_type == "limit":
            size_adjustment = max(0.1, 1.0 - size_vs_displayed * 0.3)
        else:
            size_adjustment = 0.95
        return base_prob * size_adjustment

    def expected_fill_score(self, quote: Quote, side: str, quantity: int) -> float:
        displayed_size = quote.ask_size if side == "buy" else quote.bid_size
        fill_prob = self.estimate_fill_probability(
            quote.venue, "limit", quantity / max(displayed_size, 1)
        )
        price = quote.ask if side == "buy" else quote.bid
        net_price = price + quote.fee_per_share - quote.rebate_per_share
        latency_penalty = quote.latency_us / 1_000_000 * 0.01

        if side == "buy":
            score = -(net_price + latency_penalty) * fill_prob
        else:
            score = (net_price - latency_penalty) * fill_prob

        return score

    def route_order(self, side: str, quantity: int, quotes: list[Quote]) -> list[dict]:
        scored = [(q, self.expected_fill_score(q, side, quantity)) for q in quotes]
        scored.sort(key=lambda x: -x[1])

        routes = []
        remaining = quantity

        for q, score in scored:
            if remaining <= 0:
                break
            size = q.ask_size if side == "buy" else q.bid_size
            fill_qty = min(remaining, size)
            routes.append({
                "venue": q.venue,
                "quantity": fill_qty,
                "score": score,
                "estimated_fill_prob": self.estimate_fill_probability(
                    q.venue, "limit", fill_qty / max(size, 1)
                ),
            })
            remaining -= fill_qty

        return routes
暗池路由
暗池（Dark Pool）是不公开显示报价的交易场所，适合大单执行以减少信息泄露：


class DarkPoolRouter:
    """暗池路由策略"""

    def __init__(self, dark_pools: list[str], probe_size: int = 100,
                 min_dark_saving: float = 0.0001):
        self.dark_pools = dark_pools
        self.probe_size = probe_size
        self.min_dark_saving = min_dark_saving

    def should_try_dark(self, order_quantity: int, adv: float, nbbo_spread: float) -> bool:
        participation = order_quantity / adv
        if participation > 0.01 and nbbo_spread > 0.01:
            return True
        return False

    def dark_pool_ranking(self, symbol: str, historical_fills: dict) -> list[str]:
        fill_rates = {}
        for pool in self.dark_pools:
            history = historical_fills.get(pool, {}).get(symbol, [])
            if history:
                fill_rates[pool] = np.mean(history)
            else:
                fill_rates[pool] = 0.1

        ranked = sorted(fill_rates.items(), key=lambda x: -x[1])
        return [pool for pool, _ in ranked]

    def create_probe_orders(self, side: str, quantity: int, midpoint: float) -> list[dict]:
        probes = []
        for pool in self.dark_pools:
            probes.append({
                "venue": pool,
                "side": side,
                "quantity": min(self.probe_size, quantity),
                "price": midpoint,
                "order_type": "IOC",
                "time_in_force": "immediate",
            })
        return probes

延迟敏感路由
在高频交易环境下，路由延迟直接影响成交质量：


class LatencyAwareRouter:
    """延迟感知路由"""

    def __init__(self, venue_latencies: dict[str, int]):
        self.latencies = venue_latencies

    def stale_quote_probability(self, venue: str, quote_age_us: int) -> float:
        typical_update_interval = 100
        if quote_age_us > typical_update_interval * 3:
            return 0.8
        elif quote_age_us > typical_update_interval:
            return 0.3
        return 0.05

    def latency_adjusted_route(self, side: str, quantity: int,
                                quotes: list[Quote]) -> list[dict]:
        adjusted_quotes = []
        for q in quotes:
            stale_prob = self.stale_quote_probability(q.venue, q.latency_us)
            if side == "buy":
                adjusted_price = q.ask * (1 + stale_prob * 0.001)
            else:
                adjusted_price = q.bid * (1 - stale_prob * 0.001)
            adjusted_quotes.append((q, adjusted_price))

        if side == "buy":
            adjusted_quotes.sort(key=lambda x: x[1])
        else:
            adjusted_quotes.sort(key=lambda x: -x[1])

        routes = []
        remaining = quantity
        for q, adj_price in adjusted_quotes:
            if remaining <= 0:
                break
            size = q.ask_size if side == "buy" else q.bid_size
            fill_qty = min(remaining, size)
            routes.append({
                "venue": q.venue,
                "quantity": fill_qty,
                "quoted_price": q.ask if side == "buy" else q.bid,
                "adjusted_price": adj_price,
                "latency_us": q.latency_us,
            })
            remaining -= fill_qty

        return routes

高级订单类型
class OrderTypeStrategy:
    """订单类型选择策略"""

    def select_order_type(self, urgency: float, spread: float, volatility: float,
                          side: str, nbbo: dict) -> dict:
        if urgency > 0.8:
            return {
                "type": "market",
                "rationale": "高紧迫度，接受市场价格",
            }

        if spread < 0.01:
            if side == "buy":
                price = nbbo["nbbo_bid"] + 0.001
            else:
                price = nbbo["nbbo_ask"] - 0.001
            return {
                "type": "limit",
                "price": price,
                "rationale": "窄价差，在价差内挂单",
            }

        mid = (nbbo["nbbo_bid"] + nbbo["nbbo_ask"]) / 2
        return {
            "type": "midpoint_peg",
            "price": mid,
            "rationale": "宽价差，锚定中间价",
        }

    def iceberg_parameters(self, total_qty: int, adv: float) -> dict:
        display_ratio = min(0.05, 100 / total_qty)
        display_qty = max(100, int(total_qty * display_ratio))
        return {
            "display_quantity": display_qty,
            "total_quantity": total_qty,
            "refresh_quantity": display_qty,
        }
SOR 性能评估
class SORAnalytics:
    """SOR 性能分析"""

    def venue_fill_analysis(self, executions: pd.DataFrame) -> pd.DataFrame:
        venue_stats = executions.groupby("venue").agg(
            total_quantity=("quantity", "sum"),
            avg_price=("price", "mean"),
            n_fills=("quantity", "count"),
            total_fees=("fee", "sum"),
        )
        venue_stats["pct_of_total"] = venue_stats["total_quantity"] / venue_stats["total_quantity"].sum()
        return venue_stats

    def price_improvement_analysis(self, executions: pd.DataFrame, nbbo_at_time: pd.DataFrame) -> dict:
        merged = executions.merge(nbbo_at_time, on="timestamp", how="left")
        if executions["side"].iloc[0] == "buy":
            improvement = merged["nbbo_ask"] - merged["price"]
        else:
            improvement = merged["price"] - merged["nbbo_bid"]

        return {
            "avg_improvement_bps": (improvement / merged["price"]).mean() * 10000,
            "pct_improved": (improvement > 0).mean(),
            "pct_at_nbbo": (improvement == 0).mean(),
            "pct_worse": (improvement < 0).mean(),
        }

    def routing_efficiency(self, actual_cost_bps: float, best_possible_bps: float) -> float:
        if best_possible_bps == 0:
            return 1.0
        return 1.0 - (actual_cost_bps - best_possible_bps) / abs(best_possible_bps)

class ComplianceMonitor:
    """SOR 合规监控"""

    def check_trade_through(self, execution_price: float, side: str,
                            protected_quotes: list[Quote]) -> dict:
        if side == "buy":
            best_protected = min(q.ask for q in protected_quotes if q.ask_size > 0)
            is_violation = execution_price > best_protected
        else:
            best_protected = max(q.bid for q in protected_quotes if q.bid_size > 0)
            is_violation = execution_price < best_protected

        return {
            "is_trade_through": is_violation,
            "execution_price": execution_price,
            "best_protected_price": best_protected,
            "difference_bps": abs(execution_price - best_protected) / best_protected * 10000,
        }
总结
智能订单路由是碎片化市场环境下的核心基础设施。从最基本的价格优先路由，到考虑净费用的净价路由，再到结合成交概率的流动性感知路由和暗池策略，SOR 的复杂度与市场结构的碎片化程度正相关。在实践中，一个生产级 SOR 系统需要综合考虑价格、费用、延迟、成交概率、信息泄露风险和监管合规等多个维度。随着市场微观结构的持续演变——如 SEC 提议的 Reg NMS 改革和暗池规则收紧——SOR 技术也在不断进化，成为连接交易策略和市场执行的关键桥梁。
