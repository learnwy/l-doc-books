实施缺口与最优执行

概述

实施缺口（Implementation Shortfall，IS）是衡量交易执行成本最全面的指标，由 André Perold 于 1988 年提出。它度量的是从投资决策到交易完全执行之间的总成本——包括显性成本（佣金、税费）和隐性成本（市场冲击、时机成本、机会成本）。基于实施缺口最小化的最优执行（Optimal Execution）理论，则由 Almgren 和 Chriss（2000）系统化，成为现代交易成本分析（TCA）的理论基石。

实施缺口的定义与分解
数学定义
假设在时刻 t=0t=0t=0 做出投资决策，决策时的价格为 P0P_0P0​（Arrival Price），订单总量为 XXX 股，最终在 [0,T][0, T][0,T] 时间内执行完毕：


IS=1X∑i=1Nxi(Si−P0)IS = \frac{1}{X} \sum_{i=1}^{N} x_i (S_i - P_0)IS=X1​i=1∑N​xi​(Si​−P0​)
其中 xix_ixi​ 是第 iii 笔成交的股数，SiS_iSi​ 是对应的成交价格。对于买入订单，IS>0IS > 0IS>0 意味着执行成本为正（不利）。

import numpy as np
import pandas as pd
from dataclasses import dataclass

@dataclass
class ExecutionRecord:
    decision_price: float
    execution_prices: np.ndarray
    execution_quantities: np.ndarray
    execution_times: np.ndarray
    total_quantity: int
    side: str
    close_price: float
    commission_rate: float = 0.0003

class ImplementationShortfallAnalyzer:
    """实施缺口分析器"""

    def decompose(self, record: ExecutionRecord) -> dict:
        direction = 1.0 if record.side == "buy" else -1.0
        exec_qty = record.execution_quantities
        exec_prices = record.execution_prices
        total_exec = exec_qty.sum()
        unexecuted = record.total_quantity - total_exec

        avg_exec_price = np.average(exec_prices, weights=exec_qty)

        market_impact = direction * (avg_exec_price - record.decision_price) / record.decision_price
        timing_cost = 0.0
        if len(exec_prices) > 1:
            mid_time_price = exec_prices[len(exec_prices) // 2]
            timing_cost = direction * (mid_time_price - record.decision_price) / record.decision_price

        opportunity_cost = 0.0
        if unexecuted > 0:
            opportunity_cost = (direction * (record.close_price - record.decision_price) /
                                record.decision_price * unexecuted / record.total_quantity)

        commission = record.commission_rate
        total_is = market_impact + opportunity_cost + commission

        return {
            "total_is_bps": total_is * 10000,
            "market_impact_bps": market_impact * 10000,
            "timing_cost_bps": timing_cost * 10000,
            "opportunity_cost_bps": opportunity_cost * 10000,
            "commission_bps": commission * 10000,
            "fill_rate": total_exec / record.total_quantity,
            "avg_exec_price": avg_exec_price,
        }

class AlmgrenChrissModel:
    """Almgren-Chriss 最优执行模型"""

    def __init__(self, total_shares: int, total_time: float, n_periods: int,
                 volatility: float, eta: float, gamma: float, risk_aversion: float):
        self.X = total_shares
        self.T = total_time
        self.N = n_periods
        self.sigma = volatility
        self.eta = eta
        self.gamma = gamma
        self.lam = risk_aversion
        self.tau = total_time / n_periods

    def optimal_trajectory(self) -> np.ndarray:
        kappa_sq = (self.lam * self.sigma ** 2) / (self.eta * (1 / self.tau))
        kappa = np.sqrt(kappa_sq) if kappa_sq > 0 else 0.01

        trajectory = np.zeros(self.N + 1)
        for k in range(self.N + 1):
            t_k = k * self.tau
            trajectory[k] = self.X * np.sinh(kappa * (self.T - t_k)) / np.sinh(kappa * self.T)

        return trajectory

    def optimal_trade_list(self) -> np.ndarray:
        traj = self.optimal_trajectory()
        trades = -np.diff(traj)
        return trades

    def expected_cost(self) -> float:
        trades = self.optimal_trade_list()
        permanent_cost = self.gamma * np.sum(trades ** 2) / (2 * self.tau)
        temporary_cost = self.eta * np.sum(trades ** 2) / self.tau
        return permanent_cost + temporary_cost

    def cost_variance(self) -> float:
        traj = self.optimal_trajectory()
        variance = self.sigma ** 2 * self.tau * np.sum(traj[:-1] ** 2)
        return variance

    def efficient_frontier(self, n_points: int = 50) -> pd.DataFrame:
        lambdas = np.logspace(-3, 3, n_points)
        results = []
        for lam in lambdas:
            self.lam = lam
            expected = self.expected_cost()
            variance = self.cost_variance()
            results.append({
                "risk_aversion": lam,
                "expected_cost": expected,
                "cost_std": np.sqrt(variance),
            })
        return pd.DataFrame(results)

实践中常用的市场冲击模型形式为幂律函数：


Impact=α⋅σ⋅(QV)βImpact = \alpha \cdot \sigma \cdot \left(\frac{Q}{V}\right)^{\beta}Impact=α⋅σ⋅(VQ​)β
其中 σ\sigmaσ 是波动率，Q/VQ/VQ/V 是订单量与日均成交量之比，α\alphaα 和 β\betaβ 是经验参数。


class MarketImpactModel:
    """市场冲击模型"""

    def __init__(self, alpha: float = 0.1, beta: float = 0.5, gamma_perm: float = 0.1):
        self.alpha = alpha
        self.beta = beta
        self.gamma_perm = gamma_perm

    def temporary_impact(self, trade_rate: float, volatility: float, adv: float) -> float:
        participation = trade_rate / adv
        return self.alpha * volatility * participation ** self.beta

    def permanent_impact(self, total_quantity: float, adv: float, volatility: float) -> float:
        return self.gamma_perm * volatility * (total_quantity / adv) ** 0.5

    def total_impact(self, trade_schedule: np.ndarray, volatility: float,
                     adv: float, tau: float) -> float:
        temp_costs = sum(
            self.temporary_impact(q / tau, volatility, adv) * q
            for q in trade_schedule
        )
        perm_cost = self.permanent_impact(trade_schedule.sum(), adv, volatility)
        total_qty = trade_schedule.sum()
        return (temp_costs + perm_cost * total_qty) / total_qty

自适应 IS 算法
在实际执行中，标准 Almgren-Chriss 模型假设参数已知且恒定，这与实际情况有差距。自适应 IS 算法在执行过程中实时更新参数：


class AdaptiveISAlgorithm:
    """自适应实施缺口算法"""

    def __init__(self, total_shares: int, total_time: float, urgency: float = 0.5):
        self.total_shares = total_shares
        self.total_time = total_time
        self.urgency = urgency
        self.executed = 0
        self.arrival_price = None

    def update_state(self, current_price: float, current_volume: float,
                     current_volatility: float, elapsed_time: float) -> dict:
        if self.arrival_price is None:
            self.arrival_price = current_price

        remaining = self.total_shares - self.executed
        remaining_time = max(self.total_time - elapsed_time, 0.001)
        time_pct = elapsed_time / self.total_time
        completion_pct = self.executed / self.total_shares

        price_drift = (current_price - self.arrival_price) / self.arrival_price
        is_favorable = price_drift < 0

        if is_favorable:
            aggression = min(1.0, self.urgency * 1.5)
        elif completion_pct < time_pct * 0.7:
            aggression = min(1.0, self.urgency * 1.3)
        else:
            aggression = self.urgency * 0.8

        base_rate = remaining / remaining_time
        adjusted_rate = base_rate * aggression

        return {
            "target_rate": adjusted_rate,
            "aggression": aggression,
            "remaining_shares": remaining,
            "is_favorable": is_favorable,
            "price_drift_bps": price_drift * 10000,
        }

事后分析框架
class TCAFramework:
    """交易成本分析框架"""

    def pre_trade_estimate(self, order_size: int, adv: float, volatility: float,
                           spread: float) -> dict:
        participation = order_size / adv
        estimated_impact = 0.1 * volatility * participation ** 0.5
        estimated_spread_cost = spread / 2
        total_estimate = estimated_impact + estimated_spread_cost

        return {
            "estimated_impact_bps": estimated_impact * 10000,
            "estimated_spread_bps": estimated_spread_cost * 10000,
            "total_estimate_bps": total_estimate * 10000,
            "participation_rate": participation,
            "risk_rating": "high" if participation > 0.2 else "medium" if participation > 0.05 else "low",
        }

    def post_trade_analysis(self, executions: pd.DataFrame, benchmarks: dict) -> dict:
        exec_vwap = np.average(executions["price"], weights=executions["quantity"])
        total_qty = executions["quantity"].sum()

        vs_arrival = (exec_vwap - benchmarks["arrival"]) / benchmarks["arrival"] * 10000
        vs_vwap = (exec_vwap - benchmarks["vwap"]) / benchmarks["vwap"] * 10000
        vs_close = (exec_vwap - benchmarks["close"]) / benchmarks["close"] * 10000

        return {
            "exec_vwap": exec_vwap,
            "vs_arrival_bps": vs_arrival,
            "vs_vwap_bps": vs_vwap,
            "vs_close_bps": vs_close,
            "total_quantity": total_qty,
            "n_fills": len(executions),
            "avg_fill_size": total_qty / len(executions),
        }

    def peer_comparison(self, my_results: dict, peer_results: list) -> dict:
        peer_costs = [p["vs_arrival_bps"] for p in peer_results]
        my_cost = my_results["vs_arrival_bps"]
        percentile = sum(1 for p in peer_costs if p > my_cost) / len(peer_costs) * 100
        return {
            "my_cost_bps": my_cost,
            "peer_median_bps": np.median(peer_costs),
            "percentile": percentile,
            "better_than_median": my_cost < np.median(peer_costs),
        }

多资产联合执行
当需要同时执行多个相关资产的订单时（如组合再平衡），需要考虑资产间的相关性：


class PortfolioExecution:
    """组合级最优执行"""

    def __init__(self, n_assets: int, correlation_matrix: np.ndarray):
        self.n_assets = n_assets
        self.corr = correlation_matrix

    def joint_impact(self, trade_list: np.ndarray, individual_impacts: np.ndarray) -> float:
        cross_impact = trade_list @ self.corr @ trade_list.T
        direct_impact = np.sum(individual_impacts * np.abs(trade_list))
        return direct_impact + 0.1 * cross_impact

    def netting_opportunity(self, buy_orders: dict, sell_orders: dict) -> dict:
        netted = {}
        for symbol in set(buy_orders.keys()) & set(sell_orders.keys()):
            buy_qty = buy_orders[symbol]
            sell_qty = sell_orders[symbol]
            net = buy_qty - sell_qty
            saved = min(buy_qty, sell_qty)
            netted[symbol] = {"net_quantity": net, "saved_quantity": saved}
        return netted
总结
实施缺口是衡量执行质量最全面的指标，它将市场冲击、时机成本、机会成本和显性成本统一纳入一个框架。Almgren-Chriss 模型为最优执行提供了理论基础，通过在风险（价格不确定性）和成本（市场冲击）之间的权衡，给出最优执行轨迹。在实践中，自适应算法根据实时市场状况动态调整执行策略，TCA 框架则提供事前估计和事后分析的闭环反馈。理解这些概念是构建专业级执行系统的基础——执行成本往往决定了一个好策略能否在实盘中真正盈利。
