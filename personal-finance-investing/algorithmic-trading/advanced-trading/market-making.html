做市策略与库存管理

概述

做市（Market Making）是金融市场中最古老也最核心的交易角色之一。做市商同时在买方和卖方报价，为市场提供流动性，并通过买卖价差（Bid-Ask Spread）获取利润。在电子化交易时代，算法做市已经取代了传统的人工做市，高频做市商（如 Citadel Securities、Virtu Financial、Jump Trading）在全球主要市场中扮演着关键角色。

做市策略的核心挑战在于库存管理——做市商必须在提供流动性（赚取价差）和控制库存风险（价格不利变动）之间寻找最优平衡。Avellaneda 和 Stoikov（2008）的经典模型为这一平衡提供了理论框架。

逆向选择
逆向选择（Adverse Selection）是做市面临的最大风险。当知情交易者（Informed Trader）拥有做市商不知道的信息时，做市商的报价会系统性地处于不利地位：


import numpy as np
import pandas as pd
from dataclasses import dataclass

class AdverseSelectionEstimator:
    """逆向选择估计"""

    def kyle_lambda(self, trades: pd.DataFrame) -> float:
        price_changes = trades["mid_price"].diff()
        signed_volume = trades["volume"] * trades["direction"]
        model = np.polyfit(signed_volume.dropna(), price_changes.dropna(), 1)
        return model[0]

    def roll_spread(self, prices: pd.Series) -> float:
        delta_p = prices.diff()
        autocovariance = delta_p.autocorr(lag=1) * delta_p.var()
        if autocovariance < 0:
            return 2 * np.sqrt(-autocovariance)
        return 0.0

    def pin_estimate(self, buy_counts: np.ndarray, sell_counts: np.ndarray) -> dict:
        total_buys = buy_counts.sum()
        total_sells = sell_counts.sum()
        total = total_buys + total_sells
        imbalance = abs(total_buys - total_sells)
        pin = imbalance / total if total > 0 else 0
        return {
            "pin": pin,
            "buy_intensity": total_buys / len(buy_counts),
            "sell_intensity": total_sells / len(sell_counts),
        }
Avellaneda-Stoikov 模型
模型框架
Avellaneda 和 Stoikov（2008）的做市模型是最经典的库存管理框架。做市商的目标是最大化终端时刻的效用函数：


max⁡E[−exp⁡(−γ(XT+qTST))]\max E[-\exp(-\gamma (X_T + q_T S_T))]maxE[−exp(−γ(XT​+qT​ST​))]
其中 γ\gammaγ 是风险规避系数，XTX_TXT​ 是终端现金，qTq_TqT​ 是终端库存，STS_TST​ 是终端价格。


最优报价公式
在一定假设下，最优买卖报价为：

其中 rrr 是保留价格（Reservation Price），sss 是中间价，qqq 是当前库存。


class AvellanedaStoikov:
    """Avellaneda-Stoikov 做市模型"""

    def __init__(self, gamma: float = 0.1, sigma: float = 0.02,
                 kappa: float = 1.5, dt: float = 1 / 252):
        self.gamma = gamma
        self.sigma = sigma
        self.kappa = kappa
        self.dt = dt

    def reservation_price(self, mid_price: float, inventory: float,
                          time_remaining: float) -> float:
        return mid_price - inventory * self.gamma * self.sigma ** 2 * time_remaining

    def optimal_spread(self, time_remaining: float) -> float:
        base_spread = (2 / self.gamma) * np.log(1 + self.gamma / self.kappa)
        inventory_spread = self.gamma * self.sigma ** 2 * time_remaining
        return base_spread + inventory_spread

    def optimal_quotes(self, mid_price: float, inventory: float,
                       time_remaining: float) -> dict:
        r = self.reservation_price(mid_price, inventory, time_remaining)
        half_spread = self.optimal_spread(time_remaining) / 2
        bid = r - half_spread
        ask = r + half_spread
        return {
            "bid": bid,
            "ask": ask,
            "spread": ask - bid,
            "reservation_price": r,
            "mid_price": mid_price,
            "inventory_skew": mid_price - r,
        }

    def simulate(self, n_steps: int = 1000, initial_price: float = 100.0) -> pd.DataFrame:
        price = initial_price
        inventory = 0.0
        cash = 0.0
        records = []

        for step in range(n_steps):
            time_remaining = (n_steps - step) / n_steps
            quotes = self.optimal_quotes(price, inventory, time_remaining)

            buy_prob = self.kappa * np.exp(-self.kappa * (price - quotes["bid"]))
            sell_prob = self.kappa * np.exp(-self.kappa * (quotes["ask"] - price))

            buy_prob = min(buy_prob * self.dt, 0.5)
            sell_prob = min(sell_prob * self.dt, 0.5)

            if np.random.random() < buy_prob:
                inventory += 1
                cash -= quotes["bid"]

            if np.random.random() < sell_prob:
                inventory -= 1
                cash += quotes["ask"]

            price += self.sigma * np.random.normal() * np.sqrt(self.dt)
            pnl = cash + inventory * price

            records.append({
                "step": step,
                "mid_price": price,
                "bid": quotes["bid"],
                "ask": quotes["ask"],
                "inventory": inventory,
                "cash": cash,
                "pnl": pnl,
                "spread": quotes["spread"],
            })

        return pd.DataFrame(records)
库存管理策略
库存偏斜（Inventory Skew）
库存偏斜是做市策略中最基本的库存管理手段——当库存为正（多头）时，降低买入报价、抬高卖出报价以鼓励平仓；反之亦然：


class InventoryManager:
    """库存管理器"""

    def __init__(self, max_inventory: int = 100, skew_factor: float = 0.001,
                 urgency_threshold: float = 0.8):
        self.max_inventory = max_inventory
        self.skew_factor = skew_factor
        self.urgency_threshold = urgency_threshold

    def compute_skew(self, current_inventory: int) -> dict:
        inventory_ratio = current_inventory / self.max_inventory
        bid_skew = -self.skew_factor * current_inventory
        ask_skew = self.skew_factor * current_inventory

        if abs(inventory_ratio) > self.urgency_threshold:
            urgency_multiplier = 2.0
            if current_inventory > 0:
                ask_skew *= urgency_multiplier
            else:
                bid_skew *= urgency_multiplier

        return {
            "bid_adjustment": bid_skew,
            "ask_adjustment": ask_skew,
            "inventory_ratio": inventory_ratio,
            "is_urgent": abs(inventory_ratio) > self.urgency_threshold,
        }

    def position_limit_check(self, current_inventory: int, proposed_trade: int) -> dict:
        new_inventory = current_inventory + proposed_trade
        if abs(new_inventory) > self.max_inventory:
            allowed_trade = np.sign(proposed_trade) * (self.max_inventory - abs(current_inventory))
            return {
                "allowed": int(allowed_trade),
                "is_limited": True,
            }
        return {
            "allowed": proposed_trade,
            "is_limited": False,
        }
多级报价
class MultiLevelQuoter:
    """多级报价策略"""

    def __init__(self, n_levels: int = 5, level_spacing: float = 0.0001,
                 size_decay: float = 0.7):
        self.n_levels = n_levels
        self.level_spacing = level_spacing
        self.size_decay = size_decay

    def generate_quotes(self, mid_price: float, base_size: int,
                        inventory_skew: dict) -> list[dict]:
        quotes = []
        bid_adj = inventory_skew["bid_adjustment"]
        ask_adj = inventory_skew["ask_adjustment"]

        for level in range(self.n_levels):
            size = int(base_size * self.size_decay ** level)
            if size < 1:
                break

            offset = (level + 1) * self.level_spacing * mid_price
            bid_price = mid_price - offset + bid_adj
            ask_price = mid_price + offset + ask_adj

            quotes.append({
                "level": level + 1,
                "bid_price": round(bid_price, 4),
                "ask_price": round(ask_price, 4),
                "bid_size": size,
                "ask_size": size,
                "spread": round(ask_price - bid_price, 4),
            })

        return quotes

class MarketMakingRiskMonitor:
    """做市风控监控"""

    def __init__(self, max_loss: float = 50000, max_inventory_duration_s: float = 600):
        self.max_loss = max_loss
        self.max_duration = max_inventory_duration_s
        self._inventory_start_time = None
        self._realized_pnl = 0.0

    def realized_spread(self, trades: list[dict]) -> float:
        if not trades:
            return 0.0
        spreads = []
        for i in range(0, len(trades) - 1, 2):
            if i + 1 < len(trades):
                buy = trades[i] if trades[i]["side"] == "buy" else trades[i + 1]
                sell = trades[i + 1] if trades[i + 1]["side"] == "sell" else trades[i]
                spreads.append(sell["price"] - buy["price"])
        return np.mean(spreads) if spreads else 0.0

    def adverse_selection_metric(self, fills: list[dict], mid_prices_after: list[float],
                                  horizon_s: float = 5.0) -> float:
        adverse_count = 0
        for fill, future_mid in zip(fills, mid_prices_after):
            if fill["side"] == "buy" and future_mid < fill["price"]:
                adverse_count += 1
            elif fill["side"] == "sell" and future_mid > fill["price"]:
                adverse_count += 1
        return adverse_count / max(len(fills), 1)

总结
做市策略是算法交易中最精密的策略类型之一，需要在流动性提供（赚取价差）和库存风险管理之间寻找最优平衡。Avellaneda-Stoikov 模型提供了理论基础——通过库存偏斜调整报价，让做市商在持有大量库存时自然倾向于减仓。实践中的关键包括：准确估计逆向选择风险以合理定价价差、多级报价提高成交概率和价差捕获、严格的库存限制和持续时间控制防止尾部风险、以及毫秒甚至微秒级的系统延迟以保持竞争力。做市是一个规模效应显著的业务，少数头部做市商凭借技术优势和资本实力占据了绝大部分市场份额。
