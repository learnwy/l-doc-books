强化学习在交易中的应用

概述

强化学习（Reinforcement Learning, RL）为算法交易提供了一种全新的范式。与传统的监督学习方法不同，RL 直接优化交易者在动态市场环境中的序列决策——智能体（Agent）通过与市场环境的反复交互，学习在不同市场状态下应采取的最优行动（买入、卖出、持有），以最大化长期累计收益。这种"试错学习"的范式天然契合交易决策的序列性和不确定性。

近年来（2020-2026），随着深度强化学习技术的成熟和算力成本的下降，RL 在交易中的应用从学术实验走向了实际部署。主要应用场景包括执行优化、投资组合管理和做市策略。

交易问题的 MDP 建模
状态、动作、奖励定义
将交易决策建模为马尔可夫决策过程（MDP）是应用 RL 的第一步：


| MDP 要素          | 交易场景定义                 | 设计考量                   |
| ----------------- | ---------------------------- | -------------------------- | ------------------ |
| 状态 sts_tst​        | 价格、持仓、技术指标、订单簿 | 维度适中，包含决策所需信息 |
| 动作 ata_tat​        | 交易方向和数量               | 离散或连续动作空间         |
| 奖励 rtr_trt​        | 已实现 PnL、风险调整收益     | 需要平衡短期与长期         |
| 转移 P(s_t+1∣st,at)P(s\_{t+1}  | s_t, a_t)P(s_t+1∣st​,at​)                   | 市场动态 + 交易影响        | 部分可观察、非平稳 |
| 折扣因子 γ\gammaγ | 0.95-0.999                   | 控制远期收益的权重         |


环境实现
import numpy as np
import pandas as pd
from dataclasses import dataclass

@dataclass
class TradingState:
    prices: np.ndarray
    position: float
    cash: float
    unrealized_pnl: float
    step: int
    features: np.ndarray

class TradingEnvironment:
    """交易环境"""

    def __init__(self, price_data: pd.DataFrame, initial_cash: float = 1_000_000,
                 max_position: float = 1.0, transaction_cost: float = 0.001,
                 feature_window: int = 30):
        self.price_data = price_data
        self.initial_cash = initial_cash
        self.max_position = max_position
        self.transaction_cost = transaction_cost
        self.feature_window = feature_window
        self.n_steps = len(price_data)
        self._reset_state()

    def _reset_state(self):
        self.position = 0.0
        self.cash = self.initial_cash
        self.step = self.feature_window
        self.entry_price = 0.0
        self.portfolio_values = [self.initial_cash]

    def reset(self) -> np.ndarray:
        self._reset_state()
        return self._get_observation()

    def _get_observation(self) -> np.ndarray:
        window = self.price_data.iloc[self.step - self.feature_window:self.step]
        prices = window["close"].values
        returns = np.diff(prices) / prices[:-1]
        volatility = np.std(returns)
        ma_fast = np.mean(prices[-5:])
        ma_slow = np.mean(prices[-20:])
        rsi = self._compute_rsi(prices)
        features = np.array([
            returns[-1] if len(returns) > 0 else 0,
            volatility,
            (ma_fast - ma_slow) / ma_slow if ma_slow > 0 else 0,
            rsi / 100 - 0.5,
            self.position,
            (self.cash / self.initial_cash) - 1,
        ])
        return features

    def _compute_rsi(self, prices: np.ndarray, period: int = 14) -> float:
        deltas = np.diff(prices[-period - 1:])
        gains = np.maximum(deltas, 0)
        losses = np.abs(np.minimum(deltas, 0))
        avg_gain = np.mean(gains) if len(gains) > 0 else 0
        avg_loss = np.mean(losses) if len(losses) > 0 else 0
        if avg_loss == 0:
            return 100
        rs = avg_gain / avg_loss
        return 100 - 100 / (1 + rs)

    def step_env(self, action: int) -> tuple:
        current_price = self.price_data.iloc[self.step]["close"]
        action_map = {0: -1.0, 1: -0.5, 2: 0.0, 3: 0.5, 4: 1.0}
        target_position = action_map.get(action, 0.0) * self.max_position
        trade_size = target_position - self.position

        cost = abs(trade_size) * current_price * self.transaction_cost
        self.cash -= trade_size * current_price + cost
        self.position = target_position

        self.step += 1
        done = self.step >= self.n_steps - 1

        next_price = self.price_data.iloc[self.step]["close"]
        portfolio_value = self.cash + self.position * next_price
        self.portfolio_values.append(portfolio_value)

        reward = self._compute_reward(portfolio_value)
        obs = self._get_observation() if not done else np.zeros(6)

        return obs, reward, done, {"portfolio_value": portfolio_value}

    def _compute_reward(self, current_value: float) -> float:
        if len(self.portfolio_values) < 2:
            return 0.0
        ret = (current_value - self.portfolio_values[-2]) / self.portfolio_values[-2]
        risk_penalty = -0.5 * ret ** 2 if ret < 0 else 0
        return ret + risk_penalty
常用 RL 算法
DQN（Deep Q-Network）
DQN 适用于离散动作空间的交易决策：


import torch
import torch.nn as nn
import torch.optim as optim
from collections import deque
import random

class QNetwork(nn.Module):
    def __init__(self, state_dim: int, action_dim: int, hidden_dim: int = 128):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
        )

    def forward(self, x):
        return self.net(x)

class DQNAgent:
    """DQN 交易智能体"""

    def __init__(self, state_dim: int, action_dim: int, lr: float = 1e-3,
                 gamma: float = 0.99, epsilon: float = 1.0, epsilon_decay: float = 0.995,
                 epsilon_min: float = 0.01, buffer_size: int = 100_000, batch_size: int = 64):
        self.action_dim = action_dim
        self.gamma = gamma
        self.epsilon = epsilon
        self.epsilon_decay = epsilon_decay
        self.epsilon_min = epsilon_min
        self.batch_size = batch_size

        self.q_network = QNetwork(state_dim, action_dim)
        self.target_network = QNetwork(state_dim, action_dim)
        self.target_network.load_state_dict(self.q_network.state_dict())
        self.optimizer = optim.Adam(self.q_network.parameters(), lr=lr)
        self.replay_buffer = deque(maxlen=buffer_size)

    def select_action(self, state: np.ndarray) -> int:
        if random.random() < self.epsilon:
            return random.randint(0, self.action_dim - 1)
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        with torch.no_grad():
            q_values = self.q_network(state_tensor)
        return q_values.argmax(dim=1).item()

    def store_transition(self, state, action, reward, next_state, done):
        self.replay_buffer.append((state, action, reward, next_state, done))

    def train_step(self) -> float:
        if len(self.replay_buffer) < self.batch_size:
            return 0.0

        batch = random.sample(self.replay_buffer, self.batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)

        states = torch.FloatTensor(np.array(states))
        actions = torch.LongTensor(actions).unsqueeze(1)
        rewards = torch.FloatTensor(rewards)
        next_states = torch.FloatTensor(np.array(next_states))
        dones = torch.FloatTensor(dones)

        current_q = self.q_network(states).gather(1, actions).squeeze()
        with torch.no_grad():
            next_q = self.target_network(next_states).max(dim=1)[0]
            target_q = rewards + (1 - dones) * self.gamma * next_q

        loss = nn.MSELoss()(current_q, target_q)
        self.optimizer.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(self.q_network.parameters(), 1.0)
        self.optimizer.step()

        self.epsilon = max(self.epsilon_min, self.epsilon * self.epsilon_decay)
        return loss.item()

    def update_target(self):
        self.target_network.load_state_dict(self.q_network.state_dict())
PPO（Proximal Policy Optimization）
PPO 是连续动作空间中最常用的 RL 算法：


class ActorCritic(nn.Module):
    def __init__(self, state_dim: int, action_dim: int, hidden_dim: int = 128):
        super().__init__()
        self.shared = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
        )
        self.actor_mean = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
            nn.Tanh(),
        )
        self.actor_log_std = nn.Parameter(torch.zeros(action_dim))
        self.critic = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1),
        )

    def forward(self, state):
        shared = self.shared(state)
        action_mean = self.actor_mean(shared)
        action_std = self.actor_log_std.exp()
        value = self.critic(shared)
        return action_mean, action_std, value

class PPOAgent:
    """PPO 交易智能体"""

    def __init__(self, state_dim: int, action_dim: int, lr: float = 3e-4,
                 gamma: float = 0.99, clip_ratio: float = 0.2, epochs: int = 10):
        self.gamma = gamma
        self.clip_ratio = clip_ratio
        self.epochs = epochs
        self.model = ActorCritic(state_dim, action_dim)
        self.optimizer = optim.Adam(self.model.parameters(), lr=lr)
        self.trajectory = []

    def select_action(self, state: np.ndarray) -> tuple:
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        with torch.no_grad():
            mean, std, value = self.model(state_tensor)
        dist = torch.distributions.Normal(mean, std)
        action = dist.sample()
        log_prob = dist.log_prob(action).sum(dim=-1)
        return action.squeeze().numpy(), log_prob.item(), value.item()

    def store_transition(self, state, action, reward, log_prob, value, done):
        self.trajectory.append({
            "state": state, "action": action, "reward": reward,
            "log_prob": log_prob, "value": value, "done": done,
        })

    def compute_gae(self, rewards, values, dones, next_value, gae_lambda=0.95):
        advantages = []
        gae = 0
        for t in reversed(range(len(rewards))):
            if t == len(rewards) - 1:
                next_val = next_value
            else:
                next_val = values[t + 1]
            delta = rewards[t] + self.gamma * next_val * (1 - dones[t]) - values[t]
            gae = delta + self.gamma * gae_lambda * (1 - dones[t]) * gae
            advantages.insert(0, gae)
        return advantages
RL 在执行优化中的应用
最优执行 RL
将 Almgren-Chriss 框架推广为 RL 问题，允许智能体学习非线性冲击模型下的最优执行策略：


class ExecutionRLEnvironment:
    """执行优化 RL 环境"""

    def __init__(self, total_shares: int, total_time: int, volatility: float,
                 impact_coefficient: float):
        self.total_shares = total_shares
        self.total_time = total_time
        self.volatility = volatility
        self.impact_coef = impact_coefficient

    def reset(self) -> np.ndarray:
        self.remaining = self.total_shares
        self.time_remaining = self.total_time
        self.avg_exec_price = 0.0
        self.total_executed = 0
        self.price = 100.0
        return self._get_state()

    def _get_state(self) -> np.ndarray:
        return np.array([
            self.remaining / self.total_shares,
            self.time_remaining / self.total_time,
            self.price / 100.0,
        ])

    def step_env(self, action: float) -> tuple:
        trade_qty = int(action * self.remaining)
        trade_qty = max(0, min(trade_qty, self.remaining))

        price_change = np.random.normal(0, self.volatility)
        impact = self.impact_coef * (trade_qty / self.total_shares) ** 0.5
        exec_price = self.price + impact + price_change

        if self.total_executed + trade_qty > 0:
            self.avg_exec_price = (
                (self.avg_exec_price * self.total_executed + exec_price * trade_qty)
                / (self.total_executed + trade_qty)
            )

        self.total_executed += trade_qty
        self.remaining -= trade_qty
        self.time_remaining -= 1
        self.price += price_change

        reward = -impact * trade_qty
        done = self.time_remaining <= 0 or self.remaining <= 0

        if done and self.remaining > 0:
            reward -= self.remaining * self.volatility * 10

        return self._get_state(), reward, done, {}

训练最佳实践
class RLTrainingPipeline:
    """RL 训练流水线"""

    def __init__(self, env_factory, agent_factory, n_envs: int = 8):
        self.envs = [env_factory() for _ in range(n_envs)]
        self.agent = agent_factory()
        self.best_reward = float("-inf")

    def train(self, n_episodes: int = 1000, eval_interval: int = 50):
        for episode in range(n_episodes):
            env = self.envs[episode % len(self.envs)]
            state = env.reset()
            total_reward = 0

            while True:
                action = self.agent.select_action(state)
                if isinstance(action, tuple):
                    action_val, log_prob, value = action
                    next_state, reward, done, info = env.step_env(action_val)
                    self.agent.store_transition(state, action_val, reward, log_prob, value, done)
                else:
                    next_state, reward, done, info = env.step_env(action)
                    self.agent.store_transition(state, action, reward, next_state, done)
                    self.agent.train_step()

                total_reward += reward
                state = next_state

                if done:
                    break

            if episode % eval_interval == 0:
                eval_reward = self._evaluate()
                if eval_reward > self.best_reward:
                    self.best_reward = eval_reward

    def _evaluate(self, n_episodes: int = 10) -> float:
        total_rewards = []
        for _ in range(n_episodes):
            env = self.envs[0]
            state = env.reset()
            ep_reward = 0
            while True:
                action = self.agent.select_action(state)
                if isinstance(action, tuple):
                    action = action[0]
                next_state, reward, done, _ = env.step_env(action)
                ep_reward += reward
                state = next_state
                if done:
                    break
            total_rewards.append(ep_reward)
        return np.mean(total_rewards)
总结
强化学习为交易决策提供了一种端到端的优化框架，将状态感知、决策和学习统一在一个体系中。DQN 适合离散动作空间的交易信号生成，PPO 在连续动作空间的仓位管理中表现突出，而在执行优化场景中 RL 能够学习适应非线性市场冲击的策略。核心挑战在于金融市场的非平稳性、部分可观察性和过拟合风险。成功应用 RL 的关键包括：精心设计状态空间和奖励函数、使用多环境训练增强泛化性、通过约束 RL 或安全层确保风控合规、以及建立严格的样本外评估流程。RL 不是传统方法的替代品，而是强有力的补充——在传统模型难以处理的复杂非线性场景中，RL 展现出独特的优势。
