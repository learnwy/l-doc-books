WebAssembly 代码生成

WebAssembly 概述

WebAssembly（WASM）是一种面向 Web 的低级字节码格式，设计为安全、高效且可移植的编译目标。它提供接近原生的执行速度，同时在浏览器沙箱中安全运行。将高级语言编译到 WASM 是现代编译器工程的重要课题。

interface WasmModule {
  types: FuncType[];
  functions: WasmFunction[];
  memories: MemoryType[];
  exports: WasmExport[];
  globals: WasmGlobal[];
}

interface FuncType {
  params: WasmValueType[];
  results: WasmValueType[];
}

type WasmValueType = 'i32' | 'i64' | 'f32' | 'f64' | 'funcref' | 'externref';

interface WasmFunction {
  name: string;
  typeIndex: number;
  locals: WasmValueType[];
  body: WasmInstruction[];
}

interface MemoryType {
  min: number;
  max?: number;
}

interface WasmExport {
  name: string;
  kind: 'function' | 'memory' | 'table' | 'global';
  index: number;
}

interface WasmGlobal {
  type: WasmValueType;
  mutable: boolean;
  init: WasmInstruction[];
}
WASM 特性 描述
类型安全 模块级验证保证类型正确
沙箱执行 无法直接访问宿主内存
可移植 与平台和浏览器无关
流式编译 可以在下载的同时编译
确定性执行 相同输入产生相同结果
紧凑格式 二进制格式比 JS 小得多 


提示： 虽然 WASM 最初为 Web 设计，但其便携性使其在非 Web 场景也大放异彩——WASI（WebAssembly System Interface）让 WASM 可以在服务器端运行，Cloudflare Workers 和 Fastly Compute 用 WASM 实现边缘计算。






WASM 指令集
WASM 使用栈式虚拟机执行模型。指令从栈上消费操作数、将结果压入栈。WASM 的指令集设计精简但完整，覆盖了算术、内存访问、控制流等核心操作。


type WasmInstruction =
  | { op: 'i32.const'; value: number }
  | { op: 'i64.const'; value: bigint }
  | { op: 'f64.const'; value: number }
  | { op: 'i32.add' }
  | { op: 'i32.sub' }
  | { op: 'i32.mul' }
  | { op: 'i32.div_s' }
  | { op: 'i32.eq' }
  | { op: 'i32.lt_s' }
  | { op: 'i32.gt_s' }
  | { op: 'local.get'; index: number }
  | { op: 'local.set'; index: number }
  | { op: 'local.tee'; index: number }
  | { op: 'global.get'; index: number }
  | { op: 'global.set'; index: number }
  | { op: 'i32.load'; offset: number; align: number }
  | { op: 'i32.store'; offset: number; align: number }
  | { op: 'call'; funcIndex: number }
  | { op: 'block'; blockType: WasmValueType | null; body: WasmInstruction[] }
  | { op: 'loop'; blockType: WasmValueType | null; body: WasmInstruction[] }
  | { op: 'if'; blockType: WasmValueType | null; then: WasmInstruction[]; else?: WasmInstruction[] }
  | { op: 'br'; label: number }
  | { op: 'br_if'; label: number }
  | { op: 'return' }
  | { op: 'drop' }
  | { op: 'memory.size' }
  | { op: 'memory.grow' };

const compileExprToWasm = (expr: Expression): WasmInstruction[] => {
  switch (expr.kind) {
    case 'NumberLiteral':
      return [{ op: 'i32.const', value: Math.trunc((expr as NumberLiteral).value) }];
    case 'BinaryExpression': {
      const bin = expr as BinaryExpression;
      const left = compileExprToWasm(bin.left);
      const right = compileExprToWasm(bin.right);
      const opMap: Record<string, WasmInstruction['op']> = {
        '+': 'i32.add',
        '-': 'i32.sub',
        '*': 'i32.mul',
        '/': 'i32.div_s',
      };
      return [...left, ...right, { op: opMap[bin.operator] } as WasmInstruction];
    }
    case 'Identifier': {
      return [{ op: 'local.get', index: 0 }];
    }
    default:
      return [];
  }
};

类型系统映射
将高级语言的类型系统映射到 WASM 的类型系统是编译到 WASM 的核心挑战之一。WASM 目前只有四种数值类型和两种引用类型。


interface TypeMapping {
  sourceType: string;
  wasmType: WasmValueType;
  strategy: string;
}

const typeMappings: TypeMapping[] = [

  { sourceType: 'string', wasmType: 'i32', strategy: '线性内存指针 + 长度' },
  { sourceType: 'array', wasmType: 'i32', strategy: '线性内存指针' },
  { sourceType: 'struct/object', wasmType: 'i32', strategy: '线性内存指针 + 字段偏移' },

const compileStringLiteral = (
  str: string,
  memoryOffset: number,
): {
  dataSegment: Uint8Array;
  ptrInstructions: WasmInstruction[];
} => {
  const encoder = new TextEncoder();
  const bytes = encoder.encode(str);
  const header = new Uint8Array(4);
  new DataView(header.buffer).setUint32(0, bytes.length, true);

  const dataSegment = new Uint8Array(4 + bytes.length);
  dataSegment.set(header, 0);
  dataSegment.set(bytes, 4);

  return {
    dataSegment,
    ptrInstructions: [{ op: 'i32.const', value: memoryOffset }],
  };
};

interface StructLayout {
  name: string;
  fields: Array<{ name: string; type: WasmValueType; offset: number; size: number }>;
  totalSize: number;
  alignment: number;
}

const computeStructLayout = (fields: Array<{ name: string; type: WasmValueType }>): StructLayout => {
  let offset = 0;
  const layoutFields = fields.map((f) => {
    const size = wasmTypeSize(f.type);
    const align = size;
    offset = Math.ceil(offset / align) * align;
    const result = { name: f.name, type: f.type, offset, size };
    offset += size;
    return result;
  });

  return {
    name: '',
    fields: layoutFields,
    totalSize: offset,
    alignment: Math.max(...layoutFields.map((f) => f.size), 1),
  };
};

const wasmTypeSize = (type: WasmValueType): number => {
  switch (type) {
    case 'i32':
    case 'f32':
      return 4;
    case 'i64':
    case 'f64':
      return 8;
    default:
      return 4;
  }
};

内存管理
WASM 使用线性内存（Linear Memory）作为程序的地址空间。编译器需要实现内存分配器来管理这块连续的内存区域。


class WasmMemoryAllocator {
  private freePtr: number;
  private memorySize: number;
  private freeList: Array<{ start: number; size: number }> = [];

  constructor(initialOffset: number, memoryPages: number) {
    this.freePtr = initialOffset;
    this.memorySize = memoryPages * 65536;
  }

  alloc(size: number, alignment: number = 4): number {
    const alignedPtr = Math.ceil(this.freePtr / alignment) * alignment;

    for (let i = 0; i < this.freeList.length; i++) {
      const block = this.freeList[i];
      if (block.size >= size) {
        this.freeList.splice(i, 1);
        if (block.size > size + 16) {
          this.freeList.push({ start: block.start + size, size: block.size - size });
        }
        return block.start;
      }
    }

    if (alignedPtr + size > this.memorySize) {
      return -1;
    }

    this.freePtr = alignedPtr + size;
    return alignedPtr;
  }

  free(ptr: number, size: number): void {
    this.freeList.push({ start: ptr, size });
    this.coalesceFreeList();
  }

  private coalesceFreeList(): void {
    this.freeList.sort((a, b) => a.start - b.start);
    for (let i = 0; i < this.freeList.length - 1; i++) {
      const current = this.freeList[i];
      const next = this.freeList[i + 1];
      if (current.start + current.size === next.start) {
        current.size += next.size;
        this.freeList.splice(i + 1, 1);
        i--;
      }
    }
  }

  generateAllocFunction(): WasmFunction {
    return {
      name: '__alloc',
      typeIndex: 0,
      locals: ['i32'],
      body: [
        { op: 'global.get', index: 0 },
        { op: 'local.set', index: 1 },
        { op: 'global.get', index: 0 },
        { op: 'local.get', index: 0 },
        { op: 'i32.add' },
        { op: 'global.set', index: 0 },
        { op: 'local.get', index: 1 },
      ],
    };
  }
}

小结
    

  • WebAssembly 是安全、高效、可移植的编译目标，使用栈式虚拟机执行模型。
    • 

  • WASM 指令集使用结构化控制流（block/loop/if），而非传统的任意跳转。
    • 

  • 高级类型到 WASM 的映射是编译的核心挑战——字符串、数组、闭包都需要通过线性内存间接表示。
    • 

  • WASM 的内存管理需要编译器自行实现分配器，或依赖 WASM GC 提案的原生支持。
    • 

  • WASI 将 WASM 从浏览器带到服务器端，开启了"编译一次、到处运行"的新范式。
