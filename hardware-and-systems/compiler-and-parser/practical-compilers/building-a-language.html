构建一门语言

语言设计

从零构建一门编程语言是理解编译原理的最佳实践。本节将端到端实现一门名为 Tiny 的小型语言——它支持变量声明、函数定义、条件分支和基本表达式，足以展示编译器的完整管线。

// Tiny 语言示例：
// let max = fn(a, b) {
//   if (a > b) { return a; }
//   return b;
// };
// let result = max(10, 20);

interface TinyToken {
  kind: TinyTokenKind;
  lexeme: string;
  span: { start: number; end: number };
}

enum TinyTokenKind {
  Let,
  Fn,
  If,
  Else,
  Return,
  While,
  Identifier,
  Number,
  String,
  Plus,
  Minus,
  Star,
  Slash,
  Eq,
  EqEq,
  Bang,
  BangEq,
  Lt,
  Gt,
  LtEq,
  GtEq,
  LParen,
  RParen,
  LBrace,
  RBrace,
  Comma,
  Semi,
  EOF,
  Error,
}

const TINY_KEYWORDS: Record<string, TinyTokenKind> = {
  let: TinyTokenKind.Let,
  fn: TinyTokenKind.Fn,
  if: TinyTokenKind.If,
  else: TinyTokenKind.Else,
  return: TinyTokenKind.Return,
  while: TinyTokenKind.While,
};
语言特性 Tiny 支持 示例
变量声明 ✓ let x = 42;
函数定义 ✓ let f = fn(x) { ... };
条件分支 ✓ if (x > 0) { ... }
循环 ✓ while (x > 0) { ... }
算术运算 ✓ a + b * c
比较运算 ✓ a > b, a == b
函数调用 ✓ f(1, 2)
类型系统 ✗ 动态类型
闭包 ✗ 简化处理 


提示： 设计语言时，先确定语法和语义，再实现编译器。一个好的做法是先写出若干个你想让语言支持的示例程序，用这些程序驱动语言设计和编译器实现。






词法分析器与解析器
class TinyLexer {
  private src: string;
  private pos: number = 0;

  constructor(src: string) {
    this.src = src;
  }

  private peek(): string {
    return this.pos < this.src.length ? this.src[this.pos] : '\0';
  }
  private advance(): string {
    return this.src[this.pos++];
  }

  private skipWhitespace(): void {
    while (this.pos < this.src.length && ' \t\r\n'.includes(this.peek())) this.advance();
  }

  scan(): TinyToken[] {
    const tokens: TinyToken[] = [];
    while (this.pos < this.src.length) {
      this.skipWhitespace();
      if (this.pos >= this.src.length) break;
      const start = this.pos;
      const ch = this.advance();

      if (ch >= '0' && ch <= '9') {
        while (this.peek() >= '0' && this.peek() <= '9') this.advance();
        tokens.push(this.tok(TinyTokenKind.Number, start));
      } else if ((ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') || ch === '_') {
        while ('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_'.includes(this.peek())) this.advance();
        const lexeme = this.src.slice(start, this.pos);
        const kind = TINY_KEYWORDS[lexeme] ?? TinyTokenKind.Identifier;
        tokens.push({ kind, lexeme, span: { start, end: this.pos } });
      } else {
        tokens.push(this.scanPunctuation(ch, start));
      }
    }
    tokens.push({ kind: TinyTokenKind.EOF, lexeme: '', span: { start: this.pos, end: this.pos } });
    return tokens;
  }

  private scanPunctuation(ch: string, start: number): TinyToken {
    const next = this.peek();
    if (ch === '=' && next === '=') {
      this.advance();
      return this.tok(TinyTokenKind.EqEq, start);
    }
    if (ch === '!' && next === '=') {
      this.advance();
      return this.tok(TinyTokenKind.BangEq, start);
    }
    if (ch === '<' && next === '=') {
      this.advance();
      return this.tok(TinyTokenKind.LtEq, start);
    }
    if (ch === '>' && next === '=') {
      this.advance();
      return this.tok(TinyTokenKind.GtEq, start);
    }

    const map: Record<string, TinyTokenKind> = {
      '+': TinyTokenKind.Plus,
      '-': TinyTokenKind.Minus,
      '*': TinyTokenKind.Star,
      '/': TinyTokenKind.Slash,
      '(': TinyTokenKind.LParen,
      ')': TinyTokenKind.RParen,
      '{': TinyTokenKind.LBrace,
      '}': TinyTokenKind.RBrace,
      ',': TinyTokenKind.Comma,
      ';': TinyTokenKind.Semi,
      '=': TinyTokenKind.Eq,
      '!': TinyTokenKind.Bang,
      '<': TinyTokenKind.Lt,
      '>': TinyTokenKind.Gt,
    };
    return this.tok(map[ch] ?? TinyTokenKind.Error, start);
  }

  private tok(kind: TinyTokenKind, start: number): TinyToken {
    return { kind, lexeme: this.src.slice(start, this.pos), span: { start, end: this.pos } };
  }
}
type TinyExpr =
  | { kind: 'number'; value: number }
  | { kind: 'ident'; name: string }
  | { kind: 'binary'; op: string; left: TinyExpr; right: TinyExpr }
  | { kind: 'call'; callee: TinyExpr; args: TinyExpr[] }
  | { kind: 'fn'; params: string[]; body: TinyStmt[] };

type TinyStmt =
  | { kind: 'let'; name: string; init: TinyExpr }
  | { kind: 'return'; value: TinyExpr | null }
  | { kind: 'if'; cond: TinyExpr; then: TinyStmt[]; else?: TinyStmt[] }
  | { kind: 'while'; cond: TinyExpr; body: TinyStmt[] }
  | { kind: 'expr'; expr: TinyExpr };

class TinyParser {
  private tokens: TinyToken[];
  private pos: number = 0;

  constructor(tokens: TinyToken[]) {
    this.tokens = tokens;
  }
  private peek(): TinyToken {
    return this.tokens[this.pos];
  }
  private advance(): TinyToken {
    return this.tokens[this.pos++];
  }
  private expect(kind: TinyTokenKind): TinyToken {
    const t = this.peek();
    if (t.kind !== kind) throw new Error(`期望 ${TinyTokenKind[kind]}，得到 ${t.lexeme}`);
    return this.advance();
  }

  parseProgram(): TinyStmt[] {
    const stmts: TinyStmt[] = [];
    while (this.peek().kind !== TinyTokenKind.EOF) stmts.push(this.parseStmt());
    return stmts;
  }

  private parseStmt(): TinyStmt {
    if (this.peek().kind === TinyTokenKind.Let) return this.parseLet();
    if (this.peek().kind === TinyTokenKind.Return) return this.parseReturn();
    if (this.peek().kind === TinyTokenKind.If) return this.parseIf();
    if (this.peek().kind === TinyTokenKind.While) return this.parseWhile();
    const expr = this.parseExpr(0);
    this.expect(TinyTokenKind.Semi);
    return { kind: 'expr', expr };
  }

  private parseLet(): TinyStmt {
    this.advance();
    const name = this.expect(TinyTokenKind.Identifier).lexeme;
    this.expect(TinyTokenKind.Eq);
    const init = this.parseExpr(0);
    this.expect(TinyTokenKind.Semi);
    return { kind: 'let', name, init };
  }

  private parseReturn(): TinyStmt {
    this.advance();
    const value = this.peek().kind === TinyTokenKind.Semi ? null : this.parseExpr(0);
    this.expect(TinyTokenKind.Semi);
    return { kind: 'return', value };
  }

  private parseIf(): TinyStmt {
    this.advance();
    this.expect(TinyTokenKind.LParen);
    const cond = this.parseExpr(0);
    this.expect(TinyTokenKind.RParen);
    const then = this.parseBlock();
    let elseBlock: TinyStmt[] | undefined;
    if (this.peek().kind === TinyTokenKind.Else) {
      this.advance();
      elseBlock = this.parseBlock();
    }
    return { kind: 'if', cond, then, else: elseBlock };
  }

  private parseWhile(): TinyStmt {
    this.advance();
    this.expect(TinyTokenKind.LParen);
    const cond = this.parseExpr(0);
    this.expect(TinyTokenKind.RParen);
    return { kind: 'while', cond, body: this.parseBlock() };
  }

  private parseBlock(): TinyStmt[] {
    this.expect(TinyTokenKind.LBrace);
    const stmts: TinyStmt[] = [];
    while (this.peek().kind !== TinyTokenKind.RBrace) stmts.push(this.parseStmt());
    this.expect(TinyTokenKind.RBrace);
    return stmts;
  }

  private parseExpr(minPrec: number): TinyExpr {
    let left = this.parsePrimary();
    while (true) {
      const prec = this.getPrec(this.peek());
      if (prec <= minPrec) break;
      const op = this.advance().lexeme;
      const right = this.parseExpr(prec);
      left = { kind: 'binary', op, left, right };
    }
    return left;
  }

  private parsePrimary(): TinyExpr {
    const t = this.peek();
    if (t.kind === TinyTokenKind.Number) {
      this.advance();
      return { kind: 'number', value: Number(t.lexeme) };
    }
    if (t.kind === TinyTokenKind.Fn) return this.parseFn();
    if (t.kind === TinyTokenKind.Identifier) {
      this.advance();
      let expr: TinyExpr = { kind: 'ident', name: t.lexeme };
      if (this.peek().kind === TinyTokenKind.LParen) {
        this.advance();
        const args: TinyExpr[] = [];
        while (this.peek().kind !== TinyTokenKind.RParen) {
          args.push(this.parseExpr(0));
          if (this.peek().kind === TinyTokenKind.Comma) this.advance();
        }
        this.expect(TinyTokenKind.RParen);
        expr = { kind: 'call', callee: expr, args };
      }
      return expr;
    }
    if (t.kind === TinyTokenKind.LParen) {
      this.advance();
      const e = this.parseExpr(0);
      this.expect(TinyTokenKind.RParen);
      return e;
    }
    throw new Error(`意外的 Token: ${t.lexeme}`);
  }

  private parseFn(): TinyExpr {
    this.advance();
    this.expect(TinyTokenKind.LParen);
    const params: string[] = [];
    while (this.peek().kind !== TinyTokenKind.RParen) {
      params.push(this.expect(TinyTokenKind.Identifier).lexeme);
      if (this.peek().kind === TinyTokenKind.Comma) this.advance();
    }
    this.expect(TinyTokenKind.RParen);
    return { kind: 'fn', params, body: this.parseBlock() };
  }

  private getPrec(t: TinyToken): number {
    const map: Record<string, number> = {
      '==': 1,
      '!=': 1,
      '<': 2,
      '>': 2,
      '<=': 2,
      '>=': 2,
      '+': 3,
      '-': 3,
      '*': 4,
      '/': 4,
    };
    return map[t.lexeme] ?? 0;
  }
}


树遍历解释器
type TinyValue = number | boolean | TinyFunction | null;
interface TinyFunction {
  params: string[];
  body: TinyStmt[];
  closure: TinyEnv;
}
type TinyEnv = Map<string, TinyValue>;

class ReturnSignal {
  constructor(public value: TinyValue) {}
}

class TinyInterpreter {
  eval(stmts: TinyStmt[], env: TinyEnv = new Map()): TinyValue {
    let result: TinyValue = null;
    for (const stmt of stmts) result = this.execStmt(stmt, env);
    return result;
  }

  private execStmt(stmt: TinyStmt, env: TinyEnv): TinyValue {
    switch (stmt.kind) {
      case 'let':
        env.set(stmt.name, this.evalExpr(stmt.init, env));
        return null;
      case 'return':
        throw new ReturnSignal(stmt.value ? this.evalExpr(stmt.value, env) : null);
      case 'if': {
        const cond = this.evalExpr(stmt.cond, env);
        if (cond) return this.eval(stmt.then, new Map([...env]));
        if (stmt.else) return this.eval(stmt.else, new Map([...env]));
        return null;
      }
      case 'while': {
        while (this.evalExpr(stmt.cond, env)) this.eval(stmt.body, new Map([...env]));
        return null;
      }
      case 'expr':
        return this.evalExpr(stmt.expr, env);
    }
  }

  private evalExpr(expr: TinyExpr, env: TinyEnv): TinyValue {
    switch (expr.kind) {
      case 'number':
        return expr.value;
      case 'ident':
        return env.get(expr.name) ?? null;
      case 'binary':
        return this.evalBinary(expr.op, this.evalExpr(expr.left, env)!, this.evalExpr(expr.right, env)!);
      case 'fn':
        return { params: expr.params, body: expr.body, closure: new Map([...env]) };
      case 'call': {
        const callee = this.evalExpr(expr.callee, env) as TinyFunction;
        const args = expr.args.map((a) => this.evalExpr(a, env));
        const callEnv = new Map([...callee.closure]);
        callee.params.forEach((p, i) => callEnv.set(p, args[i]));
        try {
          this.eval(callee.body, callEnv);
          return null;
        } catch (e) {
          if (e instanceof ReturnSignal) return e.value;
          throw e;
        }
      }
    }
  }

  private evalBinary(op: string, a: TinyValue, b: TinyValue): TinyValue {
    const ops: Record<string, (x: number, y: number) => TinyValue> = {
      '+': (x, y) => x + y,
      '-': (x, y) => x - y,
      '*': (x, y) => x * y,
      '/': (x, y) => x / y,
      '==': (x, y) => x === y,
      '!=': (x, y) => x !== y,
      '<': (x, y) => x < y,
      '>': (x, y) => x > y,
      '<=': (x, y) => x <= y,
      '>=': (x, y) => x >= y,
    };
    return ops[op]?.(a as number, b as number) ?? null;
  }
}


提示： 树遍历解释器是实现语言最快的路径——无需生成中间代码或字节码，直接在 AST 上递归求值即可。Ruby MRI 直到 1.8 版本都使用树遍历解释器。性能要求高时，可以后续添加字节码编译阶段。






小结
    

  • 从零构建语言应先确定示例程序，用目标驱动语言设计和编译器实现。
    • 

  • 词法分析器将源码字符流转换为 Token 序列，关键字通过查表区分。
    • 

  • 递归下降 + Pratt 解析的组合可以简洁地处理语句和表达式。
    • 

  • 树遍历解释器是最快的实现路径，无需字节码即可执行程序。
    • 

  • 用 throw + catch 模式实现 return 语句的跨函数体返回是一个实用技巧。
