LR 与 GLR 解析

LR 解析概述

LR 解析是一类自底向上的解析技术，通过移进（Shift）和归约（Reduce）操作将 Token 序列归约为语法树。LR 解析器由一个有限状态自动机和一张解析表驱动，能处理比 LL 更广泛的文法类别。

type Action =
  | { type: 'shift'; state: number }
  | { type: 'reduce'; rule: number; lhs: string; rhsLength: number }
  | { type: 'accept' }
  | { type: 'error' };

interface ParseTable {
  action: Map<string, Action>[];
  goto: Map<string, number>[];
}

interface GrammarRule {
  lhs: string;
  rhs: string[];
  index: number;
}

class LRParser {
  private table: ParseTable;
  private rules: GrammarRule[];
  private stateStack: number[] = [0];
  private nodeStack: ASTNode[] = [];

  constructor(table: ParseTable, rules: GrammarRule[]) {
    this.table = table;
    this.rules = rules;
  }

  parse(tokens: Token[]): ASTNode {
    let pos = 0;

    while (true) {
      const state = this.stateStack[this.stateStack.length - 1];
      const token = tokens[pos];
      const action =
        this.table.action[state].get(token.lexeme) ?? this.table.action[state].get(tokenKindToString(token.kind));

      if (!action || action.type === 'error') {
        throw new Error(`语法错误: 状态 ${state}，Token '${token.lexeme}' at ${token.line}:${token.column}`);
      }

      if (action.type === 'shift') {
        this.stateStack.push(action.state);
        this.nodeStack.push({ kind: 'Terminal', token });
        pos++;
      } else if (action.type === 'reduce') {
        const rule = this.rules[action.rule];
        const children = this.nodeStack.splice(-rule.rhs.length);
        this.stateStack.splice(-rule.rhs.length);
        const newNode: ASTNode = { kind: rule.lhs, children };
        this.nodeStack.push(newNode);
        const topState = this.stateStack[this.stateStack.length - 1];
        const gotoState = this.table.goto[topState].get(rule.lhs);
        if (gotoState === undefined) throw new Error(`GOTO 表缺失: 状态 ${topState}，非终结符 ${rule.lhs}`);
        this.stateStack.push(gotoState);
      } else {
        return this.nodeStack[0];
      }
    }
  }
}

const tokenKindToString = (kind: TokenKind): string => TokenKind[kind];

移进-归约机制
移进-归约解析维护两个核心数据结构：状态栈和符号栈。移进操作将 Token 压栈，归约操作将栈顶若干符号替换为一个非终结符。


interface ShiftReduceTrace {
  step: number;
  stack: string[];
  input: string[];
  action: string;
}

const traceShiftReduce = (tokens: Token[], table: ParseTable, rules: GrammarRule[]): ShiftReduceTrace[] => {
  const traces: ShiftReduceTrace[] = [];
  const stateStack: number[] = [0];
  const symbolStack: string[] = [];
  let pos = 0;
  let step = 0;

  while (true) {
    const state = stateStack[stateStack.length - 1];
    const token = tokens[pos];
    const remaining = tokens.slice(pos).map((t) => t.lexeme);
    const action = table.action[state].get(token.lexeme);

    if (!action) break;

    if (action.type === 'shift') {
      traces.push({
        step: step++,
        stack: [...symbolStack],
        input: [...remaining],
        action: `移进 '${token.lexeme}'，转到状态 ${action.state}`,
      });
      stateStack.push(action.state);
      symbolStack.push(token.lexeme);
      pos++;
    } else if (action.type === 'reduce') {
      const rule = rules[action.rule];
      traces.push({
        step: step++,
        stack: [...symbolStack],
        input: [...remaining],
        action: `归约 ${rule.lhs} → ${rule.rhs.join(' ')}`,
      });
      symbolStack.splice(-rule.rhs.length);
      stateStack.splice(-rule.rhs.length);
      symbolStack.push(rule.lhs);
      const topState = stateStack[stateStack.length - 1];
      stateStack.push(table.goto[topState].get(rule.lhs)!);
    } else {
      traces.push({ step: step++, stack: [...symbolStack], input: [...remaining], action: '接受' });
      break;
    }
  }
  return traces;
};

解析表构造
LR 解析表的构造是 LR 解析理论的核心。以 LR(0) 项集（Item Set）为基础，通过闭包和转移操作构建规范项集族，再从中提取 Action 表和 Goto 表。


interface LR0Item {
  ruleIndex: number;
  dotPos: number;
}

interface ItemSet {
  id: number;
  items: LR0Item[];
}

const closure = (items: LR0Item[], rules: GrammarRule[]): LR0Item[] => {
  const result = [...items];
  const added = new Set<string>();

  for (const item of items) {
    added.add(`${item.ruleIndex}:${item.dotPos}`);
  }

  let changed = true;
  while (changed) {
    changed = false;
    for (const item of [...result]) {
      const rule = rules[item.ruleIndex];
      if (item.dotPos >= rule.rhs.length) continue;

      const nextSymbol = rule.rhs[item.dotPos];
      for (const r of rules) {
        if (r.lhs === nextSymbol) {
          const key = `${r.index}:0`;
          if (!added.has(key)) {
            result.push({ ruleIndex: r.index, dotPos: 0 });
            added.add(key);
            changed = true;
          }
        }
      }
    }
  }
  return result;
};

const gotoSet = (items: LR0Item[], symbol: string, rules: GrammarRule[]): LR0Item[] => {
  const moved: LR0Item[] = [];
  for (const item of items) {
    const rule = rules[item.ruleIndex];
    if (item.dotPos < rule.rhs.length && rule.rhs[item.dotPos] === symbol) {
      moved.push({ ruleIndex: item.ruleIndex, dotPos: item.dotPos + 1 });
    }
  }
  return closure(moved, rules);
};

GLR 解析
当 LR 解析表存在冲突时，GLR（Generalized LR）解析器不会报错，而是分叉——同时沿多条路径解析，最终产出一个共享打包解析森林（Shared Packed Parse Forest，SPPF）。


interface GLRStack {
  id: number;
  state: number;
  parent: GLRStack | null;
  node: ASTNode | null;
}

interface GLRProcess {
  stack: GLRStack;
  pos: number;
}

class GLRParser {
  private table: ParseTable;
  private rules: GrammarRule[];

  constructor(table: ParseTable, rules: GrammarRule[]) {
    this.table = table;
    this.rules = rules;
  }

  parse(tokens: Token[]): ASTNode[] {
    let processes: GLRProcess[] = [{ stack: { id: 0, state: 0, parent: null, node: null }, pos: 0 }];

    while (processes.length > 0) {
      const nextProcesses: GLRProcess[] = [];
      const accepted: ASTNode[] = [];

      for (const proc of processes) {
        const state = proc.stack.state;
        const token = tokens[proc.pos];
        const actions = this.getAllActions(state, token);

        for (const action of actions) {
          if (action.type === 'shift') {
            const newStack: GLRStack = {
              id: proc.stack.id + 1,
              state: action.state,
              parent: proc.stack,
              node: { kind: 'Terminal', token },
            };
            nextProcesses.push({ stack: newStack, pos: proc.pos + 1 });
          } else if (action.type === 'reduce') {
            const reduced = this.reduceGLR(proc.stack, action.rule);
            if (reduced) nextProcesses.push({ ...proc, stack: reduced });
          } else if (action.type === 'accept') {
            if (proc.stack.node) accepted.push(proc.stack.node);
          }
        }
      }

      if (nextProcesses.length === 0) {
        return processes
          .filter((p) => tokens[p.pos]?.kind === TokenKind.EOF)
          .map((p) => p.stack.node!)
          .filter(Boolean);
      }
      processes = nextProcesses;
    }
    return [];
  }

  private getAllActions(state: number, token: Token): Action[] {
    const key = token.lexeme;
    const action = this.table.action[state].get(key);
    return action ? [action] : [];
  }

  private reduceGLR(stack: GLRStack, ruleIndex: number): GLRStack | null {
    const rule = this.rules[ruleIndex];
    let current: GLRStack | null = stack;
    for (let i = 0; i < rule.rhs.length; i++) {
      if (!current) return null;
      current = current.parent;
    }
    if (!current) return null;
    const gotoState = this.table.goto[current.state].get(rule.lhs);
    if (gotoState === undefined) return null;
    return { id: stack.id + 1, state: gotoState, parent: current, node: { kind: rule.lhs } };
  }
}

小结
    

  • LR 解析通过移进-归约机制自底向上构建语法树，由解析表驱动。
    • 

  • LALR(1) 是实践中最常用的 LR 变体，平衡了解析能力与表大小。
    • 

  • 移进-归约冲突可通过优先级和结合性声明解决，归约-归约冲突通常需要重写文法。
    • 

  • GLR 通过分叉解析处理任意上下文无关文法，对无歧义输入性能接近 O(n)。
    • 

  • tree-sitter 使用 GLR 解析保证对所有文法的鲁棒性，即使文法存在局部歧义也能正确解析。
