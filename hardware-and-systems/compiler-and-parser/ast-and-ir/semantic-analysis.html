语义分析

符号表

符号表（Symbol Table）是编译器在语义分析阶段维护的核心数据结构，记录程序中所有声明的标识符及其属性（类型、作用域、可变性等）。每次遇到声明，编译器向符号表中插入条目；每次遇到标识符使用，编译器查询符号表进行名称解析。

interface Symbol {
  name: string;
  kind: 'variable' | 'function' | 'parameter' | 'type';
  type: Type;
  mutable: boolean;
  declSpan: { start: number; end: number };
  scope: Scope;
}

type Type =
  | { kind: 'primitive'; name: string }
  | { kind: 'function'; params: Type[]; returnType: Type }
  | { kind: 'array'; element: Type }
  | { kind: 'generic'; name: string; typeParams: Type[] }
  | { kind: 'union'; members: Type[] }
  | { kind: 'unknown' };

interface Scope {
  id: number;
  parent: Scope | null;
  symbols: Map<string, Symbol>;
  children: Scope[];
  kind: 'global' | 'function' | 'block' | 'class';
}

class SymbolTable {
  private currentScope: Scope;
  private scopeCounter: number = 0;

  constructor() {
    this.currentScope = this.createScope(null, 'global');
  }

  private createScope(parent: Scope | null, kind: Scope['kind']): Scope {
    const scope: Scope = {
      id: this.scopeCounter++,
      parent,
      symbols: new Map(),
      children: [],
      kind,
    };
    if (parent) parent.children.push(scope);
    return scope;
  }

  enterScope(kind: Scope['kind']): void {
    this.currentScope = this.createScope(this.currentScope, kind);
  }

  exitScope(): void {
    if (this.currentScope.parent) {
      this.currentScope = this.currentScope.parent;
    }
  }

  declare(name: string, symbol: Omit<Symbol, 'scope'>): Symbol | null {
    if (this.currentScope.symbols.has(name)) {
      return null;
    }
    const sym: Symbol = { ...symbol, scope: this.currentScope } as Symbol;
    this.currentScope.symbols.set(name, sym);
    return sym;
  }

  resolve(name: string): Symbol | null {
    let scope: Scope | null = this.currentScope;
    while (scope !== null) {
      const sym = scope.symbols.get(name);
      if (sym) return sym;
      scope = scope.parent;
    }
    return null;
  }
}

作用域解析
作用域解析（Scope Resolution）是将源代码中的每个标识符使用绑定到其对应声明的过程。这一步也称为名称解析（Name Resolution），是语义分析的第一步。


interface ResolvedReference {
  name: string;
  symbol: Symbol;
  useSpan: { start: number; end: number };
}

class NameResolver {
  private symbolTable: SymbolTable;
  private references: ResolvedReference[] = [];
  private errors: string[] = [];

  constructor(symbolTable: SymbolTable) {
    this.symbolTable = symbolTable;
  }

  resolveProgram(program: Program): void {
    for (const stmt of program.body) {
      this.resolveStatement(stmt);
    }
  }

  private resolveStatement(stmt: Statement): void {
    switch (stmt.kind) {
      case 'VariableDeclaration':
        this.resolveVarDecl(stmt as VariableDeclaration);
        break;
      case 'FunctionDeclaration':
        this.resolveFuncDecl(stmt as FunctionDeclaration);
        break;
      case 'BlockStatement':
        this.symbolTable.enterScope('block');
        for (const s of (stmt as BlockStatement).body) this.resolveStatement(s);
        this.symbolTable.exitScope();
        break;
    }
  }

  private resolveVarDecl(decl: VariableDeclaration): void {
    if (decl.init) this.resolveExpression(decl.init);
    const result = this.symbolTable.declare(decl.name.name, {
      name: decl.name.name,
      kind: 'variable',
      type: { kind: 'unknown' },
      mutable: decl.mutable,
      declSpan: decl.span,
    });
    if (!result) {
      this.errors.push(`重复声明: '${decl.name.name}'`);
    }
  }

  private resolveFuncDecl(decl: FunctionDeclaration): void {
    this.symbolTable.declare(decl.name.name, {
      name: decl.name.name,
      kind: 'function',
      type: { kind: 'function', params: [], returnType: { kind: 'unknown' } },
      mutable: false,
      declSpan: decl.span,
    });
    this.symbolTable.enterScope('function');
    for (const param of decl.params) {
      this.symbolTable.declare(param.name, {
        name: param.name,
        kind: 'parameter',
        type: { kind: 'unknown' },
        mutable: false,
        declSpan: param.span,
      });
    }
    this.resolveStatement(decl.body);
    this.symbolTable.exitScope();
  }

  private resolveExpression(expr: Expression): void {
    switch (expr.kind) {
      case 'Identifier': {
        const sym = this.symbolTable.resolve((expr as Identifier).name);
        if (!sym) {
          this.errors.push(`未定义的标识符: '${(expr as Identifier).name}'`);
        } else {
          this.references.push({ name: (expr as Identifier).name, symbol: sym, useSpan: expr.span });
        }
        break;
      }
      case 'BinaryExpression': {
        const bin = expr as BinaryExpression;
        this.resolveExpression(bin.left);
        this.resolveExpression(bin.right);
        break;
      }
      case 'CallExpression': {
        const call = expr as CallExpression;
        this.resolveExpression(call.callee);
        for (const arg of call.arguments) this.resolveExpression(arg);
        break;
      }
    }
  }
}


类型检查
类型检查（Type Checking）验证程序中所有操作的类型正确性。类型检查器遍历 AST，为每个表达式推断或验证类型，报告类型不匹配的错误。


class TypeChecker {
  private symbolTable: SymbolTable;
  private errors: string[] = [];

  constructor(symbolTable: SymbolTable) {
    this.symbolTable = symbolTable;
  }

  checkExpression(expr: Expression): Type {
    switch (expr.kind) {
      case 'NumberLiteral':
        return { kind: 'primitive', name: 'number' };
      case 'StringLiteral':
        return { kind: 'primitive', name: 'string' };
      case 'Identifier': {
        const sym = this.symbolTable.resolve((expr as Identifier).name);
        return sym?.type ?? { kind: 'unknown' };
      }
      case 'BinaryExpression':
        return this.checkBinary(expr as BinaryExpression);
      case 'CallExpression':
        return this.checkCall(expr as CallExpression);
      default:
        return { kind: 'unknown' };
    }
  }

  private checkBinary(expr: BinaryExpression): Type {
    const leftType = this.checkExpression(expr.left);
    const rightType = this.checkExpression(expr.right);

    if (expr.operator === '+') {
      if (leftType.kind === 'primitive' && rightType.kind === 'primitive') {
        if (leftType.name === 'number' && rightType.name === 'number') {
          return { kind: 'primitive', name: 'number' };
        }
        if (leftType.name === 'string' || rightType.name === 'string') {
          return { kind: 'primitive', name: 'string' };
        }
      }
      this.errors.push(`运算符 '+' 不能应用于类型 '${typeToString(leftType)}' 和 '${typeToString(rightType)}'`);
    }

    if (['-', '*', '/'].includes(expr.operator)) {
      if (
        !this.isAssignableTo(leftType, { kind: 'primitive', name: 'number' }) ||
        !this.isAssignableTo(rightType, { kind: 'primitive', name: 'number' })
      ) {
        this.errors.push(`运算符 '${expr.operator}' 需要数字类型`);
      }
      return { kind: 'primitive', name: 'number' };
    }

    return { kind: 'unknown' };
  }

  private checkCall(expr: CallExpression): Type {
    const calleeType = this.checkExpression(expr.callee);
    if (calleeType.kind !== 'function') {
      this.errors.push('不能调用非函数类型的值');
      return { kind: 'unknown' };
    }
    if (expr.arguments.length !== calleeType.params.length) {
      this.errors.push(`期望 ${calleeType.params.length} 个参数，实际提供 ${expr.arguments.length} 个`);
    }
    for (let i = 0; i < Math.min(expr.arguments.length, calleeType.params.length); i++) {
      const argType = this.checkExpression(expr.arguments[i]);
      if (!this.isAssignableTo(argType, calleeType.params[i])) {
        this.errors.push(
          `参数类型不匹配: 期望 '${typeToString(calleeType.params[i])}'，实际 '${typeToString(argType)}'`,
        );
      }
    }
    return calleeType.returnType;
  }

  private isAssignableTo(source: Type, target: Type): boolean {
    if (target.kind === 'unknown' || source.kind === 'unknown') return true;
    if (source.kind === 'primitive' && target.kind === 'primitive') return source.name === target.name;
    return false;
  }
}

const typeToString = (type: Type): string => {
  switch (type.kind) {
    case 'primitive':
      return type.name;
    case 'function':
      return `(${type.params.map(typeToString).join(', ')}) => ${typeToString(type.returnType)}`;
    case 'array':
      return `${typeToString(type.element)}[]`;
    case 'unknown':
      return 'unknown';
    default:
      return '<type>';
  }
};

Hindley-Milner 类型推断
Hindley-Milner（HM）类型推断系统可以在没有类型注解的情况下推导出表达式的最通用类型。它基于合一（Unification）算法，是函数式语言（Haskell、OCaml、ML）类型系统的理论基础。


type HMType =
  | { tag: 'TVar'; name: string }
  | { tag: 'TCon'; name: string }
  | { tag: 'TFun'; param: HMType; ret: HMType };

type Substitution = Map<string, HMType>;

const applySubst = (subst: Substitution, type: HMType): HMType => {
  switch (type.tag) {
    case 'TVar':
      return subst.get(type.name) ?? type;
    case 'TCon':
      return type;
    case 'TFun':
      return { tag: 'TFun', param: applySubst(subst, type.param), ret: applySubst(subst, type.ret) };
  }
};

const occursIn = (name: string, type: HMType): boolean => {
  switch (type.tag) {
    case 'TVar':
      return type.name === name;
    case 'TCon':
      return false;
    case 'TFun':
      return occursIn(name, type.param) || occursIn(name, type.ret);
  }
};

const unify = (a: HMType, b: HMType): Substitution => {
  if (a.tag === 'TVar') {
    if (a.name === (b.tag === 'TVar' ? b.name : '')) return new Map();
    if (occursIn(a.name, b)) throw new Error(`无限类型: ${a.name} 出现在 ${hmTypeToString(b)} 中`);
    return new Map([[a.name, b]]);
  }
  if (b.tag === 'TVar') return unify(b, a);
  if (a.tag === 'TCon' && b.tag === 'TCon') {
    if (a.name !== b.name) throw new Error(`类型不匹配: ${a.name} ≠ ${b.name}`);
    return new Map();
  }
  if (a.tag === 'TFun' && b.tag === 'TFun') {
    const s1 = unify(a.param, b.param);
    const s2 = unify(applySubst(s1, a.ret), applySubst(s1, b.ret));
    return composeSubst(s1, s2);
  }
  throw new Error(`无法合一: ${hmTypeToString(a)} 和 ${hmTypeToString(b)}`);
};

const composeSubst = (s1: Substitution, s2: Substitution): Substitution => {
  const result = new Map<string, HMType>();
  for (const [k, v] of s2) result.set(k, applySubst(s1, v));
  for (const [k, v] of s1) if (!result.has(k)) result.set(k, v);
  return result;
};

const hmTypeToString = (type: HMType): string => {
  switch (type.tag) {
    case 'TVar':
      return type.name;
    case 'TCon':
      return type.name;
    case 'TFun':
      return `(${hmTypeToString(type.param)} → ${hmTypeToString(type.ret)})`;
  }
};


提示： TypeScript 的类型推断比经典 HM 系统更强大也更复杂——它支持子类型、联合类型、条件类型和模板字面量类型。但 HM 合一的核心思想（生成约束、求解约束）仍然是其类型推断引擎的理论基石。






小结
    

  • 符号表记录所有声明的标识符及其属性，通过链式作用域支持嵌套查找。
    • 

  • 名称解析将每个标识符使用绑定到其对应的声明，是语义分析的第一步。
    • 

  • 类型检查验证操作的类型正确性，为表达式推断类型并报告不匹配错误。
    • 

  • Hindley-Milner 合一算法可以在没有类型注解的情况下推导最通用类型，是函数式语言类型系统的基石。
    • 

  • 工业级类型检查器（TypeScript）在 HM 基础上扩展了子类型、联合类型、条件类型等高级特性。
