多传感器融合

核心概念

多传感器融合（Multi-Sensor Fusion）是将来自不同传感器的信息组合起来，生成比任何单一传感器更准确、更可靠的状态估计。在机器人定位中，不同传感器各有优劣——GPS 提供全局位置但精度有限，IMU 提供高频姿态但会漂移，LiDAR 提供精确距离但受环境影响，视觉提供丰富语义但受光照影响。融合的目标是利用各传感器的互补特性，弥补各自的不足。

多传感器融合按融合层次可分为三个级别：数据级融合（原始数据层面的合并）、特征级融合（提取特征后的融合）和决策级融合（各传感器独立决策后的融合）。在机器人定位中，特征级和决策级融合最为常用。

IMU 预积分与惯性导航
IMU（惯性测量单元）是多传感器融合中的核心传感器，它以极高的频率（200-1000Hz）提供加速度和角速度测量。IMU 的优势是完全自主、不依赖外部信号，劣势是积分过程会累积误差（漂移）。


IMU 预积分（Preintegration）技术解决了在优化框架中高效使用 IMU 数据的问题。传统方法需要在每次状态优化后重新积分 IMU 数据，预积分将两个关键帧之间的 IMU 测量预先积分为一个相对运动约束，这个约束独立于起始状态，只需在状态更新后简单修正。


import numpy as np
from scipy.spatial.transform import Rotation

class IMUPreintegration:
    def __init__(self, acc_noise=0.01, gyro_noise=0.001, acc_bias_noise=0.001, gyro_bias_noise=0.0001):
        self.delta_R = np.eye(3)
        self.delta_v = np.zeros(3)
        self.delta_p = np.zeros(3)
        self.dt_sum = 0.0

        self.acc_bias = np.zeros(3)
        self.gyro_bias = np.zeros(3)

        self.covariance = np.zeros((15, 15))
        self.acc_noise = acc_noise
        self.gyro_noise = gyro_noise

        self.jacobian = np.eye(15)

    def integrate(self, acc, gyro, dt):
        corrected_acc = acc - self.acc_bias
        corrected_gyro = gyro - self.gyro_bias

        dR = Rotation.from_rotvec(corrected_gyro * dt).as_matrix()

        self.delta_p += self.delta_v * dt + 0.5 * self.delta_R @ corrected_acc * dt**2
        self.delta_v += self.delta_R @ corrected_acc * dt
        self.delta_R = self.delta_R @ dR
        self.dt_sum += dt

        self._update_covariance(corrected_acc, corrected_gyro, dt)

    def _update_covariance(self, acc, gyro, dt):
        A = np.eye(15)
        A[0:3, 3:6] = -self.delta_R @ self._skew(acc) * dt
        A[3:6, 0:3] = np.eye(3) * dt
        A[6:9, 0:3] = 0.5 * np.eye(3) * dt**2
        A[6:9, 3:6] = np.eye(3) * dt

        B = np.zeros((15, 12))
        B[0:3, 0:3] = self.delta_R * dt
        B[3:6, 3:6] = self.delta_R * dt
        B[9:12, 6:9] = np.eye(3) * dt
        B[12:15, 9:12] = np.eye(3) * dt

        noise_cov = np.diag([
            *([self.gyro_noise**2] * 3),
            *([self.acc_noise**2] * 3),
            *([0.001**2] * 3),
            *([0.0001**2] * 3)
        ])

        self.covariance = A @ self.covariance @ A.T + B @ noise_cov @ B.T
        self.jacobian = A @ self.jacobian

    def _skew(self, v):
        return np.array([
            [0, -v[2], v[1]],
            [v[2], 0, -v[0]],
            [-v[1], v[0], 0]
        ])

    def predict(self, state_R, state_v, state_p, gravity):
        predicted_R = state_R @ self.delta_R
        predicted_v = state_v + gravity * self.dt_sum + state_R @ self.delta_v
        predicted_p = (state_p + state_v * self.dt_sum +
                      0.5 * gravity * self.dt_sum**2 +
                      state_R @ self.delta_p)
        return predicted_R, predicted_v, predicted_p
预积分的数学基础建立在 SO(3) 流形上。旋转部分在李群上进行积分，使用 BCH（Baker-Campbell-Hausdorff）近似处理小旋转的复合。这确保了旋转积分的数值精度和一致性，避免了欧拉角或旋转矩阵直接积分带来的数值问题。


松耦合与紧耦合融合
松耦合融合（Loosely-Coupled Fusion）将每个传感器子系统视为独立的定位系统，各自输出位姿估计，然后在位姿层面进行融合。松耦合的优势是模块化程度高、实现简单、容错性好——单个传感器失效不会导致整个系统崩溃。


import numpy as np

class LooseCoupledFusion:
    def __init__(self, dim_state=15):
        self.x = np.zeros(dim_state)
        self.P = np.eye(dim_state) * 0.1
        self.Q = np.eye(dim_state) * 0.01

    def imu_predict(self, acc, gyro, dt):
        F = np.eye(15)
        F[6:9, 3:6] = np.eye(3) * dt
        F[6:9, 0:3] = 0.5 * np.eye(3) * dt**2
        F[3:6, 0:3] = np.eye(3) * dt

        self.x = F @ self.x
        self.x[3:6] += acc * dt
        self.x[6:9] += self.x[3:6] * dt
        self.P = F @ self.P @ F.T + self.Q * dt

    def gps_update(self, gps_pos, R_gps):
        H = np.zeros((3, 15))
        H[0:3, 6:9] = np.eye(3)

        y = gps_pos - H @ self.x
        S = H @ self.P @ H.T + R_gps
        K = self.P @ H.T @ np.linalg.inv(S)
        self.x += K @ y
        self.P = (np.eye(15) - K @ H) @ self.P

    def lidar_update(self, lidar_pose, R_lidar):
        H = np.zeros((6, 15))
        H[0:3, 0:3] = np.eye(3)
        H[3:6, 6:9] = np.eye(3)

        z_pred = H @ self.x
        y = lidar_pose - z_pred
        S = H @ self.P @ H.T + R_lidar
        K = self.P @ H.T @ np.linalg.inv(S)
        self.x += K @ y
        self.P = (np.eye(15) - K @ H) @ self.P

    def get_state(self):
        return {
            'orientation': self.x[0:3],
            'velocity': self.x[3:6],
            'position': self.x[6:9],
            'acc_bias': self.x[9:12],
            'gyro_bias': self.x[12:15]
        }
紧耦合融合（Tightly-Coupled Fusion）直接在原始观测层面进行融合，将所有传感器的原始数据纳入统一的优化框架。紧耦合能够利用更多信息（如 GPS 的伪距、视觉的特征点位置），精度更高，但实现复杂度也更高。


VINS-Mono 是紧耦合视觉惯性导航的经典系统。它将视觉特征点的重投影误差和 IMU 预积分约束纳入统一的非线性优化问题中，使用滑动窗口（Sliding Window）策略限制优化规模。


LIO-SAM 是紧耦合 LiDAR-惯性里程计的代表系统。它以因子图（Factor Graph）为优化框架，IMU 预积分提供帧间约束，LiDAR 扫描匹配提供观测约束，GPS 提供全局约束，回环检测提供长期一致性约束。所有约束统一在 GTSAM 因子图中进行增量优化（iSAM2）。


时间同步与标定
多传感器融合的工程实践中，时间同步和外参标定是两个关键的基础问题。传感器之间的时间偏移和空间偏移如果不精确估计，会严重降低融合精度。


时间同步包括硬件同步和软件同步。硬件同步通过 PPS（Pulse Per Second）信号或 PTP（Precision Time Protocol）实现微秒级同步。软件同步通过时间戳对齐和插值实现，常用的方法包括最近邻匹配和线性插值。


import numpy as np
from collections import deque

class SensorSynchronizer:
    def __init__(self, max_delay=0.05):
        self.max_delay = max_delay
        self.buffers = {}

    def add_buffer(self, sensor_name, max_size=100):
        self.buffers[sensor_name] = deque(maxlen=max_size)

    def push(self, sensor_name, timestamp, data):
        self.buffers[sensor_name].append((timestamp, data))

    def get_synced(self, reference_sensor, target_sensor, ref_timestamp):
        target_buf = self.buffers[target_sensor]

        if len(target_buf) < 2:
            return None

        best_idx = None
        best_diff = float('inf')

        for i, (ts, _) in enumerate(target_buf):
            diff = abs(ts - ref_timestamp)
            if diff < best_diff:
                best_diff = diff
                best_idx = i

        if best_diff > self.max_delay:
            return None

        if best_idx > 0 and best_idx < len(target_buf) - 1:
            ts0, data0 = target_buf[best_idx - 1]
            ts1, data1 = target_buf[best_idx]
            ts2, data2 = target_buf[best_idx + 1]

            if abs(ts1 - ref_timestamp) < 1e-6:
                return data1

            if ref_timestamp < ts1 and ts1 > ts0:
                alpha = (ref_timestamp - ts0) / (ts1 - ts0)
                return self._interpolate(data0, data1, alpha)
            elif ref_timestamp > ts1 and ts2 > ts1:
                alpha = (ref_timestamp - ts1) / (ts2 - ts1)
                return self._interpolate(data1, data2, alpha)

        return target_buf[best_idx][1]

    def _interpolate(self, data0, data1, alpha):
        if isinstance(data0, np.ndarray):
            return data0 * (1 - alpha) + data1 * alpha
        return data0


class ExtrinsicCalibration:
    def __init__(self):
        self.R = np.eye(3)
        self.t = np.zeros(3)
        self.time_offset = 0.0

    def calibrate_hand_eye(self, motions_A, motions_B):
        n = len(motions_A)
        A_rotations = [m[:3, :3] for m in motions_A]
        B_rotations = [m[:3, :3] for m in motions_B]

        M = np.zeros((3, 3))
        for i in range(n):
            from scipy.spatial.transform import Rotation
            ra = Rotation.from_matrix(A_rotations[i]).as_rotvec()
            rb = Rotation.from_matrix(B_rotations[i]).as_rotvec()
            M += np.outer(rb, ra)

        U, _, Vt = np.linalg.svd(M)
        self.R = U @ Vt

        A_trans = np.array([m[:3, 3] for m in motions_A])
        B_trans = np.array([m[:3, 3] for m in motions_B])

        A_matrix = np.vstack([
            np.eye(3) - A_rotations[i] for i in range(n)
        ])
        b_vector = np.hstack([
            A_trans[i] - self.R @ B_trans[i] for i in range(n)
        ])

        self.t, _, _, _ = np.linalg.lstsq(A_matrix, b_vector, rcond=None)
        return self.R, self.t
外参标定确定传感器之间的相对位姿关系（旋转 R 和平移 t）。手眼标定（Hand-Eye Calibration）是经典方法，通过收集多组运动对（如机器人运动 + 相机观测到的运动），求解 AX=XB 方程组。在线标定方法将外参作为状态变量的一部分纳入优化，在系统运行过程中自动估计和更新。

小结
    

  • 多传感器融合利用不同传感器的互补特性，提供比单一传感器更准确可靠的估计
    • 

  • IMU 预积分是紧耦合融合的关键技术，将高频 IMU 数据高效纳入优化框架
    • 

  • 松耦合融合模块化程度高、容错性好；紧耦合融合精度更高但实现复杂
    • 

  • VINS-Mono 和 LIO-SAM 是紧耦合融合的代表系统，基于滑动窗口或因子图优化
    • 

  • 时间同步和外参标定是融合系统的工程基础，直接影响融合精度
    • 

  • 因子图（Factor Graph）提供了统一的多传感器融合优化框架
    • 

  • 实际系统需要处理传感器失效、退化场景和异常值，鲁棒性设计至关重要
