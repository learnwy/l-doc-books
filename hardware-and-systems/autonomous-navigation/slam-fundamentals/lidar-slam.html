激光 SLAM

核心概念

激光 SLAM（LiDAR SLAM）利用激光雷达（LiDAR）的距离测量数据同时完成定位和地图构建。与视觉 SLAM 相比，激光 SLAM 对光照变化免疫，直接提供精确的距离信息，无需特征匹配，在结构化环境中精度更高。

激光 SLAM 根据激光雷达类型分为 2D 和 3D 两大类。2D 激光 SLAM 使用单线扫描数据构建平面占据栅格地图，适合室内移动机器人；3D 激光 SLAM 使用多线或旋转式激光雷达构建三维点云地图，适合自动驾驶和野外环境。

扫描匹配算法
扫描匹配是激光 SLAM 的核心操作，目标是找到当前扫描与参考地图/扫描之间的最优对齐变换。ICP（Iterative Closest Point）及其变体是最基础的扫描匹配算法。


import numpy as np

class ICP:
    def __init__(self, max_iterations=50, tolerance=1e-6, max_correspondence_dist=1.0):
        self.max_iter = max_iterations
        self.tolerance = tolerance
        self.max_dist = max_correspondence_dist

    def align(self, source, target, initial_guess=np.eye(3)):
        T = initial_guess.copy()
        transformed = self._transform_points(source, T)

        for iteration in range(self.max_iter):
            correspondences = self._find_correspondences(transformed, target)

            if len(correspondences) < 3:
                break

            src_matched = np.array([source[i] for i, _ in correspondences])
            tgt_matched = np.array([target[j] for _, j in correspondences])

            T_update = self._compute_transform(src_matched, tgt_matched)
            T = T_update @ T
            transformed = self._transform_points(source, T)

            mean_error = np.mean([
                np.linalg.norm(transformed[i] - target[j])
                for i, j in correspondences
            ])

            if mean_error < self.tolerance:
                break

        return T, mean_error

    def _find_correspondences(self, source, target):
        from scipy.spatial import KDTree
        tree = KDTree(target)
        correspondences = []

        for i, point in enumerate(source):
            dist, idx = tree.query(point)
            if dist < self.max_dist:
                correspondences.append((i, idx))

        return correspondences

    def _compute_transform(self, source, target):
        centroid_s = np.mean(source, axis=0)
        centroid_t = np.mean(target, axis=0)

        s_centered = source - centroid_s
        t_centered = target - centroid_t

        H = s_centered.T @ t_centered
        U, S, Vt = np.linalg.svd(H)

        R = Vt.T @ U.T
        if np.linalg.det(R) < 0:
            Vt[-1, :] *= -1
            R = Vt.T @ U.T

        t = centroid_t - R @ centroid_s

        T = np.eye(3)
        T[:2, :2] = R[:2, :2]
        T[:2, 2] = t[:2]
        return T

    def _transform_points(self, points, T):
        R = T[:2, :2]
        t = T[:2, 2]
        return (R @ points[:, :2].T).T + t

class NDT:
    def __init__(self, cell_size=1.0, max_iterations=30):
        self.cell_size = cell_size
        self.max_iter = max_iterations

    def build_grid(self, points):
        grid = {}
        for point in points:
            key = (
                int(point[0] / self.cell_size),
                int(point[1] / self.cell_size)
            )
            if key not in grid:
                grid[key] = []
            grid[key].append(point)

        ndt_cells = {}
        for key, cell_points in grid.items():
            if len(cell_points) >= 3:
                pts = np.array(cell_points)
                mean = np.mean(pts, axis=0)
                cov = np.cov(pts.T) + np.eye(pts.shape[1]) * 0.01
                ndt_cells[key] = {'mean': mean, 'cov': cov, 'cov_inv': np.linalg.inv(cov)}

        return ndt_cells

    def score(self, point, cell):
        diff = point - cell['mean']
        exponent = -0.5 * diff @ cell['cov_inv'] @ diff
        return np.exp(exponent)
子图与增量建图
大规模环境的激光 SLAM 采用子图（Submap）策略：将地图分成多个局部子图，每个子图独立优化，通过子图间的约束维护全局一致性。Google Cartographer 是这一策略的典型实现。


import numpy as np

class Submap:
    def __init__(self, submap_id, origin_pose, resolution=0.05, size=200):
        self.id = submap_id
        self.origin = origin_pose.copy()
        self.resolution = resolution
        self.size = size
        self.grid = np.full((size, size), -1, dtype=np.int8)
        self.insertion_count = 0
        self.is_finished = False
        self.max_insertions = 90

    def insert_scan(self, scan_points, scan_pose):
        relative_pose = np.linalg.inv(self.origin) @ scan_pose

        robot_cell = self._world_to_grid(relative_pose[:2, 2])

        for point in scan_points:
            point_in_submap = relative_pose[:2, :2] @ point[:2] + relative_pose[:2, 2]
            point_cell = self._world_to_grid(point_in_submap)

            if self._in_bounds(point_cell):
                self._update_cell(point_cell, occupied=True)

            self._ray_trace(robot_cell, point_cell)

        self.insertion_count += 1
        if self.insertion_count >= self.max_insertions:
            self.is_finished = True

    def _world_to_grid(self, position):
        cx = int(position[0] / self.resolution + self.size / 2)
        cy = int(position[1] / self.resolution + self.size / 2)
        return np.array([cx, cy])

    def _in_bounds(self, cell):
        return 0 <= cell[0] < self.size and 0 <= cell[1] < self.size

    def _update_cell(self, cell, occupied):
        x, y = cell
        if occupied:
            self.grid[y, x] = min(self.grid[y, x] + 10, 100) if self.grid[y, x] >= 0 else 55
        else:
            self.grid[y, x] = max(self.grid[y, x] - 5, 0) if self.grid[y, x] >= 0 else 45

    def _ray_trace(self, start, end):
        dx = abs(end[0] - start[0])
        dy = abs(end[1] - start[1])
        sx = 1 if start[0] < end[0] else -1
        sy = 1 if start[1] < end[1] else -1
        err = dx - dy
        x, y = start[0], start[1]

        while (x, y) != (end[0], end[1]):
            if self._in_bounds(np.array([x, y])):
                self._update_cell(np.array([x, y]), occupied=False)
            e2 = 2 * err
            if e2 > -dy:
                err -= dy
                x += sx
            if e2 < dx:
                err += dx
                y += sy

class SubmapManager:
    def __init__(self, resolution=0.05):
        self.submaps = []
        self.resolution = resolution
        self.active_submaps = []

    def create_new_submap(self, pose):
        submap_id = len(self.submaps)
        submap = Submap(submap_id, pose, self.resolution)
        self.submaps.append(submap)
        self.active_submaps.append(submap)

        if len(self.active_submaps) > 2:
            self.active_submaps.pop(0)

        return submap

    def insert_scan(self, scan_points, scan_pose):
        for submap in self.active_submaps:
            if not submap.is_finished:
                submap.insert_scan(scan_points, scan_pose)

        all_finished = all(s.is_finished for s in self.active_submaps)
        if all_finished or not self.active_submaps:
            self.create_new_submap(scan_pose)
3D 点云特征提取
3D 激光 SLAM 需要从点云中提取几何特征（边缘、平面）用于扫描匹配。LOAM 系列算法的特征提取策略是在点云中区分边缘点和平面点。


import numpy as np

class PointCloudFeatureExtractor:
    def __init__(self, edge_threshold=1.0, planar_threshold=0.1):
        self.edge_threshold = edge_threshold
        self.planar_threshold = planar_threshold
        self.n_sectors = 6
        self.max_edge_per_sector = 20
        self.max_planar_per_sector = 40

    def extract_features(self, points):
        n = len(points)
        curvatures = np.zeros(n)

        for i in range(5, n - 5):
            diff_x = sum(points[i-5:i, 0]) + sum(points[i+1:i+6, 0]) - 10 * points[i, 0]
            diff_y = sum(points[i-5:i, 1]) + sum(points[i+1:i+6, 1]) - 10 * points[i, 1]
            diff_z = sum(points[i-5:i, 2]) + sum(points[i+1:i+6, 2]) - 10 * points[i, 2]
            curvatures[i] = diff_x**2 + diff_y**2 + diff_z**2

        edge_points = []
        planar_points = []

        angles = np.arctan2(points[:, 1], points[:, 0])
        sector_size = 2 * np.pi / self.n_sectors

        for sector in range(self.n_sectors):
            angle_min = -np.pi + sector * sector_size
            angle_max = angle_min + sector_size
            mask = (angles >= angle_min) & (angles < angle_max) & (curvatures > 0)

            sector_indices = np.where(mask)[0]
            sector_curvatures = curvatures[sector_indices]
            sorted_idx = np.argsort(sector_curvatures)[::-1]

            edge_count = 0
            for idx in sorted_idx:
                i = sector_indices[idx]
                if curvatures[i] > self.edge_threshold and edge_count < self.max_edge_per_sector:
                    edge_points.append(points[i])
                    edge_count += 1

            planar_count = 0
            for idx in sorted_idx[::-1]:
                i = sector_indices[idx]
                if curvatures[i] < self.planar_threshold and planar_count < self.max_planar_per_sector:
                    planar_points.append(points[i])
                    planar_count += 1

        return np.array(edge_points), np.array(planar_points)

class PointCloudPreprocessor:
    def __init__(self, min_range=0.5, max_range=100.0, voxel_size=0.1):
        self.min_range = min_range
        self.max_range = max_range
        self.voxel_size = voxel_size

    def preprocess(self, points):
        distances = np.linalg.norm(points[:, :3], axis=1)
        mask = (distances > self.min_range) & (distances < self.max_range)
        filtered = points[mask]

        downsampled = self.voxel_downsample(filtered)
        return downsampled

    def voxel_downsample(self, points):
        voxel_indices = np.floor(points[:, :3] / self.voxel_size).astype(int)

        voxel_dict = {}
        for i, key in enumerate(map(tuple, voxel_indices)):
            if key not in voxel_dict:
                voxel_dict[key] = []
            voxel_dict[key].append(points[i])

        downsampled = []
        for voxel_points in voxel_dict.values():
            centroid = np.mean(voxel_points, axis=0)
            downsampled.append(centroid)

        return np.array(downsampled)
小结
    

  • 激光 SLAM 利用 LiDAR 的精确距离测量实现高精度定位和建图，对光照免疫
    • 

  • ICP 及 NDT 是最基础的扫描匹配算法，NDT 在大规模场景中效率更高
    • 

  • 子图策略将大地图分解为局部子图，支持增量建图和全局优化
    • 

  • 3D 点云特征提取区分边缘点和平面点，用于 LOAM 系列的几何匹配
    • 

  • 点云预处理（距离滤波、体素降采样）是保证实时性的关键步骤
    • 

  • 现代激光 SLAM（Cartographer、LIO-SAM）已成为工业级导航的基础技术
