图优化

核心概念

图优化（Graph Optimization）是 SLAM 后端的核心技术，将定位和建图问题建模为一个因子图（Factor Graph），通过最小化所有约束的残差平方和来求解全局最优的机器人位姿和地图点位置。图优化替代了早期 SLAM 中的扩展卡尔曼滤波（EKF），在大规模问题上具有更好的可扩展性和精度。

在因子图中，节点表示待估计的状态变量（位姿、路标点），边（因子）表示节点间的约束关系（里程计测量、传感器观测、回环约束）。优化的目标是找到使所有因子残差最小化的节点状态值。

Gauss-Newton 优化
Gauss-Newton 方法是图优化中最常用的求解算法。它将非线性最小二乘问题在当前估计点线性化，通过迭代求解线性方程组逼近全局最优。


import numpy as np
from scipy.sparse import lil_matrix, csc_matrix
from scipy.sparse.linalg import spsolve

class PoseGraphOptimizer:
    def __init__(self):
        self.poses = {}
        self.edges = []

    def add_pose(self, pose_id, x, y, theta):
        self.poses[pose_id] = np.array([x, y, theta])

    def add_edge(self, from_id, to_id, dx, dy, dtheta, info_matrix):
        self.edges.append({
            'from': from_id, 'to': to_id,
            'measurement': np.array([dx, dy, dtheta]),
            'info': info_matrix
        })

    def optimize(self, max_iterations=30, tolerance=1e-6):
        n = len(self.poses)
        pose_ids = sorted(self.poses.keys())
        id_to_idx = {pid: i for i, pid in enumerate(pose_ids)}

        for iteration in range(max_iterations):
            H = lil_matrix((3 * n, 3 * n))
            b = np.zeros(3 * n)
            total_error = 0

            for edge in self.edges:
                i = id_to_idx[edge['from']]
                j = id_to_idx[edge['to']]

                xi = self.poses[edge['from']]
                xj = self.poses[edge['to']]
                z = edge['measurement']
                omega = edge['info']

                e, Ji, Jj = self._compute_error_and_jacobian(xi, xj, z)
                total_error += e @ omega @ e

                H[3*i:3*i+3, 3*i:3*i+3] += Ji.T @ omega @ Ji
                H[3*i:3*i+3, 3*j:3*j+3] += Ji.T @ omega @ Jj
                H[3*j:3*j+3, 3*i:3*i+3] += Jj.T @ omega @ Ji
                H[3*j:3*j+3, 3*j:3*j+3] += Jj.T @ omega @ Jj

                b[3*i:3*i+3] += Ji.T @ omega @ e
                b[3*j:3*j+3] += Jj.T @ omega @ e

            H[0:3, 0:3] += np.eye(3) * 1e6

            H_csc = csc_matrix(H)
            dx = spsolve(H_csc, -b)

            for pid in pose_ids:
                idx = id_to_idx[pid]
                self.poses[pid] += dx[3*idx:3*idx+3]
                self.poses[pid][2] = self._normalize_angle(self.poses[pid][2])

            if np.linalg.norm(dx) < tolerance:
                break

        return total_error

    def _compute_error_and_jacobian(self, xi, xj, z):
        c = np.cos(xi[2])
        s = np.sin(xi[2])
        R = np.array([[c, s], [-s, c]])

        dt = xj[:2] - xi[:2]
        e = np.zeros(3)
        e[:2] = R @ dt - z[:2]
        e[2] = self._normalize_angle(xj[2] - xi[2] - z[2])

        Ji = np.zeros((3, 3))
        Ji[:2, :2] = -R
        Ji[0, 2] = -s * dt[0] + c * dt[1]
        Ji[1, 2] = -c * dt[0] - s * dt[1]
        Ji[2, 2] = -1

        Jj = np.zeros((3, 3))
        Jj[:2, :2] = R
        Jj[2, 2] = 1

        return e, Ji, Jj

    def _normalize_angle(self, angle):
        while angle > np.pi:
            angle -= 2 * np.pi
        while angle < -np.pi:
            angle += 2 * np.pi
        return angle
增量式图优化（iSAM2）
增量式平滑与建图（iSAM2）通过维护贝叶斯树结构，避免每次新增约束时重新优化整个图，实现了近乎恒定时间的增量更新。这对于大规模实时 SLAM 至关重要。


import numpy as np

class BayesTreeNode:
    def __init__(self, clique_variables):
        self.variables = clique_variables
        self.factors = []
        self.children = []
        self.parent = None
        self.conditional = None

    def add_factor(self, factor):
        self.factors.append(factor)

class IncrementalOptimizer:
    def __init__(self):
        self.variables = {}
        self.factors = []
        self.affected_variables = set()
        self.elimination_order = []

    def add_variable(self, var_id, initial_value):
        self.variables[var_id] = initial_value.copy()
        self.affected_variables.add(var_id)

    def add_factor(self, factor):
        self.factors.append(factor)
        for var_id in factor['connected_variables']:
            self.affected_variables.add(var_id)

    def update(self):
        if not self.affected_variables:
            return

        affected_factors = []
        for factor in self.factors:
            if any(v in self.affected_variables for v in factor['connected_variables']):
                affected_factors.append(factor)

        for var_id in self.affected_variables:
            self._optimize_variable(var_id, affected_factors)

        self.affected_variables.clear()

    def _optimize_variable(self, var_id, factors):
        relevant = [f for f in factors if var_id in f['connected_variables']]

        if not relevant:
            return

        x = self.variables[var_id].copy()

        for _ in range(5):
            H = np.zeros((3, 3))
            b = np.zeros(3)

            for factor in relevant:
                e, J = factor['error_fn'](self.variables, var_id)
                omega = factor['info']
                H += J.T @ omega @ J
                b += J.T @ omega @ e

            H += np.eye(3) * 1e-4
            dx = np.linalg.solve(H, -b)

            self.variables[var_id] += dx

            if np.linalg.norm(dx) < 1e-6:
                break
g2o 与 GTSAM 实践
g2o 和 GTSAM 是 SLAM 领域两个最主流的图优化库。g2o 以 C++ 效率著称，GTSAM 以因子图模型的表达力见长。


import numpy as np

class G2OStyleOptimizer:
    def __init__(self):
        self.vertices = {}
        self.edges = []

    def add_vertex(self, vid, estimate, fixed=False):
        self.vertices[vid] = {
            'estimate': estimate.copy(),
            'fixed': fixed,
            'dimension': len(estimate)
        }

    def add_edge(self, v_from, v_to, measurement, information):
        self.edges.append({
            'vertices': (v_from, v_to),
            'measurement': measurement.copy(),
            'information': information.copy()
        })

    def optimize(self, iterations=20):
        free_vertices = {k: v for k, v in self.vertices.items() if not v['fixed']}
        n_vars = sum(v['dimension'] for v in free_vertices.values())
        var_offset = {}
        offset = 0
        for vid in sorted(free_vertices.keys()):
            var_offset[vid] = offset
            offset += free_vertices[vid]['dimension']

        for it in range(iterations):
            H = np.zeros((n_vars, n_vars))
            b = np.zeros(n_vars)
            chi2 = 0

            for edge in self.edges:
                vi, vj = edge['vertices']
                xi = self.vertices[vi]['estimate']
                xj = self.vertices[vj]['estimate']
                z = edge['measurement']
                omega = edge['information']

                e = self._edge_error(xi, xj, z)
                Ji, Jj = self._edge_jacobians(xi, xj, z)
                chi2 += e @ omega @ e

                if vi in var_offset:
                    oi = var_offset[vi]
                    d = free_vertices[vi]['dimension']
                    H[oi:oi+d, oi:oi+d] += Ji.T @ omega @ Ji
                    b[oi:oi+d] += Ji.T @ omega @ e

                    if vj in var_offset:
                        oj = var_offset[vj]
                        dj = free_vertices[vj]['dimension']
                        H[oi:oi+d, oj:oj+dj] += Ji.T @ omega @ Jj
                        H[oj:oj+dj, oi:oi+d] += Jj.T @ omega @ Ji

                if vj in var_offset:
                    oj = var_offset[vj]
                    dj = free_vertices[vj]['dimension']
                    H[oj:oj+dj, oj:oj+dj] += Jj.T @ omega @ Jj
                    b[oj:oj+dj] += Jj.T @ omega @ e

            H += np.eye(n_vars) * 1e-6
            dx = np.linalg.solve(H, -b)

            for vid in sorted(free_vertices.keys()):
                o = var_offset[vid]
                d = free_vertices[vid]['dimension']
                self.vertices[vid]['estimate'] += dx[o:o+d]

            if np.linalg.norm(dx) < 1e-8:
                break

        return chi2

    def _edge_error(self, xi, xj, z):
        c, s = np.cos(xi[2]), np.sin(xi[2])
        R = np.array([[c, s], [-s, c]])
        dt = xj[:2] - xi[:2]
        e = np.zeros(3)
        e[:2] = R @ dt - z[:2]
        e[2] = xj[2] - xi[2] - z[2]
        while e[2] > np.pi: e[2] -= 2*np.pi
        while e[2] < -np.pi: e[2] += 2*np.pi
        return e

    def _edge_jacobians(self, xi, xj, z):
        c, s = np.cos(xi[2]), np.sin(xi[2])
        dt = xj[:2] - xi[:2]
        Ji = np.array([
            [-c, -s, -s*dt[0]+c*dt[1]],
            [s, -c, -c*dt[0]-s*dt[1]],
            [0, 0, -1]
        ])
        Jj = np.array([
            [c, s, 0],
            [-s, c, 0],
            [0, 0, 1]
        ])
        return Ji, Jj
鲁棒核函数
实际 SLAM 中存在错误的回环检测或异常观测值，这些外点会严重影响优化结果。鲁棒核函数（Robust Kernel）通过减小大误差的权重来降低外点的影响。


import numpy as np

class RobustKernel:
    @staticmethod
    def huber(delta, e):
        abs_e = abs(e)
        if abs_e <= delta:
            return e * e
        else:
            return 2 * delta * abs_e - delta * delta

    @staticmethod
    def cauchy(delta, e):
        return delta * delta * np.log(1 + (e / delta) ** 2)

    @staticmethod
    def tukey(delta, e):
        if abs(e) <= delta:
            return (delta**2 / 6) * (1 - (1 - (e/delta)**2)**3)
        else:
            return delta**2 / 6

    @staticmethod
    def weight(kernel_type, delta, e):
        abs_e = max(abs(e), 1e-10)
        if kernel_type == 'huber':
            return min(1.0, delta / abs_e)
        elif kernel_type == 'cauchy':
            return 1.0 / (1 + (e / delta) ** 2)
        elif kernel_type == 'tukey':
            if abs_e <= delta:
                return (1 - (e/delta)**2)**2
            return 0.0
        return 1.0
