视觉 SLAM

核心概念

视觉 SLAM（Simultaneous Localization and Mapping）利用相机图像同时估计机器人位姿和构建环境地图。与激光 SLAM 相比，视觉 SLAM 的传感器成本更低、信息更丰富（包含纹理、颜色和语义），但计算复杂度更高且对光照变化敏感。

视觉 SLAM 系统由五个核心模块组成：传感器数据采集、前端视觉里程计、后端优化、回环检测和地图构建。前端负责帧间运动估计，后端负责全局一致性优化。

特征提取与匹配
特征提取是视觉 SLAM 前端的第一步，目标是从图像中检测稳定的关键点并计算描述子。ORB（Oriented FAST and Rotated BRIEF）是当前 SLAM 系统中最常用的特征，兼顾了计算效率和匹配鲁棒性。


import cv2
import numpy as np

class FeatureExtractor:
    def __init__(self, n_features=2000, scale_factor=1.2, n_levels=8):
        self.orb = cv2.ORB_create(
            nfeatures=n_features,
            scaleFactor=scale_factor,
            nlevels=n_levels,
            edgeThreshold=31,
            firstLevel=0,
            WTA_K=2,
            scoreType=cv2.ORB_HARRIS_SCORE,
            patchSize=31
        )

        self.bf_matcher = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=False)

    def detect_and_compute(self, image):
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) if len(image.shape) == 3 else image

        clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
        gray = clahe.apply(gray)

        keypoints, descriptors = self.orb.detectAndCompute(gray, None)
        return keypoints, descriptors

    def match_features(self, desc1, desc2, ratio_threshold=0.75):
        matches = self.bf_matcher.knnMatch(desc1, desc2, k=2)

        good_matches = []
        for m, n in matches:
            if m.distance < ratio_threshold * n.distance:
                good_matches.append(m)

        return good_matches

    def geometric_verification(self, kp1, kp2, matches, ransac_threshold=3.0):
        pts1 = np.float32([kp1[m.queryIdx].pt for m in matches])
        pts2 = np.float32([kp2[m.trainIdx].pt for m in matches])

        F, mask = cv2.findFundamentalMat(pts1, pts2, cv2.FM_RANSAC, ransac_threshold)

        inlier_matches = [m for m, flag in zip(matches, mask.ravel()) if flag]
        return inlier_matches, F
视觉里程计
视觉里程计（Visual Odometry, VO）通过连续帧之间的特征对应关系估计相机的增量运动。根据特征表示方式分为特征点法和直接法两大流派。


import cv2
import numpy as np

class VisualOdometry:
    def __init__(self, camera_matrix, dist_coeffs=None):
        self.K = camera_matrix
        self.dist_coeffs = dist_coeffs if dist_coeffs is not None else np.zeros(5)
        self.feature_extractor = FeatureExtractor()

        self.current_pose = np.eye(4)
        self.prev_frame = None
        self.prev_keypoints = None
        self.prev_descriptors = None

    def process_frame(self, frame):
        keypoints, descriptors = self.feature_extractor.detect_and_compute(frame)

        if self.prev_frame is None:
            self.prev_frame = frame
            self.prev_keypoints = keypoints
            self.prev_descriptors = descriptors
            return self.current_pose.copy()

        matches = self.feature_extractor.match_features(
            self.prev_descriptors, descriptors
        )

        if len(matches) < 8:
            return self.current_pose.copy()

        inliers, F = self.feature_extractor.geometric_verification(
            self.prev_keypoints, keypoints, matches
        )

        E = self.K.T @ F @ self.K

        pts1 = np.float32([self.prev_keypoints[m.queryIdx].pt for m in inliers])
        pts2 = np.float32([keypoints[m.trainIdx].pt for m in inliers])

        _, R, t, mask = cv2.recoverPose(E, pts1, pts2, self.K)

        T_relative = np.eye(4)
        T_relative[:3, :3] = R
        T_relative[:3, 3] = t.flatten()

        self.current_pose = self.current_pose @ T_relative

        self.prev_frame = frame
        self.prev_keypoints = keypoints
        self.prev_descriptors = descriptors

        return self.current_pose.copy()

    def triangulate_points(self, kp1, kp2, matches, R, t):
        pts1 = np.float32([kp1[m.queryIdx].pt for m in matches])
        pts2 = np.float32([kp2[m.trainIdx].pt for m in matches])

        P1 = self.K @ np.hstack([np.eye(3), np.zeros((3, 1))])
        P2 = self.K @ np.hstack([R, t])

        points_4d = cv2.triangulatePoints(P1, P2, pts1.T, pts2.T)
        points_3d = (points_4d[:3] / points_4d[3]).T

        valid = points_3d[:, 2] > 0
        return points_3d[valid]

class StereoVisualOdometry:
    def __init__(self, K_left, K_right, baseline):
        self.K_left = K_left
        self.K_right = K_right
        self.baseline = baseline
        self.feature_extractor = FeatureExtractor()
        self.current_pose = np.eye(4)
        self.prev_kp = None
        self.prev_desc = None
        self.prev_points_3d = None

    def process_stereo_pair(self, left_frame, right_frame):
        kp_left, desc_left = self.feature_extractor.detect_and_compute(left_frame)
        kp_right, desc_right = self.feature_extractor.detect_and_compute(right_frame)

        stereo_matches = self.feature_extractor.match_features(desc_left, desc_right)

        points_3d = []
        for m in stereo_matches:
            pl = kp_left[m.queryIdx].pt
            pr = kp_right[m.trainIdx].pt
            disparity = pl[0] - pr[0]
            if disparity > 0:
                depth = self.baseline * self.K_left[0, 0] / disparity
                x = (pl[0] - self.K_left[0, 2]) * depth / self.K_left[0, 0]
                y = (pl[1] - self.K_left[1, 2]) * depth / self.K_left[1, 1]
                points_3d.append([x, y, depth])

        points_3d = np.array(points_3d)

        if self.prev_kp is not None:
            temporal_matches = self.feature_extractor.match_features(
                self.prev_desc, desc_left
            )

            if len(temporal_matches) >= 6:
                obj_pts = np.array([self.prev_points_3d[m.queryIdx] for m in temporal_matches], dtype=np.float32)
                img_pts = np.array([kp_left[m.trainIdx].pt for m in temporal_matches], dtype=np.float32)

                success, rvec, tvec, inliers = cv2.solvePnPRansac(
                    obj_pts, img_pts, self.K_left, None
                )

                if success:
                    R, _ = cv2.Rodrigues(rvec)
                    T = np.eye(4)
                    T[:3, :3] = R
                    T[:3, 3] = tvec.flatten()
                    self.current_pose = self.current_pose @ np.linalg.inv(T)

        self.prev_kp = kp_left
        self.prev_desc = desc_left
        self.prev_points_3d = points_3d

        return self.current_pose.copy()
VIO：视觉惯性里程计
视觉惯性里程计（Visual-Inertial Odometry）融合相机和 IMU 数据，利用 IMU 的高频角速度和加速度测量弥补视觉在快速运动和纹理缺乏场景下的不足。


import numpy as np

class VIOPreintegration:
    def __init__(self, gravity=np.array([0, 0, -9.81])):
        self.gravity = gravity
        self.delta_R = np.eye(3)
        self.delta_v = np.zeros(3)
        self.delta_p = np.zeros(3)
        self.dt_sum = 0

    def integrate(self, gyro, accel, dt):
        R_update = self._exp_map(gyro * dt)
        self.delta_p += self.delta_v * dt + 0.5 * self.delta_R @ accel * dt**2
        self.delta_v += self.delta_R @ accel * dt
        self.delta_R = self.delta_R @ R_update
        self.dt_sum += dt

    def _exp_map(self, omega):
        theta = np.linalg.norm(omega)
        if theta < 1e-10:
            return np.eye(3)
        axis = omega / theta
        K = np.array([
            [0, -axis[2], axis[1]],
            [axis[2], 0, -axis[0]],
            [-axis[1], axis[0], 0]
        ])
        return np.eye(3) + np.sin(theta) * K + (1 - np.cos(theta)) * K @ K

    def predict_state(self, R_prev, v_prev, p_prev):
        R_new = R_prev @ self.delta_R
        v_new = v_prev + self.gravity * self.dt_sum + R_prev @ self.delta_v
        p_new = (p_prev + v_prev * self.dt_sum +
                 0.5 * self.gravity * self.dt_sum**2 + R_prev @ self.delta_p)
        return R_new, v_new, p_new

class SimpleVIO:
    def __init__(self, camera_matrix, imu_rate=200):
        self.K = camera_matrix
        self.imu_rate = imu_rate
        self.vo = VisualOdometry(camera_matrix)
        self.preintegrator = VIOPreintegration()

        self.state = {
            'R': np.eye(3),
            'v': np.zeros(3),
            'p': np.zeros(3),
            'bg': np.zeros(3),
            'ba': np.zeros(3)
        }

    def process_imu(self, gyro, accel):
        corrected_gyro = gyro - self.state['bg']
        corrected_accel = accel - self.state['ba']
        dt = 1.0 / self.imu_rate
        self.preintegrator.integrate(corrected_gyro, corrected_accel, dt)

    def process_image(self, frame):
        vo_pose = self.vo.process_frame(frame)

        R_pred, v_pred, p_pred = self.preintegrator.predict_state(
            self.state['R'], self.state['v'], self.state['p']
        )

        alpha = 0.7
        R_vo = vo_pose[:3, :3]
        p_vo = vo_pose[:3, 3]

        self.state['R'] = R_pred
        self.state['v'] = v_pred
        self.state['p'] = alpha * p_pred + (1 - alpha) * p_vo

        self.preintegrator = VIOPreintegration()

        return self.get_pose()

    def get_pose(self):
        T = np.eye(4)
        T[:3, :3] = self.state['R']
        T[:3, 3] = self.state['p']
        return T
地图构建与表示
视觉 SLAM 构建的地图根据密度可分为稀疏地图、半稠密地图和稠密地图。不同的地图表示方式服务于不同的下游任务。


import numpy as np

class SparseMap:
    def __init__(self):
        self.map_points = {}
        self.keyframes = {}
        self.point_id_counter = 0

    def add_point(self, position, descriptor, keyframe_id, keypoint_idx):
        point_id = self.point_id_counter
        self.point_id_counter += 1

        self.map_points[point_id] = {
            'position': position.copy(),
            'descriptor': descriptor.copy(),
            'observations': [(keyframe_id, keypoint_idx)],
            'normal': np.zeros(3)
        }
        return point_id

    def add_keyframe(self, frame_id, pose, keypoints, descriptors):
        self.keyframes[frame_id] = {
            'pose': pose.copy(),
            'keypoints': keypoints,
            'descriptors': descriptors,
            'point_ids': []
        }

    def get_local_map(self, pose, radius=10.0):
        position = pose[:3, 3]
        local_points = {}

        for pid, point in self.map_points.items():
            dist = np.linalg.norm(point['position'] - position)
            if dist < radius:
                local_points[pid] = point

        return local_points

class OccupancyGrid:
    def __init__(self, resolution=0.05, width=200, height=200):
        self.resolution = resolution
        self.width = width
        self.height = height
        self.grid = np.full((height, width), 50, dtype=np.int8)
        self.origin_x = -width * resolution / 2
        self.origin_y = -height * resolution / 2

    def update_from_pointcloud(self, points, robot_pose):
        robot_x, robot_y = robot_pose[:2]

        for point in points:
            px, py = point[:2]
            gx = int((px - self.origin_x) / self.resolution)
            gy = int((py - self.origin_y) / self.resolution)

            if 0 <= gx < self.width and 0 <= gy < self.height:
                self.grid[gy, gx] = min(self.grid[gy, gx] + 10, 100)

            self._bresenham_free(
                int((robot_x - self.origin_x) / self.resolution),
                int((robot_y - self.origin_y) / self.resolution),
                gx, gy
            )

    def _bresenham_free(self, x0, y0, x1, y1):
        dx, dy = abs(x1 - x0), abs(y1 - y0)
        sx = 1 if x0 < x1 else -1
        sy = 1 if y0 < y1 else -1
        err = dx - dy

        while (x0, y0) != (x1, y1):
            if 0 <= x0 < self.width and 0 <= y0 < self.height:
                self.grid[y0, x0] = max(self.grid[y0, x0] - 5, 0)
            e2 = 2 * err
            if e2 > -dy:
                err -= dy
                x0 += sx
            if e2 < dx:
                err += dx
                y0 += sy
小结
    

  • 视觉 SLAM 利用相机图像同时定位和建图，ORB 特征是当前最常用的特征表示
    • 

  • 视觉里程计通过帧间特征匹配估计增量运动，双目和 RGB-D 可直接获取深度
    • 

  • VIO 融合 IMU 预积分和视觉观测，在快速运动和弱纹理场景中显著提升鲁棒性
    • 

  • 地图表示从稀疏点云到稠密占据栅格，不同密度服务于不同下游任务
    • 

  • 特征提取、匹配和几何验证构成前端的核心流水线，RANSAC 去除外点至关重要
    • 

  • 现代视觉 SLAM 系统（ORB-SLAM3、VINS-Fusion）已在多种场景中达到实用水平
    •
