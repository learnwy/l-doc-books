逆运动学

核心概念

逆运动学（Inverse Kinematics, IK）是正运动学的逆问题：给定末端执行器的目标位姿，求解满足约束的关节角度组合。这是机器人运动规划和轨迹跟踪的核心环节，直接决定了机器人能否到达期望的工作位置。

逆运动学的难度远超正运动学，主要体现在三个方面：解可能不存在（目标超出工作空间）、解可能不唯一（同一位姿对应多组关节角）、以及非线性方程组的求解复杂度。

解析法求解
解析法（Analytical/Closed-form Solution）通过代数运算直接求出关节角的封闭表达式，是效率最高的逆运动学求解方法。其适用条件是机器人的运动学结构满足 Pieper 准则：6 自由度机器人中相邻 3 个旋转轴交于一点（腕部中心），或存在足够多的平行轴。


以 6 自由度工业机器人（如 UR 系列）为例，利用腕部解耦将问题分解为位置逆解和姿态逆解：


import numpy as np

def ik_analytical_6dof(T_target, dh_params):
    px, py, pz = T_target[:3, 3]
    R_target = T_target[:3, :3]

    d1, a2, a3, d4, d5, d6 = [p[1] for p in dh_params[:6]]

    wrist_center = T_target[:3, 3] - d6 * R_target[:3, 2]
    xc, yc, zc = wrist_center

    theta1_a = np.arctan2(yc, xc)
    theta1_b = np.arctan2(-yc, -xc)

    r = np.sqrt(xc**2 + yc**2)
    s = zc - d1
    D = (r**2 + s**2 - a2**2 - a3**2) / (2 * a2 * a3)

    if abs(D) > 1:
        return None

    theta3_a = np.arctan2(np.sqrt(1 - D**2), D)
    theta3_b = np.arctan2(-np.sqrt(1 - D**2), D)

    for theta1 in [theta1_a, theta1_b]:
        for theta3 in [theta3_a, theta3_b]:
            k1 = a2 + a3 * np.cos(theta3)
            k2 = a3 * np.sin(theta3)
            theta2 = np.arctan2(s, r) - np.arctan2(k2, k1)

            R_03 = compute_R03(theta1, theta2, theta3)
            R_36 = R_03.T @ R_target
            theta4, theta5, theta6 = euler_from_rotation(R_36)

            yield [theta1, theta2, theta3, theta4, theta5, theta6]

def compute_R03(t1, t2, t3):
    c1, s1 = np.cos(t1), np.sin(t1)
    c23, s23 = np.cos(t2 + t3), np.sin(t2 + t3)
    return np.array([
        [c1 * c23, -c1 * s23, s1],
        [s1 * c23, -s1 * s23, -c1],
        [s23,       c23,       0]
    ])

def euler_from_rotation(R):
    theta5 = np.arccos(R[2, 2])
    if abs(np.sin(theta5)) > 1e-6:
        theta4 = np.arctan2(R[1, 2], R[0, 2])
        theta6 = np.arctan2(R[2, 1], -R[2, 0])
    else:
        theta4 = 0
        theta6 = np.arctan2(-R[0, 1], R[0, 0])
    return theta4, theta5, theta6
解析法的关键挑战是奇异构型的处理。当机器人处于奇异位置时（如手臂完全伸展或腕部轴对齐），某些关节角变得不确定，分母趋近于零。工程上通常通过设置保护阈值和退化求解策略来应对。


数值迭代法
数值迭代法是最通用的逆运动学求解方法，适用于任意构型的机器人。其核心思想是将逆运动学转化为优化问题：最小化当前末端位姿与目标位姿之间的误差。


雅可比矩阵迭代法是最经典的数值方法，利用速度级运动学关系 x˙=J(q)q˙\dot{x} = J(q) \dot{q}x˙=J(q)q˙​ 的逆关系进行迭代：


import numpy as np

def ik_jacobian(robot, T_target, q_init, max_iter=100, tol=1e-6):
    q = q_init.copy()

    for i in range(max_iter):
        T_current = robot.fkine(q)

        pos_error = T_target[:3, 3] - T_current[:3, 3]
        R_error = T_target[:3, :3] @ T_current[:3, :3].T
        angle_axis = rotation_to_angle_axis(R_error)

        error = np.concatenate([pos_error, angle_axis])

        if np.linalg.norm(error) < tol:
            return q, True, i

        J = robot.jacobian(q)

        dq = damped_least_squares(J, error, damping=0.01)
        q = q + dq
        q = clamp_joints(q, robot.joint_limits)

    return q, False, max_iter

def damped_least_squares(J, error, damping=0.01):
    JJT = J @ J.T
    n = JJT.shape[0]
    return J.T @ np.linalg.solve(JJT + damping**2 * np.eye(n), error)

def rotation_to_angle_axis(R):
    angle = np.arccos(np.clip((np.trace(R) - 1) / 2, -1, 1))
    if angle < 1e-10:
        return np.zeros(3)
    axis = np.array([
        R[2, 1] - R[1, 2],
        R[0, 2] - R[2, 0],
        R[1, 0] - R[0, 1]
    ]) / (2 * np.sin(angle))
    return angle * axis

def clamp_joints(q, limits):
    return np.clip(q, limits[:, 0], limits[:, 1])

冗余机器人逆运动学
当机器人的自由度数大于任务空间维度时（如 7-DOF 机器人在 6-DOF 任务中），存在无穷多组逆运动学解，形成零空间运动。冗余度的利用是高级机器人控制的核心课题。


零空间投影方法允许在满足主任务的同时优化次级目标（如避障、关节限位避免、最大化可操作度）：


import numpy as np

def ik_redundant(robot, T_target, q_init, secondary_objective,
                 max_iter=200, tol=1e-6, alpha=0.5):
    q = q_init.copy()
    n_joints = len(q)

    for i in range(max_iter):
        T_current = robot.fkine(q)
        error = compute_task_error(T_target, T_current)

        if np.linalg.norm(error) < tol:
            return q, True

        J = robot.jacobian(q)
        J_pinv = np.linalg.pinv(J)

        dq_primary = J_pinv @ error

        N = np.eye(n_joints) - J_pinv @ J
        grad = secondary_objective.gradient(q)
        dq_secondary = N @ grad

        dq = dq_primary + alpha * dq_secondary
        q = q + dq

    return q, False

class JointLimitAvoidance:
    def __init__(self, limits):
        self.q_min = limits[:, 0]
        self.q_max = limits[:, 1]
        self.q_mid = (self.q_min + self.q_max) / 2

    def gradient(self, q):
        n = len(q)
        grad = np.zeros(n)
        for i in range(n):
            range_i = self.q_max[i] - self.q_min[i]
            grad[i] = -(q[i] - self.q_mid[i]) / (range_i / 2)**2
        return grad / np.linalg.norm(grad + 1e-10)
基于优化的全局求解
传统迭代法的局限在于对初始值敏感，容易陷入局部最优。基于优化的全局求解方法通过构造合适的目标函数和约束，在更大的搜索空间中寻找最优解。


from scipy.optimize import minimize, differential_evolution

def ik_optimization(robot, T_target, joint_limits):
    def objective(q):
        T_current = robot.fkine(q)
        pos_error = np.linalg.norm(T_target[:3, 3] - T_current[:3, 3])
        R_error = T_target[:3, :3] @ T_current[:3, :3].T
        angle_error = np.arccos(np.clip((np.trace(R_error) - 1) / 2, -1, 1))
        return pos_error**2 + 0.5 * angle_error**2

    bounds = [(l, u) for l, u in joint_limits]

    result = differential_evolution(
        objective,
        bounds=bounds,
        maxiter=1000,
        tol=1e-8,
        seed=42,
        polish=True
    )

    if result.fun < 1e-6:
        return result.x, True
    return result.x, False

def ik_with_constraints(robot, T_target, joint_limits, obstacles):
    def objective(q):
        T_current = robot.fkine(q)
        return pose_distance(T_target, T_current)

    constraints = []
    for obs in obstacles:
        def collision_constraint(q, obstacle=obs):
            min_dist = robot.min_distance_to_obstacle(q, obstacle)
            return min_dist - 0.05
        constraints.append({
            'type': 'ineq',
            'fun': collision_constraint
        })

    q0 = np.mean(joint_limits, axis=1)
    bounds = [(l, u) for l, u in joint_limits]

    result = minimize(
        objective, q0, method='SLSQP',
        bounds=bounds, constraints=constraints
    )
    return result.x, result.success
小结
    

  • 逆运动学是机器人运动规划的核心，将任务空间目标转化为关节空间指令
    • 

  • 解析法适用于满足 Pieper 准则的特定构型，求解速度快但适用范围有限
    • 

  • 数值迭代法通用性强，阻尼最小二乘法有效处理奇异性问题
    • 

  • 冗余机器人可利用零空间运动优化次级目标，如避障和关节限位
    • 

  • 全局优化方法（如差分进化）可避免局部最优，但计算开销较大
    • 

  • 工程实践中通常组合多种方法：解析法快速求解 + 数值法精细调整
    •
