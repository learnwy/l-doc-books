工作空间分析

核心概念

工作空间（Workspace）是机器人末端执行器在操作空间中所能到达的全部位置和姿态的集合。工作空间分析是机器人设计、选型和任务规划的关键环节，直接决定了机器人能否完成指定的操作任务。

工作空间可分为可达工作空间（Reachable Workspace）和灵巧工作空间（Dexterous Workspace）。可达工作空间是末端至少以一种姿态能到达的所有位置集合；灵巧工作空间是末端以任意姿态都能到达的位置集合，通常是可达工作空间的子集。

数值法求解工作空间
数值法通过对关节空间进行系统采样，利用正运动学计算对应的末端位置，从而近似描绘工作空间的边界。这种方法适用于任意构型的机器人，是最通用的工作空间分析手段。


import numpy as np
from itertools import product

def compute_workspace_numerical(robot_fk, joint_limits, samples_per_joint=20):
    n_joints = len(joint_limits)
    joint_ranges = [
        np.linspace(limits[0], limits[1], samples_per_joint)
        for limits in joint_limits
    ]

    positions = []
    configs = []

    for joint_values in product(*joint_ranges):
        q = np.array(joint_values)
        T = robot_fk(q)
        pos = T[:3, 3]
        positions.append(pos)
        configs.append(q)

    positions = np.array(positions)
    return positions, configs

def workspace_boundary(positions, resolution=0.01):
    from scipy.spatial import ConvexHull

    try:
        hull = ConvexHull(positions)
        volume = hull.volume
        surface_area = hull.area

        center = np.mean(positions, axis=0)
        distances = np.linalg.norm(positions - center, axis=1)
        max_reach = np.max(distances)
        min_reach = np.min(distances)

        return {
            'volume': volume,
            'surface_area': surface_area,
            'max_reach': max_reach,
            'min_reach': min_reach,
            'center': center,
            'hull': hull
        }
    except Exception:
        return None

蒙特卡洛方法是另一种高效的数值方法，通过随机采样关节空间来统计工作空间的覆盖情况：


import numpy as np

def workspace_monte_carlo(robot_fk, joint_limits, n_samples=100000):
    n_joints = len(joint_limits)

    random_joints = np.random.uniform(
        low=[l[0] for l in joint_limits],
        high=[l[1] for l in joint_limits],
        size=(n_samples, n_joints)
    )

    positions = np.zeros((n_samples, 3))
    for i, q in enumerate(random_joints):
        T = robot_fk(q)
        positions[i] = T[:3, 3]

    return positions

def workspace_density_map(positions, grid_resolution=0.02):
    x_range = (positions[:, 0].min(), positions[:, 0].max())
    y_range = (positions[:, 1].min(), positions[:, 1].max())
    z_range = (positions[:, 2].min(), positions[:, 2].max())

    x_bins = int((x_range[1] - x_range[0]) / grid_resolution)
    y_bins = int((y_range[1] - y_range[0]) / grid_resolution)
    z_bins = int((z_range[1] - z_range[0]) / grid_resolution)

    density, edges = np.histogramdd(
        positions, bins=(x_bins, y_bins, z_bins),
        range=(x_range, y_range, z_range)
    )

    occupied = np.sum(density > 0)
    total = density.size
    coverage = occupied / total

    return density, edges, coverage
解析法求解工作空间边界
对于结构较简单的机器人（如平面 2-DOF、3-DOF），可以通过解析方法精确描述工作空间边界。解析法的优势是能给出精确的边界方程，便于后续的数学分析。

import numpy as np

def analytical_workspace_2dof(L1, L2, theta1_range, theta2_range):
    r_max = L1 + L2
    r_min = abs(L1 - L2)

    theta1_min, theta1_max = theta1_range
    theta2_min, theta2_max = theta2_range

    boundary_points = []
    n_points = 500

    for theta2 in [theta2_min, theta2_max]:
        for theta1 in np.linspace(theta1_min, theta1_max, n_points):
            x = L1 * np.cos(theta1) + L2 * np.cos(theta1 + theta2)
            y = L1 * np.sin(theta1) + L2 * np.sin(theta1 + theta2)
            boundary_points.append([x, y])

    for theta1 in [theta1_min, theta1_max]:
        for theta2 in np.linspace(theta2_min, theta2_max, n_points):
            x = L1 * np.cos(theta1) + L2 * np.cos(theta1 + theta2)
            y = L1 * np.sin(theta1) + L2 * np.sin(theta1 + theta2)
            boundary_points.append([x, y])

    return np.array(boundary_points), r_min, r_max

def workspace_area_2dof(L1, L2, theta1_range, theta2_range):
    if theta2_range == (-np.pi, np.pi):
        return np.pi * ((L1 + L2)**2 - (L1 - L2)**2)

    delta_theta1 = theta1_range[1] - theta1_range[0]
    delta_theta2 = theta2_range[1] - theta2_range[0]
    area = delta_theta1 * (
        L1**2 * delta_theta2 +
        2 * L1 * L2 * (np.sin(theta2_range[1]) - np.sin(theta2_range[0]))
    )
    return abs(area)
奇异性与工作空间质量
奇异构型是工作空间分析中不可忽视的因素。在奇异点附近，机器人的末端运动能力急剧下降，某些方向的速度或力传递能力趋近于零。


可操作度椭球（Manipulability Ellipsoid）是量化工作空间质量的核心工具。其数学基础是雅可比矩阵的奇异值分解：


import numpy as np

def manipulability_analysis(robot, q):
    J = robot.jacobian(q)

    U, sigma, Vt = np.linalg.svd(J)

    w_yoshikawa = np.prod(sigma)

    condition_number = sigma[0] / sigma[-1] if sigma[-1] > 1e-10 else np.inf

    w_isotropy = sigma[-1] / sigma[0] if sigma[0] > 1e-10 else 0

    return {
        'manipulability': w_yoshikawa,
        'condition_number': condition_number,
        'isotropy': w_isotropy,
        'singular_values': sigma,
        'ellipsoid_axes': U,
        'is_singular': sigma[-1] < 1e-6
    }

def workspace_quality_map(robot_fk, robot_jacobian, joint_limits, grid_size=20):
    quality_map = {}

    joint_ranges = [
        np.linspace(l[0], l[1], grid_size)
        for l in joint_limits
    ]

    for q in np.array(list(product(*joint_ranges))):
        T = robot_fk(q)
        pos = tuple(np.round(T[:3, 3], 3))

        J = robot_jacobian(q)
        w = np.sqrt(np.linalg.det(J @ J.T))

        if pos not in quality_map or w > quality_map[pos]:
            quality_map[pos] = w

    return quality_map

工作空间优化设计
在机器人设计阶段，连杆参数的选择直接影响工作空间的大小和形状。工作空间优化的目标是在满足任务需求的前提下，找到最优的连杆尺寸和关节配置。


from scipy.optimize import minimize

def workspace_optimization(task_points, n_joints=6):
    def objective(design_params):
        link_lengths = design_params[:n_joints]

        workspace_volume = estimate_workspace_volume(link_lengths)

        reachability = 0
        for point in task_points:
            if is_reachable(link_lengths, point):
                reachability += 1
        reachability_ratio = reachability / len(task_points)

        total_length = sum(link_lengths)

        return -(0.6 * reachability_ratio +
                 0.3 * workspace_volume / (4/3 * np.pi * total_length**3) -
                 0.1 * total_length)

    bounds = [(0.05, 0.8)] * n_joints

    result = minimize(
        objective,
        x0=[0.3] * n_joints,
        bounds=bounds,
        method='L-BFGS-B'
    )

    return result.x

def collision_free_workspace(robot_fk, joint_limits, obstacles, n_samples=50000):
    positions = []

    for _ in range(n_samples):
        q = np.random.uniform(
            [l[0] for l in joint_limits],
            [l[1] for l in joint_limits]
        )

        T = robot_fk(q)
        pos = T[:3, 3]

        collision = False
        for obs_center, obs_radius in obstacles:
            if np.linalg.norm(pos - obs_center) < obs_radius + 0.05:
                collision = True
                break

        if not collision:
            positions.append(pos)

    return np.array(positions)
小结
    

  • 工作空间分析确定机器人末端执行器的可达范围，是选型和任务规划的基础
    • 

  • 数值法（网格采样、蒙特卡洛）适用于任意构型，解析法仅适用于简单结构
    • 

  • 可达工作空间和灵巧工作空间的区分对精密操作任务至关重要
    • 

  • 奇异性分析通过可操作度椭球量化工作空间质量，避免在低质量区域规划任务
    • 

  • 工作空间优化需在可达性、灵巧性和结构紧凑性之间权衡
    • 

  • 实际应用中需考虑障碍物约束，计算无碰撞工作空间
