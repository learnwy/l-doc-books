传感器类型与选型

核心概念

传感器是机器人感知外部世界和自身状态的基础器件。机器人传感器按功能可分为内部传感器（感知自身状态）和外部传感器（感知环境信息）。传感器的选型直接影响机器人的感知能力、控制精度和任务执行效果。

传感器选型需要综合考虑测量范围、精度、响应频率、环境适应性和成本等因素。不同应用场景对传感器的要求差异巨大——工业焊接机器人需要高精度位置传感器，而服务机器人更需要丰富的环境感知传感器。

位置与角度传感器
位置传感器是机器人控制系统中最基础的传感器，提供关节角度或线性位移的精确测量，构成闭环控制的反馈信号。


光电编码器是工业机器人中最常用的位置传感器，分为增量式和绝对式两种：


import numpy as np

class IncrementalEncoder:
    def __init__(self, ppr, gear_ratio=1.0):
        self.ppr = ppr
        self.gear_ratio = gear_ratio
        self.resolution = 2 * np.pi / (ppr * 4 * gear_ratio)
        self.count = 0
        self.prev_a = 0
        self.prev_b = 0

    def quadrature_decode(self, a, b):
        state = (a << 1) | b
        prev_state = (self.prev_a << 1) | self.prev_b

        transition = (prev_state << 2) | state

        forward = {0b0001, 0b0111, 0b1110, 0b1000}
        backward = {0b0010, 0b1011, 0b1101, 0b0100}

        if transition in forward:
            self.count += 1
        elif transition in backward:
            self.count -= 1

        self.prev_a = a
        self.prev_b = b

        return self.count * self.resolution

    def get_velocity(self, dt, method='differentiation'):
        if method == 'pulse_counting':
            return self.count * self.resolution / dt
        return 0.0

class AbsoluteEncoder:
    def __init__(self, bits, multi_turn_bits=0):
        self.bits = bits
        self.single_turn_resolution = 2 * np.pi / (2 ** bits)
        self.multi_turn_bits = multi_turn_bits
        self.total_range = 2 * np.pi * (2 ** multi_turn_bits)

    def read_position(self, raw_value):
        single_turn = (raw_value & ((1 << self.bits) - 1)) * self.single_turn_resolution

        if self.multi_turn_bits > 0:
            turns = (raw_value >> self.bits) & ((1 << self.multi_turn_bits) - 1)
            return turns * 2 * np.pi + single_turn

        return single_turn

    def gray_to_binary(self, gray):
        binary = gray
        mask = gray >> 1
        while mask:
            binary ^= mask
            mask >>= 1
        return binary

视觉传感器
视觉传感器为机器人提供丰富的环境信息，是实现目标识别、抓取规划和导航的核心传感器。主流的视觉传感器包括 RGB 相机、深度相机和事件相机。


import numpy as np

class DepthCamera:
    def __init__(self, width, height, fx, fy, cx, cy):
        self.width = width
        self.height = height
        self.fx = fx
        self.fy = fy
        self.cx = cx
        self.cy = cy
        self.K = np.array([
            [fx, 0,  cx],
            [0,  fy, cy],
            [0,  0,  1 ]
        ])

    def depth_to_pointcloud(self, depth_image, color_image=None):
        u, v = np.meshgrid(
            np.arange(self.width),
            np.arange(self.height)
        )

        z = depth_image.astype(np.float64) / 1000.0
        valid = z > 0

        x = (u - self.cx) * z / self.fx
        y = (v - self.cy) * z / self.fy

        points = np.stack([x[valid], y[valid], z[valid]], axis=-1)

        if color_image is not None:
            colors = color_image[valid] / 255.0
            return points, colors

        return points

    def project_3d_to_2d(self, points_3d):
        x, y, z = points_3d[:, 0], points_3d[:, 1], points_3d[:, 2]
        valid = z > 0

        u = self.fx * x[valid] / z[valid] + self.cx
        v = self.fy * y[valid] / z[valid] + self.cy

        return np.stack([u, v], axis=-1), valid

class StereoVision:
    def __init__(self, camera_left, camera_right, baseline):
        self.left = camera_left
        self.right = camera_right
        self.baseline = baseline

    def compute_disparity(self, left_image, right_image, block_size=5, num_disparities=64):
        height, width = left_image.shape[:2]
        disparity = np.zeros((height, width))
        half = block_size // 2

        for y in range(half, height - half):
            for x in range(num_disparities + half, width - half):
                left_block = left_image[
                    y-half:y+half+1, x-half:x+half+1
                ].astype(np.float64)

                min_cost = float('inf')
                best_d = 0

                for d in range(num_disparities):
                    right_block = right_image[
                        y-half:y+half+1, x-d-half:x-d+half+1
                    ].astype(np.float64)

                    cost = np.sum((left_block - right_block) ** 2)
                    if cost < min_cost:
                        min_cost = cost
                        best_d = d

                disparity[y, x] = best_d

        return disparity

    def disparity_to_depth(self, disparity):
        depth = np.zeros_like(disparity)
        valid = disparity > 0
        depth[valid] = self.baseline * self.left.fx / disparity[valid]
        return depth
惯性测量单元（IMU）
IMU 集成了加速度计和陀螺仪，提供机器人的线加速度和角速度测量。高端 IMU 还集成磁力计，形成 9 轴惯性导航系统。IMU 在移动机器人和无人机中是不可或缺的姿态估计传感器。


import numpy as np

class IMUProcessor:
    def __init__(self, dt, accel_noise=0.01, gyro_noise=0.001):
        self.dt = dt
        self.accel_noise = accel_noise
        self.gyro_noise = gyro_noise

        self.accel_bias = np.zeros(3)
        self.gyro_bias = np.zeros(3)

        self.orientation = np.eye(3)
        self.velocity = np.zeros(3)
        self.position = np.zeros(3)

    def calibrate_bias(self, static_readings, n_samples=1000):
        accel_data = np.array([r['accel'] for r in static_readings[:n_samples]])
        gyro_data = np.array([r['gyro'] for r in static_readings[:n_samples]])

        self.gyro_bias = np.mean(gyro_data, axis=0)

        mean_accel = np.mean(accel_data, axis=0)
        gravity_dir = mean_accel / np.linalg.norm(mean_accel)
        self.accel_bias = mean_accel - 9.81 * gravity_dir

    def process_reading(self, accel_raw, gyro_raw):
        accel = accel_raw - self.accel_bias
        gyro = gyro_raw - self.gyro_bias

        omega = gyro
        theta = np.linalg.norm(omega) * self.dt

        if theta > 1e-10:
            axis = omega / np.linalg.norm(omega)
            K = np.array([
                [0, -axis[2], axis[1]],
                [axis[2], 0, -axis[0]],
                [-axis[1], axis[0], 0]
            ])
            dR = (np.eye(3) + np.sin(theta) * K +
                  (1 - np.cos(theta)) * K @ K)
            self.orientation = self.orientation @ dR

        accel_world = self.orientation @ accel - np.array([0, 0, 9.81])
        self.velocity += accel_world * self.dt
        self.position += self.velocity * self.dt

        return {
            'orientation': self.orientation.copy(),
            'velocity': self.velocity.copy(),
            'position': self.position.copy(),
            'accel_world': accel_world,
            'gyro_corrected': gyro
        }

传感器选型方法论
传感器选型是一个系统工程决策过程，需要在性能、成本、集成难度之间找到最优平衡。


class SensorSelector:
    def __init__(self):
        self.requirements = {}
        self.candidates = []

    def set_requirements(self, measurement_type, range_min, range_max,
                        accuracy, frequency, environment, budget):
        self.requirements = {
            'type': measurement_type,
            'range': (range_min, range_max),
            'accuracy': accuracy,
            'frequency': frequency,
            'environment': environment,
            'budget': budget
        }

    def evaluate_candidate(self, sensor):
        score = 0
        weights = {
            'accuracy': 0.3,
            'range': 0.2,
            'frequency': 0.15,
            'reliability': 0.15,
            'cost': 0.1,
            'integration': 0.1
        }

        if sensor['accuracy'] <= self.requirements['accuracy']:
            score += weights['accuracy'] * (1 - sensor['accuracy'] / self.requirements['accuracy'])

        if (sensor['range'][0] <= self.requirements['range'][0] and
            sensor['range'][1] >= self.requirements['range'][1]):
            score += weights['range']

        if sensor['frequency'] >= self.requirements['frequency']:
            score += weights['frequency']

        if sensor['cost'] <= self.requirements['budget']:
            score += weights['cost'] * (1 - sensor['cost'] / self.requirements['budget'])

        return score

    def recommend(self, candidates):
        scored = [(s, self.evaluate_candidate(s)) for s in candidates]
        scored.sort(key=lambda x: x[1], reverse=True)
        return scored
小结
    

  • 机器人传感器分为内部传感器（位置、速度、力）和外部传感器（视觉、距离、触觉），共同构建完整的感知系统
    • 

  • 光电编码器是关节位置测量的标准方案，绝对式编码器适合需要断电保持的场景
    • 

  • 深度相机和激光雷达提供三维环境感知，是自主导航和抓取规划的基础
    • 

  • IMU 通过加速度计和陀螺仪实现姿态估计，是移动机器人的核心传感器
    • 

  • 传感器选型需综合考虑精度、频率、环境适应性、成本和集成难度
    • 

  • 多传感器融合是提升整体感知性能的必然趋势
