执行器系统

核心概念

执行器（Actuator）是将控制信号转换为机械运动的装置，是机器人动力系统的核心组件。执行器的性能直接决定了机器人的运动速度、精度、负载能力和能量效率。根据能源类型，执行器主要分为电动、液压和气动三大类。

现代机器人绝大多数采用电动执行器，尤其是永磁同步电机（PMSM）和无刷直流电机（BLDC）。液压执行器在重载工业和特种机器人中仍有广泛应用，气动执行器则在轻载快速操作场景中占有一席之地。

电机驱动与控制
电机驱动是执行器系统的核心环节，负责将控制器的指令转换为电机的电流/电压信号。现代伺服驱动器通常采用三环控制结构：电流环（最内层）→ 速度环 → 位置环（最外层）。


import numpy as np

class BLDCDriver:
    def __init__(self, kt, ke, R, L, J, B, dt):
        self.kt = kt
        self.ke = ke
        self.R = R
        self.L = L
        self.J = J
        self.B = B
        self.dt = dt

        self.current = 0.0
        self.velocity = 0.0
        self.position = 0.0

    def step(self, voltage, load_torque=0.0):
        back_emf = self.ke * self.velocity
        di = (voltage - self.R * self.current - back_emf) / self.L
        self.current += di * self.dt

        motor_torque = self.kt * self.current
        dw = (motor_torque - load_torque - self.B * self.velocity) / self.J
        self.velocity += dw * self.dt

        self.position += self.velocity * self.dt

        return {
            'current': self.current,
            'velocity': self.velocity,
            'position': self.position,
            'torque': motor_torque
        }

class ServoController:
    def __init__(self, motor, dt):
        self.motor = motor
        self.dt = dt

        self.current_pid = PIDController(kp=10, ki=1000, kd=0, dt=dt,
                                         output_limits=(-48, 48))
        self.velocity_pid = PIDController(kp=5, ki=50, kd=0.01, dt=dt,
                                          output_limits=(-20, 20))
        self.position_pid = PIDController(kp=50, ki=5, kd=2, dt=dt,
                                          output_limits=(-100, 100))

    def position_mode(self, target_position, current_state):
        vel_cmd, _ = self.position_pid.compute(
            target_position, current_state['position']
        )

        cur_cmd, _ = self.velocity_pid.compute(
            vel_cmd, current_state['velocity']
        )

        voltage, _ = self.current_pid.compute(
            cur_cmd, current_state['current']
        )

        return self.motor.step(voltage)

    def torque_mode(self, target_torque, current_state):
        target_current = target_torque / self.motor.kt

        voltage, _ = self.current_pid.compute(
            target_current, current_state['current']
        )

        return self.motor.step(voltage)
减速器与传动机构
减速器是电机与关节之间的关键传动环节，通过降低转速、提升力矩来匹配负载需求。减速器的选型直接影响关节的精度、刚度和反向驱动特性。


import numpy as np

class HarmonicDrive:
    def __init__(self, ratio, efficiency=0.85, backlash=0.0001):
        self.ratio = ratio
        self.efficiency = efficiency
        self.backlash = backlash
        self.stiffness = 1e5

    def forward(self, motor_angle, motor_torque):
        output_angle = motor_angle / self.ratio
        output_torque = motor_torque * self.ratio * self.efficiency
        return output_angle, output_torque

    def reflected_inertia(self, motor_inertia):
        return motor_inertia * self.ratio ** 2

class CycloidalDrive:
    def __init__(self, ratio, efficiency=0.90, backlash=0.0005):
        self.ratio = ratio
        self.efficiency = efficiency
        self.backlash = backlash
        self.shock_resistance = 5.0

class PlanetaryGear:
    def __init__(self, stages, ratios_per_stage, efficiency_per_stage=0.97):
        self.stages = stages
        self.ratios = ratios_per_stage
        self.total_ratio = np.prod(ratios_per_stage)
        self.total_efficiency = efficiency_per_stage ** stages

def select_reducer(requirements):
    candidates = {
        'harmonic': {
            'ratio_range': (30, 200),
            'backlash': 0.0001,
            'efficiency': 0.85,
            'weight_factor': 0.3,
            'cost_factor': 1.5
        },
        'cycloidal': {
            'ratio_range': (6, 120),
            'backlash': 0.0005,
            'efficiency': 0.90,
            'weight_factor': 0.5,
            'cost_factor': 1.2
        },
        'planetary': {
            'ratio_range': (3, 1000),
            'backlash': 0.001,
            'efficiency': 0.94,
            'weight_factor': 0.7,
            'cost_factor': 0.8
        }
    }

    results = []
    for name, spec in candidates.items():
        if spec['ratio_range'][0] <= requirements['ratio'] <= spec['ratio_range'][1]:
            score = (
                0.3 * (1 - spec['backlash'] / requirements.get('max_backlash', 0.01)) +
                0.3 * spec['efficiency'] +
                0.2 * (1 - spec['weight_factor']) +
                0.2 * (1 - spec['cost_factor'] / 2.0)
            )
            results.append((name, score))

    return sorted(results, key=lambda x: x[1], reverse=True)

串联弹性执行器（SEA）
串联弹性执行器（Series Elastic Actuator, SEA）在电机输出端串联一个弹性元件（弹簧），通过测量弹簧变形间接测量和控制输出力矩。SEA 是协作机器人和助力外骨骼的关键技术。


import numpy as np

class SeriesElasticActuator:
    def __init__(self, motor, spring_stiffness, dt):
        self.motor = motor
        self.k = spring_stiffness
        self.dt = dt

        self.motor_pos = 0.0
        self.output_pos = 0.0
        self.motor_vel = 0.0
        self.output_vel = 0.0

    def compute_output_torque(self):
        spring_deflection = self.motor_pos - self.output_pos
        return self.k * spring_deflection

    def torque_control(self, desired_torque, external_load):
        current_torque = self.compute_output_torque()

        desired_deflection = desired_torque / self.k
        desired_motor_pos = self.output_pos + desired_deflection

        motor_state = self.motor.step_to_position(desired_motor_pos)
        self.motor_pos = motor_state['position']
        self.motor_vel = motor_state['velocity']

        spring_torque = self.compute_output_torque()
        output_accel = (spring_torque - external_load) / 0.1
        self.output_vel += output_accel * self.dt
        self.output_pos += self.output_vel * self.dt

        return {
            'output_torque': spring_torque,
            'motor_position': self.motor_pos,
            'output_position': self.output_pos,
            'spring_deflection': self.motor_pos - self.output_pos
        }

class QuasiDirectDrive:
    def __init__(self, motor, low_ratio_gear, dt):
        self.motor = motor
        self.gear = low_ratio_gear
        self.dt = dt
        self.reflected_inertia = motor.J * low_ratio_gear.ratio ** 2

    def torque_control(self, desired_torque, current_velocity):
        motor_torque = desired_torque / (self.gear.ratio * self.gear.efficiency)
        motor_current = motor_torque / self.motor.kt
        return motor_current

    def backdrivability_check(self, external_torque):
        friction_torque = self.motor.B * 0 + 0.1
        reflected_friction = friction_torque * self.gear.ratio
        return abs(external_torque) > reflected_friction
执行器集成与选型
执行器的选型和集成需要从系统层面考虑电机、减速器、驱动器、传感器的匹配关系，确保整个驱动链路的性能满足任务需求。


class ActuatorModule:
    def __init__(self, config):
        self.motor_torque = config['motor_peak_torque']
        self.motor_speed = config['motor_max_speed']
        self.gear_ratio = config['gear_ratio']
        self.gear_efficiency = config['gear_efficiency']

        self.output_torque = self.motor_torque * self.gear_ratio * self.gear_efficiency
        self.output_speed = self.motor_speed / self.gear_ratio

    def check_operating_point(self, required_torque, required_speed):
        motor_torque_needed = required_torque / (self.gear_ratio * self.gear_efficiency)
        motor_speed_needed = required_speed * self.gear_ratio

        torque_margin = 1 - motor_torque_needed / self.motor_torque
        speed_margin = 1 - motor_speed_needed / self.motor_speed

        return {
            'feasible': torque_margin > 0 and speed_margin > 0,
            'torque_margin': torque_margin,
            'speed_margin': speed_margin,
            'power': required_torque * required_speed
        }
小结
    

  • 电动执行器是现代机器人的主流选择，BLDC/PMSM 电机配合三环伺服控制实现精确运动
    • 

  • 减速器（谐波、摆线、行星）通过降速增矩匹配负载，但引入回差和效率损失
    • 

  • 串联弹性执行器通过弹性元件实现力矩感知和柔顺控制，适合人机交互场景
    • 

  • 直驱和准直驱方案消除减速器影响，提升反向驱动特性，但需要大力矩电机
    • 

  • 执行器选型需要系统匹配电机、减速器和驱动器，确保工作点在安全范围内
    • 

  • 新型执行器（人工肌肉、形状记忆合金）为软体机器人提供了新的驱动方案
    •
