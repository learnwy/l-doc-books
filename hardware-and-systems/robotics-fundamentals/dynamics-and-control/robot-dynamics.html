机器人动力学

核心概念

机器人动力学研究的是力/力矩与运动之间的关系。与运动学只关注几何和运动不同，动力学需要考虑质量、惯量、重力、摩擦等物理因素。动力学模型是机器人控制系统设计的基础，直接影响控制精度和运动性能。

机器人动力学的两个核心问题是：正动力学（给定关节力矩，求运动轨迹）和逆动力学（给定运动轨迹，求所需关节力矩）。正动力学用于仿真，逆动力学用于控制。

拉格朗日动力学
拉格朗日方法基于能量原理建立动力学方程。拉格朗日量 LLL 定义为系统动能 TTT 与势能 VVV 之差：L=T−VL = T - VL=T−V。通过 Euler-Lagrange 方程可直接得到运动方程：


ddt∂L∂q˙i−∂L∂qi=τi\frac{d}{dt}\frac{\partial L}{\partial \dot{q}_i} - \frac{\partial L}{\partial q_i} = \tau_idtd​∂q˙​i​∂L​−∂qi​∂L​=τi​
展开后得到标准形式的动力学方程：


M(q)q¨+C(q,q˙)q˙+g(q)=τM(q)\ddot{q} + C(q, \dot{q})\dot{q} + g(q) = \tauM(q)q¨​+C(q,q˙​)q˙​+g(q)=τ
import numpy as np
from sympy import symbols, cos, sin, Matrix, diff, simplify

def lagrangian_dynamics_2dof():
    q1, q2, dq1, dq2, ddq1, ddq2 = symbols('q1 q2 dq1 dq2 ddq1 ddq2')
    m1, m2, L1, L2, g_val = symbols('m1 m2 L1 L2 g', positive=True)
    Lc1, Lc2, I1, I2 = symbols('Lc1 Lc2 I1 I2', positive=True)

    x1 = Lc1 * cos(q1)
    y1 = Lc1 * sin(q1)
    dx1 = -Lc1 * sin(q1) * dq1
    dy1 = Lc1 * cos(q1) * dq1

    x2 = L1 * cos(q1) + Lc2 * cos(q1 + q2)
    y2 = L1 * sin(q1) + Lc2 * sin(q1 + q2)
    dx2 = -L1 * sin(q1) * dq1 - Lc2 * sin(q1 + q2) * (dq1 + dq2)
    dy2 = L1 * cos(q1) * dq1 + Lc2 * cos(q1 + q2) * (dq1 + dq2)

    T = (0.5 * m1 * (dx1**2 + dy1**2) + 0.5 * I1 * dq1**2 +
         0.5 * m2 * (dx2**2 + dy2**2) + 0.5 * I2 * (dq1 + dq2)**2)

    V = m1 * g_val * y1 + m2 * g_val * y2

    L_func = T - V

    tau1 = simplify(
        diff(diff(L_func, dq1), q1) * dq1 +
        diff(diff(L_func, dq1), q2) * dq2 +
        diff(diff(L_func, dq1), dq1) * ddq1 +
        diff(diff(L_func, dq1), dq2) * ddq2 -
        diff(L_func, q1)
    )

    return tau1

def numerical_dynamics_2dof(q, dq, params):
    q1, q2 = q
    dq1, dq2 = dq
    m1, m2, L1, L2, Lc1, Lc2, I1, I2, g = params

    M11 = (m1 * Lc1**2 + I1 + m2 * (L1**2 + Lc2**2 + 2*L1*Lc2*np.cos(q2)) + I2)
    M12 = m2 * (Lc2**2 + L1 * Lc2 * np.cos(q2)) + I2
    M21 = M12
    M22 = m2 * Lc2**2 + I2
    M = np.array([[M11, M12], [M21, M22]])

    h = m2 * L1 * Lc2 * np.sin(q2)
    C = np.array([
        [-h * dq2, -h * (dq1 + dq2)],
        [h * dq1,  0]
    ])

    g_vec = np.array([
        (m1 * Lc1 + m2 * L1) * g * np.cos(q1) + m2 * Lc2 * g * np.cos(q1 + q2),
        m2 * Lc2 * g * np.cos(q1 + q2)
    ])

    return M, C, g_vec
Newton-Euler 递推算法
Newton-Euler 方法通过两次递推（向外速度/加速度传播 + 向内力/力矩传播）高效计算逆动力学，计算复杂度为 O(n)O(n)O(n)，适合实时控制应用。


import numpy as np

class NewtonEulerSolver:
    def __init__(self, links):
        self.links = links
        self.n = len(links)
        self.gravity = np.array([0, 0, -9.81])

    def inverse_dynamics(self, q, dq, ddq):
        omega = [np.zeros(3)] * (self.n + 1)
        alpha = [np.zeros(3)] * (self.n + 1)
        a_c = [np.zeros(3)] * (self.n + 1)
        a_e = [np.zeros(3)] * (self.n + 1)
        a_e[0] = -self.gravity

        for i in range(self.n):
            link = self.links[i]
            R = link.rotation_matrix(q[i])
            z = link.joint_axis

            omega[i+1] = R.T @ omega[i] + dq[i] * z
            alpha[i+1] = R.T @ alpha[i] + (
                np.cross(R.T @ omega[i], dq[i] * z) + ddq[i] * z
            )
            a_e[i+1] = R.T @ (
                a_e[i] + np.cross(alpha[i], link.origin) +
                np.cross(omega[i], np.cross(omega[i], link.origin))
            )
            a_c[i+1] = (
                a_e[i+1] + np.cross(alpha[i+1], link.com) +
                np.cross(omega[i+1], np.cross(omega[i+1], link.com))
            )

        f = [np.zeros(3)] * (self.n + 2)
        n_torque = [np.zeros(3)] * (self.n + 2)
        tau = np.zeros(self.n)

        for i in range(self.n - 1, -1, -1):
            link = self.links[i]
            R_next = self.links[i+1].rotation_matrix(q[i+1]) if i < self.n-1 else np.eye(3)

            f[i+1] = (R_next @ f[i+2] + link.mass * a_c[i+1])
            n_torque[i+1] = (
                R_next @ n_torque[i+2] +
                np.cross(link.com, link.mass * a_c[i+1]) +
                np.cross(link.origin + R_next @ self.links[i+1].origin if i < self.n-1
                         else link.origin, R_next @ f[i+2]) +
                link.inertia @ alpha[i+1] +
                np.cross(omega[i+1], link.inertia @ omega[i+1])
            )

            tau[i] = n_torque[i+1] @ link.joint_axis

        return tau

动力学参数辨识
动力学模型的精度取决于参数的准确性。参数辨识通过实验数据回归出真实的动力学参数（质量、惯量、摩擦系数等）。


动力学方程可以改写为关于惯性参数的线性形式：


τ=Y(q,q˙,q¨)⋅π\tau = Y(q, \dot{q}, \ddot{q}) \cdot \piτ=Y(q,q˙​,q¨​)⋅π
其中 YYY 是回归矩阵，π\piπ 是待辨识的参数向量。


import numpy as np
from scipy.optimize import least_squares

class DynamicsIdentification:
    def __init__(self, robot):
        self.robot = robot

    def collect_trajectory_data(self, trajectory, dt=0.001):
        data = []
        for t, q, dq, ddq, tau_measured in trajectory:
            Y = self.robot.regressor_matrix(q, dq, ddq)
            data.append({
                'Y': Y,
                'tau': tau_measured,
                'q': q,
                'dq': dq,
                'ddq': ddq
            })
        return data

    def identify_parameters(self, data):
        Y_stacked = np.vstack([d['Y'] for d in data])
        tau_stacked = np.concatenate([d['tau'] for d in data])

        pi_hat, residuals, rank, sv = np.linalg.lstsq(
            Y_stacked, tau_stacked, rcond=None
        )

        tau_predicted = Y_stacked @ pi_hat
        error = tau_stacked - tau_predicted
        rmse = np.sqrt(np.mean(error**2))

        return {
            'parameters': pi_hat,
            'rmse': rmse,
            'relative_error': rmse / np.std(tau_stacked),
            'condition_number': sv[0] / sv[-1] if sv[-1] > 0 else np.inf
        }

    def exciting_trajectory(self, n_harmonics=5, duration=10.0, dt=0.001):
        t = np.arange(0, duration, dt)
        omega_f = 2 * np.pi / duration

        n_joints = self.robot.n_joints
        q = np.zeros((len(t), n_joints))

        for j in range(n_joints):
            for k in range(1, n_harmonics + 1):
                a_k = np.random.uniform(-0.5, 0.5)
                b_k = np.random.uniform(-0.5, 0.5)
                q[:, j] += (a_k * np.sin(k * omega_f * t) +
                           b_k * np.cos(k * omega_f * t)) / k**2

        return t, q
摩擦模型与补偿
摩擦是影响机器人动力学精度的重要因素。准确的摩擦建模和补偿对于高精度运动控制至关重要。


import numpy as np

class FrictionModel:
    def __init__(self, model_type='lugre'):
        self.model_type = model_type

    def coulomb_viscous(self, dq, fc, fv):
        return fc * np.sign(dq) + fv * dq

    def stribeck(self, dq, fc, fs, fv, vs):
        return (fc + (fs - fc) * np.exp(-(dq/vs)**2)) * np.sign(dq) + fv * dq

    def lugre(self, dq, z, params):
        sigma_0, sigma_1, sigma_2 = params['sigma_0'], params['sigma_1'], params['sigma_2']
        fc, fs, vs = params['fc'], params['fs'], params['vs']

        g_dq = fc + (fs - fc) * np.exp(-(dq/vs)**2)

        dz = dq - sigma_0 * np.abs(dq) / g_dq * z

        friction = sigma_0 * z + sigma_1 * dz + sigma_2 * dq

        return friction, dz

class FrictionCompensator:
    def __init__(self, n_joints):
        self.n_joints = n_joints
        self.params = [
            {'fc': 0.5, 'fs': 0.8, 'fv': 0.1, 'vs': 0.01}
            for _ in range(n_joints)
        ]

    def compensate(self, dq):
        tau_friction = np.zeros(self.n_joints)
        for i in range(self.n_joints):
            p = self.params[i]
            tau_friction[i] = (
                (p['fc'] + (p['fs'] - p['fc']) *
                 np.exp(-(dq[i]/p['vs'])**2)) * np.sign(dq[i]) +
                p['fv'] * dq[i]
            )
        return tau_friction

小结
    

  • 机器人动力学建立力/力矩与运动之间的定量关系，是控制器设计的数学基础
    • 

  • 拉格朗日法基于能量守恒，适合推导闭式动力学方程和参数辨识
    • 

  • Newton-Euler 递推法计算复杂度为 O(n)O(n)O(n)，适合实时逆动力学计算

  • 动力学参数辨识通过激励轨迹实验数据回归真实参数，需注意回归矩阵的条件数

    •
