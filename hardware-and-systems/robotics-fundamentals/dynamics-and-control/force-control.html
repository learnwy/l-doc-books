力控制与柔顺控制

核心概念

力控制（Force Control）是机器人与环境交互时的核心控制模式。与纯位置控制不同，力控制允许机器人在接触环境时调节施加的力和力矩，实现安全、精确的操作。典型应用包括装配、打磨、抛光、人机协作等需要控制接触力的场景。

力控制的核心挑战在于：位置控制和力控制在同一方向上不能同时独立实现。这一约束由环境的物理特性决定——当机器人与刚性环境接触时，微小的位置偏差会导致巨大的力变化。

阻抗控制
阻抗控制（Impedance Control）不直接控制力或位置，而是控制末端执行器的机械阻抗——即力与运动之间的动态关系。其目标是使机器人表现出期望的弹簧-阻尼-质量系统特性。


期望的阻抗关系为：


Md(x¨−x¨d)+Bd(x˙−x˙d)+Kd(x−xd)=FextM_d(\ddot{x} - \ddot{x}_d) + B_d(\dot{x} - \dot{x}_d) + K_d(x - x_d) = F_{ext}Md​(x¨−x¨d​)+Bd​(x˙−x˙d​)+Kd​(x−xd​)=Fext​
其中 MdM_dMd​、BdB_dBd​、KdK_dKd​ 分别是期望的惯量、阻尼和刚度矩阵。


import numpy as np

class ImpedanceController:
    def __init__(self, robot, Md, Bd, Kd):
        self.robot = robot
        self.Md = np.diag(Md) if np.ndim(Md) == 1 else Md
        self.Bd = np.diag(Bd) if np.ndim(Bd) == 1 else Bd
        self.Kd = np.diag(Kd) if np.ndim(Kd) == 1 else Kd

    def compute(self, x_desired, dx_desired, ddx_desired,
                x_actual, dx_actual, f_external, q, dq):
        e = x_desired - x_actual
        de = dx_desired - dx_actual

        Md_inv = np.linalg.inv(self.Md)
        ddx_cmd = ddx_desired + Md_inv @ (
            self.Bd @ de + self.Kd @ e - f_external
        )

        J = self.robot.jacobian(q)
        dJ = self.robot.jacobian_dot(q, dq)

        ddq_cmd = np.linalg.pinv(J) @ (ddx_cmd - dJ @ dq)

        M, C, g = self.robot.dynamics(q, dq)
        tau = M @ ddq_cmd + C @ dq + g

        return tau

class VariableImpedance:
    def __init__(self, Kd_min, Kd_max, Bd_min, Bd_max):
        self.Kd_min = np.array(Kd_min)
        self.Kd_max = np.array(Kd_max)
        self.Bd_min = np.array(Bd_min)
        self.Bd_max = np.array(Bd_max)

    def adapt(self, f_error, position_error):
        f_norm = np.linalg.norm(f_error)
        p_norm = np.linalg.norm(position_error)

        alpha = np.clip(f_norm / 10.0, 0, 1)
        Kd = (1 - alpha) * self.Kd_max + alpha * self.Kd_min

        beta = np.clip(p_norm / 0.05, 0, 1)
        Bd = (1 - beta) * self.Bd_min + beta * self.Bd_max

        return np.diag(Kd), np.diag(Bd)

导纳控制
导纳控制（Admittance Control）与阻抗控制互补。阻抗控制接受运动输入输出力响应，导纳控制接受力输入输出运动响应。导纳控制更适合配备力/力矩传感器的位置控制型机器人。


import numpy as np

class AdmittanceController:
    def __init__(self, Md, Bd, Kd, dt):
        self.Md = np.diag(Md)
        self.Bd = np.diag(Bd)
        self.Kd = np.diag(Kd)
        self.dt = dt

        self.x_offset = np.zeros(6)
        self.dx_offset = np.zeros(6)

    def compute(self, x_desired, f_measured, f_desired=None):
        if f_desired is None:
            f_desired = np.zeros(6)

        f_error = f_measured - f_desired

        Md_inv = np.linalg.inv(self.Md)
        ddx_offset = Md_inv @ (
            f_error - self.Bd @ self.dx_offset - self.Kd @ self.x_offset
        )

        self.dx_offset += ddx_offset * self.dt
        self.x_offset += self.dx_offset * self.dt

        max_offset = 0.1
        self.x_offset = np.clip(self.x_offset, -max_offset, max_offset)

        x_modified = x_desired + self.x_offset

        return x_modified, self.x_offset

    def reset(self):
        self.x_offset = np.zeros(6)
        self.dx_offset = np.zeros(6)

class ForceTrackingAdmittance:
    def __init__(self, Md, Bd, dt):
        self.Md = np.diag(Md)
        self.Bd = np.diag(Bd)
        self.dt = dt
        self.dx = np.zeros(6)

    def compute(self, f_measured, f_desired):
        f_error = f_desired - f_measured

        Md_inv = np.linalg.inv(self.Md)
        ddx = Md_inv @ (f_error - self.Bd @ self.dx)

        self.dx += ddx * self.dt
        dx_cmd = self.dx.copy()

        return dx_cmd
混合力位控制
混合力位控制（Hybrid Force/Position Control）由 Raibert 和 Craig 于 1981 年提出，其核心思想是在任务空间中将自由度分为力控方向和位控方向，分别施加不同的控制策略。


import numpy as np

class HybridForcePositionController:
    def __init__(self, robot, kp_pos, kd_pos, kp_force, ki_force, dt):
        self.robot = robot
        self.Kp_pos = np.diag(kp_pos)
        self.Kd_pos = np.diag(kd_pos)
        self.Kp_force = np.diag(kp_force)
        self.Ki_force = np.diag(ki_force)
        self.dt = dt
        self.force_integral = np.zeros(6)

    def compute(self, selection_matrix, x_desired, dx_desired,
                f_desired, x_actual, dx_actual, f_actual, q, dq):
        S = np.diag(selection_matrix)
        S_bar = np.eye(6) - S

        pos_error = x_desired - x_actual
        vel_error = dx_desired - dx_actual
        ddx_position = self.Kp_pos @ pos_error + self.Kd_pos @ vel_error

        force_error = f_desired - f_actual
        self.force_integral += force_error * self.dt
        ddx_force = self.Kp_force @ force_error + self.Ki_force @ self.force_integral

        ddx_cmd = S_bar @ ddx_position + S @ ddx_force

        J = self.robot.jacobian(q)
        J_inv = np.linalg.pinv(J)
        dJ = self.robot.jacobian_dot(q, dq)

        ddq_cmd = J_inv @ (ddx_cmd - dJ @ dq)

        M, C, g = self.robot.dynamics(q, dq)
        tau = M @ ddq_cmd + C @ dq + g

        return tau

力传感器信号处理
力/力矩传感器的信号质量直接影响力控制性能。原始信号中通常包含传感器噪声、动态耦合效应和重力补偿误差，需要精心处理。


import numpy as np
from collections import deque

class ForceSensorProcessor:
    def __init__(self, dt, cutoff_freq=20.0, buffer_size=100):
        self.dt = dt
        self.alpha = self._compute_alpha(cutoff_freq, dt)
        self.filtered = np.zeros(6)
        self.buffer = deque(maxlen=buffer_size)
        self.gravity_offset = np.zeros(6)
        self.tool_mass = 0
        self.tool_com = np.zeros(3)

    def _compute_alpha(self, fc, dt):
        tau = 1.0 / (2 * np.pi * fc)
        return dt / (tau + dt)

    def calibrate_gravity(self, sensor_reading, orientation_matrix):
        g = np.array([0, 0, -9.81])
        f_gravity = self.tool_mass * orientation_matrix.T @ g

        moment_gravity = np.cross(
            orientation_matrix.T @ self.tool_com,
            f_gravity
        )

        self.gravity_offset = np.concatenate([f_gravity, moment_gravity])

    def process(self, raw_reading, orientation_matrix):
        self.calibrate_gravity(raw_reading, orientation_matrix)
        compensated = raw_reading - self.gravity_offset

        self.filtered = self.alpha * compensated + (1 - self.alpha) * self.filtered

        self.buffer.append(self.filtered.copy())

        return self.filtered.copy()

    def detect_contact(self, threshold=2.0):
        force_magnitude = np.linalg.norm(self.filtered[:3])
        return force_magnitude > threshold

    def get_statistics(self):
        if len(self.buffer) < 2:
            return None
        data = np.array(self.buffer)
        return {
            'mean': np.mean(data, axis=0),
            'std': np.std(data, axis=0),
            'max': np.max(np.abs(data), axis=0)
        }
小结
    

  • 力控制使机器人能够安全精确地与环境交互，是装配和协作任务的关键技术
    • 

  • 阻抗控制通过调节机械阻抗参数实现柔顺行为，适合未知环境交互
    • 

  • 导纳控制接受力输入输出运动响应，适合位置控制型机器人
    • 

  • 混合力位控制通过选择矩阵分方向实施力和位置控制，适合约束面操作
    • 

  • 力传感器信号处理包括重力补偿、低通滤波和接触检测，是力控制的基础环节
    • 

  • 变阻抗控制根据任务状态动态调节柔顺参数，是力控制的前沿方向
    •
