PID 控制器调参

核心概念

PID 控制器（比例-积分-微分控制器）是机器人关节级控制中最广泛使用的控制算法。其核心思想是通过对误差信号的比例（P）、积分（I）和微分（D）三种运算的线性组合，生成控制输出以消除跟踪误差。

PID 控制器的数学表达式为：

其中 e(t)=qd(t)−q(t)e(t) = q_d(t) - q(t)e(t)=qd​(t)−q(t) 是期望位置与实际位置之间的误差。KpK_pKp​、KiK_iKi​、KdK_dKd​ 分别是比例、积分、微分增益。

离散化实现
实际机器人控制系统运行在数字计算机上，PID 控制器需要离散化实现。常用的离散化方法包括后向差分法、梯形积分法和双线性变换法。


import numpy as np

class PIDController:
    def __init__(self, kp, ki, kd, dt, output_limits=(-100, 100)):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.dt = dt
        self.output_limits = output_limits

        self.integral = 0.0
        self.prev_error = 0.0
        self.prev_derivative = 0.0
        self.anti_windup_limit = output_limits[1] * 0.8

    def compute(self, setpoint, measurement):
        error = setpoint - measurement

        p_term = self.kp * error

        self.integral += error * self.dt
        self.integral = np.clip(
            self.integral,
            -self.anti_windup_limit / self.ki if self.ki > 0 else -1e6,
            self.anti_windup_limit / self.ki if self.ki > 0 else 1e6
        )
        i_term = self.ki * self.integral

        alpha = 0.1
        raw_derivative = (error - self.prev_error) / self.dt
        filtered_derivative = alpha * raw_derivative + (1 - alpha) * self.prev_derivative
        d_term = self.kd * filtered_derivative
        self.prev_derivative = filtered_derivative

        self.prev_error = error

        output = p_term + i_term + d_term
        output = np.clip(output, *self.output_limits)

        return output, {'p': p_term, 'i': i_term, 'd': d_term, 'error': error}

class JointPIDController:
    def __init__(self, n_joints, dt):
        self.controllers = []
        self.n_joints = n_joints
        self.dt = dt

        for i in range(n_joints):
            self.controllers.append(
                PIDController(kp=100, ki=10, kd=5, dt=dt)
            )

    def compute(self, q_desired, q_actual):
        tau = np.zeros(self.n_joints)
        diagnostics = []

        for i in range(self.n_joints):
            tau[i], diag = self.controllers[i].compute(
                q_desired[i], q_actual[i]
            )
            diagnostics.append(diag)

        return tau, diagnostics
Ziegler-Nichols 调参法
Ziegler-Nichols 方法是最经典的 PID 参数整定方法，分为阶跃响应法和临界振荡法两种。临界振荡法（Ultimate Gain Method）在机器人控制中更为常用。


临界振荡法的步骤：先将 Ki=0K_i = 0Ki​=0、Kd=0K_d = 0Kd​=0，逐步增大 KpK_pKp​ 直到系统产生等幅振荡。记录此时的临界增益 KuK_uKu​ 和临界周期 TuT_uTu​。


import numpy as np

class ZieglerNicholsTuner:
    def __init__(self, plant_model, dt=0.001):
        self.plant = plant_model
        self.dt = dt

    def find_ultimate_gain(self, kp_range=(1, 10000), tol=0.01):
        kp_low, kp_high = kp_range

        while kp_high - kp_low > tol:
            kp_mid = (kp_low + kp_high) / 2
            is_oscillating, period = self._test_oscillation(kp_mid)

            if is_oscillating:
                kp_high = kp_mid
                ku = kp_mid
                tu = period
            else:
                kp_low = kp_mid

        return ku, tu

    def _test_oscillation(self, kp, duration=5.0):
        controller = PIDController(kp=kp, ki=0, kd=0, dt=self.dt)
        state = self.plant.reset()

        positions = []
        setpoint = 1.0

        for _ in range(int(duration / self.dt)):
            u, _ = controller.compute(setpoint, state['position'])
            state = self.plant.step(u)
            positions.append(state['position'])

        positions = np.array(positions)
        peaks = self._find_peaks(positions)

        if len(peaks) >= 4:
            amplitudes = positions[peaks]
            ratio = amplitudes[-1] / amplitudes[-3] if amplitudes[-3] != 0 else 0

            if 0.95 < ratio < 1.05:
                period = np.mean(np.diff(peaks)) * self.dt
                return True, period

        return False, None

    def compute_pid_params(self, ku, tu, controller_type='PID'):
        zn_table = {
            'P':   {'kp': 0.50 * ku, 'ki': 0,                'kd': 0},
            'PI':  {'kp': 0.45 * ku, 'ki': 0.45*ku/(0.83*tu), 'kd': 0},
            'PID': {'kp': 0.60 * ku, 'ki': 0.60*ku/(0.50*tu), 'kd': 0.60*ku*0.125*tu},
        }

        params = zn_table[controller_type]
        return params

    def _find_peaks(self, data):
        peaks = []
        for i in range(1, len(data) - 1):
            if data[i] > data[i-1] and data[i] > data[i+1]:
                peaks.append(i)
        return np.array(peaks)

自适应 PID 与增益调度
固定参数 PID 在机器人控制中面临的主要挑战是系统参数随构型变化。机器人的惯量矩阵 M(q)M(q)M(q) 随关节角度显著变化，导致固定 PID 参数在不同构型下表现不一致。


import numpy as np

class GainSchedulingPID:
    def __init__(self, n_joints, dt):
        self.n_joints = n_joints
        self.dt = dt
        self.controllers = [
            PIDController(kp=100, ki=10, kd=5, dt=dt)
            for _ in range(n_joints)
        ]

        self.gain_tables = {}

    def update_gains(self, q, M_diagonal):
        for i in range(self.n_joints):
            inertia = M_diagonal[i]

            omega_n = 50.0
            zeta = 0.8

            kp = inertia * omega_n**2
            kd = inertia * 2 * zeta * omega_n
            ki = kp * 0.1

            self.controllers[i].kp = kp
            self.controllers[i].ki = ki
            self.controllers[i].kd = kd

    def compute(self, q_desired, q_actual, M_diagonal):
        self.update_gains(q_actual, M_diagonal)

        tau = np.zeros(self.n_joints)
        for i in range(self.n_joints):
            tau[i], _ = self.controllers[i].compute(
                q_desired[i], q_actual[i]
            )
        return tau

class MITRuleAdaptivePID:
    def __init__(self, kp_init, ki_init, kd_init, gamma, dt):
        self.kp = kp_init
        self.ki = ki_init
        self.kd = kd_init
        self.gamma_p = gamma[0]
        self.gamma_i = gamma[1]
        self.gamma_d = gamma[2]
        self.dt = dt
        self.integral = 0.0
        self.prev_error = 0.0

    def compute(self, setpoint, measurement, model_output):
        error = setpoint - measurement
        model_error = model_output - measurement

        self.kp += self.gamma_p * model_error * error * self.dt
        self.ki += self.gamma_i * model_error * self.integral * self.dt
        derivative = (error - self.prev_error) / self.dt
        self.kd += self.gamma_d * model_error * derivative * self.dt

        self.kp = max(0, self.kp)
        self.ki = max(0, self.ki)
        self.kd = max(0, self.kd)

        self.integral += error * self.dt
        p_term = self.kp * error
        i_term = self.ki * self.integral
        d_term = self.kd * derivative

        self.prev_error = error
        return p_term + i_term + d_term
计算力矩控制与 PID 结合
单纯的关节级 PID 控制无法补偿机器人的非线性动力学耦合。计算力矩控制（Computed Torque Control）将动力学模型与 PID 结合，实现了关节间解耦和线性化控制。


import numpy as np

class ComputedTorqueController:
    def __init__(self, robot, kp, kd, dt):
        self.robot = robot
        self.kp = np.diag(kp)
        self.kd = np.diag(kd)
        self.dt = dt

    def compute(self, q_desired, dq_desired, ddq_desired, q_actual, dq_actual):
        M, C, g = self.robot.dynamics(q_actual, dq_actual)

        e = q_desired - q_actual
        de = dq_desired - dq_actual

        ddq_cmd = ddq_desired + self.kd @ de + self.kp @ e

        tau = M @ ddq_cmd + C @ dq_actual + g

        return tau, {'error': e, 'derror': de, 'ddq_cmd': ddq_cmd}

class RobustComputedTorque:
    def __init__(self, robot, kp, kd, uncertainty_bound):
        self.robot = robot
        self.kp = np.diag(kp)
        self.kd = np.diag(kd)
        self.rho = uncertainty_bound

    def compute(self, q_d, dq_d, ddq_d, q, dq):
        M_hat, C_hat, g_hat = self.robot.dynamics_nominal(q, dq)

        e = q_d - q
        de = dq_d - dq
        s = de + self.kd @ e

        ddq_cmd = ddq_d + self.kd @ de + self.kp @ e
        tau_nominal = M_hat @ ddq_cmd + C_hat @ dq + g_hat

        s_norm = np.linalg.norm(s)
        if s_norm > 0.001:
            tau_robust = self.rho * s / s_norm
        else:
            tau_robust = self.rho * s / 0.001

        return tau_nominal + tau_robust
小结
    

  • PID 控制器通过比例、积分、微分三项运算消除跟踪误差，是机器人关节控制的基础
    • 

  • 离散化实现需要抗积分饱和和微分滤波等工程处理，避免数值问题
    • 

  • Ziegler-Nichols 法提供系统化的参数整定起点，但通常需要进一步微调
    • 

  • 增益调度根据惯量变化动态调整 PID 参数，解决构型依赖问题
    • 

  • 计算力矩控制将动力学模型与 PID 结合，实现关节间动态解耦
    • 

  • 鲁棒计算力矩控制通过滑模项补偿模型不确定性，提升跟踪精度
