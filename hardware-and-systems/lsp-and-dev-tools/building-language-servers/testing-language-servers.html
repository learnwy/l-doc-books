测试语言服务器

协议级测试

协议级测试直接发送和接收 LSP/DAP 消息，验证服务器对协议的正确实现。这是最接近真实使用场景的测试方式。

import { ProtocolConnection, createProtocolConnection } from 'vscode-languageserver-protocol';
import { StreamMessageReader, StreamMessageWriter } from 'vscode-jsonrpc/node';
import { spawn } from 'node:child_process';

async function createTestConnection(): Promise<ProtocolConnection> {
  const serverProcess = spawn('node', ['./server/out/server.js', '--stdio']);

  const connection = createProtocolConnection(
    new StreamMessageReader(serverProcess.stdout!),
    new StreamMessageWriter(serverProcess.stdin!),
  );

  connection.listen();
  return connection;
}

async function testInitialize(connection: ProtocolConnection): Promise<void> {
  const result = await connection.sendRequest('initialize', {
    processId: process.pid,
    rootUri: 'file:///test-workspace',
    capabilities: {},
    workspaceFolders: [{ uri: 'file:///test-workspace', name: 'test' }],
  });

  assert(result.capabilities.completionProvider);
  assert(result.capabilities.definitionProvider);

  await connection.sendNotification('initialized');
}

Mock 客户端
Mock 客户端模拟编辑器行为，提供更高层的测试抽象。核心接口封装了 openDocument、editDocument、getCompletions、getDefinition 和 waitForDiagnostics 等方法。


class MockLanguageClient {
  private connection: ProtocolConnection;
  private diagnostics: Map<string, Diagnostic[]> = new Map();

  async openDocument(uri: string, text: string): Promise<void> {
    await this.connection.sendNotification('textDocument/didOpen', {
      textDocument: { uri, languageId: 'typescript', version: 1, text },
    });
  }

  async getCompletions(uri: string, position: Position): Promise<CompletionItem[]> {
    const result = await this.connection.sendRequest('textDocument/completion', {
      textDocument: { uri },
      position,
    });
    return Array.isArray(result) ? result : result.items;
  }

  async waitForDiagnostics(uri: string, timeout = 5000): Promise<Diagnostic[]> {
    const start = Date.now();
    while (Date.now() - start < timeout) {
      const diags = this.diagnostics.get(uri);
      if (diags?.length) return diags;
      await new Promise((r) => setTimeout(r, 100));
    }
    return this.diagnostics.get(uri) ?? [];
  }
}
使用 Mock 客户端编写测试：


test('completion provides local variables', async () => {
  await client.openDocument('file:///test.ts', 'const greeting = "hello";\nconsole.log(gre)');
  const items = await client.getCompletions('file:///test.ts', { line: 1, character: 15 });
  expect(items.map((i) => i.label)).toContain('greeting');
});

test('diagnostics for type errors', async () => {
  await client.openDocument('file:///error.ts', 'const x: number = "string";');
  const diagnostics = await client.waitForDiagnostics('file:///error.ts');
  expect(diagnostics).toHaveLength(1);
  expect(diagnostics[0].severity).toBe(DiagnosticSeverity.Error);
});


快照测试
快照测试将语言服务器的输出与预先保存的"快照"比对，适合验证大量输出不回退。


snapshotTest(
  'hover-function',
  'function add(a: number, b: number): number { return a + b; }\nadd(1, 2);',
  async (client, uri) => client.getHover(uri, { line: 1, character: 0 }),
);

snapshotTest(
  'document-symbols',
  'interface User { name: string; }\nfunction getUser(): User { return { name: "Alice" }; }',
  async (client, uri) => client.getDocumentSymbols(uri),
);

一致性测试
一致性测试验证语言服务器是否正确实现了 LSP 规范的各项要求。


describe('LSP Conformance', () => {
  test('responds with ServerNotInitialized before initialization', async () => {
    const rawConnection = await createRawConnection();
    try {
      await rawConnection.sendRequest('textDocument/completion', {
        textDocument: { uri: 'file:///test.ts' },
        position: { line: 0, character: 0 },
      });
      fail('Expected error');
    } catch (e) {
      expect(e.code).toBe(-32002);
    }
  });

  test('handles $/cancelRequest correctly', async () => {
    await client.openDocument('file:///big.ts', generateLargeFile(10000));

    const completionPromise = client.getCompletions('file:///big.ts', { line: 5000, character: 0 });

    await client.sendCancelRequest(completionPromise.requestId);

    try {
      await completionPromise;
    } catch (e) {
      expect(e.code).toBe(-32800);
    }
  });

  test('shutdown then exit sequence', async () => {
    await client.sendRequest('shutdown');
    await client.sendNotification('exit');
    expect(await serverProcess.exitCode).toBe(0);
  });

  test('exit without shutdown returns non-zero', async () => {
    await client.sendNotification('exit');
    expect(await serverProcess.exitCode).toBe(1);
  });
});

性能基准测试
性能测试确保语言服务器在各种规模的项目中保持可接受的响应时间。


async function benchmark(name: string, fn: () => Promise<void>, samples = 100): Promise<BenchmarkResult> {
  for (let i = 0; i < 10; i++) await fn();

  const times: number[] = [];
  for (let i = 0; i < samples; i++) {
    const start = performance.now();
    await fn();
    times.push(performance.now() - start);
  }
  times.sort((a, b) => a - b);

  return {
    name,
    samples,
    mean: times.reduce((a, b) => a + b) / times.length,
    p50: times[Math.floor(times.length * 0.5)],
    p95: times[Math.floor(times.length * 0.95)],
    p99: times[Math.floor(times.length * 0.99)],
  };
}

await benchmark('completion-latency', () => client.getCompletions('file:///main.ts', { line: 50, character: 10 }));

提示： 性能测试应在 CI 中定期运行，并与基线值对比。建议在不同规模的测试项目上运行（100 文件、1000 文件、10000 文件），确保服务器在各种规模下性能不退化。






小结
    

  • 协议级测试直接发送 LSP 消息验证服务器行为，覆盖正常和异常流程。
    • 

  • Mock 客户端提供 openDocument/editDocument/getCompletions 等高层抽象，简化测试编写。
    • 

  • 快照测试自动生成期望值并对比，适合大量输出的回归检测。
    • 

  • 一致性测试验证 LSP 规范合规性，包括初始化顺序、错误码和关闭流程。
    • 

  • 性能基准测试关注启动时间、补全延迟等关键指标，在 CI 中与基线对比防止退化。
