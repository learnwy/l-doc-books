TypeScript 语言服务器

vscode-languageserver-node 框架

vscode-languageserver-node 是构建 Node.js 语言服务器的官方框架，提供了完整的 LSP 协议实现和便捷的助手工具。

{
  "dependencies": {
    "vscode-languageserver": "^9.0.1",
    "vscode-languageserver-textdocument": "^1.0.11"
  }
}
import {
  createConnection,
  TextDocuments,
  ProposedFeatures,
  InitializeParams,
  InitializeResult,
  TextDocumentSyncKind,
  DiagnosticSeverity,
} from 'vscode-languageserver/node';

import { TextDocument } from 'vscode-languageserver-textdocument';

const connection = createConnection(ProposedFeatures.all);
const documents = new TextDocuments(TextDocument);

connection.onInitialize((params: InitializeParams): InitializeResult => {
  return {
    capabilities: {
      textDocumentSync: TextDocumentSyncKind.Incremental,
      completionProvider: {
        resolveProvider: true,
        triggerCharacters: ['.'],
      },
      definitionProvider: true,
      hoverProvider: true,
      documentSymbolProvider: true,
      codeActionProvider: {
        codeActionKinds: ['quickfix', 'refactor'],
      },
      renameProvider: { prepareProvider: true },
    },
  };
});

documents.listen(connection);
connection.listen();

请求处理器实现
以下展示补全、定义跳转和悬停信息三个核心请求的实现模式。


connection.onCompletion((params): CompletionItem[] => {
  const document = documents.get(params.textDocument.uri);
  if (!document) return [];
  const text = document.getText();
  const offset = document.offsetAt(params.position);
  const linePrefix = text.substring(text.lastIndexOf('\n', offset - 1) + 1, offset);

  const items: CompletionItem[] = [];
  if (linePrefix.endsWith('console.')) {
    items.push(
      { label: 'log', kind: CompletionItemKind.Method, detail: 'Log output' },
      { label: 'warn', kind: CompletionItemKind.Method, detail: 'Warning output' },
      { label: 'error', kind: CompletionItemKind.Method, detail: 'Error output' },
    );
  }
  return items;
});

connection.onDefinition((params): Location | null => {
  const document = documents.get(params.textDocument.uri);
  if (!document) return null;
  const word = getWordAtPosition(document, params.position);
  if (!word) return null;
  for (const [uri, doc] of documents.all()) {
    const match = getDocumentSymbols(doc).find((s) => s.name === word);
    if (match) return { uri, range: match.range };
  }
  return null;
});

connection.onHover((params): Hover | null => {
  const document = documents.get(params.textDocument.uri);
  if (!document) return null;
  const word = getWordAtPosition(document, params.position);
  const symbol = word ? findSymbolDefinition(word) : null;
  if (!symbol) return null;
  return {
    contents: {
      kind: 'markdown',
      value: `\`\`\`typescript\n${symbol.signature}\n\`\`\`\n\n${symbol.documentation ?? ''}`,
    },
  };
});


诊断推送
诊断推送是语言服务器的核心职责之一。通常在文档变更后延迟触发分析并推送结果。


documents.onDidChangeContent((change) => {
  validateDocument(change.document);
});

async function validateDocument(document: TextDocument): Promise<void> {
  const diagnostics: Diagnostic[] = [];
  const text = document.getText();
  const lines = text.split('\n');

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];

    const todoMatch = line.match(/\/\/\s*TODO\b/i);
    if (todoMatch) {
      diagnostics.push({
        severity: DiagnosticSeverity.Information,
        range: {
          start: { line: i, character: todoMatch.index! },
          end: { line: i, character: todoMatch.index! + todoMatch[0].length },
        },
        message: 'TODO comment found',
        source: 'my-language-server',
      });
    }

    if (line.length > 120) {
      diagnostics.push({
        severity: DiagnosticSeverity.Warning,
        range: {
          start: { line: i, character: 120 },
          end: { line: i, character: line.length },
        },
        message: `Line exceeds 120 characters (${line.length})`,
        source: 'my-language-server',
      });
    }
  }

  connection.sendDiagnostics({
    uri: document.uri,
    version: document.version,
    diagnostics,
  });
}

客户端扩展
语言服务器通常搭配一个编辑器扩展（客户端），负责启动服务器进程和转发 LSP 消息。


import { LanguageClient, LanguageClientOptions, ServerOptions, TransportKind } from 'vscode-languageclient/node';

let client: LanguageClient;

export function activate(context: ExtensionContext): void {
  const serverModule = context.asAbsolutePath('server/out/server.js');
  const serverOptions: ServerOptions = {
    run: { module: serverModule, transport: TransportKind.ipc },
    debug: {
      module: serverModule,
      transport: TransportKind.ipc,
      options: { execArgv: ['--nolazy', '--inspect=6009'] },
    },
  };
  const clientOptions: LanguageClientOptions = {
    documentSelector: [{ scheme: 'file', language: 'mylang' }],
    synchronize: { fileEvents: workspace.createFileSystemWatcher('**/*.mylang') },
  };
  client = new LanguageClient('myLanguageServer', 'My Language Server', serverOptions, clientOptions);
  client.start();
}

export function deactivate(): Thenable<void> | undefined {
  return client?.stop();
}

小结
    

  • vscode-languageserver-node 是构建 TypeScript 语言服务器的官方框架，TextDocuments 自动处理文档同步。
    • 

  • 核心请求处理器（onCompletion/onDefinition/onHover）接收位置参数，查询文档模型后返回结果。
    • 

  • 诊断推送通常在 onDidChangeContent 中触发，配合去抖策略避免频繁分析。
    • 

  • 客户端扩展通过 LanguageClient 启动服务器进程，documentSelector 指定处理的文件类型。
    • 

  • 开发时使用 --inspect 参数支持断点调试，IPC 传输模式性能最优。
    •
