MoveIt2 运动规划

核心概念

MoveIt2 是 ROS2 生态系统中最重要的机器人臂运动规划框架，为工业机器人、协作机器人和移动操作平台提供了完整的运动规划、碰撞检测、逆运动学和轨迹执行能力。MoveIt2 继承了 MoveIt1 的成熟架构，同时利用 ROS2 的实时通信和生命周期管理特性进行了现代化改造。

MoveIt2 的核心设计围绕"规划组"（Planning Group）展开——每个规划组定义了一组关节和链接，代表机器人的一个运动子系统（如机械臂、手爪、双臂等）。规划请求指定目标位姿，MoveIt2 通过运动规划器在配置空间中搜索无碰撞路径，然后通过时间参数化生成可执行轨迹。

规划场景与碰撞检测

规划场景（Planning Scene）是 MoveIt2 的环境模型，它维护了机器人的当前状态、已知障碍物和约束条件。碰撞检测在规划过程中被频繁调用（每次状态采样都需要检测），因此效率至关重要。

import numpy as np

class CollisionObject:
    def __init__(self, obj_id, shape, pose, dimensions):
        self.id = obj_id
        self.shape = shape
        self.pose = pose
        self.dimensions = dimensions

    def get_aabb(self):
        if self.shape == 'box':
            half = np.array(self.dimensions) / 2
            return self.pose[:3] - half, self.pose[:3] + half
        elif self.shape == 'sphere':
            r = self.dimensions[0]
            return self.pose[:3] - r, self.pose[:3] + r
        return None, None


class PlanningScene:
    def __init__(self):
        self.collision_objects = {}
        self.attached_objects = {}
        self.allowed_collision_matrix = {}

    def add_collision_object(self, obj):
        self.collision_objects[obj.id] = obj

    def remove_collision_object(self, obj_id):
        self.collision_objects.pop(obj_id, None)

    def attach_object(self, obj_id, link_name):
        if obj_id in self.collision_objects:
            self.attached_objects[obj_id] = {
                'object': self.collision_objects.pop(obj_id),
                'link': link_name
            }

    def check_collision(self, robot_state):
        for link_aabb in robot_state.get_link_aabbs():
            for obj in self.collision_objects.values():
                obj_min, obj_max = obj.get_aabb()
                if obj_min is not None:
                    if self._aabb_overlap(link_aabb, (obj_min, obj_max)):
                        return True
        return False

    def check_self_collision(self, robot_state):
        links = robot_state.get_link_aabbs()
        for i in range(len(links)):
            for j in range(i + 2, len(links)):
                link_i_name = robot_state.link_names[i]
                link_j_name = robot_state.link_names[j]
                pair = (link_i_name, link_j_name)
                if pair in self.allowed_collision_matrix:
                    continue
                if self._aabb_overlap(links[i], links[j]):
                    return True
        return False

    def _aabb_overlap(self, aabb1, aabb2):
        min1, max1 = aabb1
        min2, max2 = aabb2
        return np.all(min1 <= max2) and np.all(min2 <= max1)


class OctomapIntegration:
    def __init__(self, resolution=0.05):
        self.resolution = resolution
        self.occupied_cells = set()

    def insert_point_cloud(self, points, sensor_origin):
        for point in points:
            cell = tuple((point / self.resolution).astype(int))
            self.occupied_cells.add(cell)

    def to_collision_objects(self):
        objects = []
        for cell in self.occupied_cells:
            pos = np.array(cell) * self.resolution + self.resolution / 2
            obj = CollisionObject(
                f'octomap_{cell}', 'box', np.append(pos, [0, 0, 0, 1]),
                [self.resolution] * 3
            )
            objects.append(obj)
        return objects

Allowed Collision Matrix（ACM）是优化碰撞检测性能的关键数据结构。它记录了哪些链接对永远不需要检测碰撞（如相邻链接、始终不会接触的链接对），避免了大量无效的碰撞检测计算。SRDF 文件中通过 <disable_collisions> 标签预定义 ACM。

运动规划器

MoveIt2 支持多种运动规划器，通过插件机制可以灵活切换。OMPL（Open Motion Planning Library）是默认的规划器集合，提供了数十种基于采样的规划算法。

import numpy as np

class MotionPlanRequest:
    def __init__(self, group_name, goal_constraints, path_constraints=None):
        self.group_name = group_name
        self.goal_constraints = goal_constraints
        self.path_constraints = path_constraints
        self.planner_id = 'RRTConnectkConfigDefault'
        self.planning_time = 5.0
        self.num_planning_attempts = 10
        self.max_velocity_scaling = 1.0
        self.max_acceleration_scaling = 1.0


class SimplifiedOMPLPlanner:
    def __init__(self, planning_scene, joint_limits):
        self.scene = planning_scene
        self.joint_limits = joint_limits
        self.step_size = 0.05

    def plan(self, start_joints, goal_joints, max_iter=5000):
        tree_start = [{'joints': start_joints, 'parent': -1}]
        tree_goal = [{'joints': goal_joints, 'parent': -1}]

        for iteration in range(max_iter):
            q_rand = self._random_config()

            idx_s = self._nearest(tree_start, q_rand)
            q_new_s = self._steer(tree_start[idx_s]['joints'], q_rand)
            if q_new_s is not None and not self._in_collision(q_new_s):
                tree_start.append({'joints': q_new_s, 'parent': idx_s})

                idx_g = self._nearest(tree_goal, q_new_s)
                q_new_g = self._steer(tree_goal[idx_g]['joints'], q_new_s)
                if q_new_g is not None and not self._in_collision(q_new_g):
                    tree_goal.append({'joints': q_new_g, 'parent': idx_g})

                    if np.linalg.norm(q_new_s - q_new_g) < self.step_size:
                        return self._extract_path(tree_start, tree_goal)

            tree_start, tree_goal = tree_goal, tree_start

        return None

    def _random_config(self):
        return np.random.uniform(
            self.joint_limits[:, 0], self.joint_limits[:, 1]
        )

    def _nearest(self, tree, q):
        dists = [np.linalg.norm(node['joints'] - q) for node in tree]
        return int(np.argmin(dists))

    def _steer(self, q_from, q_to):
        diff = q_to - q_from
        dist = np.linalg.norm(diff)
        if dist <= self.step_size:
            return q_to.copy()
        return q_from + diff / dist * self.step_size

    def _in_collision(self, joints):
        return False

    def _extract_path(self, tree_s, tree_g):
        path_s = []
        idx = len(tree_s) - 1
        while idx != -1:
            path_s.append(tree_s[idx]['joints'])
            idx = tree_s[idx]['parent']

        path_g = []
        idx = len(tree_g) - 1
        while idx != -1:
            path_g.append(tree_g[idx]['joints'])
            idx = tree_g[idx]['parent']

        return path_s[::-1] + path_g

Pilz 工业运动规划器提供了确定性的运动规划，适用于对路径可预测性要求极高的工业场景。它支持三种标准工业运动类型：PTP（点对点关节空间运动）、LIN（笛卡尔空间直线运动）和 CIRC（笛卡尔空间圆弧运动）。

MoveIt2 实践模式

MoveIt2 的典型使用方式是通过 MoveGroupInterface（C++）或 MoveGroupCommander（Python）进行交互。这些高层接口封装了复杂的规划和执行流程。

class MoveItInterface:
    def __init__(self, group_name, node):
        self.group_name = group_name
        self.node = node
        self.planning_frame = 'base_link'
        self.end_effector_link = 'tool0'

    def set_joint_target(self, joint_values):
        self.target_joints = joint_values

    def set_pose_target(self, pose):
        self.target_pose = pose

    def set_path_constraints(self, constraints):
        self.path_constraints = constraints

    def plan_and_execute(self):
        plan = self._plan()
        if plan is not None:
            return self._execute(plan)
        return False

    def _plan(self):
        return {'trajectory': [], 'planning_time': 0.5}

    def _execute(self, plan):
        return True

    def pick(self, object_id, grasp_poses):
        for pose in grasp_poses:
            pre_grasp = self._offset_pose(pose, z_offset=0.1)
            self.set_pose_target(pre_grasp)
            if self.plan_and_execute():
                self.set_pose_target(pose)
                if self.plan_and_execute():
                    self._close_gripper()
                    self._attach_object(object_id)
                    self.set_pose_target(pre_grasp)
                    return self.plan_and_execute()
        return False

    def place(self, object_id, place_pose):
        pre_place = self._offset_pose(place_pose, z_offset=0.1)
        self.set_pose_target(pre_place)
        if self.plan_and_execute():
            self.set_pose_target(place_pose)
            if self.plan_and_execute():
                self._open_gripper()
                self._detach_object(object_id)
                return True
        return False

    def _offset_pose(self, pose, z_offset):
        new_pose = pose.copy()
        new_pose[2] += z_offset
        return new_pose

    def _close_gripper(self): pass
    def _open_gripper(self): pass
    def _attach_object(self, obj_id): pass
    def _detach_object(self, obj_id): pass

MoveIt Servo 是 MoveIt2 中的实时控制模块，支持通过手柄、键盘或外部控制器对机械臂进行实时遥操作。Servo 在每个控制周期接收笛卡尔速度或关节速度指令，进行碰撞检测后直接发送到关节控制器，延迟通常在 10ms 以内。

小结

  • MoveIt2 是 ROS2 中最完善的机器人臂运动规划框架，覆盖规划、碰撞检测和轨迹执行
  • 规划场景管理环境模型，ACM 优化碰撞检测性能
  • OMPL 提供丰富的基于采样的规划算法，Pilz 提供确定性工业运动规划
  • MoveGroupInterface 封装了规划-执行的完整流程，支持 pick-and-place 等高层操作
  • MoveIt Servo 实现毫秒级实时控制，适用于遥操作和示教场景
  • 碰撞检测使用 FCL/Bullet 库，Octomap 集成支持动态环境感知
  • TOPP-RA 时间参数化确保生成的轨迹满足关节速度和加速度约束
