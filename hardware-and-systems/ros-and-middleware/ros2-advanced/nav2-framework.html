Nav2 导航框架

核心概念

Nav2（Navigation2）是 ROS2 官方的导航框架，提供了移动机器人自主导航的完整解决方案。它包含全局路径规划、局部轨迹跟踪、避障、恢复行为和行为树编排等核心功能。Nav2 采用高度模块化的架构，每个功能模块都可以通过插件机制替换为自定义实现，使其适用于从差速驱动小车到全向平台、Ackermann 转向车辆等各种机器人形态。

Nav2 的核心设计理念是"行为树驱动的导航"——不同于传统的有限状态机，Nav2 使用 BehaviorTree.CPP 库实现的行为树来编排导航的各个阶段，提供了更灵活、可组合的任务逻辑。

代价地图与环境表示
Costmap 2D 是 Nav2 的环境感知基础，它将传感器数据转化为二维栅格代价地图，每个栅格的代价值反映该位置的通行难度。全局代价地图基于静态地图和传感器数据构建，局部代价地图基于实时传感器数据在机器人周围构建。


代价地图采用分层架构，每层（Layer）独立处理一种信息源，最终通过合并策略（如取最大值）生成统一的代价地图。


import numpy as np

class CostmapLayer:
    def __init__(self, width, height, resolution):
        self.width = width
        self.height = height
        self.resolution = resolution
        self.grid = np.zeros((height, width), dtype=np.uint8)

    def world_to_map(self, wx, wy, origin_x, origin_y):
        mx = int((wx - origin_x) / self.resolution)
        my = int((wy - origin_y) / self.resolution)
        return mx, my


class StaticLayer(CostmapLayer):
    def update(self, static_map):
        self.grid = static_map.copy()


class ObstacleLayer(CostmapLayer):
    def __init__(self, width, height, resolution, max_range=10.0):
        super().__init__(width, height, resolution)
        self.max_range = max_range

    def update(self, scan_points, robot_x, robot_y, origin_x, origin_y):
        self.grid.fill(0)
        for px, py in scan_points:
            mx, my = self.world_to_map(px, py, origin_x, origin_y)
            if 0 <= mx < self.width and 0 <= my < self.height:
                self.grid[my, mx] = 254

    def raytrace_clear(self, robot_x, robot_y, point_x, point_y, origin_x, origin_y):
        rx, ry = self.world_to_map(robot_x, robot_y, origin_x, origin_y)
        px, py = self.world_to_map(point_x, point_y, origin_x, origin_y)
        for x, y in self._bresenham(rx, ry, px, py):
            if 0 <= x < self.width and 0 <= y < self.height:
                self.grid[y, x] = 0

    def _bresenham(self, x0, y0, x1, y1):
        points = []
        dx, dy = abs(x1 - x0), abs(y1 - y0)
        sx = 1 if x0 < x1 else -1
        sy = 1 if y0 < y1 else -1
        err = dx - dy
        while True:
            points.append((x0, y0))
            if x0 == x1 and y0 == y1:
                break
            e2 = 2 * err
            if e2 > -dy:
                err -= dy
                x0 += sx
            if e2 < dx:
                err += dx
                y0 += sy
        return points


class InflationLayer(CostmapLayer):
    def __init__(self, width, height, resolution, inflation_radius=0.5, cost_scaling=10.0):
        super().__init__(width, height, resolution)
        self.inflation_radius = inflation_radius
        self.cost_scaling = cost_scaling
        self.inflation_cells = int(inflation_radius / resolution)

    def update(self, master_grid):
        self.grid.fill(0)
        obstacles = np.argwhere(master_grid >= 254)

        for oy, ox in obstacles:
            for dy in range(-self.inflation_cells, self.inflation_cells + 1):
                for dx in range(-self.inflation_cells, self.inflation_cells + 1):
                    ny, nx = oy + dy, ox + dx
                    if 0 <= ny < self.height and 0 <= nx < self.width:
                        dist = np.sqrt(dy**2 + dx**2) * self.resolution
                        if dist <= self.inflation_radius:
                            cost = int(252 * np.exp(-self.cost_scaling * dist))
                            self.grid[ny, nx] = max(self.grid[ny, nx], cost)
膨胀层（Inflation Layer）是路径规划的安全保障。它在障碍物周围生成渐变的代价值，使规划器生成的路径自然远离障碍物。膨胀半径通常设置为机器人外接圆半径，代价衰减函数决定了机器人保持安全距离的程度。


全局规划与控制器
Nav2 的全局规划器接收目标位置和代价地图，输出从机器人当前位置到目标的最优路径。控制器接收全局路径和实时传感器数据，生成跟踪路径的速度指令。


import numpy as np

class SmacPlannerConfig:
    def __init__(self):
        self.motion_model = 'DUBIN'
        self.minimum_turning_radius = 0.5
        self.analytic_expansion_ratio = 3.5
        self.max_iterations = 10000
        self.tolerance = 0.25
        self.allow_unknown = True
        self.max_planning_time = 5.0


class RegulatedPurePursuit:
    def __init__(self, lookahead_dist=0.6, max_linear=0.5, max_angular=1.0):
        self.lookahead_dist = lookahead_dist
        self.max_linear = max_linear
        self.max_angular = max_angular
        self.lookahead_gain = 0.5
        self.min_lookahead = 0.3
        self.max_lookahead = 1.0

    def compute_velocity(self, robot_pose, path, current_speed):
        adaptive_lookahead = max(
            self.min_lookahead,
            min(self.max_lookahead, self.lookahead_dist + self.lookahead_gain * current_speed)
        )

        lookahead_point = self._find_lookahead_point(robot_pose, path, adaptive_lookahead)
        if lookahead_point is None:
            return 0.0, 0.0

        dx = lookahead_point[0] - robot_pose[0]
        dy = lookahead_point[1] - robot_pose[1]
        local_x = dx * np.cos(robot_pose[2]) + dy * np.sin(robot_pose[2])
        local_y = -dx * np.sin(robot_pose[2]) + dy * np.cos(robot_pose[2])

        L = np.sqrt(local_x**2 + local_y**2)
        if L < 0.01:
            return 0.0, 0.0

        curvature = 2.0 * local_y / (L * L)

        linear = self.max_linear
        linear = self._regulate_linear(linear, curvature, path, robot_pose)

        angular = linear * curvature
        angular = np.clip(angular, -self.max_angular, self.max_angular)

        return linear, angular

    def _find_lookahead_point(self, pose, path, lookahead):
        min_dist_sq = float('inf')
        closest_idx = 0
        for i, pt in enumerate(path):
            d = (pt[0]-pose[0])**2 + (pt[1]-pose[1])**2
            if d < min_dist_sq:
                min_dist_sq = d
                closest_idx = i

        for i in range(closest_idx, len(path)):
            d = np.sqrt((path[i][0]-pose[0])**2 + (path[i][1]-pose[1])**2)
            if d >= lookahead:
                return path[i]

        return path[-1] if len(path) > 0 else None

    def _regulate_linear(self, speed, curvature, path, pose):
        if abs(curvature) > 0.5:
            speed *= max(0.2, 1.0 - abs(curvature) * 0.5)

        if len(path) > 0:
            goal = path[-1]
            dist_to_goal = np.sqrt((goal[0]-pose[0])**2 + (goal[1]-pose[1])**2)
            if dist_to_goal < 0.5:
                speed *= max(0.1, dist_to_goal / 0.5)

        return speed

行为树导航
行为树（Behavior Tree, BT）是 Nav2 编排导航逻辑的核心机制。与有限状态机（FSM）相比，行为树具有更好的模块性和可扩展性——每个节点独立实现单一行为，通过组合节点（Sequence、Fallback、Parallel）实现复杂逻辑。


class BTNode:
    SUCCESS = 'SUCCESS'
    FAILURE = 'FAILURE'
    RUNNING = 'RUNNING'

    def tick(self):
        raise NotImplementedError


class SequenceNode(BTNode):
    def __init__(self, children):
        self.children = children
        self.current = 0

    def tick(self):
        while self.current < len(self.children):
            status = self.children[self.current].tick()
            if status == BTNode.RUNNING:
                return BTNode.RUNNING
            if status == BTNode.FAILURE:
                self.current = 0
                return BTNode.FAILURE
            self.current += 1
        self.current = 0
        return BTNode.SUCCESS


class FallbackNode(BTNode):
    def __init__(self, children):
        self.children = children

    def tick(self):
        for child in self.children:
            status = child.tick()
            if status != BTNode.FAILURE:
                return status
        return BTNode.FAILURE


class ComputePathAction(BTNode):
    def __init__(self, planner_id='GridBased'):
        self.planner_id = planner_id
        self.path = None

    def tick(self):
        if self.path is None:
            return BTNode.RUNNING
        return BTNode.SUCCESS


class FollowPathAction(BTNode):
    def __init__(self, controller_id='FollowPath'):
        self.controller_id = controller_id

    def tick(self):
        return BTNode.RUNNING


class RecoveryAction(BTNode):
    def __init__(self, behavior='spin'):
        self.behavior = behavior

    def tick(self):
        return BTNode.SUCCESS


def build_navigate_to_pose_bt():
    main_sequence = SequenceNode([
        ComputePathAction('GridBased'),
        FollowPathAction('FollowPath'),
    ])

    recovery_fallback = FallbackNode([
        main_sequence,
        SequenceNode([
            RecoveryAction('clear_costmap'),
            main_sequence,
        ]),
        SequenceNode([
            RecoveryAction('spin'),
            main_sequence,
        ]),
        SequenceNode([
            RecoveryAction('backup'),
            main_sequence,
        ]),
    ])

    return recovery_fallback
Nav2 的默认行为树 navigate_to_pose.xml 实现了一个带恢复机制的导航循环：先尝试规划和跟踪路径，如果失败则依次尝试清除代价地图、原地旋转和后退等恢复行为，直到成功导航到目标或耗尽所有恢复选项。


小结
    

  • Nav2 是 ROS2 的官方导航框架，提供从路径规划到运动控制的完整导航能力
    • 

  • Costmap 2D 的分层架构将传感器数据、静态地图和膨胀层组合为统一的代价地图
    • 

  • 全局规划器（NavFn、Smac Hybrid A*）和局部控制器（DWB、RPP、MPPI）通过插件机制灵活替换
    • 

  • 行为树驱动的导航编排提供了比状态机更好的模块性和可扩展性
    • 

  • Regulated Pure Pursuit 在高速场景下通过曲率调节和速度约束保证安全
    • 

  • MPPI 是最新一代控制器，通过 GPU 并行采样实现高质量动态避障
    • 

  • Nav2 的恢复行为机制使机器人能够自动从导航失败中恢复
