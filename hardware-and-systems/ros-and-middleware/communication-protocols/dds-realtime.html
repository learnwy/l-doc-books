DDS 与实时通信

核心概念

DDS（Data Distribution Service）是 OMG（Object Management Group）制定的数据分发标准，是 ROS2 通信架构的底层基础。DDS 采用发布-订阅模式，提供去中心化的数据分发、丰富的 QoS 策略和内建的发现机制，使 ROS2 能够满足从研究原型到工业级部署的通信需求。

ROS2 通过 RMW（ROS Middleware Interface）抽象层与 DDS 实现交互，支持多种 DDS 厂商实现（如 Fast-DDS、Cyclone DDS、Connext DDS）。这种架构允许用户根据性能需求和许可证约束选择最适合的 DDS 实现。

QoS 策略详解
QoS（Quality of Service）策略是 DDS 通信的核心特性，允许开发者精确控制数据传输的可靠性、实时性和资源使用。ROS2 将 DDS 的 QoS 策略映射为易用的 Python/C++ 接口。


from rclpy.qos import (
    QoSProfile, QoSReliabilityPolicy, QoSHistoryPolicy,
    QoSDurabilityPolicy, QoSLivelinessPolicy
)
from rclpy.duration import Duration

sensor_qos = QoSProfile(
    reliability=QoSReliabilityPolicy.BEST_EFFORT,
    history=QoSHistoryPolicy.KEEP_LAST,
    depth=5,
    durability=QoSDurabilityPolicy.VOLATILE,
    deadline=Duration(seconds=0, nanoseconds=100_000_000),
    lifespan=Duration(seconds=1),
    liveliness=QoSLivelinessPolicy.AUTOMATIC,
    liveliness_lease_duration=Duration(seconds=5)
)

reliable_qos = QoSProfile(
    reliability=QoSReliabilityPolicy.RELIABLE,
    history=QoSHistoryPolicy.KEEP_ALL,
    durability=QoSDurabilityPolicy.TRANSIENT_LOCAL,
)

realtime_qos = QoSProfile(
    reliability=QoSReliabilityPolicy.BEST_EFFORT,
    history=QoSHistoryPolicy.KEEP_LAST,
    depth=1,
    durability=QoSDurabilityPolicy.VOLATILE,
    deadline=Duration(seconds=0, nanoseconds=10_000_000),
)
from rclpy.node import Node
from rclpy.qos import QoSProfile
from rclpy.qos_event import SubscriptionEventCallbacks, QoSRequestedDeadlineMissedInfo

class QoSAwareNode(Node):
    def __init__(self):
        super().__init__('qos_aware_node')

        event_callbacks = SubscriptionEventCallbacks(
            deadline=self.deadline_callback,
            liveliness=self.liveliness_callback,
        )

        qos = QoSProfile(depth=10)
        qos.deadline = Duration(seconds=0, nanoseconds=100_000_000)

        self.sub = self.create_subscription(
            LaserScan, '/scan',
            self.scan_callback, qos,
            event_callbacks=event_callbacks
        )

        self.deadline_miss_count = 0

    def scan_callback(self, msg):
        self.get_logger().debug('收到扫描数据')

    def deadline_callback(self, event: QoSRequestedDeadlineMissedInfo):
        self.deadline_miss_count += 1
        self.get_logger().warn(
            f'Deadline 违规! 总计: {self.deadline_miss_count}'
        )

    def liveliness_callback(self, event):
        self.get_logger().warn('发布者活跃性变更')

RTPS 协议与发现机制
RTPS（Real-Time Publish-Subscribe）是 DDS 的传输协议，定义了参与者发现、数据序列化和可靠传输等机制。理解 RTPS 有助于诊断通信问题和优化网络性能。


import os
import xml.etree.ElementTree as ET

def generate_fastdds_profile(domain_id=0, discovery_server=None):
    profiles = ET.Element('profiles')
    profiles.set('xmlns', 'http://www.eprosima.com/XMLSchemas/fastRTPS_Profiles')

    participant = ET.SubElement(profiles, 'participant', profile_name='default_participant')
    rtps = ET.SubElement(participant, 'rtps')

    builtin = ET.SubElement(rtps, 'builtin')
    discovery = ET.SubElement(builtin, 'discovery_config')

    if discovery_server:
        protocol = ET.SubElement(discovery, 'discoveryProtocol')
        protocol.text = 'CLIENT'
        servers = ET.SubElement(discovery, 'discoveryServersList')
        server = ET.SubElement(servers, 'RemoteServer', prefix='44.53.00.5f.45.50.52.4f.53.49.4d.41')
        metatraffic = ET.SubElement(server, 'metatrafficUnicastLocatorList')
        locator = ET.SubElement(metatraffic, 'locator')
        ET.SubElement(locator, 'udpv4')
        address = ET.SubElement(locator, 'address')
        address.text = discovery_server
        port = ET.SubElement(locator, 'port')
        port.text = '11811'

    transport = ET.SubElement(rtps, 'userTransports')
    descriptor = ET.SubElement(transport, 'transport_id')
    descriptor.text = 'shm_transport'

    return ET.tostring(profiles, encoding='unicode')

共享内存与零拷贝传输
对于大数据量话题（如图像、点云），传统的序列化-传输-反序列化路径会成为性能瓶颈。共享内存传输和零拷贝（Zero-Copy）技术可以显著降低延迟和 CPU 开销。


#include <rclcpp/rclcpp.hpp>
#include <sensor_msgs/msg/image.hpp>

class ZeroCopyPublisher : public rclcpp::Node {
public:
    ZeroCopyPublisher() : Node("zero_copy_publisher") {
        auto qos = rclcpp::QoS(1).best_effort();
        publisher_ = this->create_publisher<sensor_msgs::msg::Image>(
            "/camera/image", qos
        );

        timer_ = this->create_wall_timer(
            std::chrono::milliseconds(33),
            std::bind(&ZeroCopyPublisher::publish_image, this)
        );
    }

private:
    void publish_image() {
        auto msg = publisher_->borrow_loaned_message();

        auto & image = msg.get();
        image.header.stamp = this->now();
        image.header.frame_id = "camera_optical";
        image.width = 640;
        image.height = 480;
        image.encoding = "rgb8";
        image.step = 640 * 3;
        image.data.resize(640 * 480 * 3);

        fill_image_data(image.data);

        publisher_->publish(std::move(msg));
    }

    void fill_image_data(std::vector<uint8_t>& data) {
        for (size_t i = 0; i < data.size(); i += 3) {
            data[i] = 128;
            data[i+1] = 64;
            data[i+2] = 32;
        }
    }

    rclcpp::Publisher<sensor_msgs::msg::Image>::SharedPtr publisher_;
    rclcpp::TimerBase::SharedPtr timer_;
};
import numpy as np

class CommunicationBenchmark:
    def __init__(self):
        self.results = {}

    def measure_latency(self, publisher, subscriber, msg_size, n_samples=1000):
        latencies = []

        for _ in range(n_samples):
            send_time = time.monotonic_ns()
            publisher.publish(create_message(msg_size))
            recv_time = subscriber.wait_for_message(timeout=1.0)

            if recv_time:
                latency_us = (recv_time - send_time) / 1000
                latencies.append(latency_us)

        return {
            'mean_us': np.mean(latencies),
            'p50_us': np.percentile(latencies, 50),
            'p99_us': np.percentile(latencies, 99),
            'max_us': np.max(latencies),
            'jitter_us': np.std(latencies)
        }

小结
    

  • DDS 是 ROS2 的通信基础，提供去中心化发布-订阅、QoS 策略和自动发现
    • 

  • QoS 策略（可靠性、持久性、截止时间）使通信行为可精确调优
    • 

  • RTPS 协议支持多播发现和点对点传输，通过 DDS 配置文件优化网络行为
    • 

  • 共享内存和零拷贝技术将大数据传输延迟从毫秒级降低到微秒级
    • 

  • DDS 实现选择（Fast-DDS vs Cyclone DDS）对性能和功能有显著影响
    • 

  • 实时通信需要从 QoS 配置到操作系统调优的全栈协同
    •
