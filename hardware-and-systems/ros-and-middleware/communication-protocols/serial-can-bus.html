串口与 CAN 总线通信

核心概念

串口通信（Serial Communication）和 CAN 总线（Controller Area Network）是机器人底层硬件通信的两大主力协议。串口协议简单直观，广泛用于传感器、GPS 模块和微控制器的数据交换；CAN 总线以其高可靠性和多节点仲裁能力，成为工业机器人和移动机器人底盘控制的标准总线。

在机器人系统架构中，串口和 CAN 通常作为 ROS2 节点与底层硬件之间的"最后一公里"通信手段，由专门的硬件驱动节点负责协议转换。

串口通信实现
串口通信在机器人系统中主要用于连接 GPS、IMU、激光雷达等外部设备。Python 的 pyserial 库提供了跨平台的串口操作接口。


import serial
import struct
import threading
from collections import deque

class RobotSerialInterface:
    def __init__(self, port='/dev/ttyUSB0', baudrate=115200):
        self.serial = serial.Serial(
            port=port,
            baudrate=baudrate,
            bytesize=serial.EIGHTBITS,
            parity=serial.PARITY_NONE,
            stopbits=serial.STOPBITS_ONE,
            timeout=0.01
        )

        self.rx_buffer = deque(maxlen=4096)
        self.running = False
        self.callbacks = {}

    def start(self):
        self.running = True
        self.rx_thread = threading.Thread(target=self._read_loop, daemon=True)
        self.rx_thread.start()

    def _read_loop(self):
        while self.running:
            if self.serial.in_waiting:
                data = self.serial.read(self.serial.in_waiting)
                self.rx_buffer.extend(data)
                self._parse_frames()

    def _parse_frames(self):
        while len(self.rx_buffer) >= 4:
            if self.rx_buffer[0] != 0xAA or self.rx_buffer[1] != 0x55:
                self.rx_buffer.popleft()
                continue

            if len(self.rx_buffer) < 4:
                break

            msg_id = self.rx_buffer[2]
            payload_len = self.rx_buffer[3]

            total_len = 4 + payload_len + 1
            if len(self.rx_buffer) < total_len:
                break

            frame = [self.rx_buffer.popleft() for _ in range(total_len)]

            checksum = sum(frame[:-1]) & 0xFF
            if checksum != frame[-1]:
                continue

            payload = bytes(frame[4:4+payload_len])

            if msg_id in self.callbacks:
                self.callbacks[msg_id](payload)

    def send_command(self, msg_id, payload):
        frame = bytearray([0xAA, 0x55, msg_id, len(payload)])
        frame.extend(payload)
        checksum = sum(frame) & 0xFF
        frame.append(checksum)
        self.serial.write(frame)

    def register_callback(self, msg_id, callback):
        self.callbacks[msg_id] = callback

    def close(self):
        self.running = False
        if hasattr(self, 'rx_thread'):
            self.rx_thread.join(timeout=1.0)
        self.serial.close()

class OdometryParser:
    def __init__(self, serial_interface):
        self.serial = serial_interface
        self.serial.register_callback(0x01, self.parse_odometry)
        self.latest_odom = None

    def parse_odometry(self, payload):
        if len(payload) == 24:
            x, y, theta, vx, vy, omega = struct.unpack('<ffffff', payload)
            self.latest_odom = {
                'x': x, 'y': y, 'theta': theta,
                'vx': vx, 'vy': vy, 'omega': omega
            }

    def send_velocity(self, linear, angular):
        payload = struct.pack('<ff', linear, angular)
        self.serial.send_command(0x10, payload)
CAN 总线通信
CAN 总线是机器人底盘控制的首选协议。其核心优势包括：基于优先级的消息仲裁（无需主从）、错误检测和故障隔离、以及确定性的消息传输延迟。


import can
import struct
import threading

class RobotCANBus:
    def __init__(self, channel='can0', bitrate=500000):
        self.bus = can.interface.Bus(
            channel=channel,
            interface='socketcan',
            bitrate=bitrate
        )
        self.callbacks = {}
        self.running = False

    def start(self):
        self.running = True
        self.rx_thread = threading.Thread(target=self._receive_loop, daemon=True)
        self.rx_thread.start()

    def _receive_loop(self):
        while self.running:
            msg = self.bus.recv(timeout=0.01)
            if msg:
                self._dispatch(msg)

    def _dispatch(self, msg):
        arb_id = msg.arbitration_id

        if arb_id in self.callbacks:
            self.callbacks[arb_id](msg.data)

        base_id = arb_id & 0xFF80
        if base_id in self.callbacks:
            self.callbacks[base_id](msg.data, arb_id & 0x7F)

    def send(self, arb_id, data, is_extended=False):
        msg = can.Message(
            arbitration_id=arb_id,
            data=data,
            is_extended_id=is_extended
        )
        self.bus.send(msg)

    def register_callback(self, arb_id, callback):
        self.callbacks[arb_id] = callback

    def close(self):
        self.running = False
        self.bus.shutdown()

class CANMotorDriver:
    CMD_VELOCITY = 0x200
    CMD_POSITION = 0x201
    CMD_TORQUE = 0x202
    FEEDBACK_STATUS = 0x280

    def __init__(self, can_bus, motor_id):
        self.bus = can_bus
        self.motor_id = motor_id
        self.feedback = {'position': 0, 'velocity': 0, 'current': 0}

        feedback_id = self.FEEDBACK_STATUS + motor_id
        self.bus.register_callback(feedback_id, self._parse_feedback)

    def set_velocity(self, rpm):
        data = struct.pack('<hHxxxx', int(rpm * 10), 0)
        self.bus.send(self.CMD_VELOCITY + self.motor_id, data)

    def set_position(self, angle_deg):
        data = struct.pack('<ixxxx', int(angle_deg * 100))
        self.bus.send(self.CMD_POSITION + self.motor_id, data)

    def _parse_feedback(self, data):
        pos_raw, vel_raw, cur_raw = struct.unpack('<hhh', data[:6])
        self.feedback = {
            'position': pos_raw * 0.01,
            'velocity': vel_raw * 0.1,
            'current': cur_raw * 0.001
        }
CANopen 协议栈
CANopen 是建立在 CAN 总线之上的高层应用协议，定义了设备通信模型、对象字典和设备配置文件（Device Profile），广泛应用于工业自动化和机器人领域。


class CANopenNode:
    NMT_START = 0x01
    NMT_STOP = 0x02
    NMT_RESET = 0x81

    def __init__(self, can_bus, node_id):
        self.bus = can_bus
        self.node_id = node_id
        self.object_dictionary = {}
        self.pdo_mappings = {}

    def nmt_command(self, command):
        data = bytes([command, self.node_id])
        self.bus.send(0x000, data)

    def sdo_read(self, index, subindex):
        data = struct.pack('<BHBBBBB', 0x40, index, subindex, 0, 0, 0, 0)
        self.bus.send(0x600 + self.node_id, data)

    def sdo_write(self, index, subindex, value, size=4):
        cmd_byte = {1: 0x2F, 2: 0x2B, 4: 0x23}[size]
        data = struct.pack('<BHB', cmd_byte, index, subindex)
        data += struct.pack('<I', value)[:size]
        data += b'\x00' * (8 - len(data))
        self.bus.send(0x600 + self.node_id, data)

    def configure_tpdo(self, pdo_num, cob_id, mappings):
        comm_index = 0x1800 + pdo_num - 1
        map_index = 0x1A00 + pdo_num - 1

        self.sdo_write(comm_index, 0x01, cob_id | 0x80000000)

        self.sdo_write(map_index, 0x00, 0)
        for i, (index, subindex, bits) in enumerate(mappings):
            mapping_value = (index << 16) | (subindex << 8) | bits
            self.sdo_write(map_index, i + 1, mapping_value)
        self.sdo_write(map_index, 0x00, len(mappings))

        self.sdo_write(comm_index, 0x01, cob_id)

class CANopenMotorController(CANopenNode):
    MODE_VELOCITY = 3
    MODE_POSITION = 1
    MODE_TORQUE = 4

    def __init__(self, can_bus, node_id):
        super().__init__(can_bus, node_id)

    def set_operation_mode(self, mode):
        self.sdo_write(0x6060, 0x00, mode, size=1)

    def enable_motor(self):
        self.sdo_write(0x6040, 0x00, 0x06, size=2)
        self.sdo_write(0x6040, 0x00, 0x07, size=2)
        self.sdo_write(0x6040, 0x00, 0x0F, size=2)

    def set_target_velocity(self, rpm):
        self.sdo_write(0x60FF, 0x00, int(rpm))

    def set_target_position(self, counts):
        self.sdo_write(0x607A, 0x00, int(counts))
        self.sdo_write(0x6040, 0x00, 0x3F, size=2)

ROS2 硬件驱动集成
将串口和 CAN 通信封装为 ROS2 节点是机器人开发的标准实践。ros2_control 框架提供了硬件抽象层（Hardware Interface），使控制算法与硬件通信解耦。


import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist
from nav_msgs.msg import Odometry
from sensor_msgs.msg import JointState

class ChassisDriverNode(Node):
    def __init__(self):
        super().__init__('chassis_driver')

        self.declare_parameter('can_channel', 'can0')
        self.declare_parameter('wheel_radius', 0.08)
        self.declare_parameter('wheel_separation', 0.4)

        channel = self.get_parameter('can_channel').value
        self.can = RobotCANBus(channel=channel)
        self.left_motor = CANMotorDriver(self.can, motor_id=1)
        self.right_motor = CANMotorDriver(self.can, motor_id=2)

        self.cmd_sub = self.create_subscription(
            Twist, '/cmd_vel', self.cmd_callback, 10
        )

        self.odom_pub = self.create_publisher(Odometry, '/odom', 10)
        self.joint_pub = self.create_publisher(JointState, '/joint_states', 10)

        self.can.start()
        self.create_timer(0.02, self.publish_feedback)

    def cmd_callback(self, msg):
        R = self.get_parameter('wheel_radius').value
        L = self.get_parameter('wheel_separation').value

        v = msg.linear.x
        omega = msg.angular.z

        v_left = (v - omega * L / 2) / R
        v_right = (v + omega * L / 2) / R

        left_rpm = v_left * 60 / (2 * 3.14159)
        right_rpm = v_right * 60 / (2 * 3.14159)

        self.left_motor.set_velocity(left_rpm)
        self.right_motor.set_velocity(right_rpm)

    def publish_feedback(self):
        joint_msg = JointState()
        joint_msg.header.stamp = self.get_clock().now().to_msg()
        joint_msg.name = ['left_wheel_joint', 'right_wheel_joint']
        joint_msg.position = [
            self.left_motor.feedback['position'],
            self.right_motor.feedback['position']
        ]
        joint_msg.velocity = [
            self.left_motor.feedback['velocity'],
            self.right_motor.feedback['velocity']
        ]
        self.joint_pub.publish(joint_msg)
小结
    

  • 串口通信简单直观，适合传感器和低速设备接口，帧协议设计需考虑同步和校验
    • 

  • CAN 总线提供优先级仲裁、错误检测和多节点支持，是机器人底盘控制的首选协议
    • 

  • CANopen 在 CAN 之上定义了标准化的设备通信模型，简化了多设备集成
    • 

  • 硬件驱动节点将底层通信协议封装为 ROS2 话题和服务，实现控制算法与硬件解耦
    • 

  • 通信协议选择需综合考虑速率、距离、节点数量和实时性要求
    • 

  • 工业级部署建议使用 CAN FD 或 EtherCAT 以获得更高的带宽和实时性
    •
