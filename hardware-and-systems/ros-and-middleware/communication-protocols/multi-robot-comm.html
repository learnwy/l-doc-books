多机器人通信架构

核心概念

多机器人系统（Multi-Robot System, MRS）的通信架构是实现机器人协作的基础设施。当多台机器人需要协同完成任务时，通信架构的设计决定了系统的可扩展性、实时性和容错能力。通信架构需要解决的核心问题包括：数据路由、带宽管理、时间同步、网络分区恢复和安全访问控制。

多机器人通信架构从拓扑结构上可分为中心化、去中心化和层次化三种模式。每种模式在延迟、可靠性和复杂度之间有不同的权衡。

ROS2 多机器人通信配置
ROS2 的 DDS 天然支持多机器人通信，通过 Domain ID 隔离和话题命名空间实现多机器人的有序通信。


import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
from std_msgs.msg import String
import json

class MultiRobotCommNode(Node):
    def __init__(self, robot_id, team_size):
        super().__init__(f'robot_{robot_id}_comm')
        self.robot_id = robot_id
        self.team_size = team_size

        self.pose_pub = self.create_publisher(
            PoseStamped, f'/robot_{robot_id}/pose', 10
        )

        self.peer_poses = {}
        for i in range(team_size):
            if i != robot_id:
                self.create_subscription(
                    PoseStamped, f'/robot_{i}/pose',
                    lambda msg, rid=i: self.peer_pose_callback(msg, rid), 10
                )

        self.broadcast_pub = self.create_publisher(
            String, '/team/broadcast', 10
        )
        self.create_subscription(
            String, '/team/broadcast', self.broadcast_callback, 10
        )

        self.task_pub = self.create_publisher(
            String, f'/robot_{robot_id}/task_status', 10
        )

        self.create_timer(0.1, self.publish_pose)

    def peer_pose_callback(self, msg, robot_id):
        self.peer_poses[robot_id] = msg

    def broadcast_callback(self, msg):
        data = json.loads(msg.data)
        if data.get('sender') != self.robot_id:
            self.process_broadcast(data)

    def publish_pose(self):
        msg = PoseStamped()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.header.frame_id = 'map'
        self.pose_pub.publish(msg)

    def broadcast_message(self, msg_type, payload):
        data = {
            'sender': self.robot_id,
            'type': msg_type,
            'payload': payload,
            'timestamp': self.get_clock().now().nanoseconds
        }
        msg = String()
        msg.data = json.dumps(data)
        self.broadcast_pub.publish(msg)

    def process_broadcast(self, data):
        msg_type = data['type']
        if msg_type == 'task_allocation':
            self.handle_task_allocation(data['payload'])
        elif msg_type == 'obstacle_alert':
            self.handle_obstacle_alert(data['payload'])

    def handle_task_allocation(self, payload):
        pass

    def handle_obstacle_alert(self, payload):
        pass
分布式共识与任务分配
多机器人协作中，任务分配需要在分布式环境下达成共识。常用的分布式算法包括拍卖算法、共识算法和市场机制。


import numpy as np
from dataclasses import dataclass

@dataclass
class Task:
    task_id: int
    position: np.ndarray
    priority: float
    required_capability: str

class AuctionBasedAllocator:
    def __init__(self, robot_id, capabilities):
        self.robot_id = robot_id
        self.capabilities = capabilities
        self.assigned_tasks = []
        self.bids = {}

    def compute_bid(self, task, current_position):
        if task.required_capability not in self.capabilities:
            return -1

        distance = np.linalg.norm(task.position - current_position)
        workload = len(self.assigned_tasks)

        bid_value = task.priority / (distance + 1) / (workload + 1)
        return bid_value

    def submit_bids(self, tasks, current_position):
        bids = {}
        for task in tasks:
            bid = self.compute_bid(task, current_position)
            if bid > 0:
                bids[task.task_id] = {
                    'robot_id': self.robot_id,
                    'bid_value': bid,
                    'task_id': task.task_id
                }
        return bids

    def process_auction_result(self, winner_id, task_id):
        if winner_id == self.robot_id:
            self.assigned_tasks.append(task_id)
            return True
        return False

class ConsensusProtocol:
    def __init__(self, robot_id, n_robots, adjacency_matrix):
        self.robot_id = robot_id
        self.n_robots = n_robots
        self.A = adjacency_matrix
        self.state = np.zeros(n_robots)

    def update(self, neighbor_states):
        new_state = self.state[self.robot_id]

        neighbors = np.where(self.A[self.robot_id] > 0)[0]
        for j in neighbors:
            if j in neighbor_states:
                weight = self.A[self.robot_id, j]
                new_state += weight * (neighbor_states[j] - self.state[self.robot_id])

        self.state[self.robot_id] = new_state
        return new_state

    def has_converged(self, tolerance=0.01):
        return np.std(self.state) < tolerance
通信质量与带宽管理
在真实部署中，无线通信面临带宽有限、延迟波动和丢包等挑战。带宽管理策略需要根据任务优先级动态分配通信资源。


import time
import numpy as np
from collections import deque

class BandwidthManager:
    def __init__(self, max_bandwidth_kbps=1000):
        self.max_bandwidth = max_bandwidth_kbps * 1024 / 8
        self.channels = {}
        self.usage_history = deque(maxlen=100)

    def register_channel(self, name, priority, max_rate_hz, msg_size_bytes):
        self.channels[name] = {
            'priority': priority,
            'max_rate': max_rate_hz,
            'msg_size': msg_size_bytes,
            'last_send': 0,
            'allocated_rate': 0,
            'dropped': 0
        }
        self._reallocate()

    def _reallocate(self):
        total_priority = sum(c['priority'] for c in self.channels.values())

        for name, channel in self.channels.items():
            share = channel['priority'] / total_priority
            max_bytes_per_sec = self.max_bandwidth * share
            allocated_rate = min(
                channel['max_rate'],
                max_bytes_per_sec / channel['msg_size']
            )
            channel['allocated_rate'] = allocated_rate

    def should_send(self, channel_name):
        channel = self.channels[channel_name]
        now = time.monotonic()

        min_interval = 1.0 / channel['allocated_rate'] if channel['allocated_rate'] > 0 else float('inf')

        if now - channel['last_send'] >= min_interval:
            channel['last_send'] = now
            return True

        channel['dropped'] += 1
        return False

class AdaptiveCompression:
    def __init__(self):
        self.compression_level = 1
        self.network_quality = 1.0

    def update_network_quality(self, rtt_ms, packet_loss_rate):
        if rtt_ms < 50 and packet_loss_rate < 0.01:
            self.network_quality = 1.0
            self.compression_level = 0
        elif rtt_ms < 100 and packet_loss_rate < 0.05:
            self.network_quality = 0.7
            self.compression_level = 1
        else:
            self.network_quality = 0.3
            self.compression_level = 2

    def compress_pointcloud(self, points, level=None):
        if level is None:
            level = self.compression_level

        if level == 0:
            return points
        elif level == 1:
            indices = np.random.choice(len(points), len(points) // 4, replace=False)
            return points[indices]
        else:
            voxel_size = 0.1
            quantized = np.round(points / voxel_size) * voxel_size
            unique_points = np.unique(quantized, axis=0)
            return unique_points

网络分区与容错机制
无线通信环境中，网络分区（某些机器人暂时失联）是常见场景。系统需要在分区期间保持局部功能，并在恢复连接后实现状态同步。


import time

class PartitionDetector:
    def __init__(self, robot_id, heartbeat_timeout=3.0):
        self.robot_id = robot_id
        self.heartbeat_timeout = heartbeat_timeout
        self.peer_heartbeats = {}
        self.partition_callbacks = []

    def update_heartbeat(self, peer_id, timestamp):
        self.peer_heartbeats[peer_id] = timestamp

    def check_partitions(self):
        now = time.monotonic()
        active_peers = set()
        lost_peers = set()

        for peer_id, last_time in self.peer_heartbeats.items():
            if now - last_time < self.heartbeat_timeout:
                active_peers.add(peer_id)
            else:
                lost_peers.add(peer_id)

        if lost_peers:
            for callback in self.partition_callbacks:
                callback(active_peers, lost_peers)

        return active_peers, lost_peers

class StateReconciliation:
    def __init__(self, robot_id):
        self.robot_id = robot_id
        self.local_log = []
        self.vector_clock = {}

    def record_event(self, event):
        self.vector_clock[self.robot_id] = self.vector_clock.get(self.robot_id, 0) + 1
        entry = {
            'event': event,
            'clock': self.vector_clock.copy(),
            'timestamp': time.monotonic()
        }
        self.local_log.append(entry)

    def merge_logs(self, remote_log, remote_id):
        for entry in remote_log:
            remote_clock = entry['clock']

            if not self._is_dominated(remote_clock):
                self.local_log.append(entry)

                for rid, count in remote_clock.items():
                    self.vector_clock[rid] = max(
                        self.vector_clock.get(rid, 0), count
                    )

    def _is_dominated(self, clock):
        for rid, count in clock.items():
            if count > self.vector_clock.get(rid, 0):
                return False
        return True

    •
