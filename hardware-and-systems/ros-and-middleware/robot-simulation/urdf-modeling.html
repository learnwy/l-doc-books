URDF 与机器人建模

核心概念

URDF（Unified Robot Description Format）是 ROS 生态系统中描述机器人模型的标准格式。它以 XML 语法定义机器人的运动链结构，包括连杆（Link）的几何形状、质量属性，以及关节（Joint）的类型、运动范围和动力学参数。URDF 是机器人仿真、运动规划、可视化和控制的基础——几乎所有 ROS 工具链都依赖 URDF 来理解机器人的物理结构。

URDF 的核心数据结构是一棵树：每个连杆通过关节连接到父连杆，从根连杆（通常是 base_link）到末端执行器形成一条运动链。URDF 不支持闭环结构（如并联机构），需要通过额外的约束或使用 SDF 格式来描述。

URDF 结构详解
一个完整的 URDF 模型由连杆和关节交替组成。每个连杆包含三个可选元素：visual（可视化几何体）、collision（碰撞几何体）和 inertial（惯性参数）。关节定义了父子连杆之间的运动关系。


import xml.etree.ElementTree as ET

class URDFBuilder:
    def __init__(self, robot_name):
        self.robot = ET.Element('robot', name=robot_name)

    def add_link(self, name, visual_geometry=None, collision_geometry=None,
                 mass=1.0, inertia=None, origin=None):
        link = ET.SubElement(self.robot, 'link', name=name)

        if visual_geometry:
            visual = ET.SubElement(link, 'visual')
            if origin:
                self._add_origin(visual, origin)
            geom = ET.SubElement(visual, 'geometry')
            self._add_geometry(geom, visual_geometry)

        if collision_geometry:
            collision = ET.SubElement(link, 'collision')
            if origin:
                self._add_origin(collision, origin)
            geom = ET.SubElement(collision, 'geometry')
            self._add_geometry(geom, collision_geometry)

        inertial = ET.SubElement(link, 'inertial')
        mass_elem = ET.SubElement(inertial, 'mass', value=str(mass))
        if inertia:
            inertia_elem = ET.SubElement(inertial, 'inertia',
                ixx=str(inertia[0]), ixy=str(inertia[1]), ixz=str(inertia[2]),
                iyy=str(inertia[3]), iyz=str(inertia[4]), izz=str(inertia[5])
            )
        else:
            default_i = mass * 0.01
            ET.SubElement(inertial, 'inertia',
                ixx=str(default_i), ixy='0', ixz='0',
                iyy=str(default_i), iyz='0', izz=str(default_i)
            )

    def add_joint(self, name, joint_type, parent, child,
                  origin=None, axis=None, limit=None, dynamics=None):
        joint = ET.SubElement(self.robot, 'joint', name=name, type=joint_type)
        ET.SubElement(joint, 'parent', link=parent)
        ET.SubElement(joint, 'child', link=child)

        if origin:
            self._add_origin(joint, origin)

        if axis:
            ET.SubElement(joint, 'axis', xyz=f'{axis[0]} {axis[1]} {axis[2]}')

        if limit:
            ET.SubElement(joint, 'limit',
                lower=str(limit.get('lower', 0)),
                upper=str(limit.get('upper', 0)),
                effort=str(limit.get('effort', 100)),
                velocity=str(limit.get('velocity', 1.0))
            )

        if dynamics:
            ET.SubElement(joint, 'dynamics',
                damping=str(dynamics.get('damping', 0.1)),
                friction=str(dynamics.get('friction', 0.0))
            )

    def add_transmission(self, name, joint_name, actuator_name,
                        trans_type='SimpleTransmission', reduction=1.0):
        trans = ET.SubElement(self.robot, 'transmission', name=name)
        type_elem = ET.SubElement(trans, 'type')
        type_elem.text = f'transmission_interface/{trans_type}'

        joint_elem = ET.SubElement(trans, 'joint', name=joint_name)
        hw = ET.SubElement(joint_elem, 'hardwareInterface')
        hw.text = 'hardware_interface/PositionJointInterface'

        actuator = ET.SubElement(trans, 'actuator', name=actuator_name)
        mech = ET.SubElement(actuator, 'mechanicalReduction')
        mech.text = str(reduction)

    def _add_origin(self, parent, origin):
        xyz = f'{origin.get("x", 0)} {origin.get("y", 0)} {origin.get("z", 0)}'
        rpy = f'{origin.get("roll", 0)} {origin.get("pitch", 0)} {origin.get("yaw", 0)}'
        ET.SubElement(parent, 'origin', xyz=xyz, rpy=rpy)

    def _add_geometry(self, parent, geometry):
        if geometry['type'] == 'box':
            ET.SubElement(parent, 'box', size=geometry['size'])
        elif geometry['type'] == 'cylinder':
            ET.SubElement(parent, 'cylinder',
                radius=str(geometry['radius']), length=str(geometry['length']))
        elif geometry['type'] == 'sphere':
            ET.SubElement(parent, 'sphere', radius=str(geometry['radius']))
        elif geometry['type'] == 'mesh':
            ET.SubElement(parent, 'mesh', filename=geometry['filename'])

    def to_string(self):
        return ET.tostring(self.robot, encoding='unicode')
惯性参数（Inertial）对物理仿真的准确性至关重要。质量和惯性张量直接影响机器人在仿真中的动力学行为。对于简单几何体，惯性张量可以通过公式计算；对于复杂形状，通常使用 CAD 软件导出。


Xacro 宏与参数化建模
Xacro（XML Macros）是 URDF 的扩展，提供了宏定义、参数化、条件判断和文件包含等功能。在实际项目中，直接编写纯 URDF 非常冗长且难以维护，Xacro 通过参数化和模块化大幅减少了重复代码。


class XacroProcessor:
    def __init__(self):
        self.properties = {}
        self.macros = {}

    def set_property(self, name, value):
        self.properties[name] = value

    def define_macro(self, name, params, body_template):
        self.macros[name] = {
            'params': params,
            'body': body_template
        }

    def expand_macro(self, name, **kwargs):
        macro = self.macros[name]
        body = macro['body']
        for param, value in kwargs.items():
            body = body.replace(f'${{{param}}}', str(value))
        for prop, value in self.properties.items():
            body = body.replace(f'${{{prop}}}', str(value))
        return body


LINK_MACRO_TEMPLATE = """
<link name="${prefix}_${name}">
  <visual>
    <geometry>
      <cylinder radius="${radius}" length="${length}"/>
    </geometry>
    <material name="${material}"/>
  </visual>
  <collision>
    <geometry>
      <cylinder radius="${radius}" length="${length}"/>
    </geometry>
  </collision>
  <inertial>
    <mass value="${mass}"/>
    <inertia ixx="${ixx}" ixy="0" ixz="0" iyy="${iyy}" iyz="0" izz="${izz}"/>
  </inertial>
</link>
"""

JOINT_MACRO_TEMPLATE = """
<joint name="${prefix}_${name}_joint" type="revolute">
  <parent link="${parent}"/>
  <child link="${prefix}_${name}"/>
  <origin xyz="${x} ${y} ${z}" rpy="${roll} ${pitch} ${yaw}"/>
  <axis xyz="${ax} ${ay} ${az}"/>
  <limit lower="${lower}" upper="${upper}" effort="${effort}" velocity="${velocity}"/>
  <dynamics damping="${damping}"/>
</joint>
"""
Xacro 的典型使用模式包括：为机械臂的每个关节定义参数化的连杆宏，为传感器安装定义可复用的安装宏，为多机器人场景定义带前缀的模型宏。通过 xacro:include 可以将大型机器人模型拆分为多个文件，各自维护不同的子系统。

Robot State Publisher 与 TF
Robot State Publisher（RSP）是将 URDF 模型转化为实时 TF 变换树的关键节点。它读取 URDF 模型和关节状态消息（/joint_states），计算所有连杆之间的变换关系并发布到 TF 树上，使其他节点能够查询任意两个坐标系之间的变换。


import numpy as np
from scipy.spatial.transform import Rotation

class SimpleRobotStatePublisher:
    def __init__(self, urdf_string):
        self.links = {}
        self.joints = {}
        self._parse_urdf(urdf_string)

    def _parse_urdf(self, urdf_string):
        import xml.etree.ElementTree as ET
        root = ET.fromstring(urdf_string)
        for link in root.findall('link'):
            self.links[link.get('name')] = {}
        for joint in root.findall('joint'):
            name = joint.get('name')
            jtype = joint.get('type')
            parent = joint.find('parent').get('link')
            child = joint.find('child').get('link')
            origin = joint.find('origin')
            axis_elem = joint.find('axis')

            xyz = [0, 0, 0]
            rpy = [0, 0, 0]
            if origin is not None:
                xyz = list(map(float, origin.get('xyz', '0 0 0').split()))
                rpy = list(map(float, origin.get('rpy', '0 0 0').split()))

            axis = [0, 0, 1]
            if axis_elem is not None:
                axis = list(map(float, axis_elem.get('xyz', '0 0 1').split()))

            self.joints[name] = {
                'type': jtype, 'parent': parent, 'child': child,
                'xyz': xyz, 'rpy': rpy, 'axis': axis
            }

    def compute_transforms(self, joint_states):
        transforms = {}
        for joint_name, joint_info in self.joints.items():
            xyz = np.array(joint_info['xyz'])
            rpy = joint_info['rpy']

            base_rot = Rotation.from_euler('xyz', rpy)

            if joint_info['type'] in ['revolute', 'continuous']:
                angle = joint_states.get(joint_name, 0.0)
                axis = np.array(joint_info['axis'])
                joint_rot = Rotation.from_rotvec(axis * angle)
                total_rot = base_rot * joint_rot
            elif joint_info['type'] == 'prismatic':
                displacement = joint_states.get(joint_name, 0.0)
                axis = np.array(joint_info['axis'])
                xyz = xyz + axis * displacement
                total_rot = base_rot
            else:
                total_rot = base_rot

            transforms[joint_info['child']] = {
                'parent': joint_info['parent'],
                'translation': xyz,
                'rotation': total_rot.as_quat()
            }
        return transforms
RSP 区分两种变换：固定关节（fixed joint）的变换通过 /tf_static 一次性发布（TRANSIENT_LOCAL QoS），活动关节的变换通过 /tf 持续发布。这种分离策略减少了 TF 树的更新量，提高了系统效率。


小结
    

  • URDF 是 ROS 生态中描述机器人模型的标准格式，定义运动链的几何、质量和运动学属性
    • 

  • 关节类型包括旋转（revolute/continuous）、平移（prismatic）和固定（fixed），覆盖大多数机构
    • 

  • Xacro 通过宏、参数和条件判断实现 URDF 的模块化和参数化，大幅减少代码重复
    • 

  • 惯性参数对仿真精度至关重要，必须根据实际零件质量分布准确计算
    • 

  • Robot State Publisher 将 URDF 和关节状态转化为实时 TF 变换树
    • 

  • Visual/Collision 几何体可以使用不同精度的模型（高精度可视化 + 简化碰撞体）
    • 

  • URDF 不支持闭环机构，需要通过 SDF 格式或额外约束来描述并联结构
    •
