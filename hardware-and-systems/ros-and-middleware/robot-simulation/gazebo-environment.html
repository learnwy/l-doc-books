Gazebo 仿真环境

核心概念

Gazebo 是机器人学中最广泛使用的物理仿真器，为机器人系统提供了完整的物理引擎、传感器仿真和环境建模能力。它允许开发者在虚拟环境中测试机器人算法，无需使用真实硬件即可进行开发、调试和验证。Gazebo 与 ROS2 的深度集成使其成为 ROS 生态系统中仿真的事实标准。

Gazebo 的最新版本（Gazebo Harmonic/Ionic，前称 Ignition Gazebo）对架构进行了全面重设计：基于实体组件系统（ECS）架构、模块化的插件系统、多物理引擎支持、现代化的渲染引擎。与旧版 Gazebo Classic 相比，新 Gazebo 更轻量、更灵活，且支持分布式仿真。

世界与模型创建
Gazebo 使用 SDF（Simulation Description Format）格式描述仿真世界和机器人模型。SDF 是 URDF 的超集，支持更丰富的物理属性、传感器配置和世界描述。在 ROS2 中，通常使用 URDF 描述机器人，通过 ros_gz 桥接包将其转换为 SDF 加载到 Gazebo 中。


import xml.etree.ElementTree as ET

class SDFWorldBuilder:
    def __init__(self, name='default_world'):
        self.world = ET.Element('sdf', version='1.9')
        self.world_elem = ET.SubElement(self.world, 'world', name=name)
        self._add_physics()
        self._add_scene()

    def _add_physics(self):
        physics = ET.SubElement(self.world_elem, 'physics', type='dart')
        max_step = ET.SubElement(physics, 'max_step_size')
        max_step.text = '0.001'
        real_time = ET.SubElement(physics, 'real_time_factor')
        real_time.text = '1.0'

    def _add_scene(self):
        scene = ET.SubElement(self.world_elem, 'scene')
        ambient = ET.SubElement(scene, 'ambient')
        ambient.text = '0.4 0.4 0.4 1.0'
        background = ET.SubElement(scene, 'background')
        background.text = '0.7 0.7 0.7 1.0'

    def add_ground_plane(self):
        model = ET.SubElement(self.world_elem, 'model', name='ground_plane')
        static = ET.SubElement(model, 'static')
        static.text = 'true'
        link = ET.SubElement(model, 'link', name='link')

        collision = ET.SubElement(link, 'collision', name='collision')
        geom = ET.SubElement(collision, 'geometry')
        plane = ET.SubElement(geom, 'plane')
        normal = ET.SubElement(plane, 'normal')
        normal.text = '0 0 1'
        size = ET.SubElement(plane, 'size')
        size.text = '100 100'

        visual = ET.SubElement(link, 'visual', name='visual')
        v_geom = ET.SubElement(visual, 'geometry')
        v_plane = ET.SubElement(v_geom, 'plane')
        v_normal = ET.SubElement(v_plane, 'normal')
        v_normal.text = '0 0 1'
        v_size = ET.SubElement(v_plane, 'size')
        v_size.text = '100 100'

    def add_box_obstacle(self, name, x, y, z, sx, sy, sz):
        model = ET.SubElement(self.world_elem, 'model', name=name)
        static = ET.SubElement(model, 'static')
        static.text = 'true'
        pose = ET.SubElement(model, 'pose')
        pose.text = f'{x} {y} {z} 0 0 0'

        link = ET.SubElement(model, 'link', name='link')
        collision = ET.SubElement(link, 'collision', name='collision')
        geom = ET.SubElement(collision, 'geometry')
        box = ET.SubElement(geom, 'box')
        size = ET.SubElement(box, 'size')
        size.text = f'{sx} {sy} {sz}'

        visual = ET.SubElement(link, 'visual', name='visual')
        v_geom = ET.SubElement(visual, 'geometry')
        v_box = ET.SubElement(v_geom, 'box')
        v_size = ET.SubElement(v_box, 'size')
        v_size.text = f'{sx} {sy} {sz}'

    def add_light(self, name='sun'):
        light = ET.SubElement(self.world_elem, 'light', name=name, type='directional')
        cast = ET.SubElement(light, 'cast_shadows')
        cast.text = 'true'
        direction = ET.SubElement(light, 'direction')
        direction.text = '-0.5 0.1 -0.9'

    def to_string(self):
        return ET.tostring(self.world, encoding='unicode')
Gazebo 模型库（Fuel）提供了大量预构建的模型资源，包括建筑、家具、车辆和机器人。开发者可以直接引用这些模型来快速构建仿真场景，极大地减少了环境建模的工作量。


传感器仿真
Gazebo 提供了丰富的传感器仿真插件，能够模拟相机、LiDAR、IMU、GPS、力/力矩传感器等多种传感器。仿真传感器输出的数据格式与真实传感器完全一致，使得算法代码无需修改即可在仿真和真实环境间切换。


class SensorConfigurator:
    def __init__(self, model_element):
        self.model = model_element

    def add_lidar(self, link_name, topic, update_rate=10, samples=360,
                  min_range=0.1, max_range=30.0, min_angle=-3.14159, max_angle=3.14159):
        link = self._get_or_create_link(link_name)
        sensor = ET.SubElement(link, 'sensor', name='lidar', type='gpu_lidar')

        rate = ET.SubElement(sensor, 'update_rate')
        rate.text = str(update_rate)

        lidar = ET.SubElement(sensor, 'lidar')
        scan = ET.SubElement(lidar, 'scan')
        horizontal = ET.SubElement(scan, 'horizontal')
        h_samples = ET.SubElement(horizontal, 'samples')
        h_samples.text = str(samples)
        h_min = ET.SubElement(horizontal, 'min_angle')
        h_min.text = str(min_angle)
        h_max = ET.SubElement(horizontal, 'max_angle')
        h_max.text = str(max_angle)

        rng = ET.SubElement(lidar, 'range')
        r_min = ET.SubElement(rng, 'min')
        r_min.text = str(min_range)
        r_max = ET.SubElement(rng, 'max')
        r_max.text = str(max_range)

        noise = ET.SubElement(lidar, 'noise')
        n_type = ET.SubElement(noise, 'type')
        n_type.text = 'gaussian'
        n_mean = ET.SubElement(noise, 'mean')
        n_mean.text = '0.0'
        n_stddev = ET.SubElement(noise, 'stddev')
        n_stddev.text = '0.01'

    def add_camera(self, link_name, topic, width=640, height=480, fps=30, fov=1.047):
        link = self._get_or_create_link(link_name)
        sensor = ET.SubElement(link, 'sensor', name='camera', type='camera')

        rate = ET.SubElement(sensor, 'update_rate')
        rate.text = str(fps)

        camera = ET.SubElement(sensor, 'camera')
        h_fov = ET.SubElement(camera, 'horizontal_fov')
        h_fov.text = str(fov)

        image = ET.SubElement(camera, 'image')
        w = ET.SubElement(image, 'width')
        w.text = str(width)
        h = ET.SubElement(image, 'height')
        h.text = str(height)
        fmt = ET.SubElement(image, 'format')
        fmt.text = 'R8G8B8'

    def add_imu(self, link_name, topic, update_rate=200):
        link = self._get_or_create_link(link_name)
        sensor = ET.SubElement(link, 'sensor', name='imu', type='imu')

        rate = ET.SubElement(sensor, 'update_rate')
        rate.text = str(update_rate)

        imu = ET.SubElement(sensor, 'imu')
        angular = ET.SubElement(imu, 'angular_velocity')
        for axis in ['x', 'y', 'z']:
            ax = ET.SubElement(angular, axis)
            noise = ET.SubElement(ax, 'noise', type='gaussian')
            mean = ET.SubElement(noise, 'mean')
            mean.text = '0.0'
            stddev = ET.SubElement(noise, 'stddev')
            stddev.text = '0.0001'

    def _get_or_create_link(self, name):
        import xml.etree.ElementTree as ET
        return ET.SubElement(self.model, 'link', name=name)

传感器噪声模型是提高仿真真实度的关键。Gazebo 支持高斯噪声、随机偏置漂移等噪声模型，参数可以根据真实传感器的数据手册进行配置。对于 IMU，还可以模拟加速度计偏置不稳定性和角速率随机游走。


ROS2 集成与 Sim-to-Real
ros_gz 包族是 Gazebo 与 ROS2 之间的桥梁。ros_gz_bridge 负责在 Gazebo 话题和 ROS2 话题之间进行消息类型转换和数据传递。ros_gz_sim 提供了从 ROS2 控制 Gazebo 仿真的接口（暂停、重置、生成模型等）。


from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, IncludeLaunchDescription
from launch.substitutions import LaunchConfiguration, PathJoinSubstitution
from launch_ros.actions import Node
from launch_ros.substitutions import FindPackageShare

def generate_launch_description():
    world_file = DeclareLaunchArgument(
        'world', default_value='warehouse.sdf'
    )

    gazebo = IncludeLaunchDescription(
        PathJoinSubstitution([
            FindPackageShare('ros_gz_sim'), 'launch', 'gz_sim.launch.py'
        ]),
        launch_arguments={
            'gz_args': ['-r ', LaunchConfiguration('world')]
        }.items()
    )

    spawn_robot = Node(
        package='ros_gz_sim',
        executable='create',
        arguments=[
            '-name', 'robot',
            '-topic', 'robot_description',
            '-x', '0', '-y', '0', '-z', '0.1'
        ],
        output='screen'
    )

    bridge = Node(
        package='ros_gz_bridge',
        executable='parameter_bridge',
        arguments=[
            '/scan@sensor_msgs/msg/LaserScan@gz.msgs.LaserScan',
            '/imu@sensor_msgs/msg/Imu@gz.msgs.IMU',
            '/camera/image@sensor_msgs/msg/Image@gz.msgs.Image',
            '/cmd_vel@geometry_msgs/msg/Twist@gz.msgs.Twist',
            '/odom@nav_msgs/msg/Odometry@gz.msgs.Odometry',
            '/clock@rosgraph_msgs/msg/Clock@gz.msgs.Clock',
        ],
        output='screen'
    )

    return LaunchDescription([
        world_file,
        gazebo,
        spawn_robot,
        bridge,
    ])

小结
    

  • Gazebo 是 ROS2 生态中最成熟的物理仿真器，提供物理引擎、传感器仿真和环境建模
    • 

  • 新 Gazebo（Harmonic）采用 ECS 架构和模块化设计，性能和扩展性显著提升
    • 

  • SDF 格式是仿真世界描述的标准，URDF 通过桥接包转换为 SDF 加载
    • 

  • 传感器仿真支持噪声模型配置，使仿真数据接近真实传感器输出
    • 

  • ros_gz_bridge 实现 Gazebo 和 ROS2 之间的无缝消息桥接
    • 

  • Sim-to-Real 迁移通过域随机化和系统辨识减小仿真与真实的差距
    • 

  • 仿真环境极大加速了机器人算法的开发和验证周期
