ROS2 服务与动作

核心概念

ROS2 中的通信模型除了话题（Topic）的发布-订阅模式外，还提供了服务（Service）和动作（Action）两种请求-响应模式。服务适用于短时间内可完成的同步请求，如查询参数、触发行为；动作适用于长时间运行的异步任务，如导航到目标点、执行抓取序列。三种通信模式构成了 ROS2 完整的节点间交互体系。

服务使用请求-响应模式：客户端发送请求，服务端处理后返回响应。动作在服务基础上增加了目标（Goal）、反馈（Feedback）和结果（Result）三要素，支持取消和进度报告。

服务通信
ROS2 服务基于 DDS 的请求-响应模式实现。每个服务由一个 .srv 文件定义接口，包含请求和响应两部分，用 --- 分隔。服务端注册后等待请求，客户端发送请求后阻塞等待或异步等待响应。


服务在 ROS2 中的底层实现使用两个隐藏的话题：请求话题和响应话题。服务端同时订阅请求话题和发布响应话题，客户端则相反。RMW（ROS Middleware）层确保请求和响应的配对关系。


import rclpy
from rclpy.node import Node
from example_interfaces.srv import AddTwoInts

class AdditionServer(Node):
    def __init__(self):
        super().__init__('addition_server')
        self.srv = self.create_service(
            AddTwoInts, 'add_two_ints', self.add_callback
        )
        self.get_logger().info('Addition service ready')

    def add_callback(self, request, response):
        response.sum = request.a + request.b
        self.get_logger().info(f'{request.a} + {request.b} = {response.sum}')
        return response


class AdditionClient(Node):
    def __init__(self):
        super().__init__('addition_client')
        self.client = self.create_client(AddTwoInts, 'add_two_ints')
        while not self.client.wait_for_service(timeout_sec=1.0):
            self.get_logger().info('Waiting for service...')

    def send_request(self, a, b):
        request = AddTwoInts.Request()
        request.a = a
        request.b = b
        future = self.client.call_async(request)
        return future


class RobotCommandService(Node):
    def __init__(self):
        super().__init__('robot_command_service')
        self.services = {}
        self._register_services()

    def _register_services(self):
        from std_srvs.srv import Trigger, SetBool

        self.services['emergency_stop'] = self.create_service(
            Trigger, 'emergency_stop', self._emergency_stop_cb
        )
        self.services['enable_motor'] = self.create_service(
            SetBool, 'enable_motor', self._enable_motor_cb
        )

    def _emergency_stop_cb(self, request, response):
        self.get_logger().warn('Emergency stop triggered!')
        response.success = True
        response.message = 'All motors stopped'
        return response

    def _enable_motor_cb(self, request, response):
        enabled = request.data
        self.get_logger().info(f'Motor {"enabled" if enabled else "disabled"}')
        response.success = True
        response.message = f'Motor {"on" if enabled else "off"}'
        return response
服务通信的重要注意事项：服务回调在单线程执行器中默认是阻塞的，长时间的服务处理会阻塞节点的其他回调。对于耗时操作，应使用多线程执行器（MultiThreadedExecutor）或将服务转换为动作。服务不支持多个并发请求的排队——如果服务端正在处理一个请求，后续请求会被缓冲但不保证顺序。


动作通信
动作（Action）是 ROS2 中处理长时间运行任务的标准机制。动作由 .action 文件定义，包含三部分：Goal（客户端发送的目标）、Result（任务完成后的结果）和 Feedback（执行过程中的进度反馈）。


动作的底层由三个服务和两个话题组成：发送目标服务、取消目标服务、获取结果服务、反馈话题和状态话题。这种组合使得动作同时支持请求-响应语义和流式反馈。


import rclpy
from rclpy.node import Node
from rclpy.action import ActionServer, ActionClient, CancelResponse, GoalResponse
from nav2_msgs.action import NavigateToPose
import time

class NavigationActionServer(Node):
    def __init__(self):
        super().__init__('navigation_server')
        self._action_server = ActionServer(
            self,
            NavigateToPose,
            'navigate_to_pose',
            execute_callback=self.execute_callback,
            goal_callback=self.goal_callback,
            cancel_callback=self.cancel_callback
        )

    def goal_callback(self, goal_request):
        self.get_logger().info(
            f'Received goal: ({goal_request.pose.pose.position.x}, '
            f'{goal_request.pose.pose.position.y})'
        )
        return GoalResponse.ACCEPT

    def cancel_callback(self, goal_handle):
        self.get_logger().info('Received cancel request')
        return CancelResponse.ACCEPT

    async def execute_callback(self, goal_handle):
        self.get_logger().info('Executing navigation...')
        feedback_msg = NavigateToPose.Feedback()

        target_x = goal_handle.request.pose.pose.position.x
        target_y = goal_handle.request.pose.pose.position.y

        for step in range(100):
            if goal_handle.is_cancel_requested:
                goal_handle.canceled()
                self.get_logger().info('Navigation canceled')
                return NavigateToPose.Result()

            progress = (step + 1) / 100.0
            feedback_msg.distance_remaining = (1.0 - progress) * 10.0
            goal_handle.publish_feedback(feedback_msg)
            time.sleep(0.1)

        goal_handle.succeed()
        result = NavigateToPose.Result()
        self.get_logger().info('Navigation completed')
        return result


class NavigationClient(Node):
    def __init__(self):
        super().__init__('navigation_client')
        self._action_client = ActionClient(
            self, NavigateToPose, 'navigate_to_pose'
        )

    def send_goal(self, x, y):
        self._action_client.wait_for_server()

        goal_msg = NavigateToPose.Goal()
        goal_msg.pose.pose.position.x = x
        goal_msg.pose.pose.position.y = y

        send_goal_future = self._action_client.send_goal_async(
            goal_msg, feedback_callback=self.feedback_callback
        )
        send_goal_future.add_done_callback(self.goal_response_callback)

    def goal_response_callback(self, future):
        goal_handle = future.result()
        if not goal_handle.accepted:
            self.get_logger().info('Goal rejected')
            return
        self.get_logger().info('Goal accepted')
        result_future = goal_handle.get_result_async()
        result_future.add_done_callback(self.result_callback)

    def feedback_callback(self, feedback_msg):
        distance = feedback_msg.feedback.distance_remaining
        self.get_logger().info(f'Distance remaining: {distance:.1f}m')

    def result_callback(self, future):
        self.get_logger().info('Navigation finished')
动作服务端的状态机管理了目标的完整生命周期：ACCEPTED → EXECUTING → (SUCCEEDED | CANCELED | ABORTED)。多个目标可以同时存在，服务端通过策略（如抢占式、排队式）决定如何处理新目标与正在执行的目标之间的冲突。Nav2 导航系统默认使用抢占策略——新目标会取消当前正在执行的目标。


自定义接口与最佳实践
ROS2 支持自定义消息（msg）、服务（srv）和动作（action）接口。接口定义文件使用 IDL（Interface Definition Language）语法，编译后生成 Python 和 C++ 的类型绑定代码。


import rclpy
from rclpy.node import Node
from rclpy.callback_group import MutuallyExclusiveCallbackGroup, ReentrantCallbackGroup
from rclpy.executors import MultiThreadedExecutor

class RobotController(Node):
    def __init__(self):
        super().__init__('robot_controller')

        self.service_group = MutuallyExclusiveCallbackGroup()
        self.action_group = ReentrantCallbackGroup()
        self.timer_group = MutuallyExclusiveCallbackGroup()

        from std_srvs.srv import Trigger
        self.create_service(
            Trigger, 'get_status', self._get_status_cb,
            callback_group=self.service_group
        )

        self.create_timer(
            0.1, self._control_loop,
            callback_group=self.timer_group
        )

        self.state = {'mode': 'idle', 'battery': 100.0, 'position': [0.0, 0.0]}

    def _get_status_cb(self, request, response):
        response.success = True
        response.message = f"Mode: {self.state['mode']}, Battery: {self.state['battery']:.1f}%"
        return response

    def _control_loop(self):
        self.state['battery'] -= 0.01
        if self.state['battery'] < 10.0:
            self.get_logger().warn('Low battery!')


def main():
    rclpy.init()
    node = RobotController()
    executor = MultiThreadedExecutor(num_threads=4)
    executor.add_node(node)
    executor.spin()
    rclpy.shutdown()
回调组（Callback Group）是 ROS2 中控制并发执行的核心机制。MutuallyExclusiveCallbackGroup 保证组内回调互斥执行（适合共享资源），ReentrantCallbackGroup 允许组内回调并发执行（适合独立任务）。正确配置回调组对于避免死锁和资源竞争至关重要。

小结
    

  • ROS2 提供话题、服务和动作三种通信模式，覆盖不同的交互需求
    • 

  • 服务适用于短时同步请求，动作适用于长时异步任务并支持进度反馈和取消
    • 

  • 动作底层由三个服务和两个话题组合实现，提供完整的目标生命周期管理
    • 

  • 回调组和多线程执行器是控制 ROS2 并发行为的核心机制
    • 

  • 自定义接口通过 IDL 定义，自动生成 Python 和 C++ 绑定
    • 

  • 在生产系统中，正确的通信模式选择和并发配置对系统稳定性至关重要
    • 

  • Nav2 导航框架大量使用动作接口实现导航、路径规划等长时任务
    •
