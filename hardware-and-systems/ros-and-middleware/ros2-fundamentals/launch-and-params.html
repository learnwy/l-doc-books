ROS2 启动系统与参数管理

核心概念

ROS2 的启动系统（Launch System）负责以声明式方式管理多节点应用的启动、配置和编排。一个完整的机器人系统通常包含数十个节点，手动逐一启动既低效又容易出错。启动系统通过 Python 或 XML 编写的启动文件（Launch File），实现一键部署整个机器人软件栈。

参数管理（Parameter Management）与启动系统紧密配合，为节点提供运行时配置能力。ROS2 参数支持动态修改，允许在不重启节点的情况下调整行为——这对于机器人调试和在线优化至关重要。

Launch 文件编写
ROS2 推荐使用 Python 编写 Launch 文件，相比 ROS1 的 XML 格式，Python 提供了完整的编程能力，可以实现条件启动、环境变量读取和复杂的节点编排逻辑。


from launch import LaunchDescription
from launch.actions import (
    DeclareLaunchArgument, GroupAction, IncludeLaunchDescription,
    TimerAction, RegisterEventHandler
)
from launch.conditions import IfCondition, UnlessCondition
from launch.substitutions import (
    LaunchConfiguration, PathJoinSubstitution, PythonExpression
)
from launch.event_handlers import OnProcessStart
from launch_ros.actions import Node, PushRosNamespace
from launch_ros.substitutions import FindPackageShare

def generate_launch_description():
    use_sim = DeclareLaunchArgument(
        'use_sim', default_value='false',
        description='是否使用仿真模式'
    )

    robot_name = DeclareLaunchArgument(
        'robot_name', default_value='my_robot',
        description='机器人名称'
    )

    params_file = PathJoinSubstitution([
        FindPackageShare('my_robot_bringup'),
        'config', 'robot_params.yaml'
    ])

    sensor_driver = Node(
        package='sensor_driver',
        executable='lidar_node',
        name='lidar',
        parameters=[params_file],
        remappings=[('/scan_raw', '/scan')],
        condition=UnlessCondition(LaunchConfiguration('use_sim'))
    )

    slam_node = Node(
        package='slam_toolbox',
        executable='async_slam_toolbox_node',
        name='slam',
        parameters=[
            params_file,
            {'use_sim_time': LaunchConfiguration('use_sim')}
        ],
        output='screen'
    )

    navigation_group = GroupAction([
        PushRosNamespace(LaunchConfiguration('robot_name')),
        Node(
            package='nav2_controller',
            executable='controller_server',
            name='controller_server',
            parameters=[params_file]
        ),
        Node(
            package='nav2_planner',
            executable='planner_server',
            name='planner_server',
            parameters=[params_file]
        ),
    ])

    delayed_nav = TimerAction(
        period=5.0,
        actions=[navigation_group]
    )

    return LaunchDescription([
        use_sim,
        robot_name,
        sensor_driver,
        slam_node,
        delayed_nav,
    ])

from rclpy.node import Node
from rcl_interfaces.msg import ParameterDescriptor, FloatingPointRange

class ConfigurableNode(Node):
    def __init__(self):
        super().__init__('configurable_node')

        speed_descriptor = ParameterDescriptor(
            description='最大线速度 (m/s)',
            floating_point_range=[FloatingPointRange(
                from_value=0.0,
                to_value=2.0,
                step=0.1
            )]
        )
        self.declare_parameter('max_speed', 0.5, speed_descriptor)

        self.add_on_set_parameters_callback(self.parameter_callback)

    def parameter_callback(self, params):
        from rcl_interfaces.msg import SetParametersResult

        for param in params:
            self.get_logger().info(f'参数变更: {param.name} = {param.value}')

            if param.name == 'max_speed' and param.value < 0:
                return SetParametersResult(
                    successful=False,
                    reason='速度不能为负值'
                )

        return SetParametersResult(successful=True)
生命周期节点
生命周期节点（Lifecycle Node）是 ROS2 的重要特性，为节点提供了标准化的状态机管理。生命周期节点有四个主要状态：Unconfigured、Inactive、Active、Finalized，通过转换函数在状态间切换。


import rclpy
from rclpy.lifecycle import Node as LifecycleNode, TransitionCallbackReturn
from rclpy.lifecycle import State

class ManagedSensorNode(LifecycleNode):
    def __init__(self):
        super().__init__('managed_sensor')
        self.sensor_handle = None
        self.publisher = None

    def on_configure(self, state: State) -> TransitionCallbackReturn:
        self.get_logger().info('配置中...')

        self.declare_parameter('device_port', '/dev/ttyUSB0')
        self.declare_parameter('baud_rate', 115200)

        port = self.get_parameter('device_port').value
        baud = self.get_parameter('baud_rate').value

        try:
            self.sensor_handle = self._open_sensor(port, baud)
            self.publisher = self.create_publisher(LaserScan, '/scan', 10)
            return TransitionCallbackReturn.SUCCESS
        except Exception as e:
            self.get_logger().error(f'传感器配置失败: {e}')
            return TransitionCallbackReturn.FAILURE

    def on_activate(self, state: State) -> TransitionCallbackReturn:
        self.get_logger().info('激活中...')
        self.timer = self.create_timer(0.1, self.publish_data)
        return TransitionCallbackReturn.SUCCESS

    def on_deactivate(self, state: State) -> TransitionCallbackReturn:
        self.get_logger().info('停用中...')
        self.destroy_timer(self.timer)
        return TransitionCallbackReturn.SUCCESS

    def on_cleanup(self, state: State) -> TransitionCallbackReturn:
        self.get_logger().info('清理中...')
        if self.sensor_handle:
            self._close_sensor(self.sensor_handle)
        self.destroy_publisher(self.publisher)
        return TransitionCallbackReturn.SUCCESS

    def on_shutdown(self, state: State) -> TransitionCallbackReturn:
        self.get_logger().info('关闭中...')
        return TransitionCallbackReturn.SUCCESS

    def publish_data(self):
        if self.sensor_handle:
            data = self._read_sensor(self.sensor_handle)
            self.publisher.publish(data)

    def _open_sensor(self, port, baud):
        return {'port': port, 'baud': baud}

    def _close_sensor(self, handle):
        pass

    def _read_sensor(self, handle):
        return LaserScan()

多机器人启动编排
在多机器人系统中，Launch 系统需要管理命名空间隔离、话题重映射和参数差异化配置：


from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, OpaqueFunction
from launch_ros.actions import Node, PushRosNamespace

def launch_robot(context, robot_id):
    ns = f'robot_{robot_id}'

    return [
        PushRosNamespace(ns),
        Node(
            package='robot_driver',
            executable='base_driver',
            name='base_driver',
            parameters=[{
                'robot_id': int(robot_id),
                'initial_x': float(robot_id) * 2.0,
                'initial_y': 0.0,
            }],
            remappings=[
                ('/cmd_vel', f'/{ns}/cmd_vel'),
                ('/odom', f'/{ns}/odom'),
            ]
        ),
        Node(
            package='robot_navigation',
            executable='navigator',
            name='navigator',
            parameters=[{
                'robot_id': int(robot_id),
                'map_topic': '/shared_map',
            }]
        ),
    ]

def generate_launch_description():
    num_robots = DeclareLaunchArgument('num_robots', default_value='3')

    def launch_all_robots(context):
        n = int(context.launch_configurations['num_robots'])
        actions = []
        for i in range(n):
            actions.extend(launch_robot(context, str(i)))
        return actions

    return LaunchDescription([
        num_robots,
        OpaqueFunction(function=launch_all_robots),
    ])
