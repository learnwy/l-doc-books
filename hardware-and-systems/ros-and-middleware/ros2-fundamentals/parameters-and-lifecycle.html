ROS2 参数与生命周期

核心概念

ROS2 的参数系统（Parameters）和生命周期管理（Lifecycle）是构建可靠机器人系统的两大基础设施。参数系统提供了运行时配置的统一机制，支持动态修改、类型检查和参数回调；生命周期节点（Managed Node）定义了节点的状态转移协议，使系统能够以确定性的方式初始化、配置和关闭节点。

参数系统取代了 ROS1 中的 rosparam 和 dynamic_reconfigure，提供了更统一、更安全的配置方案。生命周期节点解决了 ROS1 中节点启动顺序不可控的问题，特别适用于需要协调初始化的复杂系统。

参数声明与管理
ROS2 中的参数必须先声明再使用，这确保了参数的可发现性和类型安全。参数支持的类型包括 bool、int、double、string 以及对应的数组类型。声明时可以指定默认值、描述和取值范围约束。


import rclpy
from rclpy.node import Node
from rcl_interfaces.msg import ParameterDescriptor, FloatingPointRange, IntegerRange

class ConfigurableController(Node):
    def __init__(self):
        super().__init__('configurable_controller')
        self._declare_parameters()
        self._setup_parameter_callback()

    def _declare_parameters(self):
        self.declare_parameter('control_frequency', 50.0,
            ParameterDescriptor(
                description='Control loop frequency in Hz',
                floating_point_range=[FloatingPointRange(
                    from_value=1.0, to_value=1000.0, step=1.0
                )]
            )
        )

        self.declare_parameter('max_velocity', 1.0,
            ParameterDescriptor(description='Maximum linear velocity (m/s)')
        )

        self.declare_parameter('pid_gains.kp', 1.0)
        self.declare_parameter('pid_gains.ki', 0.1)
        self.declare_parameter('pid_gains.kd', 0.01)
        self.declare_parameter('enabled', True)
        self.declare_parameter('controller_type', 'pid')

    def _setup_parameter_callback(self):
        self.add_on_set_parameters_callback(self._on_parameter_change)

    def _on_parameter_change(self, params):
        from rcl_interfaces.msg import SetParametersResult
        for param in params:
            if param.name == 'max_velocity' and param.value < 0:
                return SetParametersResult(
                    successful=False,
                    reason='max_velocity must be non-negative'
                )

            if param.name == 'controller_type':
                valid_types = ['pid', 'mpc', 'lqr']
                if param.value not in valid_types:
                    return SetParametersResult(
                        successful=False,
                        reason=f'controller_type must be one of {valid_types}'
                    )

        self.get_logger().info(f'Parameters updated: {[p.name for p in params]}')
        return SetParametersResult(successful=True)

    def get_config(self):
        return {
            'frequency': self.get_parameter('control_frequency').value,
            'max_vel': self.get_parameter('max_velocity').value,
            'kp': self.get_parameter('pid_gains.kp').value,
            'ki': self.get_parameter('pid_gains.ki').value,
            'kd': self.get_parameter('pid_gains.kd').value,
            'enabled': self.get_parameter('enabled').value,
            'type': self.get_parameter('controller_type').value,
        }
YAML 配置文件是 ROS2 参数的主要外部来源。配置文件结构遵循 <node_name>: ros__parameters: 的格式。Launch 文件通过 parameters 参数将配置文件传递给节点。


参数回调（Parameter Callback）是参数动态更新的核心机制。回调函数在参数被修改前调用，可以验证新值的合法性并选择接受或拒绝。这种机制使得在线调参（如调整 PID 增益）变得安全可控。


生命周期节点
生命周期节点（Lifecycle Node）定义了四种主要状态和七种转移，使节点的初始化和关闭过程可控、可编排。


四种主要状态：Unconfigured（刚创建，未配置）、Inactive（已配置但未激活，不处理数据）、Active（正常运行）、Finalized（已终止）。


import rclpy
from rclpy.lifecycle import LifecycleNode, TransitionCallbackReturn, LifecycleState
from rclpy.lifecycle import Publisher as LifecyclePublisher

class LifecycleSensorDriver(LifecycleNode):
    def __init__(self):
        super().__init__('sensor_driver')
        self.sensor = None
        self.publisher_ = None
        self.timer_ = None

    def on_configure(self, state: LifecycleState) -> TransitionCallbackReturn:
        self.get_logger().info('Configuring...')

        self.declare_parameter('port', '/dev/ttyUSB0')
        self.declare_parameter('baud_rate', 115200)
        self.declare_parameter('publish_rate', 10.0)

        port = self.get_parameter('port').value
        baud = self.get_parameter('baud_rate').value

        try:
            self.sensor = self._init_sensor(port, baud)
        except Exception as e:
            self.get_logger().error(f'Failed to initialize sensor: {e}')
            return TransitionCallbackReturn.FAILURE

        from sensor_msgs.msg import Imu
        self.publisher_ = self.create_lifecycle_publisher(Imu, 'imu/data', 10)

        self.get_logger().info('Configuration complete')
        return TransitionCallbackReturn.SUCCESS

    def on_activate(self, state: LifecycleState) -> TransitionCallbackReturn:
        self.get_logger().info('Activating...')

        rate = self.get_parameter('publish_rate').value
        self.timer_ = self.create_timer(1.0 / rate, self._publish_data)

        self.get_logger().info('Active and publishing')
        return super().on_activate(state)

    def on_deactivate(self, state: LifecycleState) -> TransitionCallbackReturn:
        self.get_logger().info('Deactivating...')

        if self.timer_:
            self.timer_.cancel()
            self.timer_ = None

        return super().on_deactivate(state)

    def on_cleanup(self, state: LifecycleState) -> TransitionCallbackReturn:
        self.get_logger().info('Cleaning up...')

        if self.sensor:
            self.sensor = None
        self.publisher_ = None

        return TransitionCallbackReturn.SUCCESS

    def on_shutdown(self, state: LifecycleState) -> TransitionCallbackReturn:
        self.get_logger().info('Shutting down...')
        if self.sensor:
            self.sensor = None
        return TransitionCallbackReturn.SUCCESS

    def _init_sensor(self, port, baud):
        return {'port': port, 'baud': baud, 'connected': True}

    def _publish_data(self):
        if self.publisher_ and self.publisher_.is_activated:
            from sensor_msgs.msg import Imu
            msg = Imu()
            self.publisher_.publish(msg)
生命周期节点的关键特性是生命周期发布者（Lifecycle Publisher）。与普通发布者不同，生命周期发布者只在节点处于 Active 状态时才实际发送消息，在 Inactive 状态下消息会被丢弃。这防止了未完全初始化的节点向系统发送错误数据。


Launch 系统与系统编排
ROS2 的 Launch 系统使用 Python 脚本实现，相比 ROS1 的 XML 格式更加灵活强大。Launch 文件可以描述节点的启动顺序、参数配置、条件判断和事件响应，是系统级编排的核心工具。


from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, GroupAction, TimerAction
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node, LifecycleNode
from launch_ros.event_handlers import OnStateTransition
from launch.event_handlers import OnProcessStart
from lifecycle_msgs.msg import Transition

def generate_launch_description():
    use_sim = DeclareLaunchArgument('use_sim', default_value='false')
    config_file = DeclareLaunchArgument(
        'config', default_value='config/robot.yaml'
    )

    sensor_node = LifecycleNode(
        package='robot_drivers',
        executable='sensor_driver',
        name='imu_driver',
        namespace='sensors',
        parameters=[LaunchConfiguration('config')],
        output='screen'
    )

    controller_node = Node(
        package='robot_control',
        executable='controller',
        name='motion_controller',
        parameters=[{
            'use_sim_time': LaunchConfiguration('use_sim'),
            'control_frequency': 100.0,
        }],
        remappings=[
            ('cmd_vel', '/robot/cmd_vel'),
            ('odom', '/robot/odom'),
        ],
    )

    configure_event = TimerAction(
        period=2.0,
        actions=[
            Node(
                package='lifecycle_utils',
                executable='lifecycle_manager',
                arguments=['--configure', 'imu_driver'],
            )
        ]
    )

    return LaunchDescription([
        use_sim,
        config_file,
        sensor_node,
        controller_node,
        configure_event,
    ])
Launch 系统的事件模型支持复杂的启动编排：节点 A 启动完成后再启动节点 B，传感器驱动配置成功后自动激活控制器，任何节点崩溃时自动重启。这些功能对于生产级机器人系统至关重要。

小结
    

  • ROS2 参数系统提供强类型、可验证、支持运行时修改的统一配置方案
    • 

  • 参数回调机制允许在参数修改前验证合法性，确保在线调参安全
    • 

  • 生命周期节点通过四状态七转移模型，使节点初始化和关闭过程可控可编排
    • 

  • 生命周期发布者仅在 Active 状态下发送消息，防止未初始化数据污染系统
    • 

  • Launch 系统使用 Python 脚本实现灵活的系统级编排，支持条件启动和事件响应
    • 

  • 参数系统与生命周期的结合使得配置在不同状态下有不同行为成为可能
    • 

  • 这些基础设施是构建生产级、可维护的机器人系统的关键
