ROS2 坐标变换（TF2）

核心概念

TF2 是 ROS2 中管理坐标系变换的核心库。在机器人系统中，不同组件（传感器、执行器、末端工具）各自拥有局部坐标系，TF2 负责维护这些坐标系之间的变换关系树（Transform Tree），使得任意两个坐标系之间的相对位姿可以随时查询。

一个典型的移动机器人可能有 10-30 个坐标系：世界坐标系、里程计坐标系、底盘坐标系、各传感器坐标系等。TF2 将这些坐标系组织成一棵树，每条边表示父子坐标系间的时变变换。

变换发布与监听
TF2 采用发布-监听模式，变换数据通过 /tf 和 /tf_static 两个话题传输。动态变换需要持续发布（通常与传感器同频），静态变换只需发布一次。


import rclpy
from rclpy.node import Node
import tf2_ros
from geometry_msgs.msg import TransformStamped
import numpy as np
from tf_transformations import quaternion_from_euler

class TFPublisher(Node):
    def __init__(self):
        super().__init__('tf_publisher')

        self.static_broadcaster = tf2_ros.StaticTransformBroadcaster(self)
        self.publish_static_transforms()

        self.dynamic_broadcaster = tf2_ros.TransformBroadcaster(self)
        self.timer = self.create_timer(0.02, self.publish_dynamic_tf)

        self.odom_x = 0.0
        self.odom_y = 0.0
        self.odom_theta = 0.0

    def publish_static_transforms(self):
        transforms = []

        laser_tf = TransformStamped()
        laser_tf.header.stamp = self.get_clock().now().to_msg()
        laser_tf.header.frame_id = 'base_link'
        laser_tf.child_frame_id = 'laser_frame'
        laser_tf.transform.translation.x = 0.2
        laser_tf.transform.translation.z = 0.15
        q = quaternion_from_euler(0, 0, 0)
        laser_tf.transform.rotation.x = q[0]
        laser_tf.transform.rotation.y = q[1]
        laser_tf.transform.rotation.z = q[2]
        laser_tf.transform.rotation.w = q[3]
        transforms.append(laser_tf)

        camera_tf = TransformStamped()
        camera_tf.header.stamp = self.get_clock().now().to_msg()
        camera_tf.header.frame_id = 'base_link'
        camera_tf.child_frame_id = 'camera_link'
        camera_tf.transform.translation.x = 0.15
        camera_tf.transform.translation.z = 0.3
        q = quaternion_from_euler(0, 0.1, 0)
        camera_tf.transform.rotation.x = q[0]
        camera_tf.transform.rotation.y = q[1]
        camera_tf.transform.rotation.z = q[2]
        camera_tf.transform.rotation.w = q[3]
        transforms.append(camera_tf)

        self.static_broadcaster.sendTransform(transforms)

    def publish_dynamic_tf(self):
        t = TransformStamped()
        t.header.stamp = self.get_clock().now().to_msg()
        t.header.frame_id = 'odom'
        t.child_frame_id = 'base_link'
        t.transform.translation.x = self.odom_x
        t.transform.translation.y = self.odom_y
        q = quaternion_from_euler(0, 0, self.odom_theta)
        t.transform.rotation.x = q[0]
        t.transform.rotation.y = q[1]
        t.transform.rotation.z = q[2]
        t.transform.rotation.w = q[3]

        self.dynamic_broadcaster.sendTransform(t)

class TFListener(Node):
    def __init__(self):
        super().__init__('tf_listener')

        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)

        self.timer = self.create_timer(0.1, self.query_transform)

    def query_transform(self):
        try:
            transform = self.tf_buffer.lookup_transform(
                'map', 'laser_frame',
                rclpy.time.Time(),
                timeout=rclpy.duration.Duration(seconds=1.0)
            )

            pos = transform.transform.translation
            self.get_logger().info(
                f'laser 在 map 中: ({pos.x:.3f}, {pos.y:.3f}, {pos.z:.3f})'
            )
        except tf2_ros.LookupException:
            self.get_logger().warn('变换未找到')
        except tf2_ros.ExtrapolationException:
            self.get_logger().warn('变换外推错误')
URDF 与 Robot State Publisher
URDF（Unified Robot Description Format）是描述机器人几何结构和物理属性的 XML 格式。Robot State Publisher 根据 URDF 模型和关节状态，自动发布所有连杆间的 TF 变换。


<?xml version="1.0"?>
<robot name="mobile_manipulator" xmlns:xacro="http://ros.org/wiki/xacro">

  <xacro:property name="base_width" value="0.4"/>
  <xacro:property name="base_length" value="0.6"/>
  <xacro:property name="wheel_radius" value="0.08"/>

  <link name="base_link">
    <visual>
      <geometry>
        <box size="${base_length} ${base_width} 0.15"/>
      </geometry>
    </visual>
    <collision>
      <geometry>
        <box size="${base_length} ${base_width} 0.15"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="20.0"/>
      <inertia ixx="0.5" ixy="0" ixz="0" iyy="0.8" iyz="0" izz="0.6"/>
    </inertial>
  </link>

  <xacro:macro name="wheel" params="prefix x_offset y_offset">
    <link name="${prefix}_wheel">
      <visual>
        <geometry><cylinder radius="${wheel_radius}" length="0.04"/></geometry>
        <origin rpy="${pi/2} 0 0"/>
      </visual>
      <inertial>
        <mass value="0.5"/>
        <inertia ixx="0.001" ixy="0" ixz="0" iyy="0.001" iyz="0" izz="0.001"/>
      </inertial>
    </link>
    <joint name="${prefix}_wheel_joint" type="continuous">
      <parent link="base_link"/>
      <child link="${prefix}_wheel"/>
      <origin xyz="${x_offset} ${y_offset} -0.05" rpy="0 0 0"/>
      <axis xyz="0 1 0"/>
    </joint>
  </xacro:macro>

  <xacro:wheel prefix="front_left" x_offset="0.2" y_offset="0.22"/>
  <xacro:wheel prefix="front_right" x_offset="0.2" y_offset="-0.22"/>
  <xacro:wheel prefix="rear_left" x_offset="-0.2" y_offset="0.22"/>
  <xacro:wheel prefix="rear_right" x_offset="-0.2" y_offset="-0.22"/>

</robot>
from launch import LaunchDescription
from launch_ros.actions import Node
from launch.substitutions import Command
from launch_ros.substitutions import FindPackageShare

def generate_launch_description():
    urdf_file = FindPackageShare('my_robot_description') + '/urdf/robot.urdf.xacro'

    robot_description = Command(['xacro ', urdf_file])

    robot_state_publisher = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        parameters=[{'robot_description': robot_description}]
    )

    joint_state_publisher = Node(
        package='joint_state_publisher',
        executable='joint_state_publisher',
        parameters=[{'source_list': ['/arm/joint_states']}]
    )

    return LaunchDescription([
        robot_state_publisher,
        joint_state_publisher,
    ])
传感器数据坐标变换
传感器数据（点云、激光扫描、图像目标检测结果）通常需要从传感器坐标系变换到基准坐标系才能进行融合和规划。TF2 提供了专门的数据变换接口。


import rclpy
from rclpy.node import Node
import tf2_ros
import tf2_geometry_msgs
from sensor_msgs.msg import PointCloud2
from geometry_msgs.msg import PointStamped
import numpy as np

class SensorTransformer(Node):
    def __init__(self):
        super().__init__('sensor_transformer')

        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)

        self.create_subscription(
            PointStamped, '/detection/point',
            self.transform_point, 10
        )

        self.pub = self.create_publisher(
            PointStamped, '/detection/point_map', 10
        )

    def transform_point(self, msg):
        try:
            transformed = self.tf_buffer.transform(
                msg, 'map',
                timeout=rclpy.duration.Duration(seconds=0.5)
            )
            self.pub.publish(transformed)

        except (tf2_ros.LookupException,
                tf2_ros.ConnectivityException,
                tf2_ros.ExtrapolationException) as e:
            self.get_logger().warn(f'变换失败: {e}')

    def transform_pointcloud_manual(self, cloud_points, source_frame, target_frame, stamp):
        try:
            transform = self.tf_buffer.lookup_transform(
                target_frame, source_frame, stamp
            )

            t = transform.transform.translation
            r = transform.transform.rotation

            translation = np.array([t.x, t.y, t.z])
            R = self.quaternion_to_matrix(r.x, r.y, r.z, r.w)

            transformed = (R @ cloud_points.T).T + translation
            return transformed

        except Exception as e:
            self.get_logger().error(f'点云变换失败: {e}')
            return None

    def quaternion_to_matrix(self, x, y, z, w):
        return np.array([
            [1-2*(y*y+z*z), 2*(x*y-w*z), 2*(x*z+w*y)],
            [2*(x*y+w*z), 1-2*(x*x+z*z), 2*(y*z-w*x)],
            [2*(x*z-w*y), 2*(y*z+w*x), 1-2*(x*x+y*y)]
        ])
小结
    

  • TF2 维护机器人所有坐标系间的变换关系树，是 ROS2 空间推理的基础设施
    • 

  • 静态变换（传感器安装位置）和动态变换（里程计、SLAM）通过不同机制发布
    • 

  • URDF 定义机器人模型结构，Robot State Publisher 自动生成连杆间的 TF 变换
    • 

  • Xacro 宏语言简化了重复结构的 URDF 编写，支持参数化模型生成
    • 

  • 传感器数据坐标变换是多传感器融合的前置步骤，TF2 提供时间同步的变换查询
    • 

  • 正确的 TF 树结构设计（map→odom→base_link 链路）是导航系统正常工作的前提
    •
