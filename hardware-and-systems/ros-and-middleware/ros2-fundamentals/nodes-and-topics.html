ROS2 节点与话题

核心概念

ROS2（Robot Operating System 2）是机器人软件开发的主流中间件框架，节点（Node）和话题（Topic）是其最基本的通信构建块。节点是 ROS2 中最小的可执行单元，每个节点负责一个独立的功能模块（如传感器驱动、路径规划、电机控制）。话题是节点间异步通信的命名通道，基于发布-订阅（Pub-Sub）模式实现松耦合的数据流传输。

ROS2 相比 ROS1 的核心改进包括：基于 DDS（Data Distribution Service）的去中心化通信、实时性支持、多平台兼容、以及内建的安全机制。这些改进使 ROS2 适用于从研究原型到工业部署的全生命周期。

节点创建与生命周期
ROS2 节点使用 rclpy（Python）或 rclcpp（C++）客户端库创建。每个节点拥有独立的命名空间、参数空间和执行上下文。


import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan
from geometry_msgs.msg import Twist
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy

class ObstacleAvoidanceNode(Node):
    def __init__(self):
        super().__init__('obstacle_avoidance')

        sensor_qos = QoSProfile(
            reliability=ReliabilityPolicy.BEST_EFFORT,
            history=HistoryPolicy.KEEP_LAST,
            depth=5
        )

        self.scan_sub = self.create_subscription(
            LaserScan, '/scan', self.scan_callback, sensor_qos
        )

        self.cmd_pub = self.create_publisher(Twist, '/cmd_vel', 10)

        self.timer = self.create_timer(0.1, self.control_loop)

        self.declare_parameter('safety_distance', 0.5)
        self.declare_parameter('max_linear_speed', 0.5)
        self.declare_parameter('max_angular_speed', 1.0)

        self.latest_scan = None
        self.get_logger().info('避障节点已启动')

    def scan_callback(self, msg):
        self.latest_scan = msg

    def control_loop(self):
        if self.latest_scan is None:
            return

        safety_dist = self.get_parameter('safety_distance').value
        max_linear = self.get_parameter('max_linear_speed').value
        max_angular = self.get_parameter('max_angular_speed').value

        ranges = list(self.latest_scan.ranges)
        n = len(ranges)

        front_ranges = ranges[n//4 : 3*n//4]
        min_front = min(r for r in front_ranges if r > 0.1)

        cmd = Twist()

        if min_front > safety_dist:
            cmd.linear.x = max_linear
            cmd.angular.z = 0.0
        else:
            cmd.linear.x = 0.0
            left_avg = sum(ranges[:n//4]) / (n//4)
            right_avg = sum(ranges[3*n//4:]) / (n//4)

            if left_avg > right_avg:
                cmd.angular.z = max_angular
            else:
                cmd.angular.z = -max_angular

        self.cmd_pub.publish(cmd)

def main():
    rclpy.init()
    node = ObstacleAvoidanceNode()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()
C++ 版本的节点实现展示了 ROS2 的性能优势：


#include <rclcpp/rclcpp.hpp>
#include <sensor_msgs/msg/laser_scan.hpp>
#include <geometry_msgs/msg/twist.hpp>

class ObstacleAvoidanceNode : public rclcpp::Node {
public:
    ObstacleAvoidanceNode() : Node("obstacle_avoidance") {
        auto sensor_qos = rclcpp::QoS(5).best_effort();

        scan_sub_ = this->create_subscription<sensor_msgs::msg::LaserScan>(
            "/scan", sensor_qos,
            std::bind(&ObstacleAvoidanceNode::scan_callback, this, std::placeholders::_1)
        );

        cmd_pub_ = this->create_publisher<geometry_msgs::msg::Twist>("/cmd_vel", 10);

        timer_ = this->create_wall_timer(
            std::chrono::milliseconds(100),
            std::bind(&ObstacleAvoidanceNode::control_loop, this)
        );

        this->declare_parameter("safety_distance", 0.5);
    }

private:
    void scan_callback(const sensor_msgs::msg::LaserScan::SharedPtr msg) {
        latest_scan_ = msg;
    }

    void control_loop() {
        if (!latest_scan_) return;

        auto safety_dist = this->get_parameter("safety_distance").as_double();
        auto cmd = geometry_msgs::msg::Twist();

        float min_range = std::numeric_limits<float>::max();
        for (size_t i = latest_scan_->ranges.size() / 4;
             i < 3 * latest_scan_->ranges.size() / 4; ++i) {
            if (latest_scan_->ranges[i] > 0.1f) {
                min_range = std::min(min_range, latest_scan_->ranges[i]);
            }
        }

        if (min_range > safety_dist) {
            cmd.linear.x = 0.5;
        } else {
            cmd.angular.z = 1.0;
        }

        cmd_pub_->publish(cmd);
    }

    rclcpp::Subscription<sensor_msgs::msg::LaserScan>::SharedPtr scan_sub_;
    rclcpp::Publisher<geometry_msgs::msg::Twist>::SharedPtr cmd_pub_;
    rclcpp::TimerBase::SharedPtr timer_;
    sensor_msgs::msg::LaserScan::SharedPtr latest_scan_;
};

多节点组合与执行器
ROS2 提供了组件（Component）机制，允许将多个节点加载到同一进程中，通过共享内存实现零拷贝通信，显著降低延迟和 CPU 开销。


import rclpy
from rclpy.executors import MultiThreadedExecutor, SingleThreadedExecutor
from rclpy.node import Node

class SensorNode(Node):
    def __init__(self):
        super().__init__('sensor_node')
        self.publisher = self.create_publisher(LaserScan, '/scan', 10)
        self.timer = self.create_timer(0.05, self.publish_scan)

    def publish_scan(self):
        msg = LaserScan()
        msg.header.stamp = self.get_clock().now().to_msg()
        self.publisher.publish(msg)

class ProcessingNode(Node):
    def __init__(self):
        super().__init__('processing_node')
        self.subscription = self.create_subscription(
            LaserScan, '/scan', self.callback, 10
        )

    def callback(self, msg):
        self.get_logger().debug('处理扫描数据')

def main():
    rclpy.init()

    sensor = SensorNode()
    processing = ProcessingNode()
    avoidance = ObstacleAvoidanceNode()

    executor = MultiThreadedExecutor(num_threads=4)
    executor.add_node(sensor)
    executor.add_node(processing)
    executor.add_node(avoidance)

    try:
        executor.spin()
    finally:
        executor.shutdown()
        rclpy.shutdown()

ROS2 话题诊断与调试
话题的监控和诊断是保障机器人系统可靠运行的重要手段。ROS2 提供了丰富的命令行工具和编程接口用于话题调试。


ros2 topic list
ros2 topic info /scan --verbose
ros2 topic hz /scan
ros2 topic bw /scan
ros2 topic echo /cmd_vel --once
ros2 topic pub /cmd_vel geometry_msgs/msg/Twist "{linear: {x: 0.5}, angular: {z: 0.1}}"
from rclpy.node import Node
from diagnostic_msgs.msg import DiagnosticArray, DiagnosticStatus, KeyValue

class TopicMonitor(Node):
    def __init__(self, topic_name, expected_hz):
        super().__init__('topic_monitor')
        self.topic_name = topic_name
        self.expected_hz = expected_hz
        self.msg_count = 0
        self.last_time = self.get_clock().now()

        self.create_subscription(
            LaserScan, topic_name, self.msg_callback, 10
        )

        self.diag_pub = self.create_publisher(
            DiagnosticArray, '/diagnostics', 10
        )
        self.create_timer(1.0, self.check_health)

    def msg_callback(self, msg):
        self.msg_count += 1

    def check_health(self):
        now = self.get_clock().now()
        elapsed = (now - self.last_time).nanoseconds / 1e9
        actual_hz = self.msg_count / elapsed if elapsed > 0 else 0

        diag = DiagnosticArray()
        diag.header.stamp = now.to_msg()

        status = DiagnosticStatus()
        status.name = f'Topic: {self.topic_name}'
        status.values = [
            KeyValue(key='actual_hz', value=f'{actual_hz:.1f}'),
            KeyValue(key='expected_hz', value=f'{self.expected_hz:.1f}'),
        ]

        if actual_hz >= self.expected_hz * 0.8:
            status.level = DiagnosticStatus.OK
            status.message = '频率正常'
        elif actual_hz >= self.expected_hz * 0.5:
            status.level = DiagnosticStatus.WARN
            status.message = '频率偏低'
        else:
            status.level = DiagnosticStatus.ERROR
            status.message = '频率严重不足'

        diag.status.append(status)
        self.diag_pub.publish(diag)

        self.msg_count = 0
        self.last_time = now
小结
    

  • ROS2 的节点和话题构成发布-订阅通信模型，实现模块间松耦合数据传输
    • 

  • 节点是最小的可执行单元，每个节点聚焦单一功能，通过话题、服务、动作进行交互
    • 

  • QoS 策略（可靠性、历史深度）允许针对不同场景调优通信行为
    • 

  • 自定义消息类型通过 IDL 定义，支持跨语言（C++/Python）使用
    • 

  • 组件机制和多线程执行器支持高性能的多节点组合部署
    • 

  • 话题诊断和监控工具是系统可靠运行的保障
