ROS2 Launch 系统

核心概念

ROS2 的 Launch 系统是机器人系统部署和编排的核心工具。它负责启动多个节点、配置参数、设置命名空间和话题重映射，以及管理节点间的依赖关系。ROS2 的 Launch 系统使用 Python 脚本（也支持 XML 和 YAML），提供了远超 ROS1 XML Launch 文件的灵活性和表达能力。

Launch 系统的设计理念是"声明式编排"——开发者描述系统应该是什么样的，Launch 系统负责实现。它支持条件执行、事件驱动、参数替换和嵌套组合，能够表达复杂的系统启动逻辑。

Launch 文件核心组件
Launch 文件由 Actions（动作）、Substitutions（替换）和 Conditions（条件）三大核心组件构成。Actions 是 Launch 系统执行的基本单元，Substitutions 提供延迟求值的参数替换，Conditions 实现条件逻辑。


from launch import LaunchDescription, LaunchContext
from launch.actions import (
    DeclareLaunchArgument, GroupAction, IncludeLaunchDescription,
    ExecuteProcess, TimerAction, RegisterEventHandler, LogInfo,
    OpaqueFunction
)
from launch.conditions import IfCondition, UnlessCondition
from launch.substitutions import (
    LaunchConfiguration, PathJoinSubstitution,
    PythonExpression, EnvironmentVariable
)
from launch_ros.actions import Node, SetParameter, PushRosNamespace
from launch_ros.substitutions import FindPackageShare
import os

def generate_launch_description():
    pkg_share = FindPackageShare('robot_bringup')

    robot_name_arg = DeclareLaunchArgument(
        'robot_name', default_value='robot_1',
        description='Name of the robot instance'
    )
    use_sim_arg = DeclareLaunchArgument(
        'use_sim', default_value='false',
        description='Use simulation time'
    )
    config_file_arg = DeclareLaunchArgument(
        'config_file',
        default_value=PathJoinSubstitution([pkg_share, 'config', 'default.yaml'])
    )

    set_sim_time = SetParameter(
        name='use_sim_time',
        value=LaunchConfiguration('use_sim')
    )

    driver_group = GroupAction([
        PushRosNamespace(LaunchConfiguration('robot_name')),
        Node(
            package='robot_drivers',
            executable='lidar_driver',
            name='lidar',
            parameters=[LaunchConfiguration('config_file')],
            output='screen',
            condition=UnlessCondition(LaunchConfiguration('use_sim'))
        ),
        Node(
            package='robot_drivers',
            executable='imu_driver',
            name='imu',
            parameters=[LaunchConfiguration('config_file')],
            remappings=[('imu/data_raw', 'imu/data')],
            condition=UnlessCondition(LaunchConfiguration('use_sim'))
        ),
    ])

    sim_group = GroupAction([
        PushRosNamespace(LaunchConfiguration('robot_name')),
        Node(
            package='gazebo_ros',
            executable='spawn_entity.py',
            arguments=['-entity', LaunchConfiguration('robot_name'),
                      '-topic', 'robot_description'],
            condition=IfCondition(LaunchConfiguration('use_sim'))
        ),
    ])

    navigation_launch = IncludeLaunchDescription(
        PathJoinSubstitution([pkg_share, 'launch', 'navigation.launch.py']),
        launch_arguments={
            'namespace': LaunchConfiguration('robot_name'),
            'use_sim_time': LaunchConfiguration('use_sim'),
        }.items()
    )

    return LaunchDescription([
        robot_name_arg,
        use_sim_arg,
        config_file_arg,
        set_sim_time,
        driver_group,
        sim_group,
        navigation_launch,
    ])
PythonExpression 替换允许在 Launch 文件中执行任意 Python 表达式，实现复杂的参数计算。OpaqueFunction 允许在 Launch 上下文中执行自定义 Python 函数，可以根据运行时条件动态生成 Actions。


事件系统与依赖管理
Launch 系统的事件机制使得节点间的启动依赖可以被精确描述。常见的事件包括进程启动/退出、生命周期状态转移、定时器触发等。通过事件处理器，可以实现"节点 A 启动后再启动节点 B"这样的依赖关系。


from launch import LaunchDescription
from launch.actions import (
    RegisterEventHandler, EmitEvent, TimerAction, LogInfo
)
from launch.event_handlers import OnProcessStart, OnProcessExit
from launch.events import Shutdown
from launch_ros.actions import Node, LifecycleNode
from launch_ros.event_handlers import OnStateTransition
from lifecycle_msgs.msg import Transition

def generate_launch_description():
    sensor_driver = LifecycleNode(
        package='robot_drivers',
        executable='sensor_node',
        name='sensor',
        output='screen'
    )

    controller = Node(
        package='robot_control',
        executable='controller_node',
        name='controller',
        output='screen'
    )

    configure_sensor = RegisterEventHandler(
        OnProcessStart(
            target_action=sensor_driver,
            on_start=[
                TimerAction(
                    period=1.0,
                    actions=[LogInfo(msg='Configuring sensor...')]
                )
            ]
        )
    )

    start_controller_on_sensor_active = RegisterEventHandler(
        OnStateTransition(
            target_lifecycle_node=sensor_driver,
            goal_state='active',
            entities=[
                LogInfo(msg='Sensor active, starting controller'),
                controller
            ]
        )
    )

    handle_controller_crash = RegisterEventHandler(
        OnProcessExit(
            target_action=controller,
            on_exit=[
                LogInfo(msg='Controller crashed! Restarting...'),
                TimerAction(
                    period=3.0,
                    actions=[controller]
                )
            ]
        )
    )

    return LaunchDescription([
        sensor_driver,
        configure_sensor,
        start_controller_on_sensor_active,
        handle_controller_crash,
    ])
进程监控和自动重启是生产系统的关键能力。OnProcessExit 事件处理器可以检测节点崩溃并触发重启逻辑。结合 TimerAction 可以实现退避重启策略，避免快速循环重启。


多机器人 Launch 编排
多机器人场景需要为每个机器人实例创建独立的节点集合，同时避免话题和服务名称冲突。ROS2 的命名空间（Namespace）机制和 Launch 的 GroupAction 组合是解决方案。


from launch import LaunchDescription
from launch.actions import GroupAction, DeclareLaunchArgument, OpaqueFunction
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node, PushRosNamespace
from launch_ros.substitutions import FindPackageShare

def spawn_robot(context, robot_name, x, y, yaw):
    ns = robot_name

    return [
        GroupAction([
            PushRosNamespace(ns),
            Node(
                package='robot_state_publisher',
                executable='robot_state_publisher',
                name='robot_state_publisher',
                parameters=[{'robot_description': f'<robot name="{ns}"/>'}],
            ),
            Node(
                package='robot_drivers',
                executable='base_driver',
                name='base_driver',
                parameters=[{
                    'initial_x': float(x),
                    'initial_y': float(y),
                    'initial_yaw': float(yaw),
                }],
            ),
            Node(
                package='robot_navigation',
                executable='navigator',
                name='navigator',
                remappings=[
                    ('/tf', 'tf'),
                    ('/tf_static', 'tf_static'),
                ],
            ),
        ])
    ]

def generate_launch_description():
    num_robots_arg = DeclareLaunchArgument(
        'num_robots', default_value='3'
    )

    def launch_setup(context):
        num_robots = int(LaunchConfiguration('num_robots').perform(context))
        actions = []

        positions = [
            (0.0, 0.0, 0.0),
            (2.0, 0.0, 1.57),
            (0.0, 2.0, 3.14),
            (2.0, 2.0, -1.57),
        ]

        for i in range(min(num_robots, len(positions))):
            x, y, yaw = positions[i]
            robot_name = f'robot_{i+1}'
            actions.extend(spawn_robot(context, robot_name, x, y, yaw))

        return actions

    return LaunchDescription([
        num_robots_arg,
        OpaqueFunction(function=launch_setup),
    ])

OpaqueFunction 在多机器人场景中尤其有用——它允许在运行时根据参数动态生成任意数量的节点组，这是纯声明式 Action 无法实现的。


小结
    

  • ROS2 Launch 系统使用 Python 脚本实现声明式系统编排，支持复杂的启动逻辑
    • 

  • 核心组件包括 Actions（执行单元）、Substitutions（参数替换）和 Conditions（条件判断）
    • 

  • 事件系统支持进程监控、生命周期转移响应和节点间依赖管理
    • 

  • GroupAction + PushRosNamespace 实现多机器人的命名空间隔离
    • 

  • OpaqueFunction 提供运行时动态生成能力，适合参数化的系统部署
    • 

  • IncludeLaunchDescription 支持 Launch 文件的模块化组合和复用
    • 

  • 生产系统必须包含进程监控、自动重启和错误处理的 Launch 配置
    •
