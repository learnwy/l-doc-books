SCADA 系统

核心概念

SCADA（Supervisory Control and Data Acquisition，数据采集与监视控制系统）是工业自动化中的上层监控平台，负责对分布在不同位置的 PLC、RTU（远程终端单元）和智能仪表进行集中监控和管理。SCADA 系统实时采集现场数据、显示工艺流程图、记录历史趋势、管理报警事件，为操作员提供全局视野和远程控制能力。

SCADA 处于自动化金字塔的监控层（Level 2），位于 PLC 控制层（Level 1）之上，MES 层（Level 3）之下。它是连接车间现场和企业管理的关键枢纽。

数据采集与通信协议
SCADA 系统通过工业通信协议与现场设备交互。OPC UA（Open Platform Communications Unified Architecture）是当前工业物联网的标准通信框架，提供跨平台、安全、面向服务的数据交换能力。Modbus 是历史最悠久的工业协议，简单可靠，仍广泛用于传感器和小型设备。


import struct
import time

class ModbusRTUClient:
    def __init__(self, port, baudrate=9600, slave_id=1):
        self.port = port
        self.baudrate = baudrate
        self.slave_id = slave_id

    def read_holding_registers(self, start_address, count):
        request = struct.pack('>BBhh',
            self.slave_id, 0x03, start_address, count)
        crc = self._compute_crc(request)
        frame = request + struct.pack('<H', crc)
        return self._send_receive(frame, count * 2 + 5)

    def write_single_register(self, address, value):
        request = struct.pack('>BBhh',
            self.slave_id, 0x06, address, value)
        crc = self._compute_crc(request)
        frame = request + struct.pack('<H', crc)
        return self._send_receive(frame, 8)

    def _compute_crc(self, data):
        crc = 0xFFFF
        for byte in data:
            crc ^= byte
            for _ in range(8):
                if crc & 0x0001:
                    crc = (crc >> 1) ^ 0xA001
                else:
                    crc >>= 1
        return crc

    def _send_receive(self, frame, expected_length):
        return b'\x00' * expected_length


class OPCUADataCollector:
    def __init__(self, server_url):
        self.server_url = server_url
        self.subscriptions = {}
        self.data_buffer = {}

    def connect(self):
        self.connected = True

    def subscribe(self, node_id, callback, interval_ms=1000):
        self.subscriptions[node_id] = {
            'callback': callback,
            'interval': interval_ms,
            'last_value': None
        }

    def read_value(self, node_id):
        return self.data_buffer.get(node_id, {
            'value': 0.0,
            'timestamp': time.time(),
            'quality': 'Good'
        })

    def write_value(self, node_id, value):
        self.data_buffer[node_id] = {
            'value': value,
            'timestamp': time.time(),
            'quality': 'Good'
        }

    def poll_cycle(self):
        for node_id, sub in self.subscriptions.items():
            current = self.read_value(node_id)
            if current['value'] != sub['last_value']:
                sub['callback'](node_id, current)
                sub['last_value'] = current['value']
OPC UA 相对 Modbus 的核心优势在于：信息模型（不仅传输数据值，还传输数据的语义描述、工程单位、访问权限）、安全机制（证书认证、加密传输、审计日志）和发现机制（服务端自动发现、浏览信息模型）。


报警管理与历史趋势
报警管理是 SCADA 系统的关键功能，它监控过程变量是否超出正常范围，并及时通知操作员采取行动。ISA-18.2 标准定义了报警管理的最佳实践，包括报警优先级分类、抑制策略和报警合理化。


import time
from collections import deque
from enum import Enum

class AlarmPriority(Enum):
    CRITICAL = 1
    HIGH = 2
    MEDIUM = 3
    LOW = 4
    INFO = 5

class AlarmState(Enum):
    NORMAL = 'normal'
    ACTIVE_UNACK = 'active_unacknowledged'
    ACTIVE_ACK = 'active_acknowledged'
    CLEARED_UNACK = 'cleared_unacknowledged'

class AlarmManager:
    def __init__(self):
        self.alarm_definitions = {}
        self.active_alarms = {}
        self.alarm_history = deque(maxlen=10000)
        self.shelved_alarms = set()

    def define_alarm(self, tag, description, priority, condition):
        self.alarm_definitions[tag] = {
            'description': description,
            'priority': priority,
            'condition': condition,
            'deadband': 0.0,
            'delay_seconds': 0
        }

    def evaluate(self, tag, value, timestamp=None):
        if tag not in self.alarm_definitions:
            return
        if tag in self.shelved_alarms:
            return

        defn = self.alarm_definitions[tag]
        is_alarm = defn['condition'](value)
        ts = timestamp or time.time()

        if is_alarm and tag not in self.active_alarms:
            self.active_alarms[tag] = {
                'state': AlarmState.ACTIVE_UNACK,
                'priority': defn['priority'],
                'value': value,
                'timestamp': ts,
                'description': defn['description']
            }
            self._log_event(tag, 'ACTIVATED', value, ts)

        elif not is_alarm and tag in self.active_alarms:
            alarm = self.active_alarms[tag]
            if alarm['state'] == AlarmState.ACTIVE_ACK:
                del self.active_alarms[tag]
            else:
                alarm['state'] = AlarmState.CLEARED_UNACK
            self._log_event(tag, 'CLEARED', value, ts)

    def acknowledge(self, tag, operator):
        if tag in self.active_alarms:
            alarm = self.active_alarms[tag]
            if alarm['state'] == AlarmState.ACTIVE_UNACK:
                alarm['state'] = AlarmState.ACTIVE_ACK
            elif alarm['state'] == AlarmState.CLEARED_UNACK:
                del self.active_alarms[tag]
            self._log_event(tag, 'ACKNOWLEDGED', None, time.time(), operator)

    def _log_event(self, tag, event_type, value, timestamp, operator=None):
        self.alarm_history.append({
            'tag': tag, 'event': event_type, 'value': value,
            'timestamp': timestamp, 'operator': operator
        })

    def get_active_summary(self):
        summary = {p: 0 for p in AlarmPriority}
        for alarm in self.active_alarms.values():
            summary[alarm['priority']] += 1
        return summary
历史趋势系统将过程数据存储在时序数据库中，支持操作员查看任意时间段的数据变化曲线。数据压缩算法（如旋转门压缩 Swinging Door Trending）在保证数据精度的前提下大幅减少存储量。

小结
    

  • SCADA 是工业自动化的监控中枢，集中管理分布式现场设备
    • 

  • OPC UA 是现代工业通信标准，提供语义信息模型和安全机制
    • 

  • Modbus 简单可靠，仍广泛用于传感器和小型设备通信
    • 

  • 报警管理遵循 ISA-18.2 标准，包括优先级分类、状态机管理和抑制策略
    • 

  • 历史数据库使用时序数据库存储，旋转门压缩算法优化存储效率
    • 

  • SCADA 安全是工控安全的关键环节，需要网络隔离和访问控制
    • 

  • 现代 SCADA 正向 Web 化、云端化和 AI 赋能方向演进
