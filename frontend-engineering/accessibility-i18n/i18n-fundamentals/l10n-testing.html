本地化测试

核心概念

本地化（L10n）测试验证应用在不同语言和地区设置下是否正确工作。它不仅检查翻译的准确性，还验证布局适配、格式化正确性、文本截断和编码问题。本地化测试是 i18n 工程的质量保证环节——即使代码中的 i18n 架构完美，翻译质量或布局适配问题也会导致糟糕的用户体验。

伪本地化
伪本地化（Pseudo-localization）是在不需要真实翻译的情况下测试 i18n 完整性的技术。它通过算法转换原文字符，保持可读性同时模拟翻译后的各种特征。


const pseudoLocalize = (text: string, options: PseudoOptions = {}): string => {
  const { expandRatio = 0.3, addAccents = true, wrapBrackets = true } = options;

  let result = text;

  if (addAccents) {
    const accentMap: Record<string, string> = {
      a: 'à',
      e: 'è',
      i: 'ì',
      o: 'ò',
      u: 'ù',
      A: 'À',
      E: 'È',
      I: 'Ì',
      O: 'Ò',
      U: 'Ù',
    };
    result = result
      .split('')
      .map((c) => accentMap[c] || c)
      .join('');
  }

  if (expandRatio > 0) {
    const expansion = Math.ceil(text.length * expandRatio);
    result = result + '~'.repeat(expansion);
  }

  if (wrapBrackets) {
    result = `[${result}]`;
  }

  return result;
};

// 伪本地化示例
pseudoLocalize('Save changes');
// "[Sàvè chàngès~~~~~~~~~]"

pseudoLocalize('OK');
// "[ÒK~]"

// 集成到 i18n 系统
const enablePseudoLocale = () => {
  i18n.addResourceBundle(
    'pseudo',
    'common',
    Object.fromEntries(Object.entries(enMessages).map(([key, value]) => [key, pseudoLocalize(value)])),
  );
};

自动化 L10n 测试
自动化测试覆盖翻译完整性、格式化正确性和基础布局检查。


import { test, expect } from '@playwright/test';

const locales = ['en-US', 'zh-CN', 'ja-JP', 'de-DE'];

for (const locale of locales) {
  test.describe(`${locale} 本地化测试`, () => {
    test.beforeEach(async ({ page }) => {
      await page.goto(`/?lang=${locale}`);
      await page.waitForLoadState('networkidle');
    });

    test('页面无缺失翻译键', async ({ page }) => {
      const missingKeys = await page.evaluate(() => {
        const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
        const missing: string[] = [];
        while (walker.nextNode()) {
          const text = walker.currentNode.textContent?.trim() ?? '';
          if (/^[a-z]+\.[a-z]+(\.[a-z]+)*$/.test(text)) {
            missing.push(text);
          }
        }
        return missing;
      });
      expect(missingKeys, `发现未翻译的键: ${missingKeys.join(', ')}`).toEqual([]);
    });

    test('文本不溢出容器', async ({ page }) => {
      const overflows = await page.evaluate(() => {
        const elements = document.querySelectorAll('p, span, h1, h2, h3, button, a');
        const issues: string[] = [];
        elements.forEach((el) => {
          const htmlEl = el as HTMLElement;
          if (htmlEl.scrollWidth > htmlEl.clientWidth + 1) {
            issues.push(`${el.tagName}: "${el.textContent?.slice(0, 50)}"`);
          }
        });
        return issues;
      });
      expect(overflows).toEqual([]);
    });

    test('日期格式化正确', async ({ page }) => {
      const dateTexts = await page.locator('[data-testid="formatted-date"]').allTextContents();
      for (const text of dateTexts) {
        expect(text).not.toContain('Invalid Date');
        expect(text).not.toContain('NaN');
      }
    });
  });
}
视觉回归测试
视觉回归测试通过截图对比发现本地化导致的布局问题。不同语言的截图作为基线，每次变更后对比差异。


import { test, expect } from '@playwright/test';

const testLocales = [
  { code: 'en-US', name: 'English' },
  { code: 'zh-CN', name: 'Chinese' },
  { code: 'ar-SA', name: 'Arabic', dir: 'rtl' },
  { code: 'de-DE', name: 'German' },
];

for (const locale of testLocales) {
  test(`首页视觉回归 - ${locale.name}`, async ({ page }) => {
    await page.goto(`/?lang=${locale.code}`);
    await page.waitForLoadState('networkidle');

    await expect(page).toHaveScreenshot(`home-${locale.code}.png`, {
      maxDiffPixelRatio: 0.01,
      fullPage: true,
    });
  });
}

// 组件级视觉测试
test.describe('Button 组件多语言视觉', () => {
  const buttonTexts = {
    'en-US': 'Submit',
    'zh-CN': '提交',
    'de-DE': 'Einreichen',
    'ar-SA': 'إرسال',
  };

  for (const [locale, text] of Object.entries(buttonTexts)) {
    test(`Button - ${locale}`, async ({ page }) => {
      await page.setContent(
        `<html lang="${locale}" dir="${locale === 'ar-SA' ? 'rtl' : 'ltr'}">
          <button style="padding: 8px 16px">${text}</button>
        </html>`,
      );
      await expect(page.locator('button')).toHaveScreenshot(`button-${locale}.png`);
    });
  }
});
