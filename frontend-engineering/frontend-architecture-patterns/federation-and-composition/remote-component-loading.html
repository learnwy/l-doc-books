远程组件加载

概述

远程组件加载是将 UI 组件的代码托管在远端服务器，在运行时按需获取并渲染的架构模式。它将传统的"构建时打包一切"模式转变为"运行时按需组装"，使得功能交付从"整体发版"细化到"组件级发布"。2026 年的生产级方案已经解决了类型安全、版本管理、降级策略和预加载优化等关键问题，让远程组件可以像本地组件一样可靠地使用。

实战指南
通用远程组件加载器
// lib/remote-loader.ts
import { init, loadRemote } from '@module-federation/enhanced/runtime';

interface RemoteConfig {
  name: string;
  entry: string; // mf-manifest.json URL
  module: string; // 暴露的模块路径
}

const loadedRemotes = new Map<string, React.ComponentType>();

export async function loadRemoteComponent<P = any>(config: RemoteConfig): Promise<React.ComponentType<P>> {
  const cacheKey = `${config.name}/${config.module}`;

  if (loadedRemotes.has(cacheKey)) {
    return loadedRemotes.get(cacheKey)! as React.ComponentType<P>;
  }

  // 动态注册远程
  init({
    name: 'host',
    remotes: [{ name: config.name, entry: config.entry }],
  });

  const module = await loadRemote<{ default: React.ComponentType<P> }>(`${config.name}/${config.module}`);

  if (!module?.default) {
    throw new Error(`Remote module ${cacheKey} has no default export`);
  }

  loadedRemotes.set(cacheKey, module.default);
  return module.default;
}
React Hook 封装
// hooks/useRemoteComponent.ts
import { useState, useEffect, useRef } from 'react';
import { loadRemoteComponent } from '../lib/remote-loader';

interface UseRemoteOptions {
  name: string;
  entry: string;
  module: string;
  fallback?: React.ComponentType;
  retryCount?: number;
  retryDelay?: number;
  timeout?: number;
}

export function useRemoteComponent<P = any>(options: UseRemoteOptions) {
  const { name, entry, module, retryCount = 3, retryDelay = 1000, timeout = 10000 } = options;
  const [Component, setComponent] = useState<React.ComponentType<P> | null>(null);
  const [error, setError] = useState<Error | null>(null);
  const [loading, setLoading] = useState(true);
  const retriesRef = useRef(0);

  useEffect(() => {
    let cancelled = false;
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeout);

    async function load() {
      try {
        const Comp = await loadRemoteComponent<P>({ name, entry, module });
        if (!cancelled) {
          setComponent(() => Comp);
          setLoading(false);
        }
      } catch (err) {
        if (cancelled) return;
        if (retriesRef.current < retryCount) {
          retriesRef.current++;
          setTimeout(load, retryDelay * retriesRef.current);
        } else {
          setError(err as Error);
          setLoading(false);
        }
      }
    }

    load();
    return () => { cancelled = true; clearTimeout(timer); };
  }, [name, entry, module, retryCount, retryDelay, timeout]);

  return { Component, error, loading };
}

// 使用示例
function RemoteWidget() {
  const { Component, error, loading } = useRemoteComponent({
    name: 'widgets',
    entry: 'https://cdn.example.com/widgets/mf-manifest.json',
    module: 'RecommendationPanel',
    retryCount: 2,
  });

  if (loading) return <WidgetSkeleton />;
  if (error) return <WidgetFallback error={error} />;
  if (!Component) return null;

  return <Component userId="current" maxItems={5} />;
}
版本管理与灰度发布
// lib/version-resolver.ts
interface ComponentManifest {
  name: string;
  versions: {
    stable: string; // 稳定版 URL
    canary: string; // 金丝雀版 URL
    rollback: string; // 回滚版 URL
  };
  features: Record<string, boolean>; // Feature flags
}

async function resolveRemoteEntry(componentName: string, userId: string): Promise<string> {
  const manifest = (await fetch('/api/component-registry/' + componentName).then((r) => r.json())) as ComponentManifest;

  // 灰度策略: 10% 用户使用 canary 版本
  const isCanaryUser = hashUserId(userId) % 100 < 10;

  if (isCanaryUser && manifest.versions.canary) {
    return manifest.versions.canary;
  }

  return manifest.versions.stable;
}

function NavLink({ to, remoteName, remoteEntry }: NavLinkProps) {
  const handleMouseEnter = () => {
    // 预加载远程 manifest 和入口文件
    preloadRemote({ name: remoteName, entry: remoteEntry });
  };

  return (
    <Link to={to} onMouseEnter={handleMouseEnter}>
      {children}
    </Link>
  );
}

// 2. Intersection Observer 预加载 - 即将进入视口
function LazyRemoteSection({ config }: { config: RemoteConfig }) {
  const ref = useRef<HTMLDivElement>(null);
  const [shouldLoad, setShouldLoad] = useState(false);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setShouldLoad(true);
          observer.disconnect();
        }
      },
      { rootMargin: '200px' } // 提前 200px 开始加载
    );
    if (ref.current) observer.observe(ref.current);
    return () => observer.disconnect();
  }, []);

  return (
    <div ref={ref}>
      {shouldLoad ? <RemoteComponent config={config} /> : <Placeholder />}
    </div>
  );
}

总结
远程组件加载将前端应用从"编译时整体"转变为"运行时组合"，让功能交付的粒度从"整个应用"细化到"单个组件"。生产级实现需要关注四个关键维度：可靠性（重试 + 降级 + 缓存回退）、性能（预加载 + 共享依赖）、安全性（CSP + SRI + 沙箱）、可观测性（加载成功率 + 耗时监控）。Module Federation 2.0 的 manifest 协议和运行时 API 提供了目前最完整的解决方案，但更轻量的场景（如嵌入第三方 widget）可以使用 ESM CDN import 或 SystemJS 等更简单的方案。选择远程加载的前提是：该组件的发布周期确实需要独立于宿主应用。
