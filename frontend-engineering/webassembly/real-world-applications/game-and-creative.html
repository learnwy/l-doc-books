// Bevy 游戏引擎 WASM 示例
use bevy::prelude::*;

fn main() {
    App::new()
        .add_plugins(DefaultPlugins.set(WindowPlugin {
            primary_window: Some(Window {
                title: "WASM 游戏".to_string(),
                canvas: Some("#game-canvas".to_string()),
                ..default()
            }),
            ..default()
        }))
        .add_systems(Startup, setup)
        .add_systems(Update, (move_player, check_collisions))
        .run();
}

fn setup(
    mut commands: Commands,
    mut meshes: ResMut<Assets<Mesh>>,
    mut materials: ResMut<Assets<ColorMaterial>>,
) {
    commands.spawn(Camera2dBundle::default());
    commands.spawn((
        MaterialMesh2dBundle {
            mesh: meshes.add(Circle::new(25.0)).into(),
            material: materials.add(Color::srgb(0.2, 0.7, 0.9)),
            ..default()
        },
        Player { speed: 200.0 },
    ));
}

#[derive(Component)]
struct Player {
    speed: f32,
}

fn move_player(
    time: Res<Time>,
    input: Res<ButtonInput<KeyCode>>,
    mut query: Query<(&mut Transform, &Player)>,
) {
    for (mut transform, player) in &mut query {
        let mut direction = Vec3::ZERO;
        if input.pressed(KeyCode::ArrowUp) { direction.y += 1.0; }
        if input.pressed(KeyCode::ArrowDown) { direction.y -= 1.0; }
        if input.pressed(KeyCode::ArrowLeft) { direction.x -= 1.0; }
        if input.pressed(KeyCode::ArrowRight) { direction.x += 1.0; }
        transform.translation += direction.normalize_or_zero()
            * player.speed * time.delta_seconds();
    }
}

fn check_collisions() {}

音频处理（DSP）
WebAssembly 非常适合实时音频处理——每秒需要处理 44100 个采样点，延迟要求在毫秒级。


use wasm_bindgen::prelude::*;

#[wasm_bindgen]
pub struct AudioProcessor {
    sample_rate: f32,
    phase: f32,
}

#[wasm_bindgen]
impl AudioProcessor {
    #[wasm_bindgen(constructor)]
    pub fn new(sample_rate: f32) -> AudioProcessor {
        AudioProcessor { sample_rate, phase: 0.0 }
    }

    pub fn generate_sine(&mut self, output: &mut [f32], frequency: f32) {
        let phase_inc = frequency / self.sample_rate;
        for sample in output.iter_mut() {
            *sample = (self.phase * std::f32::consts::TAU).sin();
            self.phase += phase_inc;
            if self.phase >= 1.0 { self.phase -= 1.0; }
        }
    }

    pub fn apply_lowpass(&self, buffer: &mut [f32], cutoff: f32) {
        let rc = 1.0 / (cutoff * std::f32::consts::TAU);
        let alpha = (1.0 / self.sample_rate) / (rc + 1.0 / self.sample_rate);
        let mut prev = buffer[0];
        for sample in buffer.iter_mut().skip(1) {
            *sample = prev + alpha * (*sample - prev);
            prev = *sample;
        }
    }

    pub fn apply_gain(buffer: &mut [f32], gain: f32) {
        for sample in buffer.iter_mut() {
            *sample *= gain;
            *sample = sample.clamp(-1.0, 1.0);
        }
    }
}
class WasmAudioWorklet extends AudioWorkletProcessor {
  private processor: AudioProcessor | null = null;

  constructor() {
    super();
    this.port.onmessage = async (e) => {
      if (e.data.type === 'init') {
        const module = await import('./pkg/audio_wasm.js');
        await module.default();
        this.processor = new module.AudioProcessor(sampleRate);
      }
    };
  }

  process(inputs: Float32Array[][], outputs: Float32Array[][]): boolean {
    if (!this.processor) return true;
    const output = outputs[0][0];
    this.processor.generate_sine(output, 440.0);
    return true;
  }
}

物理引擎
将成熟的物理引擎（如 Rapier、Box2D）编译到 WASM，在浏览器中实现高精度物理模拟。


use rapier2d::prelude::*;
use wasm_bindgen::prelude::*;

#[wasm_bindgen]
pub struct PhysicsWorld {
    pipeline: PhysicsPipeline,
    gravity: Vector<f32>,
    integration_params: IntegrationParameters,
    islands: IslandManager,
    broad_phase: DefaultBroadPhase,
    narrow_phase: NarrowPhase,
    bodies: RigidBodySet,
    colliders: ColliderSet,
    impulse_joints: ImpulseJointSet,
    multibody_joints: MultibodyJointSet,
    ccd_solver: CCDSolver,
}

#[wasm_bindgen]
impl PhysicsWorld {
    #[wasm_bindgen(constructor)]
    pub fn new() -> PhysicsWorld {
        PhysicsWorld {
            pipeline: PhysicsPipeline::new(),
            gravity: vector![0.0, -9.81],
            integration_params: IntegrationParameters::default(),
            islands: IslandManager::new(),
            broad_phase: DefaultBroadPhase::new(),
            narrow_phase: NarrowPhase::new(),
            bodies: RigidBodySet::new(),
            colliders: ColliderSet::new(),
            impulse_joints: ImpulseJointSet::new(),
            multibody_joints: MultibodyJointSet::new(),
            ccd_solver: CCDSolver::new(),
        }
    }

    pub fn add_ball(&mut self, x: f32, y: f32, radius: f32) -> u32 {
        let body = RigidBodyBuilder::dynamic().translation(vector![x, y]).build();
        let handle = self.bodies.insert(body);
        let collider = ColliderBuilder::ball(radius).restitution(0.7).build();
        self.colliders.insert_with_parent(collider, handle, &mut self.bodies);
        handle.0.into_raw_parts().0
    }
}

提示： 在 WASM 中通过 web-sys 调用 WebGL 有显著的跨边界开销（每次 GL 调用都是一次 JS↔WASM 交互）。对于 draw call 密集的场景，建议在 WASM 中计算渲染数据，在 JS 侧批量提交 GL 命令。






小结
    

  • 主流游戏引擎（Unity/Unreal/Godot/Bevy）均支持编译到 WASM + WebGL。
    • 

  • WASM 在 AudioWorklet 中实现实时 DSP 处理，满足 2.9ms/128 采样的延迟预算。
    • 

  • Rapier 等 Rust 物理引擎编译到 WASM 后体积仅 ~200KB，提供高精度 2D/3D 模拟。
    • 

  • WASM 与 WebGL 的互操作建议采用WASM 计算 + JS 渲染的分工模式，减少跨边界 GL 调用。
    • 

  • WebGPU 作为 WebGL 的继任者，将为 WASM 图形应用带来更强大的 GPU 计算能力。
