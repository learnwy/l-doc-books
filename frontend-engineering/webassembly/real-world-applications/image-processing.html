图像处理

浏览器端图像处理架构

在浏览器中使用 WASM 进行图像处理，可以获得接近原生的性能，同时保持 Web 的便捷性和跨平台能力。

async function processImage(file: File): Promise<ImageData> {
  const bitmap = await createImageBitmap(file);
  const canvas = new OffscreenCanvas(bitmap.width, bitmap.height);
  const ctx = canvas.getContext('2d')!;
  ctx.drawImage(bitmap, 0, 0);
  const imageData = ctx.getImageData(0, 0, bitmap.width, bitmap.height);

  const wasmModule = await import('./pkg/image_wasm.js');
  await wasmModule.default();

  wasmModule.apply_filter(new Uint8Array(imageData.data.buffer), imageData.width, imageData.height);

  return imageData;
}

像素级操作
WASM 在逐像素操作上的性能优势最为显著，尤其是涉及大量数值计算的滤镜。


use wasm_bindgen::prelude::*;

#[wasm_bindgen]
pub fn grayscale(pixels: &mut [u8]) {
    for chunk in pixels.chunks_exact_mut(4) {
        let r = chunk[0] as f32;
        let g = chunk[1] as f32;
        let b = chunk[2] as f32;
        let gray = (r * 0.299 + g * 0.587 + b * 0.114) as u8;
        chunk[0] = gray;
        chunk[1] = gray;
        chunk[2] = gray;
    }
}

#[wasm_bindgen]
pub fn box_blur(pixels: &mut [u8], width: u32, height: u32, radius: u32) {
    let w = width as usize;
    let h = height as usize;
    let r = radius as usize;
    let mut output = pixels.to_vec();

    for y in 0..h {
        for x in 0..w {
            let mut sum_r = 0u32;
            let mut sum_g = 0u32;
            let mut sum_b = 0u32;
            let mut count = 0u32;

            let y_start = y.saturating_sub(r);
            let y_end = (y + r + 1).min(h);
            let x_start = x.saturating_sub(r);
            let x_end = (x + r + 1).min(w);

            for ky in y_start..y_end {
                for kx in x_start..x_end {
                    let idx = (ky * w + kx) * 4;
                    sum_r += pixels[idx] as u32;
                    sum_g += pixels[idx + 1] as u32;
                    sum_b += pixels[idx + 2] as u32;
                    count += 1;
                }
            }

            let idx = (y * w + x) * 4;
            output[idx] = (sum_r / count) as u8;
            output[idx + 1] = (sum_g / count) as u8;
            output[idx + 2] = (sum_b / count) as u8;
        }
    }

    pixels.copy_from_slice(&output);
}

#[wasm_bindgen]
pub fn adjust_brightness(pixels: &mut [u8], factor: f32) {
    for chunk in pixels.chunks_exact_mut(4) {
        chunk[0] = ((chunk[0] as f32 * factor).min(255.0)) as u8;
        chunk[1] = ((chunk[1] as f32 * factor).min(255.0)) as u8;
        chunk[2] = ((chunk[2] as f32 * factor).min(255.0)) as u8;
    }
}

SIMD 加速
WASM SIMD（128 位向量指令）可以同时处理多个像素通道，大幅提升图像处理性能。


#[cfg(target_arch = "wasm32")]
use std::arch::wasm32::*;

#[wasm_bindgen]
pub fn simd_brighten(pixels: &mut [u8], amount: u8) {
    #[cfg(target_arch = "wasm32")]
    {
        let add_vec = u8x16_splat(amount);
        let chunks = pixels.chunks_exact_mut(16);
        for chunk in chunks {
            let pixel_vec = v128_load(chunk.as_ptr() as *const v128);
            let result = u8x16_add_sat(pixel_vec, add_vec);
            v128_store(chunk.as_mut_ptr() as *mut v128, result);
        }
    }
}

#[wasm_bindgen]
pub fn simd_invert(pixels: &mut [u8]) {
    #[cfg(target_arch = "wasm32")]
    {
        let max_vec = u8x16_splat(255);
        let alpha_mask = u8x16(255,255,255,0, 255,255,255,0, 255,255,255,0, 255,255,255,0);
        let chunks = pixels.chunks_exact_mut(16);
        for chunk in chunks {
            let pixel_vec = v128_load(chunk.as_ptr() as *const v128);
            let inverted = v128_xor(pixel_vec, alpha_mask);
            v128_store(chunk.as_mut_ptr() as *mut v128, inverted);
        }
    }
}

Canvas 集成
将 WASM 处理后的像素数据与 HTML Canvas API 集成，实现实时图像预览和编辑。


class ImageEditor {
  private canvas: HTMLCanvasElement;
  private ctx: CanvasRenderingContext2D;
  private originalPixels: Uint8ClampedArray | null = null;
  private wasmModule: any;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d')!;
  }

  async init(): Promise<void> {
    this.wasmModule = await import('./pkg/image_wasm.js');
    await this.wasmModule.default();
  }

  async loadImage(src: string): Promise<void> {
    const img = new Image();
    img.src = src;
    await img.decode();
    this.canvas.width = img.width;
    this.canvas.height = img.height;
    this.ctx.drawImage(img, 0, 0);
    this.originalPixels = this.ctx
      .getImageData(0, 0, img.width, img.height)
      .data.slice() as unknown as Uint8ClampedArray;
  }

  applyFilter(filter: 'grayscale' | 'blur' | 'brighten', params?: any): void {
    if (!this.originalPixels) return;
    const pixels = new Uint8Array(this.originalPixels);

    switch (filter) {
      case 'grayscale':
        this.wasmModule.grayscale(pixels);
        break;
      case 'blur':
        this.wasmModule.box_blur(pixels, this.canvas.width, this.canvas.height, params?.radius ?? 3);
        break;
      case 'brighten':
        this.wasmModule.adjust_brightness(pixels, params?.factor ?? 1.2);
        break;
    }

    const imageData = new ImageData(new Uint8ClampedArray(pixels), this.canvas.width, this.canvas.height);
    this.ctx.putImageData(imageData, 0, 0);
  }
}

小结
    

  • 浏览器端图像处理采用 JS 解码 → WASM 处理 → Canvas 渲染 的架构。
    • 

  • 像素级操作中 WASM 比 JS 快 2-10 倍，计算复杂度越高优势越明显。
    • 

  • WASM SIMD 可同时处理 4 个像素（16 字节），需编译时启用 +simd128。
    • 

  • Canvas 的 getImageData/putImageData 实现 JS 与 WASM 之间的像素数据交换。
    • 

  • 对于耗时操作，使用 OffscreenCanvas + Worker 在后台线程处理，保持 UI 响应。
