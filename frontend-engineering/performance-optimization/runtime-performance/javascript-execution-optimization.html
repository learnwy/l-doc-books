JavaScript 执行优化

主线程瓶颈

JavaScript 在浏览器中是单线程执行的——主线程同时承担着 JavaScript 执行、样式计算、布局、绘制和用户事件处理等职责。当一个 JavaScript 任务执行时间过长（超过 50ms），就会成为"长任务"（Long Task），阻塞主线程，导致用户交互无法及时得到响应。



长任务拆分
使用 setTimeout 拆分
最简单的长任务拆分方式是使用 setTimeout(fn, 0) 将大任务分割为多个小任务，让浏览器有机会在任务间处理用户事件和渲染更新。


function processLargeArray<T>(items: T[], callback: (item: T) => void): Promise<void> {
  return new Promise((resolve) => {
    const CHUNK_SIZE = 100;
    let index = 0;

    function processChunk(): void {
      const end = Math.min(index + CHUNK_SIZE, items.length);
      for (; index < end; index++) {
        callback(items[index]);
      }
      if (index < items.length) {
        setTimeout(processChunk, 0);
      } else {
        resolve();
      }
    }

    processChunk();
  });
}
使用 scheduler.yield()
scheduler.yield() 是新的浏览器 API，专门为长任务拆分设计。它比 setTimeout(0) 更高效，因为它会将后续任务放回任务队列的前面，而不是末尾。


async function processItemsWithYield<T>(items: T[], process: (item: T) => void, deadline: number = 5): Promise<void> {
  let lastYield = performance.now();

  for (const item of items) {
    process(item);

    if (performance.now() - lastYield > deadline) {
      if ('scheduler' in globalThis && 'yield' in (globalThis as any).scheduler) {
        await (globalThis as any).scheduler.yield();
      } else {
        await new Promise<void>((r) => setTimeout(r, 0));
      }
      lastYield = performance.now();
    }
  }
}
使用 requestIdleCallback
requestIdleCallback 在浏览器空闲时执行低优先级任务，不会影响关键交互和渲染。


interface IdleTaskQueue {
  enqueue: (task: () => void) => void;
  clear: () => void;
}

function createIdleTaskQueue(): IdleTaskQueue {
  const tasks: Array<() => void> = [];
  let scheduled = false;

  function flush(deadline: IdleDeadline): void {
    while (tasks.length > 0 && deadline.timeRemaining() > 1) {
      const task = tasks.shift();
      task?.();
    }

    if (tasks.length > 0) {
      requestIdleCallback(flush);
    } else {
      scheduled = false;
    }
  }

  return {
    enqueue(task: () => void): void {
      tasks.push(task);
      if (!scheduled) {
        scheduled = true;
        requestIdleCallback(flush);
      }
    },
    clear(): void {
      tasks.length = 0;
      scheduled = false;
    },
  };
}

const queue = createIdleTaskQueue();

queue.enqueue(() => sendAnalyticsEvent('page_view'));
queue.enqueue(() => prefetchNextPageData());
queue.enqueue(() => precomputeSearchIndex());


Web Workers
Web Workers 在独立线程中运行 JavaScript，不阻塞主线程。对于 CPU 密集型计算（数据处理、加密、图像处理等），将其转移到 Worker 中是最有效的优化手段。


封装通用 Worker 通信
type WorkerTask<T, R> = {
  id: string;
  type: string;
  payload: T;
  resolve: (value: R) => void;
  reject: (reason: Error) => void;
};

class WorkerPool<T, R> {
  private workers: Worker[] = [];
  private taskQueue: WorkerTask<T, R>[] = [];
  private activeWorkers = new Map<Worker, WorkerTask<T, R>>();

  constructor(
    private workerUrl: string | URL,
    private poolSize: number = navigator.hardwareConcurrency || 4,
  ) {
    for (let i = 0; i < poolSize; i++) {
      const worker = new Worker(workerUrl);
      worker.onmessage = (e) => this.handleMessage(worker, e);
      worker.onerror = (e) => this.handleError(worker, e);
      this.workers.push(worker);
    }
  }

  execute(type: string, payload: T): Promise<R> {
    return new Promise((resolve, reject) => {
      const task: WorkerTask<T, R> = {
        id: crypto.randomUUID(),
        type,
        payload,
        resolve,
        reject,
      };
      this.taskQueue.push(task);
      this.processQueue();
    });
  }

  private processQueue(): void {
    const available = this.workers.find((w) => !this.activeWorkers.has(w));
    if (!available || this.taskQueue.length === 0) return;

    const task = this.taskQueue.shift()!;
    this.activeWorkers.set(available, task);
    available.postMessage({ id: task.id, type: task.type, payload: task.payload });
  }

  private handleMessage(worker: Worker, event: MessageEvent): void {
    const task = this.activeWorkers.get(worker);
    if (task) {
      task.resolve(event.data.result);
      this.activeWorkers.delete(worker);
      this.processQueue();
    }
  }

  private handleError(worker: Worker, event: ErrorEvent): void {
    const task = this.activeWorkers.get(worker);
    if (task) {
      task.reject(new Error(event.message));
      this.activeWorkers.delete(worker);
      this.processQueue();
    }
  }

  terminate(): void {
    this.workers.forEach((w) => w.terminate());
  }
}


事件处理优化
防抖（Debounce）
function debounce<T extends (...args: any[]) => void>(fn: T, delay: number): (...args: Parameters<T>) => void {
  let timer: ReturnType<typeof setTimeout>;

  return (...args: Parameters<T>) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}

const handleSearch = debounce((query: string) => {
  fetch(`/api/search?q=${encodeURIComponent(query)}`)
    .then((res) => res.json())
    .then((data) => renderResults(data));
}, 300);
节流（Throttle）
function throttle<T extends (...args: any[]) => void>(fn: T, interval: number): (...args: Parameters<T>) => void {
  let lastTime = 0;
  let timer: ReturnType<typeof setTimeout> | null = null;

  return (...args: Parameters<T>) => {
    const now = Date.now();
    const remaining = interval - (now - lastTime);

    if (remaining <= 0) {
      if (timer) {
        clearTimeout(timer);
        timer = null;
      }
      lastTime = now;
      fn(...args);
    } else if (!timer) {
      timer = setTimeout(() => {
        lastTime = Date.now();
        timer = null;
        fn(...args);
      }, remaining);
    }
  };
}

const handleScroll = throttle(() => {
  const scrollTop = window.scrollY;
  updateScrollProgress(scrollTop);
}, 16);
requestAnimationFrame 节流
function rafThrottle<T extends (...args: any[]) => void>(fn: T): (...args: Parameters<T>) => void {
  let rafId: number | null = null;

  return (...args: Parameters<T>) => {
    if (rafId !== null) return;

    rafId = requestAnimationFrame(() => {
      fn(...args);
      rafId = null;
    });
  };
}

const handleResize = rafThrottle(() => {
  recalculateLayout();
});

window.addEventListener('resize', handleResize);


计算结果缓存
Memoization
function memoize<Args extends any[], Result>(
  fn: (...args: Args) => Result,
  keyFn?: (...args: Args) => string,
): (...args: Args) => Result {
  const cache = new Map<string, Result>();

  return (...args: Args): Result => {
    const key = keyFn ? keyFn(...args) : JSON.stringify(args);

    if (cache.has(key)) {
      return cache.get(key)!;
    }

    const result = fn(...args);
    cache.set(key, result);
    return result;
  };
}

const expensiveSort = memoize(
  (items: number[], order: 'asc' | 'desc') => {
    console.log('Sorting...');
    return [...items].sort((a, b) => (order === 'asc' ? a - b : b - a));
  },
  (items, order) => `${items.join(',')}_${order}`,
);
LRU 缓存
class LRUCache<K, V> {
  private cache = new Map<K, V>();

  constructor(private maxSize: number) {}

  get(key: K): V | undefined {
    if (!this.cache.has(key)) return undefined;
    const value = this.cache.get(key)!;
    this.cache.delete(key);
    this.cache.set(key, value);
    return value;
  }

  set(key: K, value: V): void {
    if (this.cache.has(key)) {
      this.cache.delete(key);
    } else if (this.cache.size >= this.maxSize) {
      const firstKey = this.cache.keys().next().value;
      if (firstKey !== undefined) this.cache.delete(firstKey);
    }
    this.cache.set(key, value);
  }

  get size(): number {
    return this.cache.size;
  }
}


小结
    

  • 主线程上超过 50ms 的任务是"长任务"，会阻塞用户交互——必须拆分。
    • 

  • scheduler.yield() > setTimeout(0) > requestIdleCallback，根据任务优先级选择拆分方式。
    • 

  • CPU 密集型计算应该转移到 Web Workers 中执行。
    • 

  • 高频事件（scroll、resize、input）必须使用防抖、节流或 rAF 节流。
    • 

  • 计算结果缓存（Memoization/LRU Cache）避免重复的昂贵计算。
