内存管理与泄漏

JavaScript 内存模型

JavaScript 使用自动垃圾回收（Garbage Collection, GC）管理内存。引擎会自动追踪对象的引用关系，当一个对象不再被任何"根"对象可达时，就会被标记为垃圾并回收。理解引用链和可达性是检测内存泄漏的基础。

上图中，对象 E 和 F 虽然互相引用，但从 GC 根不可达，因此会被回收。现代 JavaScript 引擎使用标记-清除（Mark-and-Sweep）算法，已经能够处理循环引用问题。

常见内存泄漏模式
1. 未清理的事件监听器
function createComponent(element: HTMLElement): void {
  const handler = () => {
    console.log('clicked', element.id);
  };
  window.addEventListener('scroll', handler);
}

function createComponentFixed(element: HTMLElement): () => void {
  const handler = () => {
    console.log('clicked', element.id);
  };
  window.addEventListener('scroll', handler);

  return () => {
    window.removeEventListener('scroll', handler);
  };
}

3. 分离的 DOM 节点
let cachedElement: HTMLElement | null = null;

function showDialog(): void {
  const dialog = document.createElement('div');
  dialog.className = 'dialog';
  dialog.innerHTML = '<p>Dialog content</p>';
  document.body.appendChild(dialog);
  cachedElement = dialog;
}

function hideDialog(): void {
  if (cachedElement) {
    document.body.removeChild(cachedElement);
  }
}

function hideDialogFixed(): void {
  if (cachedElement) {
    document.body.removeChild(cachedElement);
    cachedElement = null;
  }
}
4. 定时器未清理
function startPolling(url: string): void {
  setInterval(async () => {
    const data = await fetch(url).then((r) => r.json());
    updateUI(data);
  }, 5000);
}

function startPollingFixed(url: string): () => void {
  const timer = setInterval(async () => {
    const data = await fetch(url).then((r) => r.json());
    updateUI(data);
  }, 5000);

  return () => clearInterval(timer);
}
5. 全局变量累积
const cache: Record<string, any> = {};

function fetchWithCache(url: string): Promise<any> {
  if (cache[url]) return Promise.resolve(cache[url]);
  return fetch(url)
    .then((r) => r.json())
    .then((data) => {
      cache[url] = data;
      return data;
    });
}

class BoundedCache<K, V> {
  private cache = new Map<K, V>();

  constructor(private maxSize: number = 100) {}

  get(key: K): V | undefined {
    return this.cache.get(key);
  }

  set(key: K, value: V): void {
    if (this.cache.size >= this.maxSize) {
      const firstKey = this.cache.keys().next().value;
      if (firstKey !== undefined) this.cache.delete(firstKey);
    }
    this.cache.set(key, value);
  }
}


React 中的内存泄漏
异步操作未取消
import { useState, useEffect } from 'react';

function UserProfile({ userId }: { userId: string }) {
  const [user, setUser] = useState<any>(null);

  useEffect(() => {
    const controller = new AbortController();

    fetch(`/api/users/${userId}`, { signal: controller.signal })
      .then((res) => res.json())
      .then((data) => setUser(data))
      .catch((err) => {
        if (err.name !== 'AbortError') console.error(err);
      });

    return () => controller.abort();
  }, [userId]);

  return user ? <div>{user.name}</div> : <div>Loading...</div>;
}
订阅未清理
import { useEffect, useRef } from 'react';

function useWebSocket(url: string, onMessage: (data: any) => void): void {
  const wsRef = useRef<WebSocket | null>(null);
  const callbackRef = useRef(onMessage);
  callbackRef.current = onMessage;

  useEffect(() => {
    const ws = new WebSocket(url);
    wsRef.current = ws;

    ws.onmessage = (event) => {
      callbackRef.current(JSON.parse(event.data));
    };

    return () => {
      ws.close();
      wsRef.current = null;
    };
  }, [url]);
}


内存泄漏检测
Chrome DevTools Memory 面板
编程式内存监控
interface MemorySnapshot {
  timestamp: number;
  usedJSHeapSize: number;
  totalJSHeapSize: number;
  jsHeapSizeLimit: number;
}

class MemoryMonitor {
  private snapshots: MemorySnapshot[] = [];
  private timer: ReturnType<typeof setInterval> | null = null;

  start(interval: number = 5000): void {
    this.timer = setInterval(() => {
      this.takeSnapshot();
    }, interval);
  }

  stop(): void {
    if (this.timer) {
      clearInterval(this.timer);
      this.timer = null;
    }
  }

  private takeSnapshot(): void {
    if (!('memory' in performance)) return;

    const memory = (performance as any).memory;
    this.snapshots.push({
      timestamp: Date.now(),
      usedJSHeapSize: memory.usedJSHeapSize,
      totalJSHeapSize: memory.totalJSHeapSize,
      jsHeapSizeLimit: memory.jsHeapSizeLimit,
    });

    this.checkForLeaks();
  }

  private checkForLeaks(): void {
    if (this.snapshots.length < 10) return;

    const recent = this.snapshots.slice(-10);
    const trend = recent.map((s) => s.usedJSHeapSize);
    const isIncreasing = trend.every((val, i) => i === 0 || val >= trend[i - 1]);

    if (isIncreasing) {
      const growth = trend[trend.length - 1] - trend[0];
      const growthMB = (growth / 1024 / 1024).toFixed(2);
      console.warn(`Potential memory leak: heap grew by ${growthMB} MB over last ${recent.length} snapshots`);
    }
  }

  getReport(): { current: string; peak: string; snapshots: number } {
    const current = this.snapshots.at(-1);
    const peak = Math.max(...this.snapshots.map((s) => s.usedJSHeapSize));

    return {
      current: current ? `${(current.usedJSHeapSize / 1024 / 1024).toFixed(2)} MB` : 'N/A',
      peak: `${(peak / 1024 / 1024).toFixed(2)} MB`,
      snapshots: this.snapshots.length,
    };
  }
}
WeakRef 与 FinalizationRegistry
const leakDetector = new FinalizationRegistry<string>((label) => {
  console.log(`Object "${label}" was garbage collected`);
});

function trackObject(obj: object, label: string): void {
  leakDetector.register(obj, label);
}

class WeakCache<K extends object, V> {
  private cache = new Map<string, WeakRef<V & object>>();
  private registry = new FinalizationRegistry<string>((key) => {
    this.cache.delete(key);
  });

  set(key: string, value: V & object): void {
    const ref = new WeakRef(value);
    this.cache.set(key, ref);
    this.registry.register(value, key);
  }

  get(key: string): V | undefined {
    const ref = this.cache.get(key);
    if (!ref) return undefined;
    const value = ref.deref();
    if (!value) {
      this.cache.delete(key);
      return undefined;
    }
    return value;
  }
}

小结
    

  • JavaScript 使用标记-清除算法进行垃圾回收，理解对象可达性是内存管理的基础。
    • 

  • 五大常见泄漏：未清理的事件监听器、闭包引用、分离的 DOM 节点、定时器、全局变量累积。
    • 

  • React 中最常见的泄漏是组件卸载后异步操作仍尝试更新状态——使用 AbortController 解决。
    • 

  • Chrome DevTools Memory 面板的 Heap Snapshot 对比是检测内存泄漏的最有效工具。
    • 

  • WeakRef 和 FinalizationRegistry 提供了更精细的内存管理能力。
