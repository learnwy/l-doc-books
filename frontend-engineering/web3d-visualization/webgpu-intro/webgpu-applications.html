WebGPU 应用实践

核心概念

WebGPU 的应用范围远超传统的 3D 图形渲染。凭借原生计算着色器支持和现代 GPU API 的设计优势，WebGPU 正在重塑浏览器端高性能计算的格局。从大规模数据可视化到机器学习推理，从物理模拟到科学计算，WebGPU 为 Web 应用带来了前所未有的计算能力。

WebGPU 的实际价值在于将原本需要服务器端 GPU 或原生应用才能完成的计算，带到了浏览器中——零安装、跨平台、即时可用。


大规模数据可视化
WebGPU 的高 draw call 能力和计算着色器使得百万级数据点的实时可视化成为可能。


const createScatterPlot = async (
  device: GPUDevice,
  context: GPUCanvasContext,
  format: GPUTextureFormat,
  points: Float32Array,
): Promise<{ render: () => void }> => {
  const vertexBuffer = device.createBuffer({
    size: points.byteLength,
    usage: GPUBufferUsage.VERTEX | GPUBufferUsage.COPY_DST,
  });
  device.queue.writeBuffer(vertexBuffer, 0, points);

  const shaderModule = device.createShaderModule({
    code: `
      struct Uniforms {
        viewMatrix: mat4x4<f32>,
        pointSize: f32,
      };
      @group(0) @binding(0) var<uniform> uniforms: Uniforms;

      struct VertexInput {
        @location(0) position: vec2<f32>,
        @location(1) value: f32,
      };

      struct VertexOutput {
        @builtin(position) position: vec4<f32>,
        @location(0) color: vec3<f32>,
        @builtin(point_size) pointSize: f32,
      };

      fn valueToColor(v: f32) -> vec3<f32> {
        let r = clamp(v * 2.0, 0.0, 1.0);
        let g = clamp(2.0 - v * 2.0, 0.0, 1.0);
        let b = 0.3;
        return vec3<f32>(r, g, b);
      }

      @vertex
      fn vs_main(input: VertexInput) -> VertexOutput {
        var output: VertexOutput;
        output.position = uniforms.viewMatrix * vec4<f32>(input.position, 0.0, 1.0);
        output.color = valueToColor(input.value);
        output.pointSize = uniforms.pointSize;
        return output;
      }

      @fragment
      fn fs_main(input: VertexOutput) -> @location(0) vec4<f32> {
        return vec4<f32>(input.color, 0.7);
      }
    `,
  });

  const pipeline = device.createRenderPipeline({
    layout: 'auto',
    vertex: {
      module: shaderModule,
      entryPoint: 'vs_main',
      buffers: [
        {
          arrayStride: 3 * 4,
          attributes: [
            { shaderLocation: 0, offset: 0, format: 'float32x2' },
            { shaderLocation: 1, offset: 8, format: 'float32' },
          ],
        },
      ],
    },
    fragment: {
      module: shaderModule,
      entryPoint: 'fs_main',
      targets: [{ format }],
    },
    primitive: { topology: 'point-list' },
  });

  const pointCount = points.length / 3;

  return {
    render: () => {
      const encoder = device.createCommandEncoder();
      const pass = encoder.beginRenderPass({
        colorAttachments: [
          {
            view: context.getCurrentTexture().createView(),
            loadOp: 'clear',
            storeOp: 'store',
            clearValue: { r: 0.1, g: 0.1, b: 0.15, a: 1.0 },
          },
        ],
      });

      pass.setPipeline(pipeline);
      pass.setVertexBuffer(0, vertexBuffer);
      pass.draw(pointCount);
      pass.end();

      device.queue.submit([encoder.finish()]);
    },
  };
};
GPU 粒子系统
粒子系统是计算着色器最直观的应用——每个粒子的位置、速度和生命周期在 GPU 上并行计算。


const particleComputeShader = `
struct Particle {
  position: vec2<f32>,
  velocity: vec2<f32>,
  life: f32,
  padding: f32,
};

struct SimParams {
  deltaTime: f32,
  gravity: f32,
  damping: f32,
  particleCount: u32,
};

@group(0) @binding(0) var<uniform> params: SimParams;
@group(0) @binding(1) var<storage, read_write> particles: array<Particle>;

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) id: vec3<u32>) {
  let index = id.x;
  if (index >= params.particleCount) { return; }

  var p = particles[index];

  p.velocity.y -= params.gravity * params.deltaTime;
  p.velocity *= params.damping;
  p.position += p.velocity * params.deltaTime;
  p.life -= params.deltaTime;

  if (p.position.y < -1.0) {
    p.position.y = -1.0;
    p.velocity.y *= -0.6;
  }
  if (p.position.x < -1.0 || p.position.x > 1.0) {
    p.velocity.x *= -1.0;
  }

  particles[index] = p;
}
`;

浏览器端 ML 推理
WebGPU 为浏览器端机器学习推理提供了原生加速能力，多个主流框架已支持 WebGPU 后端。


interface TensorBuffer {
  buffer: GPUBuffer;
  shape: number[];
  dtype: 'float32' | 'float16';
}

const createTensorBuffer = (device: GPUDevice, data: Float32Array, shape: number[]): TensorBuffer => {
  const buffer = device.createBuffer({
    size: data.byteLength,
    usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST | GPUBufferUsage.COPY_SRC,
  });
  device.queue.writeBuffer(buffer, 0, data);
  return { buffer, shape, dtype: 'float32' };
};

const reluShader = `
@group(0) @binding(0) var<storage, read_write> data: array<f32>;

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) id: vec3<u32>) {
  let index = id.x;
  if (index >= arrayLength(&data)) { return; }
  data[index] = max(data[index], 0.0);
}
`;
