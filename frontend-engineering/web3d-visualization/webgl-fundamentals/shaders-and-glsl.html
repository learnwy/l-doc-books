着色器与 GLSL

核心概念

着色器（Shader）是运行在 GPU 上的程序，使用 GLSL（OpenGL Shading Language）编写。WebGL 中有两种主要着色器：顶点着色器负责变换顶点位置，片段着色器负责计算像素颜色。GLSL 是一种类 C 语言，专为并行图形计算设计。

每个顶点独立执行顶点着色器，每个片段独立执行片段着色器，这种并行特性使得 GPU 能以极高的吞吐量处理图形数据。

GLSL 数据类型与运算
GLSL 提供了丰富的向量和矩阵类型，这些类型直接映射到 GPU 寄存器，运算效率极高。


基本标量类型包括 float、int、bool。向量类型 vec2、vec3、vec4 是最常用的，支持分量访问（swizzling）：


const swizzleDemo = `
  vec4 color = vec4(1.0, 0.5, 0.3, 1.0);
  vec3 rgb = color.rgb;          // (1.0, 0.5, 0.3)
  vec2 rg = color.rg;            // (1.0, 0.5)
  float r = color.r;             // 1.0
  vec3 bgr = color.bgr;          // (0.3, 0.5, 1.0) 重排
  vec4 rrbb = color.rrbb;        // (1.0, 1.0, 0.3, 0.3) 重复

  // 位置分量别名：xyzw, stpq, rgba 三套可互换
  vec3 pos = vec3(1.0, 2.0, 3.0);
  float y = pos.y;               // 等同于 pos.g 或 pos.t
`;
矩阵类型 mat2、mat3、mat4 按列主序存储，与 JavaScript 的 TypedArray 对应：


const matrixDemo = `
  mat4 model = mat4(1.0);  // 单位矩阵

  // 矩阵与向量乘法
  vec4 worldPos = model * vec4(position, 1.0);

  // 矩阵分量访问
  vec4 column0 = model[0];  // 第一列
  float m01 = model[0][1];  // 第一列第二行
`;
GLSL 内置了大量数学函数，直接在 GPU 硬件上执行：


const builtinFunctions = `
  // 常用数学函数
  float a = mix(0.0, 1.0, 0.5);      // 线性插值 → 0.5
  float b = clamp(1.5, 0.0, 1.0);    // 钳制 → 1.0
  float c = smoothstep(0.0, 1.0, x);  // 平滑插值
  float d = step(0.5, x);             // 阶跃函数
  float e = fract(2.7);               // 小数部分 → 0.7

  // 向量运算
  float dp = dot(v1, v2);             // 点积
  vec3 cp = cross(v1, v2);            // 叉积
  vec3 n = normalize(v);              // 归一化
  float len = length(v);              // 长度
  vec3 r = reflect(I, N);             // 反射
  vec3 t = refract(I, N, eta);        // 折射
`;
顶点着色器编程
顶点着色器的主要职责是将模型空间的顶点变换到裁剪空间。典型的变换链是：模型矩阵 → 视图矩阵 → 投影矩阵。


const phongVertexShader = `#version 300 es
  in vec3 a_position;
  in vec3 a_normal;
  in vec2 a_texCoord;

  uniform mat4 u_model;
  uniform mat4 u_view;
  uniform mat4 u_projection;
  uniform mat3 u_normalMatrix;

  out vec3 v_worldPosition;
  out vec3 v_normal;
  out vec2 v_texCoord;

  void main() {
    vec4 worldPos = u_model * vec4(a_position, 1.0);
    v_worldPosition = worldPos.xyz;
    v_normal = normalize(u_normalMatrix * a_normal);
    v_texCoord = a_texCoord;
    gl_Position = u_projection * u_view * worldPos;
  }
`;
法线矩阵（Normal Matrix）是模型矩阵逆转置的 3x3 部分，确保非均匀缩放下法线方向正确：


const calculateNormalMatrix = (modelMatrix: Float32Array): Float32Array => {
  const m = new Float32Array(9);
  const det =
    modelMatrix[0] * (modelMatrix[5] * modelMatrix[10] - modelMatrix[6] * modelMatrix[9]) -
    modelMatrix[1] * (modelMatrix[4] * modelMatrix[10] - modelMatrix[6] * modelMatrix[8]) +
    modelMatrix[2] * (modelMatrix[4] * modelMatrix[9] - modelMatrix[5] * modelMatrix[8]);

  const invDet = 1.0 / det;
  m[0] = (modelMatrix[5] * modelMatrix[10] - modelMatrix[6] * modelMatrix[9]) * invDet;
  m[1] = (modelMatrix[2] * modelMatrix[9] - modelMatrix[1] * modelMatrix[10]) * invDet;
  m[2] = (modelMatrix[1] * modelMatrix[6] - modelMatrix[2] * modelMatrix[5]) * invDet;
  m[3] = (modelMatrix[6] * modelMatrix[8] - modelMatrix[4] * modelMatrix[10]) * invDet;
  m[4] = (modelMatrix[0] * modelMatrix[10] - modelMatrix[2] * modelMatrix[8]) * invDet;
  m[5] = (modelMatrix[2] * modelMatrix[4] - modelMatrix[0] * modelMatrix[6]) * invDet;
  m[6] = (modelMatrix[4] * modelMatrix[9] - modelMatrix[5] * modelMatrix[8]) * invDet;
  m[7] = (modelMatrix[1] * modelMatrix[8] - modelMatrix[0] * modelMatrix[9]) * invDet;
  m[8] = (modelMatrix[0] * modelMatrix[5] - modelMatrix[1] * modelMatrix[4]) * invDet;
  return m;
};
片段着色器与光照模型
片段着色器决定每个像素的最终颜色。Phong 光照模型是最经典的实时光照模型，由环境光、漫反射和镜面反射三个分量组成。


const phongFragmentShader = `#version 300 es
  precision highp float;

  in vec3 v_worldPosition;
  in vec3 v_normal;
  in vec2 v_texCoord;

  uniform vec3 u_lightPosition;
  uniform vec3 u_cameraPosition;
  uniform vec3 u_lightColor;
  uniform sampler2D u_diffuseMap;

  out vec4 fragColor;

  void main() {
    vec3 normal = normalize(v_normal);
    vec3 lightDir = normalize(u_lightPosition - v_worldPosition);
    vec3 viewDir = normalize(u_cameraPosition - v_worldPosition);
    vec3 halfDir = normalize(lightDir + viewDir);

    // 环境光
    vec3 ambient = 0.1 * u_lightColor;

    // 漫反射（Lambert）
    float diff = max(dot(normal, lightDir), 0.0);
    vec3 diffuse = diff * u_lightColor;

    // 镜面反射（Blinn-Phong）
    float spec = pow(max(dot(normal, halfDir), 0.0), 32.0);
    vec3 specular = spec * u_lightColor;

    vec3 texColor = texture(u_diffuseMap, v_texCoord).rgb;
    vec3 result = (ambient + diffuse) * texColor + specular;
    fragColor = vec4(result, 1.0);
  }
`;
预处理器与着色器组织
GLSL 支持 C 风格的预处理器指令，可用于条件编译和代码复用。在大型项目中，着色器代码的组织和复用至关重要。


const shaderChunks: Record<string, string> = {
  lighting: `
    vec3 calcPointLight(vec3 pos, vec3 normal, vec3 lightPos, vec3 lightColor) {
      vec3 dir = normalize(lightPos - pos);
      float distance = length(lightPos - pos);
      float attenuation = 1.0 / (1.0 + 0.09 * distance + 0.032 * distance * distance);
      float diff = max(dot(normal, dir), 0.0);
      return diff * lightColor * attenuation;
    }
  `,
  fog: `
    vec3 applyFog(vec3 color, float dist) {
      float fogFactor = exp(-dist * 0.01);
      vec3 fogColor = vec3(0.7, 0.8, 0.9);
      return mix(fogColor, color, clamp(fogFactor, 0.0, 1.0));
    }
  `,
};

const assembleShader = (template: string, chunks: Record<string, string>): string => {
  return template.replace(/#include\s+<(\w+)>/g, (_, name) => {
    const chunk = chunks[name];
    if (!chunk) throw new Error(`Unknown shader chunk: ${name}`);
    return chunk;
  });
};

const mainShader = `#version 300 es
  precision highp float;
  #include <lighting>
  #include <fog>

  void main() {
    // 使用 chunk 中定义的函数
  }
`;
利用预处理器实现多变体着色器：


const buildVariant = (defines: Record<string, boolean | number>): string => {
  return Object.entries(defines)
    .map(([key, value]) => {
      if (typeof value === 'boolean') {
        return value ? `#define ${key}` : '';
      }
      return `#define ${key} ${value}`;
    })
    .filter(Boolean)
    .join('\n');
};
