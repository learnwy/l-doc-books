缓冲区与纹理

核心概念

缓冲区（Buffer）和纹理（Texture）是 WebGL 中两种最重要的 GPU 内存资源。缓冲区存储顶点数据和索引数据，纹理存储图像数据和采样信息。高效管理这两种资源直接决定了渲染性能。

WebGL 中的数据流是单向的：JavaScript 通过 API 将数据上传到 GPU 显存，GPU 在渲染时读取这些数据。理解缓冲区类型和纹理格式是优化渲染性能的关键。

顶点缓冲区与数据布局
顶点数据可以使用两种布局方式：交错布局（Interleaved）将所有属性紧密排列，分离布局（Separate）将每种属性存储在独立缓冲区中。交错布局对缓存更友好。


interface VertexAttribute {
  name: string;
  size: number;
  type: number;
  normalized: boolean;
  offset: number;
}

const createInterleavedBuffer = (
  gl: WebGL2RenderingContext,
  data: Float32Array,
  attributes: VertexAttribute[],
  program: WebGLProgram,
): WebGLBuffer => {
  const buffer = gl.createBuffer();
  if (!buffer) throw new Error('Failed to create buffer');

  gl.bindBuffer(gl.ARRAY_BUFFER, buffer);
  gl.bufferData(gl.ARRAY_BUFFER, data, gl.STATIC_DRAW);

  const stride = attributes.reduce((sum, attr) => sum + attr.size * 4, 0);

  for (const attr of attributes) {
    const location = gl.getAttribLocation(program, attr.name);
    if (location === -1) continue;

    gl.enableVertexAttribArray(location);
    gl.vertexAttribPointer(location, attr.size, attr.type, attr.normalized, stride, attr.offset);
  }

  return buffer;
};

纹理创建与采样
纹理是 GPU 中最重要的图像资源。WebGL 2.0 支持多种纹理类型和格式，正确配置纹理参数对渲染质量影响巨大。


const loadTexture = async (gl: WebGL2RenderingContext, url: string): Promise<WebGLTexture> => {
  const texture = gl.createTexture();
  if (!texture) throw new Error('Failed to create texture');

  gl.bindTexture(gl.TEXTURE_2D, texture);
  gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, 1, 1, 0, gl.RGBA, gl.UNSIGNED_BYTE, new Uint8Array([128, 128, 128, 255]));

  const image = new Image();
  image.crossOrigin = 'anonymous';

  return new Promise((resolve, reject) => {
    image.onload = () => {
      gl.bindTexture(gl.TEXTURE_2D, texture);
      gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, image);

      if (isPowerOf2(image.width) && isPowerOf2(image.height)) {
        gl.generateMipmap(gl.TEXTURE_2D);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR);
      } else {
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
      }
      gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);

      resolve(texture);
    };
    image.onerror = reject;
    image.src = url;
  });
};

const isPowerOf2 = (value: number): boolean => (value & (value - 1)) === 0;
纹理采样参数控制着放大/缩小时的插值方式和边缘处理：


const configureSampler = (gl: WebGL2RenderingContext): WebGLSampler => {
  const sampler = gl.createSampler();
  if (!sampler) throw new Error('Failed to create sampler');

  gl.samplerParameteri(sampler, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR);
  gl.samplerParameteri(sampler, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
  gl.samplerParameteri(sampler, gl.TEXTURE_WRAP_S, gl.REPEAT);
  gl.samplerParameteri(sampler, gl.TEXTURE_WRAP_T, gl.REPEAT);
  gl.samplerParameterf(sampler, gl.TEXTURE_MAX_ANISOTROPY_EXT, 16);

  return sampler;
};
统一缓冲对象与批量传输
WebGL 2.0 引入的 UBO（Uniform Buffer Object）允许将多个 uniform 变量打包到一个缓冲区中，减少 API 调用次数并支持多程序共享。


interface CameraUniforms {
  viewMatrix: Float32Array;
  projectionMatrix: Float32Array;
  cameraPosition: Float32Array;
  nearFar: Float32Array;
}

const createUBO = (
  gl: WebGL2RenderingContext,
  program: WebGLProgram,
  blockName: string,
  bindingPoint: number,
): WebGLBuffer => {
  const blockIndex = gl.getUniformBlockIndex(program, blockName);
  gl.uniformBlockBinding(program, blockIndex, bindingPoint);

  const blockSize = gl.getActiveUniformBlockParameter(program, blockIndex, gl.UNIFORM_BLOCK_DATA_SIZE);

  const buffer = gl.createBuffer();
  if (!buffer) throw new Error('Failed to create UBO');

  gl.bindBuffer(gl.UNIFORM_BUFFER, buffer);
  gl.bufferData(gl.UNIFORM_BUFFER, blockSize, gl.DYNAMIC_DRAW);
  gl.bindBufferBase(gl.UNIFORM_BUFFER, bindingPoint, buffer);

  return buffer;
};

const updateCameraUBO = (gl: WebGL2RenderingContext, buffer: WebGLBuffer, camera: CameraUniforms) => {
  gl.bindBuffer(gl.UNIFORM_BUFFER, buffer);
  gl.bufferSubData(gl.UNIFORM_BUFFER, 0, camera.viewMatrix);
  gl.bufferSubData(gl.UNIFORM_BUFFER, 64, camera.projectionMatrix);
  gl.bufferSubData(gl.UNIFORM_BUFFER, 128, camera.cameraPosition);
  gl.bufferSubData(gl.UNIFORM_BUFFER, 144, camera.nearFar);
};
UBO 的内存布局遵循 std140 规则，需要注意对齐要求：标量 4 字节对齐，vec3 按 16 字节对齐，矩阵按列存储且每列 16 字节对齐。


压缩纹理与内存优化
对于大型 3D 应用，纹理内存占用是最大的性能瓶颈。压缩纹理直接在 GPU 解码，不需要解压到 RGBA 格式。


const loadCompressedTexture = async (gl: WebGL2RenderingContext, url: string): Promise<WebGLTexture> => {
  const ext =
    gl.getExtension('WEBGL_compressed_texture_s3tc') ||
    gl.getExtension('WEBGL_compressed_texture_etc') ||
    gl.getExtension('WEBGL_compressed_texture_astc');

  if (!ext) throw new Error('No compressed texture support');

  const response = await fetch(url);
  const buffer = await response.arrayBuffer();
  const header = parseKTXHeader(buffer);

  const texture = gl.createTexture();
  if (!texture) throw new Error('Failed to create texture');

  gl.bindTexture(gl.TEXTURE_2D, texture);

  for (let level = 0; level < header.mipLevels; level++) {
    const levelData = extractMipLevel(buffer, header, level);
    gl.compressedTexImage2D(
      gl.TEXTURE_2D,
      level,
      header.internalFormat,
      header.width >> level,
      header.height >> level,
      0,
      levelData,
    );
  }

  return texture;
};

interface KTXHeader {
  width: number;
  height: number;
  mipLevels: number;
  internalFormat: number;
}

const parseKTXHeader = (buffer: ArrayBuffer): KTXHeader => {
  const view = new DataView(buffer);
  return {
    width: view.getUint32(36, true),
    height: view.getUint32(40, true),
    mipLevels: view.getUint32(56, true),
    internalFormat: view.getUint32(28, true),
  };
};

const extractMipLevel = (buffer: ArrayBuffer, header: KTXHeader, level: number): Uint8Array => {
  let offset = 64 + header.mipLevels * 4;
  for (let i = 0; i < level; i++) {
    const size = new DataView(buffer).getUint32(64 + i * 4, true);
    offset += size;
  }
  const size = new DataView(buffer).getUint32(64 + level * 4, true);
  return new Uint8Array(buffer, offset, size);
};
小结
    

  • 顶点缓冲区支持交错和分离两种布局，交错布局对 GPU 缓存更友好
    • 

  • 纹理加载需处理 2 的幂尺寸限制，Mipmap 是提升远距离纹理质量的关键
    • 

  • WebGL 2.0 的 UBO 减少 uniform 更新的 API 调用，支持多着色器程序共享
    • 

  • 压缩纹理（S3TC/ETC/ASTC）在 GPU 直接解码，节省 4-8 倍显存
    • 

  • 缓冲区的 usage hint 帮助驱动优化内存分配策略
