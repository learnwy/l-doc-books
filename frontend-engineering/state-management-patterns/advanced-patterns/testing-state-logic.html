测试状态逻辑

核心概念

测试状态逻辑确保状态管理按预期工作，防止回归问题，提升代码质量。

为什么测试状态逻辑

原因 说明
防止回归 修改代码后确保功能正常
文档作用 测试用例说明预期行为
重构信心 有测试可以大胆重构
设计反馈 难测试的代码通常设计有问题

测试纯函数

测试 Reducer

import { describe, it, expect } from 'vitest';
import counterReducer, { increment, decrement, incrementByAmount } from './counterSlice';

describe('counter reducer', () => {
  const initialState = { value: 0, status: 'idle', error: null };

  it('should handle initial state', () => {
    expect(counterReducer(undefined, { type: 'unknown' })).toEqual(initialState);
  });

  it('should handle increment', () => {
    const actual = counterReducer(initialState, increment());
    expect(actual.value).toEqual(1);
  });

  it('should handle decrement', () => {
    const state = { ...initialState, value: 3 };
    const actual = counterReducer(state, decrement());
    expect(actual.value).toEqual(2);
  });

  it('should handle incrementByAmount', () => {
    const actual = counterReducer(initialState, incrementByAmount(5));
    expect(actual.value).toEqual(5);
  });
});

测试选择器

import { describe, it, expect } from 'vitest';
import { selectPostsWithAuthors, selectPostWithAuthor } from './selectors';

describe('posts selectors', () => {
  const mockState = {
    posts: {
      byId: {
        '1': { id: '1', title: 'Post 1', authorId: '1' },
      },
      allIds: ['1'],
    },
    users: {
      byId: {
        '1': { id: '1', name: 'Alice' },
      },
      allIds: ['1'],
    },
  };

  it('should select posts with authors', () => {
    const posts = selectPostsWithAuthors(mockState);
    expect(posts).toHaveLength(1);
    expect(posts[0].author).toEqual({ id: '1', name: 'Alice' });
  });

  it('should select single post with author', () => {
    const post = selectPostWithAuthor('1')(mockState);
    expect(post?.title).toEqual('Post 1');
    expect(post?.author?.name).toEqual('Alice');
  });
});

测试 React 组件

使用 React Testing Library

import { describe, it, expect } from 'vitest'
import { render, screen, fireEvent } from '@testing-library/react'
import { Provider } from 'react-redux'
import { configureStore } from '@reduxjs/toolkit'
import counterReducer from './counterSlice'
import Counter from './Counter'

function renderWithProviders(
  ui: React.ReactElement,
  {
    preloadedState = {},
    store = configureStore({
      reducer: { counter: counterReducer },
      preloadedState
    })
  } = {}
) {
  function Wrapper({ children }: { children: React.ReactNode }) {
    return <Provider store={store}>{children}</Provider>
  }

  return { store, ...render(ui, { wrapper: Wrapper }) }
}

describe('Counter', () => {
  it('should render initial count', () => {
    renderWithProviders(<Counter />)
    expect(screen.getByText(/count:/i)).toHaveTextContent('Count: 0')
  })

  it('should increment counter', () => {
    renderWithProviders(<Counter />)
    fireEvent.click(screen.getByText('+'))
    expect(screen.getByText(/count:/i)).toHaveTextContent('Count: 1')
  })

  it('should decrement counter', () => {
    renderWithProviders(<Counter />, {
      preloadedState: { counter: { value: 5, status: 'idle', error: null } }
    })
    fireEvent.click(screen.getByText('-'))
    expect(screen.getByText(/count:/i)).toHaveTextContent('Count: 4')
  })
})

测试异步逻辑

测试 Thunk

import { describe, it, expect, vi } from 'vitest';
import { configureStore } from '@reduxjs/toolkit';
import userReducer, { fetchUser } from './userSlice';

const mockUser = { id: '1', name: 'Alice', email: 'alice@example.com' };

global.fetch = vi.fn(() =>
  Promise.resolve({
    ok: true,
    json: () => Promise.resolve(mockUser),
  }),
) as jest.Mock;

describe('user thunks', () => {
  it('should fetch user successfully', async () => {
    const store = configureStore({ reducer: { user: userReducer } });

    await store.dispatch(fetchUser('1'));

    const state = store.getState().user;
    expect(state.status).toEqual('succeeded');
    expect(state.user).toEqual(mockUser);
  });

  it('should handle fetch error', async () => {
    (global.fetch as jest.Mock).mockImplementationOnce(() =>
      Promise.resolve({
        ok: false,
      }),
    );

    const store = configureStore({ reducer: { user: userReducer } });

    await store.dispatch(fetchUser('1'));

    const state = store.getState().user;
    expect(state.status).toEqual('failed');
    expect(state.error).toBeDefined();
  });
});

测试 Zustand Store

import { describe, it, expect, vi } from 'vitest';
import { act } from '@testing-library/react';
import { useCounterStore } from './store';

describe('counter store', () => {
  it('should increment counter', () => {
    const { result } = renderHook(() => useCounterStore());

    act(() => {
      result.current.increment();
    });

    expect(result.current.count).toBe(1);
  });

  it('should decrement counter', () => {
    const { result } = renderHook(() => useCounterStore());

    act(() => {
      result.current.increment();
      result.current.increment();
      result.current.decrement();
    });

    expect(result.current.count).toBe(1);
  });
});

常见误区

❌ 误区 1：测试实现细节

测试行为，不测试实现细节。

❌ 误区 2：过度 mock

不要 mock 太多，保持测试接近真实场景。

❌ 误区 3：不测试边界情况

测试空值、错误、边界条件。

小结

    

  • Reducer 是纯函数，最容易测试——输入确定，输出确定
    • 

  • Store 集成测试验证多个 Action 的组合行为
    • 

  • Selector 测试关注派生值的正确性和缓存行为
    • 

  • Hook 测试使用 renderHook 隔离测试 Hook 逻辑
