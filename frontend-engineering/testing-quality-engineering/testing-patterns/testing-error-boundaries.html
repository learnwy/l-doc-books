错误边界测试

核心概念

错误边界（Error Boundary）是 React 应用的防御性机制，捕获子组件树中的 JavaScript 错误并展示降级 UI，防止整个应用崩溃。测试错误边界确保应用在异常情况下的韧性——正确捕获错误、显示友好的降级界面、支持错误恢复。

实现错误边界
import { Component, type ReactNode, type ErrorInfo } from 'react'

interface ErrorBoundaryProps {
  fallback?: ReactNode | ((error: Error, reset: () => void) => ReactNode)
  onError?: (error: Error, errorInfo: ErrorInfo) => void
  children: ReactNode
}

interface ErrorBoundaryState {
  hasError: boolean
  error: Error | null
}

class ErrorBoundary extends Component<ErrorBoundaryProps, ErrorBoundaryState> {
  state: ErrorBoundaryState = { hasError: false, error: null }

  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo): void {
    this.props.onError?.(error, errorInfo)
  }

  reset = (): void => {
    this.setState({ hasError: false, error: null })
  }

  render(): ReactNode {
    if (this.state.hasError && this.state.error) {
      if (typeof this.props.fallback === 'function') {
        return this.props.fallback(this.state.error, this.reset)
      }
      return this.props.fallback || <div role="alert">出错了</div>
    }
    return this.props.children
  }
}
基础错误边界测试
import { render, screen } from '@testing-library/react'
import { describe, it, expect, vi, beforeEach } from 'vitest'

function BrokenComponent(): JSX.Element {
  throw new Error('组件渲染崩溃')
}

function GoodComponent(): JSX.Element {
  return <div>正常内容</div>
}

describe('ErrorBoundary 基础测试', () => {
  beforeEach(() => {
    vi.spyOn(console, 'error').mockImplementation(() => {})
  })

  it('无错误时正常渲染子组件', () => {
    render(
      <ErrorBoundary>
        <GoodComponent />
      </ErrorBoundary>
    )

    expect(screen.getByText('正常内容')).toBeInTheDocument()
  })

  it('子组件错误时显示默认降级 UI', () => {
    render(
      <ErrorBoundary>
        <BrokenComponent />
      </ErrorBoundary>
    )

    expect(screen.getByRole('alert')).toBeInTheDocument()
    expect(screen.getByText('出错了')).toBeInTheDocument()
    expect(screen.queryByText('正常内容')).not.toBeInTheDocument()
  })

  it('显示自定义降级 UI', () => {
    render(
      <ErrorBoundary fallback={<div role="alert">系统维护中，请稍后再试</div>}>
        <BrokenComponent />
      </ErrorBoundary>
    )

    expect(screen.getByText('系统维护中，请稍后再试')).toBeInTheDocument()
  })
})
测试错误上报
import { render } from '@testing-library/react'
import { describe, it, expect, vi, beforeEach } from 'vitest'

function ThrowWithMessage({ message }: { message: string }): JSX.Element {
  throw new Error(message)
}

describe('错误上报', () => {
  beforeEach(() => {
    vi.spyOn(console, 'error').mockImplementation(() => {})
  })

  it('应调用 onError 回调', () => {
    const onError = vi.fn()

    render(
      <ErrorBoundary onError={onError}>
        <ThrowWithMessage message="测试错误" />
      </ErrorBoundary>
    )

    expect(onError).toHaveBeenCalledOnce()
    expect(onError).toHaveBeenCalledWith(
      expect.objectContaining({ message: '测试错误' }),
      expect.objectContaining({ componentStack: expect.any(String) })
    )
  })

  it('onError 接收组件堆栈信息', () => {
    const onError = vi.fn()

    render(
      <ErrorBoundary onError={onError}>
        <ThrowWithMessage message="堆栈测试" />
      </ErrorBoundary>
    )

    const [, errorInfo] = onError.mock.calls[0]
    expect(errorInfo.componentStack).toBeDefined()
    expect(typeof errorInfo.componentStack).toBe('string')
  })
})
测试错误恢复
import { render, screen, fireEvent } from '@testing-library/react'
import { describe, it, expect, vi, beforeEach } from 'vitest'
import { useState } from 'react'

function ConditionalCrash({ shouldCrash }: { shouldCrash: boolean }): JSX.Element {
  if (shouldCrash) throw new Error('条件崩溃')
  return <div>正常内容</div>
}

function RecoverableApp(): JSX.Element {
  const [crashed, setCrashed] = useState(true)

  return (
    <ErrorBoundary
      fallback={(error, reset) => (
        <div role="alert">
          <p>错误: {error.message}</p>
          <button
            onClick={() => {
              setCrashed(false)
              reset()
            }}
          >
            重试
          </button>
        </div>
      )}
    >
      <ConditionalCrash shouldCrash={crashed} />
    </ErrorBoundary>
  )
}

describe('错误恢复', () => {
  beforeEach(() => {
    vi.spyOn(console, 'error').mockImplementation(() => {})
  })

  it('render prop 降级 UI 显示错误信息', () => {
    render(<RecoverableApp />)

    expect(screen.getByRole('alert')).toBeInTheDocument()
    expect(screen.getByText('错误: 条件崩溃')).toBeInTheDocument()
    expect(screen.getByRole('button', { name: '重试' })).toBeInTheDocument()
  })

  it('点击重试应恢复正常渲染', () => {
    render(<RecoverableApp />)

    expect(screen.getByText('错误: 条件崩溃')).toBeInTheDocument()

    fireEvent.click(screen.getByRole('button', { name: '重试' }))

    expect(screen.getByText('正常内容')).toBeInTheDocument()
    expect(screen.queryByRole('alert')).not.toBeInTheDocument()
  })
})
嵌套错误边界测试
import { render, screen } from '@testing-library/react'
import { describe, it, expect, vi, beforeEach } from 'vitest'

function InnerBroken(): JSX.Element {
  throw new Error('内层错误')
}

describe('嵌套错误边界', () => {
  beforeEach(() => {
    vi.spyOn(console, 'error').mockImplementation(() => {})
  })

  it('内层边界捕获错误，外层不受影响', () => {
    render(
      <ErrorBoundary fallback={<div>外层降级</div>}>
        <div>外层内容</div>
        <ErrorBoundary fallback={<div role="alert">内层降级</div>}>
          <InnerBroken />
        </ErrorBoundary>
      </ErrorBoundary>
    )

    expect(screen.getByText('外层内容')).toBeInTheDocument()
    expect(screen.getByText('内层降级')).toBeInTheDocument()
    expect(screen.queryByText('外层降级')).not.toBeInTheDocument()
  })
})
异步错误处理
错误边界不能直接捕获异步错误，但可以通过状态触发同步重渲染来间接处理。


import { render, screen, waitFor } from '@testing-library/react'
import { describe, it, expect, vi, beforeEach } from 'vitest'
import { useState, useEffect } from 'react'

function AsyncErrorComponent({ url }: { url: string }): JSX.Element {
  const [, setData] = useState(null)
  const [error, setError] = useState<Error | null>(null)

  useEffect(() => {
    fetch(url)
      .then((res) => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`)
        return res.json()
      })
      .then(setData)
      .catch(setError)
  }, [url])

  if (error) throw error

  return <div>加载中...</div>
}

describe('异步错误处理', () => {
  beforeEach(() => {
    vi.spyOn(console, 'error').mockImplementation(() => {})
  })

  it('异步错误通过状态传递给错误边界', async () => {
    vi.stubGlobal('fetch', vi.fn().mockResolvedValue({
      ok: false,
      status: 500,
    }))

    render(
      <ErrorBoundary fallback={<div role="alert">请求失败</div>}>
        <AsyncErrorComponent url="/api/data" />
      </ErrorBoundary>
    )

    await waitFor(() => {
      expect(screen.getByText('请求失败')).toBeInTheDocument()
    })

    vi.unstubAllGlobals()
  })
})
测试策略总结
常见误区与纠正
误区一：不抑制 console.error
React 在错误边界捕获错误时会输出 console.error。测试中应 mock 掉以保持输出干净。


误区二：只测试降级 UI
还需要测试 onError 回调（错误上报）、错误恢复（reset）、嵌套边界的隔离性。


误区三：依赖错误边界捕获事件处理器错误
事件处理器中的错误不会被错误边界捕获。应使用 try-catch 在事件处理器内部处理。


误区四：忽略异步错误路径
异步错误需要通过 state 转换为同步渲染错误。测试中需要 waitFor 等待异步状态更新。


小结
错误边界是 React 应用韧性工程的核心组件。


核心要点回顾：


    

  • 错误边界捕获渲染和生命周期错误，不捕获事件处理器和异步错误
    • 

  • 测试需覆盖：正常渲染、降级 UI、错误上报、恢复重试
    • 

  • render prop 形式的 fallback 支持显示错误信息和重试按钮
    • 

  • 嵌套错误边界实现局部降级，不影响外层正常内容
    • 

  • 异步错误通过 state 传递给错误边界间接捕获
    • 

  • vi.spyOn(console, 'error') 抑制 React 错误输出
    • 

  • 错误上报回调应包含 Error 对象和 componentStack 堆栈信息
    • 

  • 重试功能需要 reset 错误边界状态 + 修复错误源
