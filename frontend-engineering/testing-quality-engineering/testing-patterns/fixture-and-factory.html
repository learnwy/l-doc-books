Fixture 与 Factory

核心概念

测试数据管理是测试工程中经常被低估的领域。Fixture（固定数据）和 Factory（工厂函数）是两种核心的测试数据策略。Fixture 提供静态、可预测的数据集；Factory 动态生成数据，支持灵活覆盖。结合 Builder 模式和 faker.js，可以构建高效、隔离、可维护的测试数据体系。

Fixture：固定数据
简单 Fixture
import { describe, it, expect } from 'vitest';

interface Product {
  id: string;
  name: string;
  price: number;
  category: string;
  inStock: boolean;
}

const PRODUCT_FIXTURES: Record<string, Product> = {
  keyboard: {
    id: 'prod-001',
    name: '机械键盘',
    price: 599,
    category: '外设',
    inStock: true,
  },
  mouse: {
    id: 'prod-002',
    name: '无线鼠标',
    price: 199,
    category: '外设',
    inStock: true,
  },
  monitor: {
    id: 'prod-003',
    name: '4K 显示器',
    price: 2999,
    category: '显示器',
    inStock: false,
  },
};

describe('Product Fixture 使用', () => {
  it('计算在库商品总价', () => {
    const products = Object.values(PRODUCT_FIXTURES);
    const inStockTotal = products.filter((p) => p.inStock).reduce((sum, p) => sum + p.price, 0);

    expect(inStockTotal).toBe(798);
  });

  it('按分类筛选', () => {
    const peripherals = Object.values(PRODUCT_FIXTURES).filter((p) => p.category === '外设');
    expect(peripherals).toHaveLength(2);
  });
});
JSON Fixture 文件
import { describe, it, expect } from 'vitest';
import { readFileSync } from 'fs';
import { join } from 'path';

interface ApiResponse {
  data: Array<{ id: string; name: string }>;
  pagination: { page: number; total: number };
}

function loadFixture<T>(name: string): T {
  const path = join(__dirname, '__fixtures__', `${name}.json`);
  return JSON.parse(readFileSync(path, 'utf-8'));
}

describe('JSON Fixture', () => {
  it('加载用户列表响应', () => {
    const response = loadFixture<ApiResponse>('user-list-response');
    expect(response.data).toBeInstanceOf(Array);
    expect(response.pagination.total).toBeGreaterThan(0);
  });
});
Factory：工厂函数
简单工厂
import { describe, it, expect } from 'vitest';

interface User {
  id: string;
  name: string;
  email: string;
  role: 'admin' | 'editor' | 'viewer';
  isActive: boolean;
  createdAt: Date;
}

let userIdCounter = 0;

function createUser(overrides: Partial<User> = {}): User {
  userIdCounter++;
  return {
    id: `user-${userIdCounter}`,
    name: `测试用户 ${userIdCounter}`,
    email: `user${userIdCounter}@test.com`,
    role: 'viewer',
    isActive: true,
    createdAt: new Date('2024-01-01'),
    ...overrides,
  };
}

function createAdmin(overrides: Partial<User> = {}): User {
  return createUser({ role: 'admin', ...overrides });
}

function createInactiveUser(overrides: Partial<User> = {}): User {
  return createUser({ isActive: false, ...overrides });
}

describe('简单工厂', () => {
  it('创建默认用户', () => {
    const user = createUser();
    expect(user.role).toBe('viewer');
    expect(user.isActive).toBe(true);
  });

  it('覆盖特定属性', () => {
    const user = createUser({ name: 'Alice', role: 'editor' });
    expect(user.name).toBe('Alice');
    expect(user.role).toBe('editor');
  });

  it('快捷工厂', () => {
    const admin = createAdmin();
    expect(admin.role).toBe('admin');

    const inactive = createInactiveUser();
    expect(inactive.isActive).toBe(false);
  });

  it('每次生成独立对象', () => {
    const user1 = createUser();
    const user2 = createUser();
    expect(user1.id).not.toBe(user2.id);
    expect(user1.email).not.toBe(user2.email);
  });
});
关联数据工厂
import { describe, it, expect } from 'vitest';

interface Order {
  id: string;
  userId: string;
  items: OrderItem[];
  status: 'pending' | 'paid' | 'shipped' | 'delivered';
  total: number;
}

interface OrderItem {
  productId: string;
  name: string;
  price: number;
  quantity: number;
}

let orderIdCounter = 0;

function createOrderItem(overrides: Partial<OrderItem> = {}): OrderItem {
  return {
    productId: `prod-${Math.random().toString(36).slice(2, 8)}`,
    name: '测试商品',
    price: 100,
    quantity: 1,
    ...overrides,
  };
}

function createOrder(overrides: Partial<Order> = {}): Order {
  orderIdCounter++;
  const items = overrides.items || [createOrderItem()];
  const total = items.reduce((sum, item) => sum + item.price * item.quantity, 0);

  return {
    id: `order-${orderIdCounter}`,
    userId: 'user-1',
    items,
    status: 'pending',
    total,
    ...overrides,
  };
}

describe('关联数据工厂', () => {
  it('自动计算订单总价', () => {
    const order = createOrder({
      items: [createOrderItem({ price: 100, quantity: 2 }), createOrderItem({ price: 50, quantity: 3 })],
    });

    expect(order.total).toBe(350);
    expect(order.items).toHaveLength(2);
  });

  it('默认订单有一个商品', () => {
    const order = createOrder();
    expect(order.items).toHaveLength(1);
    expect(order.status).toBe('pending');
  });
});
Builder 模式
Builder 模式提供链式 API，适合构建复杂对象。


import { describe, it, expect } from 'vitest';

interface UserProfile {
  id: string;
  name: string;
  email: string;
  role: 'admin' | 'editor' | 'viewer';
  permissions: string[];
  settings: {
    theme: 'light' | 'dark';
    language: string;
    notifications: boolean;
  };
  metadata: Record<string, unknown>;
}

class UserProfileBuilder {
  private profile: UserProfile = {
    id: 'default-id',
    name: '默认用户',
    email: 'default@test.com',
    role: 'viewer',
    permissions: [],
    settings: {
      theme: 'light',
      language: 'zh-CN',
      notifications: true,
    },
    metadata: {},
  };

  withId(id: string): this {
    this.profile.id = id;
    return this;
  }

  withName(name: string): this {
    this.profile.name = name;
    return this;
  }

  withEmail(email: string): this {
    this.profile.email = email;
    return this;
  }

  asAdmin(): this {
    this.profile.role = 'admin';
    this.profile.permissions = ['read', 'write', 'delete', 'admin'];
    return this;
  }

  asEditor(): this {
    this.profile.role = 'editor';
    this.profile.permissions = ['read', 'write'];
    return this;
  }

  withDarkTheme(): this {
    this.profile.settings.theme = 'dark';
    return this;
  }

  withLanguage(lang: string): this {
    this.profile.settings.language = lang;
    return this;
  }

  withNotificationsDisabled(): this {
    this.profile.settings.notifications = false;
    return this;
  }

  withPermissions(...permissions: string[]): this {
    this.profile.permissions = permissions;
    return this;
  }

  withMetadata(key: string, value: unknown): this {
    this.profile.metadata[key] = value;
    return this;
  }

  build(): UserProfile {
    return { ...this.profile, settings: { ...this.profile.settings } };
  }
}

function aUserProfile(): UserProfileBuilder {
  return new UserProfileBuilder();
}

describe('Builder 模式', () => {
  it('构建管理员', () => {
    const admin = aUserProfile().withName('Admin Alice').withEmail('alice@admin.com').asAdmin().withDarkTheme().build();

    expect(admin.name).toBe('Admin Alice');
    expect(admin.role).toBe('admin');
    expect(admin.permissions).toContain('admin');
    expect(admin.settings.theme).toBe('dark');
  });

  it('构建编辑者', () => {
    const editor = aUserProfile().withName('Editor Bob').asEditor().withLanguage('en-US').build();

    expect(editor.role).toBe('editor');
    expect(editor.permissions).toEqual(['read', 'write']);
    expect(editor.settings.language).toBe('en-US');
  });

  it('构建自定义权限用户', () => {
    const user = aUserProfile()
      .withPermissions('read', 'comment')
      .withNotificationsDisabled()
      .withMetadata('department', '工程部')
      .build();

    expect(user.permissions).toEqual(['read', 'comment']);
    expect(user.settings.notifications).toBe(false);
    expect(user.metadata.department).toBe('工程部');
  });

  it('每次 build 生成独立对象', () => {
    const builder = aUserProfile().withName('Same Name');
    const user1 = builder.build();
    const user2 = builder.build();

    user1.name = 'Modified';
    expect(user2.name).toBe('Same Name');
  });
});
使用 faker.js 生成随机数据
pnpm add -D @faker-js/faker
import { describe, it, expect } from 'vitest';
import { faker } from '@faker-js/faker/locale/zh_CN';

interface Employee {
  id: string;
  name: string;
  email: string;
  phone: string;
  department: string;
  salary: number;
  hireDate: Date;
  address: {
    city: string;
    street: string;
    zipCode: string;
  };
}

function createEmployee(overrides: Partial<Employee> = {}): Employee {
  return {
    id: faker.string.uuid(),
    name: faker.person.fullName(),
    email: faker.internet.email(),
    phone: faker.phone.number(),
    department: faker.helpers.arrayElement(['工程部', '产品部', '设计部', '市场部']),
    salary: faker.number.int({ min: 8000, max: 50000 }),
    hireDate: faker.date.past({ years: 5 }),
    address: {
      city: faker.location.city(),
      street: faker.location.street(),
      zipCode: faker.location.zipCode(),
    },
    ...overrides,
  };
}

function createEmployees(count: number, overrides: Partial<Employee> = {}): Employee[] {
  return Array.from({ length: count }, () => createEmployee(overrides));
}

describe('faker.js 工厂', () => {
  it('生成随机员工', () => {
    const employee = createEmployee();
    expect(employee.id).toBeDefined();
    expect(employee.name).toBeTruthy();
    expect(employee.email).toContain('@');
    expect(employee.salary).toBeGreaterThanOrEqual(8000);
    expect(employee.salary).toBeLessThanOrEqual(50000);
  });

  it('批量生成并验证唯一性', () => {
    const employees = createEmployees(50);
    const uniqueIds = new Set(employees.map((e) => e.id));
    expect(uniqueIds.size).toBe(50);
  });

  it('覆盖指定部门', () => {
    const engineers = createEmployees(10, { department: '工程部' });
    expect(engineers.every((e) => e.department === '工程部')).toBe(true);
  });
});
可重现的随机数据
import { describe, it, expect, beforeEach } from 'vitest';
import { faker } from '@faker-js/faker';

describe('可重现的随机数据', () => {
  beforeEach(() => {
    faker.seed(12345);
  });

  it('相同种子产生相同数据', () => {
    const name1 = faker.person.fullName();

    faker.seed(12345);
    const name2 = faker.person.fullName();

    expect(name1).toBe(name2);
  });
});
数据隔离策略
import { describe, it, expect, beforeEach } from 'vitest';

interface TodoItem {
  id: string;
  text: string;
  done: boolean;
}

class TodoStore {
  private items: TodoItem[] = [];

  add(text: string): TodoItem {
    const item: TodoItem = {
      id: String(this.items.length + 1),
      text,
      done: false,
    };
    this.items.push(item);
    return item;
  }

  getAll(): TodoItem[] {
    return [...this.items];
  }

  toggle(id: string): void {
    const item = this.items.find((i) => i.id === id);
    if (item) item.done = !item.done;
  }

  clear(): void {
    this.items = [];
  }
}

describe('数据隔离 - beforeEach 重置', () => {
  let store: TodoStore;

  beforeEach(() => {
    store = new TodoStore();
    store.add('任务一');
    store.add('任务二');
  });

  it('初始状态有两个任务', () => {
    expect(store.getAll()).toHaveLength(2);
  });

  it('添加任务不影响其他测试', () => {
    store.add('任务三');
    expect(store.getAll()).toHaveLength(3);
  });

  it('这个测试仍然只有两个任务', () => {
    expect(store.getAll()).toHaveLength(2);
  });
});
常见误区与纠正
误区一：共享可变 Fixture
多个测试共享同一个可变对象，一个测试修改了数据会影响其他测试。应使用工厂或深拷贝。


误区二：Factory 生成不必要的属性
工厂函数应该只生成测试需要的属性。不要为了"完整"而生成大量无关数据，增加阅读成本。


误区三：随机数据导致不确定性
faker.js 生成的随机数据可能导致偶发失败。使用 faker.seed() 固定种子，或对随机值做范围断言而非精确断言。


误区四：Builder 链过长
如果 Builder 链超过 5-6 个调用，考虑提取为命名工厂（如 createAdmin()）。


小结
Fixture 和 Factory 是测试数据管理的两大支柱。


核心要点回顾：


    

  • Fixture 适合静态、少量、可预测的测试数据
    • 

  • Factory 适合动态、大量、需要变化的实体数据
    • 

  • Builder 模式为复杂对象提供链式构建 API
    • 

  • faker.js 生成逼真的随机数据，seed() 保证可重现
    • 

  • 关联数据工厂自动维护对象之间的一致性
    • 

  • beforeEach 重置状态保证测试间数据隔离
    • 

  • 每个测试应使用独立的数据实例，避免共享可变状态
