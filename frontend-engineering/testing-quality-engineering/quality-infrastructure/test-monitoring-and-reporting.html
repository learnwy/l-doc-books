测试监控与报告

核心概念

测试监控与报告是测试体系的"仪表板"，让团队实时了解测试健康状况、发现趋势、快速定位问题。一个完善的报告体系不仅展示通过/失败数量，还追踪覆盖率趋势、Flaky 率、执行时间等关键指标，为质量决策提供数据支撑。

自定义报告器
import type { Reporter, TestCase, TestModule, TestSuite } from 'vitest/reporters';

interface TestMetrics {
  total: number;
  passed: number;
  failed: number;
  skipped: number;
  duration: number;
  slowTests: Array<{ name: string; duration: number }>;
}

class MetricsReporter implements Reporter {
  private metrics: TestMetrics = {
    total: 0,
    passed: 0,
    failed: 0,
    skipped: 0,
    duration: 0,
    slowTests: [],
  };

  onTestCaseResult(test: TestCase): void {
    this.metrics.total++;

    const result = test.result;
    if (!result) return;

    if (result.state === 'passed') this.metrics.passed++;
    else if (result.state === 'failed') this.metrics.failed++;
    else if (result.state === 'skipped') this.metrics.skipped++;

    const duration = result.duration ?? 0;
    this.metrics.duration += duration;

    if (duration > 1000) {
      this.metrics.slowTests.push({
        name: test.name,
        duration,
      });
    }
  }

  onFinished(): void {
    const { total, passed, failed, skipped, duration, slowTests } = this.metrics;

    console.log('\n📊 测试指标摘要');
    console.log('─'.repeat(50));
    console.log(`总数: ${total}`);
    console.log(`通过: ${passed} (${((passed / total) * 100).toFixed(1)}%)`);
    console.log(`失败: ${failed}`);
    console.log(`跳过: ${skipped}`);
    console.log(`总耗时: ${(duration / 1000).toFixed(2)}s`);
    console.log(`平均耗时: ${(duration / total).toFixed(0)}ms`);

    if (slowTests.length > 0) {
      console.log(`\n🐢 慢测试 (> 1s):`);
      slowTests
        .sort((a, b) => b.duration - a.duration)
        .slice(0, 10)
        .forEach((t) => {
          console.log(`  ${(t.duration / 1000).toFixed(2)}s - ${t.name}`);
        });
    }
  }
}

export default MetricsReporter;
测试结果分析
解析 JSON 结果
import { readFileSync } from 'fs';

interface VitestResult {
  numTotalTests: number;
  numPassedTests: number;
  numFailedTests: number;
  numPendingTests: number;
  testResults: Array<{
    name: string;
    status: 'passed' | 'failed' | 'pending';
    duration: number;
    assertionResults: Array<{
      title: string;
      status: string;
      duration: number;
    }>;
  }>;
}

function analyzeResults(resultPath: string): void {
  const data: VitestResult = JSON.parse(readFileSync(resultPath, 'utf-8'));

  const passRate = ((data.numPassedTests / data.numTotalTests) * 100).toFixed(1);

  console.log(`通过率: ${passRate}%`);
  console.log(`总数: ${data.numTotalTests}`);

  const slowFiles = data.testResults.filter((r) => r.duration > 5000).sort((a, b) => b.duration - a.duration);

  if (slowFiles.length > 0) {
    console.log('\n慢测试文件:');
    slowFiles.forEach((f) => {
      console.log(`  ${(f.duration / 1000).toFixed(1)}s - ${f.name}`);
    });
  }

  const failedTests = data.testResults.filter((r) => r.status === 'failed');
  if (failedTests.length > 0) {
    console.log('\n失败的测试:');
    failedTests.forEach((f) => {
      console.log(`  ❌ ${f.name}`);
      f.assertionResults
        .filter((a) => a.status === 'failed')
        .forEach((a) => {
          console.log(`     - ${a.title}`);
        });
    });
  }
}
趋势追踪
interface DailyMetrics {
  date: string;
  totalTests: number;
  passRate: number;
  coveragePercent: number;
  avgDurationMs: number;
  flakyCount: number;
}

function calculateTrend(history: DailyMetrics[]): void {
  if (history.length < 2) return;

  const latest = history[history.length - 1];
  const previous = history[history.length - 2];

  const trends = {
    passRate: latest.passRate - previous.passRate,
    coverage: latest.coveragePercent - previous.coveragePercent,
    duration: latest.avgDurationMs - previous.avgDurationMs,
    flaky: latest.flakyCount - previous.flakyCount,
  };

  console.log('\n📈 趋势分析');
  console.log(`通过率: ${formatTrend(trends.passRate)}%`);
  console.log(`覆盖率: ${formatTrend(trends.coverage)}%`);
  console.log(`平均耗时: ${formatTrend(trends.duration)}ms`);
  console.log(`Flaky 数: ${formatTrend(-trends.flaky)}`);
}

function formatTrend(diff: number): string {
  if (diff > 0) return `↑ +${diff.toFixed(1)}`;
  if (diff < 0) return `↓ ${diff.toFixed(1)}`;
  return '→ 0';
}

import type { Reporter, TestCase, TestResult, FullConfig, Suite } from '@playwright/test/reporter';

class SummaryReporter implements Reporter {
  private results: Array<{ name: string; status: string; duration: number }> = [];

  onTestEnd(test: TestCase, result: TestResult): void {
    this.results.push({
      name: test.title,
      status: result.status,
      duration: result.duration,
    });
  }

  onEnd(): void {
    const total = this.results.length;
    const passed = this.results.filter((r) => r.status === 'passed').length;
    const failed = this.results.filter((r) => r.status === 'failed').length;
    const avgDuration = this.results.reduce((sum, r) => sum + r.duration, 0) / total;

    console.log('\n🎭 E2E 测试摘要');
    console.log(`总数: ${total} | 通过: ${passed} | 失败: ${failed}`);
    console.log(`平均耗时: ${(avgDuration / 1000).toFixed(1)}s`);

    const slowest = this.results.sort((a, b) => b.duration - a.duration).slice(0, 5);

    console.log('\n最慢测试:');
    slowest.forEach((t) => {
      console.log(`  ${(t.duration / 1000).toFixed(1)}s - ${t.name}`);
    });
  }
}

export default SummaryReporter;

Slack 通知
interface SlackMessage {
  text: string;
  blocks: Array<{
    type: string;
    text?: { type: string; text: string };
    fields?: Array<{ type: string; text: string }>;
  }>;
}

function buildSlackMessage(metrics: {
  passRate: number;
  totalTests: number;
  failedTests: number;
  duration: number;
  coverage: number;
  branch: string;
}): SlackMessage {
  const emoji = metrics.failedTests === 0 ? '✅' : '❌';
  const status = metrics.failedTests === 0 ? '全部通过' : `${metrics.failedTests} 个失败`;

  return {
    text: `${emoji} 测试结果: ${status}`,
    blocks: [
      {
        type: 'header',
        text: {
          type: 'plain_text',
          text: `${emoji} 测试报告 - ${metrics.branch}`,
        },
      },
      {
        type: 'section',
        fields: [
          { type: 'mrkdwn', text: `*通过率:* ${metrics.passRate.toFixed(1)}%` },
          { type: 'mrkdwn', text: `*总测试数:* ${metrics.totalTests}` },
          { type: 'mrkdwn', text: `*覆盖率:* ${metrics.coverage.toFixed(1)}%` },
          { type: 'mrkdwn', text: `*耗时:* ${(metrics.duration / 1000).toFixed(1)}s` },
        ],
      },
    ],
  };
}

async function sendToSlack(webhookUrl: string, message: SlackMessage): Promise<void> {
  await fetch(webhookUrl, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(message),
  });
}

常见误区与纠正
误区一：只看通过/失败
单纯的通过/失败不够。需要分析趋势、定位慢测试、追踪 Flaky。


误区二：不追踪历史数据
没有历史数据就无法发现趋势。应持久化每次运行的指标，至少保留 90 天。


误区三：报告只给开发看
测试报告应面向全团队：开发看详情、管理看趋势、QA 看覆盖率。


误区四：手动生成报告
报告应完全自动化生成和分发，不需要人工干预。


小结
测试监控与报告让质量状况透明可见。
