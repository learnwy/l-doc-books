编写有效断言

核心概念

断言（Assertions）是测试的核心，它定义了"期望值与实际值之间的关系"。有效的断言不仅验证代码行为的正确性，还能在测试失败时提供清晰、可操作的错误信息。编写好的断言是写好测试的基础技能。

Vitest expect API 全景

相等性断言
toBe vs toEqual vs toStrictEqual
import { describe, it, expect } from 'vitest';

describe('相等性断言的区别', () => {
  it('toBe - 严格引用相等（===）', () => {
    expect(1 + 1).toBe(2);
    expect('hello').toBe('hello');
    expect(true).toBe(true);
    expect(null).toBe(null);

    const obj = { a: 1 };
    const ref = obj;
    expect(ref).toBe(obj);
  });

  it('toEqual - 深度值相等（递归比较属性值）', () => {
    expect({ a: 1, b: { c: 2 } }).toEqual({ a: 1, b: { c: 2 } });
    expect([1, [2, 3]]).toEqual([1, [2, 3]]);

    expect({ a: 1, b: undefined }).toEqual({ a: 1 });
  });

  it('toStrictEqual - 严格深度相等', () => {
    expect({ a: 1 }).toStrictEqual({ a: 1 });

    expect({ a: 1, b: undefined }).not.toStrictEqual({ a: 1 });

    class User {
      constructor(public name: string) {}
    }
    expect(new User('Alice')).not.toStrictEqual({ name: 'Alice' });
  });
});

布尔值与存在性断言
import { describe, it, expect } from 'vitest';

describe('布尔值和存在性', () => {
  it('truthy / falsy', () => {
    expect(1).toBeTruthy();
    expect('non-empty').toBeTruthy();
    expect([]).toBeTruthy();
    expect({}).toBeTruthy();

    expect(0).toBeFalsy();
    expect('').toBeFalsy();
    expect(null).toBeFalsy();
    expect(undefined).toBeFalsy();
    expect(NaN).toBeFalsy();
  });

  it('null / undefined / defined', () => {
    expect(null).toBeNull();
    expect(undefined).toBeUndefined();
    expect('value').toBeDefined();
    expect(undefined).not.toBeDefined();
  });

  it('NaN', () => {
    expect(NaN).toBeNaN();
    expect(0 / 0).toBeNaN();
    expect(1).not.toBeNaN();
  });

  it('类型检查', () => {
    expect(typeof 'hello').toBe('string');
    expect(typeof 42).toBe('number');
    expect(typeof true).toBe('boolean');
    expect(Array.isArray([1, 2])).toBe(true);
  });
});
数值断言
import { describe, it, expect } from 'vitest';

describe('数值断言', () => {
  it('大小比较', () => {
    expect(10).toBeGreaterThan(5);
    expect(10).toBeGreaterThanOrEqual(10);
    expect(5).toBeLessThan(10);
    expect(5).toBeLessThanOrEqual(5);
  });

  it('浮点数精度 - toBeCloseTo', () => {
    expect(0.1 + 0.2).toBeCloseTo(0.3, 5);
    expect(Math.PI).toBeCloseTo(3.14159, 4);
    expect(1 / 3).toBeCloseTo(0.333, 2);
  });

  it('范围验证', () => {
    const value = 42;
    expect(value).toBeGreaterThanOrEqual(0);
    expect(value).toBeLessThanOrEqual(100);
  });

  it('Infinity', () => {
    expect(1 / 0).toBe(Infinity);
    expect(-1 / 0).toBe(-Infinity);
    expect(Number.isFinite(42)).toBe(true);
    expect(Number.isFinite(Infinity)).toBe(false);
  });
});
字符串断言
import { describe, it, expect } from 'vitest';

describe('字符串断言', () => {
  const message = 'Hello, World! Welcome to Vitest.';

  it('包含子串', () => {
    expect(message).toContain('World');
    expect(message).toContain('Vitest');
    expect(message).not.toContain('Jest');
  });

  it('正则匹配', () => {
    expect(message).toMatch(/Hello/);
    expect(message).toMatch(/^Hello/);
    expect(message).toMatch(/Vitest\.$/i);
    expect('user@example.com').toMatch(/^[\w.]+@[\w.]+\.\w+$/);
  });

  it('长度', () => {
    expect('hello').toHaveLength(5);
    expect('').toHaveLength(0);
  });

  it('模板字符串验证', () => {
    const name = 'Alice';
    const greeting = `你好, ${name}！`;
    expect(greeting).toBe('你好, Alice！');
    expect(greeting).toMatch(new RegExp(`${name}`));
  });
});
数组与集合断言
import { describe, it, expect } from 'vitest';

describe('数组断言', () => {
  const numbers = [1, 2, 3, 4, 5];
  const users = [
    { id: '1', name: 'Alice', role: 'admin' },
    { id: '2', name: 'Bob', role: 'user' },
    { id: '3', name: 'Charlie', role: 'user' },
  ];

  it('包含与长度', () => {
    expect(numbers).toContain(3);
    expect(numbers).not.toContain(6);
    expect(numbers).toHaveLength(5);
  });

  it('对象数组包含', () => {
    expect(users).toContainEqual({ id: '1', name: 'Alice', role: 'admin' });
    expect(users).toContainEqual(expect.objectContaining({ name: 'Bob' }));
  });

  it('expect.arrayContaining - 子集匹配', () => {
    expect(numbers).toEqual(expect.arrayContaining([1, 3, 5]));
    expect(numbers).not.toEqual(expect.arrayContaining([6, 7]));
  });

  it('验证数组结构', () => {
    expect(users).toEqual(
      expect.arrayContaining([
        expect.objectContaining({
          id: expect.any(String),
          name: expect.any(String),
          role: expect.stringMatching(/^(admin|user)$/),
        }),
      ]),
    );
  });

  it('排序验证', () => {
    const sorted = [...numbers].sort((a, b) => a - b);
    expect(sorted).toEqual([1, 2, 3, 4, 5]);
  });
});
对象断言
import { describe, it, expect } from 'vitest';

describe('对象断言', () => {
  const user = {
    id: '1',
    name: 'Alice',
    email: 'alice@example.com',
    address: {
      city: '北京',
      zip: '100000',
    },
    tags: ['admin', 'vip'],
  };

  it('属性存在性', () => {
    expect(user).toHaveProperty('name');
    expect(user).toHaveProperty('address.city');
    expect(user).toHaveProperty(['address', 'zip']);
    expect(user).not.toHaveProperty('phone');
  });

  it('属性值', () => {
    expect(user).toHaveProperty('name', 'Alice');
    expect(user).toHaveProperty('address.city', '北京');
    expect(user).toHaveProperty('tags', ['admin', 'vip']);
  });

  it('部分匹配 - toMatchObject', () => {
    expect(user).toMatchObject({
      name: 'Alice',
      address: { city: '北京' },
    });
  });

  it('expect.objectContaining', () => {
    expect(user).toEqual(
      expect.objectContaining({
        id: expect.any(String),
        email: expect.stringContaining('@'),
        tags: expect.arrayContaining(['admin']),
      }),
    );
  });
});
异常断言
import { describe, it, expect } from 'vitest';

class ValidationError extends Error {
  constructor(
    message: string,
    public field: string,
    public code: string,
  ) {
    super(message);
    this.name = 'ValidationError';
  }
}

function parseAge(input: string): number {
  const age = parseInt(input, 10);
  if (isNaN(age)) {
    throw new ValidationError('年龄必须是数字', 'age', 'INVALID_NUMBER');
  }
  if (age < 0 || age > 150) {
    throw new ValidationError('年龄必须在 0-150 之间', 'age', 'OUT_OF_RANGE');
  }
  return age;
}

describe('异常断言', () => {
  it('基本异常捕获', () => {
    expect(() => parseAge('abc')).toThrow();
    expect(() => parseAge('abc')).toThrow('年龄必须是数字');
    expect(() => parseAge('abc')).toThrow(/数字/);
    expect(() => parseAge('abc')).toThrow(ValidationError);
  });

  it('验证异常属性', () => {
    try {
      parseAge('abc');
    } catch (error) {
      expect(error).toBeInstanceOf(ValidationError);
      const validationError = error as ValidationError;
      expect(validationError.field).toBe('age');
      expect(validationError.code).toBe('INVALID_NUMBER');
    }
  });

  it('验证不同类型的异常', () => {
    expect(() => parseAge('-1')).toThrow('0-150');
    expect(() => parseAge('200')).toThrow('OUT_OF_RANGE');
    expect(() => parseAge('25')).not.toThrow();
  });

  it('异步异常', async () => {
    async function fetchData(): Promise<never> {
      throw new Error('网络错误');
    }

    await expect(fetchData()).rejects.toThrow('网络错误');
    await expect(fetchData()).rejects.toThrow(Error);
    await expect(fetchData()).rejects.toBeInstanceOf(Error);
  });
});
自定义 Matcher
import { expect, describe, it } from 'vitest';

interface CustomMatchers<R = unknown> {
  toBeValidEmail(): R;
  toBeBetween(min: number, max: number): R;
  toBeEmptyArray(): R;
  toHaveBeenCalledBefore(laterFn: ReturnType<typeof vi.fn>): R;
}

declare module 'vitest' {
  interface Assertion<T = any> extends CustomMatchers<T> {}
  interface AsymmetricMatchersContaining extends CustomMatchers {}
}

expect.extend({
  toBeValidEmail(received: string) {
    const emailRegex = /^[\w.+-]+@[\w-]+\.[\w.]+$/;
    const pass = emailRegex.test(received);
    return {
      pass,
      message: () => (pass ? `期望 "${received}" 不是有效邮箱` : `期望 "${received}" 是有效邮箱`),
    };
  },

  toBeBetween(received: number, min: number, max: number) {
    const pass = received >= min && received <= max;
    return {
      pass,
      message: () =>
        pass ? `期望 ${received} 不在 ${min} 和 ${max} 之间` : `期望 ${received} 在 ${min} 和 ${max} 之间`,
    };
  },

  toBeEmptyArray(received: unknown[]) {
    const pass = Array.isArray(received) && received.length === 0;
    return {
      pass,
      message: () => (pass ? `期望不是空数组，但收到空数组` : `期望是空数组，但收到 [${received}]`),
    };
  },
});

describe('自定义 Matcher', () => {
  it('邮箱验证', () => {
    expect('alice@example.com').toBeValidEmail();
    expect('user.name+tag@domain.co.uk').toBeValidEmail();
    expect('not-an-email').not.toBeValidEmail();
    expect('@missing-local.com').not.toBeValidEmail();
  });

  it('范围验证', () => {
    expect(50).toBeBetween(0, 100);
    expect(0).toBeBetween(0, 100);
    expect(100).toBeBetween(0, 100);
    expect(101).not.toBeBetween(0, 100);
  });

  it('空数组验证', () => {
    expect([]).toBeEmptyArray();
    expect([1]).not.toBeEmptyArray();
  });
});
断言模式与最佳实践
Arrange-Act-Assert（AAA 模式）
import { describe, it, expect } from 'vitest';

describe('AAA 模式', () => {
  it('计算含税总价', () => {
    const items = [
      { name: '商品A', price: 100, quantity: 2 },
      { name: '商品B', price: 50, quantity: 1 },
    ];
    const taxRate = 0.1;

    const subtotal = items.reduce((sum, item) => sum + item.price * item.quantity, 0);
    const total = subtotal * (1 + taxRate);

    expect(total).toBe(275);
  });
});
每个测试一个逻辑断言
import { describe, it, expect } from 'vitest';

interface User {
  name: string;
  email: string;
  isActive: boolean;
}

function createUser(name: string, email: string): User {
  return { name, email, isActive: true };
}

describe('单一逻辑断言', () => {
  it('新用户应有正确的名称', () => {
    const user = createUser('Alice', 'alice@test.com');
    expect(user.name).toBe('Alice');
  });

  it('新用户应有正确的邮箱', () => {
    const user = createUser('Alice', 'alice@test.com');
    expect(user.email).toBe('alice@test.com');
  });

  it('新用户应默认激活', () => {
    const user = createUser('Alice', 'alice@test.com');
    expect(user.isActive).toBe(true);
  });

  it('或使用 toMatchObject 验证整体结构', () => {
    const user = createUser('Alice', 'alice@test.com');
    expect(user).toMatchObject({
      name: 'Alice',
      email: 'alice@test.com',
      isActive: true,
    });
  });
});
避免测试实现细节
import { describe, it, expect } from 'vitest';

class UserService {
  private cache = new Map<string, { name: string }>();

  getUser(id: string): { name: string } | undefined {
    if (this.cache.has(id)) return this.cache.get(id);
    const user = { name: `User-${id}` };
    this.cache.set(id, user);
    return user;
  }
}

describe('测试行为而非实现', () => {
  it('应返回用户信息', () => {
    const service = new UserService();

    const user = service.getUser('1');
    expect(user).toEqual({ name: 'User-1' });
  });

  it('多次获取同一用户应返回相同数据', () => {
    const service = new UserService();

    const first = service.getUser('1');
    const second = service.getUser('1');

    expect(first).toEqual(second);
  });
});
expect.assertions 确保异步断言执行
import { describe, it, expect } from 'vitest';

describe('expect.assertions', () => {
  it('确保 catch 分支中的断言被执行', async () => {
    expect.assertions(2);

    try {
      await Promise.reject(new Error('失败'));
    } catch (error) {
      expect(error).toBeInstanceOf(Error);
      expect((error as Error).message).toBe('失败');
    }
  });

  it('expect.hasAssertions 确保至少有一个断言', () => {
    expect.hasAssertions();
    expect(true).toBe(true);
  });
});
