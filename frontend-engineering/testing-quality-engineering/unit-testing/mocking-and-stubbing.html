Mock 与 Stub 技术

核心概念

Mock 和 Stub 是隔离被测代码依赖的核心手段。它们让你在不依赖真实外部服务、数据库或第三方 API 的情况下，专注测试业务逻辑。正确使用测试替身是编写可靠、快速单元测试的关键技能。

基础 Mock
Mock 函数
import { vi, describe, it, expect } from 'vitest';

describe('mocks', () => {
  it('should create mock function', () => {
    const mockFn = vi.fn();

    mockFn('hello');
    mockFn(1, 2, 3);

    expect(mockFn).toHaveBeenCalled();
    expect(mockFn).toHaveBeenCalledTimes(2);
    expect(mockFn).toHaveBeenCalledWith('hello');
    expect(mockFn).toHaveBeenLastCalledWith(1, 2, 3);
  });

  it('should return mock value', () => {
    const mockFn = vi.fn(() => 'default');
    mockFn.mockReturnValueOnce('first');
    mockFn.mockReturnValueOnce('second');

    expect(mockFn()).toBe('first');
    expect(mockFn()).toBe('second');
    expect(mockFn()).toBe('default');
  });

  it('should resolve promise', async () => {
    const mockFn = vi.fn().mockResolvedValue('success');

    const result = await mockFn();
    expect(result).toBe('success');
  });

  it('should reject promise', async () => {
    const mockFn = vi.fn().mockRejectedValue(new Error('Failed'));

    await expect(mockFn()).rejects.toThrow('Failed');
  });
});
Mock 模块
import { vi, describe, it, expect } from 'vitest';
import { fetchUser } from './api';
import { processUser } from './user-service';

vi.mock('./api', () => ({
  fetchUser: vi.fn(),
}));

describe('user service', () => {
  it('should process user', async () => {
    vi.mocked(fetchUser).mockResolvedValue({
      id: '1',
      name: 'Alice',
    });

    const result = await processUser('1');

    expect(fetchUser).toHaveBeenCalledWith('1');
    expect(result).toEqual({
      id: '1',
      name: 'Alice',
      processed: true,
    });
  });
});
Spy 技术
Spy 方法
import { vi, describe, it, expect } from 'vitest';
import { UserService } from './user-service';

describe('spies', () => {
  it('should spy on method', () => {
    const service = new UserService();
    const spy = vi.spyOn(service, 'logMessage');

    service.doSomething();

    expect(spy).toHaveBeenCalled();
    expect(spy).toHaveBeenCalledWith('Did something');

    spy.mockRestore(); // 恢复原始方法
  });

  it('should mock implementation', () => {
    const service = new UserService();
    vi.spyOn(service, 'getUser').mockImplementation((id) => ({
      id,
      name: 'Mocked User',
    }));

    const user = service.getUser('1');
    expect(user.name).toBe('Mocked User');
  });
});
Fake 实现
内存数据库
import { describe, it, expect, beforeEach } from 'vitest';

class FakeDatabase {
  private users: Map<string, any> = new Map();

  async connect() {
    // 不需要实际连接
  }

  async disconnect() {
    // 不需要实际断开
  }

  async insertUser(user: any) {
    this.users.set(user.id, user);
  }

  async getUser(id: string) {
    return this.users.get(id) || null;
  }

  async clear() {
    this.users.clear();
  }
}

describe('with fake db', () => {
  let db: FakeDatabase;

  beforeEach(() => {
    db = new FakeDatabase();
  });

  it('should insert and retrieve user', async () => {
    await db.insertUser({ id: '1', name: 'Alice' });
    const user = await db.getUser('1');
    expect(user).toEqual({ id: '1', name: 'Alice' });
  });
});

❌ 避免的写法
// 不要 mock 太多
test('over-mocked', () => {
  vi.mock('./module1');
  vi.mock('./module2');
  vi.mock('./module3');
  // 测试变得脆弱
});

// 不要 mock 你自己的代码
test('mock own code', () => {
  vi.mock('./utils', () => ({
    add: vi.fn(() => 3), // 应该直接测试 add 函数
  }));
});
✅ 推荐的写法
// 只 mock 外部依赖
test('mock external', () => {
  vi.mock('axios'); // mock 第三方库
  // 测试你的代码
});

// 使用 Fake 替代复杂依赖
test('use fake', () => {
  const db = new FakeDatabase();
  // 使用 Fake 进行测试
});
小结
    

  • vi.fn() 创建可追踪调用的模拟函数
    • 

  • vi.spyOn() 监听并可选替换已有方法
    • 

  • vi.mock() 在模块级别替换整个依赖
    • 

  • MSW 是网络请求 mock 的推荐方案
    • 

  • 遵循最少 mock 原则，只 mock 真正的外部依赖
