异步代码测试

核心概念

异步代码测试是前端测试中的高频场景——API 调用、定时器、事件回调、数据流订阅都涉及异步操作。Vitest 提供了完善的异步测试支持，包括 async/await、Promise 断言和 Fake Timer。掌握这些工具是写出可靠异步测试的基础。

Promise 测试
async/await
import { describe, it, expect } from 'vitest';
import { fetchUser, fetchUsers } from './api';

describe('async functions', () => {
  it('should fetch user with async/await', async () => {
    const user = await fetchUser('1');
    expect(user.id).toBe('1');
    expect(user.name).toBe('Alice');
  });

  it('should handle error with try/catch', async () => {
    try {
      await fetchUser('invalid-id');
      // 如果没有抛出异常，测试失败
      expect.fail('Should have thrown an error');
    } catch (error) {
      expect(error).toBeInstanceOf(Error);
      expect(error.message).toContain('not found');
    }
  });
});
.then()/.catch()
import { describe, it, expect } from 'vitest';
import { fetchUser } from './api';

describe('async with then', () => {
  it('should fetch user with .then()', () => {
    return fetchUser('1').then((user) => {
      expect(user.id).toBe('1');
    });
  });

  it('should handle error with .catch()', () => {
    return fetchUser('invalid').catch((error) => {
      expect(error).toBeDefined();
    });
  });
});
resolves/rejects
import { describe, it, expect } from 'vitest';
import { fetchUser } from './api';

describe('resolves and rejects', () => {
  it('should resolve with user', async () => {
    await expect(fetchUser('1')).resolves.toEqual({
      id: '1',
      name: 'Alice',
    });
  });

  it('should reject with error', async () => {
    await expect(fetchUser('invalid')).rejects.toThrow('not found');
  });
});
定时器测试
使用 fakeTimers
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';

describe('timers', () => {
  beforeEach(() => {
    vi.useFakeTimers();
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  it('should execute setTimeout', () => {
    const callback = vi.fn();
    setTimeout(callback, 1000);

    expect(callback).not.toHaveBeenCalled();

    vi.advanceTimersByTime(1000);

    expect(callback).toHaveBeenCalledTimes(1);
  });

  it('should execute setInterval', () => {
    const callback = vi.fn();
    setInterval(callback, 1000);

    vi.advanceTimersByTime(3000);

    expect(callback).toHaveBeenCalledTimes(3);
  });

  it('should run all timers', () => {
    const callback = vi.fn();
    setTimeout(callback, 1000);
    setTimeout(callback, 2000);

    vi.runAllTimers();

    expect(callback).toHaveBeenCalledTimes(2);
  });
});
事件测试
测试事件监听
import { describe, it, expect, vi } from 'vitest';
import { EventEmitter } from 'events';

describe('events', () => {
  it('should emit and listen to event', () => {
    const emitter = new EventEmitter();
    const callback = vi.fn();

    emitter.on('data', callback);
    emitter.emit('data', 'hello');

    expect(callback).toHaveBeenCalledWith('hello');
  });

  it('should listen once', () => {
    const emitter = new EventEmitter();
    const callback = vi.fn();

    emitter.once('data', callback);
    emitter.emit('data', 'first');
    emitter.emit('data', 'second');

    expect(callback).toHaveBeenCalledTimes(1);
    expect(callback).toHaveBeenCalledWith('first');
  });
});
React 异步测试
测试异步组件
import { describe, it, expect, vi } from 'vitest'
import { render, screen, waitFor } from '@testing-library/react'
import UserProfile from './UserProfile'

vi.mock('./api', () => ({
  fetchUser: vi.fn()
}))

describe('async components', () => {
  it('should render loading state', () => {
    render(<UserProfile userId="1" />)
    expect(screen.getByText('Loading...')).toBeInTheDocument()
  })

  it('should render user data', async () => {
    vi.mocked(fetchUser).mockResolvedValue({
      id: '1',
      name: 'Alice'
    })

    render(<UserProfile userId="1" />)

    await waitFor(() => {
      expect(screen.getByText('Alice')).toBeInTheDocument()
    })
  })

  it('should render error state', async () => {
    vi.mocked(fetchUser).mockRejectedValue(new Error('Failed'))

    render(<UserProfile userId="1" />)

    await waitFor(() => {
      expect(screen.getByText('Error loading user')).toBeInTheDocument()
    })
  })
})
