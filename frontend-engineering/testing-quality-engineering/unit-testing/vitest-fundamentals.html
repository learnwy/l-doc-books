Vitest 基础与配置

核心概念

Vitest 是基于 Vite 的下一代测试框架，利用 Vite 的即时模块转换和热更新能力，提供极速的测试体验。它与 Jest API 高度兼容，支持原生 ESM、TypeScript 和多线程并行，是现代前端项目的首选测试工具。

为什么选择 Vitest

维度 Vitest Jest 说明
速度 ⚡️ 极快 🐢 较慢 复用 Vite 的转换管道，无需单独配置 Babel
配置 复用 vite.config 独立配置 减少配置文件和重复设置
ESM 支持 原生支持 实验性 无需 --experimental-vm-modules
TypeScript 开箱即用 需要 ts-jest 无需额外转换器
HMR 支持 Watch 模式 Watch 模式较慢 文件变更时只重新运行相关测试
UI 界面 @vitest/ui 无 浏览器中交互式查看测试结果
多线程 Worker Threads Worker Threads 默认隔离每个测试文件

describe / it / expect

测试组织结构

import { describe, it, expect, test } from 'vitest';

describe('StringUtils', () => {
  describe('capitalize', () => {
    it('应将首字母大写', () => {
      expect(capitalize('hello')).toBe('Hello');
    });

    it('空字符串返回空字符串', () => {
      expect(capitalize('')).toBe('');
    });

    it('已大写的字符串保持不变', () => {
      expect(capitalize('Hello')).toBe('Hello');
    });
  });

  describe('slugify', () => {
    it('应将空格替换为连字符', () => {
      expect(slugify('hello world')).toBe('hello-world');
    });

    it('应转为小写', () => {
      expect(slugify('Hello World')).toBe('hello-world');
    });

    it('应移除特殊字符', () => {
      expect(slugify('hello@world!')).toBe('helloworld');
    });
  });
});

function capitalize(str: string): string {
  if (!str) return str;
  return str.charAt(0).toUpperCase() + str.slice(1);
}

function slugify(str: string): string {
  return str
    .toLowerCase()
    .replace(/\s+/g, '-')
    .replace(/[^a-z0-9-]/g, '');
}

test.each 参数化测试

import { describe, it, expect } from 'vitest';

function add(a: number, b: number): number {
  return a + b;
}

describe('add', () => {
  it.each([
    [1, 2, 3],
    [0, 0, 0],
    [-1, 1, 0],
    [100, 200, 300],
    [0.1, 0.2, 0.3],
  ])('add(%d, %d) = %d', (a, b, expected) => {
    expect(add(a, b)).toBeCloseTo(expected);
  });

  it.each`
    a     | b     | expected
    ${1}  | ${2}  | ${3}
    ${-1} | ${-1} | ${-2}
    ${0}  | ${0}  | ${0}
  `('add($a, $b) = $expected', ({ a, b, expected }) => {
    expect(add(a, b)).toBe(expected);
  });
});

生命周期钩子

import { describe, it, expect, beforeAll, afterAll, beforeEach, afterEach } from 'vitest';

interface TodoItem {
  id: string;
  text: string;
  done: boolean;
}

class TodoStore {
  private items: Map<string, TodoItem> = new Map();
  private counter = 0;

  add(text: string): TodoItem {
    const item: TodoItem = {
      id: String(++this.counter),
      text,
      done: false,
    };
    this.items.set(item.id, item);
    return item;
  }

  toggle(id: string): void {
    const item = this.items.get(id);
    if (item) item.done = !item.done;
  }

  remove(id: string): void {
    this.items.delete(id);
  }

  getAll(): TodoItem[] {
    return [...this.items.values()];
  }

  clear(): void {
    this.items.clear();
    this.counter = 0;
  }
}

describe('TodoStore', () => {
  let store: TodoStore;

  beforeAll(() => {
    store = new TodoStore();
  });

  afterAll(() => {
    store.clear();
  });

  beforeEach(() => {
    store.clear();
  });

  afterEach(() => {});

  it('应添加待办项', () => {
    const item = store.add('学习 Vitest');
    expect(item.text).toBe('学习 Vitest');
    expect(item.done).toBe(false);
    expect(store.getAll()).toHaveLength(1);
  });

  it('应切换完成状态', () => {
    const item = store.add('写测试');
    store.toggle(item.id);
    expect(store.getAll()[0].done).toBe(true);

    store.toggle(item.id);
    expect(store.getAll()[0].done).toBe(false);
  });

  it('应删除待办项', () => {
    const item = store.add('删除我');
    store.remove(item.id);
    expect(store.getAll()).toHaveLength(0);
  });
});
测试异步代码
import { describe, it, expect, vi } from 'vitest';

interface ApiResponse<T> {
  data: T;
  status: number;
}

async function fetchUserById(id: string): Promise<ApiResponse<{ name: string }>> {
  const response = await fetch(`/api/users/${id}`);
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  const data = await response.json();
  return { data, status: response.status };
}

async function fetchWithRetry<T>(fn: () => Promise<T>, retries: number = 3): Promise<T> {
  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (attempt === retries) throw error;
    }
  }
  throw new Error('不可达');
}

describe('异步测试', () => {
  it('async/await 方式', async () => {
    vi.stubGlobal(
      'fetch',
      vi.fn().mockResolvedValue({
        ok: true,
        status: 200,
        json: () => Promise.resolve({ name: 'Alice' }),
      }),
    );

    const result = await fetchUserById('1');
    expect(result.data.name).toBe('Alice');
    expect(result.status).toBe(200);

    vi.unstubAllGlobals();
  });

  it('验证异步错误', async () => {
    vi.stubGlobal(
      'fetch',
      vi.fn().mockResolvedValue({
        ok: false,
        status: 404,
      }),
    );

    await expect(fetchUserById('999')).rejects.toThrow('HTTP 404');

    vi.unstubAllGlobals();
  });

  it('带重试的异步函数', async () => {
    let callCount = 0;
    const unstableFn = async () => {
      callCount++;
      if (callCount < 3) throw new Error('临时错误');
      return '成功';
    };

    const result = await fetchWithRetry(unstableFn, 3);
    expect(result).toBe('成功');
    expect(callCount).toBe(3);
  });
});

describe('vi.fn - 创建模拟函数', () => {

    expect(mockFn()).toBe('second');
    expect(mockFn()).toBe('default');
    expect(mockFn()).toBe('default');
  });

  it('模拟实现', () => {
    const mockFn = vi.fn().mockImplementation((a: number, b: number) => a * b);

    expect(mockFn(3, 4)).toBe(12);
    expect(mockFn(5, 6)).toBe(30);
  });
});

describe('vi.spyOn - 监视现有方法', () => {
  const calculator = {
    add: (a: number, b: number) => a + b,
    multiply: (a: number, b: number) => a * b,
  };

  beforeEach(() => {
    vi.restoreAllMocks();
  });

  it('监视方法调用', () => {
    const spy = vi.spyOn(calculator, 'add');

    calculator.add(1, 2);

    expect(spy).toHaveBeenCalledWith(1, 2);
    expect(spy).toHaveReturnedWith(3);
  });

  it('替换实现但保留监视', () => {
    vi.spyOn(calculator, 'multiply').mockReturnValue(999);

    expect(calculator.multiply(2, 3)).toBe(999);
  });
});

定时器模拟
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';

function debounce<T extends (...args: unknown[]) => void>(fn: T, delay: number): (...args: Parameters<T>) => void {
  let timer: ReturnType<typeof setTimeout>;
  return (...args: Parameters<T>) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}

describe('debounce', () => {
  beforeEach(() => {
    vi.useFakeTimers();
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  it('延迟后应执行函数', () => {
    const fn = vi.fn();
    const debounced = debounce(fn, 300);

    debounced();
    expect(fn).not.toHaveBeenCalled();

    vi.advanceTimersByTime(300);
    expect(fn).toHaveBeenCalledOnce();
  });

  it('连续调用应只执行最后一次', () => {
    const fn = vi.fn();
    const debounced = debounce(fn, 300);

    debounced();
    debounced();
    debounced();

    vi.advanceTimersByTime(300);
    expect(fn).toHaveBeenCalledOnce();
  });

  it('超过延迟后的新调用应重新计时', () => {
    const fn = vi.fn();
    const debounced = debounce(fn, 300);

    debounced();
    vi.advanceTimersByTime(300);
    expect(fn).toHaveBeenCalledOnce();

    debounced();
    vi.advanceTimersByTime(300);
    expect(fn).toHaveBeenCalledTimes(2);
  });
});
