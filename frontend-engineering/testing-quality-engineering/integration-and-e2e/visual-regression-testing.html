视觉回归测试

核心概念

视觉回归测试（Visual Regression Testing）通过截图对比来检测 UI 的意外视觉变化。它弥补了单元测试和集成测试无法覆盖的视觉层面，确保样式变更、布局调整和主题切换不会引入意外的 UI 回归。

    });
  });

组件级截图
import { test, expect } from '@playwright/test';

test.describe('组件视觉回归', () => {
  test('按钮状态', async ({ page }) => {
    await page.goto('/storybook/button');

    const primaryBtn = page.getByTestId('btn-primary');
    await expect(primaryBtn).toHaveScreenshot('btn-primary.png');

    const disabledBtn = page.getByTestId('btn-disabled');
    await expect(disabledBtn).toHaveScreenshot('btn-disabled.png');
  });

  test('表单组件', async ({ page }) => {
    await page.goto('/storybook/form');

    const form = page.getByTestId('login-form');
    await expect(form).toHaveScreenshot('login-form-empty.png');

    await page.getByLabel('邮箱').fill('user@test.com');
    await page.getByLabel('密码').fill('password');
    await expect(form).toHaveScreenshot('login-form-filled.png');
  });

  test('表单校验错误状态', async ({ page }) => {
    await page.goto('/storybook/form');

    await page.getByRole('button', { name: '提交' }).click();

    const form = page.getByTestId('login-form');
    await expect(form).toHaveScreenshot('login-form-errors.png');
  });
});

处理动态内容
import { test, expect } from '@playwright/test';

test('稳定化动态内容', async ({ page }) => {
  await page.goto('/dashboard');

  await page.evaluate(() => {
    document.querySelectorAll('[data-testid="timestamp"]').forEach((el) => {
      el.textContent = '2024-01-01 00:00:00';
    });

    document.querySelectorAll('[data-testid="avatar"]').forEach((el) => {
      (el as HTMLImageElement).src = 'data:image/svg+xml,<svg/>';
    });
  });

  await page.evaluate(() => {
    const style = document.createElement('style');
    style.textContent = '*, *::before, *::after { animation: none !important; transition: none !important; }';
    document.head.appendChild(style);
  });

  await expect(page).toHaveScreenshot('dashboard-stable.png');
});

test('等待字体加载', async ({ page }) => {
  await page.goto('/');
  await page.evaluate(() => document.fonts.ready);
  await page.waitForTimeout(500);
  await expect(page).toHaveScreenshot('with-fonts.png');
});
主题与响应式测试
import { test, expect } from '@playwright/test';

test.describe('主题视觉测试', () => {
  test('亮色主题', async ({ page }) => {
    await page.goto('/');
    await page.emulateMedia({ colorScheme: 'light' });
    await expect(page).toHaveScreenshot('theme-light.png');
  });

  test('暗色主题', async ({ page }) => {
    await page.goto('/');
    await page.emulateMedia({ colorScheme: 'dark' });
    await expect(page).toHaveScreenshot('theme-dark.png');
  });
});

test.describe('响应式布局', () => {
  const viewports = [
    { name: 'mobile', width: 375, height: 812 },
    { name: 'tablet', width: 768, height: 1024 },
    { name: 'desktop', width: 1440, height: 900 },
  ];

  for (const vp of viewports) {
    test(`${vp.name} 布局`, async ({ page }) => {
      await page.setViewportSize({ width: vp.width, height: vp.height });
      await page.goto('/');
      await expect(page).toHaveScreenshot(`layout-${vp.name}.png`);
    });
  }
});

import { test } from '@playwright/test';
import percySnapshot from '@percy/playwright';

test.describe('Percy 视觉测试', () => {
  test('首页', async ({ page }) => {
    await page.goto('/');
    await percySnapshot(page, '首页', {
      widths: [375, 768, 1280],
      minHeight: 1024,
    });
  });

    await page.goto('/products');
    await percySnapshot(page, '产品列表', {
      widths: [375, 1280],
    });
  });
});

CI 集成
name: Visual Regression Tests
on: [pull_request]

jobs:
  visual-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'pnpm'

      - run: pnpm install

      - name: Install Playwright browsers
        run: npx playwright install --with-deps chromium

      - name: Run visual tests
        run: npx playwright test --project="Desktop Chrome"

      - name: Upload screenshots diff
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: visual-diff
          path: |
            e2e/screenshots/
            test-results/

常见误区与纠正
误区一：截全页面做回归
全页面截图包含大量不稳定内容（广告、推荐、时间戳），导致频繁误报。应聚焦关键组件和布局区域。


误区二：阈值设太低
过低的阈值（如 0）会因为字体渲染、子像素差异导致大量假阳性。推荐 threshold: 0.2 起步。


误区三：不审查就更新
视觉回归测试的核心价值在于人工审查 Diff。盲目 --update-snapshots 会让真正的视觉 Bug 溜走。


误区四：忽略跨平台差异
不同操作系统的字体渲染不同。应在 CI 中使用固定环境（如 Docker + Linux），或使用云端服务统一渲染环境。


小结
视觉回归测试是 UI 质量保障的重要一环。


核心要点回顾：


    

  • Playwright toHaveScreenshot 提供内置截图对比能力
    • 

  • 页面级和组件级截图适用于不同粒度的验证
    • 

  • mask 和 evaluate 处理动态内容，避免误报
    • 

  • 主题和响应式视口需要多维度截图覆盖
    • 

  • Percy/Chromatic 提供云端团队协作审查
    • 

  • 阈值配置需要在精确性和稳定性之间平衡
    • 

  • CI 中固定运行环境，消除跨平台渲染差异
    • 

  • 截图更新必须经过人工审查，不能盲目更新
