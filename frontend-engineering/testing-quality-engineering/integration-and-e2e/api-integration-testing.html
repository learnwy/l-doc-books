API 集成测试

核心概念

API 集成测试验证客户端与服务端之间的交互是否正常工作，涵盖 HTTP 请求/响应、数据序列化、错误处理和认证流程。与单元测试不同，集成测试关注的是多个模块协作的正确性，包括网络层、数据层和业务逻辑层的集成。

使用 MSW 进行 API Mock
MSW（Mock Service Worker）在网络层拦截请求，不侵入应用代码。


安装与配置
pnpm add -D msw
定义请求处理器
import { http, HttpResponse } from 'msw';

interface User {
  id: string;
  name: string;
  email: string;
  role: 'admin' | 'user';
}

interface CreateUserInput {
  name: string;
  email: string;
}

const users: User[] = [
  { id: '1', name: 'Alice', email: 'alice@test.com', role: 'admin' },
  { id: '2', name: 'Bob', email: 'bob@test.com', role: 'user' },
];

export const handlers = [
  http.get('/api/users', ({ request }) => {
    const url = new URL(request.url);
    const role = url.searchParams.get('role');

    const filtered = role ? users.filter((u) => u.role === role) : users;

    return HttpResponse.json({
      data: filtered,
      total: filtered.length,
    });
  }),

  http.get('/api/users/:id', ({ params }) => {
    const user = users.find((u) => u.id === params.id);
    if (!user) {
      return HttpResponse.json({ error: '用户不存在' }, { status: 404 });
    }
    return HttpResponse.json({ data: user });
  }),

  http.post('/api/users', async ({ request }) => {
    const body = (await request.json()) as CreateUserInput;

    if (!body.name || !body.email) {
      return HttpResponse.json({ error: '姓名和邮箱必填' }, { status: 400 });
    }

    const newUser: User = {
      id: String(users.length + 1),
      name: body.name,
      email: body.email,
      role: 'user',
    };

    return HttpResponse.json({ data: newUser }, { status: 201 });
  }),

  http.delete('/api/users/:id', ({ params }) => {
    const user = users.find((u) => u.id === params.id);
    if (!user) {
      return HttpResponse.json({ error: '用户不存在' }, { status: 404 });
    }
    return new HttpResponse(null, { status: 204 });
  }),
];

import { describe, it, expect, beforeAll, afterAll, afterEach } from 'vitest';
import { setupServer } from 'msw/node';
import { handlers } from './handlers';

const server = setupServer(...handlers);

beforeAll(() => server.listen());
afterAll(() => server.close());
afterEach(() => server.resetHandlers());

async function apiClient<T>(path: string, options?: RequestInit): Promise<T> {
  const response = await fetch(`http://localhost${path}`, {
    headers: { 'Content-Type': 'application/json' },
    ...options,
  });

  if (!response.ok) {
    const error = await response.json();
    throw new Error(error.error || `HTTP ${response.status}`);
  }

  if (response.status === 204) return undefined as T;
  return response.json();
}

describe('用户 API 集成测试', () => {
  it('获取用户列表', async () => {
    const result = await apiClient<{ data: Array<{ name: string }>; total: number }>('/api/users');

    expect(result.data).toHaveLength(2);
    expect(result.total).toBe(2);
    expect(result.data[0].name).toBe('Alice');
  });

  it('按角色筛选用户', async () => {
    const result = await apiClient<{ data: Array<{ role: string }> }>('/api/users?role=admin');

    expect(result.data).toHaveLength(1);
    expect(result.data[0].role).toBe('admin');
  });

  it('获取单个用户', async () => {
    const result = await apiClient<{ data: { name: string } }>('/api/users/1');

    expect(result.data.name).toBe('Alice');
  });

  it('用户不存在返回 404', async () => {
    await expect(apiClient('/api/users/999')).rejects.toThrow('用户不存在');
  });

  it('创建新用户', async () => {
    const result = await apiClient<{ data: { id: string; name: string } }>('/api/users', {
      method: 'POST',
      body: JSON.stringify({ name: 'Charlie', email: 'charlie@test.com' }),
    });

    expect(result.data.name).toBe('Charlie');
    expect(result.data.id).toBeDefined();
  });

  it('创建用户缺少必填字段', async () => {
    await expect(
      apiClient('/api/users', {
        method: 'POST',
        body: JSON.stringify({ name: '' }),
      }),
    ).rejects.toThrow('姓名和邮箱必填');
  });

  it('删除用户', async () => {
    const result = await apiClient('/api/users/1', { method: 'DELETE' });
    expect(result).toBeUndefined();
  });
});
运行时覆盖处理器
import { http, HttpResponse } from 'msw';
import { describe, it, expect } from 'vitest';

describe('错误场景测试', () => {
  it('服务器 500 错误', async () => {
    server.use(
      http.get('/api/users', () => {
        return HttpResponse.json({ error: '服务器内部错误' }, { status: 500 });
      }),
    );

    await expect(apiClient('/api/users')).rejects.toThrow('服务器内部错误');
  });

  it('网络超时', async () => {
    server.use(
      http.get('/api/users', async () => {
        await new Promise((resolve) => setTimeout(resolve, 10000));
        return HttpResponse.json({ data: [] });
      }),
    );
  });

  it('空响应', async () => {
    server.use(
      http.get('/api/users', () => {
        return HttpResponse.json({ data: [], total: 0 });
      }),
    );

    const result = await apiClient<{ data: unknown[]; total: number }>('/api/users');
    expect(result.data).toHaveLength(0);
    expect(result.total).toBe(0);
  });
});
认证与授权测试
import { http, HttpResponse } from 'msw';
import { describe, it, expect } from 'vitest';

const authHandlers = [
  http.post('/api/auth/login', async ({ request }) => {
    const body = (await request.json()) as { email: string; password: string };

    if (body.email === 'admin@test.com' && body.password === 'admin123') {
      return HttpResponse.json({
        token: 'mock-jwt-token',
        user: { id: '1', name: 'Admin', role: 'admin' },
      });
    }

    return HttpResponse.json({ error: '邮箱或密码错误' }, { status: 401 });
  }),

  http.get('/api/protected/profile', ({ request }) => {
    const authHeader = request.headers.get('Authorization');

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return HttpResponse.json({ error: '未授权' }, { status: 401 });
    }

    const token = authHeader.replace('Bearer ', '');
    if (token !== 'mock-jwt-token') {
      return HttpResponse.json({ error: 'Token 无效' }, { status: 403 });
    }

    return HttpResponse.json({
      data: { id: '1', name: 'Admin', email: 'admin@test.com' },
    });
  }),
];

describe('认证流程', () => {
  beforeAll(() => {
    server.use(...authHandlers);
  });

  it('正确凭据应返回 token', async () => {
    const result = await apiClient<{ token: string }>('/api/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email: 'admin@test.com', password: 'admin123' }),
    });

    expect(result.token).toBe('mock-jwt-token');
  });

  it('错误凭据应返回 401', async () => {
    await expect(
      apiClient('/api/auth/login', {
        method: 'POST',
        body: JSON.stringify({ email: 'wrong@test.com', password: 'wrong' }),
      }),
    ).rejects.toThrow('邮箱或密码错误');
  });

  it('携带 token 应访问受保护资源', async () => {
    const result = await apiClient<{ data: { name: string } }>('/api/protected/profile', {
      headers: {
        'Content-Type': 'application/json',
        Authorization: 'Bearer mock-jwt-token',
      },
    });

    expect(result.data.name).toBe('Admin');
  });

  it('无 token 访问受保护资源应返回 401', async () => {
    await expect(apiClient('/api/protected/profile')).rejects.toThrow('未授权');
  });
});
契约测试
契约测试确保前后端对 API 接口规范的理解一致。


import { describe, it, expect } from 'vitest';

interface ApiContract {
  path: string;
  method: 'GET' | 'POST' | 'PUT' | 'DELETE';
  requestBody?: Record<string, unknown>;
  responseStatus: number;
  responseShape: Record<string, unknown>;
}

const userContracts: ApiContract[] = [
  {
    path: '/api/users',
    method: 'GET',
    responseStatus: 200,
    responseShape: {
      data: expect.arrayContaining([
        expect.objectContaining({
          id: expect.any(String),
          name: expect.any(String),
          email: expect.stringContaining('@'),
        }),
      ]),
      total: expect.any(Number),
    },
  },
  {
    path: '/api/users/1',
    method: 'GET',
    responseStatus: 200,
    responseShape: {
      data: expect.objectContaining({
        id: '1',
        name: expect.any(String),
        email: expect.any(String),
        role: expect.stringMatching(/^(admin|user)$/),
      }),
    },
  },
];

describe('API 契约测试', () => {
  userContracts.forEach((contract) => {
    it(`${contract.method} ${contract.path} 应符合契约`, async () => {
      const response = await fetch(`http://localhost${contract.path}`, {
        method: contract.method,
        headers: { 'Content-Type': 'application/json' },
        body: contract.requestBody ? JSON.stringify(contract.requestBody) : undefined,
      });

      expect(response.status).toBe(contract.responseStatus);

      const body = await response.json();
      expect(body).toEqual(contract.responseShape);
    });
  });
});
测试数据管理
import { faker } from '@faker-js/faker';

interface UserFixture {
  id: string;
  name: string;
  email: string;
  role: 'admin' | 'user';
}

function createUserFixture(overrides: Partial<UserFixture> = {}): UserFixture {
  return {
    id: faker.string.uuid(),
    name: faker.person.fullName(),
    email: faker.internet.email(),
    role: 'user',
    ...overrides,
  };
}

function createUsersFixture(count: number): UserFixture[] {
  return Array.from({ length: count }, () => createUserFixture());
}

describe('数据工厂', () => {
  it('生成单个用户', () => {
    const user = createUserFixture({ role: 'admin' });
    expect(user.role).toBe('admin');
    expect(user.email).toContain('@');
  });

  it('批量生成用户', () => {
    const users = createUsersFixture(10);
    expect(users).toHaveLength(10);

    const uniqueEmails = new Set(users.map((u) => u.email));
    expect(uniqueEmails.size).toBe(10);
  });
});
Playwright API E2E 测试
import { test, expect } from '@playwright/test';

test.describe('API E2E 测试', () => {
  test('完整的 CRUD 流程', async ({ request }) => {
    const createResponse = await request.post('/api/users', {
      data: { name: 'E2E User', email: 'e2e@test.com' },
    });
    expect(createResponse.status()).toBe(201);
    const { data: created } = await createResponse.json();
    expect(created.id).toBeDefined();

    const getResponse = await request.get(`/api/users/${created.id}`);
    expect(getResponse.status()).toBe(200);
    const { data: fetched } = await getResponse.json();
    expect(fetched.name).toBe('E2E User');

    const deleteResponse = await request.delete(`/api/users/${created.id}`);
    expect(deleteResponse.status()).toBe(204);

    const getDeletedResponse = await request.get(`/api/users/${created.id}`);
    expect(getDeletedResponse.status()).toBe(404);
  });

  test('分页 API', async ({ request }) => {
    const response = await request.get('/api/users?page=1&pageSize=10');
    expect(response.status()).toBe(200);

    const body = await response.json();
    expect(body.data).toBeInstanceOf(Array);
    expect(body.total).toBeGreaterThanOrEqual(0);
    expect(body.data.length).toBeLessThanOrEqual(10);
  });
});
常见误区与纠正
误区一：直接依赖外部 API
集成测试不应依赖真实的外部服务。使用 MSW 或类似工具在网络层 Mock，保证测试的确定性和独立性。


误区二：只测试成功路径
错误处理是 API 集成的关键部分。必须测试 4xx/5xx 状态码、网络超时、空响应等异常场景。


误区三：忽略认证流程
认证是 API 安全的基础。测试 token 有效性、过期处理、权限控制是必不可少的。


误区四：测试数据不隔离
每个测试应使用独立的测试数据，避免测试之间相互依赖。使用 afterEach 重置 MSW 处理器或数据库状态。


小结
API 集成测试是确保前后端正确协作的关键环节。


核心要点回顾：


    

  • MSW 在网络层拦截请求，不侵入应用代码
    • 

  • 定义请求处理器覆盖 CRUD 和错误场景
    • 

  • server.use() 运行时覆盖处理器测试异常路径
    • 

  • 认证/授权测试验证 token 传递和权限控制
    • 

  • 契约测试确保前后端 API 规范一致
    • 

  • 工厂函数 + faker 生成隔离的测试数据
    • 

  • Playwright request API 进行真实 E2E API 测试
    • 

  • 测试数据隔离，每个测试独立，互不影响
    •
