Playwright 基础与实战

核心概念

Playwright 是一个强大的 E2E 测试框架，支持多个浏览器和平台。它由微软开发，能驱动 Chromium、Firefox 和 WebKit 三大浏览器引擎，提供自动等待、网络拦截、多页面/多上下文等高级能力。

// playwright.config.ts
import { defineConfig, devices } from '@playwright/test';

export default defineConfig({
  testDir: './tests',
  fullyParallel: true,
  forbidOnly: !!process.env.CI,
  retries: process.env.CI ? 2 : 0,
  workers: process.env.CI ? 1 : undefined,
  reporter: 'html',
  use: {
    baseURL: 'http://localhost:3000',
    trace: 'on-first-retry',
  },
  projects: [
    {
      name: 'chromium',
      use: { ...devices['Desktop Chrome'] },
    },
    {
      name: 'firefox',
      use: { ...devices['Desktop Firefox'] },
    },
    {
      name: 'webkit',
      use: { ...devices['Desktop Safari'] },
    },
  ],
  webServer: {
    command: 'npm run dev',
    url: 'http://localhost:3000',
    reuseExistingServer: !process.env.CI,
  },
});

import { test, expect } from '@playwright/test';

test('homepage has title', async ({ page }) => {
  await page.goto('/');
  await expect(page).toHaveTitle(/My App/);
});

test('login flow', async ({ page }) => {
  await page.goto('/login');

  await page.fill('input[name="email"]', 'alice@example.com');
  await page.fill('input[name="password"]', 'password123');
  await page.click('button[type="submit"]');

  await expect(page).toHaveURL('/dashboard');
  await expect(page.locator('h1')).toContainText('Welcome');
});
Locators
import { test, expect } from '@playwright/test';

test('locators', async ({ page }) => {
  await page.goto('/');

  // 通过角色
  await page.getByRole('button', { name: 'Submit' }).click();

  // 通过文本
  await page.getByText('Click me').click();

  // 通过标签
  await page.getByLabel('Email').fill('alice@example.com');

  // 通过 placeholder
  await page.getByPlaceholder('Search').fill('query');

  // 通过 test id
  await page.getByTestId('submit-button').click();
});
断言
import { test, expect } from '@playwright/test';

test('assertions', async ({ page }) => {
  await page.goto('/');

  // URL
  await expect(page).toHaveURL('/');
  await expect(page).toHaveURL(/dashboard/);

  // 标题
  await expect(page).toHaveTitle(/My App/);

  // 元素可见
  await expect(page.locator('h1')).toBeVisible();
  await expect(page.locator('.hidden')).toBeHidden();

  // 文本
  await expect(page.locator('h1')).toContainText('Welcome');
  await expect(page.locator('h1')).toHaveText('Welcome, Alice!');

  // 输入值
  await expect(page.locator('input[name="email"]')).toHaveValue('alice@example.com');

  // 计数
  await expect(page.locator('.item')).toHaveCount(5);
});

test('waiting', async ({ page }) => {
  await page.goto('/');

  // 等待导航
  await page.waitForURL('/dashboard');

  await page.waitForSelector('.loaded');

  // 等待网络

  // 等待函数

Page Object Model
// pages/LoginPage.ts
import { type Page, type Locator } from '@playwright/test';

export class LoginPage {
  readonly page: Page;
  readonly emailInput: Locator;
  readonly passwordInput: Locator;
  readonly submitButton: Locator;
  readonly errorMessage: Locator;

  constructor(page: Page) {
    this.page = page;
    this.emailInput = page.getByLabel('Email');
    this.passwordInput = page.getByLabel('Password');
    this.submitButton = page.getByRole('button', { name: 'Login' });
    this.errorMessage = page.locator('.error');
  }

  async goto() {
    await this.page.goto('/login');
  }

  async login(email: string, password: string) {
    await this.emailInput.fill(email);
    await this.passwordInput.fill(password);
    await this.submitButton.click();
  }
}

// tests/login.spec.ts
import { test, expect } from '@playwright/test';
import { LoginPage } from '../pages/LoginPage';

test('login with POM', async ({ page }) => {
  const loginPage = new LoginPage(page);
  await loginPage.goto();
  await loginPage.login('alice@example.com', 'password123');
  await expect(page).toHaveURL('/dashboard');
});
