无障碍测试

核心概念

无障碍测试（Accessibility Testing，简称 A11y Testing）确保 Web 应用对所有用户可用，包括使用屏幕阅读器、键盘导航或有视觉障碍的用户。无障碍不仅是道德责任和法律要求，也能提升所有用户的体验。自动化无障碍测试结合手动验证，是构建包容性应用的关键。

为什么无障碍测试重要

维度 说明
法律合规 许多国家和地区要求符合 WCAG 标准（如美国 Section 508、欧盟 EN 301 549）
用户覆盖 全球约 15% 人口有某种形式的残障
SEO 优化 语义化 HTML 和 ARIA 属性对搜索引擎友好
用户体验 无障碍改进（如键盘导航）对所有用户都有益
品牌声誉 包容性设计体现企业社会责任

基础 axe 测试
import { render } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import { axe, toHaveNoViolations } from 'vitest-axe'

expect.extend(toHaveNoViolations)

function LoginForm() {
  return (
    <form aria-label="登录表单">
      <div>
        <label htmlFor="email">邮箱</label>
        <input id="email" type="email" required />
      </div>
      <div>
        <label htmlFor="password">密码</label>
        <input id="password" type="password" required />
      </div>
      <button type="submit">登录</button>
    </form>
  )
}

describe('LoginForm 无障碍', () => {
  it('不应有无障碍违规', async () => {
    const { container } = render(<LoginForm />)
    const results = await axe(container)
    expect(results).toHaveNoViolations()
  })
})
指定 axe 规则
import { render } from '@testing-library/react'
import { it, expect } from 'vitest'
import { axe } from 'vitest-axe'

it('检查特定规则', async () => {
  const { container } = render(<LoginForm />)

  const results = await axe(container, {
    rules: {
      'color-contrast': { enabled: true },
      'label': { enabled: true },
      'heading-order': { enabled: true },
    },
  })

  expect(results).toHaveNoViolations()
})

it('排除特定规则', async () => {
  const { container } = render(<LoginForm />)

  const results = await axe(container, {
    rules: {
      region: { enabled: false },
    },
  })

  expect(results).toHaveNoViolations()
})
分析违规结果
import { render } from '@testing-library/react'
import { it, expect } from 'vitest'
import { axe } from 'vitest-axe'

function BadForm() {
  return (
    <form>
      <input type="text" />
      <div onClick={() => {}}>提交</div>
    </form>
  )
}

it('分析违规详情', async () => {
  const { container } = render(<BadForm />)
  const results = await axe(container)

  if (results.violations.length > 0) {
    results.violations.forEach((violation) => {
      console.log(`规则: ${violation.id}`)
      console.log(`影响: ${violation.impact}`)
      console.log(`描述: ${violation.description}`)
      violation.nodes.forEach((node) => {
        console.log(`  元素: ${node.html}`)
        console.log(`  修复: ${node.failureSummary}`)
      })
    })
  }

  expect(results.violations.length).toBeGreaterThan(0)
})

import { describe, it, expect } from 'vitest'

function Navigation() {
  return (
    <nav aria-label="主导航">
      <ul>
        <li><a href="/">首页</a></li>
        <li><a href="/products">产品</a></li>
        <li><a href="/about">关于</a></li>
      </ul>
    </nav>
  )
}

function SearchForm() {
  return (
    <search>
      <form role="search" aria-label="全站搜索">
        <label htmlFor="search-input">搜索</label>
        <input
          id="search-input"
          type="search"
          placeholder="输入关键词"
          aria-describedby="search-hint"
        />
        <p id="search-hint">按 Enter 键开始搜索</p>
        <button type="submit">搜索</button>
      </form>
    </search>
  )
}

describe('语义化 HTML', () => {
  it('导航应有正确的角色和标签', () => {
    render(<Navigation />)

    expect(screen.getByRole('navigation', { name: '主导航' })).toBeInTheDocument()
    expect(screen.getAllByRole('link')).toHaveLength(3)
    expect(screen.getByRole('link', { name: '首页' })).toHaveAttribute('href', '/')
  })

  it('搜索表单应有正确的语义', () => {
    render(<SearchForm />)

    expect(screen.getByRole('search', { name: '全站搜索' })).toBeInTheDocument()
    expect(screen.getByLabelText('搜索')).toBeInTheDocument()
    expect(screen.getByRole('button', { name: '搜索' })).toBeInTheDocument()
  })
})

模态框无障碍测试
import { render, screen, fireEvent } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import userEvent from '@testing-library/user-event'

interface ModalProps {
  isOpen: boolean
  title: string
  onClose: () => void
  children: React.ReactNode
}

function Modal({ isOpen, title, onClose, children }: ModalProps) {
  if (!isOpen) return null

  return (
    <div
      role="dialog"
      aria-modal="true"
      aria-labelledby="modal-title"
      aria-describedby="modal-body"
    >
      <h2 id="modal-title">{title}</h2>
      <div id="modal-body">{children}</div>
      <button onClick={onClose} aria-label="关闭对话框">×</button>
    </div>
  )
}

describe('Modal 无障碍', () => {
  it('应有正确的 ARIA 属性', () => {
    render(
      <Modal isOpen title="确认删除" onClose={() => {}}>
        确定要删除吗？
      </Modal>
    )

    const dialog = screen.getByRole('dialog')
    expect(dialog).toHaveAttribute('aria-modal', 'true')
    expect(dialog).toHaveAttribute('aria-labelledby', 'modal-title')
    expect(dialog).toHaveAttribute('aria-describedby', 'modal-body')

    expect(screen.getByRole('heading', { name: '确认删除' })).toBeInTheDocument()
  })

  it('关闭按钮应有 aria-label', () => {
    render(
      <Modal isOpen title="提示" onClose={() => {}}>
        内容
      </Modal>
    )

    expect(screen.getByRole('button', { name: '关闭对话框' })).toBeInTheDocument()
  })

  it('关闭时不渲染内容', () => {
    render(
      <Modal isOpen={false} title="提示" onClose={() => {}}>
        内容
      </Modal>
    )

    expect(screen.queryByRole('dialog')).not.toBeInTheDocument()
  })
})
选项卡无障碍测试
import { render, screen } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import userEvent from '@testing-library/user-event'

function Tabs({ tabs }: { tabs: Array<{ id: string; label: string; content: string }> }) {
  const [activeId, setActiveId] = useState(tabs[0].id)

  return (
    <div>
      <div role="tablist" aria-label="信息选项卡">
        {tabs.map((tab) => (
          <button
            key={tab.id}
            role="tab"
            id={`tab-${tab.id}`}
            aria-selected={activeId === tab.id}
            aria-controls={`panel-${tab.id}`}
            tabIndex={activeId === tab.id ? 0 : -1}
            onClick={() => setActiveId(tab.id)}
          >
            {tab.label}
          </button>
        ))}
      </div>
      {tabs.map((tab) => (
        <div
          key={tab.id}
          role="tabpanel"
          id={`panel-${tab.id}`}
          aria-labelledby={`tab-${tab.id}`}
          hidden={activeId !== tab.id}
        >
          {tab.content}
        </div>
      ))}
    </div>
  )
}

describe('Tabs 无障碍', () => {
  const tabs = [
    { id: 'info', label: '基本信息', content: '用户基本信息' },
    { id: 'security', label: '安全设置', content: '密码和双因素认证' },
    { id: 'notifications', label: '通知设置', content: '通知偏好' },
  ]

  it('应有正确的 tablist 和 tab 角色', () => {
    render(<Tabs tabs={tabs} />)

    expect(screen.getByRole('tablist', { name: '信息选项卡' })).toBeInTheDocument()
    expect(screen.getAllByRole('tab')).toHaveLength(3)
  })

  it('活动 tab 应有 aria-selected', () => {
    render(<Tabs tabs={tabs} />)

    const firstTab = screen.getByRole('tab', { name: '基本信息' })
    expect(firstTab).toHaveAttribute('aria-selected', 'true')

    const secondTab = screen.getByRole('tab', { name: '安全设置' })
    expect(secondTab).toHaveAttribute('aria-selected', 'false')
  })

  it('tab 应关联 tabpanel', () => {
    render(<Tabs tabs={tabs} />)

    const firstTab = screen.getByRole('tab', { name: '基本信息' })
    expect(firstTab).toHaveAttribute('aria-controls', 'panel-info')

    const panel = screen.getByRole('tabpanel')
    expect(panel).toHaveAttribute('aria-labelledby', 'tab-info')
  })
})
键盘导航测试
import { render, screen } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import userEvent from '@testing-library/user-event'

function DropdownMenu() {
  const [isOpen, setIsOpen] = useState(false)
  const items = ['编辑', '复制', '删除']

  return (
    <div>
      <button
        aria-haspopup="true"
        aria-expanded={isOpen}
        onClick={() => setIsOpen(!isOpen)}
      >
        操作
      </button>
      {isOpen && (
        <ul role="menu">
          {items.map((item) => (
            <li key={item} role="menuitem" tabIndex={0}>
              {item}
            </li>
          ))}
        </ul>
      )}
    </div>
  )
}

describe('DropdownMenu 键盘导航', () => {
  it('Tab 键聚焦到触发按钮', async () => {
    const user = userEvent.setup()
    render(<DropdownMenu />)

    await user.tab()
    expect(screen.getByRole('button', { name: '操作' })).toHaveFocus()
  })

  it('Enter 键打开菜单', async () => {
    const user = userEvent.setup()
    render(<DropdownMenu />)

    await user.tab()
    await user.keyboard('{Enter}')

    expect(screen.getByRole('menu')).toBeInTheDocument()
    expect(screen.getAllByRole('menuitem')).toHaveLength(3)
  })

  it('按钮 aria-expanded 状态正确', async () => {
    const user = userEvent.setup()
    render(<DropdownMenu />)

    const button = screen.getByRole('button', { name: '操作' })
    expect(button).toHaveAttribute('aria-expanded', 'false')

    await user.click(button)
    expect(button).toHaveAttribute('aria-expanded', 'true')
  })
})

describe('焦点管理', () => {
  it('Tab 顺序正确', async () => {
    const user = userEvent.setup()
    render(
      <form>
        <label htmlFor="name">姓名</label>
        <input id="name" type="text" />
        <label htmlFor="email">邮箱</label>
        <input id="email" type="email" />
        <button type="submit">提交</button>
      </form>
    )

    await user.tab()
    expect(screen.getByLabelText('姓名')).toHaveFocus()

    await user.tab()
    expect(screen.getByLabelText('邮箱')).toHaveFocus()

    await user.tab()
    expect(screen.getByRole('button', { name: '提交' })).toHaveFocus()
  })

  it('Shift+Tab 反向导航', async () => {
    const user = userEvent.setup()
    render(
      <form>
        <input aria-label="第一个输入" />
        <input aria-label="第二个输入" />
        <button>提交</button>
      </form>
    )

    await user.tab()
    await user.tab()
    await user.tab()
    expect(screen.getByRole('button')).toHaveFocus()

    await user.tab({ shift: true })
    expect(screen.getByLabelText('第二个输入')).toHaveFocus()
  })

  it('Escape 键关闭弹出层', async () => {
    const user = userEvent.setup()
    render(<DropdownMenu />)

    const button = screen.getByRole('button', { name: '操作' })
    await user.click(button)
    expect(screen.getByRole('menu')).toBeInTheDocument()

    await user.keyboard('{Escape}')
  })
})
Playwright 无障碍 E2E 测试
import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

test.describe('全站无障碍', () => {
  test('首页无障碍扫描', async ({ page }) => {
    await page.goto('/');

    const results = await new AxeBuilder({ page }).withTags(['wcag2a', 'wcag2aa']).analyze();

    expect(results.violations).toEqual([]);
  });

  test('登录页无障碍扫描', async ({ page }) => {
    await page.goto('/login');

    const results = await new AxeBuilder({ page }).include('form').withTags(['wcag2a', 'wcag2aa']).analyze();

    expect(results.violations).toEqual([]);
  });

  test('键盘可以完成登录流程', async ({ page }) => {
    await page.goto('/login');

    await page.keyboard.press('Tab');
    const emailInput = page.getByLabel('邮箱');
    await expect(emailInput).toBeFocused();
    await emailInput.fill('user@test.com');

    await page.keyboard.press('Tab');
    const passwordInput = page.getByLabel('密码');
    await expect(passwordInput).toBeFocused();
    await passwordInput.fill('password');

    await page.keyboard.press('Tab');
    await page.keyboard.press('Enter');

    await expect(page).toHaveURL('/dashboard');
  });

  test('屏幕阅读器友好的错误提示', async ({ page }) => {
    await page.goto('/login');

    await page.getByRole('button', { name: '登录' }).click();

    const errorAlert = page.getByRole('alert');
    await expect(errorAlert).toBeVisible();
    await expect(errorAlert).toContainText('请填写邮箱');
  });
});

常见误区与纠正
误区一：只用 axe 就够了
axe-core 只能检测约 30-40% 的无障碍问题。键盘导航流程、屏幕阅读器体验、焦点管理等必须手动测试。


误区二：无障碍是最后补的
无障碍应从设计阶段开始，选择语义化标签、正确的 ARIA 属性。后期补救成本高得多。


误区三：ARIA 越多越好
过度使用 ARIA 反而会干扰辅助技术。原则是：优先使用原生 HTML 语义（<button> 而非 <div role="button">），只在必要时添加 ARIA。


误区四：通过自动测试就合格
即使 axe 报告零违规，也不意味着用户体验良好。真正的无障碍需要真实用户的反馈。


小结
无障碍测试是质量工程不可或缺的一环。


核心要点回顾：


    

  • 使用 axe-core 自动扫描基础无障碍问题
    • 

  • 通过 getByRole、getByLabelText 验证语义化 HTML
    • 

  • ARIA 属性（aria-modal、aria-expanded、aria-label）提升辅助技术体验
    • 

  • 键盘导航测试确保 Tab 顺序、Enter/Escape 交互正确
    • 

  • Playwright + @axe-core/playwright 实现 E2E 无障碍扫描
    • 

  • 自动化只能覆盖 30-40% 的问题，手动测试不可替代
    • 

  • 优先使用原生语义标签，只在必要时添加 ARIA
    • 

  • 无障碍从设计阶段开始，而非开发完成后补救
    •
