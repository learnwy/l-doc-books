Hooks 与 Context 测试

核心概念

React Hooks 和 Context 是现代 React 应用的核心状态管理机制。测试它们需要理解 renderHook、act 包装器和 Context Provider 注入等特殊技术。本文涵盖自定义 Hook 的完整测试策略，从简单状态到复杂异步场景，以及多 Provider 组合测试。

测试简单状态 Hook
import { renderHook, act } from '@testing-library/react';
import { describe, it, expect } from 'vitest';
import { useState, useCallback } from 'react';

function useCounter(initialValue = 0) {
  const [count, setCount] = useState(initialValue);

  const increment = useCallback(() => setCount((c) => c + 1), []);
  const decrement = useCallback(() => setCount((c) => c - 1), []);
  const reset = useCallback(() => setCount(initialValue), [initialValue]);
  const incrementBy = useCallback((n: number) => setCount((c) => c + n), []);

  return { count, increment, decrement, reset, incrementBy };
}

describe('useCounter', () => {
  it('应使用默认初始值', () => {
    const { result } = renderHook(() => useCounter());
    expect(result.current.count).toBe(0);
  });

  it('应使用指定初始值', () => {
    const { result } = renderHook(() => useCounter(10));
    expect(result.current.count).toBe(10);
  });

  it('increment 应加 1', () => {
    const { result } = renderHook(() => useCounter());

    act(() => {
      result.current.increment();
    });

    expect(result.current.count).toBe(1);
  });

  it('decrement 应减 1', () => {
    const { result } = renderHook(() => useCounter(5));

    act(() => {
      result.current.decrement();
    });

    expect(result.current.count).toBe(4);
  });

  it('reset 应恢复到初始值', () => {
    const { result } = renderHook(() => useCounter(10));

    act(() => {
      result.current.increment();
      result.current.increment();
    });
    expect(result.current.count).toBe(12);

    act(() => {
      result.current.reset();
    });
    expect(result.current.count).toBe(10);
  });

  it('incrementBy 应增加指定值', () => {
    const { result } = renderHook(() => useCounter());

    act(() => {
      result.current.incrementBy(5);
    });

    expect(result.current.count).toBe(5);
  });

  it('连续操作', () => {
    const { result } = renderHook(() => useCounter());

    act(() => {
      result.current.increment();
      result.current.increment();
      result.current.incrementBy(10);
      result.current.decrement();
    });

    expect(result.current.count).toBe(12);
  });
});
测试异步 Hook
import { renderHook, waitFor, act } from '@testing-library/react';
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { useState, useEffect, useCallback } from 'react';

interface FetchState<T> {
  data: T | null;
  loading: boolean;
  error: string | null;
  refetch: () => void;
}

function useFetch<T>(url: string): FetchState<T> {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchData = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(url);
      if (!response.ok) throw new Error(`HTTP ${response.status}`);
      const json = await response.json();
      setData(json);
    } catch (err) {
      setError(err instanceof Error ? err.message : '未知错误');
    } finally {
      setLoading(false);
    }
  }, [url]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  return { data, loading, error, refetch: fetchData };
}

describe('useFetch', () => {
  beforeEach(() => {
    vi.restoreAllMocks();
  });

  it('成功获取数据', async () => {
    vi.stubGlobal(
      'fetch',
      vi.fn().mockResolvedValue({
        ok: true,
        json: () => Promise.resolve({ name: 'Alice' }),
      }),
    );

    const { result } = renderHook(() => useFetch<{ name: string }>('/api/user'));

    expect(result.current.loading).toBe(true);

    await waitFor(() => {
      expect(result.current.loading).toBe(false);
    });

    expect(result.current.data).toEqual({ name: 'Alice' });
    expect(result.current.error).toBeNull();

    vi.unstubAllGlobals();
  });

  it('处理请求错误', async () => {
    vi.stubGlobal(
      'fetch',
      vi.fn().mockResolvedValue({
        ok: false,
        status: 404,
      }),
    );

    const { result } = renderHook(() => useFetch('/api/missing'));

    await waitFor(() => {
      expect(result.current.loading).toBe(false);
    });

    expect(result.current.data).toBeNull();
    expect(result.current.error).toBe('HTTP 404');

    vi.unstubAllGlobals();
  });

  it('处理网络错误', async () => {
    vi.stubGlobal('fetch', vi.fn().mockRejectedValue(new Error('网络断开')));

    const { result } = renderHook(() => useFetch('/api/data'));

    await waitFor(() => {
      expect(result.current.loading).toBe(false);
    });

    expect(result.current.error).toBe('网络断开');

    vi.unstubAllGlobals();
  });

  it('refetch 应重新获取数据', async () => {
    const mockFetch = vi
      .fn()
      .mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve({ count: 1 }),
      })
      .mockResolvedValueOnce({
        ok: true,
        json: () => Promise.resolve({ count: 2 }),
      });

    vi.stubGlobal('fetch', mockFetch);

    const { result } = renderHook(() => useFetch<{ count: number }>('/api/data'));

    await waitFor(() => {
      expect(result.current.data?.count).toBe(1);
    });

    act(() => {
      result.current.refetch();
    });

    await waitFor(() => {
      expect(result.current.data?.count).toBe(2);
    });

    expect(mockFetch).toHaveBeenCalledTimes(2);

    vi.unstubAllGlobals();
  });
});
测试 Hook 的 rerender
import { renderHook } from '@testing-library/react';
import { describe, it, expect } from 'vitest';
import { useMemo } from 'react';

function useFormattedPrice(amount: number, currency: string) {
  const formatted = useMemo(() => {
    return new Intl.NumberFormat('zh-CN', {
      style: 'currency',
      currency,
    }).format(amount);
  }, [amount, currency]);

  return formatted;
}

describe('useFormattedPrice', () => {
  it('应格式化价格', () => {
    const { result } = renderHook(({ amount, currency }) => useFormattedPrice(amount, currency), {
      initialProps: { amount: 99.9, currency: 'CNY' },
    });

    expect(result.current).toContain('99.90');
  });

  it('props 变更时应重新格式化', () => {
    const { result, rerender } = renderHook(({ amount, currency }) => useFormattedPrice(amount, currency), {
      initialProps: { amount: 99.9, currency: 'CNY' },
    });

    rerender({ amount: 199.9, currency: 'CNY' });
    expect(result.current).toContain('199.90');

    rerender({ amount: 50, currency: 'USD' });
    expect(result.current).toContain('50.00');
  });
});
测试 Hook 的清理（unmount）
import { renderHook } from '@testing-library/react';
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';
import { useEffect, useState } from 'react';

function useInterval(callback: () => void, delay: number | null) {
  useEffect(() => {
    if (delay === null) return;

    const id = setInterval(callback, delay);
    return () => clearInterval(id);
  }, [callback, delay]);
}

describe('useInterval', () => {
  beforeEach(() => {
    vi.useFakeTimers();
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  it('应按间隔调用回调', () => {
    const callback = vi.fn();
    renderHook(() => useInterval(callback, 1000));

    expect(callback).not.toHaveBeenCalled();

    vi.advanceTimersByTime(1000);
    expect(callback).toHaveBeenCalledTimes(1);

    vi.advanceTimersByTime(2000);
    expect(callback).toHaveBeenCalledTimes(3);
  });

  it('unmount 后应停止调用', () => {
    const callback = vi.fn();
    const { unmount } = renderHook(() => useInterval(callback, 1000));

    vi.advanceTimersByTime(2000);
    expect(callback).toHaveBeenCalledTimes(2);

    unmount();

    vi.advanceTimersByTime(3000);
    expect(callback).toHaveBeenCalledTimes(2);
  });

  it('delay 为 null 时不启动定时器', () => {
    const callback = vi.fn();
    renderHook(() => useInterval(callback, null));

    vi.advanceTimersByTime(5000);
    expect(callback).not.toHaveBeenCalled();
  });
});
测试 Context
创建通用 Wrapper
import { render, screen, fireEvent } from '@testing-library/react'
import { renderHook, act } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import { createContext, useContext, useState, type ReactNode } from 'react'

interface Theme {
  mode: 'light' | 'dark'
  primaryColor: string
}

interface ThemeContextType {
  theme: Theme
  toggleMode: () => void
  setPrimaryColor: (color: string) => void
}

const ThemeContext = createContext<ThemeContextType | null>(null)

function ThemeProvider({ children, initialMode = 'light' as const }: {
  children: ReactNode
  initialMode?: 'light' | 'dark'
}) {
  const [theme, setTheme] = useState<Theme>({
    mode: initialMode,
    primaryColor: '#1890ff',
  })

  const toggleMode = () => {
    setTheme((prev) => ({
      ...prev,
      mode: prev.mode === 'light' ? 'dark' : 'light',
    }))
  }

  const setPrimaryColor = (color: string) => {
    setTheme((prev) => ({ ...prev, primaryColor: color }))
  }

  return (
    <ThemeContext.Provider value={{ theme, toggleMode, setPrimaryColor }}>
      {children}
    </ThemeContext.Provider>
  )
}

function useTheme(): ThemeContextType {
  const context = useContext(ThemeContext)
  if (!context) throw new Error('useTheme 必须在 ThemeProvider 内使用')
  return context
}

function createThemeWrapper(initialMode: 'light' | 'dark' = 'light') {
  return function Wrapper({ children }: { children: ReactNode }) {
    return <ThemeProvider initialMode={initialMode}>{children}</ThemeProvider>
  }
}

describe('ThemeContext', () => {
  it('提供默认主题', () => {
    const { result } = renderHook(() => useTheme(), {
      wrapper: createThemeWrapper(),
    })

    expect(result.current.theme.mode).toBe('light')
    expect(result.current.theme.primaryColor).toBe('#1890ff')
  })

  it('切换主题模式', () => {
    const { result } = renderHook(() => useTheme(), {
      wrapper: createThemeWrapper(),
    })

    act(() => {
      result.current.toggleMode()
    })

    expect(result.current.theme.mode).toBe('dark')
  })

  it('修改主色调', () => {
    const { result } = renderHook(() => useTheme(), {
      wrapper: createThemeWrapper(),
    })

    act(() => {
      result.current.setPrimaryColor('#ff4d4f')
    })

    expect(result.current.theme.primaryColor).toBe('#ff4d4f')
  })

  it('使用暗色模式初始化', () => {
    const { result } = renderHook(() => useTheme(), {
      wrapper: createThemeWrapper('dark'),
    })

    expect(result.current.theme.mode).toBe('dark')
  })

  it('不在 Provider 内使用应抛出异常', () => {
    expect(() => {
      renderHook(() => useTheme())
    }).toThrow('useTheme 必须在 ThemeProvider 内使用')
  })
})
测试组件中的 Context 消费
import { render, screen, fireEvent } from '@testing-library/react'
import { describe, it, expect } from 'vitest'

function ThemeToggle() {
  const { theme, toggleMode } = useTheme()
  return (
    <button onClick={toggleMode}>
      当前: {theme.mode === 'light' ? '亮色' : '暗色'}
    </button>
  )
}

function renderWithTheme(ui: React.ReactElement, mode: 'light' | 'dark' = 'light') {
  return render(
    <ThemeProvider initialMode={mode}>{ui}</ThemeProvider>
  )
}

describe('ThemeToggle 组件', () => {
  it('显示当前主题模式', () => {
    renderWithTheme(<ThemeToggle />)
    expect(screen.getByText('当前: 亮色')).toBeInTheDocument()
  })

  it('点击后切换模式', () => {
    renderWithTheme(<ThemeToggle />)

    fireEvent.click(screen.getByRole('button'))
    expect(screen.getByText('当前: 暗色')).toBeInTheDocument()

    fireEvent.click(screen.getByRole('button'))
    expect(screen.getByText('当前: 亮色')).toBeInTheDocument()
  })

  it('从暗色模式开始', () => {
    renderWithTheme(<ThemeToggle />, 'dark')
    expect(screen.getByText('当前: 暗色')).toBeInTheDocument()
  })
})
多 Provider 组合测试
import { render, screen } from '@testing-library/react'
import { renderHook } from '@testing-library/react'
import { describe, it, expect } from 'vitest'
import { type ReactNode } from 'react'

interface AuthUser {
  id: string
  name: string
  role: 'admin' | 'user'
}

function createAllProviders({
  user = null,
  theme = 'light',
}: {
  user?: AuthUser | null
  theme?: 'light' | 'dark'
} = {}) {
  return function AllProviders({ children }: { children: ReactNode }) {
    return (
      <ThemeProvider initialMode={theme}>
        {children}
      </ThemeProvider>
    )
  }
}

function renderWithProviders(
  ui: React.ReactElement,
  options: { user?: AuthUser | null; theme?: 'light' | 'dark' } = {}
) {
  return render(ui, {
    wrapper: createAllProviders(options),
  })
}

describe('多 Provider 组合', () => {
  it('同时使用 Theme 和 Auth', () => {
    const user: AuthUser = { id: '1', name: 'Alice', role: 'admin' }

    renderWithProviders(
      <div>
        <ThemeToggle />
      </div>,
      { user, theme: 'dark' }
    )

    expect(screen.getByText('当前: 暗色')).toBeInTheDocument()
  })
})
测试模式总结
常见误区与纠正
误区一：不使用 act 包装状态更新
忽略 act 会导致 React 警告，且状态更新可能不被应用。所有触发状态变更的操作都需要 act 包装。


误区二：直接测试 Hook 内部状态
Hook 的内部实现是私有的，应通过返回的 API（方法和值）来测试行为，而非检查内部 state。


误区三：在每个测试中重复创建 Provider
应提取公共的 wrapper/renderWith 工具函数，避免重复代码。


误区四：忽略 Hook 的错误边界
当 Hook 在无 Provider 时应抛出明确错误，需要测试这一保护路径。


小结
测试 Hooks 和 Context 需要掌握正确的工具和模式。


核心要点回顾：


    

  • renderHook 是独立测试自定义 Hook 的核心工具
    • 

  • act 包装所有触发状态变更的操作
    • 

  • waitFor 处理异步 Hook 的状态更新
    • 

  • rerender 测试 props 变更对 Hook 的影响
    • 

  • unmount 验证清理逻辑（clearInterval、removeEventListener）
    • 

  • 创建可复用的 wrapper 函数注入 Context Provider
    • 

  • 多 Provider 组合使用 createAllProviders 模式
    • 

  • 测试 Hook 在无 Provider 时的错误处理
