集成测试与端到端测试

测试范围的扩展

单元测试验证独立单元的正确性，但无法保证模块之间的协作是否正常。集成测试和端到端测试将测试范围扩展到更高层次——前者验证模块间的交互，后者验证完整的用户流程。



集成测试
集成测试在前端中通常指：渲染一个包含多个子组件的页面级组件，模拟用户的完整操作流程，验证组件间的数据传递和状态变更是否正确。


页面级组件测试
// src/features/dashboard/components/Dashboard.test.tsx
import { describe, it, expect } from "vitest";
import { render, screen, within } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { setupServer } from "msw/node";
import { http, HttpResponse } from "msw";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { Dashboard } from "./Dashboard";

const mockUsers = [
  { id: 1, name: "张三", role: "admin", status: "active" },
  { id: 2, name: "李四", role: "user", status: "active" },
  { id: 3, name: "王五", role: "user", status: "inactive" },
];

const server = setupServer(
  http.get("/api/users", ({ request }) => {
    const url = new URL(request.url);
    const status = url.searchParams.get("status");

    const filtered = status
      ? mockUsers.filter((u) => u.status === status)
      : mockUsers;

    return HttpResponse.json({ users: filtered, total: filtered.length });
  }),

  http.delete("/api/users/:id", ({ params }) => {
    return HttpResponse.json({ success: true });
  })
);

beforeAll(() => server.listen());
afterEach(() => server.resetHandlers());
afterAll(() => server.close());

function renderDashboard() {
  const queryClient = new QueryClient({
    defaultOptions: {
      queries: { retry: false },
    },
  });

  return render(
    <QueryClientProvider client={queryClient}>
      <Dashboard />
    </QueryClientProvider>
  );
}

describe("Dashboard", () => {
  it("should load and display users", async () => {
    renderDashboard();

    expect(await screen.findByText("张三")).toBeInTheDocument();
    expect(screen.getByText("李四")).toBeInTheDocument();
    expect(screen.getByText("王五")).toBeInTheDocument();
  });

  it("should filter users by status", async () => {
    const user = userEvent.setup();
    renderDashboard();

    await screen.findByText("张三");

    const statusFilter = screen.getByRole("combobox", { name: "状态筛选" });
    await user.selectOptions(statusFilter, "active");

    expect(await screen.findByText("张三")).toBeInTheDocument();
    expect(screen.getByText("李四")).toBeInTheDocument();
    expect(screen.queryByText("王五")).not.toBeInTheDocument();
  });

  it("should delete a user after confirmation", async () => {
    const user = userEvent.setup();
    renderDashboard();

    await screen.findByText("张三");

    const rows = screen.getAllByRole("row");
    const firstUserRow = rows[1];
    const deleteButton = within(firstUserRow).getByRole("button", {
      name: "删除",
    });

    await user.click(deleteButton);

    const confirmButton = await screen.findByRole("button", {
      name: "确认删除",
    });
    await user.click(confirmButton);

    expect(screen.queryByText("张三")).not.toBeInTheDocument();
  });
});
路由集成测试
// src/app/router.test.tsx
import { describe, it, expect } from "vitest";
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { MemoryRouter } from "react-router-dom";
import { App } from "./App";

describe("Router", () => {
  it("should navigate from home to dashboard", async () => {
    const user = userEvent.setup();

    render(
      <MemoryRouter initialEntries={["/"]}>
        <App />
      </MemoryRouter>
    );

    expect(screen.getByText("欢迎来到首页")).toBeInTheDocument();

    await user.click(screen.getByRole("link", { name: "仪表盘" }));

    expect(await screen.findByText("仪表盘")).toBeInTheDocument();
  });

  it("should redirect to login when unauthenticated", async () => {
    render(
      <MemoryRouter initialEntries={["/dashboard"]}>
        <App />
      </MemoryRouter>
    );

    expect(await screen.findByText("请先登录")).toBeInTheDocument();
  });
});


Playwright 端到端测试
Playwright 是 Microsoft 开源的端到端测试框架，支持 Chromium、Firefox、WebKit 三大浏览器引擎。


基础配置
// playwright.config.ts
import { defineConfig, devices } from '@playwright/test';

export default defineConfig({
  testDir: './e2e',
  fullyParallel: true,
  forbidOnly: !!process.env.CI,
  retries: process.env.CI ? 2 : 0,
  workers: process.env.CI ? 1 : undefined,
  reporter: [['html'], ['json', { outputFile: 'test-results.json' }]],
  use: {
    baseURL: 'http://localhost:3000',
    trace: 'on-first-retry',
    screenshot: 'only-on-failure',
    video: 'retain-on-failure',
  },
  projects: [
    {
      name: 'chromium',
      use: { ...devices['Desktop Chrome'] },
    },
    {
      name: 'firefox',
      use: { ...devices['Desktop Firefox'] },
    },
    {
      name: 'webkit',
      use: { ...devices['Desktop Safari'] },
    },
    {
      name: 'mobile-chrome',
      use: { ...devices['Pixel 5'] },
    },
  ],
  webServer: {
    command: 'pnpm run dev',
    url: 'http://localhost:3000',
    reuseExistingServer: !process.env.CI,
  },
});
用户登录流程测试
// e2e/auth.spec.ts
import { test, expect } from '@playwright/test';

test.describe('认证流程', () => {
  test('用户可以成功登录', async ({ page }) => {
    await page.goto('/login');

    await page.getByLabel('邮箱').fill('admin@example.com');
    await page.getByLabel('密码').fill('password123');
    await page.getByRole('button', { name: '登录' }).click();

    await expect(page).toHaveURL('/dashboard');
    await expect(page.getByText('欢迎回来')).toBeVisible();
  });

  test('无效凭证显示错误', async ({ page }) => {
    await page.goto('/login');

    await page.getByLabel('邮箱').fill('wrong@example.com');
    await page.getByLabel('密码').fill('wrong');
    await page.getByRole('button', { name: '登录' }).click();

    await expect(page.getByRole('alert')).toContainText('用户名或密码错误');
    await expect(page).toHaveURL('/login');
  });

  test('未登录用户被重定向到登录页', async ({ page }) => {
    await page.goto('/dashboard');
    await expect(page).toHaveURL('/login');
  });
});
Page Object Model
// e2e/pages/LoginPage.ts
import { type Page, type Locator } from '@playwright/test';

export class LoginPage {
  readonly page: Page;
  readonly emailInput: Locator;
  readonly passwordInput: Locator;
  readonly submitButton: Locator;
  readonly errorAlert: Locator;

  constructor(page: Page) {
    this.page = page;
    this.emailInput = page.getByLabel('邮箱');
    this.passwordInput = page.getByLabel('密码');
    this.submitButton = page.getByRole('button', { name: '登录' });
    this.errorAlert = page.getByRole('alert');
  }

  async goto() {
    await this.page.goto('/login');
  }

  async login(email: string, password: string) {
    await this.emailInput.fill(email);
    await this.passwordInput.fill(password);
    await this.submitButton.click();
  }
}
// e2e/auth-pom.spec.ts
import { test, expect } from '@playwright/test';
import { LoginPage } from './pages/LoginPage';

test('使用 POM 模式进行登录测试', async ({ page }) => {
  const loginPage = new LoginPage(page);
  await loginPage.goto();
  await loginPage.login('admin@example.com', 'password123');

  await expect(page).toHaveURL('/dashboard');
});


测试策略选择




场景
推荐测试类型
理由





格式化函数
单元测试
纯函数，输入输出确定




表单组件
组件测试
验证用户交互和状态变更




列表页面（含筛选、分页）
集成测试
多组件协作，需要 Mock API




登录→跳转→操作
E2E 测试
完整用户流程，涉及路由和鉴权




支付流程
E2E 测试
关键业务路径，需端到端验证







小结
    

  • 集成测试验证模块间的协作，填补单元测试和 E2E 测试之间的空白。
    • 

  • React Testing Library 的集成测试以用户视角驱动，配合 MSW 模拟网络请求。
    • 

  • Playwright 支持多浏览器 E2E 测试，Page Object Model 提升测试代码的可维护性。
    • 

  • 测试策略的选择取决于被测目标的范围和复杂度——不是所有场景都需要 E2E 测试。
