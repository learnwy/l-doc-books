数据采集与 ETL

知识要点

数据采集与 ETL（Extract-Transform-Load）是数据工程的第一个环节——"垃圾进、垃圾出"，数据源的质量直接决定了下游所有分析和模型的可信度。大数据工程师需要掌握从多源异构系统中高效、可靠地获取数据并转化为可用格式的全套技能。

Web 爬虫与数据抓取

Web 爬虫是获取互联网公开数据的核心技术：

import requests
from bs4 import BeautifulSoup
import time
import random

class ProductCrawler:
    def __init__(self, base_url, headers=None):
        self.base_url = base_url
        self.session = requests.Session()
        self.session.headers.update(headers or {
            'User-Agent': 'Mozilla/5.0 (compatible; DataBot/1.0)'
        })
    
    def fetch_page(self, url, retries=3):
        for attempt in range(retries):
            try:
                resp = self.session.get(url, timeout=10)
                resp.raise_for_status()
                time.sleep(random.uniform(1, 3))  # 礼貌爬取
                return resp.text
            except requests.RequestException as e:
                if attempt == retries - 1:
                    raise
                time.sleep(2 ** attempt)
    
    def parse_product(self, html):
        soup = BeautifulSoup(html, 'lxml')
        return {
            'title': soup.select_one('.product-title').text.strip(),
            'price': float(soup.select_one('.price').text.replace('¥', '')),
            'rating': float(soup.select_one('.rating').get('data-score', 0)),
            'review_count': int(soup.select_one('.reviews').text.split()[0])
        }

爬虫伦理与合规：

  • 遵守 robots.txt 规则
  • 控制请求频率（每秒不超过 1-2 次）
  • 不爬取需要登录的私有数据
  • 注意数据版权和个人隐私保护

API 数据采集

现代数据集成越来越依赖 API 而非爬虫：

import asyncio
import aiohttp
from datetime import datetime, timedelta

class APICollector:
    def __init__(self, api_key, base_url):
        self.api_key = api_key
        self.base_url = base_url
        self.semaphore = asyncio.Semaphore(5)  # 并发控制
    
    async def fetch_batch(self, session, endpoint, params):
        async with self.semaphore:
            headers = {'Authorization': f'Bearer {self.api_key}'}
            async with session.get(
                f"{self.base_url}/{endpoint}",
                params=params,
                headers=headers
            ) as resp:
                if resp.status == 429:  # Rate limit
                    await asyncio.sleep(60)
                    return await self.fetch_batch(session, endpoint, params)
                return await resp.json()
    
    async def collect_date_range(self, start_date, end_date):
        async with aiohttp.ClientSession() as session:
            tasks = []
            current = start_date
            while current <= end_date:
                params = {'date': current.strftime('%Y-%m-%d'), 'limit': 1000}
                tasks.append(self.fetch_batch(session, 'events', params))
                current += timedelta(days=1)
            return await asyncio.gather(*tasks)

Airflow 工作流调度
Apache Airflow 是数据工程领域最流行的工作流编排工具：


from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.operators.bash import BashOperator
from airflow.sensors.sql import SqlSensor
from datetime import datetime, timedelta

default_args = {
    'owner': 'data-team',
    'retries': 3,
    'retry_delay': timedelta(minutes=5),
    'email_on_failure': True,
    'email': ['data-alerts@company.com'],
}

with DAG(
    'daily_order_etl',
    default_args=default_args,
    schedule_interval='0 2 * * *',  # 每天凌晨 2 点
    start_date=datetime(2026, 1, 1),
    catchup=False,
    tags=['etl', 'orders'],
) as dag:
    
    # 等待上游数据就绪
    wait_source = SqlSensor(
        task_id='wait_source_ready',
        conn_id='mysql_source',
        sql="SELECT COUNT(*) FROM orders WHERE DATE(created_at) = '{{ ds }}'",
        mode='poke',
        poke_interval=300,
        timeout=3600,
    )
    
    # 抽取
    extract = PythonOperator(
        task_id='extract_orders',
        python_callable=extract_daily_orders,
        op_kwargs={'date': '{{ ds }}'},
    )
    
    # 转换
    transform = PythonOperator(
        task_id='transform_orders',
        python_callable=transform_orders,
    )
    
    # 加载到 Hive
    load = BashOperator(
        task_id='load_to_hive',
        bash_command='hive -f /opt/sql/load_orders.hql --hivevar dt={{ ds }}',
    )
    
    # 数据质量检查
    quality_check = PythonOperator(
        task_id='data_quality_check',
        python_callable=run_quality_checks,
    )
    
    wait_source >> extract >> transform >> load >> quality_check
数据质量监控
数据质量是 ETL 流水线可信度的最终保障：


class DataQualityChecker:
    def __init__(self, df, table_name):
        self.df = df
        self.table_name = table_name
        self.issues = []
    
    def check_completeness(self, columns, threshold=0.95):
        """检查非空率"""
        for col in columns:
            rate = self.df[col].notna().mean()
            if rate < threshold:
                self.issues.append(f"{col} 非空率 {rate:.2%} < {threshold:.2%}")
    
    def check_uniqueness(self, key_columns):
        """检查主键唯一性"""
        dup_count = self.df.duplicated(subset=key_columns).sum()
        if dup_count > 0:
            self.issues.append(f"主键重复 {dup_count} 行")
    
    def check_freshness(self, time_col, max_delay_hours=4):
        """检查数据新鲜度"""
        latest = self.df[time_col].max()
        delay = (datetime.now() - latest).total_seconds() / 3600
        if delay > max_delay_hours:
            self.issues.append(f"数据延迟 {delay:.1f} 小时")
    
    def check_value_range(self, col, min_val, max_val):
        """检查值域范围"""
        outliers = self.df[(self.df[col] < min_val) | (self.df[col] > max_val)]
        if len(outliers) > 0:
            self.issues.append(f"{col} 有 {len(outliers)} 条超出范围 [{min_val}, {max_val}]")
    
    def report(self):
        if self.issues:
            raise DataQualityError(f"{self.table_name} 质量检查失败:\n" + "\n".join(self.issues))
        return True

小结
数据采集与 ETL 是数据工程的起点和生命线。Web 爬虫和 API 采集从外部获取数据，ETL 流程将异构数据源转化为统一格式，Airflow 编排复杂的数据管道依赖关系，数据质量监控确保下游分析的可靠性。掌握这些技能后，你将能独立构建从数据源到数据仓库的完整数据流水线。
