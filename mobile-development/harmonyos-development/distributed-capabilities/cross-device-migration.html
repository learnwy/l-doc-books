跨设备迁移与流转

概述

跨设备迁移是 HarmonyOS 分布式能力中最直接面向用户体验的特性。它允许用户将正在进行的任务（如正在浏览的文章、填写到一半的表单）从一台设备无缝转移到另一台设备继续操作。这种能力对用户来说是"接力使用"，对开发者来说是 UIAbility 状态的序列化、传输与恢复过程。

源端状态保存
export default class ReadingAbility extends UIAbility {
  // 当前阅读状态
  private currentArticleId: string = '';
  private scrollPosition: number = 0;
  private highlights: string[] = [];

  onContinue(wantParam: Record<string, Object>): AbilityConstant.OnContinueResult {
    // 1. 保存需要迁移的状态数据
    wantParam['articleId'] = this.currentArticleId;
    wantParam['scrollPosition'] = this.scrollPosition;
    wantParam['highlights'] = JSON.stringify(this.highlights);
    wantParam['migrateTimestamp'] = Date.now();

    // 2. 验证数据完整性
    if (!this.currentArticleId) {
      // 没有有效状态，拒绝迁移
      return AbilityConstant.OnContinueResult.REJECT;
    }

    // 3. 同意迁移
    return AbilityConstant.OnContinueResult.AGREE;
  }

  // 迁移完成后源端清理（可选）
  onNewWant(want: Want): void {
    const reason = want.parameters?.['ohos.aafwk.param.launchReason'] as number;
    if (reason === AbilityConstant.LaunchReason.CONTINUATION) {
      // 迁移成功，关闭源端页面
      this.context.terminateSelf();
    }
  }
}
目标端状态恢复
export default class ReadingAbility extends UIAbility {
  onCreate(want: Want, launchParam: AbilityConstant.LaunchParam): void {
    if (launchParam.launchReason === AbilityConstant.LaunchReason.CONTINUATION) {
      // 这是一次迁移恢复
      this.restoreFromMigration(want);
    } else {
      // 正常启动
      this.normalLaunch(want);
    }
  }

  private restoreFromMigration(want: Want): void {
    const params = want.parameters;
    if (!params) return;

    const articleId = params['articleId'] as string;
    const scrollPosition = params['scrollPosition'] as number;
    const highlights = JSON.parse((params['highlights'] as string) || '[]');

    // 将恢复数据传递给 UI 层
    AppStorage.setOrCreate('restoredArticleId', articleId);
    AppStorage.setOrCreate('restoredScrollPosition', scrollPosition);
    AppStorage.setOrCreate('restoredHighlights', highlights);

    console.info(`迁移恢复: 文章=${articleId}, 位置=${scrollPosition}`);
  }

  private normalLaunch(want: Want): void {
    // 正常启动逻辑
  }
}
协同流转
协同流转允许应用在多台设备上同时运行并协作：


import { distributedDeviceManager } from '@kit.DistributedServiceKit';

class CollaborationManager {
  private deviceManager: distributedDeviceManager.DeviceManager | null = null;

  async initialize(): Promise<void> {
    this.deviceManager = distributedDeviceManager.createDeviceManager('com.example.myapp');
  }

  // 发现可用设备
  async discoverDevices(): Promise<DeviceInfo[]> {
    const devices = this.deviceManager?.getAvailableDeviceListSync() || [];
    return devices.map((d) => ({
      id: d.networkId,
      name: d.deviceName,
      type: d.deviceType,
    }));
  }

  // 启动协同流转
  async startCollaboration(targetDeviceId: string, taskData: object): Promise<void> {
    const context = getContext(this) as common.UIAbilityContext;

    const want: Want = {
      deviceId: targetDeviceId,
      bundleName: 'com.example.myapp',
      abilityName: 'CollabAbility',
      parameters: {
        collaborationMode: 'multi_screen',
        taskData: JSON.stringify(taskData),
      },
    };

    await context.startAbility(want);
  }
}

// 大数据迁移：先存文件，再传路径
import { fileIo } from '@kit.CoreFileKit';

async function prepareLargeData(data: ArrayBuffer): Promise<string> {
  const dir = getContext().distributedFilesDir;
  const filePath = `${dir}/migration_${Date.now()}.dat`;

  const file = fileIo.openSync(filePath, fileIo.OpenMode.CREATE | fileIo.OpenMode.WRITE_ONLY);
  fileIo.writeSync(file.fd, data);
  fileIo.closeSync(file.fd);

  return filePath; // 将路径通过 wantParam 传递
}

  onContinue(wantParam: Record<string, Object>): AbilityConstant.OnContinueResult {
    try {
      // 检查迁移前提条件
      if (!this.canMigrate()) {
        return AbilityConstant.OnContinueResult.REJECT;

      const stateSnapshot = this.captureState();
      wantParam['state'] = JSON.stringify(stateSnapshot);

小结
跨设备迁移的核心是状态的序列化与恢复。开发者需要关注三个关键点：选择合适的数据传输方式（wantParam vs 分布式文件）、处理迁移异常（网络中断、版本不兼容）、以及优化用户体验（迁移过程的 Loading 提示、失败后的回退策略）。协同流转则进一步扩展了迁移的概念，使多设备能同时参与同一个任务，为超级终端体验奠定基础。
