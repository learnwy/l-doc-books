从 Android 迁移到 HarmonyOS

概述

随着 HarmonyOS NEXT 生态的成熟，越来越多的 Android 应用面临迁移需求。由于 HarmonyOS NEXT 完全脱离 AOSP，迁移不是简单的适配——而是基于新框架的重新实现。本文提供系统性的迁移方法论，从技术栈映射、架构重设计到具体实施步骤，帮助团队高效完成迁移。

// HarmonyOS (ArkTS):
@Observed
class ArticleViewModel {
  articles: ArticleInfo[] = []
  isLoading: boolean = false
  errorMessage: string = ''

  private repository: ArticleRepository = new ArticleRepository()

  async loadArticles(): Promise<void> {
    this.isLoading = true
    this.errorMessage = ''

    try {
      this.articles = await this.repository.getArticles()
    } catch (error) {
      this.errorMessage = `加载失败: ${(error as Error).message}`
    } finally {
      this.isLoading = false
    }
  }

  async refreshArticles(): Promise<void> {
    try {
      this.articles = await this.repository.getArticles(true)
    } catch (error) {
      console.error(`刷新失败: ${error}`)
    }
  }
}

// 组件中使用
@Entry
@Component
struct ArticlePage {
  @State viewModel: ArticleViewModel = new ArticleViewModel()

  aboutToAppear(): void {
    this.viewModel.loadArticles()
  }

  build() {
    Column() {
      if (this.viewModel.isLoading) {
        LoadingProgress().width(48).height(48)
      } else if (this.viewModel.errorMessage) {
        Text(this.viewModel.errorMessage).fontColor(Color.Red)
      } else {
        List() {
          ForEach(this.viewModel.articles, (item: ArticleInfo) => {
            ListItem() { ArticleCard({ article: item }) }
          }, (item: ArticleInfo) => item.id)
        }
      }
    }
  }
}

列表页迁移
// Android RecyclerView + ListAdapter 模式 → HarmonyOS List + LazyForEach

// 数据源（替代 ListAdapter 的 DiffUtil）
class ProductDataSource implements IDataSource {
  private items: ProductInfo[] = []
  private listeners: DataChangeListener[] = []

  totalCount(): number { return this.items.length }
  getData(index: number): ProductInfo { return this.items[index] }

  registerDataChangeListener(listener: DataChangeListener): void {
    this.listeners.push(listener)
  }
  unregisterDataChangeListener(listener: DataChangeListener): void {
    const idx = this.listeners.indexOf(listener)
    if (idx >= 0) this.listeners.splice(idx, 1)
  }

  updateList(newItems: ProductInfo[]): void {
    this.items = newItems
    this.listeners.forEach(l => l.onDataReloaded())
  }
}

@Component
struct ProductListPage {
  private dataSource: ProductDataSource = new ProductDataSource()

  build() {
    List({ space: 8 }) {
      LazyForEach(this.dataSource, (item: ProductInfo) => {
        ListItem() {
          this.ProductCard(item)
        }
      }, (item: ProductInfo) => item.id)
    }
    .divider({ strokeWidth: 0.5, color: '#F0F0F0' })
    .edgeEffect(EdgeEffect.Spring)
    .cachedCount(5)
  }

  @Builder
  ProductCard(item: ProductInfo) {
    Row() {
      Image(item.imageUrl)
        .width(80).height(80)
        .borderRadius(8)
      Column() {
        Text(item.name).fontSize(16).maxLines(2)
        Text(`¥${item.price}`).fontColor('#FF4444').fontSize(18)
      }
      .layoutWeight(1)
      .margin({ left: 12 })
      .alignItems(HorizontalAlign.Start)
    }
    .padding(12)
    .width('100%')
  }
}

小结
从 Android 迁移到 HarmonyOS 本质上是一次基于新平台的应用重构，而非简单移植。成功迁移的关键是：先充分理解 HarmonyOS 的设计哲学（声明式 UI + Stage 模型 + 分布式），再有计划地分层迁移（数据层→组件库→页面），同时利用迁移机会优化原有架构中的技术债。
