原生 API 与系统服务集成

概述

HarmonyOS NEXT 提供了丰富的系统原生 API，覆盖网络通信、文件管理、媒体服务、通知推送、安全能力等核心领域。这些 API 以 Kit（能力套件）的形式组织，开发者通过 import { xxx } from '@kit.XxxKit' 统一引入。相比 Android 的碎片化 API，HarmonyOS 的 Kit 体系设计更加一致和现代化。

网络通信
HTTP 请求
import { http } from '@kit.NetworkKit';

class ApiService {
  private baseUrl: string = 'https://api.example.com';

  async get<T>(path: string, params?: Record<string, string>): Promise<T> {
    const httpRequest = http.createHttp();

    let url = `${this.baseUrl}${path}`;
    if (params) {
      const query = Object.entries(params)
        .map(([k, v]) => `${k}=${encodeURIComponent(v)}`)
        .join('&');
      url += `?${query}`;
    }

    try {
      const response = await httpRequest.request(url, {
        method: http.RequestMethod.GET,
        header: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${AppStorage.get<string>('token')}`,
        },
        connectTimeout: 10000,
        readTimeout: 30000,
      });

      if (response.responseCode === 200) {
        return JSON.parse(response.result as string) as T;
      }
      throw new Error(`HTTP ${response.responseCode}`);
    } finally {
      httpRequest.destroy();
    }
  }

  async post<T>(path: string, body: object): Promise<T> {
    const httpRequest = http.createHttp();

    try {
      const response = await httpRequest.request(`${this.baseUrl}${path}`, {
        method: http.RequestMethod.POST,
        header: { 'Content-Type': 'application/json' },
        extraData: JSON.stringify(body),
      });
      return JSON.parse(response.result as string) as T;
    } finally {
      httpRequest.destroy();
    }
  }
}
WebSocket 长连接
import { webSocket } from '@kit.NetworkKit';

class MessageChannel {
  private ws: webSocket.WebSocket | null = null;
  private reconnectTimer: number = 0;

  connect(url: string): void {
    this.ws = webSocket.createWebSocket();

    this.ws.on('open', () => {
      console.info('WebSocket 连接已建立');
    });

    this.ws.on('message', (err, data) => {
      if (typeof data === 'string') {
        const message = JSON.parse(data);
        this.handleMessage(message);
      }
    });

    this.ws.on('close', () => {
      console.info('WebSocket 连接关闭，准备重连');
      this.scheduleReconnect(url);
    });

    this.ws.connect(url);
  }

  send(data: object): void {
    this.ws?.send(JSON.stringify(data));
  }

  private handleMessage(message: object): void {
    emitter.emit({ eventId: 2001 }, { data: message });
  }

  private scheduleReconnect(url: string): void {
    this.reconnectTimer = setTimeout(() => {
      this.connect(url);
    }, 3000) as unknown as number;
  }
}

import { fileIo, picker } from '@kit.CoreFileKit';

class FileManager {
  // 写入文件
  async writeFile(fileName: string, content: string): Promise<void> {
    const dir = getContext().filesDir;
    const filePath = `${dir}/${fileName}`;

    const file = fileIo.openSync(
      filePath,
      fileIo.OpenMode.CREATE | fileIo.OpenMode.WRITE_ONLY | fileIo.OpenMode.TRUNCATE,
    );
    fileIo.writeSync(file.fd, content);
    fileIo.closeSync(file.fd);
  }

  // 读取文件
  async readFile(fileName: string): Promise<string> {
    const dir = getContext().filesDir;
    const filePath = `${dir}/${fileName}`;

    const file = fileIo.openSync(filePath, fileIo.OpenMode.READ_ONLY);
    const stat = fileIo.statSync(filePath);
    const buffer = new ArrayBuffer(stat.size);
    fileIo.readSync(file.fd, buffer);
    fileIo.closeSync(file.fd);

    const decoder = new util.TextDecoder();
    return decoder.decodeWithStream(new Uint8Array(buffer));
  }

  // 用户选择文件（通过 picker）
  async pickImage(): Promise<string[]> {
    const photoPicker = new picker.PhotoViewPicker();
    const result = await photoPicker.select({
      MIMEType: picker.PhotoViewMIMETypes.IMAGE_TYPE,
      maxSelectNumber: 9,
    });
    return result.photoUris;
  }
}
通知服务
import { notificationManager } from '@kit.NotificationKit';
import { wantAgent } from '@kit.AbilityKit';

class NotificationService {
  async sendBasicNotification(title: string, text: string, targetPage?: string): Promise<void> {
    // 创建点击跳转的 WantAgent
    let agent: WantAgent | undefined;
    if (targetPage) {
      const wantAgentInfo: wantAgent.WantAgentInfo = {
        wants: [
          {
            bundleName: 'com.example.myapp',
            abilityName: 'MainAbility',
            parameters: { targetPage },
          },
        ],
        actionType: wantAgent.OperationType.START_ABILITY,
        requestCode: 0,
      };
      agent = await wantAgent.getWantAgent(wantAgentInfo);
    }

    const request: notificationManager.NotificationRequest = {
      id: Date.now(),
      content: {
        notificationContentType: notificationManager.ContentType.NOTIFICATION_CONTENT_BASIC_TEXT,
        normal: {
          title: title,
          text: text,
        },
      },
      wantAgent: agent,
    };

    await notificationManager.publish(request);
  }

  async sendProgressNotification(title: string, progress: number): Promise<void> {
    const request: notificationManager.NotificationRequest = {
      id: 1001, // 固定 ID 用于更新进度
      content: {
        notificationContentType: notificationManager.ContentType.NOTIFICATION_CONTENT_BASIC_TEXT,
        normal: {
          title: title,
          text: `${progress}%`,
        },
      },
      template: {
        name: 'downloadTemplate',
        data: { progressValue: progress, progressMaxValue: 100 },
      },
    };
    await notificationManager.publish(request);
  }
}

import { abilityAccessCtrl, bundleManager, Permissions } from '@kit.AbilityKit';

class PermissionManager {
  async checkAndRequest(permissions: Permissions[]): Promise<boolean> {
    const atManager = abilityAccessCtrl.createAtManager();
    const bundleInfo = await bundleManager.getBundleInfoForSelf(
      bundleManager.BundleFlag.GET_BUNDLE_INFO_WITH_APPLICATION,
    );
    const tokenId = bundleInfo.appInfo.accessTokenId;

    // 检查哪些权限还未授予
    const ungrantedPermissions: Permissions[] = [];
    for (const permission of permissions) {
      const status = atManager.checkAccessTokenSync(tokenId, permission);
      if (status !== abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED) {
        ungrantedPermissions.push(permission);
      }
    }

    if (ungrantedPermissions.length === 0) return true;

    // 请求未授予的权限
    const context = getContext(this) as common.UIAbilityContext;
    const result = await atManager.requestPermissionsFromUser(context, ungrantedPermissions);

    return result.authResults.every((r) => r === abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED);
  }
}

安全能力
import { cryptoFramework } from '@kit.CryptoArchitectureKit';
import { userAuthentication } from '@kit.UserAuthenticationKit';

class SecurityService {
  // AES 加密
  async encrypt(plainText: string, key: string): Promise<ArrayBuffer> {
    const cipher = cryptoFramework.createCipher('AES256|CBC|PKCS7');
    const symKey = await this.generateSymKey(key);
    const iv = cryptoFramework.createIvParamsSpec(new Uint8Array(16));

    await cipher.init(cryptoFramework.CryptoMode.ENCRYPT_MODE, symKey, iv);
    const input: cryptoFramework.DataBlob = { data: new Uint8Array(new TextEncoder().encode(plainText)) };
    const output = await cipher.doFinal(input);
    return output.data.buffer;
  }

  // 生物认证
  async biometricAuth(): Promise<boolean> {
    const authParam: userAuthentication.AuthParam = {
      challenge: new Uint8Array(32),
      authType: [userAuthentication.UserAuthType.FINGERPRINT, userAuthentication.UserAuthType.FACE],
      authTrustLevel: userAuthentication.AuthTrustLevel.ATL3,
    };

    try {
      const authInstance = userAuthentication.getUserAuthInstance(authParam, {
        title: '身份验证',
        navigationButtonText: '取消',
      });
      const result = await authInstance.start();
      return result.result === userAuthentication.UserAuthResultCode.SUCCESS;
    } catch {
      return false;
    }
  }

  private async generateSymKey(seed: string): Promise<cryptoFramework.SymKey> {
    const keyGenerator = cryptoFramework.createSymKeyGenerator('AES256');
    return await keyGenerator.generateSymKey();
  }
}
小结
HarmonyOS 的 Kit 体系为开发者提供了统一、现代化的系统能力接口。关键 Kit 包括：NetworkKit（网络通信）、CoreFileKit（文件管理）、NotificationKit（通知推送）、CryptoArchitectureKit（加密安全）、UserAuthenticationKit（生物认证）。相比 Android 的碎片化 API，这些 Kit 设计一致性更强，且深度集成了分布式能力。在集成系统服务时，需特别注意权限申请的时机和异常处理的完备性。
