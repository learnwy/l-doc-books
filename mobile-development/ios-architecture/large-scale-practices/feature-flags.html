功能开关

功能开关（Feature Flags）使团队能够在不发版的情况下控制功能的启用与禁用。本文介绍远程配置、A/B 测试、开关生命周期管理以及灰度发布策略。

远程配置
通过远程配置服务动态控制功能开关：


protocol FeatureFlagServiceProtocol {
    func isEnabled(_ flag: FeatureFlag) -> Bool
    func value<T>(for flag: FeatureFlag) -> T?
    func refresh() async
}

enum FeatureFlag: String, CaseIterable {
    case newCheckoutFlow = "new_checkout_flow"
    case darkMode = "dark_mode"
    case aiRecommendation = "ai_recommendation"
    case socialSharing = "social_sharing"
    case newOnboarding = "new_onboarding"
}

final class RemoteFeatureFlagService: FeatureFlagServiceProtocol {
    private var flags: [String: Any] = [:]
    private let httpClient: HTTPClientProtocol
    private let cache: UserDefaults
    private let lock = NSLock()

    init(httpClient: HTTPClientProtocol, cache: UserDefaults = .standard) {
        self.httpClient = httpClient
        self.cache = cache
        loadCachedFlags()
    }

    func isEnabled(_ flag: FeatureFlag) -> Bool {
        lock.lock()
        defer { lock.unlock() }
        return flags[flag.rawValue] as? Bool ?? false
    }

    func value<T>(for flag: FeatureFlag) -> T? {
        lock.lock()
        defer { lock.unlock() }
        return flags[flag.rawValue] as? T
    }

    func refresh() async {
        do {
            let response: [String: Any] = try await httpClient.get("/api/feature-flags")
            lock.lock()
            flags = response
            lock.unlock()
            cacheFlags(response)
        } catch {
            // 使用缓存值
        }
    }

    private func loadCachedFlags() {
        if let cached = cache.dictionary(forKey: "feature_flags") {
            flags = cached
        }
    }

    private func cacheFlags(_ flags: [String: Any]) {
        cache.set(flags, forKey: "feature_flags")
    }
}


在代码中使用功能开关
struct HomeView: View {
    @Environment(\.featureFlags) private var featureFlags

    var body: some View {
        VStack {
            if featureFlags.isEnabled(.aiRecommendation) {
                AIRecommendationView()
            } else {
                ClassicRecommendationView()
            }

            if featureFlags.isEnabled(.socialSharing) {
                ShareButton()
            }
        }
    }
}
在 ViewModel 中使用：


final class CheckoutViewModel: ObservableObject {
    private let featureFlags: FeatureFlagServiceProtocol

    init(featureFlags: FeatureFlagServiceProtocol) {
        self.featureFlags = featureFlags
    }

    var checkoutSteps: [CheckoutStep] {
        var steps: [CheckoutStep] = [.cart, .address, .payment]

        if featureFlags.isEnabled(.newCheckoutFlow) {
            steps.insert(.review, at: 2)
        }

        return steps
    }
}


A/B 测试
功能开关与 A/B 测试结合，支持多变体实验：


enum ExperimentVariant: String {
    case control
    case variantA
    case variantB
}

protocol ExperimentServiceProtocol {
    func variant(for experiment: String) -> ExperimentVariant
    func trackExposure(experiment: String, variant: ExperimentVariant)
}

final class ExperimentService: ExperimentServiceProtocol {
    private let featureFlagService: FeatureFlagServiceProtocol
    private let analytics: AnalyticsServiceProtocol
    private let userId: String

    init(featureFlagService: FeatureFlagServiceProtocol, analytics: AnalyticsServiceProtocol, userId: String) {
        self.featureFlagService = featureFlagService
        self.analytics = analytics
        self.userId = userId
    }

    func variant(for experiment: String) -> ExperimentVariant {
        guard let rawValue: String = featureFlagService.value(for: FeatureFlag(rawValue: experiment) ?? .newCheckoutFlow) else {
            return .control
        }
        return ExperimentVariant(rawValue: rawValue) ?? .control
    }

    func trackExposure(experiment: String, variant: ExperimentVariant) {
        analytics.track(event: "experiment_exposure", properties: [
            "experiment": experiment,
            "variant": variant.rawValue,
            "user_id": userId
        ])
    }
}
struct PricingView: View {
    let experimentService: ExperimentServiceProtocol

    var body: some View {
        let variant = experimentService.variant(for: "pricing_page")

        Group {
            switch variant {
            case .control:
                ClassicPricingView()
            case .variantA:
                SimplifiedPricingView()
            case .variantB:
                ComparisonPricingView()
            }
        }
        .onAppear {
            experimentService.trackExposure(experiment: "pricing_page", variant: variant)
        }
    }
}

Kill Switch（紧急开关）
Kill Switch 用于紧急禁用有问题的功能：


final class KillSwitchManager {
    private let featureFlags: FeatureFlagServiceProtocol
    private let refreshInterval: TimeInterval

    init(featureFlags: FeatureFlagServiceProtocol, refreshInterval: TimeInterval = 60) {
        self.featureFlags = featureFlags
        self.refreshInterval = refreshInterval
    }

    func startMonitoring() {
        Task {
            while !Task.isCancelled {
                await featureFlags.refresh()
                try? await Task.sleep(for: .seconds(refreshInterval))
            }
        }
    }

    func isKilled(_ feature: FeatureFlag) -> Bool {
        !featureFlags.isEnabled(feature)
    }
}


灰度发布策略
struct RolloutConfig {
    let percentage: Int
    let targetUserGroups: [String]
    let targetRegions: [String]
    let startDate: Date
    let endDate: Date?
}

func shouldEnableForUser(userId: String, config: RolloutConfig) -> Bool {
    let hash = abs(userId.hashValue) % 100
    return hash < config.percentage
}


小结
    

  • 功能开关分为 Release、Experiment、Ops、Permission 四种类型
    • 

  • 远程配置服务动态下发开关值，本地缓存保证离线可用
    • 

  • A/B 测试通过多变体开关实现，需配合曝光埋点分析效果
    • 

  • 开关有明确的生命周期，全量上线后必须清理相关代码
    • 

  • Kill Switch 用于紧急禁用有问题的功能，需高频轮询刷新
    • 

  • 灰度发布通过用户哈希分桶实现，支持按比例、地区、用户组控制
    •
