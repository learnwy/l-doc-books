属性包装器注入

Swift 的属性包装器（Property Wrapper）为依赖注入提供了优雅的声明式语法。本文介绍如何实现 @Injected 属性包装器，以及与 SwiftUI Environment 的集成方式。

@Injected 属性包装器

通过自定义属性包装器实现声明式的依赖注入：

@propertyWrapper
struct Injected<T> {
    private var value: T

    init() {
        self.value = DependencyResolver.shared.resolve(T.self)
    }

    var wrappedValue: T {
        get { value }
        mutating set { value = newValue }
    }
}

使用方式非常简洁：

final class OrderListViewModel: ObservableObject {
    @Injected private var orderService: OrderServiceProtocol
    @Injected private var analytics: AnalyticsServiceProtocol

    @Published var orders: [Order] = []

    func loadOrders() async {
        analytics.track(event: "orders_loaded", properties: [:])
        orders = (try? await orderService.fetchOrders()) ?? []
    }
}

容器支撑的解析器

@Injected 需要一个全局解析器作为后端：

final class DependencyResolver: @unchecked Sendable {
    static let shared = DependencyResolver()

    private var factories: [String: () -> Any] = [:]
    private var singletons: [String: Any] = [:]
    private let lock = NSRecursiveLock()

    private init() {}

    func register<T>(_ type: T.Type, singleton: Bool = false, factory: @escaping () -> T) {
        lock.lock()
        defer { lock.unlock() }
        let key = String(describing: type)
        if singleton {
            singletons[key] = factory()
        } else {
            factories[key] = factory
        }
    }

    func resolve<T>(_ type: T.Type) -> T {
        lock.lock()
        defer { lock.unlock() }
        let key = String(describing: type)

        if let singleton = singletons[key] as? T {
            return singleton
        }

        guard let factory = factories[key], let instance = factory() as? T else {
            fatalError("依赖未注册: \(key)")
        }

        return instance
    }

    func reset() {
        lock.lock()
        defer { lock.unlock() }
        factories.removeAll()
        singletons.removeAll()
    }
}

在 App 启动时注册所有依赖：

@main
struct MyApp: App {
    init() {
        registerDependencies()
    }

    var body: some Scene {
        WindowGroup {
            ContentView()
        }
    }

    private func registerDependencies() {
        let resolver = DependencyResolver.shared

        resolver.register(HTTPClientProtocol.self, singleton: true) {
            URLSessionHTTPClient()
        }

        resolver.register(OrderServiceProtocol.self) {
            OrderService(httpClient: DependencyResolver.shared.resolve(HTTPClientProtocol.self))
        }

        resolver.register(AnalyticsServiceProtocol.self, singleton: true) {
            FirebaseAnalyticsService()
        }
    }
}

带键值的注入

支持通过 key 注入同一类型的不同实现：

@propertyWrapper
struct InjectedNamed<T> {
    private var value: T

    init(_ name: String) {
        self.value = NamedDependencyResolver.shared.resolve(T.self, name: name)
    }

    var wrappedValue: T {
        get { value }
        mutating set { value = newValue }
    }
}

final class NamedDependencyResolver: @unchecked Sendable {
    static let shared = NamedDependencyResolver()
    private var factories: [String: () -> Any] = [:]
    private let lock = NSRecursiveLock()

    func register<T>(_ type: T.Type, name: String, factory: @escaping () -> T) {
        lock.lock()
        defer { lock.unlock() }
        let key = "\(String(describing: type))_\(name)"
        factories[key] = factory
    }

    func resolve<T>(_ type: T.Type, name: String) -> T {
        lock.lock()
        defer { lock.unlock() }
        let key = "\(String(describing: type))_\(name)"
        guard let factory = factories[key], let instance = factory() as? T else {
            fatalError("命名依赖未注册: \(key)")
        }
        return instance
    }
}

使用示例：

final class NetworkViewModel: ObservableObject {
    @InjectedNamed("production") private var prodClient: HTTPClientProtocol
    @InjectedNamed("staging") private var stagingClient: HTTPClientProtocol
}

SwiftUI Environment 集成

SwiftUI 的 @Environment 和 @EnvironmentObject 本身就是依赖注入机制：

private struct HTTPClientKey: EnvironmentKey {
    static let defaultValue: HTTPClientProtocol = URLSessionHTTPClient()
}

private struct AnalyticsKey: EnvironmentKey {
    static let defaultValue: AnalyticsServiceProtocol = NoOpAnalyticsService()
}

extension EnvironmentValues {
    var httpClient: HTTPClientProtocol {
        get { self[HTTPClientKey.self] }
        set { self[HTTPClientKey.self] = newValue }
    }

    var analytics: AnalyticsServiceProtocol {
        get { self[AnalyticsKey.self] }
        set { self[AnalyticsKey.self] = newValue }
    }
}

在视图层注入和使用：

struct RootView: View {
    var body: some View {
        ContentView()
            .environment(\.httpClient, URLSessionHTTPClient())
            .environment(\.analytics, FirebaseAnalyticsService())
    }
}

struct OrderListView: View {
    @Environment(\.httpClient) private var httpClient
    @Environment(\.analytics) private var analytics

    var body: some View {
        List {
            Text("订单列表")
        }
        .onAppear {
            analytics.track(event: "order_list_viewed", properties: [:])
        }
    }
}

@Injected vs @Environment 对比

特性 @Injected @Environment
作用范围 全局可用 SwiftUI 视图树
注册方式 全局容器注册 视图树传递
适用层 ViewModel / Service View
测试替换 替换全局容器 .environment() 修改器
类型安全 运行时检查 编译期检查
线程模型 需自行保证 MainActor 


提示： 在 SwiftUI 项目中，View 层优先使用 @Environment；非 View 层（ViewModel、Service）使用 @Injected 或构造器注入。






测试中替换依赖
final class OrderListViewModelTests: XCTestCase {
    override func setUp() {
        super.setUp()
        DependencyResolver.shared.register(OrderServiceProtocol.self) {
            MockOrderService()
        }
        DependencyResolver.shared.register(AnalyticsServiceProtocol.self) {
            MockAnalyticsService()
        }
    }

    override func tearDown() {
        DependencyResolver.shared.reset()
        super.tearDown()
    }

    func testLoadOrders() async {
        let viewModel = OrderListViewModel()
        await viewModel.loadOrders()
        XCTAssertFalse(viewModel.orders.isEmpty)
    }
}


小结
    

  • @Injected 属性包装器提供了声明式的依赖注入语法
    • 

  • 底层通过全局 DependencyResolver 单例实现注册和解析
    • 

  • 命名注入支持同一协议的多个实现
    • 

  • SwiftUI @Environment 是视图层的原生 DI 机制，编译期类型安全
    • 

  • View 层用 @Environment，非 View 层用 @Injected 或构造器注入
    • 

  • 测试时通过替换全局容器中的注册来注入 Mock 实现
