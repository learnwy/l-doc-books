基于 DI 的测试

依赖注入的最大收益之一是让代码变得可测试。本文介绍 Mock 注入策略、基于协议的 Mock 创建、Test Double 分类以及结合 DI 的快照测试。

Mock 注入原理

DI 使得我们可以在测试中用 Mock 替换真实依赖，从而隔离被测对象：



基于协议的 Mock
定义协议后，Mock 的创建非常简单：


protocol ProductServiceProtocol {
    func fetchProducts(category: String) async throws -> [Product]
    func fetchProductDetail(id: String) async throws -> Product
    func addToFavorites(productId: String) async throws
}

final class MockProductService: ProductServiceProtocol {
    var fetchProductsResult: Result<[Product], Error> = .success([])
    var fetchProductDetailResult: Result<Product, Error> = .failure(NSError(domain: "", code: 0))
    var addToFavoritesCalled = false
    var addToFavoritesProductId: String?

    func fetchProducts(category: String) async throws -> [Product] {
        try fetchProductsResult.get()
    }

    func fetchProductDetail(id: String) async throws -> Product {
        try fetchProductDetailResult.get()
    }

    func addToFavorites(productId: String) async throws {
        addToFavoritesCalled = true
        addToFavoritesProductId = productId
    }
}

// Dummy — 仅满足编译
final class DummyLogger: LoggerProtocol {
    func log(_ message: String) {}
    func error(_ message: String) {}
}

// Stub — 返回固定值
final class StubUserRepository: UserRepositoryProtocol {
    func fetchCurrentUser() async throws -> User {
        User(id: "stub-id", name: "测试用户", email: "test@example.com")
    }
}

// Spy — 记录调用
final class SpyAnalytics: AnalyticsServiceProtocol {
    private(set) var trackedEvents: [(event: String, properties: [String: Any])] = []

    func track(event: String, properties: [String: Any]) {
        trackedEvents.append((event, properties))
    }
}

// Fake — 简化实现
final class FakeUserRepository: UserRepositoryProtocol {
    private var users: [String: User] = [:]

    func save(_ user: User) async throws {
        users[user.id] = user
    }

    func fetchCurrentUser() async throws -> User {
        guard let user = users.values.first else {
            throw RepositoryError.notFound
        }
        return user
    }

    func fetchUser(id: String) async throws -> User {
        guard let user = users[id] else {
            throw RepositoryError.notFound
        }
        return user
    }
}


ViewModel 测试示例
使用 DI 进行 ViewModel 的完整测试：


@MainActor
final class ProductListViewModelTests: XCTestCase {
    private var mockService: MockProductService!
    private var spyAnalytics: SpyAnalytics!
    private var sut: ProductListViewModel!

    override func setUp() {
        super.setUp()
        mockService = MockProductService()
        spyAnalytics = SpyAnalytics()
        sut = ProductListViewModel(
            productService: mockService,
            analytics: spyAnalytics
        )
    }

    func testLoadProductsSuccess() async {
        let expectedProducts = [
            Product(id: "1", name: "iPhone", price: 999),
            Product(id: "2", name: "iPad", price: 799)
        ]
        mockService.fetchProductsResult = .success(expectedProducts)

        await sut.loadProducts(category: "electronics")

        XCTAssertEqual(sut.products.count, 2)
        XCTAssertEqual(sut.products.first?.name, "iPhone")
        XCTAssertFalse(sut.isLoading)
        XCTAssertNil(sut.error)
    }

    func testLoadProductsFailure() async {
        mockService.fetchProductsResult = .failure(URLError(.notConnectedToInternet))

        await sut.loadProducts(category: "electronics")

        XCTAssertTrue(sut.products.isEmpty)
        XCTAssertNotNil(sut.error)
    }

    func testLoadProductsTracksAnalytics() async {
        mockService.fetchProductsResult = .success([])

        await sut.loadProducts(category: "electronics")

        XCTAssertEqual(spyAnalytics.trackedEvents.count, 1)
        XCTAssertEqual(spyAnalytics.trackedEvents.first?.event, "products_loaded")
    }

    func testAddToFavorites() async throws {
        let product = Product(id: "123", name: "MacBook", price: 1999)

        try await sut.addToFavorites(product: product)

        XCTAssertTrue(mockService.addToFavoritesCalled)
        XCTAssertEqual(mockService.addToFavoritesProductId, "123")
    }
}


结合 DI 的快照测试
快照测试中，通过 DI 控制 ViewModel 的状态，确保 UI 渲染的一致性：


import XCTest
import SnapshotTesting
import SwiftUI

final class ProductListSnapshotTests: XCTestCase {
    func testProductListLoaded() {
        let mockService = MockProductService()
        mockService.fetchProductsResult = .success([
            Product(id: "1", name: "iPhone 16", price: 999),
            Product(id: "2", name: "MacBook Pro", price: 2499),
            Product(id: "3", name: "AirPods Pro", price: 249),
        ])

        let viewModel = ProductListViewModel(
            productService: mockService,
            analytics: DummyAnalytics()
        )

        let view = ProductListView(viewModel: viewModel)
        let hostingController = UIHostingController(rootView: view)

        assertSnapshot(of: hostingController, as: .image(on: .iPhone13))
    }

    func testProductListEmpty() {
        let mockService = MockProductService()
        mockService.fetchProductsResult = .success([])

        let viewModel = ProductListViewModel(
            productService: mockService,
            analytics: DummyAnalytics()
        )

        let view = ProductListView(viewModel: viewModel)
        let hostingController = UIHostingController(rootView: view)

        assertSnapshot(of: hostingController, as: .image(on: .iPhone13))
    }

    func testProductListError() {
        let mockService = MockProductService()
        mockService.fetchProductsResult = .failure(URLError(.notConnectedToInternet))

        let viewModel = ProductListViewModel(
            productService: mockService,
            analytics: DummyAnalytics()
        )
        viewModel.error = "网络连接失败"

        let view = ProductListView(viewModel: viewModel)
        let hostingController = UIHostingController(rootView: view)

        assertSnapshot(of: hostingController, as: .image(on: .iPhone13))
    }
}

提示： 遵循测试金字塔原则——大量单元测试 + 适量集成测试 + 少量 UI 测试。DI 使得单元测试编写成本极低。






小结
    

  • DI 让每一层都可以独立测试，通过 Mock 替换真实依赖实现隔离
    • 

  • Test Double 分为 Dummy、Stub、Spy、Mock、Fake 五类，按需选择
    • 

  • 基于协议的 Mock 创建模式简洁、类型安全、易于维护
    • 

  • ViewModel 测试通过构造器注入 Mock 即可完成，无需 UI 环境
    • 

  • 快照测试通过 DI 控制 ViewModel 状态，确保 UI 在各种状态下的渲染正确性
    • 

  • 遵循测试金字塔原则，以单元测试为主，充分利用 DI 带来的可测试性
