容器模式

DI 容器（Dependency Injection Container）是管理对象创建和生命周期的核心组件。本文介绍如何从零实现一个类型安全的 DI 容器，涵盖作用域管理、惰性解析和线程安全。

基本容器实现

一个最简 DI 容器只需要注册和解析两个操作：

final class DIContainer: @unchecked Sendable {
    private var factories: [String: Any] = [:]
    private let lock = NSRecursiveLock()

    func register<T>(_ type: T.Type, factory: @escaping (DIContainer) -> T) {
        lock.lock()
        defer { lock.unlock() }
        let key = String(describing: type)
        factories[key] = factory
    }

    func resolve<T>(_ type: T.Type) -> T {
        lock.lock()
        defer { lock.unlock() }
        let key = String(describing: type)
        guard let factory = factories[key] as? (DIContainer) -> T else {
            fatalError("未注册类型: \(key)")
        }
        return factory(self)
    }
}

使用示例：

let container = DIContainer()

container.register(HTTPClientProtocol.self) { _ in
    URLSessionHTTPClient()
}

container.register(UserRepositoryProtocol.self) { container in
    UserRepository(httpClient: container.resolve(HTTPClientProtocol.self))
}

let repo = container.resolve(UserRepositoryProtocol.self)

作用域管理

不同的依赖需要不同的生命周期管理策略：

enum Scope {
    case transient
    case singleton
    case scoped(String)
}

final class ScopedDIContainer: @unchecked Sendable {
    private var factories: [String: (Scope, (ScopedDIContainer) -> Any)] = [:]
    private var singletons: [String: Any] = [:]
    private var scopedInstances: [String: [String: Any]] = [:]
    private let lock = NSRecursiveLock()

    func register<T>(_ type: T.Type, scope: Scope = .transient, factory: @escaping (ScopedDIContainer) -> T) {
        lock.lock()
        defer { lock.unlock() }
        let key = String(describing: type)
        factories[key] = (scope, factory)
    }

    func resolve<T>(_ type: T.Type) -> T {
        lock.lock()
        defer { lock.unlock() }
        let key = String(describing: type)

        guard let (scope, factory) = factories[key] else {
            fatalError("未注册类型: \(key)")
        }

        switch scope {
        case .transient:
            return factory(self) as! T

        case .singleton:
            if let existing = singletons[key] as? T {
                return existing
            }
            let instance = factory(self) as! T
            singletons[key] = instance
            return instance

        case .scoped(let scopeName):
            if let existing = scopedInstances[scopeName]?[key] as? T {
                return existing
            }
            let instance = factory(self) as! T
            scopedInstances[scopeName, default: [:]][key] = instance
            return instance
        }
    }

    func releaseScope(_ name: String) {
        lock.lock()
        defer { lock.unlock() }
        scopedInstances.removeValue(forKey: name)
    }
}

惰性解析
某些重量级依赖应延迟到首次使用时才创建：


final class LazyDIContainer: @unchecked Sendable {
    private var factories: [String: (ScopedDIContainer) -> Any] = [:]
    private let lock = NSRecursiveLock()

    func registerLazy<T>(_ type: T.Type, factory: @escaping (ScopedDIContainer) -> T) {
        lock.lock()
        defer { lock.unlock() }
        let key = String(describing: type)
        factories[key] = factory
    }

    func makeLazy<T>(_ type: T.Type, container: ScopedDIContainer) -> LazyDependency<T> {
        LazyDependency { [weak container] in
            guard let container else { fatalError("容器已释放") }
            return container.resolve(type)
        }
    }
}

final class LazyDependency<T>: @unchecked Sendable {
    private let factory: () -> T
    private var instance: T?
    private let lock = NSLock()

    init(_ factory: @escaping () -> T) {
        self.factory = factory
    }

    var value: T {
        lock.lock()
        defer { lock.unlock() }
        if let instance { return instance }
        let newInstance = factory()
        instance = newInstance
        return newInstance
    }
}
使用惰性解析：


final class AppCoordinator {
    private let lazyAnalytics: LazyDependency<AnalyticsServiceProtocol>

    init(lazyAnalytics: LazyDependency<AnalyticsServiceProtocol>) {
        self.lazyAnalytics = lazyAnalytics
    }

    func userDidCompleteOnboarding() {
        lazyAnalytics.value.track(event: "onboarding_complete", properties: [:])
    }
}


线程安全
DI 容器在多线程环境下必须保证线程安全：


actor ActorDIContainer {
    private var factories: [String: @Sendable (ActorDIContainer) -> Any] = [:]
    private var singletons: [String: Any] = [:]

    func register<T>(_ type: T.Type, factory: @escaping @Sendable (ActorDIContainer) -> T) {
        let key = String(describing: type)
        factories[key] = factory
    }

    func resolve<T>(_ type: T.Type) -> T {
        let key = String(describing: type)

        if let existing = singletons[key] as? T {
            return existing
        }

        guard let factory = factories[key] as? @Sendable (ActorDIContainer) -> T else {
            fatalError("未注册类型: \(key)")
        }

        let instance = factory(self)
        singletons[key] = instance
        return instance
    }
}

子容器
大型应用中可通过子容器实现模块级别的依赖隔离：


extension ScopedDIContainer {
    func createChild() -> ScopedDIContainer {
        let child = ScopedDIContainer()
        child.parent = self
        return child
    }

    private(set) weak var parent: ScopedDIContainer?

    func resolveWithFallback<T>(_ type: T.Type) -> T {
        let key = String(describing: type)
        if factories[key] != nil {
            return resolve(type)
        }
        guard let parent else {
            fatalError("未注册类型: \(key)")
        }
        return parent.resolveWithFallback(type)
    }
}


小结
    

  • DI 容器通过注册/解析机制管理对象的创建和依赖关系
    • 

  • Transient、Singleton、Scoped 三种作用域覆盖大多数生命周期需求
    • 

  • 惰性解析延迟重量级依赖的创建，优化启动性能
    • 

  • 线程安全是 DI 容器的必备特性，推荐使用 NSRecursiveLock 或 Actor
    • 

  • 子容器机制实现模块级依赖隔离，支持依赖覆盖
    • 

  • 在生产项目中，推荐使用成熟的 DI 框架而非手写容器
