深度链接路由

深度链接（Deep Link）使用户能够通过 URL 直接跳转到应用内的特定页面。本文介绍 URL Scheme、Universal Links、路由解析以及延迟深度链接的实现。

深度链接类型





类型
格式
需要服务端
未安装行为





URL Scheme
myapp://path
❌
无响应




Universal Links
https://domain/path
✅
打开网页




延迟深度链接
任意
✅
安装后跳转







URL Scheme 配置
在 Info.plist 中注册自定义 URL Scheme：


// Info.plist (概念等价的 Swift 表示)
// CFBundleURLTypes:
//   - CFBundleURLSchemes: ["myapp"]
//     CFBundleURLName: "com.example.myapp"
处理 URL Scheme 入口：


@main
struct MyApp: App {
    var body: some Scene {
        WindowGroup {
            ContentView()
                .onOpenURL { url in
                    DeepLinkRouter.shared.handle(url)
                }
        }
    }
}


Universal Links 配置
Universal Links 需要在服务端配置 apple-app-site-association 文件：


// 服务端 https://example.com/.well-known/apple-app-site-association
// {
//   "applinks": {
//     "apps": [],
//     "details": [
//       {
//         "appID": "TEAMID.com.example.myapp",
//         "paths": ["/product/*", "/user/*", "/order/*"]
//       }
//     ]
//   }
// }
在 Xcode 中启用 Associated Domains：


Signing & Capabilities → + Associated Domains
→ applinks:example.com


路由解析
将 URL 解析为强类型的路由枚举：


enum DeepLinkRoute: Equatable {
    case product(id: String)
    case productList(category: String)
    case userProfile(userId: String)
    case order(orderId: String)
    case settings
    case unknown
}

struct DeepLinkParser {
    func parse(_ url: URL) -> DeepLinkRoute {
        guard let components = URLComponents(url: url, resolvingAgainstBaseURL: true) else {
            return .unknown
        }

        let pathComponents = components.path
            .split(separator: "/")
            .map(String.init)

        switch pathComponents {
        case ["product", let id]:
            return .product(id: id)

        case ["products"]:
            let category = components.queryItems?
                .first(where: { $0.name == "category" })?.value ?? "all"
            return .productList(category: category)

        case ["user", let userId]:
            return .userProfile(userId: userId)

        case ["order", let orderId]:
            return .order(orderId: orderId)

        case ["settings"]:
            return .settings

        default:
            return .unknown
        }
    }
}


路由处理器
将解析后的路由分发到对应的 Coordinator：


@MainActor
final class DeepLinkRouter: ObservableObject {
    static let shared = DeepLinkRouter()

    private let parser = DeepLinkParser()
    private var pendingRoute: DeepLinkRoute?
    weak var coordinator: AppCoordinatorProtocol?

    func handle(_ url: URL) {
        let route = parser.parse(url)

        guard let coordinator else {
            pendingRoute = route
            return
        }

        navigate(to: route, using: coordinator)
    }

    func processPendingRoute() {
        guard let route = pendingRoute, let coordinator else { return }
        pendingRoute = nil
        navigate(to: route, using: coordinator)
    }

    private func navigate(to route: DeepLinkRoute, using coordinator: AppCoordinatorProtocol) {
        switch route {
        case .product(let id):
            coordinator.navigateToProduct(id: id)

        case .productList(let category):
            coordinator.navigateToProductList(category: category)

        case .userProfile(let userId):
            coordinator.navigateToUserProfile(userId: userId)

        case .order(let orderId):
            coordinator.navigateToOrder(orderId: orderId)

        case .settings:
            coordinator.navigateToSettings()

        case .unknown:
            break
        }
    }
}


延迟深度链接
延迟深度链接处理用户尚未安装应用的场景：


final class DeferredDeepLinkManager {
    private let userDefaults: UserDefaults
    private let clipboardKey = "deferred_deeplink"

    init(userDefaults: UserDefaults = .standard) {
        self.userDefaults = userDefaults
    }

    func checkForDeferredLink() async -> DeepLinkRoute? {
        if let savedURLString = userDefaults.string(forKey: clipboardKey),
           let url = URL(string: savedURLString) {
            userDefaults.removeObject(forKey: clipboardKey)
            return DeepLinkParser().parse(url)
        }

        return await checkPasteboardLink()
    }

    private func checkPasteboardLink() async -> DeepLinkRoute? {
        guard UIPasteboard.general.hasURLs,
              let url = UIPasteboard.general.url,
              isAppLink(url) else {
            return nil
        }

        UIPasteboard.general.urls = []
        return DeepLinkParser().parse(url)
    }

    private func isAppLink(_ url: URL) -> Bool {
        url.host == "example.com" || url.scheme == "myapp"
    }
}


路由测试
final class DeepLinkParserTests: XCTestCase {
    let parser = DeepLinkParser()

    func testParseProductURL() {
        let url = URL(string: "myapp://product/abc123")!
        XCTAssertEqual(parser.parse(url), .product(id: "abc123"))
    }

    func testParseUniversalLink() {
        let url = URL(string: "https://example.com/user/user456")!
        XCTAssertEqual(parser.parse(url), .userProfile(userId: "user456"))
    }

    func testParseProductListWithCategory() {
        let url = URL(string: "myapp://products?category=electronics")!
        XCTAssertEqual(parser.parse(url), .productList(category: "electronics"))
    }

    func testParseUnknownURL() {
        let url = URL(string: "myapp://unknown/path")!
        XCTAssertEqual(parser.parse(url), .unknown)
    }
}


提示： 深度链接解析逻辑应集中在一个 Parser 中，便于测试和维护。避免在多处硬编码 URL 路径匹配。






小结
    

  • 深度链接包括 URL Scheme、Universal Links 和延迟深度链接三种类型
    • 

  • URL Scheme 配置简单但不安全，Universal Links 需要服务端配合但更可靠
    • 

  • 路由解析将 URL 转换为强类型的枚举，消除字符串匹配的脆弱性
    • 

  • DeepLinkRouter 负责将解析后的路由分发给对应的 Coordinator
    • 

  • 延迟深度链接处理用户安装后首次打开时的跳转恢复
    • 

  • 路由解析逻辑是纯函数，应编写完整的单元测试覆盖各种场景
