SwiftUI 导航架构

SwiftUI 的导航 API 经历了从 NavigationView 到 NavigationStack 的重大演进。本文介绍基于 NavigationStack 的状态驱动导航、路径管理以及类型安全导航方案。

NavigationStack 核心概念

NavigationStack 引入了基于值类型的路由系统，导航状态完全由数据驱动：





概念
说明





NavigationStack
替代 NavigationView 的容器




NavigationPath
类型擦除的路由路径




navigationDestination
注册路由到视图的映射




NavigationLink(value:)
基于值的导航触发







基于路径的路由
定义强类型的路由枚举，配合 NavigationPath 管理导航栈：


enum AppRoute: Hashable {
    case productList(category: String)
    case productDetail(id: String)
    case cart
    case checkout
    case orderConfirmation(orderId: String)
    case userProfile(userId: String)
    case settings
    case settingsDetail(section: SettingsSection)
}

enum SettingsSection: String, Hashable {
    case account
    case notifications
    case privacy
    case about
}


路由管理器
集中管理导航状态的 Router：


@MainActor
@Observable
final class AppRouter {
    var path = NavigationPath()
    var presentedSheet: AppRoute?
    var presentedFullScreen: AppRoute?
    var alertItem: AlertItem?

    func push(_ route: AppRoute) {
        path.append(route)
    }

    func pop() {
        guard !path.isEmpty else { return }
        path.removeLast()
    }

    func popToRoot() {
        path = NavigationPath()
    }

    func pop(count: Int) {
        let removeCount = min(count, path.count)
        path.removeLast(removeCount)
    }

    func presentSheet(_ route: AppRoute) {
        presentedSheet = route
    }

    func presentFullScreen(_ route: AppRoute) {
        presentedFullScreen = route
    }

    func dismiss() {
        if presentedSheet != nil {
            presentedSheet = nil
        } else if presentedFullScreen != nil {
            presentedFullScreen = nil
        }
    }

    func reset() {
        path = NavigationPath()
        presentedSheet = nil
        presentedFullScreen = nil
    }
}


视图层集成
将 Router 与 NavigationStack 绑定：


struct AppRootView: View {
    @State private var router = AppRouter()

    var body: some View {
        NavigationStack(path: $router.path) {
            HomeView()
                .navigationDestination(for: AppRoute.self) { route in
                    destinationView(for: route)
                }
        }
        .sheet(item: $router.presentedSheet) { route in
            NavigationStack {
                destinationView(for: route)
            }
        }
        .fullScreenCover(item: $router.presentedFullScreen) { route in
            NavigationStack {
                destinationView(for: route)
            }
        }
        .environment(router)
    }

    @ViewBuilder
    private func destinationView(for route: AppRoute) -> some View {
        switch route {
        case .productList(let category):
            ProductListView(category: category)
        case .productDetail(let id):
            ProductDetailView(productId: id)
        case .cart:
            CartView()
        case .checkout:
            CheckoutView()
        case .orderConfirmation(let orderId):
            OrderConfirmationView(orderId: orderId)
        case .userProfile(let userId):
            UserProfileView(userId: userId)
        case .settings:
            SettingsView()
        case .settingsDetail(let section):
            SettingsDetailView(section: section)
        }
    }
}
在子视图中使用 Router：


struct ProductDetailView: View {
    let productId: String
    @Environment(AppRouter.self) private var router

    var body: some View {
        VStack {
            Text("商品详情: \(productId)")

            Button("加入购物车") {
                router.presentSheet(.cart)
            }

            Button("立即购买") {
                router.push(.checkout)
            }

            Button("查看更多商品") {
                router.pop()
            }
        }
    }
}


类型安全导航
通过泛型约束确保导航参数的类型安全：


protocol Routable: Hashable, Identifiable {}

extension AppRoute: Identifiable {
    var id: String {
        switch self {
        case .productList(let category): return "productList_\(category)"
        case .productDetail(let id): return "productDetail_\(id)"
        case .cart: return "cart"
        case .checkout: return "checkout"
        case .orderConfirmation(let id): return "orderConfirmation_\(id)"
        case .userProfile(let id): return "userProfile_\(id)"
        case .settings: return "settings"
        case .settingsDetail(let section): return "settingsDetail_\(section.rawValue)"
        }
    }
}


Tab + Navigation 组合
多 Tab 应用中每个 Tab 需要独立的导航栈：


@MainActor
@Observable
final class TabRouter {
    enum Tab: String, CaseIterable {
        case home
        case search
        case cart
        case profile
    }

    var selectedTab: Tab = .home
    var homePath = NavigationPath()
    var searchPath = NavigationPath()
    var cartPath = NavigationPath()
    var profilePath = NavigationPath()

    func path(for tab: Tab) -> Binding<NavigationPath> {
        switch tab {
        case .home: return Binding(get: { self.homePath }, set: { self.homePath = $0 })
        case .search: return Binding(get: { self.searchPath }, set: { self.searchPath = $0 })
        case .cart: return Binding(get: { self.cartPath }, set: { self.cartPath = $0 })
        case .profile: return Binding(get: { self.profilePath }, set: { self.profilePath = $0 })
        }
    }

    func popToRoot(tab: Tab) {
        switch tab {
        case .home: homePath = NavigationPath()
        case .search: searchPath = NavigationPath()
        case .cart: cartPath = NavigationPath()
        case .profile: profilePath = NavigationPath()
        }
    }
}

struct MainTabView: View {
    @State private var tabRouter = TabRouter()

    var body: some View {
        TabView(selection: $tabRouter.selectedTab) {
            NavigationStack(path: tabRouter.path(for: .home)) {
                HomeView()
                    .navigationDestination(for: AppRoute.self) { route in
                        RouteDestination(route: route)
                    }
            }
            .tag(TabRouter.Tab.home)
            .tabItem { Label("首页", systemImage: "house") }

            NavigationStack(path: tabRouter.path(for: .search)) {
                SearchView()
                    .navigationDestination(for: AppRoute.self) { route in
                        RouteDestination(route: route)
                    }
            }
            .tag(TabRouter.Tab.search)
            .tabItem { Label("搜索", systemImage: "magnifyingglass") }

            NavigationStack(path: tabRouter.path(for: .cart)) {
                CartView()
                    .navigationDestination(for: AppRoute.self) { route in
                        RouteDestination(route: route)
                    }
            }
            .tag(TabRouter.Tab.cart)
            .tabItem { Label("购物车", systemImage: "cart") }

            NavigationStack(path: tabRouter.path(for: .profile)) {
                ProfileView()
                    .navigationDestination(for: AppRoute.self) { route in
                        RouteDestination(route: route)
                    }
            }
            .tag(TabRouter.Tab.profile)
            .tabItem { Label("我的", systemImage: "person") }
        }
        .environment(tabRouter)
    }
}


提示： 每个 Tab 保持独立的 NavigationPath，切换 Tab 不会丢失导航历史。






小结
    

  • NavigationStack 取代 NavigationView，提供基于值的状态驱动导航
    • 

  • NavigationPath 是类型擦除的路由路径，支持 push/pop/popToRoot 操作
    • 

  • AppRouter 集中管理所有导航状态（push、sheet、fullScreenCover）
    • 

  • 通过 @Environment 将 Router 传递到整个视图树
    • 

  • 多 Tab 应用中每个 Tab 维护独立的 NavigationPath
    • 

  • 类型安全的路由枚举消除了字符串匹配的脆弱性
    •
