模块化应用中的导航

当应用被拆分为独立模块后，模块之间无法直接引用彼此的视图类型，跨模块导航成为核心挑战。本文介绍跨模块导航的抽象方案、Feature 注册机制以及导航协议设计。

导航抽象协议
定义通用的导航接口模块：


// NavigationInterface 模块
public enum NavigationTarget: Hashable {
    case productDetail(id: String)
    case userProfile(userId: String)
    case orderDetail(orderId: String)
    case settings
    case webView(url: URL)
}

public protocol NavigationServiceProtocol: AnyObject {
    func navigate(to target: NavigationTarget)
    func present(target: NavigationTarget)
    func pop()
    func popToRoot()
    func dismiss()
}
Feature 模块使用导航服务而不知道目标模块的实现：


// FeatureHome 模块 — 只依赖 NavigationInterface
import NavigationInterface

final class HomeViewModel: ObservableObject {
    private let navigationService: NavigationServiceProtocol

    init(navigationService: NavigationServiceProtocol) {
        self.navigationService = navigationService
    }

    func didTapProduct(id: String) {
        navigationService.navigate(to: .productDetail(id: id))
    }

    func didTapProfile(userId: String) {
        navigationService.navigate(to: .userProfile(userId: userId))
    }
}


Feature 注册机制
每个 Feature 模块向路由注册表注册自己的视图工厂：


// NavigationInterface 模块
public protocol FeatureRegistration {
    func register(in registry: FeatureRegistry)
}

@MainActor
public final class FeatureRegistry {
    public typealias ViewFactory = (NavigationTarget) -> AnyView?

    private var factories: [ViewFactory] = []

    public init() {}

    public func register(factory: @escaping ViewFactory) {
        factories.append(factory)
    }

    public func view(for target: NavigationTarget) -> AnyView? {
        for factory in factories {
            if let view = factory(target) {
                return view
            }
        }
        return nil
    }
}
Feature 模块提供注册实现：


// FeatureProduct 模块
import NavigationInterface
import SwiftUI

public struct ProductFeatureRegistration: FeatureRegistration {
    private let productService: ProductServiceProtocol

    public init(productService: ProductServiceProtocol) {
        self.productService = productService
    }

    public func register(in registry: FeatureRegistry) {
        registry.register { target in
            switch target {
            case .productDetail(let id):
                let viewModel = ProductDetailViewModel(
                    productId: id,
                    productService: productService
                )
                return AnyView(ProductDetailView(viewModel: viewModel))
            default:
                return nil
            }
        }
    }
}


App 壳工程装配
在 App 壳工程中将所有模块注册到路由表：


@MainActor
final class AppNavigationService: NavigationServiceProtocol {
    private let registry: FeatureRegistry
    private let coordinator: AppCoordinator

    init(coordinator: AppCoordinator) {
        self.registry = FeatureRegistry()
        self.coordinator = coordinator

        registerFeatures()
    }

    private func registerFeatures() {
        let productRegistration = ProductFeatureRegistration(
            productService: container.resolve(ProductServiceProtocol.self)
        )
        productRegistration.register(in: registry)

        let profileRegistration = ProfileFeatureRegistration(
            profileService: container.resolve(ProfileServiceProtocol.self)
        )
        profileRegistration.register(in: registry)

        let orderRegistration = OrderFeatureRegistration(
            orderService: container.resolve(OrderServiceProtocol.self)
        )
        orderRegistration.register(in: registry)
    }

    func navigate(to target: NavigationTarget) {
        guard let view = registry.view(for: target) else {
            assertionFailure("未注册的导航目标: \(target)")
            return
        }
        coordinator.push(view)
    }

    func present(target: NavigationTarget) {
        guard let view = registry.view(for: target) else { return }
        coordinator.present(view)
    }

    func pop() { coordinator.pop() }
    func popToRoot() { coordinator.popToRoot() }
    func dismiss() { coordinator.dismiss() }
}

类型安全的模块间通信
除了导航，模块间还需要数据通信。可以通过事件总线实现：


// NavigationInterface 模块
public enum ModuleEvent {
    case userDidLogin(userId: String)
    case userDidLogout
    case cartUpdated(itemCount: Int)
    case themeChanged
}

public protocol ModuleEventBusProtocol {
    func publish(_ event: ModuleEvent)
    func subscribe(_ handler: @escaping (ModuleEvent) -> Void) -> AnyCancellable
}

@MainActor
public final class ModuleEventBus: ModuleEventBusProtocol {
    private let subject = PassthroughSubject<ModuleEvent, Never>()

    public init() {}

    public func publish(_ event: ModuleEvent) {
        subject.send(event)
    }

    public func subscribe(_ handler: @escaping (ModuleEvent) -> Void) -> AnyCancellable {
        subject.sink(receiveValue: handler)
    }
}


提示： 导航接口模块应尽量保持轻量，只包含协议和枚举定义。将所有具体实现放在 App 壳工程中。






小结
    

  • 模块化应用中跨模块导航需要通过抽象层解耦
    • 

  • NavigationTarget 枚举统一定义所有可导航目标
    • 

  • FeatureRegistry 提供视图工厂注册机制，支持动态路由
    • 

  • Feature 模块通过 FeatureRegistration 协议注册自己的视图
    • 

  • App 壳工程负责装配所有 Feature 注册和导航服务
    • 

  • ModuleEventBus 提供模块间的事件通信能力，补充导航之外的交互需求
