Coordinator 模式

Coordinator 模式将导航逻辑从 ViewController/View 中抽离，形成独立的导航层。本文介绍 Coordinator 协议设计、父子关系管理，以及 UIKit 和 SwiftUI 下的实现方式。

为什么需要 Coordinator

在没有 Coordinator 的项目中，导航逻辑散布在各个 ViewController 中：





问题
Coordinator 如何解决





VC 间导航耦合
统一由 Coordinator 管理跳转




无法复用 VC
VC 不知道下一个目标，可在不同流程中复用




深度链接困难
Coordinator 可直接跳转到任意节点




测试导航逻辑
Coordinator 可独立于 UI 测试







Coordinator 协议
protocol Coordinator: AnyObject {
    var childCoordinators: [Coordinator] { get set }
    func start()
}

extension Coordinator {
    func addChild(_ coordinator: Coordinator) {
        childCoordinators.append(coordinator)
    }

    func removeChild(_ coordinator: Coordinator) {
        childCoordinators.removeAll { $0 === coordinator }
    }
}

UIKit Coordinator 实现
final class AppCoordinator: Coordinator {
    var childCoordinators: [Coordinator] = []
    private let window: UIWindow
    private let navigationController: UINavigationController

    init(window: UIWindow) {
        self.window = window
        self.navigationController = UINavigationController()
    }

    func start() {
        window.rootViewController = navigationController
        window.makeKeyAndVisible()

        if AuthManager.shared.isLoggedIn {
            showMainFlow()
        } else {
            showAuthFlow()
        }
    }

    private func showAuthFlow() {
        let authCoordinator = AuthCoordinator(navigationController: navigationController)
        authCoordinator.onLoginComplete = { [weak self] in
            self?.removeChild(authCoordinator)
            self?.showMainFlow()
        }
        addChild(authCoordinator)
        authCoordinator.start()
    }

    private func showMainFlow() {
        let mainCoordinator = MainTabCoordinator(window: window)
        addChild(mainCoordinator)
        mainCoordinator.start()
    }
}

final class AuthCoordinator: Coordinator {
    var childCoordinators: [Coordinator] = []
    var onLoginComplete: (() -> Void)?
    private let navigationController: UINavigationController

    init(navigationController: UINavigationController) {
        self.navigationController = navigationController
    }

    func start() {
        let loginVC = LoginViewController()
        loginVC.onLoginTapped = { [weak self] in
            self?.onLoginComplete?()
        }
        loginVC.onRegisterTapped = { [weak self] in
            self?.showRegister()
        }
        navigationController.setViewControllers([loginVC], animated: false)
    }

    private func showRegister() {
        let registerVC = RegisterViewController()
        registerVC.onRegisterComplete = { [weak self] in
            self?.onLoginComplete?()
        }
        navigationController.pushViewController(registerVC, animated: true)
    }
}


SwiftUI Coordinator 实现
SwiftUI 中 Coordinator 通过 ObservableObject 管理导航状态：


@MainActor
final class SwiftUICoordinator: ObservableObject {
    enum Route: Hashable {
        case productList
        case productDetail(id: String)
        case cart
        case checkout
    }

    @Published var path = NavigationPath()
    @Published var presentedSheet: Route?

    func push(_ route: Route) {
        path.append(route)
    }

    func pop() {
        guard !path.isEmpty else { return }
        path.removeLast()
    }

    func popToRoot() {
        path = NavigationPath()
    }

    func present(_ route: Route) {
        presentedSheet = route
    }

    func dismissSheet() {
        presentedSheet = nil
    }
}
在 View 中使用：


struct ShopCoordinatorView: View {
    @StateObject private var coordinator = SwiftUICoordinator()

    var body: some View {
        NavigationStack(path: $coordinator.path) {
            ProductListView()
                .navigationDestination(for: SwiftUICoordinator.Route.self) { route in
                    switch route {
                    case .productList:
                        ProductListView()
                    case .productDetail(let id):
                        ProductDetailView(productId: id)
                    case .cart:
                        CartView()
                    case .checkout:
                        CheckoutView()
                    }
                }
        }
        .sheet(item: $coordinator.presentedSheet) { route in
            sheetContent(for: route)
        }
        .environmentObject(coordinator)
    }

    @ViewBuilder
    private func sheetContent(for route: SwiftUICoordinator.Route) -> some View {
        switch route {
        case .cart:
            CartView()
        default:
            EmptyView()
        }
    }
}

小结
    

  • Coordinator 模式将导航逻辑从 View/ViewController 中抽离，形成独立导航层
    • 

  • 父子 Coordinator 形成树状结构，每个流程有独立的 Coordinator 管理
    • 

  • UIKit Coordinator 通过 UINavigationController 和闭包回调实现
    • 

  • SwiftUI Coordinator 通过 ObservableObject + NavigationPath 实现状态驱动导航
    • 

  • Coordinator 使得深度链接、A/B 测试路由和导航逻辑测试变得简单
    • 

  • 在模块化项目中，每个 Feature 模块可以有自己的 Coordinator
    •
