集成测试

集成测试验证多个组件协同工作的正确性，是单元测试和 E2E 测试之间的关键桥梁。本文介绍 XCTest 异步期望、网络存根、数据库测试以及 UI 集成测试。

XCTest 异步期望
集成测试中经常需要等待异步操作完成：


final class ArticleFlowIntegrationTests: XCTestCase {
    private var repository: FakeArticleRepository!
    private var viewModel: ArticleListViewModel!

    override func setUp() {
        super.setUp()
        repository = FakeArticleRepository()
        viewModel = ArticleListViewModel(
            fetchArticlesUseCase: FetchArticlesUseCase(articleRepository: repository)
        )
    }

    @MainActor
    func testLoadAndRefreshArticles() async {
        repository.articles = [
            Article(id: "1", title: "第一篇", content: "内容1", author: .mock, publishedAt: Date(), tags: ["swift"]),
            Article(id: "2", title: "第二篇", content: "内容2", author: .mock, publishedAt: Date(), tags: ["ios"])
        ]

        await viewModel.loadNextPage()

        XCTAssertEqual(viewModel.articles.count, 2)
        XCTAssertEqual(viewModel.articles.first?.title, "第一篇")
        XCTAssertFalse(viewModel.isLoading)
    }

    @MainActor
    func testLoadArticlesWithNetworkError() async {
        repository.shouldFail = true

        await viewModel.loadNextPage()

        XCTAssertTrue(viewModel.articles.isEmpty)
        XCTAssertNotNil(viewModel.error)
    }
}
传统回调式异步测试：


func testNotificationReceived() {
    let expectation = XCTestExpectation(description: "收到通知")

    let observer = NotificationCenter.default.addObserver(
        forName: .articleSaved,
        object: nil,
        queue: .main
    ) { notification in
        XCTAssertNotNil(notification.userInfo?["articleId"])
        expectation.fulfill()
    }

    repository.save(Article.mock)

    wait(for: [expectation], timeout: 5.0)
    NotificationCenter.default.removeObserver(observer)
}


网络存根（Network Stubbing）
使用 URLProtocol 子类拦截网络请求：


final class MockURLProtocol: URLProtocol {
    static var stubResponses: [String: (Data, HTTPURLResponse)] = [:]

    override class func canInit(with request: URLRequest) -> Bool {
        true
    }

    override class func canonicalRequest(for request: URLRequest) -> URLRequest {
        request
    }

    override func startLoading() {
        guard let url = request.url?.absoluteString,
              let (data, response) = Self.stubResponses[url] else {
            client?.urlProtocol(self, didFailWithError: URLError(.fileDoesNotExist))
            return
        }

        client?.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
        client?.urlProtocol(self, didLoad: data)
        client?.urlProtocolDidFinishLoading(self)
    }

    override func stopLoading() {}
}

extension URLSession {
    static var stubbed: URLSession {
        let config = URLSessionConfiguration.ephemeral
        config.protocolClasses = [MockURLProtocol.self]
        return URLSession(configuration: config)
    }
}
在集成测试中使用：


final class APIClientIntegrationTests: XCTestCase {
    var apiClient: APIClient!

    override func setUp() {
        super.setUp()
        apiClient = APIClient(session: .stubbed)
    }

    override func tearDown() {
        MockURLProtocol.stubResponses.removeAll()
        super.tearDown()
    }

    func testFetchProducts() async throws {
        let mockData = """
        [
            {"id": "1", "name": "iPhone", "price": 999},
            {"id": "2", "name": "iPad", "price": 799}
        ]
        """.data(using: .utf8)!

        let response = HTTPURLResponse(
            url: URL(string: "https://api.example.com/products")!,
            statusCode: 200,
            httpVersion: nil,
            headerFields: ["Content-Type": "application/json"]
        )!

        MockURLProtocol.stubResponses["https://api.example.com/products"] = (mockData, response)

        let products: [Product] = try await apiClient.get("/products")

        XCTAssertEqual(products.count, 2)
        XCTAssertEqual(products.first?.name, "iPhone")
    }
}


数据库集成测试
使用内存数据库进行 Core Data / SwiftData 集成测试：


final class ArticleStorageIntegrationTests: XCTestCase {
    var storage: ArticleLocalStorage!
    var container: ModelContainer!

    override func setUp() {
        super.setUp()
        let config = ModelConfiguration(isStoredInMemoryOnly: true)
        container = try! ModelContainer(
            for: ArticleModel.self,
            configurations: config
        )
        storage = ArticleLocalStorage(modelContainer: container)
    }

    func testSaveAndFetchArticle() async throws {
        let article = Article(
            id: "test-1",
            title: "测试文章",
            content: "这是测试内容",
            author: Author(id: "a1", name: "测试作者", avatarURL: nil),
            publishedAt: Date(),
            tags: ["test"]
        )

        try await storage.save(article)
        let fetched = try await storage.fetch(id: "test-1")

        XCTAssertEqual(fetched?.title, "测试文章")
        XCTAssertEqual(fetched?.author.name, "测试作者")
    }

    func testDeleteArticle() async throws {
        let article = Article.mock
        try await storage.save(article)
        try await storage.delete(id: article.id)

        let fetched = try await storage.fetch(id: article.id)
        XCTAssertNil(fetched)
    }

    func testFetchAllArticlesSorted() async throws {
        let articles = (1...5).map { i in
            Article(
                id: "id-\(i)",
                title: "文章 \(i)",
                content: "内容",
                author: .mock,
                publishedAt: Date().addingTimeInterval(TimeInterval(i * 3600)),
                tags: []
            )
        }

        for article in articles {
            try await storage.save(article)
        }

        let result = try await storage.fetchAll(sortedBy: .publishedAt, ascending: false)
        XCTAssertEqual(result.count, 5)
        XCTAssertEqual(result.first?.title, "文章 5")
    }
}


UI 集成测试
验证 ViewModel 和 View 的集成行为：


import XCTest
import ViewInspector
import SwiftUI

final class ProductListUIIntegrationTests: XCTestCase {
    @MainActor
    func testProductListDisplaysProducts() async throws {
        let mockService = MockProductService()
        mockService.fetchProductsResult = .success([
            Product(id: "1", name: "iPhone", price: 999),
            Product(id: "2", name: "iPad", price: 799)
        ])

        let viewModel = ProductListViewModel(
            productService: mockService,
            analytics: DummyAnalytics()
        )

        await viewModel.loadProducts(category: "all")

        XCTAssertEqual(viewModel.products.count, 2)
        XCTAssertFalse(viewModel.isLoading)
    }
}


提示： 集成测试不应该覆盖所有细节场景（那是单元测试的工作），而应该聚焦于组件间的协作是否正确。

小结
    

  • 集成测试验证多个组件协同工作的正确性，填补单元测试和 E2E 测试之间的空白
    • 

  • XCTest 的 async/await 和 XCTestExpectation 支持异步集成测试
    • 

  • URLProtocol 子类化是拦截网络请求的标准方式，无需修改生产代码
    • 

  • 内存数据库（isStoredInMemoryOnly）使数据库测试快速且隔离
    • 

  • 集成测试应聚焦组件间的协作，而非重复单元测试已覆盖的逻辑
    • 

  • 合理的集成测试策略按垂直切片组织，每层使用适当的真实/Mock 组件
