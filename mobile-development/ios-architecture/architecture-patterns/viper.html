VIPER 架构

VIPER 是 iOS 社区中一种高度模块化的架构模式，将每个屏幕拆分为五个独立组件：View、Interactor、Presenter、Entity、Router。本文详细解析各组件的职责、模块生成方式以及 VIPER 的优缺点。

VIPER 组件概览





组件
职责
持有关系





View
UI 展示与用户交互
持有 Presenter




Interactor
业务逻辑与数据获取
被 Presenter 持有




Presenter
协调 View 与 Interactor
持有 Interactor 和 Router




Entity
数据模型
被 Interactor 使用




Router
页面跳转与模块装配
被 Presenter 持有







协议定义
VIPER 通过协议明确每个组件的接口边界：


protocol UserListViewProtocol: AnyObject {
    var presenter: UserListPresenterProtocol? { get set }
    func showUsers(_ users: [UserEntity])
    func showError(_ message: String)
    func showLoading()
    func hideLoading()
}

protocol UserListPresenterProtocol: AnyObject {
    var view: UserListViewProtocol? { get set }
    var interactor: UserListInteractorInputProtocol? { get set }
    var router: UserListRouterProtocol? { get set }
    func viewDidLoad()
    func didSelectUser(_ user: UserEntity)
}

protocol UserListInteractorInputProtocol: AnyObject {
    var presenter: UserListInteractorOutputProtocol? { get set }
    func fetchUsers()
}

protocol UserListInteractorOutputProtocol: AnyObject {
    func didFetchUsers(_ users: [UserEntity])
    func didFailFetchingUsers(_ error: Error)
}

protocol UserListRouterProtocol: AnyObject {
    static func createModule() -> UIViewController
    func navigateToUserDetail(from view: UserListViewProtocol, with user: UserEntity)
}


各组件实现
Entity
struct UserEntity {
    let id: Int
    let name: String
    let email: String
    let avatarURL: URL?
}
Interactor
final class UserListInteractor: UserListInteractorInputProtocol {
    weak var presenter: UserListInteractorOutputProtocol?
    private let apiService: APIServiceProtocol

    init(apiService: APIServiceProtocol) {
        self.apiService = apiService
    }

    func fetchUsers() {
        Task {
            do {
                let users = try await apiService.fetchUsers()
                await MainActor.run {
                    presenter?.didFetchUsers(users)
                }
            } catch {
                await MainActor.run {
                    presenter?.didFailFetchingUsers(error)
                }
            }
        }
    }
}
Presenter
final class UserListPresenter: UserListPresenterProtocol {
    weak var view: UserListViewProtocol?
    var interactor: UserListInteractorInputProtocol?
    var router: UserListRouterProtocol?

    func viewDidLoad() {
        view?.showLoading()
        interactor?.fetchUsers()
    }

    func didSelectUser(_ user: UserEntity) {
        guard let view = view else { return }
        router?.navigateToUserDetail(from: view, with: user)
    }
}

extension UserListPresenter: UserListInteractorOutputProtocol {
    func didFetchUsers(_ users: [UserEntity]) {
        view?.hideLoading()
        view?.showUsers(users)
    }

    func didFailFetchingUsers(_ error: Error) {
        view?.hideLoading()
        view?.showError(error.localizedDescription)
    }
}
Router（模块装配）
final class UserListRouter: UserListRouterProtocol {
    static func createModule() -> UIViewController {
        let view = UserListViewController()
        let presenter = UserListPresenter()
        let interactor = UserListInteractor(apiService: APIService())
        let router = UserListRouter()

        view.presenter = presenter
        presenter.view = view
        presenter.interactor = interactor
        presenter.router = router
        interactor.presenter = presenter

        return view
    }

    func navigateToUserDetail(from view: UserListViewProtocol, with user: UserEntity) {
        let detailVC = UserDetailRouter.createModule(with: user)
        guard let sourceVC = view as? UIViewController else { return }
        sourceVC.navigationController?.pushViewController(detailVC, animated: true)
    }
}

小结
    

  • VIPER 将屏幕拆分为 View/Interactor/Presenter/Entity/Router 五个组件
    • 

  • 通过协议定义组件间接口，实现彻底的关注点分离
    • 

  • Router 承担模块装配和页面导航职责，是 VIPER 模块的入口
    • 

  • 代码生成工具可以大幅降低 VIPER 的样板代码负担
    • 

  • VIPER 最适合 UIKit 大型团队项目，对 SwiftUI 项目不推荐
    • 

  • 选择架构时需权衡项目规模、团队经验和技术栈适配度
