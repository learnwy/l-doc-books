平台通道（Platform Channels）

核心概念

Flutter 通过平台通道（Platform Channels）实现 Dart 代码与原生平台（iOS/Android）之间的双向通信。这是 Flutter 访问平台特有能力（如传感器、相机、系统设置）的唯一桥梁。Flutter 提供三种通道类型，搭配标准编解码器处理跨语言的数据序列化，并通过 Pigeon 工具实现类型安全的代码生成。

MethodChannel
MethodChannel 是最常用的通道类型，采用方法名 + 参数的 RPC 风格调用，支持 Dart 调用原生和原生调用 Dart 的双向通信。


Dart 端封装
class BatteryService {
  static const _channel = MethodChannel('com.example.app/battery');

  Future<int> getBatteryLevel() async {
    final level = await _channel.invokeMethod<int>('getBatteryLevel');
    if (level == null) throw PlatformException(code: 'NULL_RESULT');
    return level;
  }

  Future<BatteryState> getBatteryState() async {
    final stateMap = await _channel.invokeMapMethod<String, dynamic>(
      'getBatteryState',
    );
    if (stateMap == null) throw PlatformException(code: 'NULL_RESULT');

    return BatteryState(
      level: stateMap['level'] as int,
      isCharging: stateMap['isCharging'] as bool,
      temperature: (stateMap['temperature'] as num).toDouble(),
    );
  }

  Future<bool> enableBatterySaver({required bool enable}) async {
    final result = await _channel.invokeMethod<bool>(
      'setBatterySaver',
      {'enable': enable},
    );
    return result ?? false;
  }
}

class BatteryState {
  final int level;
  final bool isCharging;
  final double temperature;

  BatteryState({
    required this.level,
    required this.isCharging,
    required this.temperature,
  });
}
Android 端实现（Kotlin）
class MainActivity : FlutterActivity() {
    private val BATTERY_CHANNEL = "com.example.app/battery"

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)

        MethodChannel(
            flutterEngine.dartExecutor.binaryMessenger,
            BATTERY_CHANNEL
        ).setMethodCallHandler { call, result ->
            when (call.method) {
                "getBatteryLevel" -> {
                    val level = getBatteryLevel()
                    if (level != -1) {
                        result.success(level)
                    } else {
                        result.error("UNAVAILABLE", "Battery level not available", null)
                    }
                }
                "getBatteryState" -> {
                    result.success(mapOf(
                        "level" to getBatteryLevel(),
                        "isCharging" to isCharging(),
                        "temperature" to getBatteryTemperature()
                    ))
                }
                "setBatterySaver" -> {
                    val enable = call.argument<Boolean>("enable") ?: false
                    result.success(setBatterySaver(enable))
                }
                else -> result.notImplemented()
            }
        }
    }

    private fun getBatteryLevel(): Int {
        val manager = getSystemService(BATTERY_SERVICE) as BatteryManager
        return manager.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY)
    }
}
iOS 端实现（Swift）
@UIApplicationMain
@objc class AppDelegate: FlutterAppDelegate {
    override func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?
    ) -> Bool {
        let controller = window?.rootViewController as! FlutterViewController
        let batteryChannel = FlutterMethodChannel(
            name: "com.example.app/battery",
            binaryMessenger: controller.binaryMessenger
        )

        batteryChannel.setMethodCallHandler { [weak self] call, result in
            switch call.method {
            case "getBatteryLevel":
                UIDevice.current.isBatteryMonitoringEnabled = true
                let level = Int(UIDevice.current.batteryLevel * 100)
                result(level)
            case "getBatteryState":
                UIDevice.current.isBatteryMonitoringEnabled = true
                result([
                    "level": Int(UIDevice.current.batteryLevel * 100),
                    "isCharging": UIDevice.current.batteryState == .charging,
                    "temperature": 25.0
                ])
            case "setBatterySaver":
                result(FlutterError(
                    code: "UNSUPPORTED",
                    message: "iOS does not support programmatic battery saver",
                    details: nil
                ))
            default:
                result(FlutterMethodNotImplemented)
            }
        }

        GeneratedPluginRegistrant.register(with: self)
        return super.application(application, didFinishLaunchingWithOptions: launchOptions)
    }
}

class NativeCallbackHandler {
  static const _channel = MethodChannel('com.example.app/callbacks');

  static void init() {
    _channel.setMethodCallHandler((call) async {
      return switch (call.method) {
        'onDeepLink' => _handleDeepLink(call.arguments as String),
        'onPushNotification' => _handlePush(call.arguments as Map),
        _ => throw MissingPluginException(),
      };
    });
  }

  static Future<String> _handleDeepLink(String uri) async {
    return 'handled';
  }

  static Future<void> _handlePush(Map args) async {
    final title = args['title'] as String;
    final body = args['body'] as String;
  }
}


EventChannel
EventChannel 用于从原生平台向 Dart 持续推送事件流，Dart 端接收为 Stream。适用于传感器数据、定位更新、蓝牙设备扫描等持续性数据源。


Dart 端
class SensorService {
  static const _accelerometerChannel =
      EventChannel('com.example.app/accelerometer');
  static const _locationChannel =
      EventChannel('com.example.app/location');

  Stream<AccelerometerEvent> get accelerometerEvents {
    return _accelerometerChannel.receiveBroadcastStream().map((event) {
      final data = event as Map;
      return AccelerometerEvent(
        x: (data['x'] as num).toDouble(),
        y: (data['y'] as num).toDouble(),
        z: (data['z'] as num).toDouble(),
        timestamp: DateTime.fromMillisecondsSinceEpoch(data['timestamp'] as int),
      );
    });
  }

  Stream<LocationUpdate> get locationUpdates {
    return _locationChannel
        .receiveBroadcastStream({'minDistance': 10.0, 'interval': 5000})
        .map((event) {
      final data = event as Map;
      return LocationUpdate(
        latitude: (data['lat'] as num).toDouble(),
        longitude: (data['lng'] as num).toDouble(),
        accuracy: (data['accuracy'] as num).toDouble(),
      );
    });
  }
}

class AccelerometerEvent {
  final double x, y, z;
  final DateTime timestamp;
  AccelerometerEvent({
    required this.x,
    required this.y,
    required this.z,
    required this.timestamp,
  });
}

class LocationUpdate {
  final double latitude, longitude, accuracy;
  LocationUpdate({
    required this.latitude,
    required this.longitude,
    required this.accuracy,
  });
}
Android 端 EventChannel（Kotlin）
class AccelerometerStreamHandler(private val context: Context) :
    EventChannel.StreamHandler {

    private var sensorManager: SensorManager? = null
    private var listener: SensorEventListener? = null

    override fun onListen(arguments: Any?, events: EventChannel.EventSink?) {
        sensorManager = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager
        val accelerometer = sensorManager?.getDefaultSensor(Sensor.TYPE_ACCELEROMETER)

        listener = object : SensorEventListener {
            override fun onSensorChanged(event: SensorEvent?) {
                event?.let {
                    events?.success(mapOf(
                        "x" to it.values[0].toDouble(),
                        "y" to it.values[1].toDouble(),
                        "z" to it.values[2].toDouble(),
                        "timestamp" to System.currentTimeMillis()
                    ))
                }
            }

            override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {}
        }

        sensorManager?.registerListener(
            listener, accelerometer, SensorManager.SENSOR_DELAY_UI
        )
    }

    override fun onCancel(arguments: Any?) {
        sensorManager?.unregisterListener(listener)
        listener = null
    }
}

BasicMessageChannel
BasicMessageChannel 提供最底层的消息传递能力，支持自定义编解码器，适合传输非结构化或大块数据。


class ClipboardBridge {
  static final _channel = BasicMessageChannel<String>(
    'com.example.app/clipboard',
    StringCodec(),
  );

  static void init() {
    _channel.setMessageHandler((message) async {
      if (message == null) return null;
      return 'received: $message';
    });
  }

  static Future<String?> getClipboard() async {
    return _channel.send('get');
  }
}

class ImageBridge {
  static const _channel = BasicMessageChannel<ByteData>(
    'com.example.app/image',
    BinaryCodec(),
  );

  static Future<Uint8List?> getProcessedImage(Uint8List rawImage) async {
    final input = ByteData.sublistView(rawImage);
    final result = await _channel.send(input);
    if (result == null) return null;
    return result.buffer.asUint8List();
  }
}

Pigeon 类型安全代码生成
Pigeon 是 Flutter 官方提供的代码生成工具，通过定义 Dart 接口自动生成 Dart、Kotlin、Swift 三端的类型安全通信代码，彻底消除手写通道名称和类型转换的错误风险。


接口定义
import 'package:pigeon/pigeon.dart';

class DeviceInfo {
  String? model;
  String? osVersion;
  int? batteryLevel;
  bool? isCharging;
  double? storageUsedGB;
  double? storageTotalGB;
}

class AppInfo {
  String? packageName;
  String? version;
  String? buildNumber;
}

@ConfigurePigeon(PigeonOptions(
  dartOut: 'lib/src/platform/api.g.dart',
  kotlinOut: 'android/app/src/main/kotlin/com/example/app/Api.kt',
  kotlinOptions: KotlinOptions(package: 'com.example.app'),
  swiftOut: 'ios/Runner/Api.swift',
))
@HostApi()
abstract class DeviceApi {
  DeviceInfo getDeviceInfo();
  AppInfo getAppInfo();
  bool openSystemSettings(String section);
  @async
  String getUniqueId();
}

@FlutterApi()
abstract class AppLifecycleApi {
  void onLowMemory();
  void onTrimMemory(int level);
}

Kotlin 实现
class DeviceApiImpl(private val context: Context) : DeviceApi {
    override fun getDeviceInfo(): DeviceInfo {
        val batteryManager = context.getSystemService(Context.BATTERY_SERVICE) as BatteryManager
        return DeviceInfo(
            model = Build.MODEL,
            osVersion = "Android ${Build.VERSION.RELEASE}",
            batteryLevel = batteryManager.getIntProperty(
                BatteryManager.BATTERY_PROPERTY_CAPACITY
            ).toLong(),
            isCharging = batteryManager.isCharging,
            storageUsedGB = getUsedStorage(),
            storageTotalGB = getTotalStorage()
        )
    }

    override fun getAppInfo(): AppInfo {
        val pInfo = context.packageManager.getPackageInfo(context.packageName, 0)
        return AppInfo(
            packageName = context.packageName,
            version = pInfo.versionName,
            buildNumber = pInfo.longVersionCode.toString()
        )
    }

    override fun openSystemSettings(section: String): Boolean {
        return try {
            val intent = Intent(Settings.ACTION_SETTINGS)
            context.startActivity(intent)
            true
        } catch (e: Exception) {
            false
        }
    }

    override fun getUniqueId(callback: (Result<String>) -> Unit) {
        callback(Result.success(Settings.Secure.getString(
            context.contentResolver, Settings.Secure.ANDROID_ID
        )))
    }
}

class PlatformService {
  static const _channel = MethodChannel('com.example.app/platform');

  Future<String> getPlatformInfo() async {
    try {
      final info = await _channel.invokeMethod<String>('getPlatformInfo');
      return info ?? 'Unknown';
    } on PlatformException catch (e) {
      return 'Error: ${e.message}';
    } on MissingPluginException {
      return 'Plugin not available on this platform';
    }
  }

  Future<T> invokeWithFallback<T>({
    required String method,
    required T fallback,
    Map<String, dynamic>? arguments,
  }) async {
    try {
      final result = await _channel.invokeMethod<T>(method, arguments);
      return result ?? fallback;
    } on PlatformException {
      return fallback;
    } on MissingPluginException {
      return fallback;
    }
  }
}

小结
平台通道是 Flutter 与原生世界沟通的桥梁，三种通道类型覆盖了从单次调用到持续事件流的全部场景。


核心要点回顾：


    

  • MethodChannel 是最常用的通道，适合方法调用和一次性数据获取
    • 

  • EventChannel 将原生持续事件映射为 Dart Stream，适合传感器和定位场景
    • 

  • BasicMessageChannel 提供最底层消息传递，搭配 BinaryCodec 实现高性能数据传输
    • 

  • StandardMessageCodec 支持 Dart 与原生之间的标准类型映射，是默认编解码器
    • 

  • Pigeon 通过单一接口定义自动生成三端代码，消除手写通道的类型安全隐患
    • 

  • 插件开发应始终处理 PlatformException 和 MissingPluginException，提供降级方案
    •
