Flutter 状态管理

核心概念

状态管理是 Flutter 应用开发中最核心的架构决策之一。Flutter 生态提供了从轻量的 setState 到成熟的 Bloc、Riverpod 等多种方案，每种方案在复杂度、可测试性、学习曲线和适用规模上各有取舍。选择正确的方案取决于应用规模、团队经验和具体需求。

graph LR
    A[Flutter 状态管理] --> B[内置方案]
    A --> C[社区方案]
    B --> B1[setState]
    B --> B2[InheritedWidget]
    B --> B3[ValueNotifier]
    C --> C1[Provider]
    C --> C2[Riverpod]
    C --> C3[Bloc / Cubit]
    C --> C4[GetX]

setState：局部状态管理

setState 是 Flutter 内置的最基础状态管理方式。它直接修改 State 对象的字段并触发当前 Widget 子树重建。适用于单个 Widget 内部的 UI 状态。

class SearchBar extends StatefulWidget {
  final ValueChanged<String> onSearch;

  const SearchBar({super.key, required this.onSearch});

  @override
  State<SearchBar> createState() => _SearchBarState();
}

class _SearchBarState extends State<SearchBar> {
  final _controller = TextEditingController();
  bool _showClear = false;
  bool _isLoading = false;

  @override
  void initState() {
    super.initState();
    _controller.addListener(() {
      setState(() {
        _showClear = _controller.text.isNotEmpty;
      });
    });
  }

  Future<void> _performSearch() async {
    setState(() => _isLoading = true);
    widget.onSearch(_controller.text);
    await Future.delayed(const Duration(milliseconds: 500));
    setState(() => _isLoading = false);
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Row(
      children: [
        Expanded(
          child: TextField(
            controller: _controller,
            decoration: InputDecoration(
              hintText: '搜索...',
              suffixIcon: _showClear
                  ? IconButton(
                      icon: const Icon(Icons.clear),
                      onPressed: () => _controller.clear(),
                    )
                  : null,
            ),
            onSubmitted: (_) => _performSearch(),
          ),
        ),
        if (_isLoading)
          const Padding(
            padding: EdgeInsets.all(8),
            child: SizedBox(
              width: 20,
              height: 20,
              child: CircularProgressIndicator(strokeWidth: 2),
            ),
          ),
      ],
    );
  }
}

setState 适用边界

适合 不适合
单 Widget 的 UI 开关 跨 Widget 共享状态
表单输入的即时反馈 需要持久化的数据
动画控制器状态 复杂业务逻辑
临时加载/错误状态 需要撤销/重做的状态

Provider

Provider 是 Flutter 官方推荐的状态管理方案，基于 InheritedWidget 封装，提供依赖注入和自动生命周期管理。它是 Flutter 生态中使用最广泛的状态管理库。

ChangeNotifier + Provider

class CartModel extends ChangeNotifier {
  final List<CartItem> _items = [];

  List<CartItem> get items => List.unmodifiable(_items);
  int get itemCount => _items.length;

  double get totalPrice =>
      _items.fold(0, (sum, item) => sum + item.price * item.quantity);

  void addItem(Product product) {
    final existingIndex = _items.indexWhere((i) => i.productId == product.id);
    if (existingIndex >= 0) {
      _items[existingIndex] = _items[existingIndex].copyWith(
        quantity: _items[existingIndex].quantity + 1,
      );
    } else {
      _items.add(CartItem(
        productId: product.id,
        name: product.name,
        price: product.price,
        quantity: 1,
      ));
    }
    notifyListeners();
  }

  void removeItem(String productId) {
    _items.removeWhere((i) => i.productId == productId);
    notifyListeners();
  }

  void clear() {
    _items.clear();
    notifyListeners();
  }
}

Riverpod

Riverpod 是 Provider 作者推出的下一代状态管理方案。它解决了 Provider 的诸多限制：不依赖 BuildContext、支持编译期安全、Provider 可覆盖测试、支持自动销毁。

graph TB
    A[Riverpod Provider 类型] --> B[Provider]
    A --> C[StateProvider]
    A --> D[FutureProvider]
    A --> E[StreamProvider]
    A --> F[NotifierProvider]
    A --> G[AsyncNotifierProvider]
    B --> B1[只读计算值]
    C --> C1[简单可变状态]
    D --> D1[异步一次性数据]
    E --> E1[实时流数据]
    F --> F1[复杂同步逻辑]
    G --> G1[复杂异步逻辑]

NotifierProvider 实现

class TodoListNotifier extends Notifier<List<Todo>> {
  @override
  List<Todo> build() {
    return [];
  }

  void addTodo(String title) {
    state = [
      ...state,
      Todo(id: DateTime.now().millisecondsSinceEpoch.toString(), title: title),
    ];
  }

  void toggleTodo(String id) {
    state = [
      for (final todo in state)
        if (todo.id == id)
          todo.copyWith(completed: !todo.completed)
        else
          todo,
    ];
  }

  void removeTodo(String id) {
    state = state.where((t) => t.id != id).toList();
  }
}

final todoListProvider =
    NotifierProvider<TodoListNotifier, List<Todo>>(TodoListNotifier.new);

final completedCountProvider = Provider<int>((ref) {
  final todos = ref.watch(todoListProvider);
  return todos.where((t) => t.completed).length;
});

final filteredTodosProvider = Provider<List<Todo>>((ref) {
  final todos = ref.watch(todoListProvider);
  final filter = ref.watch(todoFilterProvider);

  return switch (filter) {
    TodoFilter.all => todos,
    TodoFilter.active => todos.where((t) => !t.completed).toList(),
    TodoFilter.completed => todos.where((t) => t.completed).toList(),
  };
});

AsyncNotifierProvider 异步数据

class UserProfileNotifier extends AsyncNotifier<UserProfile> {
  @override
  Future<UserProfile> build() async {
    final authState = ref.watch(authProvider);
    if (authState case Authenticated(userId: final id)) {
      return ref.read(apiClientProvider).fetchProfile(id);
    }
    throw UnauthorizedException();
  }

  Future<void> updateDisplayName(String name) async {
    state = const AsyncLoading();
    state = await AsyncValue.guard(() async {
      final updated = await ref.read(apiClientProvider).updateProfile(
            displayName: name,
          );
      return updated;
    });
  }
}

final userProfileProvider =
    AsyncNotifierProvider<UserProfileNotifier, UserProfile>(
  UserProfileNotifier.new,
);

Riverpod UI 集成

class TodoPage extends ConsumerWidget {
  const TodoPage({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final todos = ref.watch(filteredTodosProvider);
    final completedCount = ref.watch(completedCountProvider);

    return Scaffold(
      appBar: AppBar(title: Text('待办 (已完成: $completedCount)')),
      body: ListView.builder(
        itemCount: todos.length,
        itemBuilder: (context, index) {
          final todo = todos[index];
          return ListTile(
            leading: Checkbox(
              value: todo.completed,
              onChanged: (_) {
                ref.read(todoListProvider.notifier).toggleTodo(todo.id);
              },
            ),
            title: Text(
              todo.title,
              style: TextStyle(
                decoration:
                    todo.completed ? TextDecoration.lineThrough : null,
              ),
            ),
            trailing: IconButton(
              icon: const Icon(Icons.delete_outline),
              onPressed: () {
                ref.read(todoListProvider.notifier).removeTodo(todo.id);
              },
            ),
          );
        },
      ),
    );
  }
}

Bloc / Cubit

Bloc（Business Logic Component）采用严格的事件驱动架构，将 UI 事件与状态变更分离。Cubit 是 Bloc 的简化版本，用方法调用替代事件类。

graph LR
    subgraph Bloc 模式
        A1[UI Event] --> B1[Bloc]
        B1 --> C1[State]
        C1 --> D1[UI 重建]
    end
    subgraph Cubit 模式
        A2[方法调用] --> B2[Cubit]
        B2 --> C2[State]
        C2 --> D2[UI 重建]
    end

完整 Bloc 实现

sealed class AuthEvent {}

class LoginRequested extends AuthEvent {
  final String email;
  final String password;
  LoginRequested({required this.email, required this.password});
}

class LogoutRequested extends AuthEvent {}

class TokenRefreshRequested extends AuthEvent {}

sealed class AuthState {
  const AuthState();
}

class AuthInitial extends AuthState {
  const AuthInitial();
}

class AuthLoading extends AuthState {
  const AuthLoading();
}

class Authenticated extends AuthState {
  final User user;
  final String token;
  const Authenticated({required this.user, required this.token});
}

class AuthError extends AuthState {
  final String message;
  const AuthError(this.message);
}

class AuthBloc extends Bloc<AuthEvent, AuthState> {
  final AuthRepository _authRepo;

  AuthBloc({required AuthRepository authRepo})
      : _authRepo = authRepo,
        super(const AuthInitial()) {
    on<LoginRequested>(_onLogin);
    on<LogoutRequested>(_onLogout);
    on<TokenRefreshRequested>(_onTokenRefresh);
  }

  Future<void> _onLogin(LoginRequested event, Emitter<AuthState> emit) async {
    emit(const AuthLoading());
    try {
      final result = await _authRepo.login(event.email, event.password);
      emit(Authenticated(user: result.user, token: result.token));
    } on AuthException catch (e) {
      emit(AuthError(e.message));
    }
  }

  Future<void> _onLogout(LogoutRequested event, Emitter<AuthState> emit) async {
    await _authRepo.logout();
    emit(const AuthInitial());
  }

  Future<void> _onTokenRefresh(
    TokenRefreshRequested event,
    Emitter<AuthState> emit,
  ) async {
    if (state case Authenticated(token: final currentToken)) {
      try {
        final newToken = await _authRepo.refreshToken(currentToken);
        emit(Authenticated(
          user: (state as Authenticated).user,
          token: newToken,
        ));
      } on AuthException {
        emit(const AuthInitial());
      }
    }
  }
}

Cubit 简化版

class ThemeCubit extends Cubit<ThemeMode> {
  final SharedPreferences _prefs;

  ThemeCubit({required SharedPreferences prefs})
      : _prefs = prefs,
        super(_loadInitialTheme(prefs));

  static ThemeMode _loadInitialTheme(SharedPreferences prefs) {
    final value = prefs.getString('theme');
    return switch (value) {
      'dark' => ThemeMode.dark,
      'light' => ThemeMode.light,
      _ => ThemeMode.system,
    };
  }

  Future<void> setTheme(ThemeMode mode) async {
    await _prefs.setString('theme', mode.name);
    emit(mode);
  }

  void toggleTheme() {
    final next = state == ThemeMode.light ? ThemeMode.dark : ThemeMode.light;
    setTheme(next);
  }
}

Bloc UI 集成

class LoginPage extends StatelessWidget {
  const LoginPage({super.key});

  @override
  Widget build(BuildContext context) {
    return BlocConsumer<AuthBloc, AuthState>(
      listenWhen: (previous, current) =>
          current is Authenticated || current is AuthError,
      listener: (context, state) {
        if (state is Authenticated) {
          Navigator.of(context).pushReplacementNamed('/home');
        } else if (state is AuthError) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text(state.message)),
          );
        }
      },
      builder: (context, state) {
        return Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            if (state is AuthLoading)
              const CircularProgressIndicator()
            else
              ElevatedButton(
                onPressed: () {
                  context.read<AuthBloc>().add(
                        LoginRequested(
                          email: 'user@example.com',
                          password: 'password',
                        ),
                      );
                },
                child: const Text('登录'),
              ),
          ],
        );
      },
    );
  }
}

GetX

GetX 是一个集状态管理、路由、依赖注入于一体的全功能框架。它以极简 API 和零模板代码著称，但同时也带来强耦合和难以测试的风险。

class ProductController extends GetxController {
  final products = <Product>[].obs;
  final isLoading = false.obs;
  final selectedCategory = '全部'.obs;

  List<Product> get filteredProducts {
    if (selectedCategory.value == '全部') return products;
    return products.where((p) => p.category == selectedCategory.value).toList();
  }

  @override
  void onInit() {
    super.onInit();
    fetchProducts();
  }

  Future<void> fetchProducts() async {
    isLoading.value = true;
    try {
      final result = await Get.find<ApiService>().getProducts();
      products.assignAll(result);
    } finally {
      isLoading.value = false;
    }
  }
}

class ProductListPage extends StatelessWidget {
  const ProductListPage({super.key});

  @override
  Widget build(BuildContext context) {
    final controller = Get.put(ProductController());

    return Obx(() {
      if (controller.isLoading.value) {
        return const Center(child: CircularProgressIndicator());
      }

      return ListView.builder(
        itemCount: controller.filteredProducts.length,
        itemBuilder: (context, index) {
          final product = controller.filteredProducts[index];
          return ListTile(
            title: Text(product.name),
            subtitle: Text('¥${product.price}'),
          );
        },
      );
    });
  }
}

场景推荐总结

场景 推荐方案 理由

中大型团队项目 Bloc 架构统一，状态可追溯

小结

Flutter 状态管理没有银弹，关键在于匹配项目需求和团队能力。

graph LR
    subgraph 简单
        A1[setState]
        A2[ValueNotifier]
    end
    subgraph 中等
        B1[Provider]
        B2[GetX]
    end
    subgraph 高级
        C1[Riverpod]
        C2[Bloc]
    end
    A1 -->|规模增长| B1
    B1 -->|需要更强能力| C1
    B1 -->|需要事件追溯| C2

核心要点回顾：

  • setState 仅适用于单 Widget 内部的临时 UI 状态
  • Provider 基于 InheritedWidget，是官方推荐的入门级跨 Widget 状态方案
  • Riverpod 不依赖 BuildContext，支持编译期安全和自动缓存，是现代 Flutter 项目的首选
  • Bloc 通过 Event → State 单向流实现严格分层，适合需要状态可追溯的中大型项目
  • GetX 以极简 API 换取架构灵活性，适合快速原型但不推荐用于长期维护项目
  • 选型决策应综合考虑项目规模、团队经验、可测试性需求和长期维护成本
