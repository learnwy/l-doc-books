Dart 语言基础

核心概念

Dart 是 Google 开发的客户端优化语言，作为 Flutter 的唯一编程语言，它融合了静态类型安全、健全的空安全体系、一流的异步支持以及 Dart 3 引入的模式匹配等现代特性。掌握 Dart 的核心机制是构建高质量 Flutter 应用的前提。



类型系统与 Sound Null Safety
Dart 2.12 起引入健全的空安全（Sound Null Safety），在编译阶段彻底消除空指针异常。所有类型默认不可为 null，必须显式标注才能持有 null 值。


可空与不可空

int length = name.length;
int? nickLength = nickname?.length;

String display = nickname ?? 'Anonymous';
String forced = nickname!;

泛型与类型推断
Dart 的泛型在运行时保留类型信息（具体化泛型），不同于 Java 的类型擦除。


class Result<T> {
  final T? data;
  final String? error;

  Result.success(this.data) : error = null;
  Result.failure(this.error) : data = null;

  bool get isSuccess => data != null;

  R when<R>({
    required R Function(T data) success,
    required R Function(String error) failure,
  }) {
    if (data != null) return success(data as T);
    return failure(error!);
  }
}

void main() {
  final result = Result<int>.success(42);

  final message = result.when(
    success: (data) => 'Value: $data',
    failure: (error) => 'Error: $error',
  );

  print(result is Result<int>);
  print(result is Result<String>);
}

Stream 生成器与转换
Stream<int> countdown(int from) async* {
  for (var i = from; i >= 0; i--) {
    await Future.delayed(const Duration(seconds: 1));
    yield i;
  }
}

Stream<String> searchResults(Stream<String> queries) {
  return queries
      .distinct()
      .debounce(const Duration(milliseconds: 300))

StreamController 自定义流
class EventBus {
  final _controller = StreamController<AppEvent>.broadcast();

  Stream<T> on<T extends AppEvent>() {
    return _controller.stream.whereType<T>();
  }

  void emit(AppEvent event) {
    _controller.add(event);
  }

  void dispose() {
    _controller.close();
  }
}

类、继承与构造函数
abstract class Shape {
  double get area;
  double get perimeter;

  @override
  String toString() => '$runtimeType(area: ${area.toStringAsFixed(2)})';
}

class Circle extends Shape {
  final double radius;

  Circle(this.radius);

  @override
  double get area => 3.14159 * radius * radius;

  @override
  double get perimeter => 2 * 3.14159 * radius;
}

class Rectangle extends Shape {
  final double width;
  final double height;

  Rectangle(this.width, this.height);

  Rectangle.square(double side) : this(side, side);

  factory Rectangle.fromJson(Map<String, dynamic> json) {
    return Rectangle(
      (json['width'] as num).toDouble(),
      (json['height'] as num).toDouble(),
    );
  }

  @override
  double get area => width * height;

  @override
  double get perimeter => 2 * (width + height);
}


Mixin 混入
Mixin 提供代码复用，不需要继承关系，可以在不破坏类层次的前提下注入行为。


mixin Loggable {
  void log(String message) {
    print('[${DateTime.now().toIso8601String()}] $runtimeType: $message');
  }
}

mixin Cacheable<T> {
  final Map<String, T> _cache = {};

  T? getFromCache(String key) => _cache[key];

  void putInCache(String key, T value) {
    _cache[key] = value;
  }

  void clearCache() => _cache.clear();
}

mixin Validatable {
  List<String> validate();

  bool get isValid => validate().isEmpty;
}

class UserRepository with Loggable, Cacheable<User> {
  Future<User> getUser(String id) async {
    final cached = getFromCache(id);
    if (cached != null) {
      log('Cache hit for user $id');
      return cached;
    }

    log('Fetching user $id from API');
    final user = await fetchUserFromApi(id);
    putInCache(id, user);
    return user;
  }
}


扩展方法
扩展方法在不修改原类的前提下为其添加功能，广泛用于为基础类型和第三方类型添加领域方法。


extension StringX on String {
  String get capitalized {
    if (isEmpty) return this;
    return '${this[0].toUpperCase()}${substring(1)}';
  }

  String truncate(int maxLength, {String suffix = '...'}) {
    if (length <= maxLength) return this;
    return '${substring(0, maxLength - suffix.length)}$suffix';
  }

  bool get isValidEmail {
    return RegExp(r'^[\w-.]+@([\w-]+\.)+[\w-]{2,4}$').hasMatch(this);
  }
}

extension DateTimeX on DateTime {
  String get relative {
    final diff = DateTime.now().difference(this);
    if (diff.inDays > 365) return '${diff.inDays ~/ 365}年前';
    if (diff.inDays > 30) return '${diff.inDays ~/ 30}个月前';
    if (diff.inDays > 0) return '${diff.inDays}天前';
    if (diff.inHours > 0) return '${diff.inHours}小时前';
    if (diff.inMinutes > 0) return '${diff.inMinutes}分钟前';
    return '刚刚';
  }
}

extension IterableX<T> on Iterable<T> {
  Map<K, List<T>> groupBy<K>(K Function(T) keySelector) {
    return fold<Map<K, List<T>>>({}, (map, item) {
      final key = keySelector(item);
      (map[key] ??= []).add(item);
      return map;
    });
  }

  T? firstWhereOrNull(bool Function(T) test) {
    for (final element in this) {
      if (test(element)) return element;
    }
    return null;
  }
}


模式匹配（Dart 3）
Dart 3 引入了完整的模式匹配系统，包括 Records、解构、sealed class 穷举和 guard 子句。


Records 与解构
(String, int) getUserInfo() {
  return ('Alice', 30);
}

({String name, int age}) getNamedUserInfo() {
  return (name: 'Bob', age: 25);

void main() {
  final (name, age) = getUserInfo();
  print('$name is $age years old');

  final (:name as userName, :age as userAge) = getNamedUserInfo();

Sealed Class 与穷举匹配
sealed class NetworkState {}

class Loading extends NetworkState {}

class Success extends NetworkState {
  final String data;
  Success(this.data);
}

class Error extends NetworkState {
  final String message;
  final int? statusCode;
  Error(this.message, [this.statusCode]);
}

Widget buildContent(NetworkState state) {
  return switch (state) {
    Loading()                          => const CircularProgressIndicator(),
    Success(data: final d)             => Text(d),
    Error(message: final m, statusCode: 404) => Text('未找到: $m'),
    Error(message: final m, statusCode: final c?) => Text('错误 $c: $m'),
    Error(message: final m)            => Text('未知错误: $m'),
  };
}
if-case 与 Guard 子句
void processResponse(Map<String, dynamic> json) {
  if (json case {'status': 'ok', 'data': Map<String, dynamic> data}) {
    handleData(data);
  }

  if (json case {'items': List<dynamic> items} when items.isNotEmpty) {
    processItems(items);
  }

  switch (json) {
    case {'type': 'text', 'content': String content} when content.length > 100:
      handleLongText(content);
    case {'type': 'text', 'content': String content}:
      handleShortText(content);
    case {'type': 'image', 'url': String url}:
      handleImage(url);
    default:
      handleUnknown(json);
  }
}

小结
Dart 作为 Flutter 的基石语言，提供了从类型安全到异步模型再到模式匹配的完整现代语言能力。


核心要点回顾：


    

  • Sound Null Safety 从类型层面消除空指针，?、!、late 是三种核心工具
    • 

  • Future 处理单次异步，Stream 处理持续事件流，Future.wait 实现并发
    • 

  • Mixin 通过 with 实现多行为注入，扩展方法通过 extension 为现有类型添加功能
    • 

  • Dart 3 的 sealed class + switch 表达式实现穷举模式匹配，Records 提供轻量元组
    • 

  • 集合扩展运算符 ... 和条件插入 if/for 在 Flutter UI 构建中极为常用
