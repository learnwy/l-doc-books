RN 新架构

核心概念

React Native 新架构（New Architecture）是 Meta 从 2018 年开始规划、2022 年正式推出的底层重构。新架构的三大支柱是 Fabric 渲染器、TurboModules 和 JSI（JavaScript Interface），辅以 Codegen 代码生成工具实现类型安全的跨语言通信。新架构从根本上解决了旧架构的性能瓶颈——JSON 序列化桥接、同步原生模块初始化和单线程渲染限制，将 React Native 的性能上限提升到了接近原生的水平。

JSI（JavaScript Interface）
JSI 是新架构的基石，它在 JS 运行时和 C++ 之间建立了直接的内存级通信通道，彻底消除了 JSON 序列化开销。


JSI 核心 API
#include <jsi/jsi.h>
#include <ReactCommon/CallInvoker.h>

using namespace facebook::jsi;

class CryptoModule : public HostObject {
 public:
  explicit CryptoModule(std::shared_ptr<CallInvoker> jsInvoker)
    : jsInvoker_(std::move(jsInvoker)) {}

  Value get(Runtime& runtime, const PropNameID& name) override {
    auto methodName = name.utf8(runtime);

    if (methodName == "sha256") {
      return Function::createFromHostFunction(
        runtime,
        PropNameID::forAscii(runtime, "sha256"),
        1,
        [](Runtime& rt, const Value&, const Value* args, size_t count) -> Value {
          auto input = args[0].asString(rt).utf8(rt);
          auto hash = computeSHA256(input);
          return String::createFromUtf8(rt, hash);
        }
      );
    }

    if (methodName == "randomBytes") {
      return Function::createFromHostFunction(
        runtime,
        PropNameID::forAscii(runtime, "randomBytes"),
        1,
        [this](Runtime& rt, const Value&, const Value* args, size_t) -> Value {
          auto size = static_cast<int>(args[0].asNumber());
          auto bytes = generateRandomBytes(size);

          auto buffer = ArrayBuffer(rt, size);
          std::memcpy(buffer.data(rt), bytes.data(), size);
          return std::move(buffer);
        }
      );
    }

    return Value::undefined();
  }

 private:
  std::shared_ptr<CallInvoker> jsInvoker_;
};

void installCryptoModule(Runtime& runtime, std::shared_ptr<CallInvoker> invoker) {
  auto module = std::make_shared<CryptoModule>(invoker);
  auto obj = Object::createFromHostObject(runtime, module);
  runtime.global().setProperty(runtime, "NativeCrypto", std::move(obj));
}
Fabric 渲染器
Fabric 是新架构的渲染系统，替代了旧架构的 UIManager。其核心特性是并发渲染——Fabric 可以在后台线程准备新的 UI 树，同时不阻塞当前帧的渲染和用户交互。


Fabric 并发渲染
import { useTransition, useDeferredValue, useState, Suspense } from 'react';
import { View, Text, TextInput, FlatList, ActivityIndicator } from 'react-native';

function SearchScreen() {
  const [query, setQuery] = useState('');
  const [isPending, startTransition] = useTransition();
  const deferredQuery = useDeferredValue(query);

  const handleSearch = (text: string) => {
    setQuery(text);
    startTransition(() => {
      performExpensiveSearch(text);
    });
  };

  return (
    <View style={{ flex: 1 }}>
      <TextInput style={styles.searchInput} value={query} onChangeText={handleSearch} placeholder="搜索..." />

      {isPending && <ActivityIndicator style={styles.indicator} />}

      <Suspense fallback={<SearchSkeleton />}>
        <SearchResults query={deferredQuery} />
      </Suspense>
    </View>
  );
}

function SearchResults({ query }: { query: string }) {
  const results = useSearchResults(query);

  return (
    <FlatList
      data={results}
      keyExtractor={(item) => item.id}
      renderItem={({ item }) => (
        <View style={styles.resultItem}>
          <Text style={styles.resultTitle}>{item.title}</Text>
          <Text style={styles.resultDesc}>{item.description}</Text>
        </View>
      )}
      initialNumToRender={10}
      maxToRenderPerBatch={5}
      windowSize={5}
    />
  );
}
Fabric 组件注册
package com.myapp.ui

import com.facebook.react.uimanager.SimpleViewManager
import com.facebook.react.uimanager.ThemedReactContext
import com.facebook.react.uimanager.ViewManagerDelegate
import com.facebook.react.uimanager.annotations.ReactProp
import com.facebook.react.viewmanagers.GradientViewManagerDelegate
import com.facebook.react.viewmanagers.GradientViewManagerInterface

class GradientViewManager :
    SimpleViewManager<GradientView>(),
    GradientViewManagerInterface<GradientView> {

    private val delegate = GradientViewManagerDelegate(this)

    override fun getDelegate(): ViewManagerDelegate<GradientView> = delegate

    override fun getName(): String = REACT_CLASS

    override fun createViewInstance(context: ThemedReactContext): GradientView {
        return GradientView(context)
    }

    @ReactProp(name = "colors")
    override fun setColors(view: GradientView, colors: ReadableArray?) {
        colors?.let { view.setGradientColors(it) }
    }

    @ReactProp(name = "direction")
    override fun setDirection(view: GradientView, direction: String?) {
        direction?.let { view.setDirection(it) }
    }

    companion object {
        const val REACT_CLASS = "GradientView"
    }
}
TurboModules
TurboModules 替代了旧架构的 NativeModules，核心改进是懒加载和类型安全。


TurboModule Spec 定义
import type { TurboModule } from 'react-native';
import { TurboModuleRegistry } from 'react-native';

export interface Spec extends TurboModule {
  multiply(a: number, b: number): number;

  getString(key: string): Promise<string | null>;
  setString(key: string, value: string): Promise<void>;

  getConstants(): {
    appVersion: string;
    buildNumber: string;
    isDebug: boolean;
  };

  addListener(eventName: string): void;
  removeListeners(count: number): void;
}

export default TurboModuleRegistry.getEnforcing<Spec>('AppConfig');
Android 实现
package com.myapp.config

import android.content.pm.PackageManager
import com.facebook.react.bridge.Promise
import com.facebook.react.bridge.ReactApplicationContext
import com.facebook.react.module.annotations.ReactModule

@ReactModule(name = AppConfigModule.NAME)
class AppConfigModule(reactContext: ReactApplicationContext) :
    NativeAppConfigSpec(reactContext) {

    override fun getName(): String = NAME

    override fun multiply(a: Double, b: Double): Double = a * b

    override fun getString(key: String, promise: Promise) {
        val prefs = reactApplicationContext
            .getSharedPreferences("app_config", Context.MODE_PRIVATE)
        val value = prefs.getString(key, null)
        promise.resolve(value)
    }

    override fun setString(key: String, value: String, promise: Promise) {
        val prefs = reactApplicationContext
            .getSharedPreferences("app_config", Context.MODE_PRIVATE)
        prefs.edit().putString(key, value).apply()
        promise.resolve(null)
    }

    override fun getConstants(): MutableMap<String, Any> {
        val pm = reactApplicationContext.packageManager
        val info = pm.getPackageInfo(reactApplicationContext.packageName, 0)

        return mutableMapOf(
            "appVersion" to (info.versionName ?: ""),
            "buildNumber" to info.longVersionCode.toString(),
            "isDebug" to BuildConfig.DEBUG,
        )
    }

    companion object {
        const val NAME = "AppConfig"
    }
}

兼容层 — 同时支持新旧架构
import { Platform, TurboModuleRegistry, NativeModules } from 'react-native';

interface AppConfigInterface {
  multiply(a: number, b: number): number;
  getString(key: string): Promise<string | null>;
  setString(key: string, value: string): Promise<void>;
}

function getAppConfig(): AppConfigInterface {
  const turboModule = TurboModuleRegistry.get<AppConfigInterface>('AppConfig');
  if (turboModule) {
    return turboModule;
  }
  return NativeModules.AppConfig;
}

const AppConfig = getAppConfig();

export { AppConfig };

小结
React Native 新架构是一次彻底的底层重构，JSI 消除了 JSON 序列化瓶颈，Fabric 实现了并发渲染，TurboModules 带来了懒加载和类型安全，Codegen 保证了跨语言通信的正确性。迁移到新架构虽然需要一定的工程投入，但带来的性能提升和开发体验改善是显著的。


核心要点回顾：


    

  • JSI 通过 C++ 直接绑定实现 JS 和原生零开销通信，替代 JSON Bridge
    • 

  • Fabric 渲染器支持并发渲染和 React 18 特性（Suspense/useTransition）
    • 

  • TurboModules 懒加载原生模块，减少 30-50% 启动时间
    • 

  • Codegen 从 TypeScript 类型定义自动生成跨语言桥接代码
    • 

  • Hermes 引擎通过 AOT 编译优化启动速度和内存占用
    • 

  • 迁移策略：评估兼容性 → 逐步迁移 NativeModules → 启用 Fabric → 灰度验证
    •
