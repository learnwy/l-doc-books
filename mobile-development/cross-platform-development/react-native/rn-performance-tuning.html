RN 性能调优

核心概念

React Native 应用的性能调优是一个系统工程，涉及 JS 线程优化、渲染效率提升、列表滚动流畅度、图片资源管理、内存泄漏排查等多个维度。新架构（Fabric + JSI + Hermes）为性能提供了更高的上限，但要充分发挥这些能力，仍需要开发者在应用层面采取正确的优化策略。本文从六大核心场景出发，提供可落地的性能优化方案。

性能瓶颈全景

import { FlatList, View, Text, Image, StyleSheet } from 'react-native';
import { memo, useCallback, useMemo } from 'react';

interface FeedItem {
  id: string;
  title: string;
  description: string;
  image: string;
  author: string;
  timestamp: number;
}

const FeedCard = memo(function FeedCard({ item }: { item: FeedItem }) {
  return (
    <View style={styles.card}>
      <Image source={{ uri: item.image }} style={styles.cardImage} resizeMode="cover" />
      <View style={styles.cardContent}>
        <Text style={styles.cardTitle} numberOfLines={2}>
          {item.title}
        </Text>
        <Text style={styles.cardDesc} numberOfLines={3}>
          {item.description}
        </Text>
        <View style={styles.cardFooter}>
          <Text style={styles.cardAuthor}>{item.author}</Text>
          <Text style={styles.cardTime}>{formatTime(item.timestamp)}</Text>
        </View>
      </View>
    </View>
  );
});

function OptimizedFeedList({ data }: { data: FeedItem[] }) {
  const keyExtractor = useCallback((item: FeedItem) => item.id, []);

  const renderItem = useCallback(({ item }: { item: FeedItem }) => <FeedCard item={item} />, []);

  const getItemLayout = useCallback(
    (_data: any, index: number) => ({
      length: CARD_HEIGHT,
      offset: CARD_HEIGHT * index,
      index,
    }),
    [],
  );

  const ListEmptyComponent = useMemo(
    () => (
      <View style={styles.empty}>
        <Text style={styles.emptyText}>暂无数据</Text>
      </View>
    ),
    [],
  );

  return (
    <FlatList
      data={data}
      renderItem={renderItem}
      keyExtractor={keyExtractor}
      getItemLayout={getItemLayout}
      initialNumToRender={8}
      maxToRenderPerBatch={5}
      updateCellsBatchingPeriod={50}
      windowSize={5}
      removeClippedSubviews={true}
      ListEmptyComponent={ListEmptyComponent}
      showsVerticalScrollIndicator={false}
      contentContainerStyle={styles.listContent}
    />
  );
}

const CARD_HEIGHT = 280;

使用 useRef 避免闭包陷阱
import { useRef, useEffect, useCallback } from 'react';

function useStableCallback<T extends (...args: any[]) => any>(callback: T): T {
  const callbackRef = useRef(callback);
  callbackRef.current = callback;

  return useCallback(((...args: any[]) => callbackRef.current(...args)) as T, []);
}

function useInterval(callback: () => void, delay: number | null) {
  const savedCallback = useRef(callback);
  savedCallback.current = callback;

  useEffect(() => {
    if (delay === null) return;

    const id = setInterval(() => savedCallback.current(), delay);
    return () => clearInterval(id);
  }, [delay]);
}
图片优化
import { Image, Platform, PixelRatio } from 'react-native';
import FastImage from 'react-native-fast-image';

function buildOptimizedUri(baseUri: string, width: number, height: number): string {
  const pixelWidth = PixelRatio.getPixelSizeForLayoutSize(width);
  const pixelHeight = PixelRatio.getPixelSizeForLayoutSize(height);
  const format = Platform.OS === 'android' ? 'webp' : 'webp';

  return `${baseUri}?w=${pixelWidth}&h=${pixelHeight}&fmt=${format}&q=80`;
}

function OptimizedImage({
  uri,
  width,
  height,
  priority = 'normal',
}: {
  uri: string;
  width: number;
  height: number;
  priority?: 'low' | 'normal' | 'high';
}) {
  const optimizedUri = buildOptimizedUri(uri, width, height);

  return (
    <FastImage
      source={{
        uri: optimizedUri,
        priority: FastImage.priority[priority],
        cache: FastImage.cacheControl.immutable,
      }}
      style={{ width, height }}
      resizeMode={FastImage.resizeMode.cover}
    />
  );
}

function preloadImages(uris: string[]) {
  FastImage.preload(
    uris.map((uri) => ({
      uri,
      priority: FastImage.priority.normal,
    })),
  );
}

JS 线程优化
长任务分片执行
import { InteractionManager } from 'react-native';

async function processLargeDataset<T, R>(data: T[], processor: (item: T) => R, chunkSize = 100): Promise<R[]> {
  const results: R[] = [];

  for (let i = 0; i < data.length; i += chunkSize) {
    const chunk = data.slice(i, i + chunkSize);

    await new Promise<void>((resolve) => {
      InteractionManager.runAfterInteractions(() => {
        const chunkResults = chunk.map(processor);
        results.push(...chunkResults);
        resolve();
      });
    });
  }

  return results;
}

async function processInBackground<T>(task: () => T): Promise<T> {
  return new Promise((resolve) => {
    InteractionManager.runAfterInteractions(() => {
      requestAnimationFrame(() => {
        const result = task();
        resolve(result);
      });
    });
  });
}

内存泄漏排查
常见泄漏模式与修复
import { useState, useEffect, useRef, useCallback } from 'react';
import { AppState, AppStateStatus } from 'react-native';

function MemorySafeScreen({ userId }: { userId: string }) {
  const [data, setData] = useState<UserData | null>(null);
  const mountedRef = useRef(true);
  const abortControllerRef = useRef<AbortController | null>(null);
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

  useEffect(() => {
    mountedRef.current = true;
    abortControllerRef.current = new AbortController();

    const fetchData = async () => {
      try {
        const response = await fetch(`/api/users/${userId}`, {
          signal: abortControllerRef.current!.signal,
        });
        const json = await response.json();
        if (mountedRef.current) {
          setData(json);
        }
      } catch (error: any) {
        if (error.name !== 'AbortError' && mountedRef.current) {
          console.error('Fetch error:', error);
        }
      }
    };

    fetchData();

    intervalRef.current = setInterval(() => {
      if (mountedRef.current) {
        fetchData();
      }
    }, 30000);

    const appStateSubscription = AppState.addEventListener('change', (nextState: AppStateStatus) => {
      if (nextState === 'active' && mountedRef.current) {
        fetchData();
      }
    });

    return () => {
      mountedRef.current = false;
      abortControllerRef.current?.abort();
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
      appStateSubscription.remove();
    };
  }, [userId]);

  if (!data) return <LoadingView />;

  return (
    <View style={styles.container}>
      <Text style={styles.name}>{data.name}</Text>
    </View>
  );
}

小结
React Native 性能调优需要同时关注 JS 线程和原生线程。FlatList 的正确配置是列表性能的基础；memo/useMemo/useCallback 是防止过度重渲染的三板斧；图片优化需要 CDN 裁剪 + WebP + 预加载的组合策略；内存泄漏的防治核心是 useEffect 清理函数的正确编写。配合 Hermes 引擎和新架构，React Native 应用可以达到接近原生的性能水平。


核心要点回顾：


    

  • FlatList 必须配置 getItemLayout、initialNumToRender、windowSize 等关键参数
    • 

  • 使用 memo + useCallback + useMemo 三板斧防止不必要的重渲染
    • 

  • 图片优化：CDN 按需裁剪 + WebP 格式 + FastImage 缓存 + 预加载
    • 

  • 长任务使用 InteractionManager.runAfterInteractions 分片执行
    • 

  • Hermes 引擎通过 AOT 字节码编译减少 30-40% 启动时间
    • 

  • 内存泄漏防治的核心是 useEffect cleanup 中取消所有副作用
