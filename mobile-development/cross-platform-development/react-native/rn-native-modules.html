RN 原生模块

核心概念

原生模块（Native Modules）是 React Native 中 JavaScript 调用平台原生代码的核心机制。当 JS 侧需要使用平台特有能力（如生物识别、蓝牙、文件系统、推送通知等）时，必须通过原生模块来桥接。React Native 新架构引入了 TurboModules，在保持原有功能的基础上实现了懒加载、类型安全和同步调用。本文详细讲解 iOS/Android 原生模块的开发、TurboModule 迁移、事件通信和线程模型。

Objective-C 实现
#import <React/RCTBridgeModule.h>
#import <React/RCTEventEmitter.h>

@interface BiometricModule : RCTEventEmitter <RCTBridgeModule>
@end

@implementation BiometricModule

RCT_EXPORT_MODULE();

+ (BOOL)requiresMainQueueSetup {
  return NO;
}

- (NSArray<NSString *> *)supportedEvents {
  return @[@"onBiometricResult"];
}

RCT_EXPORT_METHOD(isAvailable:(RCTPromiseResolveBlock)resolve
                  rejecter:(RCTPromiseRejectBlock)reject)
{
  LAContext *context = [[LAContext alloc] init];
  NSError *error;
  BOOL available = [context canEvaluatePolicy:LAPolicyDeviceOwnerAuthenticationWithBiometrics
                                        error:&error];
  if (available) {
    NSString *type;
    if (@available(iOS 11.0, *)) {
      switch (context.biometryType) {
        case LABiometryTypeFaceID:
          type = @"FaceID";
          break;
        case LABiometryTypeTouchID:
          type = @"TouchID";
          break;
        default:
          type = @"None";
          break;
      }
    } else {
      type = @"TouchID";
    }
    resolve(@{@"available": @YES, @"type": type});
  } else {
    reject(@"BIOMETRIC_UNAVAILABLE", error.localizedDescription, error);
  }
}

RCT_EXPORT_METHOD(authenticate:(NSString *)reason
                  resolver:(RCTPromiseResolveBlock)resolve
                  rejecter:(RCTPromiseRejectBlock)reject)
{
  LAContext *context = [[LAContext alloc] init];
  [context evaluatePolicy:LAPolicyDeviceOwnerAuthenticationWithBiometrics
          localizedReason:reason
                    reply:^(BOOL success, NSError *error) {
    dispatch_async(dispatch_get_main_queue(), ^{
      if (success) {
        resolve(@{@"success": @YES});
        [self sendEventWithName:@"onBiometricResult"
                           body:@{@"authenticated": @YES}];
      } else {
        reject(@"AUTH_FAILED", error.localizedDescription, error);
      }
    });
  }];
}

@end
Swift 实现
import LocalAuthentication

@objc(BiometricModuleSwift)
class BiometricModuleSwift: RCTEventEmitter {

  override static func moduleName() -> String! { "BiometricModuleSwift" }

  override static func requiresMainQueueSetup() -> Bool { false }

  override func supportedEvents() -> [String]! {
    ["onBiometricResult"]
  }

  @objc func isAvailable(
    _ resolve: @escaping RCTPromiseResolveBlock,
    rejecter reject: @escaping RCTPromiseRejectBlock
  ) {
    let context = LAContext()
    var error: NSError?
    let available = context.canEvaluatePolicy(
      .deviceOwnerAuthenticationWithBiometrics,
      error: &error
    )

    if available {
      let type: String
      if #available(iOS 11.0, *) {
        type = context.biometryType == .faceID ? "FaceID" : "TouchID"
      } else {
        type = "TouchID"
      }
      resolve(["available": true, "type": type])
    } else {
      reject("BIOMETRIC_UNAVAILABLE", error?.localizedDescription, error)
    }
  }

  @objc func authenticate(
    _ reason: String,
    resolver resolve: @escaping RCTPromiseResolveBlock,
    rejecter reject: @escaping RCTPromiseRejectBlock
  ) {
    let context = LAContext()
    context.evaluatePolicy(
      .deviceOwnerAuthenticationWithBiometrics,
      localizedReason: reason
    ) { success, error in
      DispatchQueue.main.async {
        if success {
          resolve(["success": true])
          self.sendEvent(withName: "onBiometricResult", body: ["authenticated": true])
        } else {
          reject("AUTH_FAILED", error?.localizedDescription, error)
        }
      }
    }
  }
}
Android 原生模块开发
package com.myapp.biometric

import android.os.Build
import androidx.biometric.BiometricManager
import androidx.biometric.BiometricPrompt
import androidx.core.content.ContextCompat
import androidx.fragment.app.FragmentActivity
import com.facebook.react.bridge.*
import com.facebook.react.modules.core.DeviceEventManagerModule

class BiometricModule(reactContext: ReactApplicationContext) :
    ReactContextBaseJavaModule(reactContext) {

    override fun getName(): String = "BiometricModule"

    @ReactMethod
    fun isAvailable(promise: Promise) {
        val biometricManager = BiometricManager.from(reactApplicationContext)
        val result = biometricManager.canAuthenticate(
            BiometricManager.Authenticators.BIOMETRIC_STRONG
        )

        val map = Arguments.createMap()
        when (result) {
            BiometricManager.BIOMETRIC_SUCCESS -> {
                map.putBoolean("available", true)
                map.putString("type", if (Build.VERSION.SDK_INT >= 30) "BiometricStrong" else "Fingerprint")
                promise.resolve(map)
            }
            else -> {
                map.putBoolean("available", false)
                map.putString("type", "None")
                promise.resolve(map)
            }
        }
    }

    @ReactMethod
    fun authenticate(reason: String, promise: Promise) {
        val activity = currentActivity as? FragmentActivity
        if (activity == null) {
            promise.reject("NO_ACTIVITY", "Activity not available")
            return
        }

        val executor = ContextCompat.getMainExecutor(reactApplicationContext)

        val callback = object : BiometricPrompt.AuthenticationCallback() {
            override fun onAuthenticationSucceeded(result: BiometricPrompt.AuthenticationResult) {
                val map = Arguments.createMap()
                map.putBoolean("success", true)
                promise.resolve(map)
                sendEvent("onBiometricResult", Arguments.createMap().apply {
                    putBoolean("authenticated", true)
                })
            }

            override fun onAuthenticationError(errorCode: Int, errString: CharSequence) {
                promise.reject("AUTH_ERROR", errString.toString())
            }

            override fun onAuthenticationFailed() {
                promise.reject("AUTH_FAILED", "Authentication failed")
            }
        }

        activity.runOnUiThread {
            val prompt = BiometricPrompt(activity, executor, callback)
            val info = BiometricPrompt.PromptInfo.Builder()
                .setTitle("身份验证")
                .setSubtitle(reason)
                .setNegativeButtonText("取消")
                .build()
            prompt.authenticate(info)
        }
    }

    private fun sendEvent(eventName: String, params: WritableMap) {
        reactApplicationContext
            .getJSModule(DeviceEventManagerModule.RCTDeviceEventEmitter::class.java)
            .emit(eventName, params)
    }
}

import type { TurboModule } from 'react-native';
import { TurboModuleRegistry } from 'react-native';

type BiometricInfo = {
  available: boolean;
  type: string;
};

  success: boolean;
};

  authenticate(reason: string): Promise<AuthResult>;
  addListener(eventName: string): void;
  removeListeners(count: number): void;
}

export default TurboModuleRegistry.getEnforcing<Spec>('BiometricModule');
JS 侧调用封装
import NativeBiometric from './NativeBiometricModule';
import { NativeEventEmitter, Platform } from 'react-native';

const emitter = new NativeEventEmitter(NativeBiometric);

type BiometricType = 'FaceID' | 'TouchID' | 'Fingerprint' | 'BiometricStrong' | 'None';

interface BiometricStatus {
  available: boolean;
  type: BiometricType;
}

class BiometricService {
  async checkAvailability(): Promise<BiometricStatus> {
    const result = await NativeBiometric.isAvailable();
    return result as BiometricStatus;
  }

  async authenticate(reason?: string): Promise<boolean> {
    const defaultReason = Platform.select({
      ios: '请验证身份以继续',
      android: '身份验证',
      default: '验证身份',
    });

    const result = await NativeBiometric.authenticate(reason || defaultReason);
    return result.success;
  }

  onResult(callback: (authenticated: boolean) => void): () => void {
    const subscription = emitter.addListener('onBiometricResult', (event: { authenticated: boolean }) => {
      callback(event.authenticated);
    });

    return () => subscription.remove();
  }
}

export const biometric = new BiometricService();
事件通信
事件管理最佳实践
import { useEffect, useRef, useCallback } from 'react';
import { NativeEventEmitter, NativeModule } from 'react-native';

function useNativeEvent<T>(
  module: NativeModule,
  eventName: string,
  handler: (data: T) => void,
) {
  const handlerRef = useRef(handler);
  handlerRef.current = handler;

  useEffect(() => {
    const emitter = new NativeEventEmitter(module);
    const subscription = emitter.addListener(eventName, (data: T) => {
      handlerRef.current(data);
    });

    return () => {
      subscription.remove();
    };
  }, [module, eventName]);
}

function LocationTracker() {
  const [location, setLocation] = useState<Location | null>(null);

  useNativeEvent<{ latitude: number; longitude: number }>(
    NativeModules.LocationModule,
    'onLocationUpdate',
    useCallback((data) => {
      setLocation({ lat: data.latitude, lng: data.longitude });
    }, []),
  );

  return (
    <View>
      {location && (
        <Text>
          位置: {location.lat.toFixed(6)}, {location.lng.toFixed(6)}
        </Text>
      )}
    </View>
  );
}

线程安全示例
class ThreadSafeModule(reactContext: ReactApplicationContext) :
    ReactContextBaseJavaModule(reactContext) {

    private val scope = CoroutineScope(Dispatchers.Default + SupervisorJob())
    private val dbLock = Mutex()

    override fun getName(): String = "ThreadSafeModule"

    @ReactMethod
    fun heavyComputation(input: String, promise: Promise) {
        scope.launch {
            try {
                val result = withContext(Dispatchers.Default) {
                    performComputation(input)
                }

                withContext(Dispatchers.Main) {
                    promise.resolve(result)
                }
            } catch (e: Exception) {
                promise.reject("COMPUTATION_ERROR", e.message)
            }
        }
    }

    @ReactMethod
    fun writeToDatabase(key: String, value: String, promise: Promise) {
        scope.launch {
            dbLock.withLock {
                try {
                    val db = getDatabase()
                    db.put(key, value)
                    promise.resolve(null)
                } catch (e: Exception) {
                    promise.reject("DB_ERROR", e.message)
                }
            }
        }
    }

    override fun invalidate() {
        scope.cancel()
        super.invalidate()
    }
}

小结
原生模块是 React Native 连接 JS 和平台原生能力的桥梁。iOS 使用 Objective-C/Swift 实现，Android 使用 Kotlin/Java 实现，两端通过统一的 JS 接口暴露给业务层。TurboModules 在此基础上带来了懒加载、类型安全和 100 倍的性能提升。事件通信采用 EventEmitter 模式，线程安全则需要开发者根据场景合理选择执行线程。


核心要点回顾：


    

  • iOS 原生模块通过 RCT_EXPORT_MODULE 和 RCT_EXPORT_METHOD 宏注册
    • 

  • Android 原生模块继承 ReactContextBaseJavaModule，通过 @ReactMethod 注解暴露方法
    • 

  • TurboModule 通过 TypeScript Spec 定义接口，Codegen 生成桥接代码
    • 

  • 事件通信使用 NativeEventEmitter，需在 useEffect 清理函数中取消订阅
    • 

  • 原生方法默认在 Module Thread 执行，UI 操作需显式切到 Main Thread
    • 

  • TurboModules 支持同步调用和懒加载，性能提升 100 倍
