RN 导航模式

核心概念

导航是移动应用的骨架，它定义了用户在不同页面和功能间的流转方式。React Native 生态中，React Navigation 是事实标准的导航库，提供了栈导航（Stack）、标签导航（Tab）、抽屉导航（Drawer）等多种模式。同时，原生导航方案（如 react-native-screens）通过复用平台原生导航控制器来获得更佳的性能和过渡动画。本文系统讲解各种导航模式的实现、类型安全、Deep Link 集成和性能优化。

导航架构全景

类型安全的导航配置
import type { NativeStackScreenProps } from '@react-navigation/native-stack';
import type { BottomTabScreenProps } from '@react-navigation/bottom-tabs';
import type { DrawerScreenProps } from '@react-navigation/drawer';
import type { CompositeScreenProps, NavigatorScreenParams } from '@react-navigation/native';

type HomeStackParamList = {
  Feed: undefined;
  PostDetail: { postId: string; title: string };
  UserProfile: { userId: string };
  Settings: undefined;
};

type SearchStackParamList = {
  SearchHome: undefined;
  SearchResults: { query: string; filters?: SearchFilters };
  CategoryDetail: { categoryId: string; name: string };
};

type MainTabParamList = {
  HomeTab: NavigatorScreenParams<HomeStackParamList>;
  SearchTab: NavigatorScreenParams<SearchStackParamList>;
  CreateTab: undefined;
  NotificationTab: undefined;
  ProfileTab: { userId: string };
};

type RootDrawerParamList = {
  MainTabs: NavigatorScreenParams<MainTabParamList>;
  Favorites: undefined;
  History: undefined;
  Settings: undefined;
};

type HomeStackScreenProps<T extends keyof HomeStackParamList> = CompositeScreenProps<
  NativeStackScreenProps<HomeStackParamList, T>,
  CompositeScreenProps<BottomTabScreenProps<MainTabParamList>, DrawerScreenProps<RootDrawerParamList>>
>;
栈导航（Stack Navigation）
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { NavigationContainer } from '@react-navigation/native';

const HomeStack = createNativeStackNavigator<HomeStackParamList>();

function HomeStackNavigator() {
  return (
    <HomeStack.Navigator
      initialRouteName="Feed"
      screenOptions={{
        headerStyle: { backgroundColor: '#ffffff' },
        headerTintColor: '#1a1a1a',
        headerTitleStyle: { fontWeight: '600', fontSize: 17 },
        headerShadowVisible: false,
        animation: 'slide_from_right',
        gestureEnabled: true,
        fullScreenGestureEnabled: true,
      }}
    >
      <HomeStack.Screen name="Feed" component={FeedScreen} options={{ headerTitle: '首页' }} />
      <HomeStack.Screen
        name="PostDetail"
        component={PostDetailScreen}
        options={({ route }) => ({
          headerTitle: route.params.title,
          headerBackTitle: '返回',
        })}
      />
      <HomeStack.Screen
        name="UserProfile"
        component={UserProfileScreen}
        options={{
          headerTransparent: true,
          headerTitle: '',
          headerTintColor: '#ffffff',
        }}
      />
      <HomeStack.Screen
        name="Settings"
        component={SettingsScreen}
        options={{
          presentation: 'modal',
          animation: 'slide_from_bottom',
          headerTitle: '设置',
        }}
      />
    </HomeStack.Navigator>
  );
}

function FeedScreen({ navigation }: HomeStackScreenProps<'Feed'>) {
  const handlePostPress = (post: Post) => {
    navigation.navigate('PostDetail', {
      postId: post.id,
      title: post.title,
    });
  };

  const handleUserPress = (userId: string) => {
    navigation.navigate('UserProfile', { userId });
  };

  return (
    <FlatList
      data={posts}
      renderItem={({ item }) => (
        <PostCard
          post={item}
          onPress={() => handlePostPress(item)}
          onUserPress={() => handleUserPress(item.authorId)}
        />
      )}
      keyExtractor={(item) => item.id}
    />
  );
}

function PostDetailScreen({ route, navigation }: HomeStackScreenProps<'PostDetail'>) {
  const { postId, title } = route.params;

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>{title}</Text>
      <PostContent postId={postId} />
      <Pressable
        style={styles.relatedButton}
        onPress={() =>
          navigation.push('PostDetail', {
            postId: 'related-1',
            title: '相关文章',
          })
        }
      >
        <Text>查看相关文章</Text>
      </Pressable>
    </ScrollView>
  );
}

import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';

const MainTab = createBottomTabNavigator<MainTabParamList>();

function MainTabNavigator() {
  return (
    <MainTab.Navigator
      screenOptions={({ route }) => ({
        headerShown: false,
        tabBarActiveTintColor: '#1890ff',
        tabBarInactiveTintColor: '#999999',
        tabBarStyle: {
          backgroundColor: '#ffffff',

      })}
    >
      <MainTab.Screen name="HomeTab" component={HomeStackNavigator} options={{ tabBarLabel: '首页' }} />
      <MainTab.Screen name="SearchTab" component={SearchStackNavigator} options={{ tabBarLabel: '搜索' }} />
      <MainTab.Screen
        name="CreateTab"
        component={CreateScreen}
        options={{
          tabBarLabel: '发布',
          tabBarButton: (props) => <CreateTabButton {...props} />,
        }}
      />
      <MainTab.Screen
        name="NotificationTab"
        component={NotificationScreen}
        options={{
          tabBarLabel: '通知',
          tabBarBadge: 3,
        }}
      />
      <MainTab.Screen
        name="ProfileTab"
        component={ProfileScreen}
        initialParams={{ userId: 'me' }}
        options={{ tabBarLabel: '我的' }}
      />
    </MainTab.Navigator>
  );
}
抽屉导航（Drawer Navigation）
import { createDrawerNavigator } from '@react-navigation/drawer';

const RootDrawer = createDrawerNavigator<RootDrawerParamList>();

function RootNavigator() {
  return (
    <RootDrawer.Navigator
      screenOptions={{
        headerShown: false,
        drawerType: 'slide',
        drawerStyle: { width: 280, backgroundColor: '#ffffff' },
        overlayColor: 'rgba(0,0,0,0.4)',
        swipeEdgeWidth: 50,
      }}
      drawerContent={(props) => <CustomDrawerContent {...props} />}
    >
      <RootDrawer.Screen name="MainTabs" component={MainTabNavigator} />
      <RootDrawer.Screen name="Favorites" component={FavoritesScreen} options={{ drawerLabel: '我的收藏' }} />
      <RootDrawer.Screen name="History" component={HistoryScreen} options={{ drawerLabel: '浏览历史' }} />
      <RootDrawer.Screen name="Settings" component={SettingsScreen} options={{ drawerLabel: '设置' }} />
    </RootDrawer.Navigator>
  );
}

function CustomDrawerContent(props: DrawerContentComponentProps) {
  const { state, descriptors, navigation } = props;

  return (
    <SafeAreaView style={styles.drawer}>
      <View style={styles.drawerHeader}>
        <Image source={{ uri: user.avatar }} style={styles.drawerAvatar} />
        <Text style={styles.drawerName}>{user.name}</Text>
        <Text style={styles.drawerEmail}>{user.email}</Text>
      </View>

      <View style={styles.drawerBody}>
        {state.routes.map((route, index) => {
          const { options } = descriptors[route.key];
          const focused = state.index === index;

          return (
            <Pressable
              key={route.key}
              style={[styles.drawerItem, focused && styles.drawerItemActive]}
              onPress={() => navigation.navigate(route.name)}
            >
              <Text style={[styles.drawerLabel, focused && styles.drawerLabelActive]}>
                {options.drawerLabel as string}
              </Text>
            </Pressable>
          );
        })}
      </View>
    </SafeAreaView>
  );
}
Deep Link
Deep Link 配置
import { NavigationContainer, LinkingOptions } from '@react-navigation/native';
import { Linking } from 'react-native';

const linking: LinkingOptions<RootDrawerParamList> = {
  prefixes: ['myapp://', 'https://myapp.example.com'],

  config: {
    screens: {
      MainTabs: {
        screens: {
          HomeTab: {
            screens: {
              Feed: 'feed',
              PostDetail: 'post/:postId',
              UserProfile: 'user/:userId',
            },
          },
          SearchTab: {
            screens: {
              SearchResults: 'search/:query',
            },
          },
          ProfileTab: 'profile/:userId',
        },
      },
      Favorites: 'favorites',
      History: 'history',
      Settings: 'settings',
    },
  },

  async getInitialURL() {
    const url = await Linking.getInitialURL();
    if (url != null) return url;
    return null;
  },

  subscribe(listener) {
    const subscription = Linking.addEventListener('url', ({ url }) => {
      listener(url);
    });
    return () => subscription.remove();
  },
};

function App() {
  return (
    <NavigationContainer linking={linking} fallback={<LoadingScreen />}>
      <RootNavigator />
    </NavigationContainer>
  );
}
导航工具函数
import { createNavigationContainerRef, StackActions } from '@react-navigation/native';

export const navigationRef = createNavigationContainerRef<RootDrawerParamList>();

export function navigate(name: string, params?: object) {
  if (navigationRef.isReady()) {
    navigationRef.navigate(name as any, params);
  }
}

export function goBack() {
  if (navigationRef.isReady() && navigationRef.canGoBack()) {
    navigationRef.goBack();
  }
}

export function resetToHome() {
  if (navigationRef.isReady()) {
    navigationRef.reset({
      index: 0,
      routes: [{ name: 'MainTabs' }],
    });
  }
}

export function pushScreen(name: string, params?: object) {
  if (navigationRef.isReady()) {
    navigationRef.dispatch(StackActions.push(name, params));
  }
}

import { memo, useCallback, lazy, Suspense } from 'react';

const LazySettingsScreen = lazy(() => import('./screens/SettingsScreen'));

function SettingsScreenWrapper(props: any) {
  return (
    <Suspense fallback={<ActivityIndicator />}>

const MemoizedFeedScreen = memo(FeedScreen);

function usePreloadRoute() {
  const navigation = useNavigation();

  const preload = useCallback((routeName: string) => {
    const routes = navigation.getState()?.routes;
    if (!routes?.find(r => r.name === routeName)) {
      navigation.dispatch(
        CommonActions.preload(routeName)
      );
    }
  }, [navigation]);

  return preload;
}

小结
React Navigation 提供了完整的导航解决方案，从栈导航的页面跳转到标签导航的功能区切换，从抽屉导航的侧边菜单到 Deep Link 的外部跳转，覆盖了移动应用的所有导航场景。类型安全的参数定义、灵活的嵌套组合和性能优化策略是构建复杂导航结构的关键。


核心要点回顾：


    

  • 使用 TypeScript ParamList 类型定义实现类型安全的导航参数传递
    • 

  • 栈导航使用 createNativeStackNavigator 获得原生过渡动画和手势
    • 

  • Tab 导航保持所有标签页的状态，适合主功能区切换
    • 

  • Drawer 导航适合低频访问的二级功能入口
    • 

  • Deep Link 通过 linking 配置实现 URL 到页面的映射
    • 

  • 性能优化：懒加载页面、memo 化组件、预加载路由、冻结非活跃页面
