Tauri：Rust 后端与 Web 前端

核心概念

Tauri 是新一代桌面应用框架，使用 Rust 作为后端、系统 WebView 作为渲染引擎。相比 Electron 内嵌 Chromium 的方案，Tauri 应用体积可缩小至 3-10MB，内存占用降低 60-70%。Tauri 2.0 进一步引入移动端支持和增强的插件系统。本文深入解析 Tauri 的命令系统、事件通信、插件开发、安全模型以及与 Electron 的全方位对比。

命令系统（Commands）
Tauri 命令是前后端通信的核心，每个命令都是一个 Rust 函数，通过属性宏标注：


基础命令
use serde::{Deserialize, Serialize};
use std::path::PathBuf;
use tauri::State;

#[derive(Debug, Serialize, Deserialize)]
struct FileInfo {
    name: String,
    size: u64,
    is_dir: bool,
    modified: Option<u64>,
}

#[derive(Debug, Serialize, Deserialize)]
struct AppConfig {
    theme: String,
    language: String,
    auto_save: bool,
}

#[tauri::command]
fn list_directory(path: String) -> Result<Vec<FileInfo>, String> {
    let dir_path = PathBuf::from(&path);
    if !dir_path.exists() {
        return Err(format!("Path not found: {}", path));
    }

    let entries = std::fs::read_dir(&dir_path)
        .map_err(|e| e.to_string())?
        .filter_map(|entry| {
            let entry = entry.ok()?;
            let metadata = entry.metadata().ok()?;
            Some(FileInfo {
                name: entry.file_name().to_string_lossy().to_string(),
                size: metadata.len(),
                is_dir: metadata.is_dir(),
                modified: metadata
                    .modified()
                    .ok()
                    .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
                    .map(|d| d.as_secs()),
            })
        })
        .collect();

    Ok(entries)
}
异步命令与状态管理
use std::sync::Mutex;
use tauri::State;
use tokio::fs;

struct AppState {
    config: Mutex<AppConfig>,
    db_pool: sqlx::SqlitePool,
}

#[tauri::command]
async fn read_file_async(path: String) -> Result<String, String> {
    fs::read_to_string(&path)
        .await
        .map_err(|e| format!("Failed to read {}: {}", path, e))
}

#[tauri::command]
async fn save_config(
    state: State<'_, AppState>,
    config: AppConfig,
) -> Result<(), String> {
    let mut current = state.config.lock().map_err(|e| e.to_string())?;
    *current = config;
    Ok(())
}

#[tauri::command]
async fn query_records(
    state: State<'_, AppState>,
    search: String,
    limit: i64,
) -> Result<Vec<Record>, String> {
    sqlx::query_as::<_, Record>("SELECT * FROM records WHERE title LIKE ? LIMIT ?")
        .bind(format!("%{}%", search))
        .bind(limit)
        .fetch_all(&state.db_pool)
        .await
        .map_err(|e| e.to_string())
}
前端调用
import { invoke } from '@tauri-apps/api/core';

interface FileInfo {
  name: string;
  size: number;
  is_dir: boolean;
  modified: number | null;
}

interface AppConfig {
  theme: string;
  language: string;
  auto_save: boolean;
}

async function listDirectory(path: string): Promise<FileInfo[]> {
  return invoke<FileInfo[]>('list_directory', { path });
}

async function readFileAsync(path: string): Promise<string> {
  return invoke<string>('read_file_async', { path });
}

async function saveConfig(config: AppConfig): Promise<void> {
  return invoke('save_config', { config });
}

async function searchRecords(search: string, limit: number) {
  return invoke('query_records', { search, limit });
}
事件系统
Tauri 事件系统支持全局事件和窗口级事件，实现前后端双向实时通信：


Rust 端事件发射
use tauri::{AppHandle, Emitter, Manager};
use std::time::Duration;

#[tauri::command]
async fn process_large_file(
    app: AppHandle,
    file_path: String,
) -> Result<String, String> {
    let file_size = std::fs::metadata(&file_path)
        .map_err(|e| e.to_string())?
        .len();

    let chunk_size: u64 = 1024 * 1024;
    let total_chunks = (file_size + chunk_size - 1) / chunk_size;

    for i in 0..total_chunks {
        tokio::time::sleep(Duration::from_millis(50)).await;

        let progress = ((i + 1) as f64 / total_chunks as f64 * 100.0) as u32;
        app.emit("processing:progress", serde_json::json!({
            "chunk": i + 1,
            "total": total_chunks,
            "percent": progress
        })).map_err(|e| e.to_string())?;
    }

    Ok("Processing complete".to_string())
}

fn start_file_watcher(app: &AppHandle, watch_path: String) {
    let app_handle = app.clone();
    std::thread::spawn(move || {
        let (tx, rx) = std::sync::mpsc::channel();
        let mut watcher = notify::recommended_watcher(tx).unwrap();
        watcher.watch(
            std::path::Path::new(&watch_path),
            notify::RecursiveMode::Recursive,
        ).unwrap();

        for event in rx {
            if let Ok(event) = event {
                let _ = app_handle.emit("fs:changed", serde_json::json!({
                    "paths": event.paths,
                    "kind": format!("{:?}", event.kind)
                }));
            }
        }
    });
}
前端事件监听
import { listen, emit } from '@tauri-apps/api/event';

interface ProgressPayload {
  chunk: number;
  total: number;
  percent: number;
}

interface FileChangePayload {
  paths: string[];
  kind: string;
}

async function setupEventListeners() {
  const unlistenProgress = await listen<ProgressPayload>('processing:progress', (event) => {
    updateProgressBar(event.payload.percent);
    updateStatusText(`处理中: ${event.payload.chunk}/${event.payload.total}`);
  });

  const unlistenFileChange = await listen<FileChangePayload>('fs:changed', (event) => {
    refreshFileList(event.payload.paths);
  });

  return () => {
    unlistenProgress();
    unlistenFileChange();
  };
}

async function requestProcessing(filePath: string) {
  await emit('file:upload', { path: filePath });
}

Rust 插件实现
use tauri::{
    plugin::{Builder, TauriPlugin},
    Manager, Runtime,
};

mod commands;
mod error;

pub use error::Error;
type Result<T> = std::result::Result<T, Error>;

struct ClipboardState {
    history: std::sync::Mutex<Vec<String>>,
    max_history: usize,
}

pub fn init<R: Runtime>() -> TauriPlugin<R> {
    Builder::new("clipboard")
        .invoke_handler(tauri::generate_handler![
            commands::read_text,
            commands::write_text,
            commands::read_image,
            commands::get_history,
            commands::clear_history,
        ])
        .setup(|app, _api| {
            app.manage(ClipboardState {
                history: std::sync::Mutex::new(Vec::new()),
                max_history: 50,
            });
            Ok(())
        })
        .build()
}
use tauri::State;
use crate::{ClipboardState, Result};

#[tauri::command]
pub fn read_text() -> Result<String> {
    let mut clipboard = arboard::Clipboard::new()
        .map_err(|e| crate::Error::Clipboard(e.to_string()))?;
    clipboard
        .get_text()
        .map_err(|e| crate::Error::Clipboard(e.to_string()))
}

#[tauri::command]
pub fn write_text(
    state: State<'_, ClipboardState>,
    text: String,
) -> Result<()> {
    let mut clipboard = arboard::Clipboard::new()
        .map_err(|e| crate::Error::Clipboard(e.to_string()))?;
    clipboard
        .set_text(&text)
        .map_err(|e| crate::Error::Clipboard(e.to_string()))?;

    let mut history = state.history.lock().unwrap();
    history.insert(0, text);
    history.truncate(state.max_history);
    Ok(())
}

#[tauri::command]
pub fn get_history(state: State<'_, ClipboardState>) -> Vec<String> {
    state.history.lock().unwrap().clone()
}
插件注册
fn main() {
    tauri::Builder::default()
        .plugin(tauri_plugin_clipboard::init())
        .plugin(tauri_plugin_shell::init())
        .plugin(tauri_plugin_dialog::init())
        .plugin(tauri_plugin_fs::init())
        .invoke_handler(tauri::generate_handler![
            list_directory,
            read_file_async,
            save_config,
        ])
        .setup(|app| {
            let state = AppState {
                config: Mutex::new(AppConfig::default()),
            };
            app.manage(state);
            Ok(())
        })
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}
安全模型
Tauri 采用默认最小权限原则，通过 capabilities 配置精确控制权限：


权限配置
{
  "identifier": "main-window",
  "description": "Main window capabilities",
  "windows": ["main"],
  "permissions": [
    "core:default",
    "dialog:open",
    "dialog:save",
    "fs:read-files",
    "fs:write-files",
    "clipboard:read-text",
    "clipboard:write-text",
    {
      "identifier": "fs:scope",
      "allow": [{ "path": "$DOCUMENT/**" }, { "path": "$DOWNLOAD/**" }],
      "deny": [{ "path": "$HOME/.ssh/**" }, { "path": "$HOME/.gnupg/**" }]
    }
  ]
}

技术选型决策树
小结
Tauri 代表了桌面应用开发的新方向——用 Rust 的安全性和性能取代 Node.js，用系统 WebView 取代内嵌 Chromium。命令系统提供了类型安全的前后端通信，事件系统实现了实时双向推送，插件架构允许封装和复用原生功能。最小权限安全模型从根本上减少了攻击面。然而 Electron 在生态成熟度和跨平台渲染一致性方面仍有优势。选择取决于团队技术栈、包体积需求和目标平台覆盖范围。


核心要点回顾：


    

  • Rust 后端 + 系统 WebView 实现极致轻量
    • 

  • Command 系统提供类型安全的 IPC 通信
    • 

  • 事件系统支持全局和窗口级实时双向通信
    • 

  • 插件架构封装可复用的原生功能
    • 

  • 默认最小权限 + Capability 配置确保安全
    • 

  • Tauri 2.0 新增 iOS/Android 移动端支持
