WebView 应用架构

核心概念

WebView 是操作系统提供的内嵌浏览器组件，允许应用在原生容器中渲染 Web 内容。不同平台提供了不同的 WebView 实现——macOS 的 WKWebView、Windows 的 WebView2、Linux 的 WebKitGTK。基于 WebView 的桌面应用框架（如 Tauri、Neutralino、Wails）利用系统 WebView 取代内嵌 Chromium，大幅减小应用体积。本文深入分析各平台 WebView 引擎、通信机制、离线支持方案和混合架构设计。

import WebKit

class NativeWebViewController: NSViewController, WKScriptMessageHandler {
    private var webView: WKWebView!

    override func loadView() {
        let config = WKWebViewConfiguration()

        let contentController = WKUserContentController()
        contentController.add(self, name: "nativeAPI")
        contentController.add(self, name: "fileSystem")
        contentController.add(self, name: "notifications")
        config.userContentController = contentController

        let prefs = WKWebpagePreferences()
        prefs.allowsContentJavaScript = true
        config.defaultWebpagePreferences = prefs

        config.preferences.isElementFullscreenEnabled = true
        config.setValue(true, forKey: "developerExtrasEnabled")

        webView = WKWebView(frame: .zero, configuration: config)
        webView.navigationDelegate = self
        self.view = webView
    }

    override func viewDidLoad() {
        super.viewDidLoad()
        if let htmlURL = Bundle.main.url(forResource: "index", withExtension: "html", subdirectory: "web") {
            webView.loadFileURL(htmlURL, allowingReadAccessTo: htmlURL.deletingLastPathComponent())
        }
    }

    func userContentController(
        _ userContentController: WKUserContentController,
        didReceive message: WKScriptMessage
    ) {
        guard let body = message.body as? [String: Any],
              let action = body["action"] as? String else { return }

        switch message.name {
        case "nativeAPI":
            handleNativeAPI(action: action, params: body)
        case "fileSystem":
            handleFileSystem(action: action, params: body)
        case "notifications":
            handleNotification(action: action, params: body)
        default:
            break
        }
    }

    private func handleNativeAPI(action: String, params: [String: Any]) {
        switch action {
        case "getSystemInfo":
            let info: [String: Any] = [
                "platform": "macOS",
                "version": ProcessInfo.processInfo.operatingSystemVersionString,
                "memory": ProcessInfo.processInfo.physicalMemory,
                "processors": ProcessInfo.processInfo.processorCount
            ]
            callJavaScript("window.__native__.onSystemInfo", args: [info])

        case "showSaveDialog":
            let panel = NSSavePanel()
            panel.allowedContentTypes = [.json, .plainText]
            panel.begin { [weak self] response in
                if response == .OK, let url = panel.url {
                    self?.callJavaScript("window.__native__.onFileSaved", args: [url.path])
                }
            }

        default:
            break
        }
    }

    private func callJavaScript(_ function: String, args: [Any]) {
        let jsonData = try! JSONSerialization.data(withJSONObject: args)
        let jsonString = String(data: jsonData, encoding: .utf8)!
        webView.evaluateJavaScript("\(function)(\(jsonString))")
    }
}

using Microsoft.UI.Xaml;
using Microsoft.Web.WebView2.Core;
using System;
using System.IO;
using System.Text.Json;
using System.Threading.Tasks;

public sealed partial class MainWindow : Window
{
    private CoreWebView2 _coreWebView;

    public MainWindow()
    {
        InitializeComponent();
        InitializeWebView();
    }

    private async void InitializeWebView()
    {
        var env = await CoreWebView2Environment.CreateAsync(
            null,
            Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                "MyApp", "WebView2Cache"
            ),
            new CoreWebView2EnvironmentOptions
            {
                AdditionalBrowserArguments = "--disable-web-security=false",
                AllowSingleSignOnUsingOSPrimaryAccount = false,
            }
        );

        await WebView.EnsureCoreWebView2Async(env);
        _coreWebView = WebView.CoreWebView2;

        _coreWebView.WebMessageReceived += OnWebMessageReceived;

        _coreWebView.Settings.IsScriptEnabled = true;
        _coreWebView.Settings.AreDefaultScriptDialogsEnabled = false;
        _coreWebView.Settings.IsWebMessageEnabled = true;
        _coreWebView.Settings.AreDevToolsEnabled =
#if DEBUG
            true;
#else
            false;
#endif

        var appDir = Path.Combine(AppContext.BaseDirectory, "web");
        _coreWebView.SetVirtualHostNameToFolderMapping(
            "app.local",
            appDir,
            CoreWebView2HostResourceAccessKind.Allow
        );

        _coreWebView.Navigate("https://app.local/index.html");
    }

    private async void OnWebMessageReceived(
        CoreWebView2 sender,
        CoreWebView2WebMessageReceivedEventArgs args)
    {
        var message = JsonSerializer.Deserialize<NativeMessage>(
            args.WebMessageAsJson
        );

        switch (message?.Type)
        {
            case "readFile":
                await HandleReadFile(message.Payload);
                break;
            case "writeFile":
                await HandleWriteFile(message.Payload);
                break;
            case "getSystemInfo":
                SendToWeb("systemInfo", GetSystemInfo());
                break;
        }
    }

    private void SendToWeb(string type, object data)
    {
        var message = JsonSerializer.Serialize(new { type, data });
        _coreWebView.PostWebMessageAsJson(message);
    }

    private async Task HandleReadFile(JsonElement payload)
    {
        var path = payload.GetProperty("path").GetString();
        var content = await File.ReadAllTextAsync(path!);
        SendToWeb("fileContent", new { path, content });
    }
}

通信机制
统一通信桥接层
type MessageHandler = (data: unknown) => void;

interface PendingRequest {
  resolve: (value: unknown) => void;
  reject: (reason: Error) => void;
  timeout: ReturnType<typeof setTimeout>;
}

class WebViewBridge {
  private handlers = new Map<string, MessageHandler>();
  private pendingRequests = new Map<string, PendingRequest>();
  private requestId = 0;

  constructor() {
    window.addEventListener('message', this.onMessage.bind(this));

    if ((window as any).__NATIVE_CALLBACK__) {
      (window as any).__NATIVE_CALLBACK__ = this.onMessage.bind(this);
    }
  }

  invoke<T>(method: string, params?: unknown, timeoutMs = 5000): Promise<T> {
    return new Promise((resolve, reject) => {
      const id = `req_${++this.requestId}`;

      const timeout = setTimeout(() => {
        this.pendingRequests.delete(id);
        reject(new Error(`Request ${method} timed out after ${timeoutMs}ms`));
      }, timeoutMs);

      this.pendingRequests.set(id, {
        resolve: resolve as (value: unknown) => void,
        reject,
        timeout,
      });

      this.postToNative({ id, method, params });
    });
  }

  on(event: string, handler: MessageHandler): () => void {
    this.handlers.set(event, handler);
    return () => this.handlers.delete(event);
  }

  private onMessage(event: MessageEvent | { data: string }) {
    const data =
      typeof event === 'string'
        ? JSON.parse(event as unknown as string)
        : typeof (event as MessageEvent).data === 'string'
          ? JSON.parse((event as MessageEvent).data)
          : (event as MessageEvent).data;

    if (data.id && this.pendingRequests.has(data.id)) {
      const pending = this.pendingRequests.get(data.id)!;
      this.pendingRequests.delete(data.id);
      clearTimeout(pending.timeout);

      if (data.error) {
        pending.reject(new Error(data.error));
      } else {
        pending.resolve(data.result);
      }
      return;
    }

    if (data.event && this.handlers.has(data.event)) {
      this.handlers.get(data.event)!(data.data);
    }
  }

  private postToNative(message: unknown) {
    const json = JSON.stringify(message);

    if ((window as any).webkit?.messageHandlers?.nativeAPI) {
      (window as any).webkit.messageHandlers.nativeAPI.postMessage(json);
    } else if ((window as any).chrome?.webview) {
      (window as any).chrome.webview.postMessage(json);
    } else if ((window as any).__TAURI__) {
      (window as any).__TAURI__.invoke('handle_message', { message: json });
    }
  }
}

Service Worker 缓存策略
const CACHE_VERSION = 'v1.2.0';
const STATIC_CACHE = `static-${CACHE_VERSION}`;
const DYNAMIC_CACHE = `dynamic-${CACHE_VERSION}`;
const API_CACHE = `api-${CACHE_VERSION}`;

const STATIC_ASSETS = ['/', '/index.html', '/app.js', '/app.css', '/manifest.json'];

self.addEventListener('install', (event: ExtendableEvent) => {
  event.waitUntil(caches.open(STATIC_CACHE).then((cache) => cache.addAll(STATIC_ASSETS)));
});

self.addEventListener('activate', (event: ExtendableEvent) => {
  event.waitUntil(
    caches
      .keys()
      .then((keys) => Promise.all(keys.filter((key) => !key.includes(CACHE_VERSION)).map((key) => caches.delete(key)))),
  );
});

self.addEventListener('fetch', (event: FetchEvent) => {
  const { request } = event;
  const url = new URL(request.url);

  if (STATIC_ASSETS.includes(url.pathname)) {
    event.respondWith(cacheFirst(request, STATIC_CACHE));
    return;
  }

  if (url.pathname.startsWith('/api/')) {
    event.respondWith(networkFirst(request, API_CACHE));
    return;
  }

  event.respondWith(staleWhileRevalidate(request, DYNAMIC_CACHE));
});

async function cacheFirst(request: Request, cacheName: string): Promise<Response> {
  const cached = await caches.match(request);
  if (cached) return cached;

  const response = await fetch(request);
  const cache = await caches.open(cacheName);
  cache.put(request, response.clone());
  return response;
}

async function networkFirst(request: Request, cacheName: string): Promise<Response> {
  try {
    const response = await fetch(request);
    const cache = await caches.open(cacheName);
    cache.put(request, response.clone());
    return response;
  } catch {
    const cached = await caches.match(request);
    return cached || new Response('Offline', { status: 503 });
  }
}

async function staleWhileRevalidate(request: Request, cacheName: string): Promise<Response> {
  const cached = await caches.match(request);
  const fetchPromise = fetch(request).then((response) => {
    caches.open(cacheName).then((cache) => cache.put(request, response.clone()));
    return response;
  });
  return cached || fetchPromise;
}

class OfflineStorage {
  private db: IDBDatabase | null = null;

  async init(): Promise<void> {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open('AppOfflineDB', 2);

      request.onupgradeneeded = (event) => {
        const db = (event.target as IDBOpenDBRequest).result;

        if (!db.objectStoreNames.contains('documents')) {
          const store = db.createObjectStore('documents', { keyPath: 'id' });
          store.createIndex('updatedAt', 'updatedAt');
          store.createIndex('syncStatus', 'syncStatus');
        }

        if (!db.objectStoreNames.contains('syncQueue')) {
          db.createObjectStore('syncQueue', { keyPath: 'id', autoIncrement: true });
        }
      };

      request.onsuccess = (event) => {
        this.db = (event.target as IDBOpenDBRequest).result;
        resolve();
      };

      request.onerror = () => reject(request.error);
    });
  }

  async saveWithSync<T extends { id: string }>(storeName: string, data: T): Promise<void> {
    const tx = this.db!.transaction([storeName, 'syncQueue'], 'readwrite');

    const record = { ...data, syncStatus: 'pending', updatedAt: Date.now() };
    tx.objectStore(storeName).put(record);
    tx.objectStore('syncQueue').add({
      store: storeName,
      operation: 'upsert',
      data: record,
      timestamp: Date.now(),
    });

    return new Promise((resolve, reject) => {
      tx.oncomplete = () => resolve();
      tx.onerror = () => reject(tx.error);
    });
  }

  async processSyncQueue(): Promise<{ synced: number; failed: number }> {
    const tx = this.db!.transaction('syncQueue', 'readonly');
    const store = tx.objectStore('syncQueue');
    const items: any[] = await this.getAllFromStore(store);

    let synced = 0;
    let failed = 0;

    for (const item of items) {
      try {
        await fetch('/api/sync', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(item),
        });

        const deleteTx = this.db!.transaction('syncQueue', 'readwrite');
        deleteTx.objectStore('syncQueue').delete(item.id);
        synced++;
      } catch {
        failed++;
      }
    }

    return { synced, failed };
  }

  private getAllFromStore(store: IDBObjectStore): Promise<any[]> {
    return new Promise((resolve, reject) => {
      const request = store.getAll();
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }
}

小结
WebView 方案通过复用系统浏览器引擎，在保持 Web 技术栈开发效率的同时大幅降低应用体积。WKWebView、WebView2、WebKitGTK 各有特点，跨平台一致性不如 Electron 内嵌 Chromium 但足以覆盖大多数场景。通信层设计是混合架构的核心，统一的 Bridge 抽象可以屏蔽平台差异。离线支持通过 Service Worker 和 IndexedDB 实现，配合同步队列可保证数据最终一致性。混合架构将 UI 放在 WebView、性能密集逻辑放在原生层，是当前最务实的桌面应用开发模式。


核心要点回顾：


    

  • WKWebView / WebView2 / WebKitGTK 是三大系统 WebView 引擎
    • 

  • 统一 Bridge 层屏蔽平台通信差异，支持 Promise 和超时机制
    • 

  • Service Worker + IndexedDB 实现完整离线体验
    • 

  • 混合架构：UI 用 Web，性能密集用原生，通过 IPC 桥接
    • 

  • 框架选型取决于包体积、生态需求和团队技术栈
