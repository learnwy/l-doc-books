桌面原生 API 访问

核心概念

桌面应用区别于 Web 应用的核心能力在于对操作系统原生 API 的访问——文件系统读写、系统通知推送、剪贴板操作、全局快捷键、系统托盘等。不同框架（Electron、Tauri、Wails）提供了不同的原生 API 封装方式。本文系统性地讲解各类原生 API 的实现模式，覆盖 Electron 和 Tauri 双框架对照，并提供统一抽象层设计方案。

import { ipcMain, dialog, app, BrowserWindow } from 'electron';
import fs from 'node:fs';
import path from 'node:path';
import { watch } from 'chokidar';

ipcMain.handle('fs:openFile', async (event) => {
  const window = BrowserWindow.fromWebContents(event.sender);
  const result = await dialog.showOpenDialog(window!, {
    title: '选择文件',
    defaultPath: app.getPath('documents'),
    filters: [
      { name: '文本文件', extensions: ['txt', 'md', 'json'] },
      { name: '图片', extensions: ['png', 'jpg', 'gif', 'webp'] },
      { name: '所有文件', extensions: ['*'] },
    ],
    properties: ['openFile', 'multiSelections'],
  });

  if (result.canceled) return null;

  const files = await Promise.all(
    result.filePaths.map(async (filePath) => {
      const stat = await fs.promises.stat(filePath);
      return {
        path: filePath,
        name: path.basename(filePath),
        size: stat.size,
        modified: stat.mtimeMs,
        content: stat.size < 10 * 1024 * 1024 ? await fs.promises.readFile(filePath, 'utf-8') : null,
      };
    }),
  );

  return files;
});

ipcMain.handle(
  'fs:saveFile',
  async (
    event,
    options: {
      content: string;
      defaultName: string;
      filters?: Electron.FileFilter[];
    },
  ) => {
    const window = BrowserWindow.fromWebContents(event.sender);
    const result = await dialog.showSaveDialog(window!, {
      title: '保存文件',
      defaultPath: path.join(app.getPath('documents'), options.defaultName),
      filters: options.filters || [{ name: '文本文件', extensions: ['txt'] }],
    });

    if (result.canceled || !result.filePath) return null;

    await fs.promises.writeFile(result.filePath, options.content, 'utf-8');
    return result.filePath;
  },
);

ipcMain.handle('fs:watchDirectory', async (event, dirPath: string) => {
  const window = BrowserWindow.fromWebContents(event.sender);

  const watcher = watch(dirPath, {
    ignoreInitial: true,
    ignored: /(^|[\/\\])\../,
    persistent: true,
    depth: 2,
  });

  watcher
    .on('add', (filePath) => {
      window?.webContents.send('fs:event', { type: 'add', path: filePath });
    })
    .on('change', (filePath) => {
      window?.webContents.send('fs:event', { type: 'change', path: filePath });
    })
    .on('unlink', (filePath) => {
      window?.webContents.send('fs:event', { type: 'remove', path: filePath });
    });

  window?.on('closed', () => watcher.close());

  return { watching: dirPath };
});
Tauri 文件操作
use std::path::PathBuf;
use serde::{Deserialize, Serialize};
use tauri::api::dialog::FileDialogBuilder;

#[derive(Debug, Serialize, Deserialize)]
struct FileEntry {
    path: String,
    name: String,
    size: u64,
    modified: u64,
    content: Option<String>,
}

#[tauri::command]
async fn open_files(window: tauri::Window) -> Result<Vec<FileEntry>, String> {
    let (tx, rx) = tokio::sync::oneshot::channel();

    FileDialogBuilder::new()
        .set_title("选择文件")
        .add_filter("文本文件", &["txt", "md", "json"])
        .add_filter("图片", &["png", "jpg", "gif", "webp"])
        .set_parent(&window)
        .pick_files(move |paths| {
            let _ = tx.send(paths);
        });

    let paths = rx.await.map_err(|e| e.to_string())?;
    let Some(paths) = paths else {
        return Ok(vec![]);
    };

    let mut entries = Vec::new();
    for file_path in paths {
        let metadata = std::fs::metadata(&file_path).map_err(|e| e.to_string())?;
        let content = if metadata.len() < 10 * 1024 * 1024 {
            std::fs::read_to_string(&file_path).ok()
        } else {
            None
        };

        entries.push(FileEntry {
            name: file_path.file_name()
                .unwrap_or_default()
                .to_string_lossy()
                .to_string(),
            path: file_path.to_string_lossy().to_string(),
            size: metadata.len(),
            modified: metadata.modified()
                .ok()
                .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
                .map(|d| d.as_secs())
                .unwrap_or(0),
            content,
        });
    }

    Ok(entries)
}

#[tauri::command]
async fn read_file_stream(
    path: String,
    chunk_size: usize,
    app: tauri::AppHandle,
) -> Result<(), String> {
    use tauri::Emitter;
    let data = tokio::fs::read(&path).await.map_err(|e| e.to_string())?;
    let total = data.len();

    for (i, chunk) in data.chunks(chunk_size).enumerate() {
        let progress = ((i + 1) * chunk_size).min(total) as f64 / total as f64 * 100.0;
        app.emit("file:chunk", serde_json::json!({
            "index": i,
            "data": base64::encode(chunk),
            "progress": progress,
        })).map_err(|e| e.to_string())?;
    }

    Ok(())
}

Electron 通知系统
import { Notification, nativeImage, app, ipcMain } from 'electron';

interface NotificationOptions {
  title: string;
  body: string;
  icon?: string;
  actions?: Array<{ type: 'button'; text: string }>;
  urgency?: 'normal' | 'critical' | 'low';
  silent?: boolean;
  tag?: string;
}

class NotificationManager {
  private activeNotifications = new Map<string, Notification>();

  show(id: string, options: NotificationOptions): void {
    if (this.activeNotifications.has(id)) {
      this.activeNotifications.get(id)!.close();
    }

    const notification = new Notification({
      title: options.title,
      body: options.body,
      icon: options.icon ? nativeImage.createFromPath(options.icon) : undefined,
      actions: options.actions,
      urgency: options.urgency || 'normal',
      silent: options.silent || false,
    });

    notification.on('click', () => {
      this.handleNotificationClick(id);
    });

    notification.on('action', (_event, index) => {
      this.handleActionClick(id, index);
    });

    notification.on('close', () => {
      this.activeNotifications.delete(id);
    });

    notification.show();
    this.activeNotifications.set(id, notification);
  }

  private handleNotificationClick(id: string) {
    const windows = BrowserWindow.getAllWindows();
    if (windows.length > 0) {
      const win = windows[0];
      if (win.isMinimized()) win.restore();
      win.focus();
      win.webContents.send('notification:click', { id });
    }
  }

  private handleActionClick(id: string, actionIndex: number) {
    const windows = BrowserWindow.getAllWindows();
    if (windows.length > 0) {
      windows[0].webContents.send('notification:action', { id, actionIndex });
    }
  }

  closeAll() {
    for (const [, notification] of this.activeNotifications) {
      notification.close();
    }
    this.activeNotifications.clear();
  }
}

const notificationManager = new NotificationManager();

ipcMain.handle('notification:show', (_event, id: string, options: NotificationOptions) => {
  notificationManager.show(id, options);
});

多格式剪贴板操作
import { clipboard, nativeImage, ipcMain } from 'electron';

interface ClipboardEntry {
  type: 'text' | 'html' | 'image' | 'rtf';
  content: string;
  timestamp: number;
}

class ClipboardManager {
  private history: ClipboardEntry[] = [];
  private maxHistory = 100;
  private pollInterval: ReturnType<typeof setInterval> | null = null;
  private lastContent = '';

  startWatching(intervalMs = 500): void {
    this.pollInterval = setInterval(() => {
      this.checkForChanges();
    }, intervalMs);
  }

  stopWatching(): void {
    if (this.pollInterval) {
      clearInterval(this.pollInterval);
      this.pollInterval = null;
    }
  }

  private checkForChanges(): void {
    const text = clipboard.readText();
    if (text && text !== this.lastContent) {
      this.lastContent = text;
      this.addToHistory({
        type: 'text',
        content: text,
        timestamp: Date.now(),
      });
    }
  }

  private addToHistory(entry: ClipboardEntry): void {
    this.history.unshift(entry);
    if (this.history.length > this.maxHistory) {
      this.history.pop();
    }
  }

  getHistory(): ClipboardEntry[] {
    return [...this.history];
  }

  writeText(text: string): void {
    clipboard.writeText(text);
    this.lastContent = text;
  }

  writeHtml(html: string): void {
    clipboard.writeHTML(html);
  }

  writeImage(dataUrl: string): void {
    const image = nativeImage.createFromDataURL(dataUrl);
    clipboard.writeImage(image);
  }

  readAll(): { text: string; html: string; hasImage: boolean } {
    return {
      text: clipboard.readText(),
      html: clipboard.readHTML(),
      hasImage: !clipboard.readImage().isEmpty(),
    };
  }
}

const clipboardManager = new ClipboardManager();

ipcMain.handle('clipboard:read', () => clipboardManager.readAll());
ipcMain.handle('clipboard:writeText', (_e, text: string) => clipboardManager.writeText(text));
ipcMain.handle('clipboard:writeHtml', (_e, html: string) => clipboardManager.writeHtml(html));
ipcMain.handle('clipboard:writeImage', (_e, dataUrl: string) => clipboardManager.writeImage(dataUrl));
ipcMain.handle('clipboard:history', () => clipboardManager.getHistory());
ipcMain.handle('clipboard:startWatch', () => clipboardManager.startWatching());
ipcMain.handle('clipboard:stopWatch', () => clipboardManager.stopWatching());

Electron 快捷键管理
import { globalShortcut, BrowserWindow, app, ipcMain } from 'electron';

interface ShortcutBinding {
  accelerator: string;
  action: string;
  description: string;
}

class ShortcutManager {
  private bindings = new Map<string, ShortcutBinding>();
  private mainWindow: BrowserWindow | null = null;

  setMainWindow(window: BrowserWindow) {
    this.mainWindow = window;
  }

  registerDefaults(): void {
    this.register({
      accelerator: 'CommandOrControl+Shift+Space',
      action: 'toggle-window',
      description: '显示/隐藏主窗口',
    });

    this.register({
      accelerator: 'CommandOrControl+Shift+V',
      action: 'clipboard-history',
      description: '剪贴板历史',
    });

    this.register({
      accelerator: 'CommandOrControl+Shift+F',
      action: 'global-search',
      description: '全局搜索',
    });
  }

  register(binding: ShortcutBinding): boolean {
    const success = globalShortcut.register(binding.accelerator, () => {
      this.handleShortcut(binding.action);
    });

    if (success) {
      this.bindings.set(binding.accelerator, binding);
    }

    return success;
  }

  unregister(accelerator: string): void {
    globalShortcut.unregister(accelerator);
    this.bindings.delete(accelerator);
  }

  private handleShortcut(action: string): void {
    switch (action) {
      case 'toggle-window':
        if (!this.mainWindow) return;
        if (this.mainWindow.isVisible() && this.mainWindow.isFocused()) {
          this.mainWindow.hide();
        } else {
          this.mainWindow.show();
          this.mainWindow.focus();
        }
        break;

      case 'clipboard-history':
      case 'global-search':
        this.mainWindow?.webContents.send('shortcut:triggered', action);
        this.mainWindow?.show();
        this.mainWindow?.focus();
        break;
    }
  }

  getBindings(): ShortcutBinding[] {
    return Array.from(this.bindings.values());
  }

  unregisterAll(): void {
    globalShortcut.unregisterAll();
    this.bindings.clear();
  }
}

const shortcutManager = new ShortcutManager();

app.whenReady().then(() => {
  shortcutManager.registerDefaults();
});

app.on('will-quit', () => {
  shortcutManager.unregisterAll();
});

import { Tray, Menu, nativeImage, app, BrowserWindow } from 'electron';
import path from 'node:path';

class TrayManager {
  private tray: Tray | null = null;
  private mainWindow: BrowserWindow | null = null;
  private isQuitting = false;

  create(mainWindow: BrowserWindow): void {
    this.mainWindow = mainWindow;

    const iconPath = path.join(__dirname, '..', 'assets', process.platform === 'win32' ? 'tray.ico' : 'tray.png');
    const icon = nativeImage.createFromPath(iconPath).resize({ width: 16, height: 16 });

    this.tray = new Tray(icon);
    this.tray.setToolTip(app.getName());

    this.updateMenu('idle');
    this.setupEvents();
  }

  updateMenu(status: 'idle' | 'syncing' | 'error'): void {
    const statusLabel = {
      idle: '✅ 已同步',
      syncing: '🔄 同步中...',
      error: '❌ 同步失败',
    };

    const contextMenu = Menu.buildFromTemplate([
      { label: statusLabel[status], enabled: false },
      { type: 'separator' },
      {
        label: '显示主窗口',
        click: () => {
          this.mainWindow?.show();
          this.mainWindow?.focus();
        },
      },
      {
        label: '立即同步',
        click: () => {
          this.mainWindow?.webContents.send('tray:action', 'sync');
        },
      },
      { type: 'separator' },
      {
        label: '设置',
        click: () => {
          this.mainWindow?.webContents.send('tray:action', 'settings');
          this.mainWindow?.show();
        },
      },
      { type: 'separator' },
      {
        label: '退出',
        click: () => {
          this.isQuitting = true;
          app.quit();
        },
      },
    ]);

    this.tray?.setContextMenu(contextMenu);
  }

  private setupEvents(): void {
    this.tray?.on('double-click', () => {
      this.mainWindow?.show();
      this.mainWindow?.focus();
    });

    this.mainWindow?.on('close', (event) => {
      if (!this.isQuitting) {
        event.preventDefault();
        this.mainWindow?.hide();
      }
    });
  }

  setBadge(count: number): void {
    if (process.platform === 'darwin') {
      app.dock.setBadge(count > 0 ? String(count) : '');
    }

    if (count > 0) {
      this.tray?.setTitle(` ${count}`);
    } else {
      this.tray?.setTitle('');
    }
  }

  destroy(): void {
    this.tray?.destroy();
    this.tray = null;
  }
}

interface NativeFileSystem {
  openFile(filters?: FileFilter[]): Promise<FileEntry[] | null>;
  saveFile(content: string, defaultName: string): Promise<string | null>;
  readFile(path: string): Promise<string>;
  writeFile(path: string, content: string): Promise<void>;
  watchDirectory(path: string, callback: (event: FSEvent) => void): () => void;

interface NativeNotification {
  show(options: NotifyOptions): Promise<string>;
  close(id: string): void;

interface NativeClipboard {

type PlatformType = 'electron' | 'tauri' | 'web';

function detectPlatform(): PlatformType {
  if (typeof window !== 'undefined') {
    if ((window as any).__TAURI__) return 'tauri';
    if ((window as any).electronAPI) return 'electron';
  }
  return 'web';
}

function createNativeFS(): NativeFileSystem {
  const platform = detectPlatform();
  switch (platform) {
    case 'electron':
      return new ElectronFileSystem();
    case 'tauri':
      return new TauriFileSystem();
    default:
      return new WebFileSystem();
  }
}

小结
桌面原生 API 是桌面应用的核心竞争力。Electron 通过 Node.js 提供完整的系统访问能力，API 丰富且直观；Tauri 通过 Rust 后端和插件系统提供更安全、更轻量的原生访问。在实际开发中，建议设计统一的抽象接口层，使业务代码与具体框架解耦，同时为 Web 端提供优雅的降级方案。文件系统、通知、剪贴板、快捷键和系统托盘是最常用的五类原生 API，掌握其在不同框架下的实现模式是桌面应用开发的基础。

  • 剪贴板管理器可实现多格式读写和历史记录
    • 

  • 全局快捷键需在应用退出时反注册

  • 系统托盘承载状态展示和快捷操作
