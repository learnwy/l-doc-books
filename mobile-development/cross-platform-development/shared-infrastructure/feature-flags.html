功能开关与渐进式发布

核心概念

功能开关（Feature Flags）是控制功能发布节奏的关键基础设施。通过将功能的部署和发布解耦，团队可以实现暗发布（Dark Launch）、金丝雀发布、A/B 测试和紧急回滚，而无需重新部署代码。跨平台场景下，功能开关还需处理多平台版本差异、客户端缓存和离线降级等挑战。本文深入剖析功能开关架构、渐进式发布策略、A/B 测试集成、远程配置方案及主流工具对比。

TypeScript SDK 设计
interface FeatureFlag {
  key: string;
  enabled: boolean;
  value: unknown;
  rules: EvaluationRule[];
  defaultValue: unknown;
  updatedAt: number;
}

interface EvaluationRule {
  conditions: Condition[];
  value: unknown;
  percentage?: number;
}

interface Condition {
  attribute: string;
  operator: 'eq' | 'neq' | 'gt' | 'lt' | 'contains' | 'in' | 'regex';
  value: unknown;
}

interface UserContext {
  userId: string;
  email?: string;
  platform: 'web' | 'ios' | 'android' | 'desktop';
  appVersion: string;
  country?: string;
  plan?: string;
  customAttributes?: Record<string, unknown>;
}

class FeatureFlagClient {
  private flags = new Map<string, FeatureFlag>();
  private userContext: UserContext;
  private serverUrl: string;
  private eventSource: EventSource | null = null;
  private listeners = new Map<string, Set<(value: unknown) => void>>();
  private offlineDefaults: Record<string, unknown>;

  constructor(config: {
    serverUrl: string;
    userContext: UserContext;
    offlineDefaults?: Record<string, unknown>;
    pollingInterval?: number;
  }) {
    this.serverUrl = config.serverUrl;
    this.userContext = config.userContext;
    this.offlineDefaults = config.offlineDefaults || {};
  }

  async init(): Promise<void> {
    await this.fetchFlags();
    this.startRealtimeSync();
  }

  isEnabled(key: string): boolean {
    const flag = this.flags.get(key);
    if (!flag) {
      return (this.offlineDefaults[key] as boolean) ?? false;
    }
    return this.evaluate(flag) as boolean;
  }

  getValue<T>(key: string, defaultValue: T): T {
    const flag = this.flags.get(key);
    if (!flag) {
      return (this.offlineDefaults[key] as T) ?? defaultValue;
    }
    return (this.evaluate(flag) as T) ?? defaultValue;
  }

  onChange(key: string, callback: (value: unknown) => void): () => void {
    if (!this.listeners.has(key)) {
      this.listeners.set(key, new Set());
    }
    this.listeners.get(key)!.add(callback);
    return () => this.listeners.get(key)?.delete(callback);
  }

  private evaluate(flag: FeatureFlag): unknown {
    if (!flag.enabled) return flag.defaultValue;

    for (const rule of flag.rules) {
      if (this.matchesConditions(rule.conditions)) {
        if (rule.percentage !== undefined) {
          const hash = this.hashUser(flag.key, this.userContext.userId);
          if (hash <= rule.percentage) {
            return rule.value;
          }
          continue;
        }
        return rule.value;
      }
    }

    return flag.defaultValue;
  }

  private matchesConditions(conditions: Condition[]): boolean {
    return conditions.every((condition) => {
      const userValue = this.getUserAttribute(condition.attribute);
      return this.evaluateCondition(userValue, condition.operator, condition.value);
    });
  }

  private evaluateCondition(userValue: unknown, operator: string, targetValue: unknown): boolean {
    switch (operator) {
      case 'eq':
        return userValue === targetValue;
      case 'neq':
        return userValue !== targetValue;
      case 'gt':
        return Number(userValue) > Number(targetValue);
      case 'lt':
        return Number(userValue) < Number(targetValue);
      case 'contains':
        return String(userValue).includes(String(targetValue));
      case 'in':
        return Array.isArray(targetValue) && targetValue.includes(userValue);
      case 'regex':
        return new RegExp(String(targetValue)).test(String(userValue));
      default:
        return false;
    }
  }

  private getUserAttribute(attribute: string): unknown {
    if (attribute in this.userContext) {
      return (this.userContext as Record<string, unknown>)[attribute];
    }
    return this.userContext.customAttributes?.[attribute];
  }

  private hashUser(flagKey: string, userId: string): number {
    const str = `${flagKey}:${userId}`;
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = (hash << 5) - hash + char;
      hash = hash & hash;
    }
    return Math.abs(hash) % 100;
  }

  private async fetchFlags(): Promise<void> {
    try {
      const response = await fetch(`${this.serverUrl}/flags`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ context: this.userContext }),
      });
      const data = await response.json();
      this.updateFlags(data.flags);
    } catch {
      this.loadFromCache();
    }
  }

  private startRealtimeSync(): void {
    this.eventSource = new EventSource(`${this.serverUrl}/stream?userId=${this.userContext.userId}`);

    this.eventSource.addEventListener('flag-update', (event) => {
      const update = JSON.parse(event.data);
      this.flags.set(update.key, update);
      this.persistToCache();
      this.notifyListeners(update.key);
    });

    this.eventSource.onerror = () => {
      setTimeout(() => this.startRealtimeSync(), 5000);
    };
  }

  private updateFlags(flags: FeatureFlag[]): void {
    for (const flag of flags) {
      this.flags.set(flag.key, flag);
    }
    this.persistToCache();
  }

  private persistToCache(): void {
    const data = Object.fromEntries(this.flags);
    localStorage.setItem('feature-flags', JSON.stringify(data));
  }

  private loadFromCache(): void {
    const cached = localStorage.getItem('feature-flags');
    if (cached) {
      const data = JSON.parse(cached) as Record<string, FeatureFlag>;
      for (const [key, flag] of Object.entries(data)) {
        this.flags.set(key, flag);
      }
    }
  }

  private notifyListeners(key: string): void {
    const callbacks = this.listeners.get(key);
    if (callbacks) {
      const value = this.evaluate(this.flags.get(key)!);
      callbacks.forEach((cb) => cb(value));
    }
  }

  destroy(): void {
    this.eventSource?.close();
    this.listeners.clear();
  }
}
React Hook 集成
import { createContext, useContext, useState, useEffect, type ReactNode } from 'react'

const FeatureFlagContext = createContext<FeatureFlagClient | null>(null)

function FeatureFlagProvider({
  client,
  children,
}: {
  client: FeatureFlagClient
  children: ReactNode
}) {
  return (
    <FeatureFlagContext.Provider value={client}>
      {children}
    </FeatureFlagContext.Provider>
  )
}

function useFeatureFlag(key: string): boolean {
  const client = useContext(FeatureFlagContext)
  const [enabled, setEnabled] = useState(() => client?.isEnabled(key) ?? false)

  useEffect(() => {
    if (!client) return
    setEnabled(client.isEnabled(key))
    return client.onChange(key, (value) => setEnabled(value as boolean))
  }, [client, key])

  return enabled
}

function useFeatureValue<T>(key: string, defaultValue: T): T {
  const client = useContext(FeatureFlagContext)
  const [value, setValue] = useState(() => client?.getValue(key, defaultValue) ?? defaultValue)

  useEffect(() => {
    if (!client) return
    setValue(client.getValue(key, defaultValue))
    return client.onChange(key, (v) => setValue(v as T))
  }, [client, key, defaultValue])

  return value
}

function FeatureGate({
  flag,
  children,
  fallback = null,
}: {
  flag: string
  children: ReactNode
  fallback?: ReactNode
}) {
  const enabled = useFeatureFlag(flag)
  return enabled ? children : fallback
}

interface Experiment {
  key: string;
  variants: Variant[];
  allocation: number;
  metrics: ExperimentMetric[];
  startDate: string;
  endDate: string | null;
}

interface Variant {
  key: string;
  weight: number;
  payload: Record<string, unknown>;
}

interface ExperimentMetric {
  name: string;
  type: 'conversion' | 'continuous' | 'count';
  minimumDetectableEffect: number;
}

class ExperimentClient {
  private flagClient: FeatureFlagClient;
  private analytics: AnalyticsClient;

  constructor(flagClient: FeatureFlagClient, analytics: AnalyticsClient) {
    this.flagClient = flagClient;
    this.analytics = analytics;
  }

  getVariant(experimentKey: string): Variant | null {
    const value = this.flagClient.getValue<Variant | null>(experimentKey, null);
    if (value) {
      this.analytics.track('experiment_exposure', {
        experiment: experimentKey,
        variant: value.key,
      });
    }
    return value;
  }

  trackConversion(experimentKey: string, metricName: string, value?: number): void {
    const variant = this.flagClient.getValue<Variant | null>(experimentKey, null);
    if (!variant) return;

    this.analytics.track('experiment_conversion', {
      experiment: experimentKey,
      variant: variant.key,
      metric: metricName,
      value: value ?? 1,
    });
  }
}

function useExperiment(experimentKey: string) {
  const client = useContext(ExperimentContext)
  const [variant, setVariant] = useState<Variant | null>(null)

  useEffect(() => {
    const v = client?.getVariant(experimentKey) ?? null
    setVariant(v)
  }, [client, experimentKey])

  return {
    variant,
    isControl: variant?.key === 'control',
    isTreatment: variant?.key !== 'control' && variant !== null,
    payload: variant?.payload ?? {},
  }
}

动态配置管理
interface RemoteConfig {
  key: string;
  value: unknown;
  type: 'string' | 'number' | 'boolean' | 'json';
  platform: ('web' | 'ios' | 'android' | 'desktop')[];
  minVersion?: string;
  maxVersion?: string;
}

class RemoteConfigClient {
  private configs = new Map<string, RemoteConfig>();
  private platform: string;
  private appVersion: string;

  constructor(platform: string, appVersion: string) {
    this.platform = platform;
    this.appVersion = appVersion;
  }

  getString(key: string, defaultValue: string): string {
    return this.get(key, defaultValue) as string;
  }

  getNumber(key: string, defaultValue: number): number {
    return this.get(key, defaultValue) as number;
  }

  getBoolean(key: string, defaultValue: boolean): boolean {
    return this.get(key, defaultValue) as boolean;
  }

  getJSON<T>(key: string, defaultValue: T): T {
    return this.get(key, defaultValue) as T;
  }

  private get(key: string, defaultValue: unknown): unknown {
    const config = this.configs.get(key);
    if (!config) return defaultValue;
    if (!config.platform.includes(this.platform as any)) return defaultValue;
    if (config.minVersion && this.appVersion < config.minVersion) return defaultValue;
    if (config.maxVersion && this.appVersion > config.maxVersion) return defaultValue;
    return config.value;
  }
}

小结
功能开关是现代软件发布的核心基础设施，它将代码部署与功能发布解耦，使团队能够以更低风险、更高频率发布功能。跨平台场景需要考虑多平台 SDK 支持、客户端缓存策略和离线降级。渐进式发布通过分阶段灰度控制风险，A/B 测试提供数据驱动的决策依据。远程配置扩展了开关的能力边界，支持动态调整应用行为。工具选型需权衡功能丰富度、成本和自托管需求。


核心要点回顾：


    

  • 功能开关分为发布、实验、运维、权限和平台五种类型
    • 

  • 客户端 SDK 需支持规则评估、缓存、实时同步和离线降级
    • 

  • 渐进式发布通过分阶段灰度（1% → 10% → 50% → 100%）控制风险
    • 

  • A/B 测试基于哈希分桶实现确定性的用户分配
    • 

  • 开关生命周期管理是避免技术债务的关键
    • 

  • 工具选型取决于功能需求、预算和自托管要求
