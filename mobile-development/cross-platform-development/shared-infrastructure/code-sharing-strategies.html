跨平台代码共享策略

核心概念

跨平台开发的核心价值在于代码复用——一套业务逻辑服务多个平台。然而，不同平台的 UI 框架、API 能力和运行时环境存在显著差异，如何在最大化共享的同时保持各平台的原生体验，是架构设计的关键挑战。本文深入探讨 Monorepo 架构、共享库设计、平台特定代码隔离、条件编译策略和跨平台测试方案。

API 客户端共享
type RequestInterceptor = (config: RequestConfig) => RequestConfig | Promise<RequestConfig>;
type ResponseInterceptor = (response: ApiResponse<unknown>) => ApiResponse<unknown>;

interface RequestConfig {
  url: string;
  method: 'GET' | 'POST' | 'PUT' | 'DELETE' | 'PATCH';
  headers?: Record<string, string>;
  body?: unknown;
  params?: Record<string, string>;
  timeout?: number;
}

interface HttpAdapter {
  request<T>(config: RequestConfig): Promise<ApiResponse<T>>;
}

class ApiClient {
  private adapter: HttpAdapter;
  private baseURL: string;
  private requestInterceptors: RequestInterceptor[] = [];
  private responseInterceptors: ResponseInterceptor[] = [];

  constructor(adapter: HttpAdapter, baseURL: string) {
    this.adapter = adapter;
    this.baseURL = baseURL;
  }

  addRequestInterceptor(interceptor: RequestInterceptor) {
    this.requestInterceptors.push(interceptor);
  }

  addResponseInterceptor(interceptor: ResponseInterceptor) {
    this.responseInterceptors.push(interceptor);
  }

  async request<T>(config: RequestConfig): Promise<ApiResponse<T>> {
    let finalConfig = { ...config, url: `${this.baseURL}${config.url}` };

    for (const interceptor of this.requestInterceptors) {
      finalConfig = await interceptor(finalConfig);
    }

    let response = await this.adapter.request<T>(finalConfig);

    for (const interceptor of this.responseInterceptors) {
      response = interceptor(response) as ApiResponse<T>;
    }

    return response;
  }

  get<T>(url: string, params?: Record<string, string>) {
    return this.request<T>({ url, method: 'GET', params });
  }

  post<T>(url: string, body?: unknown) {
    return this.request<T>({ url, method: 'POST', body });
  }

  put<T>(url: string, body?: unknown) {
    return this.request<T>({ url, method: 'PUT', body });
  }

  delete<T>(url: string) {
    return this.request<T>({ url, method: 'DELETE' });
  }
}
平台适配器实现
class FetchAdapter implements HttpAdapter {
  async request<T>(config: RequestConfig): Promise<ApiResponse<T>> {
    const url = new URL(config.url);
    if (config.params) {
      Object.entries(config.params).forEach(([key, value]) => {
        url.searchParams.set(key, value);
      });
    }

    const controller = new AbortController();
    const timeoutId = config.timeout ? setTimeout(() => controller.abort(), config.timeout) : null;

    try {
      const response = await fetch(url.toString(), {
        method: config.method,
        headers: {
          'Content-Type': 'application/json',
          ...config.headers,
        },
        body: config.body ? JSON.stringify(config.body) : undefined,
        signal: controller.signal,
      });

      return await response.json();
    } finally {
      if (timeoutId) clearTimeout(timeoutId);
    }
  }
}

class TauriAdapter implements HttpAdapter {
  async request<T>(config: RequestConfig): Promise<ApiResponse<T>> {
    const { invoke } = await import('@tauri-apps/api/core');
    return invoke<ApiResponse<T>>('http_request', {
      url: config.url,
      method: config.method,
      headers: config.headers,
      body: config.body,
    });
  }
}

interface StorageAdapter {
  get(key: string): Promise<string | null>;
  set(key: string, value: string): Promise<void>;
  remove(key: string): Promise<void>;
  clear(): Promise<void>;
  keys(): Promise<string[]>;
}

class WebStorage implements StorageAdapter {
  async get(key: string) {
    return localStorage.getItem(key);
  }

  async set(key: string, value: string) {
    localStorage.setItem(key, value);
  }

  async remove(key: string) {
    localStorage.removeItem(key);
  }

  async clear() {
    localStorage.clear();
  }

  async keys() {
    return Object.keys(localStorage);
  }
}

class NativeStorage implements StorageAdapter {
  private store: Map<string, string>;

  constructor() {
    this.store = new Map();
  }

  async get(key: string) {
    const AsyncStorage = (await import('@react-native-async-storage/async-storage')).default;
    return AsyncStorage.getItem(key);
  }

  async set(key: string, value: string) {
    const AsyncStorage = (await import('@react-native-async-storage/async-storage')).default;
    return AsyncStorage.setItem(key, value);
  }

  async remove(key: string) {
    const AsyncStorage = (await import('@react-native-async-storage/async-storage')).default;
    return AsyncStorage.removeItem(key);
  }

  async clear() {
    const AsyncStorage = (await import('@react-native-async-storage/async-storage')).default;
    return AsyncStorage.clear();
  }

  async keys() {
    const AsyncStorage = (await import('@react-native-async-storage/async-storage')).default;
    return AsyncStorage.getAllKeys() as Promise<string[]>;
  }
}
依赖注入容器
type Constructor<T> = new (...args: any[]) => T;

class DIContainer {
  private registry = new Map<string, Constructor<any>>();
  private instances = new Map<string, any>();

  register<T>(token: string, impl: Constructor<T>): void {
    this.registry.set(token, impl);
  }

  resolve<T>(token: string): T {
    if (this.instances.has(token)) {
      return this.instances.get(token) as T;
    }

    const Impl = this.registry.get(token);
    if (!Impl) {
      throw new Error(`No implementation registered for ${token}`);
    }

    const instance = new Impl() as T;
    this.instances.set(token, instance);
    return instance;
  }
}

const container = new DIContainer();

function setupForWeb() {
  container.register('storage', WebStorage);
  container.register('http', FetchAdapter as any);
}

function setupForNative() {
  container.register('storage', NativeStorage);
  container.register('http', FetchAdapter as any);
}

function setupForDesktop() {
  container.register('storage', WebStorage);
  container.register('http', TauriAdapter as any);
}

import { resolve } from 'node:path';
import { defineConfig } from 'vite';

function platformResolve(platform: string) {
  return {
    name: 'platform-resolve',
    resolveId(source: string) {
      if (source.includes('.platform')) {
        const platformFile = source.replace('.platform', `.${platform}`);
        return platformFile;
      }
      return null;
    },
  };
}

const viteConfig = defineConfig({
  plugins: [platformResolve('web')],
  resolve: {
    alias: {
      '@shared': resolve(__dirname, '../shared/src'),
      '@core': resolve(__dirname, '../core/src'),
    },
  },
});

共享测试用例
import { describe, it, expect } from 'vitest';

function createUserTests(getStorage: () => StorageAdapter) {
  describe('UserService', () => {
    it('should save user preferences', async () => {
      const storage = getStorage();
      const prefs = { theme: 'dark', language: 'zh-CN' };

      await storage.set('user:prefs', JSON.stringify(prefs));
      const stored = await storage.get('user:prefs');

      expect(JSON.parse(stored!)).toEqual(prefs);
    });

    it('should handle missing keys', async () => {
      const storage = getStorage();
      const result = await storage.get('nonexistent');
      expect(result).toBeNull();
    });

    it('should clear all data', async () => {
      const storage = getStorage();
      await storage.set('key1', 'value1');
      await storage.set('key2', 'value2');
      await storage.clear();

      const keys = await storage.keys();
      expect(keys).toHaveLength(0);
    });
  });
}

小结
跨平台代码共享的关键在于分层架构——将代码按共享程度分为类型/工具（100%）、业务逻辑（85-95%）、UI 逻辑（70%）和平台特定（0-50%）四个层次。Monorepo 提供统一的代码管理和依赖解析，共享包通过 workspace 协议互相引用。平台差异通过适配器模式和依赖注入隔离，条件编译实现构建时的平台定制。测试策略遵循金字塔模型——单元测试 100% 共享，集成测试部分共享，E2E 测试平台特定。


核心要点回顾：


    

  • 分层架构按共享度组织代码：types > utils > business > UI > native
    • 

  • Monorepo + pnpm workspace 实现包级别的代码复用
    • 

  • 适配器模式隔离平台差异（Storage、HTTP、Navigation）
    • 

  • 依赖注入容器在运行时绑定平台实现
    • 

  • 条件编译通过环境变量和模块解析实现构建时定制
    • 

  • 共享测试用例确保跨平台行为一致性
