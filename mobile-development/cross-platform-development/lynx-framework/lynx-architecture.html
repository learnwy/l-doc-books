Lynx 架构

核心概念

Lynx 是字节跳动开源的高性能跨平台 UI 框架，支持 iOS、Android、Web 和 HarmonyOS 平台。与 React Native 的单线程 JS 执行模型不同，Lynx 采用了独特的双线程架构——主线程（Main Thread）负责 UI 渲染和交互响应，背景线程（Background Thread）负责 JS 业务逻辑执行。这种设计使得 JS 执行不会阻塞 UI 渲染，从而实现了更流畅的用户体验。

整体架构

双线程模型
Lynx 的双线程模型是其架构的核心创新。相比 React Native 旧架构中 JS 线程阻塞会直接影响 UI 响应，Lynx 的双线程架构从设计上就隔离了逻辑计算和 UI 渲染。

线程间通信
import { useMainThread, useBackground } from '@lynx-js/react';

function CounterComponent() {
  const [count, setCount] = useBackground(0);

  const handleTap = useMainThread(() => {
    'main thread';
    setCount(prev => prev + 1);
  });

  return (
    <view className="counter" bindtap={handleTap}>
      <text className="count">{count}</text>
    </view>
  );
}
渲染管线
Lynx 的渲染管线从模板解析开始，经过布局计算，最终将渲染指令发送到各平台的原生渲染器。


PipelineOptions
PipelineOptions 是 Lynx 提供的渲染策略控制机制，允许开发者根据场景选择不同的渲染策略。


interface PipelineOptions {
  renderMode: 'sync' | 'async' | 'batch';
  enableIncrementalLayout: boolean;
  maxFrameBudget: number;
  priorityLevel: 'high' | 'normal' | 'low';
  enableRecycling: boolean;
}

const highPriorityPipeline: PipelineOptions = {
  renderMode: 'sync',
  enableIncrementalLayout: false,
  maxFrameBudget: 16,
  priorityLevel: 'high',
  enableRecycling: true,
};

const normalPipeline: PipelineOptions = {
  renderMode: 'async',
  enableIncrementalLayout: true,
  maxFrameBudget: 32,
  priorityLevel: 'normal',
  enableRecycling: true,

原生扩展机制
Lynx 支持通过原生扩展来增强框架能力，包括自定义组件（Custom Element）和原生模块（Native Module）。


自定义原生组件
class CustomVideoElement(context: Context) : LynxUI<VideoView>(context) {
    private var videoView: VideoView? = null

    override fun createView(context: Context): VideoView {
        videoView = VideoView(context)
        return videoView!!
    }

    override fun updateAttribute(key: String, value: Any?) {
        when (key) {
            "src" -> videoView?.setVideoURI(Uri.parse(value as? String))
            "autoplay" -> {
                if (value == true) videoView?.start()
            }
            "controls" -> {
                val controller = MediaController(context)
                controller.setAnchorView(videoView)
                videoView?.setMediaController(controller)
            }
        }
    }

    override fun onDetach() {
        videoView?.stopPlayback()
        super.onDetach()
    }
}
class CustomVideoElement: LynxUI<VideoPlayerView> {
    private var player: AVPlayer?

    override func createView() -> VideoPlayerView {
        let view = VideoPlayerView()
        return view
    }

    override func updateAttribute(_ key: String, value: Any?) {
        switch key {
        case "src":
            guard let urlString = value as? String,
                  let url = URL(string: urlString) else { return }
            player = AVPlayer(url: url)
            view.playerLayer.player = player
        case "autoplay":
            if value as? Bool == true {
                player?.play()
            }
        default:
            break
        }
    }

    override func onDetach() {
        player?.pause()
        player = nil
        super.onDetach()
    }
}
原生模块
import { NativeModules } from '@lynx-js/core';

interface ShareModule {
  share(params: {
    title: string;
    content: string;
    imageUrl?: string;
    platform: 'wechat' | 'weibo' | 'system';
  }): Promise<{ success: boolean; platform: string }>;

  isInstalled(platform: string): Promise<boolean>;
}

const shareModule = NativeModules.get<ShareModule>('ShareModule');

async function handleShare(data: ShareData) {
  const installed = await shareModule.isInstalled('wechat');
  if (!installed) {
    return shareModule.share({ ...data, platform: 'system' });
  }
  return shareModule.share({ ...data, platform: 'wechat' });
}

数据流架构
import { useState, useCallback, useMemo } from '@lynx-js/react';

interface FeedItem {
  id: string;
  title: string;
  cover: string;
  author: string;
  likes: number;
}

function FeedPage() {
  const [items, setItems] = useState<FeedItem[]>([]);
  const [loading, setLoading] = useState(false);
  const [page, setPage] = useState(1);

  const loadMore = useCallback(async () => {
    if (loading) return;
    setLoading(true);

    const response = await fetch(`/api/feed?page=${page}&size=20`);
    const data = await response.json();

    setItems(prev => [...prev, ...data.items]);
    setPage(prev => prev + 1);
    setLoading(false);
  }, [loading, page]);

  const sortedItems = useMemo(
    () => items.sort((a, b) => b.likes - a.likes),
    [items],
  );

  return (
    <view className="feed">
      <list
        className="feed-list"
        bindscrolltolower={loadMore}
      >
        {sortedItems.map(item => (
          <cell key={item.id} className="feed-item">
            <image src={item.cover} className="cover" />
            <view className="info">
              <text className="title">{item.title}</text>
              <text className="author">{item.author}</text>
            </view>
          </cell>
        ))}
      </list>
      {loading && <text className="loading">加载中...</text>}
    </view>
  );
}
小结
Lynx 的架构设计围绕"高性能"和"开发体验"两个核心目标展开。双线程模型确保了 UI 渲染的流畅性，PipelineOptions 提供了灵活的渲染策略控制，原生扩展机制保证了平台能力的完整覆盖。相比 React Native，Lynx 的双线程架构在首屏性能和交互流畅度上更具优势；相比 Flutter，Lynx 使用原生组件渲染，平台适配性更好且支持热更新。


核心要点回顾：


    

  • Lynx 采用双线程架构：Main Thread 负责 UI，Background Thread 负责 JS 逻辑
    • 

  • 渲染管线从模板解析到原生组件操作，通过 PipelineOptions 控制渲染策略
    • 

  • 原生扩展机制支持自定义组件和原生模块，覆盖平台特有能力
    • 

  • 相比 RN 单线程模型，双线程架构在 JS 重计算场景下 UI 不会卡顿
    • 

  • 支持 React 和 Vue 作为前端开发框架，降低了学习成本
    • 

  • 包体积增量小（~3MB），支持热更新，适合动态化场景
