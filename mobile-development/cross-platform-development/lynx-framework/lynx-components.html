Lynx 组件

核心概念

Lynx 提供了一套丰富的内置组件体系，涵盖布局容器、文本展示、图片加载、列表滚动、表单输入等常见 UI 场景。这些组件在各平台上映射为对应的原生控件，保证了渲染性能和平台体验的一致性。同时，Lynx 支持自定义组件机制，允许开发者封装可复用的 UI 单元，并通过 props 和事件实现组件间通信。

组件体系架构

容器组件 - view
view 是最基础的容器组件，等价于 Web 中的 div，在 iOS 上映射为 UIView，在 Android 上映射为 ViewGroup。

图片组件 - image
image 组件负责图片加载和展示，支持网络图片、本地图片和 base64 格式。


import { useState, useCallback } from '@lynx-js/react';

interface ImageProps {
  src: string;
  fallback?: string;
  aspectRatio?: number;
}

function SmartImage({ src, fallback, aspectRatio = 1 }: ImageProps) {
  const [status, setStatus] = useState<'loading' | 'loaded' | 'error'>('loading');

  const handleLoad = useCallback(() => {
    setStatus('loaded');
  }, []);

  const handleError = useCallback(() => {
    setStatus('error');
  }, []);

  return (
    <view className="image-wrapper" style={{ aspectRatio }}>
      {status === 'loading' && (
        <view className="placeholder">
          <text className="placeholder-text">加载中</text>
        </view>
      )}

      <image
        src={status === 'error' && fallback ? fallback : src}
        className={`image ${status === 'loaded' ? 'visible' : 'hidden'}`}
        mode="aspectFill"
        lazy-load={true}
        bindload={handleLoad}
        binderror={handleError}
      />

      {status === 'error' && !fallback && (
        <view className="error-state">
          <text className="error-text">加载失败</text>
        </view>
      )}
    </view>
  );
}

function ImageGallery() {
  const images = [
    { id: '1', url: 'https://example.com/1.jpg', ratio: 1.5 },
    { id: '2', url: 'https://example.com/2.jpg', ratio: 1 },
    { id: '3', url: 'https://example.com/3.jpg', ratio: 0.75 },
  ];

  return (
    <view className="gallery">
      {images.map((img) => (
        <SmartImage key={img.id} src={img.url} aspectRatio={img.ratio} fallback="https://example.com/placeholder.jpg" />
      ))}
    </view>
  );
}
列表组件 - list & cell
list 和 cell 是 Lynx 中性能最优的列表渲染方案。list 组件内置了虚拟化机制，只渲染可见区域内的 cell，实现了大数据量下的流畅滚动。


import { useState, useCallback } from '@lynx-js/react';

interface Product {
  id: string;
  name: string;
  price: number;
  image: string;
  category: string;
}

function ProductList() {
  const [products, setProducts] = useState<Product[]>([]);
  const [refreshing, setRefreshing] = useState(false);

  const onRefresh = useCallback(async () => {
    setRefreshing(true);
    const data = await fetchProducts(1);
    setProducts(data);
    setRefreshing(false);
  }, []);

  const onLoadMore = useCallback(async () => {
    const nextPage = Math.ceil(products.length / 20) + 1;
    const data = await fetchProducts(nextPage);
    setProducts((prev) => [...prev, ...data]);
  }, [products.length]);

  return (
    <list
      className="product-list"
      scroll-y={true}
      bindrefresh={onRefresh}
      bindscrolltolower={onLoadMore}
      lower-threshold={200}
    >
      <header className="list-header">
        <text className="header-title">商品列表</text>
        <text className="header-count">{products.length} 件商品</text>
      </header>

      {products.map((product) => (
        <cell key={product.id} className="product-cell" reuse-id="product">
          <ProductCard product={product} />
        </cell>
      ))}

      <footer className="list-footer">
        <text className="footer-text">{refreshing ? '加载中...' : '上拉加载更多'}</text>
      </footer>
    </list>
  );
}

function ProductCard({ product }: { product: Product }) {
  return (
    <view className="product-card">
      <image src={product.image} className="product-image" mode="aspectFill" />
      <view className="product-info">
        <text className="product-name">{product.name}</text>
        <text className="product-category">{product.category}</text>
        <text className="product-price">¥{product.price.toFixed(2)}</text>
      </view>
    </view>
  );
}

表单组件
import { useState, useCallback } from '@lynx-js/react';

interface FormData {
  username: string;
  email: string;
  bio: string;
  notifications: boolean;
}

function ProfileForm() {
  const [form, setForm] = useState<FormData>({
    username: '',
    email: '',
    bio: '',
    notifications: true,
  });

  const updateField = useCallback(<K extends keyof FormData>(field: K, value: FormData[K]) => {
    setForm((prev) => ({ ...prev, [field]: value }));
  }, []);

  const handleSubmit = useCallback(async () => {
    if (!form.username || !form.email) return;
    await saveProfile(form);
  }, [form]);

  return (
    <view className="form">
      <view className="field">
        <text className="label">用户名</text>
        <input
          className="input"
          type="text"
          value={form.username}
          placeholder="请输入用户名"
          bindinput={(e) => updateField('username', e.detail.value)}
          maxlength={20}
        />
      </view>

      <view className="field">
        <text className="label">邮箱</text>
        <input
          className="input"
          type="text"
          value={form.email}
          placeholder="请输入邮箱"
          bindinput={(e) => updateField('email', e.detail.value)}
        />
      </view>

      <view className="field">
        <text className="label">个人简介</text>
        <textarea
          className="textarea"
          value={form.bio}
          placeholder="介绍一下自己"
          bindinput={(e) => updateField('bio', e.detail.value)}
          maxlength={200}
        />
      </view>

      <view className="field-row">
        <text className="label">接收通知</text>
        <switch checked={form.notifications} bindchange={(e) => updateField('notifications', e.detail.value)} />
      </view>

      <view className="submit-btn" bindtap={handleSubmit}>
        <text className="btn-text">保存</text>
      </view>
    </view>
  );
}
自定义组件通信
import { createContext, useContext, useState, useCallback } from '@lynx-js/react';

interface ThemeContext {
  theme: 'light' | 'dark';
  toggle: () => void;
}

const ThemeCtx = createContext<ThemeContext>({
  theme: 'light',
  toggle: () => {},
});

function ThemeProvider({ children }: { children: any }) {
  const [theme, setTheme] = useState<'light' | 'dark'>('light');

  const toggle = useCallback(() => {
    setTheme((prev) => (prev === 'light' ? 'dark' : 'light'));
  }, []);

  return <ThemeCtx.Provider value={{ theme, toggle }}>{children}</ThemeCtx.Provider>;
}

function ThemedButton({ label, onTap }: { label: string; onTap: () => void }) {
  const { theme } = useContext(ThemeCtx);

  return (
    <view className={`btn btn-${theme}`} bindtap={onTap}>
      <text className={`btn-text text-${theme}`}>{label}</text>
    </view>
  );
}

function ThemeToggle() {
  const { theme, toggle } = useContext(ThemeCtx);

  return (
    <view className="toggle-container" bindtap={toggle}>
      <text>{theme === 'light' ? '切换到深色模式' : '切换到浅色模式'}</text>
    </view>
  );
}
组件生命周期
import { useState, useEffect, useCallback, useRef } from '@lynx-js/react';

function LifecycleDemo({ userId }: { userId: string }) {
  const [data, setData] = useState<UserProfile | null>(null);
  const timerRef = useRef<number>(0);

  useEffect(() => {
    let cancelled = false;

    async function loadProfile() {
      const profile = await fetchUserProfile(userId);
      if (!cancelled) {
        setData(profile);
      }
    }

    loadProfile();

    return () => {
      cancelled = true;
    };
  }, [userId]);

  useEffect(() => {
    timerRef.current = setInterval(() => {
      refreshOnlineStatus(userId);
    }, 30000) as unknown as number;

    return () => {
      clearInterval(timerRef.current);
    };
  }, [userId]);

  if (!data) {
    return (
      <view className="loading">
        <text>加载中...</text>
      </view>
    );
  }

  return (
    <view className="profile">
      <image src={data.avatar} className="avatar" />
      <text className="name">{data.name}</text>
      <text className="bio">{data.bio}</text>
    </view>
  );
}

小结
Lynx 的组件体系覆盖了移动端开发的绝大部分 UI 需求。内置组件映射到原生控件保证了性能，自定义组件机制（函数组件 + Hooks）提供了良好的代码复用能力，Context 和事件系统解决了组件间通信问题。掌握 list + cell 的虚拟化列表和组件生命周期管理是开发高质量 Lynx 应用的关键。


核心要点回顾：


    

  • view 是基础容器，text 是唯一能展示文字的组件
    • 

  • list + cell 实现虚拟化列表渲染，reuse-id 控制 cell 复用
    • 

  • scroll-view 适合少量内容滚动，不具备虚拟化能力
    • 

  • 自定义组件基于函数组件 + Hooks 模式，与 React 生态一致
    • 

  • Context API 实现跨层级数据共享，事件回调实现子到父通信
    • 

  • 组件生命周期通过 useEffect 管理，需注意清理副作用
    •
