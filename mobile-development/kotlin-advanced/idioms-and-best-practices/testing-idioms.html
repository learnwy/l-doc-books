测试惯用法

Kotlin 的测试生态提供了丰富的工具和惯用模式。MockK 提供原生 Kotlin 的 Mock 支持，Kotest 提供表达力强的断言和测试风格，kotlinx-coroutines-test 处理协程测试，而基于属性的测试则确保代码在各种输入下的正确性。

MockK 基础

MockK 是专为 Kotlin 设计的 Mock 框架，原生支持协程、扩展函数和对象 mock：

import io.mockk.*

interface UserRepository {
    suspend fun findById(id: Long): User?
    suspend fun save(user: User): User
    fun count(): Int
}

class UserService(private val repository: UserRepository) {
    suspend fun getUser(id: Long): User {
        return repository.findById(id)
            ?: throw NotFoundException("用户 $id 不存在")
    }
}

class UserServiceTest {
    private val repository = mockk<UserRepository>()
    private val service = UserService(repository)

    @Test
    fun `应返回已存在的用户`() = runTest {
        val expectedUser = User(1, "张三", "zhangsan@example.com")

        coEvery { repository.findById(1) } returns expectedUser

        val user = service.getUser(1)

        assertEquals("张三", user.name)
        coVerify { repository.findById(1) }
    }
}

every / coEvery

// 同步方法
every { repository.count() } returns 42

// 挂起方法
coEvery { repository.findById(any()) } returns User(1, "测试用户", "test@test.com")

// 根据参数返回不同值
every { repository.count() } returnsMany listOf(1, 2, 3)

// 抛出异常
coEvery { repository.save(any()) } throws DatabaseException("连接失败")

// 使用答案
coEvery { repository.save(any()) } coAnswers {
    firstArg<User>().copy(id = 100)
}

verify / coVerify

// 验证调用次数
coVerify(exactly = 1) { repository.findById(1) }
coVerify(atLeast = 2) { repository.save(any()) }

// 验证未调用
coVerify(exactly = 0) { repository.save(any()) }

// 验证调用顺序
coVerifyOrder {
    repository.findById(1)
    repository.save(any())
}

slot 参数捕获

@Test
fun `应正确保存用户`() = runTest {
    val slot = slot<User>()
    coEvery { repository.save(capture(slot)) } answers { slot.captured }

    service.createUser("李四", "lisi@example.com")

    assertEquals("李四", slot.captured.name)
    assertEquals("lisi@example.com", slot.captured.email)
}


Kotest 断言
Kotest 提供了丰富且可读性强的匹配器：


import io.kotest.matchers.shouldBe
import io.kotest.matchers.shouldNotBe
import io.kotest.matchers.string.*
import io.kotest.matchers.collections.*
import io.kotest.matchers.maps.*

@Test
fun `Kotest 断言示例`() {
    // 基础断言
    42 shouldBe 42
    "hello" shouldNotBe null

    // 字符串断言
    "Kotlin" shouldStartWith "Kot"
    "Kotlin" shouldContain "lin"
    "Kotlin" shouldHaveLength 6

    // 集合断言
    listOf(1, 2, 3) shouldContain 2
    listOf(1, 2, 3) shouldHaveSize 3
    listOf(1, 2, 3) shouldBeSorted()
    listOf(1, 2, 3) shouldContainAll listOf(1, 3)

    // Map 断言
    mapOf("a" to 1, "b" to 2) shouldContainKey "a"
    mapOf("a" to 1) shouldContainExactly mapOf("a" to 1)
}

协程测试
使用 kotlinx-coroutines-test 提供的 runTest 和 TestDispatcher：


import kotlinx.coroutines.test.*

class CacheServiceTest {
    @Test
    fun `缓存过期后应重新加载`() = runTest {
        val testScheduler = TestCoroutineScheduler()
        val testDispatcher = StandardTestDispatcher(testScheduler)

        val service = CacheService(
            dispatcher = testDispatcher,
            ttlMillis = 5000
        )

        val first = service.getData()
        assertEquals("数据v1", first)

        // 推进时间 6 秒
        testScheduler.advanceTimeBy(6000)

        val second = service.getData()
        assertEquals("数据v2", second) // 缓存已过期，重新加载
    }
}

基于属性的测试
基于属性的测试（Property-Based Testing）自动生成大量随机输入来验证代码属性：


import io.kotest.property.forAll
import io.kotest.property.Arb
import io.kotest.property.arbitrary.*

class StringUtilsTest : FunSpec({
    test("反转两次应得到原字符串") {
        forAll(Arb.string()) { s ->
            s.reversed().reversed() == s
        }
    }

    test("列表排序后应有序") {
        forAll(Arb.list(Arb.int())) { list ->
            val sorted = list.sorted()
            sorted.zipWithNext().all { (a, b) -> a <= b }
        }
    }

    test("用户年龄应在有效范围内") {
        val userArb = Arb.bind(
            Arb.string(minSize = 1, maxSize = 50),
            Arb.int(0..150)
        ) { name, age -> User(name, age) }

        forAll(userArb) { user ->
            user.age in 0..150
        }
    }
})

测试 Fixtures
结构化的测试数据管理：


object TestFixtures {
    fun createUser(
        id: Long = 1,
        name: String = "测试用户",
        email: String = "test@example.com",
        roles: List<String> = listOf("user")
    ) = User(id, name, email, roles)

    fun createOrder(
        id: Long = 1,
        userId: Long = 1,
        items: List<OrderItem> = listOf(createOrderItem()),
        status: OrderStatus = OrderStatus.PENDING
    ) = Order(id, userId, items, status)

    fun createOrderItem(
        product: String = "商品A",
        price: Double = 99.0,
        quantity: Int = 1
    ) = OrderItem(product, price, quantity)
}

class OrderServiceTest {
    @Test
    fun `应计算订单总价`() {
        val order = TestFixtures.createOrder(
            items = listOf(
                TestFixtures.createOrderItem(price = 100.0, quantity = 2),
                TestFixtures.createOrderItem(price = 50.0, quantity = 3)
            )
        )

        val total = orderService.calculateTotal(order)
        total shouldBe 350.0
    }
}


提示： 使用带默认参数的工厂函数创建测试数据，只覆盖测试关注的字段，保持测试简洁和意图清晰。

小结
    

  • MockK 是 Kotlin 原生的 Mock 框架，通过 every/coEvery 定义行为，verify/coVerify 验证调用，slot 捕获参数
    • 

  • Kotest 匹配器提供丰富的 shouldBe、shouldContain 等断言，覆盖字符串、集合、数字等类型
    • 

  • 协程测试使用 runTest + TestDispatcher 控制虚拟时间，Turbine 库测试 Flow 发射序列
    • 

  • 基于属性的测试通过 forAll + Arb 生成器自动生成大量随机输入，发现边界条件和不变量
    • 

  • 测试 Fixtures使用带默认参数的工厂函数管理测试数据，保持测试简洁
    • 

  • 综合使用单元测试、属性测试和集成测试构建多层次的测试策略
