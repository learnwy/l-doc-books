Part I · iOSiOS 开发 简介SwiftUI 进阶 简介自定义组件

自定义组件

概述

SwiftUI 提供了强大的组件自定义体系：ViewModifier 封装可复用的视图修饰逻辑，ButtonStyle / ToggleStyle 定义控件的外观与交互，@ViewBuilder 支持声明式的视图组合，PreferenceKey 实现子视图向父视图的反向数据传递。掌握这些机制，可以构建高度可复用、风格一致的组件库。

基础用法
ViewModifier
ViewModifier 协议将一组修饰符打包成可复用单元。定义一个 modifier，然后通过 .modifier() 或自定义扩展方法应用到任意视图。


struct CardModifier: ViewModifier {
    var cornerRadius: CGFloat = 12
    var shadowRadius: CGFloat = 4

    func body(content: Content) -> some View {
        content
            .padding()
            .background(.background, in: RoundedRectangle(cornerRadius: cornerRadius))
            .shadow(color: .black.opacity(0.1), radius: shadowRadius, y: 2)
    }
}

extension View {
    func cardStyle(cornerRadius: CGFloat = 12, shadowRadius: CGFloat = 4) -> some View {
        modifier(CardModifier(cornerRadius: cornerRadius, shadowRadius: shadowRadius))
    }
}

struct CardDemoView: View {
    var body: some View {
        VStack(spacing: 16) {
            Text("这是一张卡片")
                .cardStyle()

            Label("带图标的卡片", systemImage: "star.fill")
                .cardStyle(cornerRadius: 20, shadowRadius: 8)
        }
        .padding()
    }
}
ButtonStyle
ButtonStyle 协议让你完全控制按钮的外观和按压反馈。通过 configuration.isPressed 感知按压状态：


struct ScaleButtonStyle: ButtonStyle {
    func makeBody(configuration: Configuration) -> some View {
        configuration.label
            .padding(.horizontal, 24)
            .padding(.vertical, 12)
            .background(.blue.gradient, in: Capsule())
            .foregroundStyle(.white)
            .scaleEffect(configuration.isPressed ? 0.95 : 1.0)
            .opacity(configuration.isPressed ? 0.9 : 1.0)
            .animation(.spring(duration: 0.2), value: configuration.isPressed)
    }
}

extension ButtonStyle where Self == ScaleButtonStyle {
    static var scale: ScaleButtonStyle { ScaleButtonStyle() }
}

struct ButtonStyleDemoView: View {
    var body: some View {
        VStack(spacing: 20) {
            Button("提交订单") {}
                .buttonStyle(.scale)

            Button {} label: {
                Label("添加到购物车", systemImage: "cart.badge.plus")
            }
            .buttonStyle(.scale)
        }
    }
}
ToggleStyle
ToggleStyle 自定义开关的外观，可以实现复选框、卡片切换等非标准样式：


struct CheckboxToggleStyle: ToggleStyle {
    func makeBody(configuration: Configuration) -> some View {
        HStack {
            Image(systemName: configuration.isOn ? "checkmark.square.fill" : "square")
                .font(.title2)
                .foregroundStyle(configuration.isOn ? .blue : .gray)
                .onTapGesture {
                    withAnimation(.spring(duration: 0.2)) {
                        configuration.isOn.toggle()
                    }
                }
            configuration.label
        }
    }
}

extension ToggleStyle where Self == CheckboxToggleStyle {
    static var checkbox: CheckboxToggleStyle { CheckboxToggleStyle() }
}

struct ToggleStyleDemoView: View {
    @State private var agreeTerms = false
    @State private var receiveNewsletter = true

    var body: some View {
        VStack(alignment: .leading, spacing: 16) {
            Toggle("同意用户协议", isOn: $agreeTerms)
                .toggleStyle(.checkbox)

            Toggle("接收新闻通知", isOn: $receiveNewsletter)
                .toggleStyle(.checkbox)
        }
        .padding()
    }
}
进阶用法
@ViewBuilder 构建复合组件
@ViewBuilder 让自定义组件可以像 SwiftUI 原生容器一样接收声明式的子视图内容：


struct SectionCard<Header: View, Content: View>: View {
    let header: Header
    let content: Content

    init(
        @ViewBuilder header: () -> Header,
        @ViewBuilder content: () -> Content
    ) {
        self.header = header()
        self.content = content()
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            header
                .font(.headline)
                .foregroundStyle(.secondary)

            content
        }
        .padding()
        .frame(maxWidth: .infinity, alignment: .leading)
        .background(.background, in: RoundedRectangle(cornerRadius: 12))
        .shadow(color: .black.opacity(0.05), radius: 4, y: 2)
    }
}

struct SectionCardDemoView: View {
    var body: some View {
        ScrollView {
            VStack(spacing: 16) {
                SectionCard {
                    Label("个人信息", systemImage: "person.fill")
                } content: {
                    LabeledContent("姓名", value: "张三")
                    LabeledContent("邮箱", value: "zhang@example.com")
                    LabeledContent("手机", value: "138-0000-0001")
                }

                SectionCard {
                    Label("偏好设置", systemImage: "gearshape.fill")
                } content: {
                    Toggle("深色模式", isOn: .constant(true))
                    Toggle("推送通知", isOn: .constant(false))
                }
            }
            .padding()
        }
        .background(Color(.systemGroupedBackground))
    }
}
PreferenceKey 子视图向父视图传值
PreferenceKey 允许子视图将数据"冒泡"到父视图，常用于收集子视图的几何信息或聚合导航标题等场景。


struct WidthPreferenceKey: PreferenceKey {
    static var defaultValue: CGFloat = 0
    static func reduce(value: inout CGFloat, nextValue: () -> CGFloat) {
        value = max(value, nextValue())
    }
}

struct EqualWidthLayout: View {
    @State private var maxWidth: CGFloat = 0

    let labels = ["短", "中等长度", "这是一段比较长的文本"]

    var body: some View {
        VStack(spacing: 12) {
            ForEach(labels, id: \.self) { label in
                Text(label)
                    .padding(.horizontal, 16)
                    .padding(.vertical, 10)
                    .frame(width: maxWidth > 0 ? maxWidth : nil)
                    .background(.blue.gradient, in: Capsule())
                    .foregroundStyle(.white)
                    .background(
                        GeometryReader { geo in
                            Color.clear.preference(
                                key: WidthPreferenceKey.self,
                                value: geo.size.width
                            )
                        }
                    )
            }
        }
        .onPreferenceChange(WidthPreferenceKey.self) { value in
            maxWidth = value
        }
    }
}

完整示例
以下是一个完整的可复用组件库，包含自定义 modifier、button style、toggle style 和复合组件：


import SwiftUI

struct GlassModifier: ViewModifier {
    func body(content: Content) -> some View {
        content
            .padding()
            .background(.ultraThinMaterial, in: RoundedRectangle(cornerRadius: 16))
            .shadow(color: .black.opacity(0.08), radius: 8, y: 4)
    }
}

extension View {
    func glassCard() -> some View {
        modifier(GlassModifier())
    }
}

struct PillButtonStyle: ButtonStyle {
    let color: Color

    func makeBody(configuration: Configuration) -> some View {
        configuration.label
            .font(.subheadline.bold())
            .padding(.horizontal, 20)
            .padding(.vertical, 10)
            .background(color.gradient, in: Capsule())
            .foregroundStyle(.white)
            .scaleEffect(configuration.isPressed ? 0.96 : 1.0)
            .animation(.spring(duration: 0.15), value: configuration.isPressed)
    }
}

struct IconToggleStyle: ToggleStyle {
    let onIcon: String
    let offIcon: String

    func makeBody(configuration: Configuration) -> some View {
        Button {
            withAnimation(.spring(duration: 0.3)) {
                configuration.isOn.toggle()
            }
        } label: {
            HStack {
                Image(systemName: configuration.isOn ? onIcon : offIcon)
                    .font(.title3)
                    .foregroundStyle(configuration.isOn ? .yellow : .gray)
                    .symbolEffect(.bounce, value: configuration.isOn)
                configuration.label
            }
        }
        .buttonStyle(.plain)
    }
}

struct StatCard<Content: View>: View {
    let title: String
    let value: String
    let trend: Double
    let detail: Content

    init(
        title: String,
        value: String,
        trend: Double,
        @ViewBuilder detail: () -> Content = { EmptyView() }
    ) {
        self.title = title
        self.value = value
        self.trend = trend
        self.detail = detail()
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            Text(title)
                .font(.caption)
                .foregroundStyle(.secondary)

            HStack(alignment: .firstTextBaseline, spacing: 4) {
                Text(value)
                    .font(.title.bold())

                HStack(spacing: 2) {
                    Image(systemName: trend >= 0 ? "arrow.up.right" : "arrow.down.right")
                    Text("\(abs(trend), specifier: "%.1f")%")
                }
                .font(.caption.bold())
                .foregroundStyle(trend >= 0 ? .green : .red)
            }

            detail
        }
        .glassCard()
    }
}

struct ComponentLibraryDemo: View {
    @State private var darkMode = false
    @State private var notifications = true
    @State private var favorites = false

    var body: some View {
        ScrollView {
            VStack(spacing: 20) {
                HStack(spacing: 12) {
                    StatCard(title: "日活跃用户", value: "12.5K", trend: 8.3) {
                        Text("较昨日")
                            .font(.caption2)
                            .foregroundStyle(.secondary)
                    }
                    StatCard(title: "转化率", value: "3.2%", trend: -1.2)
                }

                VStack(alignment: .leading, spacing: 12) {
                    Text("操作")
                        .font(.headline)
                        .padding(.horizontal)

                    HStack(spacing: 12) {
                        Button("主要操作") {}
                            .buttonStyle(PillButtonStyle(color: .blue))
                        Button("次要操作") {}
                            .buttonStyle(PillButtonStyle(color: .gray))
                        Button("危险操作") {}
                            .buttonStyle(PillButtonStyle(color: .red))
                    }
                    .padding(.horizontal)
                }

                VStack(alignment: .leading, spacing: 12) {
                    Text("设置")
                        .font(.headline)

                    Toggle("深色模式", isOn: $darkMode)
                        .toggleStyle(IconToggleStyle(onIcon: "moon.fill", offIcon: "moon"))

                    Toggle("推送通知", isOn: $notifications)
                        .toggleStyle(IconToggleStyle(onIcon: "bell.fill", offIcon: "bell.slash"))

                    Toggle("收藏功能", isOn: $favorites)
                        .toggleStyle(IconToggleStyle(onIcon: "star.fill", offIcon: "star"))
                }
                .glassCard()
            }
            .padding()
        }
        .background(Color(.systemGroupedBackground))
    }
}
最佳实践




推荐做法
不推荐做法





用 ViewModifier 封装可复用的样式组合
在每个视图上重复相同的 modifier 链




为 modifier 提供 View 扩展方法
要求调用方写 .modifier(MyModifier())




ButtonStyle / ToggleStyle 保持无状态
在 Style 中持有 @State




@ViewBuilder 泛型参数提供组合灵活性
用 AnyView 擦除类型




PreferenceKey 收集几何信息做布局对齐
用 GeometryReader 在全局坐标系做复杂计算




命名遵循 SwiftUI 惯例（.xxxStyle）
自造不一致的命名模式




提供合理的默认值，减少必填参数
每个参数都要求调用方显式传入

小结
    

  • ViewModifier 将重复的修饰符链封装为可复用单元，配合 View 扩展提供优雅的调用语法
    • 

  • ButtonStyle 通过 configuration.isPressed 实现自定义按压反馈和外观
    • 

  • ToggleStyle 可以把标准开关改造为复选框、图标切换等任意样式
    • 

  • @ViewBuilder 让自定义组件支持声明式的内容插槽，像原生容器一样灵活
    • 

  • PreferenceKey 实现子视图到父视图的数据"冒泡"，是 SwiftUI 布局系统的重要补充
    • 

  • 好的组件设计遵循单一职责、可配置、API 一致、可预览、可组合五大原则
    • 

  • 将组件按功能分类组织到独立目录，配合 #Preview 确保每个组件可独立验证
    •
