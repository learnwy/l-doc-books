Part I · iOSiOS 开发 简介SwiftUI 进阶 简介导航系统

导航系统

概述

SwiftUI 从 iOS 16 开始引入了全新的导航 API——NavigationStack 和 NavigationSplitView，取代了旧版 NavigationView。新 API 支持类型安全的路由、程序化导航和深层链接，让复杂的导航逻辑变得清晰可控。

本篇将构建一个完整的多级导航应用，涵盖从基础跳转到深层链接的全部核心知识。

基础用法
NavigationStack 与 NavigationLink
NavigationStack 是导航容器，NavigationLink 负责触发跳转。iOS 16+ 推荐使用值驱动的方式声明目的地。


struct ContentView: View {
    let fruits = ["苹果", "香蕉", "橙子", "葡萄"]

    var body: some View {
        NavigationStack {
            List(fruits, id: \.self) { fruit in
                NavigationLink(fruit, value: fruit)
            }
            .navigationTitle("水果列表")
            .navigationDestination(for: String.self) { fruit in
                FruitDetailView(name: fruit)
            }
        }
    }
}

struct FruitDetailView: View {
    let name: String

    var body: some View {
        VStack(spacing: 20) {
            Text(name)
                .font(.largeTitle)
            Text("这是 \(name) 的详情页面")
                .foregroundStyle(.secondary)
        }
        .navigationTitle(name)
    }
}
核心要点：


    

  • NavigationLink(_:value:) 将导航目标与值绑定，而非直接内联视图
    • 

  • .navigationDestination(for:) 根据值的类型决定目标视图
    • 

  • 同一类型只需注册一次 destination，所有匹配的 NavigationLink 都会自动路由
    • 



多类型路由
一个 NavigationStack 可以注册多种类型的 destination：


struct Category: Hashable {
    let id: Int
    let name: String
}

struct Product: Hashable {
    let id: Int
    let title: String
    let categoryId: Int
}

struct ShopView: View {
    let categories: [Category]
    let products: [Product]

    var body: some View {
        NavigationStack {
            List {
                Section("分类") {
                    ForEach(categories, id: \.id) { category in
                        NavigationLink(category.name, value: category)
                    }
                }
                Section("商品") {
                    ForEach(products, id: \.id) { product in
                        NavigationLink(product.title, value: product)
                    }
                }
            }
            .navigationTitle("商店")
            .navigationDestination(for: Category.self) { category in
                CategoryDetailView(category: category)
            }
            .navigationDestination(for: Product.self) { product in
                ProductDetailView(product: product)
            }
        }
    }
}
进阶用法
NavigationPath 与程序化导航
NavigationPath 是一个类型擦除的导航栈，配合 @State 可以在代码中任意控制导航层级。


struct AppView: View {
    @State private var path = NavigationPath()

    var body: some View {
        NavigationStack(path: $path) {
            VStack(spacing: 16) {
                Button("直接跳转到商品详情") {
                    path.append(Product(id: 1, title: "iPhone", categoryId: 1))
                }

                Button("跳转两级：分类 → 商品") {
                    path.append(Category(id: 1, name: "电子产品"))
                    path.append(Product(id: 1, title: "iPhone", categoryId: 1))
                }

                Button("返回根视图") {
                    path.removeLast(path.count)
                }
            }
            .navigationTitle("首页")
            .navigationDestination(for: Category.self) { category in
                CategoryDetailView(category: category)
            }
            .navigationDestination(for: Product.self) { product in
                ProductDetailView(product: product)
            }
        }
    }
}
NavigationSplitView（iPad / macOS 适配）
NavigationSplitView 提供分栏布局，在 iPad 和 macOS 上呈现侧边栏 + 详情的经典结构，在 iPhone 上自动退化为 stack 导航。


struct SplitNavView: View {
    @State private var selectedCategory: Category?
    @State private var selectedProduct: Product?

    let categories: [Category]
    let productsByCategory: [Int: [Product]]

    var body: some View {
        NavigationSplitView {
            List(categories, id: \.id, selection: $selectedCategory) { category in
                Text(category.name)
            }
            .navigationTitle("分类")
        } content: {
            if let category = selectedCategory,
               let products = productsByCategory[category.id] {
                List(products, id: \.id, selection: $selectedProduct) { product in
                    Text(product.title)
                }
                .navigationTitle(category.name)
            } else {
                Text("请选择分类")
                    .foregroundStyle(.secondary)
            }
        } detail: {
            if let product = selectedProduct {
                ProductDetailView(product: product)
            } else {
                Text("请选择商品")
                    .foregroundStyle(.secondary)
            }
        }
    }
}
深层链接（Deep Linking）
深层链接的关键是将 URL 解析为导航路径，然后注入 NavigationPath。


enum DeepLink {
    case category(id: Int)
    case product(id: Int)

    static func parse(url: URL) -> [any Hashable] {
        var pathComponents: [any Hashable] = []
        let components = url.pathComponents.filter { $0 != "/" }

        var index = 0
        while index < components.count {
            switch components[index] {
            case "category":
                if index + 1 < components.count, let id = Int(components[index + 1]) {
                    pathComponents.append(Category(id: id, name: "分类\(id)"))
                    index += 2
                } else {
                    index += 1
                }
            case "product":
                if index + 1 < components.count, let id = Int(components[index + 1]) {
                    pathComponents.append(Product(id: id, title: "商品\(id)", categoryId: 0))
                    index += 2
                } else {
                    index += 1
                }
            default:
                index += 1
            }
        }
        return pathComponents
    }
}

struct DeepLinkApp: View {
    @State private var path = NavigationPath()

    var body: some View {
        NavigationStack(path: $path) {
            HomeView()
                .navigationDestination(for: Category.self) { CategoryDetailView(category: $0) }
                .navigationDestination(for: Product.self) { ProductDetailView(product: $0) }
        }
        .onOpenURL { url in
            let items = DeepLink.parse(url: url)
            path.removeLast(path.count)
            for item in items {
                if let cat = item as? Category { path.append(cat) }
                if let prod = item as? Product { path.append(prod) }
            }
        }
    }
}
完整示例
以下是一个集成 NavigationStack、程序化导航和工具栏的完整应用：


import SwiftUI

struct Note: Identifiable, Hashable {
    let id = UUID()
    var title: String
    var content: String
    var date: Date
}

@Observable
class NoteStore {
    var notes: [Note] = [
        Note(title: "会议纪要", content: "讨论了 Q2 计划", date: .now),
        Note(title: "购物清单", content: "牛奶、鸡蛋、面包", date: .now.addingTimeInterval(-86400)),
        Note(title: "读书笔记", content: "《SwiftUI 编程》第三章", date: .now.addingTimeInterval(-172800)),
    ]
}

struct NoteListView: View {
    @State private var store = NoteStore()
    @State private var path = NavigationPath()
    @State private var searchText = ""

    var filteredNotes: [Note] {
        if searchText.isEmpty { return store.notes }
        return store.notes.filter { $0.title.localizedCaseInsensitiveContains(searchText) }
    }

    var body: some View {
        NavigationStack(path: $path) {
            List(filteredNotes) { note in
                NavigationLink(value: note) {
                    VStack(alignment: .leading) {
                        Text(note.title)
                            .font(.headline)
                        Text(note.date, style: .date)
                            .font(.caption)
                            .foregroundStyle(.secondary)
                    }
                }
            }
            .navigationTitle("备忘录")
            .searchable(text: $searchText, prompt: "搜索笔记")
            .navigationDestination(for: Note.self) { note in
                NoteDetailView(note: note)
            }
            .toolbar {
                ToolbarItem(placement: .primaryAction) {
                    Button("跳转到最新") {
                        if let first = store.notes.first {
                            path.append(first)
                        }
                    }
                }
            }
        }
    }
}

struct NoteDetailView: View {
    let note: Note

    var body: some View {
        ScrollView {
            VStack(alignment: .leading, spacing: 16) {
                Text(note.title)
                    .font(.title)
                    .bold()
                Text(note.date, style: .date)
                    .foregroundStyle(.secondary)
                Divider()
                Text(note.content)
                    .font(.body)
            }
            .padding()
        }
        .navigationTitle(note.title)
        .navigationBarTitleDisplayMode(.inline)
    }
}

导航状态持久化——可以将 NavigationPath 编码为 Data 保存到 UserDefaults，在应用重启后恢复导航状态：


extension NavigationPath {
    func encoded() -> Data? {
        try? JSONEncoder().encode(self.codable)
    }

    static func decoded(from data: Data) -> NavigationPath {
        guard let codable = try? JSONDecoder().decode(
            NavigationPath.CodableRepresentation.self, from: data
        ) else { return NavigationPath() }
        return NavigationPath(codable)
    }
}
小结
    

  • NavigationStack 是 iOS 16+ 的标准导航容器，取代 NavigationView
    • 

  • NavigationLink(value:) + .navigationDestination(for:) 提供类型安全的路由
    • 

  • NavigationPath 实现程序化导航——可以 push、pop、直接跳转到任意深度
    • 

  • NavigationSplitView 为 iPad/macOS 提供分栏导航，iPhone 上自动退化为 stack
    • 

  • 深层链接的核心思路是 URL → 解析 → 构建 NavigationPath → 驱动导航
    • 

  • 导航状态应集中管理，在根视图或 App 层持有 NavigationPath
    • 

  • 路由模型（Hashable 类型）应与视图分离，保持可测试性
    •
