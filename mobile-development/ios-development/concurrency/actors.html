Actor 模型与数据隔离

概述

Actor 是 Swift 5.5 引入的引用类型，专门用于解决并发环境下的**数据竞争（Data Race）问题。Actor 通过隔离（isolation）**机制保证同一时间只有一个任务可以访问 actor 的可变状态，无需手动使用锁、信号量或串行队列。

在传统并发编程中，保护共享可变状态是最容易出错的环节——忘记加锁、锁粒度不当、死锁等问题层出不穷。Actor 将数据保护从"开发者的纪律"变为"编译器的强制约束"：从外部访问 actor 的隔离属性和方法必须使用 await，编译器会在编译期拒绝不安全的访问。配合 @MainActor、GlobalActor 和 Sendable 检查，Actor 模型构成了 Swift 并发安全的核心支柱。

核心概念
Actor 定义
actor 是一种引用类型（像 class），但编译器自动保证对其可变状态的互斥访问：


actor BankAccount {
    let accountId: String
    private(set) var balance: Decimal

    init(accountId: String, initialBalance: Decimal) {
        self.accountId = accountId
        self.balance = initialBalance
    }

    func deposit(amount: Decimal) {
        balance += amount
    }

    func withdraw(amount: Decimal) throws {
        guard balance >= amount else {
            throw BankError.insufficientFunds
        }
        balance -= amount
    }
}

enum BankError: Error {
    case insufficientFunds
}
从外部访问 actor 的隔离成员必须使用 await：


let account = BankAccount(accountId: "A001", initialBalance: 1000)

Task {
    await account.deposit(amount: 500)
    let balance = await account.balance
    print("余额：\(balance)")
}
isolated 与 nonisolated
Actor 的方法和属性默认是 isolated（隔离的）——调用时需要 await，保证互斥访问。nonisolated 关键字标记不需要隔离的成员，它们可以在不 await 的情况下同步访问：


actor UserSession {
    let userId: String
    private var token: String
    private var lastActiveDate: Date

    init(userId: String, token: String) {
        self.userId = userId
        self.token = token
        self.lastActiveDate = Date()
    }

    func updateToken(_ newToken: String) {
        token = newToken
        lastActiveDate = Date()
    }

    nonisolated func description() -> String {
        "UserSession(\(userId))"
    }
}

let session = UserSession(userId: "U123", token: "abc")
print(session.description())         // ✅ 不需要 await
// print(session.token)              // ❌ 编译错误：需要 await
let _ = await session.lastActiveDate // ✅ 需要 await
nonisolated 只能访问 actor 的不可变属性（let）或不访问任何隔离状态。编译器会在编译期阻止 nonisolated 方法访问可变状态。


让 actor 遵循协议时，如果协议的方法不需要访问可变状态，使用 nonisolated：


actor DocumentStore: CustomStringConvertible {
    let name: String
    private var documents: [Document] = []

    init(name: String) { self.name = name }

    nonisolated var description: String {
        "DocumentStore(\(name))"
    }

    func add(_ document: Document) {
        documents.append(document)
    }
}
@MainActor
@MainActor 是一个全局 Actor，将代码隔离到主线程执行。它是 UI 层安全的编译期保证，替代了手动 DispatchQueue.main.async：


@MainActor
class ProfileViewModel: ObservableObject {
    @Published var name: String = ""
    @Published var avatarURL: URL?
    @Published var isLoading = false

    func loadProfile() async {
        isLoading = true
        defer { isLoading = false }

        do {
            let profile = try await ProfileService.shared.fetch()
            name = profile.name
            avatarURL = profile.avatarURL
        } catch {
            print("加载失败：\(error)")
        }
    }
}
也可以在方法级别使用 @MainActor：


class DataProcessor {
    func processInBackground() async -> ProcessedData {
        await heavyComputation()
    }

    @MainActor
    func updateUI(with data: ProcessedData) {
        titleLabel.text = data.title
        progressView.progress = Float(data.progress)
    }
}
GlobalActor
除了 @MainActor，可以自定义全局 Actor 来创建共享的隔离域：


@globalActor
actor DatabaseActor {
    static let shared = DatabaseActor()
}

@DatabaseActor
class DatabaseManager {
    private var connection: DatabaseConnection?

    func connect(to url: String) async throws {
        connection = try await DatabaseConnection.open(url)
    }

    func execute(_ query: String) async throws -> [Row] {
        guard let connection else { throw DatabaseError.notConnected }
        return try await connection.execute(query)
    }
}

@DatabaseActor
func performMigration() async throws {
    let manager = DatabaseManager()
    try await manager.connect(to: "sqlite:///app.db")
    try await manager.execute("CREATE TABLE IF NOT EXISTS users (...)")
}
所有标注 @DatabaseActor 的代码保证在同一个隔离域中串行执行，保护数据库连接不被并发访问。


Actor 重入性（Reentrancy）
Actor 方法中的 await 是挂起点——挂起期间，actor 可以处理其他消息。这意味着 await 前后 actor 的状态可能已经改变：


actor ImageDownloader {
    private var cache: [URL: UIImage] = [:]

    func image(from url: URL) async throws -> UIImage {
        if let cached = cache[url] {
            return cached
        }

        let (data, _) = try await URLSession.shared.data(from: url)
        // ⚠️ 挂起恢复后，其他任务可能已经缓存了同一 URL
        guard let image = UIImage(data: data) else {
            throw ImageError.decodingFailed
        }

        cache[url] = cache[url] ?? image
        return cache[url]!
    }
}
防御重入的模式——使用进行中任务的字典：


actor ImageDownloader {
    private var cache: [URL: UIImage] = [:]
    private var inProgress: [URL: Task<UIImage, Error>] = [:]

    func image(from url: URL) async throws -> UIImage {
        if let cached = cache[url] {
            return cached
        }

        if let existing = inProgress[url] {
            return try await existing.value
        }

        let task = Task {
            let (data, _) = try await URLSession.shared.data(from: url)
            guard let image = UIImage(data: data) else {
                throw ImageError.decodingFailed
            }
            return image
        }

        inProgress[url] = task
        defer { inProgress[url] = nil }

        let image = try await task.value
        cache[url] = image
        return image
    }
}
Sendable 检查
跨 actor 边界传递的数据必须是 Sendable 的。编译器在严格并发检查模式（Swift 6）下强制执行：


struct Message: Sendable {
    let id: UUID
    let content: String
    let timestamp: Date
}

actor ChatRoom {
    private var messages: [Message] = []

    func post(_ message: Message) {
        messages.append(message)
    }

    func allMessages() -> [Message] {
        messages
    }
}

let room = ChatRoom()
let msg = Message(id: UUID(), content: "Hello", timestamp: Date())
await room.post(msg)         // ✅ Message 是 Sendable
let all = await room.allMessages()  // ✅ [Message] 是 Sendable
非 Sendable 类型跨 actor 传递会在 Swift 6 中编译报错：


class MutablePayload {
    var data: [String] = []
}

actor Processor {
    func process(_ payload: MutablePayload) {
        // ⚠️ Swift 5 警告 / Swift 6 编译错误
        // MutablePayload 不是 Sendable
    }
}
从 class 迁移到 actor
将线程不安全的 class 迁移到 actor 的典型步骤：


迁移前——使用串行队列保护的 class：


class CounterOld {
    private var count = 0
    private let queue = DispatchQueue(label: "com.app.counter")

    func increment() {
        queue.sync { count += 1 }
    }

    func getCount() -> Int {
        queue.sync { count }
    }

    func reset() {
        queue.sync { count = 0 }
    }
}
迁移后——使用 actor：


actor Counter {
    private var count = 0

    func increment() {
        count += 1
    }

    func getCount() -> Int {
        count
    }

    func reset() {
        count = 0
    }
}

使用方式
模式一：线程安全的缓存服务
actor CacheService<Key: Hashable & Sendable, Value: Sendable> {
    private var storage: [Key: CacheEntry<Value>] = [:]
    private let ttl: TimeInterval

    init(ttl: TimeInterval = 300) {
        self.ttl = ttl
    }

    func get(_ key: Key) -> Value? {
        guard let entry = storage[key],
              Date().timeIntervalSince(entry.createdAt) < ttl else {
            storage[key] = nil
            return nil
        }
        return entry.value
    }

    func set(_ key: Key, value: Value) {
        storage[key] = CacheEntry(value: value, createdAt: Date())
    }

    func clear() {
        storage.removeAll()
    }
}

struct CacheEntry<Value> {
    let value: Value
    let createdAt: Date
}
模式二：@MainActor ViewModel + 后台 Actor
分层隔离——ViewModel 在主线程更新 UI，Service 在后台 actor 处理数据：


actor AnalyticsService {
    private var events: [AnalyticsEvent] = []

    func track(_ event: AnalyticsEvent) {
        events.append(event)
        if events.count >= 50 {
            flush()
        }
    }

    private func flush() {
        let batch = events
        events.removeAll()
        Task.detached {
            try? await NetworkClient.shared.upload(batch)
        }
    }
}

@MainActor
class DashboardViewModel: ObservableObject {
    @Published var metrics: [Metric] = []
    private let analytics = AnalyticsService()

    func loadMetrics() async {
        let data = try? await MetricsAPI.shared.fetch()
        metrics = data ?? []
        await analytics.track(AnalyticsEvent(name: "dashboard_loaded"))
    }
}
模式三：Actor 实现读写分离
对于读多写少的场景，可以用 actor 的 nonisolated 返回快照实现无等待读取：


actor Configuration {
    private var values: [String: Any] = [:]
    private(set) var snapshot: ConfigSnapshot = ConfigSnapshot(values: [:])

    func update(key: String, value: Any) {
        values[key] = value
        snapshot = ConfigSnapshot(values: values)
    }

    func bulkUpdate(_ updates: [String: Any]) {
        for (key, value) in updates {
            values[key] = value
        }
        snapshot = ConfigSnapshot(values: values)
    }
}

struct ConfigSnapshot: Sendable {
    let values: [String: Any]

    func string(forKey key: String) -> String? {
        values[key] as? String
    }

    func integer(forKey key: String) -> Int? {
        values[key] as? Int
    }
}
常见陷阱
陷阱一：Actor 重入导致状态不一致
在 await 前后假设状态不变是最常见的错误：


actor Wallet {
    var balance: Decimal = 100

    func purchaseIfAffordable(price: Decimal) async throws -> Receipt {
        guard balance >= price else { throw WalletError.insufficientFunds }

        let receipt = try await PaymentGateway.shared.charge(amount: price)
        // ⚠️ 恢复后 balance 可能已被其他任务修改
        balance -= price  // 可能导致余额为负！
        return receipt
    }
}
正确做法——在 await 后重新检查条件：


actor Wallet {
    var balance: Decimal = 100

    func purchaseIfAffordable(price: Decimal) async throws -> Receipt {
        guard balance >= price else { throw WalletError.insufficientFunds }

        let receipt = try await PaymentGateway.shared.charge(amount: price)

        guard balance >= price else {
            try await PaymentGateway.shared.refund(receipt: receipt)
            throw WalletError.insufficientFunds
        }

        balance -= price
        return receipt
    }
}
陷阱二：在 nonisolated 方法中意外访问隔离状态
actor Logger {
    var logs: [String] = []

    nonisolated func format(_ message: String) -> String {
        // ❌ 编译错误：nonisolated 方法不能访问 isolated 属性
        // let count = logs.count
        return "[\(Date())] \(message)"
    }
}
陷阱三：过度使用 @MainActor 导致性能问题
将不需要主线程的计算标注为 @MainActor 会阻塞 UI：


@MainActor
class BadViewModel: ObservableObject {
    @Published var result: String = ""

    func process() async {
        // ❌ 耗时计算在主线程执行，阻塞 UI
        var sum = 0
        for i in 0..<10_000_000 {
            sum += i
        }
        result = "结果：\(sum)"
    }
}
正确做法——将耗时操作移出主 actor：


@MainActor
class GoodViewModel: ObservableObject {
    @Published var result: String = ""

    func process() async {
        let computed = await Task.detached(priority: .userInitiated) {
            var sum = 0
            for i in 0..<10_000_000 {
                sum += i
            }
            return sum
        }.value

        result = "结果：\(computed)"
    }
}
陷阱四：Actor 不支持继承
actor BaseService {
    func commonMethod() { }
}

// ❌ 编译错误：actor 不支持继承
// actor DerivedService: BaseService { }
如果需要共享行为，使用协议 + 扩展：


protocol ServiceProtocol: Actor {
    func healthCheck() async -> Bool
}

extension ServiceProtocol {
    func healthCheck() async -> Bool {
        true
    }
}

actor UserService: ServiceProtocol { }
actor OrderService: ServiceProtocol { }
小结
    

  • Actor 是引用类型，通过编译器强制的隔离机制保护可变状态，同一时间只有一个任务可以访问 actor 内部。
    • 

  • 从外部访问 actor 的 isolated 成员必须使用 await，编译器在编译期拒绝不安全访问。
    • 

  • nonisolated 标记不需要隔离的成员（只访问 let 属性或不访问状态），可同步调用。
    • 

  • @MainActor 是全局 Actor，保证代码在主线程执行，是 UI 安全更新的编译期保证。
    • 

  • GlobalActor 允许自定义隔离域，将分散在多个类型中的代码统一到同一个 actor 上下文。
    • 

  • Actor 重入性意味着 await 前后状态可能改变，必须在 await 恢复后重新验证条件。
    • 

  • 跨 actor 边界传递的数据必须满足 Sendable 约束，Swift 6 严格模式下为编译错误。
    • 

  • Actor 不支持继承，使用协议 + 扩展实现共享行为。
    • 

  • 从 class + DispatchQueue 迁移到 actor：删除手动同步代码，添加 await，确保 Sendable 合规。
    •
