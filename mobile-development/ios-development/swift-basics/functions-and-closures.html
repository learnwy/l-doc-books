Part I · iOSiOS 开发 简介Swift 基础 简介函数与闭包

函数与闭包

概述

函数是 Swift 中组织和复用代码的基本单元；闭包则是可以捕获上下文变量的自包含代码块。在 Swift 中，函数本身就是一种特殊的命名闭包，而闭包表达式是一种轻量级的匿名函数。理解二者的关系及其各种用法，是掌握 Swift 编程的关键——从 API 设计到异步回调，从集合操作到 SwiftUI 构建器，函数与闭包无处不在。

核心概念
函数定义
Swift 函数使用 func 关键字定义，包含参数标签、参数名、返回类型：


func greet(person name: String) -> String {
    return "你好，\(name)！"
}

let message = greet(person: "Alice")  // "你好，Alice！"
无返回值的函数实际上返回 Void（即空元组 ()）：


func printGreeting(to name: String) {
    print("你好，\(name)！")
}
参数标签 (Argument Label) 与参数名
Swift 函数的每个参数都有两个名称：参数标签（调用时使用）和参数名（函数体内使用）。


func move(from source: String, to destination: String) {
    print("从 \(source) 移动到 \(destination)")
}
move(from: "北京", to: "上海")
使用下划线 _ 省略参数标签：


func square(_ number: Int) -> Int {
    number * number
}
let result = square(5)     // 调用时不需要标签


设计原则：参数标签应使调用处读起来像一句自然语言。例如 move(from:to:) 比 move(source:destination:) 更流畅。




默认参数值
func connect(host: String, port: Int = 443, useTLS: Bool = true) {
    print("连接 \(host):\(port), TLS: \(useTLS)")
}

connect(host: "api.example.com")
connect(host: "api.example.com", port: 8080)
connect(host: "api.example.com", port: 80, useTLS: false)
inout 参数
默认情况下，函数参数是常量。使用 inout 可以让函数直接修改传入的变量：


func swapValues(_ a: inout Int, _ b: inout Int) {
    let temp = a
    a = b
    b = temp
}

var x = 10, y = 20
swapValues(&x, &y)         // 传入时使用 & 前缀
print("x=\(x), y=\(y)")   // x=20, y=10


inout 的语义是 copy-in copy-out：进入函数时拷贝值，函数返回时将修改后的值写回原变量。




返回元组
函数可以通过元组返回多个值：


func minMax(array: [Int]) -> (min: Int, max: Int)? {
    guard let first = array.first else { return nil }

    var currentMin = first
    var currentMax = first

    for value in array.dropFirst() {
        if value < currentMin { currentMin = value }
        if value > currentMax { currentMax = value }
    }

    return (min: currentMin, max: currentMax)
}

if let result = minMax(array: [3, 1, 4, 1, 5, 9, 2, 6]) {
    print("最小值：\(result.min), 最大值：\(result.max)")
}
函数类型
Swift 中函数也是一种类型，可以赋值给变量、作为参数传递、作为返回值：


func add(_ a: Int, _ b: Int) -> Int { a + b }
func multiply(_ a: Int, _ b: Int) -> Int { a * b }

var operation: (Int, Int) -> Int = add
print(operation(3, 4))     // 7

operation = multiply
print(operation(3, 4))     // 12

func apply(_ a: Int, _ b: Int, using op: (Int, Int) -> Int) -> Int {
    op(a, b)
}
print(apply(5, 3, using: add))       // 8
print(apply(5, 3, using: multiply))  // 15
闭包表达式
闭包表达式是一种简洁的匿名函数语法：


// 完整形式
let doubled = [1, 2, 3].map({ (number: Int) -> Int in
    return number * 2
})

// 利用类型推断省略类型
let doubled2 = [1, 2, 3].map({ number in
    return number * 2
})

// 单表达式隐式返回
let doubled3 = [1, 2, 3].map({ number in number * 2 })

// 使用简写参数名 $0, $1
let doubled4 = [1, 2, 3].map({ $0 * 2 })

// 尾随闭包语法
let doubled5 = [1, 2, 3].map { $0 * 2 }
尾随闭包 (Trailing Closure)
当闭包是函数的最后一个参数时，可以将其写在括号外面。如果闭包是唯一参数，括号可以省略：


let numbers = [5, 3, 8, 1, 9, 2]

let sorted = numbers.sorted { $0 < $1 }

let filtered = numbers.filter { $0 > 3 }

let transformed = numbers
    .filter { $0.isMultiple(of: 2) }
    .map { $0 * $0 }
    .sorted { $0 > $1 }
// [64, 16, 4]
多尾随闭包语法（Swift 5.3+）：


func animate(
    duration: Double,
    animations: () -> Void,
    completion: (Bool) -> Void
) {
    animations()
    completion(true)
}

animate(duration: 0.3) {
    print("执行动画")
} completion: { finished in
    print("动画完成：\(finished)")
}
逃逸闭包 (@escaping)
默认情况下，传入函数的闭包是非逃逸的——它只在函数执行期间有效。如果闭包需要在函数返回之后被调用（例如异步回调），必须标记为 @escaping：


typealias CompletionHandler = (Result<Data, Error>) -> Void

var pendingCallbacks: [CompletionHandler] = []

func fetchData(from url: String, completion: @escaping CompletionHandler) {
    pendingCallbacks.append(completion)
    // 闭包会在函数返回后的某个时刻被调用
}
在逃逸闭包中访问 self 的属性或方法时，必须显式写出 self：


class DataManager {
    var data: [String] = []

    func loadData() {
        fetchData(from: "https://api.example.com") { [weak self] result in
            guard let self else { return }
            switch result {
            case .success(let data):
                self.data = self.parseData(data)
            case .failure(let error):
                print("错误：\(error)")
            }
        }
    }

    func parseData(_ data: Data) -> [String] { [] }
}
捕获列表 (Capture List)
闭包会捕获其定义环境中的变量和常量。默认是引用捕获（捕获变量的引用），使用捕获列表可以改为值捕获或声明弱引用：


func makeCounter() -> () -> Int {
    var count = 0
    return {
        count += 1        // 引用捕获：闭包与外部共享同一个 count
        return count
    }
}

let counter = makeCounter()
print(counter())           // 1
print(counter())           // 2
print(counter())           // 3
值捕获（创建独立副本）：


var value = 10
let closure = { [value] in
    print(value)           // 捕获时的快照：10
}
value = 99
closure()                  // 10（不受外部修改影响）
弱引用捕获（防止循环引用）：


class ViewController {
    var onDismiss: (() -> Void)?

    func setup() {
        onDismiss = { [weak self] in
            guard let self else { return }
            self.cleanUp()
        }
    }

    func cleanUp() {
        print("清理资源")
    }
}
使用方式
模式一：高阶函数链式操作
利用 map、filter、reduce、compactMap 等高阶函数配合闭包处理集合：


struct Product {
    let name: String
    let price: Double
    let inStock: Bool
}

let products = [
    Product(name: "iPhone", price: 7999, inStock: true),
    Product(name: "iPad", price: 5499, inStock: false),
    Product(name: "MacBook", price: 12999, inStock: true),
    Product(name: "AirPods", price: 1299, inStock: true),
]

let affordableInStock = products
    .filter { $0.inStock && $0.price < 10000 }
    .map { "\($0.name) ¥\($0.price)" }
    .joined(separator: ", ")

print(affordableInStock)   // "iPhone ¥7999.0, AirPods ¥1299.0"

let totalPrice = products
    .filter(\.inStock)
    .reduce(0) { $0 + $1.price }
print("库存商品总价：¥\(totalPrice)")
模式二：回调与异步模式
enum NetworkResult {
    case success(Data)
    case failure(Error)
}

func request(
    url: String,
    method: String = "GET",
    completion: @escaping (NetworkResult) -> Void
) {
    DispatchQueue.global().async {
        let data = Data()
        DispatchQueue.main.async {
            completion(.success(data))
        }
    }
}

request(url: "https://api.example.com/users") { result in
    switch result {
    case .success(let data):
        print("收到 \(data.count) 字节")
    case .failure(let error):
        print("请求失败：\(error)")
    }
}
模式三：使用泛型函数提高复用性
func retry<T>(
    times: Int,
    delay: Double = 1.0,
    operation: () throws -> T
) rethrows -> T {
    var lastError: Error?
    for attempt in 1...times {
        do {
            return try operation()
        } catch {
            lastError = error
            print("第 \(attempt) 次尝试失败：\(error)")
        }
    }
    throw lastError!
}
常见陷阱
陷阱 1：逃逸闭包中的循环引用
class SearchManager {
    var query: String = ""
    var onResult: ((String) -> Void)?

    func search() {
        // ❌ 循环引用：self -> onResult -> self
        onResult = { result in
            self.query = result
        }

        // ✅ 使用 [weak self] 打破循环
        onResult = { [weak self] result in
            self?.query = result
        }
    }
}
陷阱 2：闭包引用捕获的陷阱
var functions: [() -> Int] = []
for i in 0..<5 {
    functions.append { i }  // 每个闭包捕获的是不同的 i（Swift 中 for-in 的 i 每次迭代是新绑定）
}
print(functions.map { $0() })  // [0, 1, 2, 3, 4] ✅

// 但如果使用 var：
var counter = 0
var closures: [() -> Int] = []
for _ in 0..<5 {
    closures.append { counter }
    counter += 1
}
print(closures.map { $0() })  // [5, 5, 5, 5, 5] ⚠️ 所有闭包共享同一个 counter
陷阱 3：inout 参数不能被闭包捕获
func process(value: inout Int) {
    let closure = {
        // print(value)       // ❌ 编译错误：逃逸闭包不能捕获 inout 参数
    }
    _ = closure

    // ✅ 先拷贝到局部变量
    let localValue = value
    let safeClosure = {
        print(localValue)
    }
    safeClosure()
}
陷阱 4：@escaping 闭包中遗忘 self
class Timer {
    var interval: Double = 1.0

    func start(callback: @escaping () -> Void) {
        // 编译器要求在 @escaping 闭包中显式使用 self
    }

    func configure() {
        start { [weak self] in
            guard let self else { return }
            print(self.interval)    // ✅ 显式 self
        }
    }
}
陷阱 5：过度简写降低可读性
// ❌ 过度简写，难以理解
let result = data.map { $0.0 }.filter { $0 > $1 }.reduce(0, +)

// ✅ 适度命名，清晰可读
let result2 = data
    .map { item in item.score }
    .filter { score in score > threshold }
    .reduce(0) { total, score in total + score }
小结
    

  • Swift 函数使用参数标签让调用处读起来像自然语言；使用 _ 可省略标签。
    • 

  • 默认参数减少重载数量；inout 允许函数修改外部变量；元组返回支持多返回值。
    • 

  • 函数是一等公民——可以作为变量、参数、返回值使用。
    • 

  • 闭包表达式可以逐步简化：省略类型 → 隐式返回 → $0 简写 → 尾随闭包。
    • 

  • 非逃逸闭包在函数执行期间有效；逃逸闭包（@escaping）可在函数返回后调用，常用于异步回调。
    • 

  • 闭包默认引用捕获；使用捕获列表 [value] 可改为值捕获，[weak self] / [unowned self] 用于防止循环引用。
    • 

  • 在逃逸闭包中务必使用 [weak self] 并做好 guard let self 检查。
    • 

  • 高阶函数（map/filter/reduce）配合闭包是 Swift 函数式编程的核心范式。
    •
