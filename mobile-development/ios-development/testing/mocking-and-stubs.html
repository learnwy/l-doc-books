Part I · iOSiOS 开发 简介测试 简介Mock 与 Stub 策略

Mock 与 Stub 策略

概述

在单元测试中，被测对象通常依赖外部组件——网络层、数据库、定位服务等。如果直接调用真实依赖，测试会变得缓慢、不确定且难以覆盖边界场景。测试替身（Test Double） 通过替换真实依赖来实现隔离测试，使每个测试只验证被测单元自身的逻辑。

Swift 没有运行时反射能力来自动生成 Mock（不同于 Java/Kotlin），因此 iOS 社区主流做法是基于 Protocol 手动构建测试替身。这种方式虽然需要更多样板代码，但带来了更清晰的依赖边界和更好的架构设计——如果一个类难以 Mock，往往说明它的依赖耦合过紧。

Protocol-Based Mocking
Swift 测试替身的核心模式：定义 Protocol → 生产代码依赖 Protocol → 测试中注入替身实现。


定义协议：


protocol UserRepository {
    func fetchUser(id: String) async throws -> User
    func saveUser(_ user: User) async throws
}
生产实现：


struct RemoteUserRepository: UserRepository {
    let session: URLSession

    func fetchUser(id: String) async throws -> User {
        let url = URL(string: "https://api.example.com/users/\(id)")!
        let (data, _) = try await session.data(from: url)
        return try JSONDecoder().decode(User.self, from: data)
    }

    func saveUser(_ user: User) async throws {
        // POST 请求保存用户
    }
}
Stub 实现：


struct StubUserRepository: UserRepository {
    var userToReturn: User?
    var errorToThrow: Error?

    func fetchUser(id: String) async throws -> User {
        if let error = errorToThrow { throw error }
        return userToReturn ?? User(id: id, name: "测试用户", email: "test@example.com")
    }

    func saveUser(_ user: User) async throws {
        if let error = errorToThrow { throw error }
    }
}
Mock 实现：


final class MockUserRepository: UserRepository {
    var fetchUserCallCount = 0
    var fetchUserLastID: String?
    var fetchUserResult: Result<User, Error> = .success(
        User(id: "1", name: "Mock用户", email: "mock@example.com")
    )

    var saveUserCallCount = 0
    var savedUsers: [User] = []

    func fetchUser(id: String) async throws -> User {
        fetchUserCallCount += 1
        fetchUserLastID = id
        return try fetchUserResult.get()
    }

    func saveUser(_ user: User) async throws {
        saveUserCallCount += 1
        savedUsers.append(user)
    }
}
Spy 实现：


final class SpyLogger: Logger {
    var loggedMessages: [(level: LogLevel, message: String)] = []

    func log(level: LogLevel, message: String) {
        loggedMessages.append((level, message))
    }
}
依赖注入
通过构造器注入依赖，使测试替身可以替换真实实现：


final class UserProfileViewModel: ObservableObject {
    @Published var user: User?
    @Published var errorMessage: String?

    private let repository: UserRepository

    init(repository: UserRepository) {
        self.repository = repository
    }

    func loadUser(id: String) async {
        do {
            user = try await repository.fetchUser(id: id)
        } catch {
            errorMessage = error.localizedDescription
        }
    }
}
测试中注入 Mock：


final class UserProfileViewModelTests: XCTestCase {

    func testLoadUser_success_setsUser() async {
        let expectedUser = User(id: "1", name: "张三", email: "zhang@example.com")
        let mock = MockUserRepository()
        mock.fetchUserResult = .success(expectedUser)
        let viewModel = UserProfileViewModel(repository: mock)

        await viewModel.loadUser(id: "1")

        XCTAssertEqual(viewModel.user?.name, "张三")
        XCTAssertNil(viewModel.errorMessage)
        XCTAssertEqual(mock.fetchUserCallCount, 1)
        XCTAssertEqual(mock.fetchUserLastID, "1")
    }

    func testLoadUser_failure_setsErrorMessage() async {
        let mock = MockUserRepository()
        mock.fetchUserResult = .failure(URLError(.notConnectedToInternet))
        let viewModel = UserProfileViewModel(repository: mock)

        await viewModel.loadUser(id: "1")

        XCTAssertNil(viewModel.user)
        XCTAssertNotNil(viewModel.errorMessage)
    }
}
网络层 Mock（URLProtocol）
对于直接使用 URLSession 的代码，可通过自定义 URLProtocol 拦截网络请求：


final class MockURLProtocol: URLProtocol {
    static var requestHandler: ((URLRequest) throws -> (HTTPURLResponse, Data))?

    override class func canInit(with request: URLRequest) -> Bool { true }

    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }

    override func startLoading() {
        guard let handler = Self.requestHandler else {
            client?.urlProtocolDidFinishLoading(self)
            return
        }
        do {
            let (response, data) = try handler(request)
            client?.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
            client?.urlProtocol(self, didLoad: data)
            client?.urlProtocolDidFinishLoading(self)
        } catch {
            client?.urlProtocol(self, didFailWithError: error)
        }
    }

    override func stopLoading() {}
}
配置测试用的 URLSession：


final class APIClientTests: XCTestCase {

    var session: URLSession!

    override func setUp() {
        super.setUp()
        let config = URLSessionConfiguration.ephemeral
        config.protocolClasses = [MockURLProtocol.self]
        session = URLSession(configuration: config)
    }

    func testFetchProducts_returnsDecodedList() async throws {
        let mockData = """
        [{"id": "1", "name": "iPhone", "price": 999.0}]
        """.data(using: .utf8)!

        MockURLProtocol.requestHandler = { request in
            let response = HTTPURLResponse(
                url: request.url!,
                statusCode: 200,
                httpVersion: nil,
                headerFields: ["Content-Type": "application/json"]
            )!
            return (response, mockData)
        }

        let client = APIClient(session: session)
        let products = try await client.fetchProducts()

        XCTAssertEqual(products.count, 1)
        XCTAssertEqual(products.first?.name, "iPhone")
    }

    func testFetchProducts_serverError_throwsError() async {
        MockURLProtocol.requestHandler = { request in
            let response = HTTPURLResponse(
                url: request.url!,
                statusCode: 500,
                httpVersion: nil,
                headerFields: nil
            )!
            return (response, Data())
        }

        let client = APIClient(session: session)

        do {
            _ = try await client.fetchProducts()
            XCTFail("应抛出服务器错误")
        } catch {
            XCTAssertTrue(error is APIError)
        }
    }
}
工作流程
步骤一：识别外部依赖
审查被测类的构造器和方法，找出所有外部依赖——网络请求、数据库操作、文件系统、系统服务（定位、推送）等。


步骤二：定义 Protocol
为每个依赖提取 Protocol，只暴露被测代码实际使用的方法（接口隔离原则）。


步骤三：构造器注入
被测类通过构造器接收 Protocol 类型的依赖，不在内部创建具体实现。


步骤四：创建测试替身
根据测试需求选择替身类型：需要控制返回值用 Stub，需要验证调用行为用 Mock，需要记录调用历史用 Spy。


步骤五：注入并断言
在测试中创建替身实例，配置预期行为，注入被测对象，执行后验证结果和交互。


实战演示
以订单提交流程为例，展示如何隔离测试包含多个依赖的 ViewModel。


协议定义：


protocol OrderService {
    func submitOrder(_ order: Order) async throws -> OrderConfirmation
}

protocol PaymentGateway {
    func charge(amount: Double, method: PaymentMethod) async throws -> PaymentReceipt
}

protocol AnalyticsTracker {
    func track(event: String, properties: [String: String])
}
被测 ViewModel：


final class CheckoutViewModel: ObservableObject {
    @Published var isProcessing = false
    @Published var confirmation: OrderConfirmation?
    @Published var errorMessage: String?

    private let orderService: OrderService
    private let paymentGateway: PaymentGateway
    private let analytics: AnalyticsTracker

    init(orderService: OrderService, paymentGateway: PaymentGateway, analytics: AnalyticsTracker) {
        self.orderService = orderService
        self.paymentGateway = paymentGateway
        self.analytics = analytics
    }

    func checkout(order: Order, payment: PaymentMethod) async {
        isProcessing = true
        defer { isProcessing = false }

        do {
            _ = try await paymentGateway.charge(amount: order.total, method: payment)
            let conf = try await orderService.submitOrder(order)
            confirmation = conf
            analytics.track(event: "order_completed", properties: ["order_id": conf.id])
        } catch {
            errorMessage = "结算失败：\(error.localizedDescription)"
            analytics.track(event: "order_failed", properties: ["error": error.localizedDescription])
        }
    }
}
测试替身：


final class MockOrderService: OrderService {
    var submitResult: Result<OrderConfirmation, Error> = .success(
        OrderConfirmation(id: "ORD-001", date: Date())
    )
    var submitCallCount = 0

    func submitOrder(_ order: Order) async throws -> OrderConfirmation {
        submitCallCount += 1
        return try submitResult.get()
    }
}

final class MockPaymentGateway: PaymentGateway {
    var chargeResult: Result<PaymentReceipt, Error> = .success(
        PaymentReceipt(transactionId: "TXN-001")
    )
    var chargedAmount: Double?

    func charge(amount: Double, method: PaymentMethod) async throws -> PaymentReceipt {
        chargedAmount = amount
        return try chargeResult.get()
    }
}

final class SpyAnalyticsTracker: AnalyticsTracker {
    var trackedEvents: [(event: String, properties: [String: String])] = []

    func track(event: String, properties: [String: String]) {
        trackedEvents.append((event, properties))
    }
}
测试代码：


final class CheckoutViewModelTests: XCTestCase {

    var orderService: MockOrderService!
    var paymentGateway: MockPaymentGateway!
    var analytics: SpyAnalyticsTracker!
    var sut: CheckoutViewModel!

    override func setUp() {
        super.setUp()
        orderService = MockOrderService()
        paymentGateway = MockPaymentGateway()
        analytics = SpyAnalyticsTracker()
        sut = CheckoutViewModel(
            orderService: orderService,
            paymentGateway: paymentGateway,
            analytics: analytics
        )
    }

    func testCheckout_success_setsConfirmation() async {
        let order = Order(items: [OrderItem(name: "键盘", price: 299)], total: 299)

        await sut.checkout(order: order, payment: .creditCard)

        XCTAssertNotNil(sut.confirmation)
        XCTAssertEqual(sut.confirmation?.id, "ORD-001")
        XCTAssertNil(sut.errorMessage)
        XCTAssertEqual(paymentGateway.chargedAmount, 299)
        XCTAssertEqual(orderService.submitCallCount, 1)
    }

    func testCheckout_paymentFails_setsError() async {
        paymentGateway.chargeResult = .failure(PaymentError.declined)
        let order = Order(items: [], total: 0)

        await sut.checkout(order: order, payment: .creditCard)

        XCTAssertNil(sut.confirmation)
        XCTAssertNotNil(sut.errorMessage)
        XCTAssertEqual(orderService.submitCallCount, 0)
    }

    func testCheckout_success_tracksAnalytics() async {
        let order = Order(items: [OrderItem(name: "鼠标", price: 99)], total: 99)

        await sut.checkout(order: order, payment: .applePay)

        XCTAssertEqual(analytics.trackedEvents.count, 1)
        XCTAssertEqual(analytics.trackedEvents.first?.event, "order_completed")
        XCTAssertEqual(analytics.trackedEvents.first?.properties["order_id"], "ORD-001")
    }

    func testCheckout_failure_tracksFailureEvent() async {
        paymentGateway.chargeResult = .failure(PaymentError.declined)
        let order = Order(items: [], total: 0)

        await sut.checkout(order: order, payment: .creditCard)

        XCTAssertEqual(analytics.trackedEvents.first?.event, "order_failed")
    }
}
常见问题
Q：Mock 和 Stub 到底有什么区别？ Stub 关注输入控制——让依赖返回特定值来驱动被测代码走不同分支。Mock 关注输出验证——断言被测代码是否正确调用了依赖（调用次数、参数等）。实践中一个测试替身经常同时具备两种能力。


Q：每个 Protocol 都要手写 Mock 吗？太多样板代码了？ 可以使用代码生成工具减少样板。社区方案包括 Sourcery 和 Mockolo，它们通过扫描 Protocol 定义自动生成 Mock 类。Swift Macro 未来也可能简化这一过程。


Q：什么时候用 Fake 而不是 Mock？ 当依赖的行为复杂且测试需要多次交互时，Fake 更合适。例如用内存字典替代数据库——它有真实的读写行为，但不依赖外部存储。Mock 适合简单的"调用-返回"场景。


Q：URLProtocol Mock 和 Protocol Mock 该选哪个？ 如果代码架构良好，网络层已抽象为 Protocol，优先用 Protocol Mock——更简单、更快。URLProtocol Mock 适用于需要测试 URLSession 配置、请求头、重试策略等底层网络行为的场景。


Q：如何避免 Mock 与真实实现不一致？ 为 Protocol 编写契约测试（Contract Test）——一套测试同时验证真实实现和 Mock 实现都符合 Protocol 预期行为。当 Protocol 接口变更时，编译器会强制更新所有实现。


小结
    

  • 测试替身分为 Dummy、Stub、Mock、Spy、Fake 五种，根据测试需求选择合适类型。
    • 

  • Swift 社区主流采用 Protocol-based Mocking——定义 Protocol、构造器注入、手动实现替身。
    • 

  • Stub 控制输入（返回固定值），Mock 验证输出（断言调用行为），实践中常合二为一。
    • 

  • 依赖注入是可测试架构的基石——通过构造器注入 Protocol 类型，测试时替换为替身。
    • 

  • URLProtocol 可拦截 URLSession 请求，适合测试底层网络行为；架构层面优先用 Protocol 抽象。
    • 

  • 如果一个类难以构建测试替身，这是架构代码异味的信号——依赖过多或耦合过紧，需要重构。
    • 

  • 使用 Sourcery / Mockolo 等工具可自动生成 Mock 代码，减少手写样板。
