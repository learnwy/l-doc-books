Part I · iOSiOS 开发 简介测试 简介单元测试

单元测试

概述

单元测试是软件质量的第一道防线——它针对最小可测试单元（函数、方法、类）进行隔离验证，确保每个模块在独立运行时行为正确。在 iOS 开发中，Apple 提供了两套测试框架：经典的 XCTest 和 Swift 6 引入的 Swift Testing（@Test / @Suite / #expect）。两者可以在同一个 Target 中共存，逐步迁移。

良好的单元测试能在 CI 流水线中快速发现回归，缩短反馈循环，降低修复成本。理解测试框架的 API、组织方式和异步测试能力，是构建可维护测试套件的基础。

核心工具
XCTest 框架
XCTest 是 Xcode 内置的测试框架，测试类继承 XCTestCase，测试方法以 test 开头：


import XCTest
@testable import MyApp

final class CalculatorTests: XCTestCase {

    var calculator: Calculator!

    override func setUp() {
        super.setUp()
        calculator = Calculator()
    }

    override func tearDown() {
        calculator = nil
        super.tearDown()
    }

    func testAddition() {
        XCTAssertEqual(calculator.add(2, 3), 5)
    }

    func testDivisionByZero() {
        XCTAssertThrowsError(try calculator.divide(10, by: 0)) { error in
            XCTAssertEqual(error as? CalcError, .divisionByZero)
        }
    }
}

Swift Testing 框架
Swift Testing 使用宏标注替代继承，语法更简洁、表达力更强：


import Testing
@testable import MyApp

@Suite("Calculator Tests")
struct CalculatorTests {

    let calculator = Calculator()

    @Test("两个正整数相加")
    func addition() {
        #expect(calculator.add(2, 3) == 5)
    }

    @Test("除以零抛出错误")
    func divisionByZero() throws {
        #expect(throws: CalcError.divisionByZero) {
            try calculator.divide(10, by: 0)
        }
    }

    @Test("参数化加法验证", arguments: [
        (2, 3, 5),
        (0, 0, 0),
        (-1, 1, 0),
        (100, 200, 300)
    ])
    func parameterizedAddition(a: Int, b: Int, expected: Int) {
        #expect(calculator.add(a, b) == expected)
    }
}

工作流程
步骤一：识别待测单元
选择纯逻辑模块作为测试目标——ViewModel、Service、工具函数等不依赖 UI 的代码最适合单元测试。


步骤二：创建测试文件
在测试 Target 中创建文件，命名规范为 被测类名 + Tests，如 OrderServiceTests.swift。使用 @testable import 访问 internal 成员。


步骤三：编写测试方法
遵循 Arrange → Act → Assert 模式：


func testApplyDiscount_withValidCoupon_reducesTotal() {
    // Arrange
    let service = PricingService()
    let coupon = Coupon(code: "SAVE20", discountPercent: 20)

    // Act
    let finalPrice = service.applyDiscount(to: 100.0, coupon: coupon)

    // Assert
    XCTAssertEqual(finalPrice, 80.0, accuracy: 0.01)
}
步骤四：运行与调试
    

  • 快捷键：⌘U 运行全部测试，点击行号旁菱形按钮运行单个测试
    • 

  • 命令行：xcodebuild test -scheme MyApp -destination 'platform=iOS Simulator,name=iPhone 16'
    • 

  • 筛选：Test Navigator（⌘6）可按状态筛选失败用例
    • 



步骤五：持续集成
在 CI 中运行测试并收集覆盖率：


xcodebuild test \
  -scheme MyApp \
  -destination 'platform=iOS Simulator,name=iPhone 16' \
  -enableCodeCoverage YES \
  -resultBundlePath TestResults.xcresult
实战演示
以一个购物车价格计算模块为例，完整展示单元测试的编写过程。


被测代码：


struct CartItem {
    let name: String
    let price: Double
    let quantity: Int
}

enum CartError: Error, Equatable {
    case emptyCart
    case invalidQuantity
}

struct CartCalculator {

    func subtotal(for items: [CartItem]) throws -> Double {
        guard !items.isEmpty else { throw CartError.emptyCart }
        guard items.allSatisfy({ $0.quantity > 0 }) else {
            throw CartError.invalidQuantity
        }
        return items.reduce(0) { $0 + $1.price * Double($1.quantity) }
    }

    func total(for items: [CartItem], taxRate: Double) throws -> Double {
        let sub = try subtotal(for: items)
        return sub * (1 + taxRate)
    }
}
测试代码（XCTest 风格）：


import XCTest
@testable import MyApp

final class CartCalculatorTests: XCTestCase {

    var sut: CartCalculator!

    override func setUp() {
        super.setUp()
        sut = CartCalculator()
    }

    override func tearDown() {
        sut = nil
        super.tearDown()
    }

    func testSubtotal_withMultipleItems_returnsCorrectSum() throws {
        let items = [
            CartItem(name: "键盘", price: 299.0, quantity: 1),
            CartItem(name: "鼠标", price: 99.0, quantity: 2)
        ]
        let result = try sut.subtotal(for: items)
        XCTAssertEqual(result, 497.0, accuracy: 0.01)
    }

    func testSubtotal_withEmptyCart_throwsEmptyCartError() {
        XCTAssertThrowsError(try sut.subtotal(for: [])) { error in
            XCTAssertEqual(error as? CartError, .emptyCart)
        }
    }

    func testSubtotal_withZeroQuantity_throwsInvalidQuantity() {
        let items = [CartItem(name: "USB", price: 49.0, quantity: 0)]
        XCTAssertThrowsError(try sut.subtotal(for: items)) { error in
            XCTAssertEqual(error as? CartError, .invalidQuantity)
        }
    }

    func testTotal_withTax_appliesTaxRate() throws {
        let items = [CartItem(name: "显示器", price: 1000.0, quantity: 1)]
        let result = try sut.total(for: items, taxRate: 0.1)
        XCTAssertEqual(result, 1100.0, accuracy: 0.01)
    }
}
测试代码（Swift Testing 风格）：


import Testing
@testable import MyApp

@Suite("购物车计算器")
struct CartCalculatorTests {

    let sut = CartCalculator()

    @Test("多件商品的小计")
    func subtotalMultipleItems() throws {
        let items = [
            CartItem(name: "键盘", price: 299.0, quantity: 1),
            CartItem(name: "鼠标", price: 99.0, quantity: 2)
        ]
        let result = try sut.subtotal(for: items)
        #expect(abs(result - 497.0) < 0.01)
    }

    @Test("空购物车抛出错误")
    func subtotalEmptyCart() {
        #expect(throws: CartError.emptyCart) {
            try sut.subtotal(for: [])
        }
    }

    @Test("含税总价", arguments: [
        (1000.0, 0.1, 1100.0),
        (500.0, 0.2, 600.0),
        (200.0, 0.0, 200.0)
    ])
    func totalWithTax(price: Double, taxRate: Double, expected: Double) throws {
        let items = [CartItem(name: "商品", price: price, quantity: 1)]
        let result = try sut.total(for: items, taxRate: taxRate)
        #expect(abs(result - expected) < 0.01)
    }
}
async 测试
对异步方法使用 async 测试：


func testFetchUser_returnsValidUser() async throws {
    let service = UserService()
    let user = try await service.fetchUser(id: "u_001")
    XCTAssertEqual(user.name, "张三")
    XCTAssertFalse(user.email.isEmpty)
}
Swift Testing 中同样支持：


@Test("异步获取用户")
func fetchUser() async throws {
    let service = UserService()
    let user = try await service.fetchUser(id: "u_001")
    #expect(user.name == "张三")
}
常见问题
Q：@testable import 导入报错？ 确保测试 Target 的 Build Settings 中 Host Application 设置正确，且被测模块的 Enable Testability 在 Debug 配置下为 YES。


Q：测试执行顺序是随机的吗？ XCTest 默认按方法名字母顺序执行，但 Xcode 提供了随机化选项（Edit Scheme → Test → Options → Randomize execution order）。Swift Testing 默认并行执行。每个测试应独立，不依赖执行顺序。


Q：测试命名有什么推荐规范？ 推荐 test<被测方法>_<场景>_<预期结果> 格式，如 testLogin_withInvalidPassword_returnsAuthError。Swift Testing 中使用 @Test("可读描述") 更直观。


Q：setUp 中创建的对象什么时候销毁？ XCTest 的 tearDown() 在每个测试方法执行结束后调用，适合将 sut 置为 nil 以释放资源。Swift Testing 使用 struct，每个测试方法自动获得独立实例。


Q：如何测试 private 方法？ 不要直接测试 private 方法。通过其 public/internal 接口间接覆盖 private 逻辑。如果 private 方法复杂到需要单独测试，考虑将其提取为独立类型。


小结
    

  • XCTest 是 iOS 测试的基础框架，通过继承 XCTestCase、test 前缀方法和 XCTAssert 断言组织测试。
    • 

  • Swift Testing 使用 @Test、@Suite、#expect 宏，语法更现代，支持参数化测试和默认并行执行。
    • 

  • 测试方法遵循 Arrange → Act → Assert 三段式结构，保持单一职责。
    • 

  • setUp() / tearDown() 管理测试前后的状态初始化与清理，避免测试间相互影响。
    • 

  • async 测试直接使用 async throws 标记方法，无需手动管理期望（expectation）。
    • 

  • 测试命名应清晰表达被测方法、场景和预期结果，提高失败时的可诊断性。
    • 

  • 两套框架可在同一 Target 共存，建议新测试优先使用 Swift Testing，逐步迁移。
