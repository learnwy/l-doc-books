Part I · iOSiOS 开发 简介SwiftUI 基础 简介状态与绑定

状态与绑定

概述

SwiftUI 是一个数据驱动的框架：状态（State）是唯一的事实来源（Single Source of Truth），UI 是状态的函数。当状态变化时，框架自动重新计算受影响的视图并更新界面。

本节将全面学习 SwiftUI 的状态管理工具：@State、@Binding、@StateObject、@ObservedObject、@EnvironmentObject、@Environment，理解它们各自的使用场景，以及数据在视图层级中的流动方向。

基础用法
@State — 视图私有状态
@State 用于视图内部的简单值类型状态（String、Int、Bool、struct 等）。SwiftUI 管理其存储，视图重建时状态不会丢失。


struct CounterView: View {
    @State private var count = 0

    var body: some View {
        VStack(spacing: 16) {
            Text("计数: \(count)")
                .font(.title)

            HStack(spacing: 20) {
                Button("减少") { count -= 1 }
                Button("增加") { count += 1 }
            }
            .buttonStyle(.bordered)
        }
    }
}
关键要点：


    

  • @State 属性应始终标记为 private
    • 

  • 仅用于视图自己拥有的状态
    • 

  • 适用于值类型（struct、enum、基本类型）
    • 



@Binding — 双向绑定
@Binding 创建对另一个状态的双向引用，子视图可以读取和修改父视图的状态：


struct ParentView: View {
    @State private var isOn = false

    var body: some View {
        VStack {
            Text(isOn ? "已开启" : "已关闭")
                .font(.title)

            CustomToggle(isOn: $isOn)
        }
    }
}

struct CustomToggle: View {
    @Binding var isOn: Bool

    var body: some View {
        Button(action: { isOn.toggle() }) {
            HStack {
                Image(systemName: isOn ? "checkmark.circle.fill" : "circle")
                Text(isOn ? "开启" : "关闭")
            }
            .padding()
            .background(isOn ? .green.opacity(0.2) : .gray.opacity(0.2))
            .clipShape(Capsule())
        }
    }
}
通过 $ 前缀获取 Binding：$isOn 生成一个 Binding<Bool>，传递给子视图。


@StateObject — 创建引用类型状态
当状态是一个类（ObservableObject）且由当前视图创建和拥有时，使用 @StateObject：


class TimerModel: ObservableObject {
    @Published var seconds = 0
    private var timer: Timer?

    func start() {
        timer = Timer.scheduledTimer(withTimeInterval: 1, repeats: true) { [weak self] _ in
            self?.seconds += 1
        }
    }

    func stop() {
        timer?.invalidate()
        timer = nil
    }

    func reset() {
        stop()
        seconds = 0
    }
}

struct TimerView: View {
    @StateObject private var model = TimerModel()

    var body: some View {
        VStack(spacing: 16) {
            Text("\(model.seconds) 秒")
                .font(.system(size: 48, weight: .bold, design: .monospaced))

            HStack(spacing: 16) {
                Button("开始") { model.start() }
                Button("停止") { model.stop() }
                Button("重置") { model.reset() }
            }
            .buttonStyle(.bordered)
        }
    }
}
@ObservedObject — 观察外部引用
当 ObservableObject 由外部创建并传入当前视图时，使用 @ObservedObject：


struct TimerDisplayView: View {
    @ObservedObject var model: TimerModel

    var body: some View {
        Text("已经过 \(model.seconds) 秒")
            .font(.headline)
    }
}

struct ContainerView: View {
    @StateObject private var timer = TimerModel()

    var body: some View {
        VStack {
            TimerDisplayView(model: timer)
            Button("开始") { timer.start() }
        }
    }
}

@EnvironmentObject — 跨层级共享
@EnvironmentObject 允许在视图层级的任何深度访问共享对象，无需逐层传递：


class AppSettings: ObservableObject {
    @Published var accentColor: Color = .blue
    @Published var fontSize: CGFloat = 16
    @Published var isDarkMode: Bool = false
}

@main
struct MyApp: App {
    @StateObject private var settings = AppSettings()

    var body: some Scene {
        WindowGroup {
            RootView()
                .environmentObject(settings)
        }
    }
}

struct RootView: View {
    var body: some View {
        NavigationStack {
            VStack {
                ContentSection()
                SettingsLink()
            }
        }
    }
}

struct ContentSection: View {
    @EnvironmentObject var settings: AppSettings

    var body: some View {
        Text("内容区域")
            .font(.system(size: settings.fontSize))
            .foregroundStyle(settings.accentColor)
    }
}

struct SettingsLink: View {
    var body: some View {
        NavigationLink("设置") {
            SettingsView()
        }
    }
}

struct SettingsView: View {
    @EnvironmentObject var settings: AppSettings

    var body: some View {
        Form {
            ColorPicker("主题色", selection: $settings.accentColor)

            Slider(value: $settings.fontSize, in: 12...32) {
                Text("字体大小: \(Int(settings.fontSize))")
            }

            Toggle("深色模式", isOn: $settings.isDarkMode)
        }
        .navigationTitle("设置")
    }
}
注意：SettingsLink 本身不使用 settings，但其子视图 SettingsView 通过环境自动获取，无需 SettingsLink 中转。


@Environment — 系统环境值
@Environment 读取系统提供的环境值，如颜色方案、尺寸类别、无障碍设置等：


struct AdaptiveView: View {
    @Environment(\.colorScheme) var colorScheme
    @Environment(\.horizontalSizeClass) var sizeClass
    @Environment(\.dynamicTypeSize) var typeSize
    @Environment(\.dismiss) var dismiss

    var body: some View {
        VStack(spacing: 12) {
            Text("当前模式: \(colorScheme == .dark ? "深色" : "浅色")")

            Text("尺寸类别: \(sizeClass == .compact ? "紧凑" : "常规")")

            if typeSize.isAccessibilitySize {
                Text("无障碍大字体已开启")
                    .foregroundStyle(.orange)
            }

            Button("关闭") {
                dismiss()
            }
        }
        .padding()
    }
}

iOS 17+ Observable 宏
iOS 17 引入了 @Observable 宏，简化了 ObservableObject 的写法：


import Observation

@Observable
class UserProfile {
    var name = ""
    var email = ""
    var avatarURL: URL?
}

struct ProfileView: View {
    @State private var profile = UserProfile()

    var body: some View {
        Form {
            TextField("姓名", text: $profile.name)
            TextField("邮箱", text: $profile.email)
        }
    }
}

完整示例
一个综合运用所有状态管理工具的购物清单应用：


import SwiftUI

class ShoppingStore: ObservableObject {
    @Published var items: [ShoppingItem] = []
    @Published var budget: Double = 100.0

    var totalCost: Double {
        items.filter(\.isPurchased).reduce(0) { $0 + $1.price }
    }

    var remainingBudget: Double {
        budget - totalCost
    }

    func addItem(_ name: String, price: Double) {
        items.append(ShoppingItem(name: name, price: price))
    }

    func togglePurchased(_ item: ShoppingItem) {
        if let index = items.firstIndex(where: { $0.id == item.id }) {
            items[index].isPurchased.toggle()
        }
    }

    func removeItems(at offsets: IndexSet) {
        items.remove(atOffsets: offsets)
    }
}

struct ShoppingItem: Identifiable {
    let id = UUID()
    var name: String
    var price: Double
    var isPurchased: Bool = false
}

struct ShoppingListApp: View {
    @StateObject private var store = ShoppingStore()

    var body: some View {
        TabView {
            ShoppingListView()
                .tabItem {
                    Label("清单", systemImage: "cart")
                }

            BudgetView()
                .tabItem {
                    Label("预算", systemImage: "dollarsign.circle")
                }
        }
        .environmentObject(store)
    }
}

struct ShoppingListView: View {
    @EnvironmentObject var store: ShoppingStore
    @State private var showAddSheet = false

    var body: some View {
        NavigationStack {
            List {
                ForEach(store.items) { item in
                    ShoppingItemRow(item: item)
                }
                .onDelete { store.removeItems(at: $0) }
            }
            .navigationTitle("购物清单")
            .toolbar {
                Button(action: { showAddSheet = true }) {
                    Image(systemName: "plus")
                }
            }
            .sheet(isPresented: $showAddSheet) {
                AddItemView()
            }
            .overlay {
                if store.items.isEmpty {
                    ContentUnavailableView(
                        "清单为空",
                        systemImage: "cart",
                        description: Text("点击 + 添加购物项目")
                    )
                }
            }
        }
    }
}

struct ShoppingItemRow: View {
    @EnvironmentObject var store: ShoppingStore
    let item: ShoppingItem

    var body: some View {
        HStack {
            Image(systemName: item.isPurchased ? "checkmark.circle.fill" : "circle")
                .foregroundStyle(item.isPurchased ? .green : .gray)
                .onTapGesture { store.togglePurchased(item) }

            VStack(alignment: .leading) {
                Text(item.name)
                    .strikethrough(item.isPurchased)
                Text("¥\(item.price, specifier: "%.2f")")
                    .font(.caption)
                    .foregroundStyle(.secondary)
            }

            Spacer()
        }
    }
}

struct AddItemView: View {
    @EnvironmentObject var store: ShoppingStore
    @Environment(\.dismiss) var dismiss
    @State private var name = ""
    @State private var priceText = ""

    private var isValid: Bool {
        !name.isEmpty && Double(priceText) != nil
    }

    var body: some View {
        NavigationStack {
            Form {
                TextField("商品名称", text: $name)
                TextField("价格", text: $priceText)
                    .keyboardType(.decimalPad)
            }
            .navigationTitle("添加商品")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("取消") { dismiss() }
                }
                ToolbarItem(placement: .confirmationAction) {
                    Button("添加") {
                        if let price = Double(priceText) {
                            store.addItem(name, price: price)
                            dismiss()
                        }
                    }
                    .disabled(!isValid)
                }
            }
        }
    }
}

struct BudgetView: View {
    @EnvironmentObject var store: ShoppingStore

    var body: some View {
        NavigationStack {
            VStack(spacing: 24) {
                VStack(spacing: 8) {
                    Text("预算")
                        .font(.headline)
                    Text("¥\(store.budget, specifier: "%.0f")")
                        .font(.system(size: 48, weight: .bold, design: .rounded))

                    Slider(value: $store.budget, in: 0...500, step: 10) {
                        Text("预算")
                    }
                    .padding(.horizontal)
                }

                HStack(spacing: 20) {
                    BudgetCard(
                        title: "已花费",
                        amount: store.totalCost,
                        color: .orange
                    )
                    BudgetCard(
                        title: "剩余",
                        amount: store.remainingBudget,
                        color: store.remainingBudget >= 0 ? .green : .red
                    )
                }
                .padding(.horizontal)

                Spacer()
            }
            .padding(.top)
            .navigationTitle("预算管理")
        }
    }
}

struct BudgetCard: View {
    let title: String
    let amount: Double
    let color: Color

    var body: some View {
        VStack(spacing: 8) {
            Text(title)
                .font(.subheadline)
                .foregroundStyle(.secondary)
            Text("¥\(amount, specifier: "%.2f")")
                .font(.title2)
                .bold()
                .foregroundStyle(color)
        }
        .frame(maxWidth: .infinity)
        .padding()
        .background(color.opacity(0.1))
        .clipShape(RoundedRectangle(cornerRadius: 12))
    }
}

小结
    

  • SwiftUI 是数据驱动框架，状态变化自动触发 UI 更新
    • 

  • @State 用于视图私有的值类型状态，始终标记为 private
    • 

  • @Binding 通过 $ 前缀创建双向绑定，连接父子视图状态
    • 

  • @StateObject 创建并拥有引用类型对象，@ObservedObject 观察外部传入的对象
    • 

  • @EnvironmentObject 实现跨层级数据共享，避免逐层传递
    • 

  • @Environment 读取系统环境值（颜色方案、dismiss 操作等）
    • 

  • iOS 17+ 的 @Observable 宏大幅简化了可观察对象的声明方式
    • 

  • 选择正确的状态工具是 SwiftUI 开发中最重要的设计决策之一
    •
