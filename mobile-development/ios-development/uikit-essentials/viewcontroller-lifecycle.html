Part I · iOSiOS 开发 简介UIKit 要点 简介ViewController 生命周期

ViewController 生命周期

概述

UIViewController 是 UIKit 中最核心的类之一。每个 ViewController 从创建到销毁都会经历一系列生命周期回调，理解这些回调的调用顺序和适用场景是编写健壮 UIKit 代码的基础。

本篇将完整梳理 ViewController 的生命周期，包括每个阶段应该做什么、父子容器 VC 的嵌套关系，以及常见陷阱。

基础用法
生命周期各阶段职责
import UIKit

class ProfileViewController: UIViewController {

    private let userId: String

    init(userId: String) {
        self.userId = userId
        super.init(nibName: nil, bundle: nil)
    }

    required init?(coder: NSCoder) {
        fatalError("init(coder:) has not been implemented")
    }

    override func loadView() {
        let customView = UIView()
        customView.backgroundColor = .systemBackground
        self.view = customView
    }

    override func viewDidLoad() {
        super.viewDidLoad()
        setupUI()
        loadInitialData()
    }

    override func viewWillAppear(_ animated: Bool) {
        super.viewWillAppear(animated)
        refreshData()
        navigationController?.setNavigationBarHidden(false, animated: animated)
    }

    override func viewDidAppear(_ animated: Bool) {
        super.viewDidAppear(animated)
        startAnimations()
        reportScreenView()
    }

    override func viewWillDisappear(_ animated: Bool) {
        super.viewWillDisappear(animated)
        pauseAnimations()
    }

    override func viewDidDisappear(_ animated: Bool) {
        super.viewDidDisappear(animated)
        cancelPendingRequests()
    }

    deinit {
        NotificationCenter.default.removeObserver(self)
    }

    private func setupUI() { }
    private func loadInitialData() { }
    private func refreshData() { }
    private func startAnimations() { }
    private func reportScreenView() { }
    private func pauseAnimations() { }
    private func cancelPendingRequests() { }
}

进阶用法
viewWillLayoutSubviews 与 viewDidLayoutSubviews
当 view 的 bounds 发生变化（旋转、键盘弹出）时，layout 回调会被触发：


class ChartViewController: UIViewController {

    private let chartLayer = CAShapeLayer()

    override func viewDidLoad() {
        super.viewDidLoad()
        view.layer.addSublayer(chartLayer)
    }

    override func viewWillLayoutSubviews() {
        super.viewWillLayoutSubviews()
        chartLayer.frame = view.bounds
    }

    override func viewDidLayoutSubviews() {
        super.viewDidLayoutSubviews()
        redrawChart()
    }

    private func redrawChart() {
        let rect = chartLayer.bounds
        let path = UIBezierPath(ovalIn: rect.insetBy(dx: 20, dy: 20))
        chartLayer.path = path.cgPath
        chartLayer.strokeColor = UIColor.systemBlue.cgColor
        chartLayer.fillColor = UIColor.clear.cgColor
        chartLayer.lineWidth = 2
    }
}
父子 ViewController 容器
UIKit 允许将一个 VC 嵌入另一个 VC 中，形成容器-子 VC 关系。添加/移除子 VC 时必须按照正确的调用顺序，否则生命周期回调不会正确传递。


class ContainerViewController: UIViewController {

    private let headerVC = HeaderViewController()
    private let contentVC = ContentViewController()

    override func viewDidLoad() {
        super.viewDidLoad()
        addChildVC(headerVC, to: view)
        addChildVC(contentVC, to: view)
        layoutChildren()
    }

    private func addChildVC(_ child: UIViewController, to containerView: UIView) {
        addChild(child)
        containerView.addSubview(child.view)
        child.view.translatesAutoresizingMaskIntoConstraints = false
        child.didMove(toParent: self)
    }

    private func removeChildVC(_ child: UIViewController) {
        child.willMove(toParent: nil)
        child.view.removeFromSuperview()
        child.removeFromParent()
    }

    private func layoutChildren() {
        NSLayoutConstraint.activate([
            headerVC.view.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor),
            headerVC.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            headerVC.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            headerVC.view.heightAnchor.constraint(equalToConstant: 120),

            contentVC.view.topAnchor.constraint(equalTo: headerVC.view.bottomAnchor),
            contentVC.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            contentVC.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            contentVC.view.bottomAnchor.constraint(equalTo: view.bottomAnchor),
        ])
    }
}
动态切换子 VC
extension ContainerViewController {

    func switchContent(to newVC: UIViewController) {
        let oldVC = contentVC

        oldVC.willMove(toParent: nil)
        addChild(newVC)

        newVC.view.frame = oldVC.view.frame
        newVC.view.alpha = 0

        transition(
            from: oldVC,
            to: newVC,
            duration: 0.3,
            options: .transitionCrossDissolve
        ) {
            newVC.view.alpha = 1
        } completion: { _ in
            oldVC.removeFromParent()
            newVC.didMove(toParent: self)
        }
    }
}
完整示例
以下是一个带有 Tab 切换的容器 VC，展示完整的生命周期日志和子 VC 管理：


import UIKit

class TabContainerViewController: UIViewController {

    private let segmentedControl = UISegmentedControl(items: ["首页", "设置"])
    private var currentChild: UIViewController?
    private let childVCs: [UIViewController]

    init() {
        self.childVCs = [
            LifecycleLoggingVC(label: "首页"),
            LifecycleLoggingVC(label: "设置"),
        ]
        super.init(nibName: nil, bundle: nil)
    }

    required init?(coder: NSCoder) {
        fatalError("init(coder:) has not been implemented")
    }

    override func viewDidLoad() {
        super.viewDidLoad()
        view.backgroundColor = .systemBackground
        setupSegmentedControl()
        showChild(at: 0)
    }

    private func setupSegmentedControl() {
        segmentedControl.selectedSegmentIndex = 0
        segmentedControl.addTarget(self, action: #selector(segmentChanged), for: .valueChanged)
        segmentedControl.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(segmentedControl)
        NSLayoutConstraint.activate([
            segmentedControl.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor, constant: 8),
            segmentedControl.centerXAnchor.constraint(equalTo: view.centerXAnchor),
        ])
    }

    @objc private func segmentChanged() {
        showChild(at: segmentedControl.selectedSegmentIndex)
    }

    private func showChild(at index: Int) {
        let newChild = childVCs[index]
        guard newChild !== currentChild else { return }

        if let old = currentChild {
            old.willMove(toParent: nil)
            old.view.removeFromSuperview()
            old.removeFromParent()
        }

        addChild(newChild)
        newChild.view.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(newChild.view)
        NSLayoutConstraint.activate([
            newChild.view.topAnchor.constraint(equalTo: segmentedControl.bottomAnchor, constant: 16),
            newChild.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            newChild.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            newChild.view.bottomAnchor.constraint(equalTo: view.bottomAnchor),
        ])
        newChild.didMove(toParent: self)
        currentChild = newChild
    }
}

class LifecycleLoggingVC: UIViewController {

    private let label: String

    init(label: String) {
        self.label = label
        super.init(nibName: nil, bundle: nil)
        print("[\(label)] init")
    }

    required init?(coder: NSCoder) {
        fatalError("init(coder:) has not been implemented")
    }

    override func viewDidLoad() {
        super.viewDidLoad()
        print("[\(label)] viewDidLoad")

        view.backgroundColor = .systemGroupedBackground
        let titleLabel = UILabel()
        titleLabel.text = label
        titleLabel.font = .preferredFont(forTextStyle: .largeTitle)
        titleLabel.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(titleLabel)
        NSLayoutConstraint.activate([
            titleLabel.centerXAnchor.constraint(equalTo: view.centerXAnchor),
            titleLabel.centerYAnchor.constraint(equalTo: view.centerYAnchor),
        ])
    }

    override func viewWillAppear(_ animated: Bool) {
        super.viewWillAppear(animated)
        print("[\(label)] viewWillAppear")
    }

    override func viewDidAppear(_ animated: Bool) {
        super.viewDidAppear(animated)
        print("[\(label)] viewDidAppear")
    }

    override func viewWillDisappear(_ animated: Bool) {
        super.viewWillDisappear(animated)
        print("[\(label)] viewWillDisappear")
    }

    override func viewDidDisappear(_ animated: Bool) {
        super.viewDidDisappear(animated)
        print("[\(label)] viewDidDisappear")
    }

    deinit {
        print("[\(label)] deinit")
    }
}
最佳实践




推荐做法
不推荐做法





在 viewDidLoad 中完成一次性 UI 初始化
在 init 中访问 self.view




在 viewWillAppear 中刷新动态数据
在 viewDidLoad 中做只需每次显示才需要的刷新




自定义 view 层级时重写 loadView
在 loadView 中调用 super.loadView()




始终调用 super 的生命周期方法
忘记调用 super.viewDidLoad() 等




使用 addChild / didMove(toParent:) 管理子 VC
直接 addSubview(child.view) 不调用容器 API




在 deinit 中清理观察者和外部引用
依赖 viewDidDisappear 做最终清理




在 viewDidLayoutSubviews 中处理依赖 frame 的布局
在 viewDidLoad 中读取 view.frame（此时尚未确定）





关于 loadView——仅在你需要完全替换默认 view 时才重写它。如果你使用 Storyboard 或只是添加子视图，直接在 viewDidLoad 中操作即可。重写 loadView 时不要调用 super.loadView()。


小结
    

  • ViewController 生命周期遵循 init → loadView → viewDidLoad → viewWillAppear → viewDidAppear → viewWillDisappear → viewDidDisappear → deinit 的固定顺序
    • 

  • viewDidLoad 只调用一次，适合一次性初始化；viewWillAppear/viewDidAppear 每次显示都会调用
    • 

  • viewWillLayoutSubviews 和 viewDidLayoutSubviews 在布局变化时触发，适合处理依赖 frame 的逻辑
    • 

  • 容器 VC 必须遵循 addChild → addSubview → didMove(toParent:) 的三步添加顺序
    • 

  • 移除子 VC 时使用 willMove(toParent: nil) → removeFromSuperview → removeFromParent
    • 

  • 始终在重写的生命周期方法中调用 super
    • 

  • deinit 是最终清理的位置，确保没有循环引用阻止其被调用
    •
