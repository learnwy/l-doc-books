UIKit 与 SwiftUI 互操作

概述

在实际项目中，UIKit 和 SwiftUI 经常需要共存。Apple 提供了一套桥接 API，让两个框架可以无缝嵌入对方：在 UIKit 中使用 SwiftUI 视图，或在 SwiftUI 中使用 UIKit 组件。

本篇将详细讲解四大桥接工具以及 Coordinator 模式和混合导航栈的实现。





桥接工具
方向
用途





UIHostingController
SwiftUI → UIKit
将 SwiftUI View 包装为 UIViewController




UIViewRepresentable
UIKit → SwiftUI
将 UIView 包装为 SwiftUI View




UIViewControllerRepresentable
UIKit → SwiftUI
将 UIViewController 包装为 SwiftUI View

基础用法
UIHostingController：在 UIKit 中使用 SwiftUI
UIHostingController 是一个泛型 UIViewController，将任意 SwiftUI View 作为其根视图。


import SwiftUI
import UIKit

struct ProfileCardView: View {
    let name: String
    let role: String

    var body: some View {
        VStack(spacing: 12) {
            Image(systemName: "person.circle.fill")
                .font(.system(size: 64))
                .foregroundStyle(.blue)
            Text(name)
                .font(.title2.bold())
            Text(role)
                .font(.subheadline)
                .foregroundStyle(.secondary)
        }
        .padding(24)
        .background(.ultraThinMaterial, in: RoundedRectangle(cornerRadius: 16))
    }
}

class SettingsViewController: UIViewController {

    override func viewDidLoad() {
        super.viewDidLoad()
        view.backgroundColor = .systemBackground
        title = "设置"
        embedSwiftUICard()
    }

    private func embedSwiftUICard() {
        let swiftUIView = ProfileCardView(name: "张三", role: "iOS 工程师")
        let hostingController = UIHostingController(rootView: swiftUIView)

        addChild(hostingController)
        hostingController.view.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(hostingController.view)

        NSLayoutConstraint.activate([
            hostingController.view.centerXAnchor.constraint(equalTo: view.centerXAnchor),
            hostingController.view.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor, constant: 32),
        ])

        hostingController.didMove(toParent: self)
    }
}
UIViewRepresentable：在 SwiftUI 中使用 UIView
import SwiftUI
import UIKit
import MapKit

struct MapView: UIViewRepresentable {

    var coordinate: CLLocationCoordinate2D

    func makeUIView(context: Context) -> MKMapView {
        let mapView = MKMapView()
        mapView.isZoomEnabled = true
        mapView.isScrollEnabled = true
        return mapView
    }

    func updateUIView(_ mapView: MKMapView, context: Context) {
        let region = MKCoordinateRegion(
            center: coordinate,
            latitudinalMeters: 5000,
            longitudinalMeters: 5000
        )
        mapView.setRegion(region, animated: true)
    }
}

struct LocationScreen: View {
    var body: some View {
        MapView(coordinate: CLLocationCoordinate2D(latitude: 39.9042, longitude: 116.4074))
            .ignoresSafeArea()
            .navigationTitle("地图")
    }
}
进阶用法
Coordinator 模式：处理 Delegate 回调
UIKit 组件大量使用 Delegate 模式，而 SwiftUI 是值类型。Coordinator 作为引用类型的中间人，将 UIKit 的回调桥接到 SwiftUI 的数据流。


import SwiftUI
import UIKit

struct ImagePicker: UIViewControllerRepresentable {

    @Binding var selectedImage: UIImage?
    @Environment(\.dismiss) private var dismiss

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIViewController(context: Context) -> UIImagePickerController {
        let picker = UIImagePickerController()
        picker.delegate = context.coordinator
        picker.sourceType = .photoLibrary
        return picker
    }

    func updateUIViewController(_ uiViewController: UIImagePickerController, context: Context) {}

    class Coordinator: NSObject, UIImagePickerControllerDelegate, UINavigationControllerDelegate {

        let parent: ImagePicker

        init(_ parent: ImagePicker) {
            self.parent = parent
        }

        func imagePickerController(
            _ picker: UIImagePickerController,
            didFinishPickingMediaWithInfo info: [UIImagePickerController.InfoKey: Any]
        ) {
            parent.selectedImage = info[.originalImage] as? UIImage
            parent.dismiss()
        }

        func imagePickerControllerDidCancel(_ picker: UIImagePickerController) {
            parent.dismiss()
        }
    }
}

struct PhotoSelectionView: View {
    @State private var selectedImage: UIImage?
    @State private var showPicker = false

    var body: some View {
        VStack(spacing: 20) {
            if let image = selectedImage {
                Image(uiImage: image)
                    .resizable()
                    .scaledToFit()
                    .frame(height: 300)
                    .clipShape(RoundedRectangle(cornerRadius: 12))
            } else {
                RoundedRectangle(cornerRadius: 12)
                    .fill(.gray.opacity(0.2))
                    .frame(height: 300)
                    .overlay {
                        Image(systemName: "photo")
                            .font(.largeTitle)
                            .foregroundStyle(.secondary)
                    }
            }

            Button("选择照片") {
                showPicker = true
            }
            .buttonStyle(.borderedProminent)
        }
        .padding()
        .sheet(isPresented: $showPicker) {
            ImagePicker(selectedImage: $selectedImage)
        }
    }
}
双向数据传递：UIKit → SwiftUI
使用 @Binding 或闭包将 UIKit 中的变化传回 SwiftUI：


struct RatingSlider: UIViewRepresentable {

    @Binding var value: Float

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIView(context: Context) -> UISlider {
        let slider = UISlider()
        slider.minimumValue = 0
        slider.maximumValue = 5
        slider.addTarget(context.coordinator, action: #selector(Coordinator.valueChanged(_:)), for: .valueChanged)
        return slider
    }

    func updateUIView(_ slider: UISlider, context: Context) {
        slider.value = value
    }

    class Coordinator: NSObject {
        var parent: RatingSlider

        init(_ parent: RatingSlider) {
            self.parent = parent
        }

        @objc func valueChanged(_ sender: UISlider) {
            parent.value = sender.value
        }
    }
}

struct RatingView: View {
    @State private var rating: Float = 2.5

    var body: some View {
        VStack(spacing: 20) {
            Text("评分：\(String(format: "%.1f", rating))")
                .font(.title)
            RatingSlider(value: $rating)
                .frame(height: 44)
                .padding(.horizontal)
        }
    }
}
混合导航栈
在 UIKit 导航栈中 push SwiftUI 页面，或在 SwiftUI NavigationStack 中嵌入 UIKit 页面：


class MainNavigationController: UINavigationController {

    override func viewDidLoad() {
        super.viewDidLoad()

        let listVC = UIHostingController(rootView: ItemListView(onSelect: { [weak self] item in
            self?.pushUIKitDetail(for: item)
        }))
        listVC.title = "商品列表"
        viewControllers = [listVC]
    }

    private func pushUIKitDetail(for item: String) {
        let detailVC = LegacyDetailViewController(item: item)
        pushViewController(detailVC, animated: true)
    }
}

struct ItemListView: View {
    let onSelect: (String) -> Void
    let items = ["iPhone", "iPad", "MacBook", "AirPods"]

    var body: some View {
        List(items, id: \.self) { item in
            Button(item) {
                onSelect(item)
            }
        }
    }
}

class LegacyDetailViewController: UIViewController {

    private let item: String

    init(item: String) {
        self.item = item
        super.init(nibName: nil, bundle: nil)
        title = item
    }

    required init?(coder: NSCoder) {
        fatalError("init(coder:) has not been implemented")
    }

    override func viewDidLoad() {
        super.viewDidLoad()
        view.backgroundColor = .systemBackground

        let label = UILabel()
        label.text = "UIKit 详情页：\(item)"
        label.font = .preferredFont(forTextStyle: .title1)
        label.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(label)

        let swiftUISection = UIHostingController(rootView: RelatedProductsView(currentItem: item))
        addChild(swiftUISection)
        swiftUISection.view.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(swiftUISection.view)
        swiftUISection.didMove(toParent: self)

        NSLayoutConstraint.activate([
            label.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor, constant: 24),
            label.centerXAnchor.constraint(equalTo: view.centerXAnchor),

            swiftUISection.view.topAnchor.constraint(equalTo: label.bottomAnchor, constant: 32),
            swiftUISection.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            swiftUISection.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            swiftUISection.view.heightAnchor.constraint(equalToConstant: 200),
        ])
    }
}

struct RelatedProductsView: View {
    let currentItem: String

    var body: some View {
        VStack(alignment: .leading) {
            Text("相关推荐")
                .font(.headline)
                .padding(.horizontal)
            ScrollView(.horizontal, showsIndicators: false) {
                HStack(spacing: 12) {
                    ForEach(["配件A", "配件B", "配件C"], id: \.self) { item in
                        RoundedRectangle(cornerRadius: 10)
                            .fill(.blue.opacity(0.15))
                            .frame(width: 120, height: 120)
                            .overlay {
                                Text(item)
                                    .font(.callout)
                            }
                    }
                }
                .padding(.horizontal)
            }
        }
    }
}
完整示例
一个完整的混合应用：SwiftUI 主界面 + UIKit 相机页面 + 双向数据同步：


import SwiftUI
import UIKit

@Observable
class AppState {
    var capturedPhotos: [UIImage] = []
    var userName: String = "用户"
}

struct HybridAppView: View {
    @State private var appState = AppState()
    @State private var showCamera = false

    var body: some View {
        NavigationStack {
            VStack(spacing: 20) {
                Text("你好，\(appState.userName)")
                    .font(.title)

                Text("已拍摄 \(appState.capturedPhotos.count) 张照片")
                    .foregroundStyle(.secondary)

                ScrollView(.horizontal) {
                    HStack {
                        ForEach(Array(appState.capturedPhotos.enumerated()), id: \.offset) { _, photo in
                            Image(uiImage: photo)
                                .resizable()
                                .scaledToFill()
                                .frame(width: 80, height: 80)
                                .clipShape(RoundedRectangle(cornerRadius: 8))
                        }
                    }
                    .padding(.horizontal)
                }

                Button("打开相机") {
                    showCamera = true
                }
                .buttonStyle(.borderedProminent)
            }
            .navigationTitle("混合应用")
            .sheet(isPresented: $showCamera) {
                CameraViewWrapper(appState: appState)
            }
        }
    }
}

struct CameraViewWrapper: UIViewControllerRepresentable {

    let appState: AppState

    func makeCoordinator() -> Coordinator {
        Coordinator(appState: appState)
    }

    func makeUIViewController(context: Context) -> UINavigationController {
        let cameraVC = SimulatedCameraViewController()
        cameraVC.delegate = context.coordinator
        return UINavigationController(rootViewController: cameraVC)
    }

    func updateUIViewController(_ uiViewController: UINavigationController, context: Context) {}

    class Coordinator: NSObject, SimulatedCameraDelegate {
        let appState: AppState

        init(appState: AppState) {
            self.appState = appState
        }

        func didCapture(image: UIImage) {
            appState.capturedPhotos.append(image)
        }
    }
}

protocol SimulatedCameraDelegate: AnyObject {
    func didCapture(image: UIImage)
}

class SimulatedCameraViewController: UIViewController {

    weak var delegate: SimulatedCameraDelegate?

    override func viewDidLoad() {
        super.viewDidLoad()
        view.backgroundColor = .black
        title = "相机"

        let captureButton = UIButton(type: .system)
        captureButton.setTitle("模拟拍照", for: .normal)
        captureButton.titleLabel?.font = .preferredFont(forTextStyle: .title2)
        captureButton.setTitleColor(.white, for: .normal)
        captureButton.backgroundColor = .systemRed
        captureButton.layer.cornerRadius = 35
        captureButton.translatesAutoresizingMaskIntoConstraints = false
        captureButton.addTarget(self, action: #selector(capture), for: .touchUpInside)
        view.addSubview(captureButton)

        NSLayoutConstraint.activate([
            captureButton.centerXAnchor.constraint(equalTo: view.centerXAnchor),
            captureButton.bottomAnchor.constraint(equalTo: view.safeAreaLayoutGuide.bottomAnchor, constant: -40),
            captureButton.widthAnchor.constraint(equalToConstant: 70),
            captureButton.heightAnchor.constraint(equalToConstant: 70),
        ])

        navigationItem.leftBarButtonItem = UIBarButtonItem(
            systemItem: .close,
            primaryAction: UIAction { [weak self] _ in
                self?.dismiss(animated: true)
            }
        )
    }

    @objc private func capture() {
        let renderer = UIGraphicsImageRenderer(size: CGSize(width: 200, height: 200))
        let image = renderer.image { ctx in
            UIColor(
                hue: CGFloat.random(in: 0...1),
                saturation: 0.6,
                brightness: 0.9,
                alpha: 1
            ).setFill()
            ctx.fill(CGRect(x: 0, y: 0, width: 200, height: 200))
        }
        delegate?.didCapture(image: image)
    }
}

用 UIHostingController 逐步迁移到 SwiftUI
一次性重写整个 UIKit 项目




通过 Coordinator 处理所有 Delegate 回调
在 UIViewRepresentable 的 struct 中直接持有引用类型

updateUIView 中检查变化再更新，避免重复操作

在 dismantleUIView 中清理定时器/观察者

混合导航使用 UINavigationController 作为容器
在 SwiftUI NavigationStack 中嵌套 UINavigationController




用 sizeThatFits (iOS 16+) 让 UIKit View 正确参与 SwiftUI 布局
给 UIHostingController.view 设固定 frame





小结
    

  • UIHostingController 将 SwiftUI View 包装为 UIViewController，是在 UIKit 项目中引入 SwiftUI 的首选方式
    • 

  • UIViewRepresentable 将 UIView 桥接到 SwiftUI，需实现 makeUIView 和 updateUIView
    • 

  • UIViewControllerRepresentable 将 UIViewController 桥接到 SwiftUI，适用于相机、地图等复杂组件
    • 

  • Coordinator 是 Delegate/Target-Action 到 SwiftUI 数据流的桥梁，在 makeCoordinator() 中创建
    • 

  • 双向数据传递通过 @Binding + Coordinator 的回调实现
    • 

  • 混合导航栈中，推荐用 UIKit 的 UINavigationController 作为顶层容器，SwiftUI 页面通过 UIHostingController 推入
    • 

  • 渐进式迁移是最佳策略——新页面用 SwiftUI，旧页面保留 UIKit，通过桥接 API 连接
    •
