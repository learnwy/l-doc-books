iOS 面试进阶题型

本篇承接 iOS 面试常见题型，覆盖四类更偏工程深度与系统思维的面试题型：性能优化、并发编程、网络通信与系统设计。每类给出核心要点与示例代码，配合前篇的语言/框架/架构题组成完整的 iOS 面试准备。

核心概念

  • 性能优化考的是定位与度量：用 Instruments/MetricKit 找到瓶颈，再针对启动、内存、UI、存储分层优化，而非凭感觉改代码。
  • 并发编程考的是安全模型：从 GCD 到 async/await 与 Actor，核心是数据隔离、避免数据竞争，以及主线程约束。
  • 网络通信考的是协议理解与工程取舍：HTTP/缓存/安全基础上的请求建模、错误处理、重试与离线策略。
  • 系统设计考的是端到端拆解：把一个端侧功能拆成 API、缓存、状态、同步与容量，并讲清权衡与边界。

五、性能优化

要点

性能题考察对 Instruments、内存管理和渲染流水线的理解：

优化方向 关键指标 工具
启动时间 Pre-main + Post-main App Launch Template

网络 请求数 / 响应时间 Network Template

示例代码

列表性能优化——图片缓存 + 预加载：

actor ImageCache {
    static let shared = ImageCache()

    private var cache: [URL: UIImage] = [:]
    private var inFlightTasks: [URL: Task<UIImage, Error>] = [:]

    func image(for url: URL) async throws -> UIImage {
        if let cached = cache[url] {
            return cached
        }

        if let existingTask = inFlightTasks[url] {
            return try await existingTask.value
        }

        let task = Task<UIImage, Error> {
            let (data, _) = try await URLSession.shared.data(from: url)
            guard let image = UIImage(data: data) else {
                throw ImageCacheError.decodingFailed
            }
            return image
        }

        inFlightTasks[url] = task

        do {
            let image = try await task.value
            cache[url] = image
            inFlightTasks[url] = nil
            return image
        } catch {
            inFlightTasks[url] = nil
            throw error
        }
    }

    func prefetch(urls: [URL]) {
        for url in urls where cache[url] == nil {
            Task {
                _ = try? await image(for: url)
            }
        }
    }

    enum ImageCacheError: Error {
        case decodingFailed
    }
}

要点

概念 核心点

示例代码

经典面试题——用 Actor 实现线程安全的计数器：

actor SafeCounter {
    private var value = 0

    func increment() -> Int {
        value += 1
        return value
    }

    func decrement() -> Int {
        value -= 1
        return value
    }

    var current: Int {
        value
    }
}

actor RateLimiter {
    private let maxRequests: Int
    private let interval: TimeInterval
    private var timestamps: [Date] = []

    init(maxRequests: Int, interval: TimeInterval) {
        self.maxRequests = maxRequests
        self.interval = interval
    }

    func shouldAllow() -> Bool {
        let now = Date()
        timestamps = timestamps.filter { now.timeIntervalSince($0) < interval }
        guard timestamps.count < maxRequests else { return false }
        timestamps.append(now)
        return true
    }
}

GCD 经典面试题——死锁场景：

func deadlockDemo() {
    let queue = DispatchQueue(label: "serial")

    queue.sync {
        print("Task 1")
        // ❌ 死锁：在串行队列的同步任务中再次 sync 到同一队列
        // queue.sync { print("Task 2") }
    }

    // ✅ 避免死锁的方式
    queue.async {
        print("Task 1")
        queue.async {
            print("Task 2")
        }
    }
}

七、网络

要点

网络题考察 URLSession、REST API 设计和错误处理：

示例代码

泛型网络层：

enum HTTPMethod: String {
    case get = "GET"
    case post = "POST"
    case put = "PUT"
    case delete = "DELETE"
}

protocol APIEndpoint {
    var path: String { get }
    var method: HTTPMethod { get }
    var headers: [String: String] { get }
    var body: Data? { get }
}

struct APIClient {
    private let baseURL: URL
    private let session: URLSession

    init(baseURL: URL, session: URLSession = .shared) {
        self.baseURL = baseURL
        self.session = session
    }

    func request<T: Decodable>(_ endpoint: APIEndpoint) async throws -> T {
        let url = baseURL.appending(path: endpoint.path)
        var request = URLRequest(url: url)
        request.httpMethod = endpoint.method.rawValue
        request.httpBody = endpoint.body
        for (key, value) in endpoint.headers {
            request.setValue(value, forHTTPHeaderField: key)
        }

        let (data, response) = try await session.data(for: request)

        guard let httpResponse = response as? HTTPURLResponse else {
            throw APIError.invalidResponse
        }

        guard (200..<300).contains(httpResponse.statusCode) else {
            throw APIError.httpError(statusCode: httpResponse.statusCode, data: data)
        }

        return try JSONDecoder().decode(T.self, from: data)
    }
}

enum APIError: Error, LocalizedError {
    case invalidResponse
    case httpError(statusCode: Int, data: Data)
    case decodingError(Error)

    var errorDescription: String? {
        switch self {
        case .invalidResponse:
            return "无效的服务器响应"
        case .httpError(let code, _):
            return "HTTP 错误: \(code)"
        case .decodingError(let error):
            return "数据解析失败: \(error.localizedDescription)"
        }
    }
}

八、系统设计

要点

系统设计题考察全局思维——如何设计一个完整的功能模块：

常见系统设计题：

  • 设计一个图片加载和缓存系统
  • 设计一个消息推送处理系统
  • 设计一个离线优先的数据同步方案
  • 设计一个路由/导航系统

示例代码

简化的图片加载系统设计：

protocol ImageLoader: Sendable {
    func load(from url: URL) async throws -> UIImage
}

actor ImagePipeline: ImageLoader {
    private let memoryCache: NSCache<NSURL, UIImage>
    private let diskCache: DiskCache
    private let downloader: ImageDownloader
    private var activeTasks: [URL: Task<UIImage, Error>] = [:]

    init(
        memoryCacheLimit: Int = 50_000_000,
        diskCachePath: String = "image_cache"
    ) {
        let cache = NSCache<NSURL, UIImage>()
        cache.totalCostLimit = memoryCacheLimit
        self.memoryCache = cache
        self.diskCache = DiskCache(directory: diskCachePath)
        self.downloader = ImageDownloader()
    }

    func load(from url: URL) async throws -> UIImage {
        if let memoryHit = memoryCache.object(forKey: url as NSURL) {
            return memoryHit
        }

        if let diskHit = await diskCache.load(for: url) {
            memoryCache.setObject(diskHit, forKey: url as NSURL)
            return diskHit
        }

        if let activeTask = activeTasks[url] {
            return try await activeTask.value
        }

        let task = Task<UIImage, Error> {
            let image = try await downloader.download(from: url)
            memoryCache.setObject(image, forKey: url as NSURL)
            await diskCache.save(image, for: url)
            return image
        }

        activeTasks[url] = task
        defer { activeTasks[url] = nil }
        return try await task.value
    }

    func clearCache() async {
        memoryCache.removeAllObjects()
        await diskCache.clearAll()
    }
}

struct DiskCache: Sendable {
    let directory: String
    func load(for url: URL) async -> UIImage? { nil }
    func save(_ image: UIImage, for url: URL) async {}
    func clearAll() async {}
}

struct ImageDownloader: Sendable {
    func download(from url: URL) async throws -> UIImage {
        let (data, _) = try await URLSession.shared.data(from: url)
        guard let image = UIImage(data: data) else {
            throw URLError(.cannotDecodeContentData)
        }
        return image
    }
}

小结

  • 性能优化先度量后动手，优先解决启动、内存与 UI 卡顿这三类用户最敏感的问题。
  • 并发的本质是用结构化的隔离（async/await、Actor）替代手工线程管理，安全永远先于性能。
  • 网络题要同时讲清协议层（HTTP/缓存/HTTPS）与工程层（建模、重试、离线、错误传播）。
  • 系统设计题用「接口—缓存—状态—同步—容量」的拆解展示完整工程视角，并主动指出取舍。
  • 这四类题与前篇的 Swift/UIKit/SwiftUI/架构题互为表里，完整清单见 iOS 面试常见题型。
