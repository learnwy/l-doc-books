Core Data 持久化

概述

Core Data 是 Apple 提供的对象图管理和持久化框架——它不是数据库，而是对象图（object graph）的管理器，底层默认使用 SQLite 作为存储引擎。本文将从核心架构组件入手，依次讲解 CRUD 操作、复杂查询、关系建模、数据迁移以及性能优化。

基础用法
初始化 NSPersistentContainer
import CoreData

class PersistenceController {
    static let shared = PersistenceController()

    let container: NSPersistentContainer

    init(inMemory: Bool = false) {
        container = NSPersistentContainer(name: "AppModel")

        if inMemory {
            container.persistentStoreDescriptions.first?.url = URL(fileURLWithPath: "/dev/null")
        }

        container.loadPersistentStores { description, error in
            if let error {
                fatalError("Core Data 加载失败: \(error)")
            }
        }

        container.viewContext.automaticallyMergesChangesFromParent = true
        container.viewContext.mergePolicy = NSMergeByPropertyObjectTrumpMergePolicy
    }

    var viewContext: NSManagedObjectContext {
        container.viewContext
    }

    func newBackgroundContext() -> NSManagedObjectContext {
        container.newBackgroundContext()
    }

    func save() {
        let context = viewContext
        guard context.hasChanges else { return }
        do {
            try context.save()
        } catch {
            print("保存失败: \(error)")
        }
    }
}
定义 NSManagedObject 子类
在 .xcdatamodeld 中创建 Entity 后，可手动或自动生成代码。手动编写方式如下：


import CoreData

@objc(TaskItem)
public class TaskItem: NSManagedObject {
    @NSManaged public var id: UUID
    @NSManaged public var title: String
    @NSManaged public var content: String?
    @NSManaged public var isCompleted: Bool
    @NSManaged public var createdAt: Date
    @NSManaged public var category: Category?
}

extension TaskItem {
    @nonobjc public class func fetchRequest() -> NSFetchRequest<TaskItem> {
        NSFetchRequest<TaskItem>(entityName: "TaskItem")
    }

    static func create(
        in context: NSManagedObjectContext,
        title: String,
        content: String? = nil
    ) -> TaskItem {
        let item = TaskItem(context: context)
        item.id = UUID()
        item.title = title
        item.content = content
        item.isCompleted = false
        item.createdAt = Date()
        return item
    }
}

NSPredicate 复杂查询
let titleFilter = NSPredicate(format: "title CONTAINS[cd] %@", "Swift")

let dateRange = NSPredicate(
    format: "createdAt >= %@ AND createdAt <= %@",
    Calendar.current.startOfDay(for: Date()) as NSDate,
    Date() as NSDate
)

let compound = NSCompoundPredicate(andPredicateWithSubpredicates: [titleFilter, dateRange])

let inSet = NSPredicate(format: "category.name IN %@", ["工作", "学习"])

let request = TaskItem.fetchRequest()

request.fetchLimit = 20

NSFetchedResultsController（UIKit 集成）
import UIKit
import CoreData

class TaskListViewController: UITableViewController, NSFetchedResultsControllerDelegate {
    private var fetchedResultsController: NSFetchedResultsController<TaskItem>!

    override func viewDidLoad() {
        super.viewDidLoad()

        let request = TaskItem.fetchRequest()
        request.sortDescriptors = [
            NSSortDescriptor(keyPath: \TaskItem.isCompleted, ascending: true),
            NSSortDescriptor(keyPath: \TaskItem.createdAt, ascending: false)
        ]

        fetchedResultsController = NSFetchedResultsController(
            fetchRequest: request,
            managedObjectContext: PersistenceController.shared.viewContext,
            sectionNameKeyPath: "isCompleted",
            cacheName: nil
        )
        fetchedResultsController.delegate = self

        try? fetchedResultsController.performFetch()
    }

    override func numberOfSections(in tableView: UITableView) -> Int {
        fetchedResultsController.sections?.count ?? 0
    }

    override func tableView(_ tableView: UITableView, numberOfRowsInSection section: Int) -> Int {
        fetchedResultsController.sections?[section].numberOfObjects ?? 0
    }

    override func tableView(_ tableView: UITableView, cellForRowAt indexPath: IndexPath) -> UITableViewCell {
        let cell = tableView.dequeueReusableCell(withIdentifier: "Cell", for: indexPath)
        let task = fetchedResultsController.object(at: indexPath)
        cell.textLabel?.text = task.title
        cell.accessoryType = task.isCompleted ? .checkmark : .none
        return cell
    }

    func controllerDidChangeContent(_ controller: NSFetchedResultsController<any NSFetchRequestResult>) {
        tableView.reloadData()
    }
}
关系建模
@objc(Category)
public class Category: NSManagedObject {
    @NSManaged public var id: UUID
    @NSManaged public var name: String
    @NSManaged public var color: String?
    @NSManaged public var tasks: NSSet?
}

extension Category {
    @objc(addTasksObject:)
    @NSManaged public func addToTasks(_ value: TaskItem)

    @objc(removeTasksObject:)
    @NSManaged public func removeFromTasks(_ value: TaskItem)

    var taskArray: [TaskItem] {
        let set = tasks as? Set<TaskItem> ?? []
        return set.sorted { $0.createdAt > $1.createdAt }
    }
}
后台上下文批量操作
func importTasks(_ dtos: [TaskDTO]) async throws {
    let context = PersistenceController.shared.newBackgroundContext()
    context.mergePolicy = NSMergeByPropertyObjectTrumpMergePolicy

    try await context.perform {
        for dto in dtos {
            let request = TaskItem.fetchRequest()
            request.predicate = NSPredicate(format: "id == %@", dto.id as CVarArg)
            request.fetchLimit = 1

            let existing = try context.fetch(request).first

            let task = existing ?? TaskItem(context: context)
            task.id = dto.id
            task.title = dto.title
            task.content = dto.content
            task.isCompleted = dto.isCompleted
            task.createdAt = dto.createdAt
        }

        try context.save()
    }
}

struct TaskDTO {
    let id: UUID
    let title: String
    let content: String?
    let isCompleted: Bool
    let createdAt: Date
}
NSBatchDeleteRequest 高效删除
func deleteAllCompletedTasks() throws {
    let request = NSFetchRequest<NSFetchRequestResult>(entityName: "TaskItem")
    request.predicate = NSPredicate(format: "isCompleted == YES")

    let batchDelete = NSBatchDeleteRequest(fetchRequest: request)
    batchDelete.resultType = .resultTypeObjectIDs

    let context = PersistenceController.shared.viewContext
    let result = try context.execute(batchDelete) as? NSBatchDeleteResult
    let objectIDs = result?.result as? [NSManagedObjectID] ?? []

    NSManagedObjectContext.mergeChanges(
        fromRemoteContextSave: [NSDeletedObjectsKey: objectIDs],
        into: [context]
    )
}

class MigrationPersistenceController {
    let container: NSPersistentContainer

    init() {
        container = NSPersistentContainer(name: "AppModel")

        let description = container.persistentStoreDescriptions.first!
        description.setOption(true as NSNumber, forKey: NSMigratePersistentStoresAutomaticallyOption)
        description.setOption(true as NSNumber, forKey: NSInferMappingModelAutomaticallyOption)

        container.loadPersistentStores { _, error in
            if let error {
                fatalError("迁移失败: \(error)")
            }
        }
    }
}
完整示例
import CoreData
import SwiftUI

struct CoreDataTaskApp: App {
    let persistence = PersistenceController.shared

    var body: some Scene {
        WindowGroup {
            TaskListView()
                .environment(\.managedObjectContext, persistence.viewContext)
        }
    }
}

struct TaskListView: View {
    @Environment(\.managedObjectContext) private var context

    @FetchRequest(
        sortDescriptors: [SortDescriptor(\TaskItem.createdAt, order: .reverse)],
        predicate: NSPredicate(format: "isCompleted == NO"),
        animation: .default
    )
    private var tasks: FetchedResults<TaskItem>

    @State private var newTitle = ""

    var body: some View {
        NavigationStack {
            List {
                HStack {
                    TextField("新任务", text: $newTitle)
                    Button("添加") {
                        addTask()
                    }
                    .disabled(newTitle.isEmpty)
                }

                ForEach(tasks) { task in
                    HStack {
                        Image(systemName: task.isCompleted ? "checkmark.circle.fill" : "circle")
                            .onTapGesture { toggleTask(task) }
                        VStack(alignment: .leading) {
                            Text(task.title)
                            if let content = task.content {
                                Text(content)
                                    .font(.caption)
                                    .foregroundStyle(.secondary)
                            }
                        }
                    }
                }
                .onDelete(perform: deleteTasks)
            }
            .navigationTitle("任务列表")
        }
    }

    private func addTask() {
        _ = TaskItem.create(in: context, title: newTitle)
        try? context.save()
        newTitle = ""
    }

    private func toggleTask(_ task: TaskItem) {
        task.isCompleted.toggle()
        try? context.save()
    }

    private func deleteTasks(at offsets: IndexSet) {
        offsets.map { tasks[$0] }.forEach(context.delete)
        try? context.save()
    }
}

extension TaskItem: Identifiable {}
最佳实践




做法
说明





✅ viewContext 仅用于读取和轻量写入
主线程上下文不应执行耗时操作




✅ 批量导入使用后台上下文
newBackgroundContext() + perform {}




✅ 启用 automaticallyMergesChangesFromParent
后台保存自动同步到 viewContext




✅ 使用 NSBatchDeleteRequest 删除大量数据
绕过对象图直接操作存储，性能远优于逐条删除




✅ 设置合理的 fetchLimit 和 fetchBatchSize
避免一次性加载全部数据到内存




✅ 使用轻量级迁移
添加属性、实体等简单变更自动推断映射




❌ 不要跨线程传递 NSManagedObject
使用 objectID 在另一个上下文中重新获取




❌ 不要忽略 mergePolicy
多上下文并发写入时可能产生冲突




❌ 不要在循环中逐条 save()
收集所有变更后统一保存一次





小结
    

  • NSPersistentContainer 是 Core Data 栈的一站式入口，封装了模型、协调器和存储的初始化。
    • 

  • NSManagedObjectContext 是所有 CRUD 操作的工作区；viewContext 在主线程，后台任务用 newBackgroundContext()。
    • 

  • NSFetchRequest + NSPredicate + NSSortDescriptor 构成查询三件套，支持复杂条件、分页和投影查询。
    • 

  • NSFetchedResultsController 为 UIKit 提供数据变更的自动通知；SwiftUI 使用 @FetchRequest 实现同等功能。
    • 

  • 关系通过 NSSet 表达，需转换为 Swift 数组使用；支持一对多和多对多。
    • 

  • NSBatchDeleteRequest 绕过对象图直接操作存储层，需手动合并变更到上下文。
    • 

  • 轻量级迁移（Lightweight Migration）可自动处理添加属性、实体等简单 schema 变更。
    • 

  • 始终在正确的线程上操作对应的 NSManagedObjectContext，跨线程传递使用 objectID。
    •
