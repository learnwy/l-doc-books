Codable 与 JSON 解析

概述

在 iOS 开发中，与后端 API 交互的核心环节就是 JSON 的序列化与反序列化。Swift 的 Codable 协议（Encodable & Decodable 的组合）提供了类型安全、编译期检查的编解码能力，让 JSON 解析从手动取值的繁琐操作变成了声明式的自动映射。本文将从最简单的自动合成开始，逐步深入到自定义 Key 映射、嵌套解析、日期策略和容错解析等高级用法。

基础用法
自动合成 Codable
当模型的所有属性都遵循 Codable 时，编译器会自动合成编解码逻辑：


struct User: Codable {
    let id: Int
    let name: String
    let email: String
    let isActive: Bool
}

let json = """
{
    "id": 1,
    "name": "Alice",
    "email": "alice@example.com",
    "isActive": true
}
""".data(using: .utf8)!

let user = try JSONDecoder().decode(User.self, from: json)
print(user.name)

let encoded = try JSONEncoder().encode(user)
print(String(data: encoded, encoding: .utf8)!)
CodingKeys 自定义映射
当 JSON 的 key 与 Swift 属性名不一致时，使用 CodingKeys 枚举进行映射：


struct Article: Codable {
    let articleId: Int
    let title: String
    let authorName: String
    let createdAt: String

    enum CodingKeys: String, CodingKey {
        case articleId = "article_id"
        case title
        case authorName = "author_name"
        case createdAt = "created_at"
    }
}
keyDecodingStrategy 全局策略
如果后端统一使用 snake_case，可以用全局策略代替逐个映射：


struct Product: Codable {
    let productName: String
    let unitPrice: Double
    let stockCount: Int
}

let decoder = JSONDecoder()
decoder.keyDecodingStrategy = .convertFromSnakeCase

let json = """
{
    "product_name": "iPhone",
    "unit_price": 999.0,
    "stock_count": 100
}
""".data(using: .utf8)!

let product = try decoder.decode(Product.self, from: json)
print(product.productName)

日期解码策略
struct Event: Codable {
    let name: String
    let date: Date
}

let decoder = JSONDecoder()

decoder.dateDecodingStrategy = .iso8601

let isoJSON = """
{"name": "WWDC", "date": "2025-06-09T10:00:00Z"}
""".data(using: .utf8)!
let event1 = try decoder.decode(Event.self, from: isoJSON)

let formatter = DateFormatter()
formatter.dateFormat = "yyyy-MM-dd"
formatter.locale = Locale(identifier: "en_US_POSIX")
decoder.dateDecodingStrategy = .formatted(formatter)

let customJSON = """
{"name": "Launch", "date": "2025-09-15"}
""".data(using: .utf8)!
let event2 = try decoder.decode(Event.self, from: customJSON)

decoder.dateDecodingStrategy = .secondsSince1970
let timestampJSON = """
{"name": "Epoch", "date": 1700000000}
""".data(using: .utf8)!
let event3 = try decoder.decode(Event.self, from: timestampJSON)
自定义 Decoder — 容错解析
后端返回的数据可能不规范，需要容错处理：


struct FlexibleValue: Codable {
    let id: Int
    let score: Double

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)

        if let intId = try? container.decode(Int.self, forKey: .id) {
            id = intId
        } else {
            let stringId = try container.decode(String.self, forKey: .id)
            guard let parsed = Int(stringId) else {
                throw DecodingError.dataCorruptedError(
                    forKey: .id, in: container,
                    debugDescription: "无法将 '\(stringId)' 转换为 Int"
                )
            }
            id = parsed
        }

        if let doubleScore = try? container.decode(Double.self, forKey: .score) {
            score = doubleScore
        } else {
            let stringScore = try container.decode(String.self, forKey: .score)
            score = Double(stringScore) ?? 0
        }
    }
}

let messyJSON = """
{"id": "42", "score": "98.5"}
""".data(using: .utf8)!

let value = try JSONDecoder().decode(FlexibleValue.self, from: messyJSON)
print(value.id, value.score)
自定义 Encoder
struct Color: Codable {
    let red: UInt8
    let green: UInt8
    let blue: UInt8

    func encode(to encoder: Encoder) throws {
        var container = encoder.singleValueContainer()
        let hex = String(format: "#%02X%02X%02X", red, green, blue)
        try container.encode(hex)
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.singleValueContainer()
        let hex = try container.decode(String.self)
        let stripped = hex.trimmingCharacters(in: CharacterSet(charactersIn: "#"))
        guard stripped.count == 6, let value = UInt(stripped, radix: 16) else {
            throw DecodingError.dataCorruptedError(
                in: container, debugDescription: "无效的颜色格式"
            )
        }
        red = UInt8((value >> 16) & 0xFF)
        green = UInt8((value >> 8) & 0xFF)
        blue = UInt8(value & 0xFF)
    }
}

let color = Color(red: 255, green: 128, blue: 0)
let data = try JSONEncoder().encode(color)
print(String(data: data, encoding: .utf8)!)
安全解码数组（跳过损坏元素）
struct SafeArray<Element: Decodable>: Decodable {
    let elements: [Element]

    init(from decoder: Decoder) throws {
        var container = try decoder.unkeyedContainer()
        var result: [Element] = []

        while !container.isAtEnd {
            if let element = try? container.decode(Element.self) {
                result.append(element)
            } else {
                _ = try? container.decode(AnyCodable.self)
            }
        }
        elements = result
    }
}

private struct AnyCodable: Decodable {}

struct Item: Decodable {
    let id: Int
    let name: String
}

let mixedJSON = """
[
    {"id": 1, "name": "A"},
    {"id": "broken", "name": "B"},
    {"id": 3, "name": "C"}
]
""".data(using: .utf8)!

let safe = try JSONDecoder().decode(SafeArray<Item>.self, from: mixedJSON)
print(safe.elements.count)
完整示例
import Foundation

struct APIResponse<T: Codable>: Codable {
    let code: Int
    let message: String
    let data: T?
    let timestamp: Date
}

struct Post: Codable {
    let id: Int
    let title: String
    let content: String
    let authorName: String
    let createdAt: Date
    let tags: [String]
    let metadata: [String: String]?

    enum CodingKeys: String, CodingKey {
        case id, title, content, tags, metadata
        case authorName = "author_name"
        case createdAt = "created_at"
    }
}

enum JSONHelper {
    static func makeDecoder() -> JSONDecoder {
        let decoder = JSONDecoder()
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd'T'HH:mm:ss"
        formatter.locale = Locale(identifier: "en_US_POSIX")
        formatter.timeZone = TimeZone(secondsFromGMT: 0)
        decoder.dateDecodingStrategy = .formatted(formatter)
        return decoder
    }

    static func makeEncoder() -> JSONEncoder {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd'T'HH:mm:ss"
        formatter.locale = Locale(identifier: "en_US_POSIX")
        formatter.timeZone = TimeZone(secondsFromGMT: 0)
        encoder.dateEncodingStrategy = .formatted(formatter)
        return encoder
    }

    static func decode<T: Codable>(_ type: APIResponse<T>.Type, from data: Data) throws -> APIResponse<T> {
        try makeDecoder().decode(type, from: data)
    }
}

let json = """
{
    "code": 200,
    "message": "success",
    "timestamp": "2025-06-01T12:00:00",
    "data": {
        "id": 1,
        "title": "Swift Codable 入门",
        "content": "本文介绍 Codable 的基本用法...",
        "author_name": "Alice",
        "created_at": "2025-05-20T08:30:00",
        "tags": ["swift", "ios", "codable"],
        "metadata": {"source": "web", "version": "2"}
    }
}
""".data(using: .utf8)!

do {
    let response = try JSONHelper.decode(APIResponse<Post>.self, from: json)
    guard response.code == 200, let post = response.data else {
        print("请求失败：\(response.message)")
        return
    }
    print("标题：\(post.title)")
    print("作者：\(post.authorName)")
    print("标签：\(post.tags.joined(separator: ", "))")

    let reEncoded = try JSONHelper.makeEncoder().encode(response)
    print(String(data: reEncoded, encoding: .utf8)!)
} catch {
    print("解析失败：\(error)")
}
最佳实践




做法
说明





✅ 优先使用 struct 作为模型
值语义更安全、Codable 合成更简单




✅ 使用 keyDecodingStrategy
全局 snake_case → camelCase 转换，减少 CodingKeys 样板代码




✅ 用 Date 替代 String 存日期
配合 dateDecodingStrategy 自动解析




✅ 后端字段可能缺失时声明为 Optional
避免因缺失字段导致整个解析失败




✅ 在容错场景下自定义 init(from:)
处理类型不一致（如 id 可能是 Int 或 String）




❌ 避免 Any 和 [String: Any]
失去类型安全，回到手动解析的老路




❌ 不要在模型中放业务逻辑
模型只做数据映射，业务逻辑放在 ViewModel 或 Service 层




❌ 避免 try! 或 try? 吞掉错误
try? 会丢失错误信息，调试困难





小结
    

  • Codable 是 Encodable & Decodable 的类型别名，编译器可为所有属性均为 Codable 的类型自动合成编解码逻辑。
    • 

  • CodingKeys 枚举用于 JSON key 到 Swift 属性的逐字段映射；keyDecodingStrategy = .convertFromSnakeCase 实现全局策略。
    • 

  • 嵌套 JSON 通过嵌套 struct 自然映射；泛型 APIResponse<T> 可复用统一的响应结构。
    • 

  • 日期解码支持 .iso8601、.formatted()、.secondsSince1970 等多种策略。
    • 

  • 自定义 init(from decoder:) 可实现容错解析，处理后端返回的不规范数据。
    • 

  • 安全数组解码（跳过损坏元素）避免单个元素异常导致整个列表解析失败。
    • 

  • JSONEncoder 的 outputFormatting 可控制输出格式（美化、排序 key）。
