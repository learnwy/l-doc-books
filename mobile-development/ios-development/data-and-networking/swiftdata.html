SwiftData 现代数据层

概述

SwiftData 是 Apple 在 WWDC23 推出的全新持久化框架，旨在用纯 Swift 语法取代 Core Data 的复杂配置。它通过 @Model 宏自动生成模型元数据，与 SwiftUI 深度集成，让数据持久化变得声明式且类型安全。本文将从基础建模出发，逐步讲解查询、关系、SwiftUI 集成以及从 Core Data 迁移的实践。

基础用法
@Model 定义模型
import SwiftData

@Model
class Task {
    var title: String
    var content: String?
    var isCompleted: Bool
    var createdAt: Date
    var priority: Int

    init(title: String, content: String? = nil, priority: Int = 0) {
        self.title = title
        self.content = content
        self.isCompleted = false
        self.createdAt = Date()
        self.priority = priority
    }
}
@Model 宏自动为类添加 PersistentModel 和 Observable 协议的实现，无需手动编写 .xcdatamodeld 文件。

基本 CRUD
import SwiftData

func basicCRUD(context: ModelContext) throws {
    let task = Task(title: "学习 SwiftData", priority: 1)
    context.insert(task)
    try context.save()

    let descriptor = FetchDescriptor<Task>(
        predicate: #Predicate { $0.isCompleted == false },
        sortBy: [SortDescriptor(\.createdAt, order: .reverse)]
    )
    let tasks = try context.fetch(descriptor)

    tasks.first?.isCompleted = true
    try context.save()

    if let toDelete = tasks.last {
        context.delete(toDelete)
        try context.save()
    }
}

@Query 与 SwiftUI 集成
import SwiftUI
import SwiftData

struct TaskListView: View {
    @Environment(\.modelContext) private var context

    @Query(
        filter: #Predicate<Task> { !$0.isCompleted },
        sort: [SortDescriptor(\Task.priority, order: .reverse)],
        animation: .default
    )
    private var pendingTasks: [Task]

    @Query(
        filter: #Predicate<Task> { $0.isCompleted },
        sort: [SortDescriptor(\Task.createdAt, order: .reverse)]
    )
    private var completedTasks: [Task]

    @State private var newTitle = ""

    var body: some View {
        NavigationStack {
            List {
                Section("待完成 (\(pendingTasks.count))") {
                    ForEach(pendingTasks) { task in
                        TaskRow(task: task)
                    }
                    .onDelete { offsets in
                        offsets.map { pendingTasks[$0] }.forEach(context.delete)
                    }
                }

                Section("已完成 (\(completedTasks.count))") {
                    ForEach(completedTasks) { task in
                        TaskRow(task: task)
                    }
                }
            }
            .navigationTitle("任务管理")
            .toolbar {
                ToolbarItem(placement: .bottomBar) {
                    HStack {
                        TextField("新任务", text: $newTitle)
                        Button("添加") {
                            context.insert(Task(title: newTitle))
                            newTitle = ""
                        }
                        .disabled(newTitle.isEmpty)
                    }
                }
            }
        }
    }
}

struct TaskRow: View {
    let task: Task

    var body: some View {
        HStack {
            Image(systemName: task.isCompleted ? "checkmark.circle.fill" : "circle")
                .foregroundStyle(task.isCompleted ? .green : .gray)
                .onTapGesture { task.isCompleted.toggle() }
            VStack(alignment: .leading) {
                Text(task.title)
                    .strikethrough(task.isCompleted)
                if let content = task.content {
                    Text(content)
                        .font(.caption)
                        .foregroundStyle(.secondary)
                }
            }
            Spacer()
            Text("P\(task.priority)")
                .font(.caption2)
                .padding(.horizontal, 6)
                .padding(.vertical, 2)
                .background(priorityColor(task.priority).opacity(0.2))
                .clipShape(Capsule())
        }
    }

    private func priorityColor(_ priority: Int) -> Color {
        switch priority {
        case 3...: return .red
        case 2: return .orange
        case 1: return .blue
        default: return .gray
        }
    }
}
关系建模
@Model
class Category {
    var name: String
    var color: String

    @Relationship(deleteRule: .cascade, inverse: \Task.category)
    var tasks: [Task]

    init(name: String, color: String = "blue") {
        self.name = name
        self.color = color
        self.tasks = []
    }
}

@Model
class Task {
    var title: String
    var content: String?
    var isCompleted: Bool
    var createdAt: Date
    var priority: Int
    var category: Category?

    @Relationship(deleteRule: .nullify, inverse: \Tag.tasks)
    var tags: [Tag]

    init(title: String, content: String? = nil, priority: Int = 0) {
        self.title = title
        self.content = content
        self.isCompleted = false
        self.createdAt = Date()
        self.priority = priority
        self.tags = []
    }
}

@Model
class Tag {
    @Attribute(.unique)
    var name: String
    var tasks: [Task]

    init(name: String) {
        self.name = name
        self.tasks = []
    }
}

动态查询（运行时切换过滤条件）
struct DynamicTaskList: View {
    @State private var searchText = ""
    @State private var showCompleted = false

    var body: some View {
        FilteredTaskList(searchText: searchText, showCompleted: showCompleted)
            .searchable(text: $searchText)
            .toolbar {
                Toggle("已完成", isOn: $showCompleted)
            }
    }
}

struct FilteredTaskList: View {
    @Query private var tasks: [Task]

    init(searchText: String, showCompleted: Bool) {
        let predicate: Predicate<Task>

        if searchText.isEmpty {
            predicate = #Predicate<Task> { task in
                showCompleted || !task.isCompleted
            }
        } else {
            predicate = #Predicate<Task> { task in
                task.title.localizedStandardContains(searchText) &&
                (showCompleted || !task.isCompleted)
            }
        }

        _tasks = Query(
            filter: predicate,
            sort: [SortDescriptor(\Task.createdAt, order: .reverse)],
            animation: .default
        )
    }

    var body: some View {
        List(tasks) { task in
            TaskRow(task: task)
        }
    }
}

import SwiftUI
import SwiftData

@Model
class Note {
    var title: String
    var body: String
    var isPinned: Bool
    var createdAt: Date
    var updatedAt: Date

    @Relationship(deleteRule: .cascade, inverse: \NoteTag.note)
    var tags: [NoteTag]

    init(title: String, body: String = "") {
        self.title = title
        self.body = body
        self.isPinned = false
        self.createdAt = Date()
        self.updatedAt = Date()
        self.tags = []
    }
}

@Model
class NoteTag {
    var name: String
    var note: Note?

    init(name: String) {
        self.name = name
    }
}

@main
struct NoteApp: App {
    var body: some Scene {
        WindowGroup {
            NoteListScreen()
        }
        .modelContainer(for: [Note.self, NoteTag.self])
    }
}

struct NoteListScreen: View {
    @Environment(\.modelContext) private var context
    @Query(sort: [
        SortDescriptor(\Note.isPinned, order: .reverse),
        SortDescriptor(\Note.updatedAt, order: .reverse)
    ])
    private var notes: [Note]

    @State private var showEditor = false

    var body: some View {
        NavigationStack {
            List {
                ForEach(notes) { note in
                    NavigationLink(destination: NoteEditorView(note: note)) {
                        VStack(alignment: .leading) {
                            HStack {
                                if note.isPinned {
                                    Image(systemName: "pin.fill")
                                        .font(.caption)
                                        .foregroundStyle(.orange)
                                }
                                Text(note.title).font(.headline)
                            }
                            Text(note.body.prefix(50))
                                .font(.caption)
                                .foregroundStyle(.secondary)
                                .lineLimit(1)
                        }
                    }
                    .swipeActions(edge: .leading) {
                        Button {
                            note.isPinned.toggle()
                        } label: {
                            Label("置顶", systemImage: note.isPinned ? "pin.slash" : "pin")
                        }
                        .tint(.orange)
                    }
                }
                .onDelete { offsets in
                    offsets.map { notes[$0] }.forEach(context.delete)
                }
            }
            .navigationTitle("笔记")
            .toolbar {
                Button {
                    context.insert(Note(title: "新笔记"))
                } label: {
                    Image(systemName: "plus")
                }
            }
        }
    }
}

struct NoteEditorView: View {
    @Bindable var note: Note
    @Environment(\.modelContext) private var context

    var body: some View {
        Form {
            TextField("标题", text: $note.title)
            TextEditor(text: $note.body)
                .frame(minHeight: 200)
            Section("信息") {
                LabeledContent("创建时间", value: note.createdAt.formatted())
                LabeledContent("更新时间", value: note.updatedAt.formatted())
            }
        }
        .navigationTitle("编辑笔记")
        .onChange(of: note.body) {
            note.updatedAt = Date()
        }
    }
}
从 Core Data 迁移
迁移策略：


import CoreData
import SwiftData

enum MigrationHelper {
    static func migrateFromCoreData(
        coreDataContainer: NSPersistentContainer,
        swiftDataContext: ModelContext
    ) throws {
        let coreDataContext = coreDataContainer.viewContext
        let request = NSFetchRequest<NSManagedObject>(entityName: "CDTask")
        request.fetchBatchSize = 50

        let coreDataTasks = try coreDataContext.fetch(request)

        for cdTask in coreDataTasks {
            let task = Task(
                title: cdTask.value(forKey: "title") as? String ?? "",
                content: cdTask.value(forKey: "content") as? String,
                priority: cdTask.value(forKey: "priority") as? Int ?? 0
            )
            task.isCompleted = cdTask.value(forKey: "isCompleted") as? Bool ?? false
            task.createdAt = cdTask.value(forKey: "createdAt") as? Date ?? Date()

            swiftDataContext.insert(task)
        }

        try swiftDataContext.save()
    }
}
最佳实践




做法
说明





✅ 使用 @Model 类而非结构体
SwiftData 要求模型为引用类型（class）




✅ @Attribute(.unique) 设置唯一约束
防止重复数据，支持 upsert 语义




✅ @Attribute(.externalStorage) 存大文件
图片、文件等二进制数据自动存储到外部




✅ @Transient 标记非持久化属性
运行时临时状态不写入存储




✅ 动态 @Query 通过 init 注入条件
实现搜索、筛选等运行时变化的查询




✅ 使用 deleteRule 管理级联行为
.cascade 删除父对象时自动清理子对象




❌ 不要在 @Model 类中使用枚举关联值
当前 SwiftData 对复杂枚举支持有限




❌ 不要忘记注册所有模型到 ModelContainer
遗漏模型会导致运行时崩溃




❌ 不要在后台线程直接操作 @MainActor 上下文
使用 ModelActor 处理后台任务





小结
    

  • @Model 宏将普通 Swift 类转换为持久化模型，自动合成 PersistentModel 和 Observable。
    • 

  • ModelContainer 管理存储配置和 schema；ModelContext 是执行 CRUD 操作的核心工作区。
    • 

  • #Predicate 提供编译期类型检查的查询语法，彻底告别字符串格式的 NSPredicate。
    • 

  • @Query 属性包装器将 SwiftData 查询与 SwiftUI 视图绑定，数据变更自动驱动 UI 刷新。
    • 

  • 关系通过 @Relationship 声明，支持 .cascade、.nullify、.deny 等删除规则。
    • 

  • @Attribute 修饰符控制唯一约束（.unique）、外部存储（.externalStorage）等存储行为。
    • 

  • 动态查询通过在子视图 init 中构造 Query 实现，适用于搜索和筛选场景。
    • 

  • 从 Core Data 迁移时，先建立等价的 @Model 类，通过一次性脚本导入数据，再逐步移除旧代码。
