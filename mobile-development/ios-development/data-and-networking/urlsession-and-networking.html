URLSession 网络请求

概述

URLSession 是 iOS 网络请求的基石——从简单的 GET 请求到复杂的上传下载，所有 HTTP 通信都建立在它之上。本文将从最简单的 URLSession.shared 开始，逐步构建一个支持 async/await、泛型解码、错误处理和拦截器的完整网络层。

基础用法
最简 GET 请求（async/await）
import Foundation

func fetchUser(id: Int) async throws -> Data {
    let url = URL(string: "https://api.example.com/users/\(id)")!
    let (data, response) = try await URLSession.shared.data(from: url)

    guard let httpResponse = response as? HTTPURLResponse,
          (200...299).contains(httpResponse.statusCode) else {
        throw URLError(.badServerResponse)
    }

    return data
}
callback 版本（兼容旧代码）
func fetchUser(id: Int, completion: @escaping (Result<Data, Error>) -> Void) {
    let url = URL(string: "https://api.example.com/users/\(id)")!

    URLSession.shared.dataTask(with: url) { data, response, error in
        if let error {
            completion(.failure(error))
            return
        }
        guard let data, let httpResponse = response as? HTTPURLResponse,
              (200...299).contains(httpResponse.statusCode) else {
            completion(.failure(URLError(.badServerResponse)))
            return
        }
        completion(.success(data))
    }.resume()
}

进阶用法
自定义错误体系
enum NetworkError: Error, LocalizedError {
    case invalidURL
    case noData
    case decodingFailed(Error)
    case httpError(statusCode: Int, data: Data)
    case unauthorized
    case serverError(String)

    var errorDescription: String? {
        switch self {
        case .invalidURL: return "无效的 URL"
        case .noData: return "响应数据为空"
        case .decodingFailed(let error): return "解码失败: \(error.localizedDescription)"
        case .httpError(let code, _): return "HTTP 错误: \(code)"
        case .unauthorized: return "未授权，请重新登录"
        case .serverError(let msg): return "服务器错误: \(msg)"
        }
    }
}
泛型网络层封装
final class NetworkClient {
    static let shared = NetworkClient()
    private let session: URLSession
    private let decoder: JSONDecoder

    private init() {
        let config = URLSessionConfiguration.default
        config.timeoutIntervalForRequest = 30
        config.waitsForConnectivity = true
        session = URLSession(configuration: config)

        decoder = JSONDecoder()
        decoder.keyDecodingStrategy = .convertFromSnakeCase
        decoder.dateDecodingStrategy = .iso8601
    }

    func request<T: Decodable>(_ endpoint: Endpoint) async throws -> T {
        let request = try endpoint.makeURLRequest()

        let (data, response) = try await session.data(for: request)

        guard let httpResponse = response as? HTTPURLResponse else {
            throw NetworkError.noData
        }

        switch httpResponse.statusCode {
        case 200...299:
            do {
                return try decoder.decode(T.self, from: data)
            } catch {
                throw NetworkError.decodingFailed(error)
            }
        case 401:
            throw NetworkError.unauthorized
        default:
            throw NetworkError.httpError(statusCode: httpResponse.statusCode, data: data)
        }
    }
}
Endpoint 抽象
struct Endpoint {
    let path: String
    let method: HTTPMethod
    let headers: [String: String]
    let queryItems: [URLQueryItem]?
    let body: Encodable?

    enum HTTPMethod: String {
        case get = "GET"
        case post = "POST"
        case put = "PUT"
        case delete = "DELETE"
    }

    private static let baseURL = "https://api.example.com"

    func makeURLRequest() throws -> URLRequest {
        var components = URLComponents(string: Self.baseURL + path)!
        components.queryItems = queryItems

        guard let url = components.url else {
            throw NetworkError.invalidURL
        }

        var request = URLRequest(url: url)
        request.httpMethod = method.rawValue

        var allHeaders = ["Content-Type": "application/json"]
        allHeaders.merge(headers) { _, new in new }
        allHeaders.forEach { request.setValue($1, forHTTPHeaderField: $0) }

        if let body {
            request.httpBody = try JSONEncoder().encode(AnyEncodable(body))
        }
        return request
    }
}

struct AnyEncodable: Encodable {
    private let encodeFunc: (Encoder) throws -> Void

    init(_ value: Encodable) {
        encodeFunc = { encoder in
            try value.encode(to: encoder)
        }
    }

    func encode(to encoder: Encoder) throws {
        try encodeFunc(encoder)
    }
}
拦截器模式
protocol RequestInterceptor {
    func adapt(_ request: URLRequest) async throws -> URLRequest
    func retry(
        _ request: URLRequest,
        dueTo error: Error,
        attemptCount: Int
    ) async -> Bool
}

extension RequestInterceptor {
    func retry(_ request: URLRequest, dueTo error: Error, attemptCount: Int) async -> Bool {
        false
    }
}

struct AuthInterceptor: RequestInterceptor {
    let tokenProvider: () async -> String?

    func adapt(_ request: URLRequest) async throws -> URLRequest {
        var request = request
        if let token = await tokenProvider() {
            request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
        }
        return request
    }
}

struct LoggingInterceptor: RequestInterceptor {
    func adapt(_ request: URLRequest) async throws -> URLRequest {
        print("➡️ \(request.httpMethod ?? "GET") \(request.url?.absoluteString ?? "")")
        return request
    }
}

struct RetryInterceptor: RequestInterceptor {
    let maxRetries: Int

    func retry(_ request: URLRequest, dueTo error: Error, attemptCount: Int) async -> Bool {
        guard attemptCount < maxRetries else { return false }
        try? await Task.sleep(nanoseconds: UInt64(pow(2.0, Double(attemptCount))) * 1_000_000_000)
        return true
    }
}
集成拦截器的网络客户端
final class InterceptableClient {
    private let session: URLSession
    private let interceptors: [RequestInterceptor]
    private let decoder: JSONDecoder

    init(interceptors: [RequestInterceptor] = []) {
        self.session = URLSession(configuration: .default)
        self.interceptors = interceptors
        self.decoder = JSONDecoder()
        decoder.keyDecodingStrategy = .convertFromSnakeCase
        decoder.dateDecodingStrategy = .iso8601
    }

    func request<T: Decodable>(_ endpoint: Endpoint) async throws -> T {
        var urlRequest = try endpoint.makeURLRequest()

        for interceptor in interceptors {
            urlRequest = try await interceptor.adapt(urlRequest)
        }

        var attemptCount = 0
        while true {
            do {
                let (data, response) = try await session.data(for: urlRequest)
                guard let http = response as? HTTPURLResponse,
                      (200...299).contains(http.statusCode) else {
                    throw NetworkError.httpError(
                        statusCode: (response as? HTTPURLResponse)?.statusCode ?? -1,
                        data: Data()
                    )
                }
                return try decoder.decode(T.self, from: data)
            } catch {
                var shouldRetry = false
                for interceptor in interceptors {
                    if await interceptor.retry(urlRequest, dueTo: error, attemptCount: attemptCount) {
                        shouldRetry = true
                        break
                    }
                }
                if !shouldRetry { throw error }
                attemptCount += 1
            }
        }
    }
}
完整示例
import Foundation

struct User: Codable {
    let id: Int
    let name: String
    let email: String
}

struct CreateUserRequest: Encodable {
    let name: String
    let email: String
}

struct APIResponse<T: Codable>: Codable {
    let code: Int
    let message: String
    let data: T?
}

final class UserService {
    private let client: InterceptableClient

    init() {
        client = InterceptableClient(interceptors: [
            LoggingInterceptor(),
            AuthInterceptor { "my-auth-token" },
            RetryInterceptor(maxRetries: 3)
        ])
    }

    func fetchUser(id: Int) async throws -> User {
        let endpoint = Endpoint(
            path: "/users/\(id)",
            method: .get,
            headers: [:],
            queryItems: nil,
            body: nil
        )
        let response: APIResponse<User> = try await client.request(endpoint)
        guard let user = response.data else {
            throw NetworkError.noData
        }
        return user
    }

    func createUser(name: String, email: String) async throws -> User {
        let endpoint = Endpoint(
            path: "/users",
            method: .post,
            headers: [:],
            queryItems: nil,
            body: CreateUserRequest(name: name, email: email)
        )
        let response: APIResponse<User> = try await client.request(endpoint)
        guard let user = response.data else {
            throw NetworkError.noData
        }
        return user
    }

    func listUsers(page: Int, pageSize: Int) async throws -> [User] {
        let endpoint = Endpoint(
            path: "/users",
            method: .get,
            headers: [:],
            queryItems: [
                URLQueryItem(name: "page", value: "\(page)"),
                URLQueryItem(name: "page_size", value: "\(pageSize)")
            ],
            body: nil
        )
        let response: APIResponse<[User]> = try await client.request(endpoint)
        return response.data ?? []
    }
}

Task {
    let service = UserService()
    do {
        let user = try await service.fetchUser(id: 1)
        print("获取用户：\(user.name)")

        let newUser = try await service.createUser(name: "Bob", email: "bob@test.com")
        print("创建用户：\(newUser.name)")

        let users = try await service.listUsers(page: 1, pageSize: 20)
        print("用户列表：\(users.count) 人")
    } catch let error as NetworkError {
        print("网络错误：\(error.errorDescription ?? "")")
    } catch {
        print("未知错误：\(error)")
    }
}
最佳实践




做法
说明





✅ 使用 async/await 而非 callback
代码线性可读，错误传播自然




✅ 抽象 Endpoint 层
将 URL 拼接、Header 设置集中管理




✅ 用枚举定义错误类型
区分网络错误、HTTP 状态码错误、解码错误




✅ 设置 timeoutInterval
避免用户长时间等待无响应




✅ 使用 waitsForConnectivity
等待网络恢复后自动发送请求




✅ 拦截器模式处理横切关注点
认证、日志、重试逻辑与业务代码解耦




❌ 不要在主线程做同步网络请求
会阻塞 UI，导致界面卡死




❌ 避免硬编码 URL 和 Header
使用配置文件或枚举统一管理




❌ 不要忽略 HTTP 状态码
URLSession 不会对 4xx/5xx 抛出错误，需手动检查





小结
    

  • URLSession.shared 是最简单的入口，适合快速原型；生产环境应使用自定义 URLSessionConfiguration。
    • 

  • async/await 版本的 data(from:) 和 data(for:) 让网络请求代码线性化、错误传播自然。
    • 

  • URLRequest 可配置 HTTP 方法、Header、Body 和超时时间。
    • 

  • 自定义 NetworkError 枚举让错误处理精确且可预测。
    • 

  • 泛型网络层（NetworkClient + Endpoint）将 URL 构建、请求配置、响应解码封装为统一接口。
    • 

  • 拦截器模式通过 RequestInterceptor 协议实现认证注入、日志记录和自动重试，与业务逻辑完全解耦。
    • 

  • 指数退避重试策略（exponential backoff）能有效应对暂时性网络故障。
