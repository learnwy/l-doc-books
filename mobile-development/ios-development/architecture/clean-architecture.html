Part I · iOSiOS 开发 简介架构 简介Clean Architecture 整洁架构

Clean Architecture 整洁架构

概述

Clean Architecture（整洁架构）是 Robert C. Martin（Uncle Bob）提出的分层架构思想，其核心原则是依赖规则（Dependency Rule）：源代码的依赖关系只能指向内层，外层的变更不应影响内层。这意味着业务逻辑（Entity、UseCase）完全独立于框架、UI、数据库等基础设施细节。

在 iOS 开发中，Clean Architecture 解决了随着项目规模增长，各层职责模糊、代码难以测试和复用的问题。它将系统划分为多个同心圆层，从内到外依次是 Entity → Use Case → Interface Adapter → Framework & Driver。每一层只知道自己内层的存在。

flowchart TB
    subgraph Outer Layer ["Framework & Driver"]
        UI["UIKit / SwiftUI"]
        DB["Core Data / Network"]
    end
    subgraph Interface Adapter Layer ["Interface Adapter"]
        P["Presenter / ViewModel"]
        R["Repository Impl"]
        GW["Gateway / API Client"]
    end
    subgraph Use Case Layer ["Use Case"]
        UC["Business Use Case<br/>Interactor"]
    end
    subgraph Core Layer ["Entity"]
        E["Domain Entity<br/>Business Rules"]
    end
    UI --> P --> UC --> E
    DB --> R --> UC
    GW --> R

核心概念

依赖规则

依赖规则是 Clean Architecture 最重要的原则：内层不知道外层的存在。内层通过 Protocol 定义接口，外层提供具体实现：

flowchart LR
    subgraph Dependency Direction
        direction LR
        FW["Framework"] -->|Dependencies| IA["Interface Adapter"]
        IA -->|Dependencies| UC["Use Case"]
        UC -->|Dependencies| EN["Entity"]
    end
    EN -.- N1["❌ No dependency on outer layers"]

Entity（领域实体）

Entity 是最内层，定义核心业务对象和业务规则。它不依赖任何框架：

struct Product {
    let id: String
    var name: String
    var price: Decimal
    var stockCount: Int

    var isAvailable: Bool {
        stockCount > 0
    }

    func discountedPrice(rate: Decimal) -> Decimal {
        price * (1 - rate)
    }
}

struct Order {
    let id: String
    var items: [OrderItem]
    var status: OrderStatus
    let createdAt: Date

    var totalAmount: Decimal {
        items.reduce(0) { $0 + $1.subtotal }
    }

    var isEditable: Bool {
        status == .draft || status == .pending
    }
}

struct OrderItem {
    let product: Product
    var quantity: Int

    var subtotal: Decimal {
        product.price * Decimal(quantity)
    }
}

enum OrderStatus: String {
    case draft, pending, confirmed, shipped, delivered, cancelled
}

Use Case（业务用例）

Use Case 层封装应用的业务场景。每个 Use Case 代表一个用户可以执行的操作。它通过 Protocol 依赖 Repository，而不是具体实现：

protocol OrderRepository {
    func save(_ order: Order) async throws
    func fetchOrder(by id: String) async throws -> Order
    func fetchOrders(for userId: String) async throws -> [Order]
}

protocol ProductRepository {
    func fetchProduct(by id: String) async throws -> Product
    func updateStock(productId: String, count: Int) async throws
}

class PlaceOrderUseCase {
    private let orderRepository: OrderRepository
    private let productRepository: ProductRepository

    init(orderRepository: OrderRepository, productRepository: ProductRepository) {
        self.orderRepository = orderRepository
        self.productRepository = productRepository
    }

    func execute(items: [(productId: String, quantity: Int)]) async throws -> Order {
        var orderItems: [OrderItem] = []

        for item in items {
            let product = try await productRepository.fetchProduct(by: item.productId)
            guard product.stockCount >= item.quantity else {
                throw OrderError.insufficientStock(product.name)
            }
            orderItems.append(OrderItem(product: product, quantity: item.quantity))
        }

        var order = Order(
            id: UUID().uuidString,
            items: orderItems,
            status: .pending,
            createdAt: Date()
        )

        try await orderRepository.save(order)

        for item in items {
            let product = try await productRepository.fetchProduct(by: item.productId)
            try await productRepository.updateStock(
                productId: item.productId,
                count: product.stockCount - item.quantity
            )
        }

        return order
    }
}

enum OrderError: LocalizedError {
    case insufficientStock(String)
    case orderNotEditable

    var errorDescription: String? {
        switch self {
        case .insufficientStock(let name): "「\(name)」库存不足"
        case .orderNotEditable: "订单已确认，无法编辑"
        }
    }
}

Repository 实现（Interface Adapter）

Repository 的具体实现位于接口适配层，负责将外部数据源（网络、数据库）适配为 Use Case 所需的接口：

class RemoteOrderRepository: OrderRepository {
    private let apiClient: APIClient

    init(apiClient: APIClient) {
        self.apiClient = apiClient
    }

    func save(_ order: Order) async throws {
        let dto = OrderDTO(from: order)
        try await apiClient.post("/orders", body: dto)
    }

    func fetchOrder(by id: String) async throws -> Order {
        let dto: OrderDTO = try await apiClient.get("/orders/\(id)")
        return dto.toDomain()
    }

    func fetchOrders(for userId: String) async throws -> [Order] {
        let dtos: [OrderDTO] = try await apiClient.get("/users/\(userId)/orders")
        return dtos.map { $0.toDomain() }
    }
}

struct OrderDTO: Codable {
    let id: String
    let items: [OrderItemDTO]
    let status: String
    let createdAt: String

    init(from domain: Order) {
        id = domain.id
        items = domain.items.map { OrderItemDTO(from: $0) }
        status = domain.status.rawValue
        createdAt = ISO8601DateFormatter().string(from: domain.createdAt)
    }

    func toDomain() -> Order {
        Order(
            id: id,
            items: items.map { $0.toDomain() },
            status: OrderStatus(rawValue: status) ?? .draft,
            createdAt: ISO8601DateFormatter().date(from: createdAt) ?? Date()
        )
    }
}

struct OrderItemDTO: Codable {
    let productId: String
    let productName: String
    let price: Double
    let quantity: Int

    init(from domain: OrderItem) {
        productId = domain.product.id
        productName = domain.product.name
        price = NSDecimalNumber(decimal: domain.product.price).doubleValue
        quantity = domain.quantity
    }

    func toDomain() -> OrderItem {
        OrderItem(
            product: Product(
                id: productId,
                name: productName,
                price: Decimal(price),
                stockCount: 0
            ),
            quantity: quantity
        )
    }
}

Presenter / ViewModel（Interface Adapter）

Presenter 或 ViewModel 位于接口适配层，负责调用 Use Case 并将结果转换为 View 可消费的状态：

@Observable
class OrderViewModel {
    private let placeOrderUseCase: PlaceOrderUseCase

    var orderItems: [(productId: String, quantity: Int)] = []
    var placedOrder: Order?
    var isProcessing = false
    var errorMessage: String?

    var formattedTotal: String {
        guard let order = placedOrder else { return "¥0.00" }
        return "¥\(order.totalAmount)"
    }

    init(placeOrderUseCase: PlaceOrderUseCase) {
        self.placeOrderUseCase = placeOrderUseCase
    }

    func placeOrder() async {
        isProcessing = true
        errorMessage = nil
        do {
            placedOrder = try await placeOrderUseCase.execute(items: orderItems)
        } catch {
            errorMessage = error.localizedDescription
        }
        isProcessing = false
    }
}

模式一：模块化项目结构

模式二：Clean Architecture + MVVM

Clean Architecture 的 Presenter 层自然对应 MVVM 中的 ViewModel。两者结合后的数据流为：

flowchart LR
    V["View<br/>(SwiftUI)"] -->|Calls| VM["ViewModel"]
    VM -->|Execute| UC["UseCase"]
    UC -->|Calls| RP["Repository<br/>(Protocol)"]
    RP -.->|Implements| RPI["RemoteRepository"]
    RPI -->|Request| API["API / DB"]
    API -->|Response| RPI
    RPI -.-> RP
    RP -->|Return Entity| UC
    UC -->|Return Result| VM
    VM -->|Update State| V

模式三：Use Case 单元测试

Use Case 依赖 Protocol，可以通过 Mock 完全隔离外部依赖：

class MockOrderRepository: OrderRepository {
    var savedOrders: [Order] = []

    func save(_ order: Order) async throws {
        savedOrders.append(order)
    }

    func fetchOrder(by id: String) async throws -> Order {
        guard let order = savedOrders.first(where: { $0.id == id }) else {
            throw URLError(.badURL)
        }
        return order
    }

    func fetchOrders(for userId: String) async throws -> [Order] {
        savedOrders
    }
}

class MockProductRepository: ProductRepository {
    var products: [Product] = []

    func fetchProduct(by id: String) async throws -> Product {
        guard let product = products.first(where: { $0.id == id }) else {
            throw URLError(.badURL)
        }
        return product
    }

    func updateStock(productId: String, count: Int) async throws {
        if let index = products.firstIndex(where: { $0.id == productId }) {
            products[index].stockCount = count
        }
    }
}

@Test
func placeOrderDeductsStock() async throws {
    let mockProduct = MockProductRepository()
    mockProduct.products = [
        Product(id: "p1", name: "iPhone", price: 7999, stockCount: 10)
    ]
    let mockOrder = MockOrderRepository()

    let useCase = PlaceOrderUseCase(
        orderRepository: mockOrder,
        productRepository: mockProduct
    )

    let order = try await useCase.execute(items: [("p1", 2)])

    #expect(order.items.count == 1)
    #expect(order.totalAmount == 15998)
    #expect(mockOrder.savedOrders.count == 1)
}

常见陷阱

陷阱一：Entity 依赖外部框架

Entity 是最内层，绝对不能引用 Foundation 以外的框架（如 CoreData、SwiftUI、Combine）：

struct User {
    // @Published var name: String  ← 错误：Entity 不应使用 Combine
    var name: String
}

如果需要持久化，应该在 Data 层创建对应的 Managed Object 或 DTO，而非让 Entity 直接继承 NSManagedObject。

陷阱二：Use Case 跳过 Repository 直接访问网络

Use Case 应该通过 Repository Protocol 获取数据，而非直接调用 URLSession。直接调用会破坏依赖规则，导致 Use Case 无法独立测试：

class BadUseCase {
    func execute() async throws -> [Product] {
        // let (data, _) = try await URLSession.shared.data(from: url)  ← 违反依赖规则
    }
}

class GoodUseCase {
    private let repository: ProductRepository

    func execute() async throws -> [Product] {
        try await repository.fetchAll()
    }
}

陷阱三：过度工程化

对于简单的 CRUD 功能，完整的 Clean Architecture（Entity → UseCase → Repository → DTO → ViewModel）会引入大量样板代码。应根据项目规模和复杂度选择合适的架构层级：

项目规模 推荐架构
个人/原型 MVC 或简单 MVVM
中型项目 MVVM + Repository
大型项目 Clean Architecture + MVVM
多团队协作 Clean Architecture + 模块化 SPM

陷阱四：DTO 与 Entity 混用

Data 层的 DTO（Data Transfer Object）负责序列化/反序列化，Entity 负责业务逻辑。两者不应混为一谈。DTO 的字段可能与 API 格式耦合（如 snake_case），而 Entity 使用 Swift 惯用命名：

struct UserDTO: Codable {
    let user_id: Int
    let display_name: String

    func toDomain() -> User {
        User(id: user_id, name: display_name)
    }
}

struct User {
    let id: Int
    var name: String
}

小结

  • Clean Architecture 的核心是依赖规则：代码依赖只能从外层指向内层，内层通过 Protocol 定义接口。
  • 四层结构：Entity（业务实体）→ Use Case（业务用例）→ Interface Adapter（ViewModel/Repository 实现）→ Framework（UIKit/SwiftUI/网络）。
  • Entity 不依赖任何外部框架，Use Case 通过 Repository Protocol 获取数据，确保业务逻辑可独立测试。
  • DTO 负责外部数据格式转换，与 Entity 严格分离，避免 API 变更影响核心业务。
  • Clean Architecture 与 MVVM 天然互补——ViewModel 充当 Presenter 角色，调用 Use Case 并驱动 View 更新。
  • 根据项目复杂度选择架构层级，避免对简单项目进行过度工程化。
  • 模块化（Swift Package Manager）可以在编译层面强制执行依赖规则，防止层级穿透。
