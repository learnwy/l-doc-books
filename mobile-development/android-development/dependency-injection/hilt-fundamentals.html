Hilt 基础

Hilt 是 Google 推荐的 Android 依赖注入框架，建立在 Dagger 之上，为 Android 应用提供标准化的 DI 方案。Hilt 自动生成 Dagger 组件和组件层级，大幅简化了配置工作。

@HiltAndroidApp

@HiltAndroidApp 注解标记 Application 类，触发 Hilt 的代码生成，创建应用级别的依赖容器。

@HiltAndroidApp
class MyApplication : Application() {
    override fun onCreate() {
        super.onCreate()
    }
}

Hilt 的初始化流程：



提示： @HiltAndroidApp 必须注解在 Application 类上，它是整个 Hilt 依赖图的根节点。






@AndroidEntryPoint
@AndroidEntryPoint 注解使 Android 组件（Activity、Fragment、Service 等）支持依赖注入。


@AndroidEntryPoint
class MainActivity : ComponentActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            MyApp()
        }
    }
}

@AndroidEntryPoint
class SyncService : Service() {

    @Inject
    lateinit var syncRepository: SyncRepository

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        CoroutineScope(Dispatchers.IO).launch {
            syncRepository.performSync()
            stopSelf()
        }
        return START_NOT_STICKY
    }
}

@Inject
@Inject 注解用于构造函数注入和字段注入两种方式。


class ArticleRepository @Inject constructor(
    private val api: ArticleApi,
    private val dao: ArticleDao,
    private val dispatcher: CoroutineDispatcher
) {
    suspend fun getArticles(): List<Article> = withContext(dispatcher) {
        try {
            val remote = api.fetchArticles()
            dao.insertArticles(remote.map { it.toEntity() })
            remote
        } catch (e: Exception) {
            dao.getAllArticles().first().map { it.toDomain() }
        }
    }
}

class AnalyticsTracker @Inject constructor(
    @ApplicationContext private val context: Context,
    private val preferences: DataStore<Preferences>
) {
    fun trackEvent(name: String, params: Map<String, String> = emptyMap()) {
        // 追踪事件
    }
}

Module 与 Binding
当无法使用构造函数注入时（如接口、第三方库类），需要通过 Module 提供绑定。


@Module
@InstallIn(SingletonComponent::class)
abstract class RepositoryModule {

    @Binds
    @Singleton
    abstract fun bindArticleRepository(
        impl: ArticleRepositoryImpl
    ): ArticleRepository

    @Binds
    abstract fun bindAnalyticsService(
        impl: FirebaseAnalyticsService
    ): AnalyticsService
}

@Module
@InstallIn(SingletonComponent::class)
object NetworkModule {

    @Provides
    @Singleton
    fun provideOkHttpClient(): OkHttpClient {
        return OkHttpClient.Builder()
            .connectTimeout(30, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .addInterceptor(HttpLoggingInterceptor().apply {
                level = HttpLoggingInterceptor.Level.BODY
            })
            .build()
    }

    @Provides
    @Singleton
    fun provideRetrofit(okHttpClient: OkHttpClient): Retrofit {
        return Retrofit.Builder()
            .baseUrl("https://api.example.com/")
            .client(okHttpClient)
            .addConverterFactory(Json.asConverterFactory("application/json".toMediaType()))
            .build()
    }

    @Provides
    @Singleton
    fun provideArticleApi(retrofit: Retrofit): ArticleApi {
        return retrofit.create(ArticleApi::class.java)
    }
}
