Hilt 测试

Hilt 提供了完善的测试支持，允许在测试中替换绑定、卸载模块以及创建自定义测试组件。正确配置 Hilt 测试环境是编写可靠 Android 测试的关键。

@HiltAndroidTest

@HiltAndroidTest 注解标记测试类，使其支持 Hilt 依赖注入。配合 HiltAndroidRule 管理 Hilt 的生命周期。

@HiltAndroidTest
@RunWith(AndroidJUnit4::class)
class UserFlowTest {

    @get:Rule(order = 0)
    val hiltRule = HiltAndroidRule(this)

    @get:Rule(order = 1)
    val composeRule = createAndroidComposeRule<MainActivity>()

    @Inject
    lateinit var userRepository: UserRepository

    @Before
    fun setup() {
        hiltRule.inject()
    }

    @Test
    fun userProfile_displaysCorrectName() {
        composeRule.onNodeWithText("用户名").assertIsDisplayed()
    }
}

测试模块
使用 @TestInstallIn 全局替换模块，或使用 @UninstallModules 在单个测试类中替换。


@Module
@TestInstallIn(
    components = [SingletonComponent::class],
    replaces = [NetworkModule::class]
)
object FakeNetworkModule {

    @Provides
    @Singleton
    fun provideArticleApi(): ArticleApi {
        return FakeArticleApi()
    }

    @Provides
    @Singleton
    fun provideOkHttpClient(): OkHttpClient {
        return OkHttpClient.Builder().build()
    }
}

class FakeArticleApi : ArticleApi {

    private val articles = mutableListOf(
        Article("1", "测试文章一", "内容一"),
        Article("2", "测试文章二", "内容二")
    )

    override suspend fun getArticles(): List<Article> = articles

    override suspend fun getArticle(id: String): Article {
        return articles.first { it.id == id }
    }

    fun addArticle(article: Article) {
        articles.add(article)
    }
}

卸载模块
@UninstallModules 在单个测试类中卸载生产模块，然后通过内嵌模块提供替代实现。


@HiltAndroidTest
@UninstallModules(RepositoryModule::class)
@RunWith(AndroidJUnit4::class)
class ArticleListScreenTest {

    @get:Rule(order = 0)
    val hiltRule = HiltAndroidRule(this)

    @get:Rule(order = 1)
    val composeRule = createAndroidComposeRule<MainActivity>()

    @Module
    @InstallIn(SingletonComponent::class)
    object TestRepositoryModule {

        @Provides
        @Singleton
        fun provideArticleRepository(): ArticleRepository {
            return FakeArticleRepository().apply {
                setArticles(
                    listOf(
                        Article("1", "Kotlin 协程指南", "详细的协程教程..."),
                        Article("2", "Compose 最佳实践", "Compose 开发技巧...")
                    )
                )
            }
        }
    }

    @Before
    fun setup() {
        hiltRule.inject()
    }

    @Test
    fun articleList_displaysAllArticles() {
        composeRule.onNodeWithText("Kotlin 协程指南").assertIsDisplayed()
        composeRule.onNodeWithText("Compose 最佳实践").assertIsDisplayed()
    }

    @Test
    fun articleList_showsEmptyState_whenNoArticles() {
        composeRule.onNodeWithText("暂无文章").assertIsDisplayed()
    }
}


自定义测试组件
对于复杂的测试场景，可以创建自定义测试 Runner 和 Application。


class CustomTestRunner : AndroidJUnitRunner() {
    override fun newApplication(
        cl: ClassLoader?,
        className: String?,
        context: Context?
    ): Application {
        return super.newApplication(cl, HiltTestApplication::class.java.name, context)
    }
}

ViewModel 的单元测试可以不使用 Hilt，直接构造：


class ArticleViewModelTest {

    @get:Rule
    val mainDispatcherRule = MainDispatcherRule()

    private lateinit var viewModel: ArticleViewModel
    private val fakeRepository = FakeArticleRepository()

    @Before
    fun setup() {
        viewModel = ArticleViewModel(
            repository = fakeRepository,
            savedStateHandle = SavedStateHandle(mapOf("articleId" to "1"))
        )
    }

    @Test
    fun `loading state is shown initially`() = runTest {
        val state = viewModel.uiState.value
        assertTrue(state.isLoading)
    }

    @Test
    fun `articles loaded successfully`() = runTest {
        fakeRepository.setArticles(listOf(Article("1", "测试", "内容")))

        advanceUntilIdle()

        val state = viewModel.uiState.value
        assertFalse(state.isLoading)
        assertEquals(1, state.articles.size)
    }

    @Test
    fun `error state on failure`() = runTest {
        fakeRepository.setShouldFail(true)

        advanceUntilIdle()

        val state = viewModel.uiState.value
        assertNotNull(state.error)
    }
}

class MainDispatcherRule(
    private val dispatcher: TestDispatcher = UnconfinedTestDispatcher()
) : TestWatcher() {
    override fun starting(description: Description) {
        Dispatchers.setMain(dispatcher)
    }

    override fun finished(description: Description) {
        Dispatchers.resetMain()
    }
}

小结
    

  • @HiltAndroidTest + HiltAndroidRule 是 Hilt 测试的基础配置
    • 

  • @TestInstallIn 全局替换模块，@UninstallModules 在单个测试类中替换
    • 

  • 内嵌的 @Module 提供测试专用的 Fake 实现
    • 

  • 自定义 AndroidJUnitRunner 配合 HiltTestApplication 初始化测试环境
    • 

  • ViewModel 和 Repository 的单元测试不需要 Hilt，直接构造更简洁高效
