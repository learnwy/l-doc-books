Compose UI 测试

Jetpack Compose 提供了内置的 UI 测试框架，通过语义树（Semantic Tree）查找和操作 UI 元素。Compose 测试运行速度比传统 Espresso 更快，API 更简洁。

ComposeTestRule

ComposeTestRule 是 Compose UI 测试的入口，提供设置 Composable 内容和查找节点的能力。

class LoginScreenTest {

    @get:Rule
    val composeRule = createComposeRule()

    @Test
    fun loginScreen_displaysAllElements() {
        composeRule.setContent {
            LoginScreen(
                onLogin = {},
                onForgotPassword = {}
            )
        }

        composeRule.onNodeWithText("用户名").assertIsDisplayed()
        composeRule.onNodeWithText("密码").assertIsDisplayed()
        composeRule.onNodeWithText("登录").assertIsDisplayed()
        composeRule.onNodeWithText("忘记密码？").assertIsDisplayed()
    }

    @Test
    fun loginButton_disabledWhenFieldsEmpty() {
        composeRule.setContent {
            LoginScreen(onLogin = {}, onForgotPassword = {})
        }

        composeRule.onNodeWithText("登录").assertIsNotEnabled()
    }

    @Test
    fun loginButton_enabledWhenFieldsFilled() {
        composeRule.setContent {
            LoginScreen(onLogin = {}, onForgotPassword = {})
        }

        composeRule.onNodeWithText("用户名").performTextInput("admin")
        composeRule.onNodeWithText("密码").performTextInput("password123")

        composeRule.onNodeWithText("登录").assertIsEnabled()
    }
}

两种 TestRule 对比：

Rule 用途 Hilt 支持
createComposeRule() 独立 Composable 测试 否
createAndroidComposeRule<Activity>() Activity 集成测试 是（配合 @HiltAndroidTest）

提示： 优先使用 createComposeRule() 测试独立 Composable，只有需要 Activity 上下文或 Hilt 注入时才使用 createAndroidComposeRule。

语义匹配器

Compose 测试通过语义属性查找 UI 节点，支持多种匹配方式。

class ProductListScreenTest {

    @get:Rule
    val composeRule = createComposeRule()

    @Test
    fun productList_displaysItems() {
        val products = listOf(
            Product("1", "Kotlin 实战", 89.0),
            Product("2", "Compose 入门", 79.0)
        )

        composeRule.setContent {
            ProductListScreen(products = products)
        }

        composeRule.onNodeWithText("Kotlin 实战").assertIsDisplayed()
        composeRule.onNodeWithText("¥89.0").assertIsDisplayed()
        composeRule.onNodeWithText("Compose 入门").assertIsDisplayed()
    }

    @Test
    fun searchBar_filtersProducts() {
        composeRule.setContent {
            ProductListScreen(products = sampleProducts)
        }

        composeRule.onNodeWithTag("search_bar").performTextInput("Kotlin")
        composeRule.onNodeWithText("Kotlin 实战").assertIsDisplayed()
        composeRule.onNodeWithText("Compose 入门").assertDoesNotExist()
    }

    @Test
    fun productCard_showsCorrectBadge() {
        composeRule.setContent {
            ProductCard(product = Product("1", "热销", 99.0, isHot = true))
        }

        composeRule
            .onNodeWithContentDescription("热销标记")
            .assertIsDisplayed()
    }
}

常用查找方法：

方法 匹配依据 示例
onNodeWithText 文本内容 onNodeWithText("登录")
onNodeWithTag 测试标签 onNodeWithTag("submit_btn")
onNodeWithContentDescription 无障碍描述 onNodeWithContentDescription("返回")
onNode(hasText("")) 语义匹配器 灵活组合条件
onAllNodesWithText 多节点匹配 onAllNodesWithText("删除") 
在代码中添加测试标签：


@Composable
fun SearchBar(query: String, onQueryChange: (String) -> Unit) {
    OutlinedTextField(
        value = query,
        onValueChange = onQueryChange,
        modifier = Modifier
            .fillMaxWidth()
            .testTag("search_bar"),
        placeholder = { Text("搜索...") }
    )
}


断言
断言验证节点的属性和状态。


@Test
fun toggleButton_switchesState() {
    var isEnabled by mutableStateOf(false)

    composeRule.setContent {
        Switch(
            checked = isEnabled,
            onCheckedChange = { isEnabled = it },
            modifier = Modifier.semantics { testTag = "toggle" }
        )
    }

    composeRule.onNodeWithTag("toggle")
        .assertIsOff()
        .performClick()

    composeRule.onNodeWithTag("toggle")
        .assertIsOn()
}

@Test
fun lazyColumn_displaysCorrectItemCount() {
    composeRule.setContent {
        LazyColumn {
            items(20) { index ->
                Text("Item $index", modifier = Modifier.testTag("item_$index"))
            }
        }
    }

    composeRule.onNodeWithTag("item_0").assertIsDisplayed()
    composeRule.onNodeWithTag("item_19").assertDoesNotExist()

    composeRule.onNodeWithTag("item_0").performScrollTo()
}

操作
操作模拟用户在 UI 上的交互。


@Test
fun swipeToDelete_removesItem() {
    val items = mutableStateListOf("Item 1", "Item 2", "Item 3")

    composeRule.setContent {
        SwipeableList(items = items, onDelete = { items.remove(it) })
    }

    composeRule.onNodeWithText("Item 2").performTouchInput {
        swipeLeft()
    }

    composeRule.onNodeWithText("Item 2").assertDoesNotExist()
    composeRule.onAllNodesWithTag("list_item").assertCountEquals(2)
}

@Test
fun longPress_showsContextMenu() {
    composeRule.setContent {
        ArticleCard(article = sampleArticle)
    }

    composeRule.onNodeWithText(sampleArticle.title).performTouchInput {
        longClick()
    }

    composeRule.onNodeWithText("编辑").assertIsDisplayed()
    composeRule.onNodeWithText("删除").assertIsDisplayed()
}

截图测试
Compose 支持与截图测试工具集成，捕获 UI 快照进行视觉回归测试。


@Test
fun productCard_snapshot() {
    composeRule.setContent {
        ProductCard(
            product = Product("1", "测试商品", 99.0, imageUrl = ""),
            onClick = {}
        )
    }

    composeRule.onNodeWithTag("product_card")
        .captureToImage()
        .assertAgainstGolden(screenshotRule, "product_card_default")
}


提示： 使用 Modifier.testTag() 为需要测试的 Composable 添加标签，这是推荐的定位方式，比依赖文本内容更稳定。






小结
    

  • ComposeTestRule 是 Compose UI 测试的入口，支持独立和集成两种模式
    • 

  • 语义匹配器通过文本、标签、内容描述等方式查找 UI 节点
    • 

  • 断言验证节点的显示状态、启用状态、文本内容等属性
    • 

  • 操作模拟点击、输入、滑动等用户交互
    • 

  • Modifier.testTag() 是推荐的节点定位方式，比文本匹配更稳定
