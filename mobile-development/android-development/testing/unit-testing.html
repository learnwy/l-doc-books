单元测试

单元测试是 Android 测试金字塔的基础，运行在 JVM 上，速度快、反馈及时。现代 Android 单元测试使用 JUnit 5 作为测试框架，MockK 作为 Kotlin 友好的 Mock 库，Turbine 测试 Flow，配合协程测试工具验证异步逻辑。

JUnit 5

JUnit 5 是 Java/Kotlin 生态的标准测试框架，提供灵活的注解、参数化测试和扩展机制。

class ArticleValidatorTest {

    private lateinit var validator: ArticleValidator

    @BeforeEach
    fun setup() {
        validator = ArticleValidator()
    }

    @Test
    fun `valid article passes validation`() {
        val article = Article(id = "1", title = "标题", content = "内容...")
        val result = validator.validate(article)
        assertTrue(result.isValid)
    }

    @Test
    fun `empty title fails validation`() {
        val article = Article(id = "1", title = "", content = "内容...")
        val result = validator.validate(article)
        assertFalse(result.isValid)
        assertEquals("标题不能为空", result.errors.first())
    }

    @ParameterizedTest
    @ValueSource(strings = ["", " ", "  "])
    fun `blank titles fail validation`(title: String) {
        val article = Article(id = "1", title = title, content = "内容")
        assertFalse(validator.validate(article).isValid)
    }

    @Nested
    inner class ContentValidation {
        @Test
        fun `content exceeding max length fails`() {
            val longContent = "a".repeat(10001)
            val article = Article(id = "1", title = "标题", content = longContent)
            assertFalse(validator.validate(article).isValid)
        }
    }
}

JUnit 5 常用注解：

注解 用途 说明
@Test 标记测试方法 基本测试
@BeforeEach / @AfterEach 每个测试前后 初始化/清理
@BeforeAll / @AfterAll 所有测试前后 共享资源
@Nested 嵌套测试类 逻辑分组
@ParameterizedTest 参数化测试 多数据集验证
@DisplayName 自定义名称 可读性

MockK

MockK 是 Kotlin 原生的 Mock 框架，支持协程、扩展函数和 DSL 语法。

class ArticleRepositoryTest {

    private val api: ArticleApi = mockk()
    private val dao: ArticleDao = mockk(relaxed = true)
    private val repository = ArticleRepository(api, dao)

    @Test
    fun `getArticles returns remote data and caches locally`() = runTest {
        val remoteArticles = listOf(
            ArticleDto("1", "远程文章一", "内容一"),
            ArticleDto("2", "远程文章二", "内容二")
        )
        coEvery { api.getArticles() } returns ApiResponse(200, "ok", remoteArticles)

        val result = repository.getArticles()

        assertEquals(2, result.size)
        coVerify { dao.insertArticles(any()) }
    }

    @Test
    fun `getArticles falls back to local data on network error`() = runTest {
        val localArticles = listOf(
            ArticleEntity("1", "本地文章", "内容", "a1", 0L)
        )
        coEvery { api.getArticles() } throws IOException("网络错误")
        every { dao.getAllArticles() } returns flowOf(localArticles)

        val result = repository.getArticles()

        assertEquals(1, result.size)
        assertEquals("本地文章", result[0].title)
    }

    @Test
    fun `deleteArticle calls API and DAO`() = runTest {
        coEvery { api.deleteArticle("1") } returns ApiResponse(200, "ok", Unit)

        repository.deleteArticle("1")

        coVerifyOrder {
            api.deleteArticle("1")
            dao.deleteArticle(any())
        }
    }
}

MockK 常用函数：

函数 用途 示例
mockk() 创建 Mock 对象 mockk<ArticleApi>()
every 定义普通函数返回值 every { dao.getAll() } returns flow
coEvery 定义挂起函数返回值 coEvery { api.get() } returns data
verify 验证普通函数调用 verify { dao.insert(any()) }
coVerify 验证挂起函数调用 coVerify { api.delete("1") }
relaxed = true 返回默认值 避免手动 stub 所有方法

Turbine（Flow 测试）

Turbine 是 Square 出品的 Flow 测试库，提供简洁的 API 验证 Flow 发射序列。

class SearchViewModelTest {

    @get:Rule
    val mainDispatcherRule = MainDispatcherRule()

    private val repository = FakeSearchRepository()
    private lateinit var viewModel: SearchViewModel

    @BeforeEach
    fun setup() {
        viewModel = SearchViewModel(SavedStateHandle(), repository)
    }

    @Test
    fun `search emits loading then results`() = runTest {
        repository.setResults(listOf(SearchItem("1", "Kotlin")))

        viewModel.uiState.test {
            assertEquals(SearchUiState(), awaitItem())

            viewModel.search("Kotlin")

            val loading = awaitItem()
            assertTrue(loading.isLoading)

            val success = awaitItem()
            assertFalse(success.isLoading)
            assertEquals(1, success.results.size)

            cancelAndIgnoreRemainingEvents()
        }
    }

    @Test
    fun `events flow emits snackbar on error`() = runTest {
        repository.setShouldFail(true)

        viewModel.events.test {
            viewModel.search("error")

            val event = awaitItem()
            assertIs<UiEvent.ShowSnackbar>(event)
            assertTrue(event.message.contains("失败"))

            cancelAndIgnoreRemainingEvents()
        }
    }
}


协程测试
kotlinx-coroutines-test 提供控制协程执行的工具。


class MainDispatcherRule(
    private val dispatcher: TestDispatcher = UnconfinedTestDispatcher()
) : TestWatcher() {
    override fun starting(description: Description) {
        Dispatchers.setMain(dispatcher)
    }
    override fun finished(description: Description) {
        Dispatchers.resetMain()
    }
}

class DataSyncServiceTest {

    @get:Rule
    val mainDispatcherRule = MainDispatcherRule()

    @Test
    fun `sync completes all tasks in order`() = runTest {
        val service = DataSyncService(
            testScheduler = StandardTestDispatcher(testScheduler)
        )

        service.startSync()
        advanceUntilIdle()

        assertTrue(service.isSyncComplete)
    }

    @Test
    fun `timeout cancels long-running operation`() = runTest {
        val service = SlowService()

        val result = withTimeoutOrNull(1000) {
            service.longOperation()
        }

        assertNull(result)
    }
}

runTest

advanceTimeBy(ms)

ViewModel 测试
结合上述工具对 ViewModel 进行完整测试。


class ArticleListViewModelTest {

    @get:Rule
    val mainDispatcherRule = MainDispatcherRule()

    private val repository = FakeArticleRepository()
    private lateinit var viewModel: ArticleListViewModel

    @BeforeEach
    fun setup() {
        viewModel = ArticleListViewModel(repository)
    }

    @Test
    fun `initial state is loading`() = runTest {
        viewModel.uiState.test {
            val initial = awaitItem()
            assertTrue(initial.isLoading)
            cancelAndIgnoreRemainingEvents()
        }
    }

    @Test
    fun `refresh updates articles`() = runTest {
        repository.setArticles(listOf(Article("1", "文章一", "内容")))

        viewModel.uiState.test {
            skipItems(1)

            val loaded = awaitItem()
            assertEquals(1, loaded.articles.size)

            repository.setArticles(listOf(
                Article("1", "文章一", "内容"),
                Article("2", "文章二", "内容")
            ))
            viewModel.refresh()

            val refreshing = awaitItem()
            assertTrue(refreshing.isRefreshing)

            val refreshed = awaitItem()
            assertEquals(2, refreshed.articles.size)

            cancelAndIgnoreRemainingEvents()
        }
    }
}


提示： 使用 UnconfinedTestDispatcher 可以让协程立即执行，适合大多数 ViewModel 测试；当需要精确控制执行顺序时使用 StandardTestDispatcher。






小结
    

  • JUnit 5 提供 @Nested、参数化测试等现代测试特性
    • 

  • MockK 是 Kotlin 原生 Mock 框架，支持协程、DSL 语法和验证调用顺序
    • 

  • Turbine 简化 Flow 测试，通过 test {} 块按序验证发射值
    • 

  • MainDispatcherRule 替换 Dispatchers.Main，使 ViewModel 可在 JVM 上测试
    • 

  • ViewModel 测试结合 Fake Repository + Turbine + 协程测试工具实现完整验证
