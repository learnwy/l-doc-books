图片加载

图片加载是 Android 应用的常见需求。Coil（Coroutine Image Loader）是为 Kotlin 和 Compose 设计的现代图片加载库，以协程为核心，API 简洁，性能优秀。

Coil 基础（Compose-first）

Coil 提供了原生的 Compose 支持，AsyncImage 是最常用的图片加载组件。

@Composable
fun UserAvatar(imageUrl: String, modifier: Modifier = Modifier) {
    AsyncImage(
        model = imageUrl,
        contentDescription = "用户头像",
        modifier = modifier
            .size(64.dp)
            .clip(CircleShape),
        contentScale = ContentScale.Crop
    )
}

@Composable
fun ArticleImage(imageUrl: String, modifier: Modifier = Modifier) {
    AsyncImage(
        model = ImageRequest.Builder(LocalContext.current)
            .data(imageUrl)
            .crossfade(true)
            .build(),
        contentDescription = "文章封面",
        modifier = modifier
            .fillMaxWidth()
            .height(200.dp)
            .clip(RoundedCornerShape(8.dp)),
        contentScale = ContentScale.Crop
    )
}

@Composable
fun ProductGallery(images: List<String>) {
    LazyRow(
        contentPadding = PaddingValues(horizontal = 16.dp),
        horizontalArrangement = Arrangement.spacedBy(8.dp)
    ) {
        items(images) { url ->
            AsyncImage(
                model = url,
                contentDescription = null,
                modifier = Modifier
                    .size(120.dp)
                    .clip(RoundedCornerShape(8.dp)),
                contentScale = ContentScale.Crop
            )
        }
    }
}

Placeholder / Error / Crossfade

完善的加载状态管理提升用户体验。

@Composable
fun RobustImage(imageUrl: String, modifier: Modifier = Modifier) {
    SubcomposeAsyncImage(
        model = ImageRequest.Builder(LocalContext.current)
            .data(imageUrl)
            .crossfade(300)
            .build(),
        contentDescription = null,
        modifier = modifier
    ) {
        val state = painter.state
        when (state) {
            is AsyncImagePainter.State.Loading -> {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .shimmerEffect(),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator(modifier = Modifier.size(24.dp))
                }
            }
            is AsyncImagePainter.State.Error -> {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(MaterialTheme.colorScheme.errorContainer),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(
                        Icons.Default.BrokenImage,
                        contentDescription = "加载失败",
                        tint = MaterialTheme.colorScheme.onErrorContainer
                    )
                }
            }
            else -> {
                SubcomposeAsyncImageContent()
            }
        }
    }
}

@Composable
fun SimpleImageWithPlaceholder(imageUrl: String, modifier: Modifier = Modifier) {
    AsyncImage(
        model = ImageRequest.Builder(LocalContext.current)
            .data(imageUrl)
            .crossfade(true)
            .placeholder(R.drawable.placeholder)
            .error(R.drawable.error_image)
            .fallback(R.drawable.default_avatar)
            .build(),
        contentDescription = null,
        modifier = modifier
    )
}

磁盘/内存缓存
Coil 提供两级缓存：内存缓存和磁盘缓存。


@Module
@InstallIn(SingletonComponent::class)
object ImageModule {

    @Provides
    @Singleton
    fun provideImageLoader(@ApplicationContext context: Context): ImageLoader {
        return ImageLoader.Builder(context)
            .memoryCache {
                MemoryCache.Builder(context)
                    .maxSizePercent(0.25)
                    .build()
            }
            .diskCache {
                DiskCache.Builder()
                    .directory(context.cacheDir.resolve("image_cache"))
                    .maxSizePercent(0.05)
                    .build()
            }
            .crossfade(true)
            .respectCacheHeaders(true)
            .logger(if (BuildConfig.DEBUG) DebugLogger() else null)
            .build()
    }
}

@Composable
fun CachedImage(imageUrl: String, cachePolicy: CachePolicy = CachePolicy.ENABLED) {
    AsyncImage(
        model = ImageRequest.Builder(LocalContext.current)
            .data(imageUrl)
            .memoryCachePolicy(cachePolicy)
            .diskCachePolicy(cachePolicy)
            .memoryCacheKey(imageUrl)
            .diskCacheKey(imageUrl)
            .build(),
        contentDescription = null
    )
}

图片变换
Coil 支持多种图片变换，可以在加载时应用。


@Composable
fun TransformedImages() {
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        AsyncImage(
            model = ImageRequest.Builder(LocalContext.current)
                .data("https://example.com/photo.jpg")
                .transformations(CircleCropTransformation())
                .build(),
            contentDescription = "圆形裁剪"
        )

        AsyncImage(
            model = ImageRequest.Builder(LocalContext.current)
                .data("https://example.com/photo.jpg")
                .transformations(RoundedCornersTransformation(16f))
                .build(),
            contentDescription = "圆角"
        )

        AsyncImage(
            model = ImageRequest.Builder(LocalContext.current)
                .data("https://example.com/photo.jpg")
                .transformations(BlurTransformation(LocalContext.current, 10f, 2f))
                .build(),
            contentDescription = "模糊"
        )

        AsyncImage(
            model = ImageRequest.Builder(LocalContext.current)
                .data("https://example.com/photo.jpg")
                .transformations(GrayscaleTransformation())
                .build(),
            contentDescription = "灰度"
        )
    }
}

也可以实现自定义变换：


class WatermarkTransformation(
    private val watermarkText: String
) : Transformation {

    override val cacheKey: String = "watermark_$watermarkText"

    override suspend fun transform(input: Bitmap, size: Size): Bitmap {
        val output = input.copy(input.config, true)
        val canvas = Canvas(output)
        val paint = Paint().apply {
            color = Color.WHITE
            alpha = 128
            textSize = 48f
            isAntiAlias = true
        }
        canvas.drawText(watermarkText, 20f, output.height - 20f, paint)
        return output
    }
}


提示： 在 Compose 中使用 Modifier.clip(CircleShape) 配合 ContentScale.Crop 通常比 CircleCropTransformation 更高效，因为避免了额外的 Bitmap 操作。






小结
    

  • Coil 是 Compose-first 的图片加载库，AsyncImage 是最常用的加载组件
    • 

  • SubcomposeAsyncImage 支持自定义加载状态、错误状态的 Composable
    • 

  • 两级缓存（内存 + 磁盘）自动管理，通过 CachePolicy 控制缓存行为
    • 

  • 内置 CircleCrop、RoundedCorners、Blur、Grayscale 等变换
    • 

  • 自定义 ImageLoader 通过 Hilt 注入，统一配置缓存大小和默认行为
