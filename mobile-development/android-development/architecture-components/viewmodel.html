ViewModel 视图模型

ViewModel 是 Android Jetpack 架构组件的核心，负责以生命周期感知的方式管理和存储 UI 相关数据。它在配置变更（如屏幕旋转）时存活，避免数据重复加载和内存泄漏。

生命周期感知

ViewModel 的生命周期与创建它的 ViewModelStoreOwner（通常是 Activity 或 Fragment）绑定，但比 Activity/Fragment 的生命周期更长。

class UserProfileViewModel(
    private val userRepository: UserRepository
) : ViewModel() {

    private val _uiState = MutableStateFlow(UserProfileUiState())
    val uiState: StateFlow<UserProfileUiState> = _uiState.asStateFlow()

    init {
        loadUserProfile()
    }

    private fun loadUserProfile() {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            userRepository.getCurrentUser()
                .onSuccess { user ->
                    _uiState.update {
                        it.copy(user = user, isLoading = false)
                    }
                }
                .onFailure { error ->
                    _uiState.update {
                        it.copy(errorMessage = error.message, isLoading = false)
                    }
                }
        }
    }

    override fun onCleared() {
        super.onCleared()
    }
}

data class UserProfileUiState(
    val user: User? = null,
    val isLoading: Boolean = false,
    val errorMessage: String? = null
)

SavedStateHandle
SavedStateHandle 让 ViewModel 能在进程死亡后恢复状态，它本质上是一个键值对存储。


class SearchViewModel(
    private val savedStateHandle: SavedStateHandle,
    private val searchRepository: SearchRepository
) : ViewModel() {

    val searchQuery = savedStateHandle.getStateFlow("query", "")

    val searchResults = searchQuery
        .debounce(300)
        .filter { it.length >= 2 }
        .flatMapLatest { query ->
            searchRepository.search(query)
        }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    fun updateQuery(query: String) {
        savedStateHandle["query"] = query
    }

    val selectedFilters: StateFlow<Set<String>> =
        savedStateHandle.getStateFlow("filters", emptySet())

    fun toggleFilter(filter: String) {
        val current = savedStateHandle.get<Set<String>>("filters") ?: emptySet()
        savedStateHandle["filters"] = if (filter in current) current - filter else current + filter
    }
}

数据共享
多个 Fragment 或 Composable 可以通过共享 ViewModel 来通信。


@HiltViewModel
class SharedCartViewModel @Inject constructor(
    private val cartRepository: CartRepository
) : ViewModel() {

    private val _cartItems = MutableStateFlow<List<CartItem>>(emptyList())
    val cartItems = _cartItems.asStateFlow()

    val totalPrice = cartItems.map { items ->
        items.sumOf { it.price * it.quantity }
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), 0.0)

    val itemCount = cartItems.map { it.size }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), 0)

    fun addItem(product: Product) {
        viewModelScope.launch { cartRepository.addToCart(product) }
    }

    fun removeItem(itemId: String) {
        viewModelScope.launch { cartRepository.removeFromCart(itemId) }
    }
}

@Composable
fun ProductScreen(
    sharedCartViewModel: SharedCartViewModel = hiltViewModel(
        viewModelStoreOwner = LocalContext.current as ComponentActivity
    )
) {
    val itemCount by sharedCartViewModel.itemCount.collectAsStateWithLifecycle()
    // Activity 级别的共享 ViewModel
}


Hilt 集成
Hilt 简化了 ViewModel 的依赖注入，通过 @HiltViewModel 和 @Inject 注解自动提供依赖。


@HiltViewModel
class OrderViewModel @Inject constructor(
    private val orderRepository: OrderRepository,
    private val analyticsTracker: AnalyticsTracker,
    savedStateHandle: SavedStateHandle
) : ViewModel() {

    private val orderId: String = checkNotNull(savedStateHandle["orderId"])

    val order = orderRepository.getOrderById(orderId)
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), null)

    fun cancelOrder() {
        viewModelScope.launch {
            orderRepository.cancelOrder(orderId)
            analyticsTracker.trackEvent("order_cancelled", mapOf("orderId" to orderId))
        }
    }
}

@Composable
fun OrderDetailScreen(viewModel: OrderViewModel = hiltViewModel()) {
    val order by viewModel.order.collectAsStateWithLifecycle()
    // UI 代码
}


测试
ViewModel 的业务逻辑独立于 Android 框架，可以通过纯 JVM 单元测试进行验证。


class SearchViewModelTest {

    @get:Rule
    val mainDispatcherRule = MainDispatcherRule()

    private lateinit var viewModel: SearchViewModel
    private val fakeRepository = FakeSearchRepository()

    @Before
    fun setup() {
        viewModel = SearchViewModel(
            savedStateHandle = SavedStateHandle(),
            searchRepository = fakeRepository
        )
    }

    @Test
    fun `updateQuery updates savedStateHandle and triggers search`() = runTest {
        viewModel.updateQuery("Kotlin")

        val results = viewModel.searchResults.first()
        assertEquals("Kotlin", viewModel.searchQuery.value)
        assertTrue(results.isNotEmpty())
    }

    @Test
    fun `short query does not trigger search`() = runTest {
        viewModel.updateQuery("K")

        advanceUntilIdle()
        val results = viewModel.searchResults.value
        assertTrue(results.isEmpty())
    }
}
