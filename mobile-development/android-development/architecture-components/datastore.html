DataStore 数据存储

DataStore 是 Android Jetpack 提供的数据存储解决方案，用于替代 SharedPreferences。它基于 Kotlin 协程和 Flow 构建，提供异步、一致性和事务性的数据存储能力。DataStore 有两种实现：Preferences DataStore 和 Proto DataStore。

Preferences DataStore

Preferences DataStore 以键值对形式存储数据，类似 SharedPreferences 但具备异步安全性。

val Context.settingsDataStore: DataStore<Preferences> by preferencesDataStore(
    name = "settings"
)

object PreferencesKeys {
    val DARK_MODE = booleanPreferencesKey("dark_mode")
    val LANGUAGE = stringPreferencesKey("language")
    val FONT_SIZE = intPreferencesKey("font_size")
    val NOTIFICATIONS_ENABLED = booleanPreferencesKey("notifications_enabled")
    val LAST_SYNC_TIME = longPreferencesKey("last_sync_time")
}

class SettingsRepository(private val dataStore: DataStore<Preferences>) {

    val settings: Flow<AppSettings> = dataStore.data
        .catch { exception ->
            if (exception is IOException) emit(emptyPreferences())
            else throw exception
        }
        .map { preferences ->
            AppSettings(
                isDarkMode = preferences[PreferencesKeys.DARK_MODE] ?: false,
                language = preferences[PreferencesKeys.LANGUAGE] ?: "zh",
                fontSize = preferences[PreferencesKeys.FONT_SIZE] ?: 14,
                notificationsEnabled = preferences[PreferencesKeys.NOTIFICATIONS_ENABLED] ?: true
            )
        }

    suspend fun updateDarkMode(enabled: Boolean) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.DARK_MODE] = enabled
        }
    }

    suspend fun updateLanguage(language: String) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.LANGUAGE] = language
        }
    }
}

data class AppSettings(
    val isDarkMode: Boolean = false,
    val language: String = "zh",
    val fontSize: Int = 14,
    val notificationsEnabled: Boolean = true
)

Preferences DataStore 支持的类型：

键类型函数 存储类型 示例
intPreferencesKey Int 字体大小
longPreferencesKey Long 时间戳
floatPreferencesKey Float 亮度级别
doublePreferencesKey Double 精确数值
booleanPreferencesKey Boolean 开关状态
stringPreferencesKey String 语言代码
stringSetPreferencesKey Set<String> 标签集合

提示： dataStore.edit 是事务性操作，保证读-改-写的原子性，不会出现竞态条件。

Proto DataStore

Proto DataStore 使用 Protocol Buffers 序列化结构化数据，提供类型安全的存储。

// user_preferences.proto
syntax = "proto3";

option java_package = "com.example.app";
option java_multiple_files = true;

message UserPreferences {
  bool dark_mode = 1;
  string language = 2;
  int32 font_size = 3;
  Theme theme = 4;

  enum Theme {
    SYSTEM = 0;
    LIGHT = 1;
    DARK = 2;
  }
}
object UserPreferencesSerializer : Serializer<UserPreferences> {
    override val defaultValue: UserPreferences = UserPreferences.getDefaultInstance()

    override suspend fun readFrom(input: InputStream): UserPreferences {
        try {
            return UserPreferences.parseFrom(input)
        } catch (exception: InvalidProtocolBufferException) {
            throw CorruptionException("Cannot read proto.", exception)
        }
    }

    override suspend fun writeTo(t: UserPreferences, output: OutputStream) {
        t.writeTo(output)
    }
}

val Context.userPreferencesStore: DataStore<UserPreferences> by dataStore(
    fileName = "user_preferences.pb",
    serializer = UserPreferencesSerializer
)

class UserPreferencesRepository(
    private val dataStore: DataStore<UserPreferences>
) {

    val userPreferences: Flow<UserPreferences> = dataStore.data

    suspend fun setTheme(theme: UserPreferences.Theme) {
        dataStore.updateData { preferences ->
            preferences.toBuilder().setTheme(theme).build()
        }
    }

    suspend fun setFontSize(size: Int) {
        dataStore.updateData { preferences ->
            preferences.toBuilder().setFontSize(size).build()
        }
    }
}

从 SharedPreferences 迁移
DataStore 提供了内置的迁移工具，可以平滑地从 SharedPreferences 迁移。


val Context.settingsDataStore: DataStore<Preferences> by preferencesDataStore(
    name = "settings",
    produceMigrations = { context ->
        listOf(SharedPreferencesMigration(context, "old_settings"))
    }
)

val Context.userPreferencesStore: DataStore<UserPreferences> by dataStore(
    fileName = "user_preferences.pb",
    serializer = UserPreferencesSerializer,
    produceMigrations = { context ->
        listOf(
            SharedPreferencesMigration(context, "old_settings") { sharedPrefs, currentData ->
                currentData.toBuilder()
                    .setDarkMode(sharedPrefs.getBoolean("dark_mode", false))
                    .setLanguage(sharedPrefs.getString("language", "zh") ?: "zh")
                    .setFontSize(sharedPrefs.getInt("font_size", 14))
                    .build()
            }
        )
    }
)


加密
对于敏感数据，可以结合 EncryptedFile 或 Tink 加密库保护 DataStore 文件。


class EncryptedUserPreferencesSerializer(
    private val tink: Aead
) : Serializer<UserPreferences> {

    override val defaultValue: UserPreferences = UserPreferences.getDefaultInstance()

    override suspend fun readFrom(input: InputStream): UserPreferences {
        val encryptedBytes = input.readBytes()
        val decryptedBytes = tink.decrypt(encryptedBytes, null)
        return UserPreferences.parseFrom(decryptedBytes)
    }

    override suspend fun writeTo(t: UserPreferences, output: OutputStream) {
        val bytes = t.toByteArray()
        val encryptedBytes = tink.encrypt(bytes, null)
        output.write(encryptedBytes)
    }
}

@Module
@InstallIn(SingletonComponent::class)
object DataStoreModule {

    @Provides
    @Singleton
    fun provideEncryptedDataStore(
        @ApplicationContext context: Context
    ): DataStore<UserPreferences> {
        val aead = AndroidKeysetManager.Builder()
            .withSharedPref(context, "master_keyset", "master_key_prefs")
            .withKeyTemplate(KeyTemplates.get("AES256_GCM"))
            .withMasterKeyUri("android-keystore://master_key")
            .build()
            .keysetHandle
            .getPrimitive(Aead::class.java)

        return DataStoreFactory.create(
            serializer = EncryptedUserPreferencesSerializer(aead),
            produceFile = { context.dataStoreFile("encrypted_prefs.pb") }
        )
    }
}


提示： DataStore 文件的读写始终在 Dispatchers.IO 上执行，不会阻塞主线程。与 SharedPreferences 不同，DataStore 不提供同步读取 API。






小结
    

  • Preferences DataStore 以键值对存储简单配置，Proto DataStore 以 protobuf 存储结构化数据
    • 

  • DataStore 基于协程和 Flow，提供异步、事务性和一致性保证
    • 

  • 内置迁移工具支持从 SharedPreferences 平滑迁移
    • 

  • 敏感数据可通过 Tink 加密保护
    • 

  • DataStore 始终异步操作，不会阻塞主线程
