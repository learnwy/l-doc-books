WorkManager 后台任务管理

WorkManager 是 Android Jetpack 中用于处理可延迟、保证执行的后台任务的推荐方案。即使应用退出或设备重启，WorkManager 也能确保任务最终执行，非常适合日志上传、数据同步、定期清理等场景。

OneTimeWork 与 PeriodicWork

WorkManager 支持两种任务类型：一次性任务和周期性任务。

class SyncWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        val userId = inputData.getString("user_id") ?: return Result.failure()

        return try {
            val repository = EntryPointAccessors
                .fromApplication<SyncEntryPoint>(applicationContext)
                .syncRepository()

            repository.syncUserData(userId)
            Result.success(workDataOf("synced_count" to 42))
        } catch (e: Exception) {
            if (runAttemptCount < 3) Result.retry()
            else Result.failure()
        }
    }
}

fun scheduleSyncWork(context: Context, userId: String) {
    val syncRequest = OneTimeWorkRequestBuilder<SyncWorker>()
        .setInputData(workDataOf("user_id" to userId))
        .setBackoffCriteria(
            BackoffPolicy.EXPONENTIAL,
            Duration.ofSeconds(30)
        )
        .addTag("sync")
        .build()

    WorkManager.getInstance(context).enqueueUniqueWork(
        "user_sync_$userId",
        ExistingWorkPolicy.REPLACE,
        syncRequest
    )
}

fun schedulePeriodicSync(context: Context) {
    val periodicSync = PeriodicWorkRequestBuilder<SyncWorker>(
        repeatInterval = 6,
        repeatIntervalTimeUnit = TimeUnit.HOURS,
        flexTimeInterval = 30,
        flexTimeIntervalTimeUnit = TimeUnit.MINUTES
    )
        .setInputData(workDataOf("user_id" to "current"))
        .addTag("periodic_sync")
        .build()

    WorkManager.getInstance(context).enqueueUniquePeriodicWork(
        "periodic_sync",
        ExistingPeriodicWorkPolicy.KEEP,
        periodicSync
    )
}
任务类型 类 特点 适用场景
一次性任务 OneTimeWorkRequest 执行一次 文件上传、数据迁移
周期性任务 PeriodicWorkRequest 最小间隔 15 分钟 日志上传、缓存清理 


约束条件
WorkManager 可以设置执行约束，确保任务在合适的条件下运行。


val constraints = Constraints.Builder()
    .setRequiredNetworkType(NetworkType.CONNECTED)
    .setRequiresBatteryNotLow(true)
    .setRequiresStorageNotLow(true)
    .setRequiresCharging(false)
    .build()

val uploadRequest = OneTimeWorkRequestBuilder<UploadWorker>()
    .setConstraints(constraints)
    .setInitialDelay(5, TimeUnit.MINUTES)
    .build()

任务链
WorkManager 支持将多个任务串联或并行执行，形成复杂的工作流。


fun scheduleImageProcessingChain(context: Context, imageUris: List<String>) {
    val downloadWork = imageUris.map { uri ->
        OneTimeWorkRequestBuilder<DownloadWorker>()
            .setInputData(workDataOf("image_uri" to uri))
            .addTag("download")
            .build()
    }

    val compressWork = OneTimeWorkRequestBuilder<CompressWorker>()
        .addTag("compress")
        .build()

    val uploadWork = OneTimeWorkRequestBuilder<UploadWorker>()
        .setConstraints(
            Constraints.Builder()
                .setRequiredNetworkType(NetworkType.UNMETERED)
                .build()
        )
        .addTag("upload")
        .build()

    val cleanupWork = OneTimeWorkRequestBuilder<CleanupWorker>()
        .addTag("cleanup")
        .build()

    WorkManager.getInstance(context)
        .beginWith(downloadWork)
        .then(compressWork)
        .then(uploadWork)
        .then(cleanupWork)
        .enqueue()
}


提示： 链中前一个任务的输出可以作为后一个任务的输入。并行任务的输出会通过 ArrayCreatingInputMerger 或 OverwritingInputMerger 合并。






加急工作
从 Android 12 开始，WorkManager 支持加急工作（Expedited Work），用于需要尽快完成的重要任务。


class CriticalSyncWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        return try {
            syncCriticalData()
            Result.success()
        } catch (e: Exception) {
            Result.failure()
        }
    }

    override suspend fun getForegroundInfo(): ForegroundInfo {
        val notification = NotificationCompat.Builder(applicationContext, "sync_channel")
            .setContentTitle("正在同步数据")
            .setSmallIcon(R.drawable.ic_sync)
            .setOngoing(true)
            .build()

        return ForegroundInfo(NOTIFICATION_ID, notification)
    }

    companion object {
        private const val NOTIFICATION_ID = 1001
    }
}

fun scheduleExpeditedSync(context: Context) {
    val request = OneTimeWorkRequestBuilder<CriticalSyncWorker>()
        .setExpedited(OutOfQuotaPolicy.RUN_AS_NON_EXPEDITED_WORK_REQUEST)
        .build()

    WorkManager.getInstance(context).enqueue(request)
}

测试
WorkManager 提供了测试工具，支持同步执行和结果验证。


@HiltAndroidTest
class SyncWorkerTest {

    @get:Rule
    val hiltRule = HiltAndroidRule(this)

    private lateinit var context: Context

    @Before
    fun setup() {
        context = ApplicationProvider.getApplicationContext()
        val config = Configuration.Builder()
            .setMinimumLoggingLevel(Log.DEBUG)
            .setExecutor(SynchronousExecutor())
            .build()
        WorkManagerTestInitHelper.initializeTestWorkManager(context, config)
    }

    @Test
    fun syncWorker_succeeds_withValidInput() {
        val inputData = workDataOf("user_id" to "test_user")
        val request = OneTimeWorkRequestBuilder<SyncWorker>()
            .setInputData(inputData)
            .build()

        val workManager = WorkManager.getInstance(context)
        workManager.enqueue(request).result.get()

        val workInfo = workManager.getWorkInfoById(request.id).get()
        assertEquals(WorkInfo.State.SUCCEEDED, workInfo.state)
    }

    @Test
    fun syncWorker_retries_onNetworkError() = runTest {
        val worker = TestListenableWorkerBuilder<SyncWorker>(context)
            .setInputData(workDataOf("user_id" to "invalid"))
            .setRunAttemptCount(0)
            .build()

        val result = worker.doWork()
        assertEquals(ListenableWorker.Result.retry(), result)
    }
}


提示： 使用 TestListenableWorkerBuilder 可以直接创建 Worker 实例进行单元测试，无需初始化完整的 WorkManager。






小结
    

  • WorkManager 处理可延迟、保证执行的后台任务，支持一次性和周期性任务
    • 

  • 约束条件确保任务在网络、电量、存储等条件满足时执行
    • 

  • 任务链支持串联和并行执行，构建复杂的后台工作流
    • 

  • 加急工作用于高优先级任务，需要实现 getForegroundInfo 提供通知
    • 

  • 使用 WorkManagerTestInitHelper 和 TestListenableWorkerBuilder 进行测试
