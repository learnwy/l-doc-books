Room 数据库

Room 是 Android Jetpack 提供的 SQLite 抽象层，通过注解处理器在编译时验证 SQL 语句，提供类型安全的数据库访问。它支持 Kotlin 协程和 Flow，与现代 Android 架构完美集成。

实体（Entities）

实体类映射数据库表，使用注解定义表结构。

@Entity(tableName = "articles")
data class ArticleEntity(
    @PrimaryKey
    val id: String,
    @ColumnInfo(name = "title")
    val title: String,
    @ColumnInfo(name = "content")
    val content: String,
    @ColumnInfo(name = "author_id")
    val authorId: String,
    @ColumnInfo(name = "created_at")
    val createdAt: Long,
    @ColumnInfo(name = "is_bookmarked")
    val isBookmarked: Boolean = false,
    @ColumnInfo(name = "tags")
    val tags: List<String> = emptyList()
)

@Entity(
    tableName = "comments",
    foreignKeys = [
        ForeignKey(
            entity = ArticleEntity::class,
            parentColumns = ["id"],
            childColumns = ["article_id"],
            onDelete = ForeignKey.CASCADE
        )
    ],
    indices = [Index("article_id")]
)
data class CommentEntity(
    @PrimaryKey(autoGenerate = true)
    val id: Long = 0,
    @ColumnInfo(name = "article_id")
    val articleId: String,
    val text: String,
    val author: String,
    val timestamp: Long
)


DAO（数据访问对象）
DAO 定义数据库操作接口，Room 在编译时生成实现。


@Dao
interface ArticleDao {

    @Query("SELECT * FROM articles ORDER BY created_at DESC")
    fun getAllArticles(): Flow<List<ArticleEntity>>

    @Query("SELECT * FROM articles WHERE id = :id")
    suspend fun getArticleById(id: String): ArticleEntity?

    @Query("SELECT * FROM articles WHERE is_bookmarked = 1 ORDER BY created_at DESC")
    fun getBookmarkedArticles(): Flow<List<ArticleEntity>>

    @Query("SELECT * FROM articles WHERE title LIKE '%' || :query || '%' OR content LIKE '%' || :query || '%'")
    fun searchArticles(query: String): Flow<List<ArticleEntity>>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertArticles(articles: List<ArticleEntity>)

    @Update
    suspend fun updateArticle(article: ArticleEntity)

    @Query("UPDATE articles SET is_bookmarked = :bookmarked WHERE id = :id")
    suspend fun setBookmarked(id: String, bookmarked: Boolean)

    @Delete
    suspend fun deleteArticle(article: ArticleEntity)

    @Query("DELETE FROM articles")
    suspend fun deleteAllArticles()

    @Transaction
    @Query("SELECT * FROM articles WHERE id = :id")
    fun getArticleWithComments(id: String): Flow<ArticleWithComments>
}

data class ArticleWithComments(
    @Embedded val article: ArticleEntity,
    @Relation(
        parentColumn = "id",
        entityColumn = "article_id"
    )
    val comments: List<CommentEntity>
)

类型转换器
Room 默认只支持基本类型，复杂类型需要通过 TypeConverter 转换。


class Converters {

    @TypeConverter
    fun fromStringList(value: List<String>): String {
        return Json.encodeToString(value)
    }

    @TypeConverter
    fun toStringList(value: String): List<String> {
        return Json.decodeFromString(value)
    }

    @TypeConverter
    fun fromInstant(value: Instant?): Long? {
        return value?.toEpochMilliseconds()
    }

    @TypeConverter
    fun toInstant(value: Long?): Instant? {
        return value?.let { Instant.fromEpochMilliseconds(it) }
    }
}

@Database(
    entities = [ArticleEntity::class, CommentEntity::class],
    version = 2,
    exportSchema = true
)
@TypeConverters(Converters::class)
abstract class AppDatabase : RoomDatabase() {
    abstract fun articleDao(): ArticleDao
    abstract fun commentDao(): CommentDao
}


提示： 设置 exportSchema = true 可以导出数据库的 JSON 格式 schema，用于自动化迁移验证。






数据库迁移
当数据库结构发生变化时，必须提供迁移策略以保护用户数据。


val MIGRATION_1_2 = object : Migration(1, 2) {
    override fun migrate(db: SupportSQLiteDatabase) {
        db.execSQL("ALTER TABLE articles ADD COLUMN tags TEXT NOT NULL DEFAULT '[]'")
    }
}

val MIGRATION_2_3 = object : Migration(2, 3) {
    override fun migrate(db: SupportSQLiteDatabase) {
        db.execSQL("""
            CREATE TABLE IF NOT EXISTS comments (
                id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
                article_id TEXT NOT NULL,
                text TEXT NOT NULL,
                author TEXT NOT NULL,
                timestamp INTEGER NOT NULL,
                FOREIGN KEY (article_id) REFERENCES articles(id) ON DELETE CASCADE
            )
        """)
        db.execSQL("CREATE INDEX IF NOT EXISTS index_comments_article_id ON comments(article_id)")
    }
}

@Module
@InstallIn(SingletonComponent::class)
object DatabaseModule {

    @Provides
    @Singleton
    fun provideDatabase(@ApplicationContext context: Context): AppDatabase {
        return Room.databaseBuilder(context, AppDatabase::class.java, "app_database")
            .addMigrations(MIGRATION_1_2, MIGRATION_2_3)
            .build()
    }

    @Provides
    fun provideArticleDao(database: AppDatabase): ArticleDao = database.articleDao()
}
Room 2.4+ 支持自动迁移：


@Database(
    entities = [ArticleEntity::class],
    version = 3,
    autoMigrations = [
        AutoMigration(from = 1, to = 2),
        AutoMigration(from = 2, to = 3, spec = Migration2To3::class)
    ]
)
abstract class AppDatabase : RoomDatabase()

@RenameColumn(tableName = "articles", fromColumnName = "desc", toColumnName = "content")
class Migration2To3 : AutoMigrationSpec


Flow / LiveData 支持与测试
Room 的 Flow 返回类型会在数据变化时自动发射新值。


class ArticleRepository(private val articleDao: ArticleDao) {

    fun getArticles(): Flow<List<Article>> =
        articleDao.getAllArticles().map { entities ->
            entities.map { it.toDomain() }
        }

    suspend fun toggleBookmark(id: String) {
        val article = articleDao.getArticleById(id) ?: return
        articleDao.setBookmarked(id, !article.isBookmarked)
    }
}

@RunWith(AndroidJUnit4::class)
class ArticleDaoTest {

    private lateinit var database: AppDatabase
    private lateinit var dao: ArticleDao

    @Before
    fun setup() {
        database = Room.inMemoryDatabaseBuilder(
            ApplicationProvider.getApplicationContext(),
            AppDatabase::class.java
        ).allowMainThreadQueries().build()
        dao = database.articleDao()
    }

    @After
    fun teardown() {
        database.close()
    }

    @Test
    fun insertAndRetrieveArticle() = runTest {
        val article = ArticleEntity(
            id = "1", title = "测试", content = "内容",
            authorId = "a1", createdAt = System.currentTimeMillis()
        )
        dao.insertArticles(listOf(article))

        val result = dao.getAllArticles().first()
        assertEquals(1, result.size)
        assertEquals("测试", result[0].title)
    }
}


小结
    

  • Room 通过注解处理器在编译时验证 SQL，提供类型安全的数据库访问
    • 

  • Entity 定义表结构，DAO 定义操作接口，Database 组装整体
    • 

  • TypeConverter 处理复杂类型的序列化与反序列化
    • 

  • 数据库迁移通过 Migration 类或 AutoMigration 注解实现
    • 

  • Flow 返回类型实现数据变化的自动通知，使用内存数据库进行 DAO 测试
