Part II · AndroidAndroid 开发 简介Jetpack Compose 简介导航

导航

Jetpack Compose 的 Navigation 组件提供声明式导航方案，支持类型安全的路由、Deep Link 和嵌套导航图。通过 Kotlin Serialization 实现的类型安全导航消除了传统字符串路由的类型不安全问题。

NavHost 与 NavGraph

NavHost 是导航的容器，NavGraph 定义了导航图中所有可达目的地和它们之间的关系。

@Composable
fun AppNavigation() {
    val navController = rememberNavController()

    NavHost(
        navController = navController,
        startDestination = "home"
    ) {
        composable("home") {
            HomeScreen(
                onNavigateToDetail = { id -> navController.navigate("detail/$id") },
                onNavigateToSettings = { navController.navigate("settings") }
            )
        }
        composable(
            route = "detail/{itemId}",
            arguments = listOf(navArgument("itemId") { type = NavType.StringType })
        ) { backStackEntry ->
            val itemId = backStackEntry.arguments?.getString("itemId") ?: return@composable
            DetailScreen(itemId = itemId)
        }
        composable("settings") {
            SettingsScreen(onBack = { navController.popBackStack() })
        }
    }
}

类型安全导航（Kotlin Serialization）
使用 Kotlin Serialization 定义路由可以实现编译时类型检查，消除字符串路由的拼写错误风险。


@Serializable
data object Home

@Serializable
data class ProductDetail(val productId: String)

@Serializable
data class CategoryList(val category: String, val sortBy: String = "name")

@Serializable
data object Settings

@Serializable
data object Profile

@Composable
fun TypeSafeNavigation() {
    val navController = rememberNavController()

    NavHost(navController = navController, startDestination = Home) {
        composable<Home> {
            HomeScreen(
                onProductClick = { id ->
                    navController.navigate(ProductDetail(productId = id))
                },
                onCategoryClick = { category ->
                    navController.navigate(CategoryList(category = category))
                }
            )
        }
        composable<ProductDetail> { backStackEntry ->
            val detail: ProductDetail = backStackEntry.toRoute()
            ProductDetailScreen(productId = detail.productId)
        }
        composable<CategoryList> { backStackEntry ->
            val list: CategoryList = backStackEntry.toRoute()
            CategoryListScreen(category = list.category, sortBy = list.sortBy)
        }
        composable<Settings> {
            SettingsScreen()
        }
    }
}

Deep Link
Deep Link 允许外部 URL 直接导航到应用内的特定页面。


@Serializable
data class Article(val articleId: String)

@Composable
fun DeepLinkNavigation() {
    val navController = rememberNavController()

    NavHost(navController = navController, startDestination = Home) {
        composable<Home> { HomeScreen() }
        composable<Article>(
            deepLinks = listOf(
                navDeepLink<Article>(basePath = "https://myapp.com/article")
            )
        ) { backStackEntry ->
            val article: Article = backStackEntry.toRoute()
            ArticleScreen(articleId = article.articleId)
        }
    }
}
在 AndroidManifest.xml 中配合声明：


<activity android:name=".MainActivity" android:exported="true">
    <intent-filter android:autoVerify="true">
        <action android:name="android.intent.action.VIEW" />
        <category android:name="android.intent.category.DEFAULT" />
        <category android:name="android.intent.category.BROWSABLE" />
        <data android:scheme="https" android:host="myapp.com" />
    </intent-filter>
</activity>


嵌套导航
嵌套导航图将相关页面组织在一起，适用于带底部导航栏或功能模块化的应用。


@Serializable data object HomeGraph
@Serializable data object HomeFeed
@Serializable data class HomeDetail(val id: String)

@Serializable data object SearchGraph
@Serializable data object SearchMain
@Serializable data class SearchResult(val query: String)

@Serializable data object ProfileGraph
@Serializable data object ProfileMain
@Serializable data object ProfileEdit

@Composable
fun MainScreen() {
    val navController = rememberNavController()

    Scaffold(
        bottomBar = {
            NavigationBar {
                val currentRoute = navController.currentBackStackEntryAsState()
                    .value?.destination?.route

                NavigationBarItem(
                    selected = currentRoute?.contains("Home") == true,
                    onClick = {
                        navController.navigate(HomeGraph) {
                            popUpTo(navController.graph.findStartDestination().id) {
                                saveState = true
                            }
                            launchSingleTop = true
                            restoreState = true
                        }
                    },
                    icon = { Icon(Icons.Default.Home, contentDescription = null) },
                    label = { Text("首页") }
                )
                NavigationBarItem(
                    selected = currentRoute?.contains("Search") == true,
                    onClick = {
                        navController.navigate(SearchGraph) {
                            popUpTo(navController.graph.findStartDestination().id) {
                                saveState = true
                            }
                            launchSingleTop = true
                            restoreState = true
                        }
                    },
                    icon = { Icon(Icons.Default.Search, contentDescription = null) },
                    label = { Text("搜索") }
                )
            }
        }
    ) { padding ->
        NavHost(
            navController = navController,
            startDestination = HomeGraph,
            modifier = Modifier.padding(padding)
        ) {
            navigation<HomeGraph>(startDestination = HomeFeed) {
                composable<HomeFeed> { HomeFeedScreen() }
                composable<HomeDetail> { DetailScreen() }
            }
            navigation<SearchGraph>(startDestination = SearchMain) {
                composable<SearchMain> { SearchScreen() }
                composable<SearchResult> { ResultScreen() }
            }
            navigation<ProfileGraph>(startDestination = ProfileMain) {
                composable<ProfileMain> { ProfileScreen() }
                composable<ProfileEdit> { EditProfileScreen() }
            }
        }
    }
}


提示： 底部导航切换时使用 saveState = true 和 restoreState = true，可以保持各 Tab 的页面状态和回退栈。






小结
    

  • NavHost 是导航容器，NavGraph 声明所有可达目的地
    • 

  • 类型安全导航使用 Kotlin Serialization 定义路由，提供编译时类型检查
    • 

  • Deep Link 通过 navDeepLink 和清单声明实现外部 URL 到应用页面的直接导航
    • 

  • 嵌套导航图通过 navigation<T> 组织相关页面，适合底部导航栏场景
    • 

  • 底部导航切换时配合 saveState / restoreState 保持各 Tab 独立状态
