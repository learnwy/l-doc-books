Part II · AndroidAndroid 开发 简介Jetpack Compose 简介布局与修饰符

布局与修饰符

Compose 提供了一套灵活的布局系统和链式修饰符机制来构建 UI。掌握布局容器和 Modifier 的使用是高效构建 Compose 界面的基础。

Column / Row / Box

这三个基础布局容器是 Compose 中最常用的布局组件，分别对应垂直排列、水平排列和层叠排列。

@Composable
fun ProfileCard(user: User) {
    Card(modifier = Modifier.fillMaxWidth().padding(8.dp)) {
        Row(
            modifier = Modifier.padding(16.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            Box(contentAlignment = Alignment.BottomEnd) {
                AsyncImage(
                    model = user.avatarUrl,
                    contentDescription = null,
                    modifier = Modifier.size(64.dp).clip(CircleShape)
                )
                if (user.isOnline) {
                    Box(
                        modifier = Modifier
                            .size(16.dp)
                            .clip(CircleShape)
                            .background(Color.Green)
                            .border(2.dp, MaterialTheme.colorScheme.surface, CircleShape)
                    )
                }
            }
            Spacer(modifier = Modifier.width(12.dp))
            Column(modifier = Modifier.weight(1f)) {
                Text(user.name, style = MaterialTheme.typography.titleMedium)
                Text(user.bio, style = MaterialTheme.typography.bodySmall, maxLines = 2)
            }
        }
    }
}

Modifier 链
Modifier 是 Compose 中装饰和配置 Composable 的核心机制。Modifier 是链式调用的，顺序非常重要。


@Composable
fun ModifierOrderDemo() {
    Column(modifier = Modifier.padding(16.dp)) {
        Text(
            text = "先 padding 后 background",
            modifier = Modifier
                .padding(16.dp)
                .background(MaterialTheme.colorScheme.primaryContainer)
                .padding(16.dp)
        )
        Spacer(modifier = Modifier.height(8.dp))
        Text(
            text = "先 background 后 padding",
            modifier = Modifier
                .background(MaterialTheme.colorScheme.primaryContainer)
                .padding(16.dp)
        )
    }
}

提示： Modifier 的顺序影响最终效果。padding 在 background 之前会创建外边距效果，在之后则是内边距效果。






自定义 Modifier
通过扩展函数和 Modifier.composed 可以创建可复用的自定义 Modifier。


fun Modifier.shimmerEffect(): Modifier = composed {
    var size by remember { mutableStateOf(IntSize.Zero) }
    val transition = rememberInfiniteTransition(label = "shimmer")
    val startOffsetX by transition.animateFloat(
        initialValue = -2 * size.width.toFloat(),
        targetValue = 2 * size.width.toFloat(),
        animationSpec = infiniteRepeatable(
            animation = tween(1000)
        ),
        label = "shimmer"
    )

    background(
        brush = Brush.linearGradient(
            colors = listOf(
                Color(0xFFB8B5B5),
                Color(0xFF8F8B8B),
                Color(0xFFB8B5B5)
            ),
            start = Offset(startOffsetX, 0f),
            end = Offset(startOffsetX + size.width.toFloat(), size.height.toFloat())
        )
    ).onGloballyPositioned { size = it.size }
}

fun Modifier.conditional(
    condition: Boolean,
    modifier: Modifier.() -> Modifier
): Modifier = if (condition) then(modifier(Modifier)) else this

@Composable
fun ShimmerPlaceholder() {
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(100.dp)
            .clip(RoundedCornerShape(8.dp))
            .shimmerEffect()
    )
}


ConstraintLayout
对于复杂的相对定位布局，可以使用 ConstraintLayout。


@Composable
fun OverlayCard(title: String, subtitle: String, imageUrl: String) {
    ConstraintLayout(
        modifier = Modifier
            .fillMaxWidth()
            .height(200.dp)
    ) {
        val (image, overlay, titleRef, subtitleRef, badge) = createRefs()

        AsyncImage(
            model = imageUrl,
            contentDescription = null,
            contentScale = ContentScale.Crop,
            modifier = Modifier.constrainAs(image) {
                width = Dimension.fillToConstraints
                height = Dimension.fillToConstraints
                top.linkTo(parent.top)
                bottom.linkTo(parent.bottom)
                start.linkTo(parent.start)
                end.linkTo(parent.end)
            }
        )

        Box(
            modifier = Modifier
                .constrainAs(overlay) {
                    width = Dimension.fillToConstraints
                    height = Dimension.value(80.dp)
                    bottom.linkTo(parent.bottom)
                    start.linkTo(parent.start)
                    end.linkTo(parent.end)
                }
                .background(
                    Brush.verticalGradient(
                        colors = listOf(Color.Transparent, Color.Black.copy(alpha = 0.7f))
                    )
                )
        )

        Text(
            text = title,
            color = Color.White,
            style = MaterialTheme.typography.titleMedium,
            modifier = Modifier.constrainAs(titleRef) {
                bottom.linkTo(subtitleRef.top, 4.dp)
                start.linkTo(parent.start, 16.dp)
            }
        )

        Text(
            text = subtitle,
            color = Color.White.copy(alpha = 0.8f),
            style = MaterialTheme.typography.bodySmall,
            modifier = Modifier.constrainAs(subtitleRef) {
                bottom.linkTo(parent.bottom, 16.dp)
                start.linkTo(parent.start, 16.dp)
            }
        )
    }
}


固有特性测量（Intrinsic Measurements）
IntrinsicSize 允许子元素在测量之前查询兄弟元素的尺寸信息，解决"先有鸡还是先有蛋"的布局问题。


@Composable
fun TwoColumnLayout() {
    Row(modifier = Modifier.height(IntrinsicSize.Min)) {
        Text(
            text = "左侧短文本",
            modifier = Modifier
                .weight(1f)
                .padding(8.dp)
        )
        Divider(
            modifier = Modifier
                .fillMaxHeight()
                .width(1.dp)
        )
        Text(
            text = "右侧较长的文本内容，可能会换行显示，左侧列和分割线会自动适应这个高度。",
            modifier = Modifier
                .weight(1f)
                .padding(8.dp)
        )
    }
}

小结
    

  • Column、Row、Box 是三大基础布局容器，分别用于垂直、水平和层叠排列
    • 

  • Modifier 链的顺序决定最终效果，padding 和 background 的先后顺序会产生不同的视觉结果
    • 

  • 自定义 Modifier 通过扩展函数实现，composed 可以在 Modifier 中使用 Composable 状态
    • 

  • ConstraintLayout 适用于复杂的相对定位场景
    • 

  • IntrinsicSize 解决子元素间尺寸相互依赖的布局问题
    •
