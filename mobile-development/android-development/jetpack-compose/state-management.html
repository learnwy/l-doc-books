Part II · AndroidAndroid 开发 简介Jetpack Compose 简介状态管理

状态管理

在 Jetpack Compose 中，UI 是状态的函数。状态管理决定了数据如何存储、流转和响应变化。正确的状态管理策略是构建可维护 Compose 应用的关键。

mutableStateOf

mutableStateOf 创建可观察的状态对象，当值变化时自动触发读取该值的 Composable 重组。

@Composable
fun LoginForm() {
    var username by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }
    var passwordVisible by remember { mutableStateOf(false) }

    Column(modifier = Modifier.padding(16.dp)) {
        OutlinedTextField(
            value = username,
            onValueChange = { username = it },
            label = { Text("用户名") },
            modifier = Modifier.fillMaxWidth()
        )
        Spacer(modifier = Modifier.height(8.dp))
        OutlinedTextField(
            value = password,
            onValueChange = { password = it },
            label = { Text("密码") },
            visualTransformation = if (passwordVisible) VisualTransformation.None
                else PasswordVisualTransformation(),
            trailingIcon = {
                IconButton(onClick = { passwordVisible = !passwordVisible }) {
                    Icon(
                        imageVector = if (passwordVisible) Icons.Default.Visibility
                            else Icons.Default.VisibilityOff,
                        contentDescription = null
                    )
                }
            },
            modifier = Modifier.fillMaxWidth()
        )
    }
}

State 的三种声明方式：

方式 语法 特点
解构声明 val (value, setValue) = remember { mutableStateOf("") } 明确区分读写
by 委托 var value by remember { mutableStateOf("") } 简洁，直接读写
直接引用 val state = remember { mutableStateOf("") } 需要 .value 访问

提示： 对于基本类型（Int、Long、Float、Double），使用专门的 mutableIntStateOf、mutableLongStateOf 等函数，避免自动装箱带来的性能开销。

derivedStateOf

derivedStateOf 用于从其他状态派生计算结果，仅当依赖的状态变化且派生结果也变化时才触发重组。

@Composable
fun FilteredProductList(products: List<Product>) {
    var searchQuery by remember { mutableStateOf("") }
    var selectedCategory by remember { mutableStateOf<Category?>(null) }

    val filteredProducts by remember(products) {
        derivedStateOf {
            products.filter { product ->
                (searchQuery.isBlank() || product.name.contains(searchQuery, ignoreCase = true)) &&
                (selectedCategory == null || product.category == selectedCategory)
            }
        }
    }

    val resultSummary by remember {
        derivedStateOf {
            "${filteredProducts.size} 个结果"
        }
    }

    Column {
        SearchBar(query = searchQuery, onQueryChange = { searchQuery = it })
        Text(text = resultSummary, modifier = Modifier.padding(8.dp))
        LazyColumn {
            items(filteredProducts, key = { it.id }) { product ->
                ProductItem(product = product)
            }
        }
    }
}
derivedStateOf 适用于输入变化频率高于输出变化频率的场景，如从快速变化的滚动位置派生"是否显示回到顶部按钮"。




rememberSaveable
rememberSaveable 在 remember 的基础上增加了状态持久化能力，支持配置变更（如屏幕旋转）和进程死亡后的状态恢复。


@Composable
fun ArticleEditor() {
    var title by rememberSaveable { mutableStateOf("") }
    var content by rememberSaveable { mutableStateOf("") }
    var selectedTab by rememberSaveable { mutableIntStateOf(0) }

    Column {
        TabRow(selectedTabIndex = selectedTab) {
            Tab(selected = selectedTab == 0, onClick = { selectedTab = 0 }) {
                Text("编辑", modifier = Modifier.padding(16.dp))
            }
            Tab(selected = selectedTab == 1, onClick = { selectedTab = 1 }) {
                Text("预览", modifier = Modifier.padding(16.dp))
            }
        }
        when (selectedTab) {
            0 -> EditorPane(title, content, { title = it }, { content = it })
            1 -> PreviewPane(title, content)
        }
    }
}
对于复杂对象，需要提供自定义 Saver：


data class SearchFilter(
    val query: String,
    val category: String,
    val minPrice: Double,
    val maxPrice: Double
)

val SearchFilterSaver = run {
    val queryKey = "query"
    val categoryKey = "category"
    val minPriceKey = "minPrice"
    val maxPriceKey = "maxPrice"

    mapSaver(
        save = { filter ->
            mapOf(
                queryKey to filter.query,
                categoryKey to filter.category,
                minPriceKey to filter.minPrice,
                maxPriceKey to filter.maxPrice
            )
        },
        restore = { map ->
            SearchFilter(
                query = map[queryKey] as String,
                category = map[categoryKey] as String,
                minPrice = map[minPriceKey] as Double,
                maxPrice = map[maxPriceKey] as Double
            )
        }
    )
}

@Composable
fun FilterScreen() {
    var filter by rememberSaveable(stateSaver = SearchFilterSaver) {
        mutableStateOf(SearchFilter("", "全部", 0.0, 1000.0))
    }
}


状态持有者（State Holder）
当组件状态逻辑变得复杂时，应将状态提取到专门的状态持有者类中。


class DrawerState(
    initialOpen: Boolean = false
) {
    var isOpen by mutableStateOf(initialOpen)
        private set

    var selectedItem by mutableStateOf<MenuItem?>(null)
        private set

    fun open() { isOpen = true }
    fun close() { isOpen = false }
    fun toggle() { isOpen = !isOpen }
    fun selectItem(item: MenuItem) { selectedItem = item }
}

@Composable
fun rememberDrawerState(initialOpen: Boolean = false): DrawerState {
    return remember { DrawerState(initialOpen) }
}

@Composable
fun AppScaffold() {
    val drawerState = rememberDrawerState()

    ModalNavigationDrawer(
        drawerState = rememberDrawerState(),
        drawerContent = {
            DrawerContent(
                selectedItem = drawerState.selectedItem,
                onItemSelected = { drawerState.selectItem(it) }
            )
        }
    ) {
        MainContent(onMenuClick = { drawerState.toggle() })
    }
}


ViewModel 集成
ViewModel 持有屏幕级别的业务状态，通过 hiltViewModel() 或 viewModel() 在 Composable 中获取。


@HiltViewModel
class ProductListViewModel @Inject constructor(
    private val repository: ProductRepository
) : ViewModel() {

    private val _uiState = MutableStateFlow(ProductListUiState())
    val uiState = _uiState.asStateFlow()

    init {
        loadProducts()
    }

    fun loadProducts() {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            repository.getProducts()
                .onSuccess { products ->
                    _uiState.update { it.copy(products = products, isLoading = false) }
                }
                .onFailure { error ->
                    _uiState.update { it.copy(error = error.message, isLoading = false) }
                }
        }
    }

    fun refresh() = loadProducts()
}

data class ProductListUiState(
    val products: List<Product> = emptyList(),
    val isLoading: Boolean = false,
    val error: String? = null
)

@Composable
fun ProductListScreen(viewModel: ProductListViewModel = hiltViewModel()) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    when {
        uiState.isLoading -> LoadingIndicator()
        uiState.error != null -> ErrorMessage(
            message = uiState.error!!,
            onRetry = viewModel::refresh
        )
        else -> ProductList(products = uiState.products)
    }
}

提示： 使用 collectAsStateWithLifecycle() 替代 collectAsState()，它会在 Lifecycle 处于非活跃状态时自动停止收集，避免不必要的资源消耗。






小结
    

  • mutableStateOf 创建可观察状态，值变化自动触发重组
    • 

  • derivedStateOf 从其他状态派生值，避免不必要的重组
    • 

  • rememberSaveable 支持跨配置变更和进程死亡的状态恢复
    • 

  • 状态持有者类封装复杂的 UI 状态逻辑，保持 Composable 简洁
    • 

  • ViewModel 管理屏幕级业务状态，通过 collectAsStateWithLifecycle() 与 Compose 集成
