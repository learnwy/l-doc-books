Part II · AndroidAndroid 开发 简介Jetpack Compose 简介动画

动画

Jetpack Compose 提供了丰富的动画 API，从简单的值动画到复杂的手势驱动动画。Compose 的动画系统声明式且可组合，让开发者以最小的代码实现流畅的过渡效果。

animateAsState

animate*AsState 是最简单的动画 API，当目标值变化时自动执行动画过渡。

@Composable
fun AnimatedFavoriteButton(isFavorite: Boolean, onClick: () -> Unit) {
    val scale by animateFloatAsState(
        targetValue = if (isFavorite) 1.2f else 1f,
        animationSpec = spring(
            dampingRatio = Spring.DampingRatioMediumBouncy,
            stiffness = Spring.StiffnessLow
        ),
        label = "scale"
    )

    val tint by animateColorAsState(
        targetValue = if (isFavorite) Color.Red else Color.Gray,
        animationSpec = tween(durationMillis = 300),
        label = "tint"
    )

    IconButton(onClick = onClick) {
        Icon(
            imageVector = if (isFavorite) Icons.Filled.Favorite else Icons.Outlined.FavoriteBorder,
            contentDescription = "收藏",
            tint = tint,
            modifier = Modifier.scale(scale)
        )
    }
}

常用动画规格对比：

AnimationSpec 说明 适用场景
spring() 弹簧物理动画 自然的反弹效果
tween() 时长+缓动曲线 精确控制时长
snap() 瞬间完成 无需动画的状态切换
keyframes 关键帧动画 分段控制动画值
repeatable() 有限次重复 按钮抖动提示
infiniteRepeatable() 无限重复 加载动画

提示： 为每个 animate*AsState 调用提供有意义的 label 参数，方便在 Android Studio 的 Animation Preview 中调试。

AnimatedVisibility

AnimatedVisibility 控制 Composable 的显示和隐藏动画。

@Composable
fun NotificationBanner(
    message: String,
    isVisible: Boolean,
    onDismiss: () -> Unit
) {
    AnimatedVisibility(
        visible = isVisible,
        enter = slideInVertically(initialOffsetY = { -it }) + fadeIn(),
        exit = slideOutVertically(targetOffsetY = { -it }) + fadeOut()
    ) {
        Card(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp),
            colors = CardDefaults.cardColors(
                containerColor = MaterialTheme.colorScheme.primaryContainer
            )
        ) {
            Row(
                modifier = Modifier.padding(16.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                Icon(Icons.Default.Info, contentDescription = null)
                Spacer(modifier = Modifier.width(8.dp))
                Text(text = message, modifier = Modifier.weight(1f))
                IconButton(onClick = onDismiss) {
                    Icon(Icons.Default.Close, contentDescription = "关闭")
                }
            }
        }
    }
}

Transition
updateTransition 管理多个动画值的状态转换，确保它们协调同步。


enum class CardState { Collapsed, Expanded }

@Composable
fun ExpandableInfoCard(title: String, content: String) {
    var cardState by remember { mutableStateOf(CardState.Collapsed) }
    val transition = updateTransition(targetState = cardState, label = "card")

    val cardElevation by transition.animateDp(
        transitionSpec = { tween(300) },
        label = "elevation"
    ) { state ->
        when (state) {
            CardState.Collapsed -> 2.dp
            CardState.Expanded -> 8.dp
        }
    }

    val cardRoundedCorner by transition.animateDp(
        transitionSpec = { tween(300) },
        label = "corner"
    ) { state ->
        when (state) {
            CardState.Collapsed -> 12.dp
            CardState.Expanded -> 4.dp
        }
    }

    val arrowRotation by transition.animateFloat(
        transitionSpec = { tween(300) },
        label = "arrow"
    ) { state ->
        when (state) {
            CardState.Collapsed -> 0f
            CardState.Expanded -> 180f
        }
    }

    Card(
        modifier = Modifier
            .fillMaxWidth()
            .padding(8.dp)
            .clickable {
                cardState = if (cardState == CardState.Collapsed)
                    CardState.Expanded else CardState.Collapsed
            },
        elevation = CardDefaults.cardElevation(defaultElevation = cardElevation),
        shape = RoundedCornerShape(cardRoundedCorner)
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(text = title, modifier = Modifier.weight(1f))
                Icon(
                    Icons.Default.KeyboardArrowDown,
                    contentDescription = null,
                    modifier = Modifier.rotate(arrowRotation)
                )
            }
            transition.AnimatedVisibility(visible = { it == CardState.Expanded }) {
                Text(
                    text = content,
                    modifier = Modifier.padding(top = 8.dp),
                    style = MaterialTheme.typography.bodyMedium
                )
            }
        }
    }
}


InfiniteTransition
rememberInfiniteTransition 创建无限循环的动画，常用于加载指示器和脉冲效果。


@Composable
fun PulsingDot(
    color: Color = MaterialTheme.colorScheme.primary,
    size: Dp = 12.dp
) {
    val infiniteTransition = rememberInfiniteTransition(label = "pulse")

    val scale by infiniteTransition.animateFloat(
        initialValue = 0.8f,
        targetValue = 1.2f,
        animationSpec = infiniteRepeatable(
            animation = tween(600, easing = FastOutSlowInEasing),
            repeatMode = RepeatMode.Reverse
        ),
        label = "scale"
    )

    val alpha by infiniteTransition.animateFloat(
        initialValue = 0.4f,
        targetValue = 1f,
        animationSpec = infiniteRepeatable(
            animation = tween(600),
            repeatMode = RepeatMode.Reverse
        ),
        label = "alpha"
    )

    Box(
        modifier = Modifier
            .size(size)
            .scale(scale)
            .alpha(alpha)
            .clip(CircleShape)
            .background(color)
    )
}

@Composable
fun LoadingIndicator() {
    Row(
        horizontalArrangement = Arrangement.spacedBy(8.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        repeat(3) { index ->
            val infiniteTransition = rememberInfiniteTransition(label = "dot$index")
            val offsetY by infiniteTransition.animateFloat(
                initialValue = 0f,
                targetValue = -10f,
                animationSpec = infiniteRepeatable(
                    animation = tween(300, delayMillis = index * 100),
                    repeatMode = RepeatMode.Reverse
                ),
                label = "offset"
            )
            Box(
                modifier = Modifier
                    .size(8.dp)
                    .offset(y = offsetY.dp)
                    .clip(CircleShape)
                    .background(MaterialTheme.colorScheme.primary)
            )
        }
    }
}


手势驱动动画
使用 Animatable 和手势检测实现交互式动画，如拖拽、滑动删除等。


@Composable
fun SwipeToDismissItem(
    onDismiss: () -> Unit,
    content: @Composable () -> Unit
) {
    val offsetX = remember { Animatable(0f) }
    val scope = rememberCoroutineScope()

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .pointerInput(Unit) {
                detectHorizontalDragGestures(
                    onDragEnd = {
                        scope.launch {
                            if (kotlin.math.abs(offsetX.value) > size.width / 3) {
                                offsetX.animateTo(
                                    targetValue = if (offsetX.value > 0) size.width.toFloat()
                                        else -size.width.toFloat(),
                                    animationSpec = tween(200)
                                )
                                onDismiss()
                            } else {
                                offsetX.animateTo(0f, spring())
                            }
                        }
                    },
                    onHorizontalDrag = { _, dragAmount ->
                        scope.launch {
                            offsetX.snapTo(offsetX.value + dragAmount)
                        }
                    }
                )
            }
            .offset { IntOffset(offsetX.value.roundToInt(), 0) }
    ) {
        content()
    }
}


提示： Animatable 支持边界限制（updateBounds），可以防止用户将元素拖出可见区域。






小结
    

  • animateAsState 是最简单的动画 API，目标值变化时自动过渡
    • 

  • AnimatedVisibility 控制元素显示/隐藏的过渡动画，支持多种 Enter/Exit 效果组合
    • 

  • updateTransition 协调多个动画值的同步状态转换
    • 

  • InfiniteTransition 用于无限循环动画，适合加载指示器等场景
    • 

  • Animatable 配合手势检测实现交互式动画，支持 snapTo 和 animateTo 两种方式
    •
