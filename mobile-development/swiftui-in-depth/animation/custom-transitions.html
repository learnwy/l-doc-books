Part I · iOSSwiftUI 深入实践 简介动画 简介自定义转场

自定义转场

概述

SwiftUI 的转场（Transition）定义了视图在插入和移除时的动画效果。内置转场包括 .opacity、.slide、.scale 等。通过 AnyTransition 的组合和非对称转场，可以创建丰富的过渡效果。iOS 17 引入了 Transition 协议，让自定义转场更加类型安全和强大。



内置转场
SwiftUI 提供多种内置转场效果：


struct BuiltInTransitions: View {
    @State private var showContent = false

    var body: some View {
        VStack(spacing: 20) {
            if showContent {
                Text("淡入淡出")
                    .padding()
                    .background(.blue.opacity(0.2))
                    .transition(.opacity)

                Text("从底部滑入")
                    .padding()
                    .background(.green.opacity(0.2))
                    .transition(.move(edge: .bottom))

                Text("缩放")
                    .padding()
                    .background(.orange.opacity(0.2))
                    .transition(.scale)

                Text("滑动")
                    .padding()
                    .background(.purple.opacity(0.2))
                    .transition(.slide)
            }

            Button(showContent ? "隐藏" : "显示") {
                withAnimation(.spring(response: 0.4)) {
                    showContent.toggle()
                }
            }
        }
    }
}

组合转场
通过 .combined(with:) 组合多个转场效果：


struct CombinedTransitionDemo: View {
    @State private var showCard = false

    var body: some View {
        ZStack {
            if showCard {
                VStack(spacing: 12) {
                    Image(systemName: "bell.fill")
                        .font(.system(size: 40))
                    Text("新通知")
                        .font(.headline)
                    Text("你有一条新消息")
                        .font(.subheadline)
                        .foregroundStyle(.secondary)
                }
                .padding(24)
                .background(.regularMaterial)
                .clipShape(RoundedRectangle(cornerRadius: 16))
                .shadow(radius: 10)
                .transition(
                    .scale(scale: 0.5)
                    .combined(with: .opacity)
                    .combined(with: .offset(y: -50))
                )
            }

            VStack {
                Spacer()
                Button(showCard ? "关闭" : "显示通知") {
                    withAnimation(.spring(response: 0.35, dampingFraction: 0.7)) {
                        showCard.toggle()
                    }
                }
                .buttonStyle(.borderedProminent)
                .padding(.bottom, 40)
            }
        }
    }
}


非对称转场
.asymmetric 为插入和移除定义不同的转场效果：


struct AsymmetricTransitionDemo: View {
    @State private var items: [String] = []
    private var nextItem: String { "项目 \(items.count + 1)" }

    var body: some View {
        VStack(spacing: 16) {
            ForEach(items, id: \.self) { item in
                HStack {
                    Text(item)
                    Spacer()
                    Button(role: .destructive) {
                        withAnimation(.spring(response: 0.3)) {
                            items.removeAll { $0 == item }
                        }
                    } label: {
                        Image(systemName: "xmark.circle.fill")
                    }
                }
                .padding()
                .background(.regularMaterial)
                .clipShape(RoundedRectangle(cornerRadius: 12))
                .transition(.asymmetric(
                    insertion: .move(edge: .trailing).combined(with: .opacity),
                    removal: .move(edge: .leading).combined(with: .opacity)
                ))
            }

            Button("添加项目") {
                withAnimation(.spring(response: 0.4)) {
                    items.append(nextItem)
                }
            }
        }
        .padding()
    }
}


自定义 Transition 协议 (iOS 17+)
iOS 17 引入了 Transition 协议，提供类型安全的自定义转场：


struct RotateAndFadeTransition: Transition {
    var angle: Angle

    func body(content: Content, phase: TransitionPhase) -> some View {
        content
            .rotationEffect(phase.isIdentity ? .zero : angle)
            .opacity(phase.isIdentity ? 1 : 0)
            .scaleEffect(phase.isIdentity ? 1 : 0.5)
    }
}

extension AnyTransition {
    static var rotateAndFade: AnyTransition {
        .modifier(
            active: RotateModifier(angle: .degrees(90), opacity: 0),
            identity: RotateModifier(angle: .zero, opacity: 1)
        )
    }
}

struct RotateModifier: ViewModifier {
    let angle: Angle
    let opacity: Double

    func body(content: Content) -> some View {
        content
            .rotationEffect(angle)
            .opacity(opacity)
    }
}

struct FlyInTransition: Transition {
    var edge: Edge

    func body(content: Content, phase: TransitionPhase) -> some View {
        content
            .offset(
                x: phase == .willAppear ? edgeOffset(edge).x : (phase == .didDisappear ? -edgeOffset(edge).x : 0),
                y: phase == .willAppear ? edgeOffset(edge).y : (phase == .didDisappear ? -edgeOffset(edge).y : 0)
            )
            .opacity(phase.isIdentity ? 1 : 0)
            .blur(radius: phase.isIdentity ? 0 : 10)
    }

    private func edgeOffset(_ edge: Edge) -> CGPoint {
        switch edge {
        case .top: return CGPoint(x: 0, y: -200)
        case .bottom: return CGPoint(x: 0, y: 200)
        case .leading: return CGPoint(x: -200, y: 0)
        case .trailing: return CGPoint(x: 200, y: 0)
        }
    }
}

struct CustomTransitionUsage: View {
    @State private var showBanner = false

    var body: some View {
        VStack {
            if showBanner {
                Text("重要公告")
                    .font(.headline)
                    .padding()
                    .frame(maxWidth: .infinity)
                    .background(.yellow)
                    .clipShape(RoundedRectangle(cornerRadius: 12))
                    .transition(FlyInTransition(edge: .top))
            }

            Spacer()

            Button(showBanner ? "隐藏" : "显示公告") {
                withAnimation(.spring(response: 0.5, dampingFraction: 0.7)) {
                    showBanner.toggle()
                }
            }
        }
        .padding()
    }
}


提示： Transition 协议（iOS 17+）相比旧的 AnyTransition.modifier(active:identity:) 方式更强大——TransitionPhase 区分了插入和移除方向，可以为两个方向定义不同的动画效果。






AnyTransition 扩展（旧版兼容）
对于需要支持 iOS 16 及更早版本的项目：


extension AnyTransition {
    static var slideAndFade: AnyTransition {
        .asymmetric(
            insertion: .move(edge: .trailing).combined(with: .opacity),
            removal: .move(edge: .leading).combined(with: .opacity)
        )
    }

    static func customScale(anchor: UnitPoint = .center) -> AnyTransition {
        .scale(scale: 0.1, anchor: anchor).combined(with: .opacity)
    }

    static var flip: AnyTransition {
        .modifier(
            active: FlipModifier(angle: 90),
            identity: FlipModifier(angle: 0)
        )
    }
}

struct FlipModifier: ViewModifier {
    let angle: Double

    func body(content: Content) -> some View {
        content
            .rotation3DEffect(.degrees(angle), axis: (x: 0, y: 1, z: 0))
            .opacity(angle == 0 ? 1 : 0)
    }
}


小结
    

  • 转场定义视图插入和移除时的动画效果
    • 

  • 内置转场包括 .opacity、.scale、.slide、.move(edge:) 和 .push(from:)
    • 

  • .combined(with:) 组合多个转场，.asymmetric 为插入和移除定义不同效果
    • 

  • iOS 17 的 Transition 协议通过 TransitionPhase 区分 willAppear/identity/didDisappear
    • 

  • 旧版兼容使用 AnyTransition.modifier(active:identity:) 自定义转场
    • 

  • 转场必须配合 withAnimation 或 .animation 才能生效
    • 

  • 自定义转场可封装为 AnyTransition 的静态扩展，方便复用
