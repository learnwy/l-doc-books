ObservableObject 可观察对象

概述

ObservableObject 协议是 SwiftUI 中管理引用类型状态的核心机制。通过 @Published 属性包装器和 objectWillChange 发布者，它为视图提供了对类实例属性变化的自动感知能力。ObservableObject 也是连接 SwiftUI 与 Combine 框架的桥梁。



@Published 属性包装器
@Published 会在属性值将要变化（willSet）时自动通过 objectWillChange 发送通知：


class ShoppingCart: ObservableObject {
    @Published var items: [CartItem] = []
    @Published var couponCode: String = ""

    var totalPrice: Double {
        items.reduce(0) { $0 + $1.price * Double($1.quantity) }
    }

    var discountedPrice: Double {
        couponCode == "SAVE20" ? totalPrice * 0.8 : totalPrice
    }

    func addItem(_ item: CartItem) {
        items.append(item)
    }

    func removeItem(at index: Int) {
        items.remove(at: index)
    }
}

struct CartItem: Identifiable {
    let id = UUID()
    let name: String
    let price: Double
    var quantity: Int
}
视图会自动响应 @Published 属性的变化：


struct CartView: View {
    @StateObject private var cart = ShoppingCart()

    var body: some View {
        NavigationStack {
            List {
                ForEach(cart.items) { item in
                    HStack {
                        Text(item.name)
                        Spacer()
                        Text("¥\(item.price, specifier: "%.2f") × \(item.quantity)")
                    }
                }
                .onDelete { indexSet in
                    for index in indexSet {
                        cart.removeItem(at: index)
                    }
                }
            }
            .navigationTitle("购物车")
            .toolbar {
                ToolbarItem(placement: .bottomBar) {
                    HStack {
                        Text("合计：¥\(cart.discountedPrice, specifier: "%.2f")")
                            .bold()
                        Spacer()
                        Button("添加测试") {
                            cart.addItem(CartItem(name: "商品\(cart.items.count + 1)", price: 29.9, quantity: 1))
                        }
                    }
                }
            }
        }
    }
}


objectWillChange 发布者
每个 ObservableObject 都自动合成了一个 objectWillChange 发布者。当使用 @Published 时，它会自动触发。也可以手动控制发布时机：


class ThrottledModel: ObservableObject {
    var searchText: String = "" {
        didSet {
            throttlePublish()
        }
    }

    @Published var displayedResults: [String] = []
    private var debounceTask: Task<Void, Never>?

    private func throttlePublish() {
        debounceTask?.cancel()
        debounceTask = Task { @MainActor in
            try? await Task.sleep(for: .milliseconds(300))
            guard !Task.isCancelled else { return }
            objectWillChange.send()
            displayedResults = performSearch(searchText)
        }
    }

    private func performSearch(_ query: String) -> [String] {
        guard !query.isEmpty else { return [] }
        return ["结果1: \(query)", "结果2: \(query)", "结果3: \(query)"]
    }
}

批量更新
当需要一次修改多个属性时，@Published 会为每个属性分别发送通知。可以通过手动控制来合并通知：


class ProfileModel: ObservableObject {
    var name: String = ""
    var email: String = ""
    var bio: String = ""

    func updateProfile(name: String, email: String, bio: String) {
        self.name = name
        self.email = email
        self.bio = bio
        objectWillChange.send()
    }
}


提示： 实际上 SwiftUI 会在同一个 RunLoop 周期内合并多次 objectWillChange 通知，所以多个 @Published 属性同时修改不一定导致多次 body 求值。但手动控制仍然更明确。






Combine 集成
ObservableObject 的 objectWillChange 是一个标准的 Combine Publisher，可以进行丰富的操作：


import Combine

class SearchViewModel: ObservableObject {
    @Published var query = ""
    @Published var results: [String] = []
    @Published var isLoading = false

    private var cancellables = Set<AnyCancellable>()

    init() {
        $query
            .debounce(for: .milliseconds(300), scheduler: RunLoop.main)
            .removeDuplicates()
            .filter { !$0.isEmpty }
            .sink { [weak self] query in
                self?.search(query)
            }
            .store(in: &cancellables)
    }

    private func search(_ query: String) {
        isLoading = true
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) { [weak self] in
            self?.results = (1...5).map { "「\(query)」结果 \($0)" }
            self?.isLoading = false
        }
    }
}

struct SearchView: View {
    @StateObject private var viewModel = SearchViewModel()

    var body: some View {
        NavigationStack {
            VStack {
                TextField("搜索...", text: $viewModel.query)
                    .textFieldStyle(.roundedBorder)
                    .padding()

                if viewModel.isLoading {
                    ProgressView()
                } else {
                    List(viewModel.results, id: \.self) { result in
                        Text(result)
                    }
                }
            }
            .navigationTitle("搜索")
        }
    }
}


嵌套 ObservableObject
@Published 不会自动追踪嵌套 ObservableObject 的属性变化：


class Settings: ObservableObject {
    @Published var theme = "light"
}

class AppState: ObservableObject {
    @Published var settings = Settings()
    @Published var userName = "用户"
}
修改 appState.settings.theme 不会触发 AppState 的 objectWillChange。解决方案：


class AppState: ObservableObject {
    var settings = Settings()
    @Published var userName = "用户"

    private var cancellables = Set<AnyCancellable>()

    init() {
        settings.objectWillChange
            .sink { [weak self] _ in
                self?.objectWillChange.send()
            }
            .store(in: &cancellables)
    }
}


提示： iOS 17 的 @Observable 宏解决了嵌套对象的追踪问题。如果目标平台支持，推荐迁移到 Observation 框架。






小结
    

  • ObservableObject 是管理引用类型状态的核心协议
    • 

  • @Published 在属性 willSet 时自动发送 objectWillChange 通知
    • 

  • 手动调用 objectWillChange.send() 可实现防抖、批量更新等高级控制
    • 

  • $property 语法获取 @Published 属性的 Combine Publisher
    • 

  • Combine 集成支持 debounce、removeDuplicates、filter 等操作符
    • 

  • 嵌套 ObservableObject 的属性变化不会自动传播，需手动转发
    • 

  • SwiftUI 会在同一 RunLoop 周期内合并多次通知，避免不必要的重复渲染
    •
