深度链接

概述

深度链接（Deep Linking）允许外部 URL 直接打开应用的特定页面。SwiftUI 通过 .onOpenURL 修饰器和 NavigationPath 提供了优雅的深度链接支持。结合 URL Scheme 和 Universal Links，可以实现从网页、推送通知、其他应用等渠道的无缝跳转。



URL Scheme 配置
在 Xcode 的 Info.plist 中注册 URL Scheme：


// Info.plist 中配置 URL Types
// URL Schemes: myapp
// 这样应用可以响应 myapp://... 格式的 URL
URL Scheme 的格式通常为：

说明





myapp://home
打开首页




myapp://product/123
查看商品 123




myapp://profile/user456
查看用户资料




myapp://settings/notifications
打开通知设置







.onOpenURL 处理
SwiftUI 使用 .onOpenURL 修饰器接收和处理深度链接：


@main
struct MyApp: App {
    @State private var appRouter = AppRouter()

    var body: some Scene {
        WindowGroup {
            RootView(router: appRouter)
                .onOpenURL { url in
                    appRouter.handleDeepLink(url)
                }
        }
    }
}

@Observable
class AppRouter {
    var selectedTab: Tab = .home
    var navigationPath = NavigationPath()

    enum Tab: String {
        case home, search, profile, settings
    }

    func handleDeepLink(_ url: URL) {
        guard let components = URLComponents(url: url, resolvingAgainstBaseURL: false) else { return }

        let pathSegments = components.path.split(separator: "/").map(String.init)

        switch url.host {
        case "home":
            selectedTab = .home
            navigationPath = NavigationPath()

        case "product":
            selectedTab = .home
            if let idString = pathSegments.first, let id = Int(idString) {
                navigationPath = NavigationPath()
                navigationPath.append(Route.product(id: id))
            }

        case "profile":
            selectedTab = .profile
            if let userId = pathSegments.first {
                navigationPath = NavigationPath()
                navigationPath.append(Route.userProfile(userId: userId))
            }

        case "settings":
            selectedTab = .settings
            if let section = pathSegments.first {
                navigationPath = NavigationPath()
                navigationPath.append(Route.settingsSection(section))
            }

        default:
            break
        }
    }
}

enum Route: Hashable {
    case product(id: Int)
    case userProfile(userId: String)
    case settingsSection(String)
}


完整导航集成
将深度链接处理与 NavigationStack 集成：


struct RootView: View {
    let router: AppRouter

    var body: some View {
        TabView(selection: Bindable(router).selectedTab) {
            HomeTab(router: router)
                .tag(AppRouter.Tab.home)
                .tabItem { Label("首页", systemImage: "house") }

            SearchTab()
                .tag(AppRouter.Tab.search)
                .tabItem { Label("搜索", systemImage: "magnifyingglass") }

            ProfileTab(router: router)
                .tag(AppRouter.Tab.profile)
                .tabItem { Label("我的", systemImage: "person") }

            SettingsTab(router: router)
                .tag(AppRouter.Tab.settings)
                .tabItem { Label("设置", systemImage: "gear") }
        }
    }
}

struct HomeTab: View {
    let router: AppRouter

    var body: some View {
        NavigationStack(path: Bindable(router).navigationPath) {
            List {
                ForEach(1...10, id: \.self) { id in
                    NavigationLink("商品 \(id)", value: Route.product(id: id))
                }
            }
            .navigationTitle("首页")
            .navigationDestination(for: Route.self) { route in
                switch route {
                case .product(let id):
                    ProductDetailView(productId: id)
                case .userProfile(let userId):
                    UserProfileView(userId: userId)
                case .settingsSection(let section):
                    SettingsSectionView(section: section)
                }
            }
        }
    }
}

struct ProductDetailView: View {
    let productId: Int

    var body: some View {
        VStack(spacing: 16) {
            Image(systemName: "shippingbox")
                .font(.system(size: 60))
                .foregroundStyle(.blue)
            Text("商品 #\(productId)")
                .font(.title)
            Text("这是通过深度链接或普通导航到达的商品详情页")
                .font(.body)
                .foregroundStyle(.secondary)
                .multilineTextAlignment(.center)
        }
        .padding()
        .navigationTitle("商品详情")
    }
}


Universal Links
Universal Links 使用 HTTPS URL，需要在服务器配置 AASA 文件：


struct UniversalLinkHandler: View {
    var body: some View {
        ContentView()
            .onOpenURL { url in
                if url.scheme == "https" && url.host == "example.com" {
                    handleUniversalLink(url)
                } else {
                    handleCustomScheme(url)
                }
            }
    }

    func handleUniversalLink(_ url: URL) {
        let path = url.pathComponents

        if path.contains("product"), let idStr = path.last, let id = Int(idStr) {
            print("打开商品: \(id)")
        } else if path.contains("user"), let userId = path.last {
            print("打开用户: \(userId)")
        }
    }

    func handleCustomScheme(_ url: URL) {
        print("处理自定义 scheme: \(url)")
    }
}

导航状态恢复
结合深度链接和 Codable 的 NavigationPath，可以实现完整的导航状态恢复：


class NavigationStateManager {
    private let key = "savedNavigationState"

    func save(tab: AppRouter.Tab, path: NavigationPath) {
        var state: [String: Any] = ["tab": tab.rawValue]
        if let codable = path.codable,
           let data = try? JSONEncoder().encode(codable) {
            state["path"] = data
        }
        UserDefaults.standard.set(state, forKey: key)
    }

    func restore() -> (tab: AppRouter.Tab, path: NavigationPath)? {
        guard let state = UserDefaults.standard.dictionary(forKey: key),
              let tabRaw = state["tab"] as? String,
              let tab = AppRouter.Tab(rawValue: tabRaw) else {
            return nil
        }

        var path = NavigationPath()
        if let data = state["path"] as? Data,
           let codable = try? JSONDecoder().decode(NavigationPath.CodableRepresentation.self, from: data) {
            path = NavigationPath(codable)
        }

        return (tab, path)
    }
}


提示： 导航状态恢复应在 scenePhase 变为 .background 时保存，在应用启动或 .active 时恢复。注意处理无效的恢复数据（如已删除的商品 ID）。






小结
    

  • URL Scheme 使用自定义协议（myapp://），配置简单但不保证唯一性
    • 

  • Universal Links 使用 HTTPS URL，需服务器配置 AASA 文件，安全性更高
    • 

  • .onOpenURL 是 SwiftUI 中处理深度链接的统一入口
    • 

  • 将 URL 解析为类型安全的 Route 枚举，再映射到 NavigationPath 操作
    • 

  • 完整的深度链接方案需要处理 Tab 切换和导航栈重建
    • 

  • 结合 NavigationPath.CodableRepresentation 可实现导航状态持久化
    • 

  • 应处理 URL 格式错误和无效参数等异常情况
