Sheet 与全屏覆盖

概述

SwiftUI 提供了 .sheet、.fullScreenCover 和 .popover 等模态呈现方式。iOS 16 引入了 Sheet Detents（半屏高度）支持，使得模态交互更加灵活。理解各种模态的触发、关闭、数据传递和自定义行为，是构建流畅交互体验的关键。



.sheet 基本用法
.sheet 以卡片形式从底部弹出：


struct SheetDemo: View {
    @State private var showProfile = false
    @State private var selectedUser: User?

    var body: some View {
        VStack(spacing: 20) {
            Button("显示个人资料") {
                showProfile = true
            }

            Button("显示用户详情") {
                selectedUser = User(name: "张三", email: "zhang@example.com")
            }
        }
        .sheet(isPresented: $showProfile) {
            ProfileSheet()
        }
        .sheet(item: $selectedUser) { user in
            UserDetailSheet(user: user)
        }
    }
}

struct User: Identifiable {
    let id = UUID()
    let name: String
    let email: String
}




触发方式
参数
说明





isPresented
Binding<Bool>
布尔值控制显示/隐藏




item
Binding<T?>
可选值非 nil 时显示







关闭 Sheet
使用 @Environment(.dismiss)
struct ProfileSheet: View {
    @Environment(\.dismiss) var dismiss

    var body: some View {
        NavigationStack {
            VStack {
                Text("个人资料")
                    .font(.title)

                Form {
                    Text("姓名：张三")
                    Text("邮箱：zhang@example.com")
                }
            }
            .navigationTitle("资料")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button("完成") {
                        dismiss()
                    }
                }
            }
        }
    }
}
onDismiss 回调
struct ParentView: View {
    @State private var showEditor = false
    @State private var savedData: String?

    var body: some View {
        VStack {
            Text("保存的数据：\(savedData ?? "无")")

            Button("编辑") { showEditor = true }
        }
        .sheet(isPresented: $showEditor, onDismiss: {
            print("Sheet 已关闭")
        }) {
            EditorSheet(savedData: $savedData)
        }
    }
}

struct EditorSheet: View {
    @Environment(\.dismiss) var dismiss
    @Binding var savedData: String?
    @State private var text = ""

    var body: some View {
        NavigationStack {
            TextField("输入内容", text: $text)
                .textFieldStyle(.roundedBorder)
                .padding()
                .navigationTitle("编辑")
                .toolbar {
                    ToolbarItem(placement: .topBarLeading) {
                        Button("取消") { dismiss() }
                    }
                    ToolbarItem(placement: .topBarTrailing) {
                        Button("保存") {
                            savedData = text
                            dismiss()
                        }
                    }
                }
        }
    }
}


Sheet Detents（半屏高度）
iOS 16+ 支持自定义 Sheet 的呈现高度：


struct DetentDemo: View {
    @State private var showSheet = false

    var body: some View {
        Button("显示半屏 Sheet") {
            showSheet = true
        }
        .sheet(isPresented: $showSheet) {
            ScrollView {
                VStack(alignment: .leading, spacing: 16) {
                    Text("评论列表")
                        .font(.title2)
                        .bold()

                    ForEach(1...20, id: \.self) { i in
                        HStack {
                            Image(systemName: "person.circle")
                            VStack(alignment: .leading) {
                                Text("用户 \(i)")
                                    .font(.headline)
                                Text("这是第 \(i) 条评论内容...")
                                    .font(.subheadline)
                                    .foregroundStyle(.secondary)
                            }
                        }
                        .padding(.vertical, 4)
                    }
                }
                .padding()
            }
            .presentationDetents([.medium, .large, .fraction(0.3)])
            .presentationDragIndicator(.visible)
            .presentationBackgroundInteraction(.enabled(upThrough: .medium))
        }
    }
}




Detent 类型
说明





.medium
屏幕高度的约一半




.large
全屏（默认）




.fraction(0.3)
屏幕高度的 30%




.height(200)
固定 200pt 高度




自定义
实现 CustomPresentationDetent





自定义 Detent
struct SmallDetent: CustomPresentationDetent {
    static func height(in context: Context) -> CGFloat? {
        max(context.maxDetentValue * 0.25, 120)
    }
}

struct CustomDetentDemo: View {
    @State private var showSheet = false

    var body: some View {
        Button("自定义高度") { showSheet = true }
            .sheet(isPresented: $showSheet) {
                Text("自定义 Detent 内容")
                    .presentationDetents([.custom(SmallDetent.self), .medium, .large])
            }
    }
}


提示： 使用 .presentationBackgroundInteraction(.enabled(upThrough:)) 可以让用户在 Sheet 未全屏时与背后的内容交互。






.fullScreenCover
全屏覆盖不支持下滑关闭，必须提供明确的关闭按钮：


struct FullScreenDemo: View {
    @State private var showOnboarding = false

    var body: some View {
        Button("显示引导页") {
            showOnboarding = true
        }
        .fullScreenCover(isPresented: $showOnboarding) {
            OnboardingView(isPresented: $showOnboarding)
        }
    }
}

struct OnboardingView: View {
    @Binding var isPresented: Bool
    @State private var currentPage = 0

    let pages = [
        ("手机", "欢迎使用我们的应用"),
        ("heart", "发现你喜欢的内容"),
        ("star", "开始你的旅程")
    ]

    var body: some View {
        VStack(spacing: 32) {
            TabView(selection: $currentPage) {
                ForEach(pages.indices, id: \.self) { index in
                    VStack(spacing: 16) {
                        Image(systemName: pages[index].0)
                            .font(.system(size: 80))
                            .foregroundStyle(.blue)
                        Text(pages[index].1)
                            .font(.title2)
                    }
                    .tag(index)
                }
            }
            .tabViewStyle(.page)

            if currentPage == pages.count - 1 {
                Button("开始使用") {
                    isPresented = false
                }
                .buttonStyle(.borderedProminent)
            }
        }
        .padding()
    }
}

防止意外关闭
struct ProtectedSheet: View {
    @Environment(\.dismiss) var dismiss
    @State private var hasUnsavedChanges = false
    @State private var text = ""
    @State private var showConfirmation = false

    var body: some View {
        NavigationStack {
            Form {
                TextField("内容", text: $text)
                    .onChange(of: text) { _, _ in
                        hasUnsavedChanges = true
                    }
            }
            .navigationTitle("编辑")
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button("保存") {
                        hasUnsavedChanges = false
                        dismiss()
                    }
                }
            }
        }
        .interactiveDismissDisabled(hasUnsavedChanges)
        .confirmationDialog("确定要放弃更改吗？", isPresented: $showConfirmation) {
            Button("放弃更改", role: .destructive) { dismiss() }
            Button("继续编辑", role: .cancel) { }
        }
    }
}


小结
    

  • .sheet 以卡片形式弹出，支持下滑关闭和 Detents 半屏高度
    • 

  • .fullScreenCover 全屏覆盖，不支持下滑关闭，适合引导页和登录页
    • 

  • 两种触发方式：isPresented（布尔值）和 item（可选值）
    • 

  • @Environment(\.dismiss) 用于在 Sheet 内部触发关闭
    • 

  • Detents 支持 .medium、.large、.fraction()、.height() 和自定义
    • 

  • .interactiveDismissDisabled() 防止用户在有未保存数据时意外关闭
    • 

  • onDismiss 回调在 Sheet 完全关闭后执行
    •
