自定义布局

概述

基于 Layout 协议，SwiftUI 允许开发者创建功能丰富的自定义布局容器。本文通过实现流式布局（Flow Layout）、径向布局（Radial Layout）等实际案例，深入展示布局优先级、动画过渡以及布局切换等高级技巧。



流式布局 (Flow Layout)
流式布局会在水平空间不足时自动换行，常用于标签（Tag）展示：


struct FlowLayout: Layout {
    var spacing: CGFloat = 8

    func sizeThatFits(
        proposal: ProposedViewSize,
        subviews: Subviews,
        cache: inout ()
    ) -> CGSize {
        let containerWidth = proposal.width ?? .infinity
        var currentX: CGFloat = 0
        var currentY: CGFloat = 0
        var lineHeight: CGFloat = 0
        var maxWidth: CGFloat = 0

        for subview in subviews {
            let size = subview.sizeThatFits(.unspecified)
            if currentX + size.width > containerWidth && currentX > 0 {
                currentX = 0
                currentY += lineHeight + spacing
                lineHeight = 0
            }
            lineHeight = max(lineHeight, size.height)
            currentX += size.width + spacing
            maxWidth = max(maxWidth, currentX - spacing)
        }

        return CGSize(width: maxWidth, height: currentY + lineHeight)
    }

    func placeSubviews(
        in bounds: CGRect,
        proposal: ProposedViewSize,
        subviews: Subviews,
        cache: inout ()
    ) {
        var currentX = bounds.minX
        var currentY = bounds.minY
        var lineHeight: CGFloat = 0

        for subview in subviews {
            let size = subview.sizeThatFits(.unspecified)
            if currentX + size.width > bounds.maxX && currentX > bounds.minX {
                currentX = bounds.minX
                currentY += lineHeight + spacing
                lineHeight = 0
            }

            subview.place(
                at: CGPoint(x: currentX, y: currentY),
                anchor: .topLeading,
                proposal: ProposedViewSize(size)
            )

            lineHeight = max(lineHeight, size.height)
            currentX += size.width + spacing
        }
    }
}
使用流式布局展示标签：


struct TagCloudView: View {
    let tags = ["SwiftUI", "iOS", "Layout", "Protocol", "动画",
                "导航", "数据流", "性能优化", "Preview", "Xcode"]

    var body: some View {
        FlowLayout(spacing: 8) {
            ForEach(tags, id: \.self) { tag in
                Text(tag)
                    .font(.callout)
                    .padding(.horizontal, 12)
                    .padding(.vertical, 6)
                    .background(.blue.opacity(0.15))
                    .clipShape(Capsule())
            }
        }
        .padding()
    }
}


径向布局 (Radial Layout)
将子视图沿圆周均匀排列：


struct RadialLayout: Layout {
    var radius: CGFloat = 100
    var startAngle: Angle = .degrees(-90)

    func sizeThatFits(
        proposal: ProposedViewSize,
        subviews: Subviews,
        cache: inout ()
    ) -> CGSize {
        let maxChildSize = subviews.reduce(CGSize.zero) { result, subview in
            let size = subview.sizeThatFits(.unspecified)
            return CGSize(
                width: max(result.width, size.width),
                height: max(result.height, size.height)
            )
        }
        let diameter = radius * 2 + max(maxChildSize.width, maxChildSize.height)
        return CGSize(width: diameter, height: diameter)
    }

    func placeSubviews(
        in bounds: CGRect,
        proposal: ProposedViewSize,
        subviews: Subviews,
        cache: inout ()
    ) {
        let angleStep = Angle.degrees(360.0 / Double(subviews.count))

        for (index, subview) in subviews.enumerated() {
            let angle = startAngle + angleStep * Double(index)
            let x = bounds.midX + radius * cos(angle.radians)
            let y = bounds.midY + radius * sin(angle.radians)

            subview.place(
                at: CGPoint(x: x, y: y),
                anchor: .center,
                proposal: .unspecified
            )
        }
    }
}
struct RadialDemo: View {
    let icons = ["house", "gear", "person", "heart", "star", "bell"]

    var body: some View {
        RadialLayout(radius: 120) {
            ForEach(icons, id: \.self) { icon in
                Image(systemName: icon)
                    .font(.title)
                    .padding(12)
                    .background(.regularMaterial)
                    .clipShape(Circle())
            }
        }
        .frame(width: 300, height: 300)
    }
}


动画布局过渡
AnyLayout 切换
AnyLayout 可以在不同布局之间进行带动画的切换：


struct LayoutTransitionView: View {
    @State private var isGrid = false

    var body: some View {
        let layout = isGrid
            ? AnyLayout(GridLayout(columns: 3))
            : AnyLayout(FlowLayout(spacing: 12))

        VStack {
            layout {
                ForEach(0..<9) { index in
                    RoundedRectangle(cornerRadius: 8)
                        .foregroundStyle(.blue.gradient)
                        .frame(width: 60, height: 60)
                        .overlay {
                            Text("\(index)")
                                .foregroundStyle(.white)
                                .bold()
                        }
                }
            }
            .animation(.spring(response: 0.5), value: isGrid)

            Button(isGrid ? "切换为流式" : "切换为网格") {
                isGrid.toggle()
            }
            .padding(.top)
        }
        .padding()
    }
}


提示： AnyLayout 切换时，子视图的标识保持不变，因此 SwiftUI 可以对位置变化应用平滑动画，而不是销毁和重建视图。






布局优先级
通过 LayoutValueKey 让子视图向布局传递优先级信息：


struct PriorityKey: LayoutValueKey {
    static let defaultValue: Int = 0
}

extension View {
    func layoutOrder(_ order: Int) -> some View {
        layoutValue(key: PriorityKey.self, value: order)
    }
}

struct PriorityHStack: Layout {
    var spacing: CGFloat = 8

    func sizeThatFits(
        proposal: ProposedViewSize,
        subviews: Subviews,
        cache: inout ()
    ) -> CGSize {
        let height = subviews.reduce(CGFloat.zero) { result, subview in
            max(result, subview.sizeThatFits(.unspecified).height)
        }
        return CGSize(width: proposal.width ?? 0, height: height)
    }

    func placeSubviews(
        in bounds: CGRect,
        proposal: ProposedViewSize,
        subviews: Subviews,
        cache: inout ()
    ) {
        let sorted = subviews.sorted { a, b in
            a[PriorityKey.self] > b[PriorityKey.self]
        }

        var remainingWidth = bounds.width
        var allocations: [CGFloat] = Array(repeating: 0, count: subviews.count)

        for subview in sorted {
            let index = subviews.firstIndex(where: { $0.id == subview.id }) ?? 0
            let idealWidth = subview.sizeThatFits(.unspecified).width
            let allocated = min(idealWidth, remainingWidth)
            allocations[index] = allocated
            remainingWidth -= allocated + spacing
        }

        var x = bounds.minX
        for (index, subview) in subviews.enumerated() {
            subview.place(
                at: CGPoint(x: x, y: bounds.midY),
                anchor: .leading,
                proposal: ProposedViewSize(width: allocations[index], height: bounds.height)
            )
            x += allocations[index] + spacing
        }
    }
}
使用带优先级的布局：


struct PriorityDemo: View {
    var body: some View {
        PriorityHStack(spacing: 8) {
            Text("重要内容")
                .layoutOrder(2)
                .padding()
                .background(.red.opacity(0.2))

            Text("普通内容")
                .layoutOrder(0)
                .padding()
                .background(.blue.opacity(0.2))

            Text("次要内容")
                .layoutOrder(1)
                .padding()
                .background(.green.opacity(0.2))
        }
        .padding()
    }
}


响应式网格布局
结合 ViewThatFits 和自定义布局，实现完全响应式的布局：


struct ResponsiveGrid: View {
    var body: some View {
        GeometryReader { geometry in
            let columns = max(1, Int(geometry.size.width / 150))

            CachedGridLayout(columns: columns) {
                ForEach(0..<12) { index in
                    RoundedRectangle(cornerRadius: 12)
                        .foregroundStyle(.blue.gradient)
                        .frame(height: 100)
                        .overlay {
                            Text("卡片 \(index + 1)")
                                .foregroundStyle(.white)
                                .bold()
                        }
                }
            }
            .padding()
        }
    }
}

小结
    

  • 流式布局在水平空间不足时自动换行，适合标签和关键词展示
    • 

  • 径向布局将子视图沿圆周均匀排列，适合菜单和仪表盘场景
    • 

  • AnyLayout 可在不同布局间带动画切换，子视图标识不变确保平滑过渡
    • 

  • LayoutValueKey 让子视图向布局容器传递优先级等自定义参数
    • 

  • 结合 GeometryReader 可以根据屏幕尺寸动态调整网格列数
    • 

  • 自定义布局是 SwiftUI 中最强大的布局工具，适合内置容器无法满足的场景
