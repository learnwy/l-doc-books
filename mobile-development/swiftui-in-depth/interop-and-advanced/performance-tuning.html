性能调优

概述

SwiftUI 的声明式范式简化了 UI 开发，但性能问题可能隐藏在不当的状态管理、过大的视图树和不必要的重绘中。本文从 Lazy Stacks、Equatable 视图、任务管理、Instruments 分析到渲染优化，系统介绍 SwiftUI 性能调优的实践方法。



Lazy Stacks
普通的 VStack / HStack 会一次性创建所有子视图。LazyVStack / LazyHStack 只在子视图即将出现在屏幕上时才创建：


struct LazyStackComparison: View {
    var body: some View {
        NavigationStack {
            List {
                NavigationLink("VStack（全量创建）") {
                    ScrollView {
                        VStack {
                            ForEach(0..<10000, id: \.self) { index in
                                ExpensiveRow(index: index)
                            }
                        }
                    }
                }

                NavigationLink("LazyVStack（按需创建）") {
                    ScrollView {
                        LazyVStack {
                            ForEach(0..<10000, id: \.self) { index in
                                ExpensiveRow(index: index)
                            }
                        }
                    }
                }
            }
            .navigationTitle("性能对比")
        }
    }
}

struct ExpensiveRow: View {
    let index: Int

    init(index: Int) {
        self.index = index
    }

    var body: some View {
        HStack {
            Circle()
                .frame(width: 40, height: 40)
                .foregroundStyle(.blue.opacity(Double(index % 10) / 10.0))
            VStack(alignment: .leading) {
                Text("项目 #\(index)")
                    .font(.headline)
                Text("描述内容")
                    .font(.caption)
                    .foregroundStyle(.secondary)
            }
            Spacer()
        }
        .padding(.vertical, 4)
    }
}

提示： LazyVStack 创建的视图不会被销毁（除非从数据源移除）。如果每个视图持有大量数据，可能导致内存增长。考虑使用 List 获得视图复用能力。






Equatable 视图
通过遵循 Equatable，可以让 SwiftUI 跳过不必要的 body 求值：


struct MetricCard: View, Equatable {
    let title: String
    let value: Double
    let unit: String

    static func == (lhs: MetricCard, rhs: MetricCard) -> Bool {
        lhs.title == rhs.title &&
        lhs.value == rhs.value &&
        lhs.unit == rhs.unit
    }

    var body: some View {
        VStack(spacing: 4) {
            Text(title)
                .font(.caption)
                .foregroundStyle(.secondary)
            Text("\(value, specifier: "%.1f")\(unit)")
                .font(.title2)
                .bold()
        }
        .padding()
        .background(.regularMaterial)
        .clipShape(RoundedRectangle(cornerRadius: 12))
    }
}

struct DashboardView: View {
    @State private var temperature = 22.5
    @State private var humidity = 65.0
    @State private var pressure = 1013.0

    var body: some View {
        VStack(spacing: 16) {
            HStack(spacing: 16) {
                EquatableView(content: MetricCard(title: "温度", value: temperature, unit: "°C"))
                EquatableView(content: MetricCard(title: "湿度", value: humidity, unit: "%"))
                EquatableView(content: MetricCard(title: "气压", value: pressure, unit: "hPa"))
            }

            Button("只更新温度") {
                temperature += 0.5
            }
        }
        .padding()
    }
}


减少不必要的更新
拆分视图
将大视图拆分为小视图，使 SwiftUI 可以独立更新各部分：


struct OptimizedParent: View {
    @State private var counter = 0
    @State private var text = "Hello"

    var body: some View {
        VStack(spacing: 16) {
            CounterSection(counter: counter)
            TextSection(text: text)

            HStack {
                Button("增加") { counter += 1 }
                Button("修改文字") { text = "World \(counter)" }
            }
        }
    }
}

struct CounterSection: View {
    let counter: Int

    var body: some View {
        Text("计数：\(counter)")
            .font(.title)
    }
}

struct TextSection: View {
    let text: String

    var body: some View {
        Text(text)
            .font(.body)
    }
}
避免 @ObservedObject 的过度通知
@Observable
class FilteredModel {
    var allItems: [String] = []
    var searchQuery = ""

    var filteredItems: [String] {
        guard !searchQuery.isEmpty else { return allItems }
        return allItems.filter { $0.localizedCaseInsensitiveContains(searchQuery) }
    }
}


任务管理
.task 与取消
struct AsyncDataView: View {
    let categoryId: String
    @State private var items: [String] = []
    @State private var isLoading = true

    var body: some View {
        Group {
            if isLoading {
                ProgressView()
            } else {
                List(items, id: \.self) { item in
                    Text(item)
                }
            }
        }
        .task(id: categoryId) {
            isLoading = true
            items = await loadItems(for: categoryId)
            isLoading = false
        }
    }

    func loadItems(for category: String) async -> [String] {
        try? await Task.sleep(for: .seconds(1))
        guard !Task.isCancelled else { return [] }
        return (1...20).map { "\(category) 项目 \($0)" }
    }
}




任务管理方式
说明





.task {}

.task(id:) {}
id 变化时取消旧任务，启动新任务




Task.isCancelled
在长操作中检查取消状态




try Task.checkCancellation()

渲染优化技巧
struct RenderOptimization: View {
    @State private var items = (0..<100).map { "项目 \($0)" }

    var body: some View {
        ScrollView {
            LazyVStack {
                ForEach(items, id: \.self) { item in
                    Text(item)
                        .padding()
                        .frame(maxWidth: .infinity, alignment: .leading)
                        .background(.regularMaterial)
                        .clipShape(RoundedRectangle(cornerRadius: 8))
                        .drawingGroup()
                }
            }
            .padding()
        }
    }
}




技巧
说明





drawingGroup()
将视图栅格化为位图，减少渲染层级




避免大量 shadow
阴影计算开销大，考虑用图片替代




减少 clipShape 嵌套
每层裁剪都增加渲染开销




使用 containerRelativeFrame
iOS 17+ 替代部分 GeometryReader




避免频繁 onChange
合并状态更新，减少回调频率







提示： 使用 Xcode 的 View Debugger（Debug → View Debugging → Capture View Hierarchy）可以直观查看视图层级深度和渲染问题。






小结
    

  • Lazy Stacks 按需创建子视图，适用于大量数据场景
    • 

  • EquatableView 通过自定义比较逻辑跳过不必要的 body 求值
    • 

  • 拆分小视图让 SwiftUI 可以独立更新各部分，减少重绘范围
    • 

  • .task(id:) 在 id 变化时自动取消旧任务，避免数据竞争
    • 

  • Instruments 的 SwiftUI 模板可追踪 body 调用次数和渲染帧率
    • 

  • drawingGroup() 将复杂视图栅格化为位图，减少渲染层级
    • 

  • 性能优化的核心是减少不必要的 body 求值和最小化渲染更新范围
