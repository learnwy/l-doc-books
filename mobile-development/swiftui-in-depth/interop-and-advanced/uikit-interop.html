UIKit 互操作

概述

SwiftUI 与 UIKit 之间的互操作性是渐进式迁移和利用成熟 UIKit 组件的关键。UIViewRepresentable 将 UIKit 视图嵌入 SwiftUI，UIViewControllerRepresentable 将整个视图控制器嵌入 SwiftUI。Coordinator 模式则负责处理 UIKit 的委托回调。理解这些桥接机制，是构建混合架构应用的基础。



UIViewRepresentable 基本结构
UIViewRepresentable 将 UIKit 的 UIView 包装为 SwiftUI 视图：


struct UIKitLabel: UIViewRepresentable {
    let text: String
    let font: UIFont

    func makeUIView(context: Context) -> UILabel {
        let label = UILabel()
        label.numberOfLines = 0
        label.textAlignment = .center
        return label
    }

    func updateUIView(_ uiView: UILabel, context: Context) {
        uiView.text = text
        uiView.font = font
    }
}

struct LabelDemo: View {
    var body: some View {
        UIKitLabel(
            text: "这是一个 UIKit UILabel",
            font: .systemFont(ofSize: 24, weight: .bold)
        )
        .frame(height: 100)
    }
}

Coordinator 模式
Coordinator 是 UIKit 委托和 SwiftUI 之间的桥梁：


struct UIKitTextField: UIViewRepresentable {
    @Binding var text: String
    var placeholder: String
    var onSubmit: (() -> Void)?

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIView(context: Context) -> UITextField {
        let textField = UITextField()
        textField.placeholder = placeholder
        textField.borderStyle = .roundedRect
        textField.delegate = context.coordinator
        textField.addTarget(
            context.coordinator,
            action: #selector(Coordinator.textChanged(_:)),
            for: .editingChanged
        )
        return textField
    }

    func updateUIView(_ uiView: UITextField, context: Context) {
        if uiView.text != text {
            uiView.text = text
        }
    }

    class Coordinator: NSObject, UITextFieldDelegate {
        var parent: UIKitTextField

        init(_ parent: UIKitTextField) {
            self.parent = parent
        }

        @objc func textChanged(_ textField: UITextField) {
            parent.text = textField.text ?? ""
        }

        func textFieldShouldReturn(_ textField: UITextField) -> Bool {
            parent.onSubmit?()
            textField.resignFirstResponder()
            return true
        }
    }
}


UIViewControllerRepresentable
包装整个 UIViewController：


struct ImagePicker: UIViewControllerRepresentable {
    @Binding var selectedImage: UIImage?
    @Environment(\.dismiss) var dismiss

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIViewController(context: Context) -> UIImagePickerController {
        let picker = UIImagePickerController()
        picker.delegate = context.coordinator
        picker.sourceType = .photoLibrary
        return picker
    }

    func updateUIViewController(_ uiViewController: UIImagePickerController, context: Context) {}

    class Coordinator: NSObject, UIImagePickerControllerDelegate, UINavigationControllerDelegate {
        let parent: ImagePicker

        init(_ parent: ImagePicker) {
            self.parent = parent
        }

        func imagePickerController(_ picker: UIImagePickerController, didFinishPickingMediaWithInfo info: [UIImagePickerController.InfoKey: Any]) {
            if let image = info[.originalImage] as? UIImage {
                parent.selectedImage = image
            }
            parent.dismiss()
        }

        func imagePickerControllerDidCancel(_ picker: UIImagePickerController) {
            parent.dismiss()
        }
    }
}

struct ImagePickerDemo: View {
    @State private var showPicker = false
    @State private var image: UIImage?

    var body: some View {
        VStack {
            if let image {
                Image(uiImage: image)
                    .resizable()
                    .scaledToFit()
                    .frame(height: 200)
            }

            Button("选择图片") { showPicker = true }
        }
        .sheet(isPresented: $showPicker) {
            ImagePicker(selectedImage: $image)
        }
    }
}

SwiftUI 嵌入 UIKit (UIHostingController)
反向操作——将 SwiftUI 视图嵌入 UIKit：


import UIKit
import SwiftUI

class MainViewController: UIViewController {
    override func viewDidLoad() {
        super.viewDidLoad()

        let swiftUIView = VStack {
            Text("来自 SwiftUI")
                .font(.title)
            Image(systemName: "swift")
                .font(.system(size: 60))
                .foregroundStyle(.orange)
        }

        let hostingController = UIHostingController(rootView: swiftUIView)
        addChild(hostingController)
        view.addSubview(hostingController.view)

        hostingController.view.translatesAutoresizingMaskIntoConstraints = false
        NSLayoutConstraint.activate([
            hostingController.view.topAnchor.constraint(equalTo: view.topAnchor),
            hostingController.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            hostingController.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            hostingController.view.bottomAnchor.constraint(equalTo: view.bottomAnchor)
        ])

        hostingController.didMove(toParent: self)
    }
}


小结
    

  • UIViewRepresentable 将 UIKit 视图嵌入 SwiftUI，需实现 makeUIView 和 updateUIView
    • 

  • UIViewControllerRepresentable 将视图控制器嵌入 SwiftUI
    • 

  • Coordinator 作为 UIKit 委托和 SwiftUI 之间的桥梁，处理回调和事件
    • 

  • dismantleUIView 是静态方法，用于视图移除时的资源清理
    • 

  • UIHostingController 实现反向嵌入——将 SwiftUI 视图嵌入 UIKit
    • 

  • updateUIView 可能被频繁调用，应检查值变化避免不必要更新
    • 

  • Coordinator 持有 parent 引用，需注意避免循环引用
