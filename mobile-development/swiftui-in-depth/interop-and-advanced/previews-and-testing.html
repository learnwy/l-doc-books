预览与测试

概述

SwiftUI 的 #Preview 宏和丰富的预览配置能力，使得开发者可以在不运行模拟器的情况下快速验证 UI。结合 ViewInspector 等测试库和快照测试，可以建立完整的 SwiftUI 测试策略。本文涵盖预览宏的高级用法、多种预览配置、单元测试和快照测试方案。



#Preview 宏
Swift 5.9 引入的 #Preview 宏替代了旧版的 PreviewProvider：


#Preview {
    ContentView()
}

#Preview("登录状态") {
    ContentView()
        .environmentObject(AuthManager.loggedInPreview)
}

#Preview("空状态") {
    ContentView()
        .environmentObject(AuthManager.emptyPreview)
}

#Preview("浅色模式") {
    SettingsCard(title: "通知", icon: "bell", isEnabled: true)
        .padding()

}

#Preview("深色模式") {
    SettingsCard(title: "通知", icon: "bell", isEnabled: true)
        .padding()

    ProfileView()

        CustomButton(title: "危险按钮", style: .destructive)
    }
    .padding()
    .previewLayout(.sizeThatFits)
}


预览中的状态管理
struct StatefulPreviewWrapper<Content: View>: View {
    @State private var value: Bool

    let content: (Binding<Bool>) -> Content

    init(_ initialValue: Bool, @ViewBuilder content: @escaping (Binding<Bool>) -> Content) {
        _value = State(initialValue: initialValue)
        self.content = content
    }

    var body: some View {
        content($value)
    }
}

#Preview("Toggle 交互") {
    StatefulPreviewWrapper(true) { isOn in
        ToggleCard(title: "推送通知", isOn: isOn)
            .padding()
    }
}

extension User {
    static let preview = User(name: "预览用户", email: "preview@test.com")
    static let emptyProfile = User(name: "", email: "")
    static let longName = User(name: "这是一个非常非常长的用户名字需要测试截断效果", email: "long@test.com")

extension [Product] {
    static let previewList: [Product] = [
        Product(name: "iPhone", price: 6999),
        Product(name: "MacBook", price: 12999),
        Product(name: "AirPods", price: 1599),
    ]
}

#Preview("商品列表") {
    ProductListView(products: .previewList)
}

#Preview("空列表") {
    ProductListView(products: [])
}


提示： 将预览数据定义为静态属性扩展，可以在多个 Preview 和测试中复用，保持数据的一致性。






ViewModel 单元测试
ViewModel 是最容易测试的层：


@Observable
class TodoViewModel {
    var items: [TodoItem] = []
    var newItemTitle = ""

    var pendingCount: Int {
        items.filter { !$0.isDone }.count
    }

    func addItem() {
        guard !newItemTitle.trimmingCharacters(in: .whitespaces).isEmpty else { return }
        items.append(TodoItem(title: newItemTitle, isDone: false))
        newItemTitle = ""
    }

    func toggleItem(_ item: TodoItem) {
        if let index = items.firstIndex(where: { $0.id == item.id }) {
            items[index].isDone.toggle()
        }
    }

    func deleteItem(_ item: TodoItem) {
        items.removeAll { $0.id == item.id }
    }
}

struct TodoItem: Identifiable, Equatable {
    let id = UUID()
    var title: String
    var isDone: Bool
}
对应的测试：


import Testing

@Suite("TodoViewModel 测试")
struct TodoViewModelTests {
    @Test("添加项目")
    func addItem() {
        let vm = TodoViewModel()
        vm.newItemTitle = "买牛奶"
        vm.addItem()

        #expect(vm.items.count == 1)
        #expect(vm.items.first?.title == "买牛奶")
        #expect(vm.newItemTitle.isEmpty)
    }

    @Test("空标题不添加")
    func addEmptyItem() {
        let vm = TodoViewModel()
        vm.newItemTitle = "   "
        vm.addItem()

        #expect(vm.items.isEmpty)
    }

    @Test("切换完成状态")
    func toggleItem() {
        let vm = TodoViewModel()
        vm.newItemTitle = "测试"
        vm.addItem()
        let item = vm.items.first!

        vm.toggleItem(item)
        #expect(vm.items.first?.isDone == true)

        vm.toggleItem(item)
        #expect(vm.items.first?.isDone == false)
    }

    @Test("待办计数")
    func pendingCount() {
        let vm = TodoViewModel()
        vm.newItemTitle = "任务1"
        vm.addItem()
        vm.newItemTitle = "任务2"
        vm.addItem()

        #expect(vm.pendingCount == 2)

        vm.toggleItem(vm.items.first!)
        #expect(vm.pendingCount == 1)
    }
}


ViewInspector 视图测试
ViewInspector 是一个第三方库，支持对 SwiftUI 视图进行检查：


import ViewInspector

extension ContentView: Inspectable {}

func testContentView() throws {
    let view = ContentView()
    let text = try view.inspect().find(text: "欢迎")
    #expect(text.string() == "欢迎")

    let button = try view.inspect().find(button: "开始")
    try button.tap()
}


快照测试
快照测试通过对比 UI 渲染结果的图片来检测 UI 回归：


import SnapshotTesting

func testProfileCard() {
    let view = ProfileCard(user: .preview)
        .frame(width: 375)
        .padding()

    assertSnapshot(of: view, as: .image(layout: .sizeThatFits))
}

func testProfileCardDarkMode() {
    let view = ProfileCard(user: .preview)
        .frame(width: 375)
        .padding()
        .preferredColorScheme(.dark)

    assertSnapshot(of: view, as: .image(layout: .sizeThatFits), named: "dark")
}

小结
    

  • #Preview 宏（Swift 5.9+）语法简洁，支持命名和多预览定义
    • 

  • 预览配置支持深色/浅色模式、动态字体、不同设备等多种环境
    • 

  • 将预览数据定义为静态属性扩展，方便在 Preview 和测试中复用
    • 

  • ViewModel 单元测试是最高效的测试方式，验证业务逻辑和数据转换
    • 

  • ViewInspector 支持对 SwiftUI 视图结构和交互的检查
    • 

  • 快照测试通过图片对比检测 UI 回归，适合组件库和设计系统
    • 

  • 测试策略应以 ViewModel 测试为核心，快照测试为辅助，UI 测试按需使用
    •
