UIViewRepresentable 详解

概述

UIViewRepresentable 是将 UIKit 视图桥接到 SwiftUI 的核心协议。本文深入探讨 makeUIView、updateUIView、dismantleUIView 的完整生命周期，以及尺寸策略、动画上下文和常见的复杂桥接场景。



makeUIView 详解
makeUIView 只在视图首次创建时调用一次，负责初始化和一次性配置：


struct RichTextEditor: UIViewRepresentable {
    @Binding var attributedText: NSAttributedString
    var isEditable: Bool = true

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIView(context: Context) -> UITextView {
        let textView = UITextView()
        textView.delegate = context.coordinator
        textView.isEditable = isEditable
        textView.isScrollEnabled = true
        textView.font = .preferredFont(forTextStyle: .body)
        textView.textContainerInset = UIEdgeInsets(top: 12, left: 8, bottom: 12, right: 8)
        textView.backgroundColor = .clear
        textView.allowsEditingTextAttributes = true
        return textView
    }

    func updateUIView(_ uiView: UITextView, context: Context) {
        if uiView.attributedText != attributedText {
            let selectedRange = uiView.selectedRange
            uiView.attributedText = attributedText
            uiView.selectedRange = selectedRange
        }
        uiView.isEditable = isEditable
    }

    class Coordinator: NSObject, UITextViewDelegate {
        var parent: RichTextEditor

        init(_ parent: RichTextEditor) {
            self.parent = parent
        }

        func textViewDidChange(_ textView: UITextView) {
            parent.attributedText = textView.attributedText
        }
    }
}

updateUIView 详解
updateUIView 在任何 SwiftUI 状态变化时都可能被调用。关键是幂等性和条件更新：


struct ProgressBar: UIViewRepresentable {
    var progress: Float
    var tintColor: UIColor

    func makeUIView(context: Context) -> UIProgressView {
        let progressView = UIProgressView(progressViewStyle: .bar)
        progressView.trackTintColor = .systemGray5
        return progressView
    }

    func updateUIView(_ uiView: UIProgressView, context: Context) {
        let animated = context.transaction.animation != nil

        if uiView.progress != progress {
            uiView.setProgress(progress, animated: animated)
        }

        if uiView.progressTintColor != tintColor {
            uiView.progressTintColor = tintColor
        }
    }
}
使用动画上下文
context.transaction 包含当前的动画信息：


struct AnimatedUIView: UIViewRepresentable {
    var isActive: Bool

    func makeUIView(context: Context) -> UIView {
        let view = UIView()
        view.layer.cornerRadius = 12
        return view
    }

    func updateUIView(_ uiView: UIView, context: Context) {
        let newColor: UIColor = isActive ? .systemBlue : .systemGray

        if let animation = context.transaction.animation {
            UIView.animate(withDuration: 0.3) {
                uiView.backgroundColor = newColor
                uiView.transform = self.isActive ? CGAffineTransform(scaleX: 1.1, y: 1.1) : .identity
            }
        } else {
            uiView.backgroundColor = newColor
            uiView.transform = isActive ? CGAffineTransform(scaleX: 1.1, y: 1.1) : .identity
        }
    }
}


dismantleUIView
静态方法 dismantleUIView 在视图被移除时调用，用于清理资源：


struct VideoPlayerView: UIViewRepresentable {
    let url: URL

    func makeUIView(context: Context) -> PlayerUIView {
        PlayerUIView(url: url)
    }

    func updateUIView(_ uiView: PlayerUIView, context: Context) {
        uiView.updateURL(url)
    }

    static func dismantleUIView(_ uiView: PlayerUIView, coordinator: ()) {
        uiView.player?.pause()
        uiView.player?.replaceCurrentItem(with: nil)
        uiView.playerLayer?.removeFromSuperlayer()
        NotificationCenter.default.removeObserver(uiView)
    }
}


尺寸策略
UIKit 视图在 SwiftUI 中的尺寸行为取决于 intrinsicContentSize：


struct SizedUIView: UIViewRepresentable {
    let text: String

    func makeUIView(context: Context) -> IntrinsicLabel {
        let label = IntrinsicLabel()
        label.numberOfLines = 0
        label.setContentHuggingPriority(.required, for: .vertical)
        label.setContentCompressionResistancePriority(.required, for: .vertical)
        return label
    }

    func updateUIView(_ uiView: IntrinsicLabel, context: Context) {
        uiView.text = text
        uiView.invalidateIntrinsicContentSize()
    }
}

class IntrinsicLabel: UILabel {
    override var intrinsicContentSize: CGSize {
        let size = super.intrinsicContentSize
        return CGSize(width: size.width + 24, height: size.height + 16)
    }
}

复杂场景：MKMapView 桥接
import MapKit

struct MapViewBridge: UIViewRepresentable {
    @Binding var region: MKCoordinateRegion
    var annotations: [MapAnnotation]
    var onAnnotationTap: ((MapAnnotation) -> Void)?

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIView(context: Context) -> MKMapView {
        let mapView = MKMapView()
        mapView.delegate = context.coordinator
        mapView.showsUserLocation = true
        return mapView
    }

    func updateUIView(_ uiView: MKMapView, context: Context) {
        if uiView.region.center.latitude != region.center.latitude ||
           uiView.region.center.longitude != region.center.longitude {
            let animated = context.transaction.animation != nil
            uiView.setRegion(region, animated: animated)
        }

        let currentIds = Set(uiView.annotations.compactMap { ($0 as? CustomAnnotation)?.id })
        let newIds = Set(annotations.map(\.id))

        if currentIds != newIds {
            uiView.removeAnnotations(uiView.annotations)
            let mkAnnotations = annotations.map { annotation in
                let mk = CustomAnnotation(annotation: annotation)
                return mk as MKAnnotation
            }
            uiView.addAnnotations(mkAnnotations)
        }
    }

    class Coordinator: NSObject, MKMapViewDelegate {
        var parent: MapViewBridge

        init(_ parent: MapViewBridge) {
            self.parent = parent
        }

        func mapView(_ mapView: MKMapView, regionDidChangeAnimated animated: Bool) {
            parent.region = mapView.region
        }

        func mapView(_ mapView: MKMapView, didSelect annotation: MKAnnotation) {
            if let custom = annotation as? CustomAnnotation {
                parent.onAnnotationTap?(custom.data)
            }
        }
    }
}

struct MapAnnotation: Identifiable {
    let id: String
    let title: String
    let coordinate: CLLocationCoordinate2D
}

class CustomAnnotation: NSObject, MKAnnotation {
    let data: MapAnnotation
    var coordinate: CLLocationCoordinate2D { data.coordinate }
    var title: String? { data.title }
    var id: String { data.id }

    init(annotation: MapAnnotation) {
        self.data = annotation
    }
}


提示： 桥接复杂 UIKit 视图时，在 updateUIView 中使用 diff 逻辑（对比新旧数据）可以避免不必要的全量更新，显著提升性能。






小结
    

  • makeUIView 只调用一次，负责初始化和一次性配置
    • 

  • updateUIView 可能频繁调用，必须是幂等的，应检查值变化后再更新
    • 

  • dismantleUIView 是静态方法，用于释放资源、移除观察者
    • 

  • context.transaction.animation 判断是否有动画上下文，据此决定 UIKit 动画
    • 

  • UIView 的 intrinsicContentSize 决定在 SwiftUI 中的默认尺寸
    • 

  • 复杂桥接场景（如地图）需要在 updateUIView 中实现增量更新逻辑
    • 

  • Coordinator 的 parent 属性在每次 updateUIView 前会自动更新
    •
