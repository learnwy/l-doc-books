自定义属性包装器

概述

属性包装器（Property Wrapper）是 Swift 中封装属性访问逻辑的强大机制。通过 @propertyWrapper 标注，开发者可以将重复的存储、验证、变换逻辑抽象为可复用的组件。属性包装器的核心是 wrappedValue 和可选的 projectedValue，二者共同构成了属性包装器的对外接口。

wrappedValue 与 projectedValue
每个属性包装器必须提供 wrappedValue 属性，可选提供 projectedValue：


@propertyWrapper
struct Clamped<Value: Comparable> {
    private var value: Value
    let range: ClosedRange<Value>

    var wrappedValue: Value {
        get { value }
        set { value = min(max(newValue, range.lowerBound), range.upperBound) }
    }

    var projectedValue: ClosedRange<Value> {
        range
    }

    init(wrappedValue: Value, _ range: ClosedRange<Value>) {
        self.range = range
        self.value = min(max(wrappedValue, range.lowerBound), range.upperBound)
    }
}
使用方式：


struct GameCharacter {
    @Clamped(0...100) var health: Int = 100
    @Clamped(0...999) var experience: Int = 0
}

var character = GameCharacter()
character.health = 150
print(character.health)       // 100（被 clamp 到上限）
print(character.$health)      // 0...100（projectedValue）

初始化模式
属性包装器支持多种初始化模式，灵活适应不同使用场景：


@propertyWrapper
struct UserDefault<Value> {
    let key: String
    let defaultValue: Value
    let container: UserDefaults

    var wrappedValue: Value {
        get { container.object(forKey: key) as? Value ?? defaultValue }
        set { container.set(newValue, forKey: key) }
    }

    init(wrappedValue: Value, _ key: String, container: UserDefaults = .standard) {
        self.key = key
        self.defaultValue = wrappedValue
        self.container = container
    }
}
不同的初始化语法：


struct AppSettings {
    @UserDefault("username") var username: String = "Guest"

    @UserDefault("launchCount") var launchCount: Int = 0

    @UserDefault("theme", container: .init(suiteName: "group.app")!)
    var theme: String = "system"
}


实现 @Throttled 包装器
@Throttled 包装器限制属性的更新频率，忽略过于频繁的赋值：


import Foundation

@propertyWrapper
struct Throttled<Value> {
    private var value: Value
    private var lastUpdate: Date = .distantPast
    let interval: TimeInterval

    var wrappedValue: Value {
        get { value }
        set {
            let now = Date()
            guard now.timeIntervalSince(lastUpdate) >= interval else { return }
            value = newValue
            lastUpdate = now
        }
    }

    var projectedValue: Date {
        lastUpdate
    }

    init(wrappedValue: Value, interval: TimeInterval) {
        self.value = wrappedValue
        self.interval = interval
    }
}
struct SensorReading {
    @Throttled(interval: 0.5) var temperature: Double = 0.0
    @Throttled(interval: 1.0) var humidity: Double = 0.0
}

var sensor = SensorReading()
sensor.temperature = 25.0  // ✅ 更新
sensor.temperature = 25.1  // ❌ 忽略（间隔不足 0.5 秒）


组合属性包装器
Swift 支持在同一属性上叠加多个属性包装器，形成包装链：


@propertyWrapper
struct Trimmed {
    private var value: String

    var wrappedValue: String {
        get { value }
        set { value = newValue.trimmingCharacters(in: .whitespacesAndNewlines) }
    }

    init(wrappedValue: String) {
        self.value = wrappedValue.trimmingCharacters(in: .whitespacesAndNewlines)
    }
}

@propertyWrapper
struct MaxLength {
    private var value: String
    let limit: Int

    var wrappedValue: String {
        get { value }
        set { value = String(newValue.prefix(limit)) }
    }

    init(wrappedValue: String, _ limit: Int) {
        self.limit = limit
        self.value = String(wrappedValue.prefix(limit))
    }
}
组合使用：


struct FormField {
    @Trimmed @MaxLength(50) var username: String = ""
    @Trimmed @MaxLength(200) var bio: String = ""
}

var field = FormField()
field.username = "   Swift Developer   "
print(field.username) // "Swift Developer"
包装器的执行顺序：




提示： 组合包装器时，外层先于内层执行 set，内层先于外层执行 get。注意执行顺序对结果的影响——先 trim 再截断和先截断再 trim 可能得到不同结果。






延迟初始化包装器
利用属性包装器实现线程安全的延迟初始化：


@propertyWrapper
struct LazyInit<Value> {
    private var storage: Value?
    private let factory: () -> Value

    var wrappedValue: Value {
        mutating get {
            if let existing = storage {
                return existing
            }
            let value = factory()
            storage = value
            return value
        }
        set {
            storage = newValue
        }
    }

    var projectedValue: Bool {
        storage != nil
    }

    init(_ factory: @autoclosure @escaping () -> Value) {
        self.factory = factory
    }
}
struct DataManager {
    @LazyInit(ExpensiveResource())
    var resource: ExpensiveResource

    func process() {
        print($resource) // false — 尚未初始化
        _ = resource      // 触发初始化
        print($resource) // true — 已初始化
    }
}


实际应用模式
属性包装器在实际项目中的常见应用：


@propertyWrapper
struct Validated<Value> {
    private var value: Value
    let validator: (Value) -> Bool
    let fallback: Value

    var wrappedValue: Value {
        get { value }
        set { value = validator(newValue) ? newValue : fallback }
    }

    init(wrappedValue: Value, validator: @escaping (Value) -> Bool, fallback: Value) {
        self.validator = validator
        self.fallback = fallback
        self.value = validator(wrappedValue) ? wrappedValue : fallback
    }
}

@propertyWrapper
struct Atomic<Value> {
    private let lock = NSLock()
    private var storage: Value

    var wrappedValue: Value {
        get {
            lock.lock()
            defer { lock.unlock() }
            return storage
        }
        set {
            lock.lock()
            defer { lock.unlock() }
            storage = newValue
        }
    }

    init(wrappedValue: Value) {
        self.storage = wrappedValue
    }
}

小结
    

  • 属性包装器通过 wrappedValue 封装属性的 get/set 逻辑，通过 projectedValue（$ 语法）暴露附加信息
    • 

  • 支持多种初始化模式：带默认值、带参数、无默认值
    • 

  • 组合包装器按从外到内的顺序应用 set，从内到外的顺序应用 get
    • 

  • 实用包装器包括 @Clamped（范围限制）、@UserDefault（持久化）、@Throttled（节流）等
    • 

  • 延迟初始化包装器可利用 @autoclosure 延迟创建开销大的资源
    • 

  • 属性包装器是 SwiftUI（@State、@Binding、@Environment）的基础设施
