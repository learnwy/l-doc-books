附加宏

概述

附加宏（Attached Macros）是以 @MacroName 形式附加到现有声明上的宏，可以为类型添加成员、生成访问器、创建扩展等。Swift 标准库的 @Observable 就是附加宏的典型代表。附加宏的强大之处在于它能读取被附加声明的完整结构，并据此生成相关代码。

@Observable 宏解析
@Observable 是 Swift Observation 框架中的核心宏，它将普通类转换为可观察对象。理解它的实现原理有助于掌握附加宏的设计模式：


@Observable
class UserSettings {
    var username = "Swift"
    var darkMode = false
}
展开后的等效代码：


class UserSettings {
    @ObservationTracked
    var username = "Swift"
    @ObservationTracked
    var darkMode = false

    @ObservationIgnored
    private let _$observationRegistrar = ObservationRegistrar()

    internal nonisolated func access<Member>(
        keyPath: KeyPath<UserSettings, Member>
    ) {
        _$observationRegistrar.access(self, keyPath: keyPath)
    }

    internal nonisolated func withMutation<Member, MutationResult>(
        keyPath: KeyPath<UserSettings, Member>,
        _ mutation: () throws -> MutationResult
    ) rethrows -> MutationResult {
        try _$observationRegistrar.withMutation(of: self, keyPath: keyPath, mutation)
    }
}

extension UserSettings: Observable {}

实现 MemberMacro
MemberMacro 可以向类型添加新的成员。以下是一个自动生成 init 方法的宏：


@attached(member, names: named(init))
public macro AutoInit() = #externalMacro(module: "MyMacrosMacros", type: "AutoInitMacro")
实现：


import SwiftSyntax
import SwiftSyntaxMacros

public struct AutoInitMacro: MemberMacro {
    public static func expansion(
        of node: AttributeSyntax,
        providingMembersOf declaration: some DeclGroupSyntax,
        conformingTo protocols: [TypeSyntax],
        in context: some MacroExpansionContext
    ) throws -> [DeclSyntax] {
        let properties = declaration.memberBlock.members
            .compactMap { $0.decl.as(VariableDeclSyntax.self) }
            .filter { $0.bindings.first?.accessorBlock == nil }
            .compactMap { variable -> (String, TypeSyntax)? in
                guard let binding = variable.bindings.first,
                      let name = binding.pattern.as(IdentifierPatternSyntax.self)?.identifier.text,
                      let type = binding.typeAnnotation?.type
                else { return nil }
                return (name, type)
            }

        let parameters = properties.map { "\($0.0): \($0.1)" }.joined(separator: ", ")
        let assignments = properties.map { "self.\($0.0) = \($0.0)" }.joined(separator: "\n        ")

        return [
            """
            init(\(raw: parameters)) {
                \(raw: assignments)
            }
            """
        ]
    }
}
使用：


@AutoInit
struct User {
    var name: String
    var age: Int
    var email: String
    // 自动生成:
    // init(name: String, age: Int, email: String) {
    //     self.name = name
    //     self.age = age
    //     self.email = email
    // }
}


实现 AccessorMacro
AccessorMacro 为属性添加 getter、setter 等访问器。以下是一个 @Logged 宏示例：


@attached(accessor, names: named(willSet))
public macro Logged() = #externalMacro(module: "MyMacrosMacros", type: "LoggedMacro")
public struct LoggedMacro: AccessorMacro {
    public static func expansion(
        of node: AttributeSyntax,
        providingAccessorsOf declaration: some DeclSyntaxProtocol,
        in context: some MacroExpansionContext
    ) throws -> [AccessorDeclSyntax] {
        guard let varDecl = declaration.as(VariableDeclSyntax.self),
              let binding = varDecl.bindings.first,
              let name = binding.pattern.as(IdentifierPatternSyntax.self)?.identifier.text
        else { return [] }

        return [
            """
            willSet {
                print("\\(Self.self).\(raw: name) 将从 \\(self.\(raw: name)) 变为 \\(newValue)")
            }
            """
        ]
    }
}


实现 ExtensionMacro
ExtensionMacro 为类型生成扩展，通常用来添加协议遵循：


@attached(extension, conformances: Equatable, Hashable, names: named(==), named(hash))
public macro AutoEquatable() = #externalMacro(module: "MyMacrosMacros", type: "AutoEquatableMacro")
public struct AutoEquatableMacro: ExtensionMacro {
    public static func expansion(
        of node: AttributeSyntax,
        attachedTo declaration: some DeclGroupSyntax,
        providingExtensionsOf type: some TypeSyntaxProtocol,
        conformingTo protocols: [TypeSyntax],
        in context: some MacroExpansionContext
    ) throws -> [ExtensionDeclSyntax] {
        let extensionDecl: DeclSyntax = """
        extension \(type.trimmed): Equatable, Hashable {}
        """
        guard let ext = extensionDecl.as(ExtensionDeclSyntax.self) else { return [] }
        return [ext]
    }
}


实现 PeerMacro
PeerMacro 在与被附加声明同级的作用域生成新声明。常见用例是为异步函数生成回调版本：


@attached(peer, names: overloaded)
public macro AddCompletionHandler() =
    #externalMacro(module: "MyMacrosMacros", type: "AddCompletionHandlerMacro")
@AddCompletionHandler
func fetchUser(id: Int) async throws -> User {
    // ...
}

// PeerMacro 生成:
// func fetchUser(id: Int, completion: @escaping (Result<User, Error>) -> Void) {
//     Task {
//         do {
//             let result = try await fetchUser(id: id)
//             completion(.success(result))
//         } catch {
//             completion(.failure(error))
//         }
//     }
// }


自定义附加宏步骤
构建一个完整的附加宏需要遵循以下步骤：


注册宏到编译器插件入口：


import SwiftCompilerPlugin
import SwiftSyntaxMacros

@main
struct MyMacrosPlugin: CompilerPlugin {
    let providingMacros: [Macro.Type] = [
        AutoInitMacro.self,
        LoggedMacro.self,
        AutoEquatableMacro.self,
    ]
}


提示： 设计宏时优先考虑组合多个角色。例如 @Observable 同时使用 member + memberAttribute + extension 三种角色，比拆分成三个独立的宏更符合用户心智模型。






小结
    

  • 附加宏以 @MacroName 形式附加到声明上，可读取被附加声明的完整结构
    • 

  • MemberMacro 向类型添加新成员（属性、方法、初始化器）
    • 

  • AccessorMacro 为属性添加 getter/setter/willSet/didSet 等访问器
    • 

  • ExtensionMacro 为类型生成扩展和协议遵循
    • 

  • PeerMacro 在同级作用域生成新声明
    • 

  • 一个宏可以组合多种角色，如 @Observable 同时使用三种角色
    • 

  • 所有宏实现需要在 CompilerPlugin 的 providingMacros 中注册
