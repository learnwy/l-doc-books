宏测试

概述

宏的正确性至关重要——一个有缺陷的宏会在所有使用点生成错误代码。Swift 提供了 SwiftSyntaxMacrosTestSupport 框架，通过 assertMacroExpansion 函数让开发者以声明式的方式验证宏展开结果。完善的测试策略包括正常展开测试、诊断信息测试和边界条件测试。

assertMacroExpansion
assertMacroExpansion 是宏测试的核心函数。它接收原始源代码、期望的展开结果和宏实现类型，自动执行展开并比较结果：


import SwiftSyntaxMacrosTestSupport
import XCTest

final class StringifyMacroTests: XCTestCase {
    func testBasicExpansion() {
        assertMacroExpansion(
            """
            #stringify(2 + 3)
            """,
            expandedSource: """
            (2 + 3, "2 + 3")
            """,
            macros: ["stringify": StringifyMacro.self]
        )
    }

    func testComplexExpression() {
        assertMacroExpansion(
            """
            #stringify(a.method(arg: value))
            """,
            expandedSource: """
            (a.method(arg: value), "a.method(arg: value)")
            """,
            macros: ["stringify": StringifyMacro.self]
        )
    }
}

测试附加宏
附加宏的测试需要包含完整的声明上下文：


final class AutoInitMacroTests: XCTestCase {
    let testMacros: [String: Macro.Type] = [
        "AutoInit": AutoInitMacro.self,
    ]

    func testStructInit() {
        assertMacroExpansion(
            """
            @AutoInit
            struct User {
                var name: String
                var age: Int
            }
            """,
            expandedSource: """
            struct User {
                var name: String
                var age: Int

                init(name: String, age: Int) {
                    self.name = name
                    self.age = age
                }
            }
            """,
            macros: testMacros
        )
    }

    func testEmptyStruct() {
        assertMacroExpansion(
            """
            @AutoInit
            struct Empty {
            }
            """,
            expandedSource: """
            struct Empty {

                init() {
                }
            }
            """,
            macros: testMacros
        )
    }
}


诊断信息测试
验证宏在错误场景下发出正确的诊断信息同样重要：


func testInvalidArgument() {
    assertMacroExpansion(
        """
        #URL(someVariable)
        """,
        expandedSource: """
        #URL(someVariable)
        """,
        diagnostics: [
            DiagnosticSpec(
                message: "此宏要求传入字符串字面量",
                line: 1,
                column: 1,
                severity: .error
            )
        ],
        macros: ["URL": URLMacro.self]
    )
}

func testInvalidURL() {
    assertMacroExpansion(
        """
        #URL("not valid ://")
        """,
        expandedSource: """
        #URL("not valid ://")
        """,
        diagnostics: [
            DiagnosticSpec(
                message: "无效的 URL: not valid ://",
                line: 1,
                column: 6,
                severity: .error
            )
        ],
        macros: ["URL": URLMacro.self]
    )
}
DiagnosticSpec 的可选参数允许精确匹配诊断位置和 Fix-It：


DiagnosticSpec(
    message: "诊断消息",
    line: 1,
    column: 5,
    severity: .warning,
    fixIts: [
        FixItSpec(message: "使用新 API")
    ]
)


Fixture 测试
对于复杂宏，基于 fixture 文件的测试可以更好地组织测试数据：


final class ComplexMacroFixtureTests: XCTestCase {
    let testMacros: [String: Macro.Type] = [
        "AutoCodable": AutoCodableMacro.self,
    ]

    func testFromFixture(_ name: String) throws {
        let inputURL = Bundle.module.url(
            forResource: name,
            withExtension: "swift",
            subdirectory: "Fixtures/Input"
        )!
        let expectedURL = Bundle.module.url(
            forResource: name,
            withExtension: "swift",
            subdirectory: "Fixtures/Expected"
        )!

        let input = try String(contentsOf: inputURL)
        let expected = try String(contentsOf: expectedURL)

        assertMacroExpansion(
            input,
            expandedSource: expected,
            macros: testMacros
        )
    }

    func testSimpleModel() throws {
        try testFromFixture("SimpleModel")
    }

    func testNestedModel() throws {
        try testFromFixture("NestedModel")
    }

    func testOptionalFields() throws {
        try testFromFixture("OptionalFields")
    }
}
Fixture 文件的组织结构：




提示： Fixture 测试适合展开结果较长的宏。将输入和期望输出分别存放在文件中，避免测试方法中出现过长的字符串字面量。在 Package.swift 中需要将 fixture 目录声明为 resources。






调试宏
使用断点调试
宏运行在独立进程中，但可以通过以下方式调试：


public struct MyMacro: ExpressionMacro {
    public static func expansion(
        of node: some FreestandingMacroExpansionSyntax,
        in context: some MacroExpansionContext
    ) throws -> ExprSyntax {
        #if DEBUG
        print("Input node: \(node.description)")
        print("Arguments: \(node.arguments.map { $0.description })")
        #endif

        // 宏实现...
        return "\(node.arguments.first!.expression)"
    }
}
在测试中调试宏最为方便，因为宏实现在测试进程内直接执行：


func testDebugExpansion() {
    // 在宏实现的 expansion 方法中设置断点
    // 测试执行时会命中断点
    assertMacroExpansion(
        "#myMacro(42)",
        expandedSource: "42",
        macros: ["myMacro": MyMacro.self]
    )
}
LLDB 调试技巧
在断点命中时，可以使用 LLDB 检查语法树：


(lldb) po node.description
(lldb) po node.arguments.first?.expression
(lldb) po declaration.memberBlock.members.map { $0.decl.description }


CI 集成
宏包作为标准的 Swift Package，可以无缝集成到 CI 流程中：


// Package.swift 中的测试目标配置
.testTarget(
    name: "MyMacrosTests",
    dependencies: [
        "MyMacrosMacros",
        .product(name: "SwiftSyntaxMacrosTestSupport", package: "swift-syntax"),
    ]
)
CI 环境中的测试命令：


// 运行所有宏测试
// swift test --filter MyMacrosTests

// 仅运行特定测试
// swift test --filter MyMacrosTests.StringifyMacroTests

// 生成代码覆盖率报告
// swift test --enable-code-coverage

小结
    

  • assertMacroExpansion 是宏测试的核心，接收源代码和期望展开结果进行比较
    • 

  • 测试应覆盖正常展开、诊断错误和边界条件三个维度
    • 

  • DiagnosticSpec 用于精确验证错误消息、位置和严重级别
    • 

  • Fixture 测试适合展开结果较长的复杂宏，将输入输出分别存储在文件中
    • 

  • 在测试进程中调试宏最方便，可直接在 expansion 方法中设置断点
    • 

  • 宏包作为标准 Swift Package 可无缝集成到 CI，建议测试多版本 swift-syntax 兼容性
    •
