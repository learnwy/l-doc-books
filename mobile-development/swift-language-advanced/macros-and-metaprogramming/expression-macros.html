表达式宏

概述

表达式宏（Expression Macros）是独立宏（Freestanding Macros）中最常用的类型，它以 #macroName 的语法出现在表达式位置，在编译期生成一个返回值。标准库中的 #stringify、#URL 等都是典型的表达式宏。表达式宏通过实现 ExpressionMacro 协议来定义展开逻辑。

ExpressionMacro 协议
所有表达式宏都必须遵循 ExpressionMacro 协议。该协议只有一个静态方法 expansion(of:in:)：


import SwiftSyntax
import SwiftSyntaxMacros

public protocol ExpressionMacro: FreestandingMacro {
    static func expansion(
        of node: some FreestandingMacroExpansionSyntax,
        in context: some MacroExpansionContext
    ) throws -> ExprSyntax
}

实现 #stringify 宏
#stringify 是 Swift 宏的经典入门示例。它接收一个表达式，返回一个元组，包含表达式的值和源代码字符串：


// 宏声明
@freestanding(expression)
public macro stringify<T>(_ value: T) -> (T, String) =
    #externalMacro(module: "MyMacrosMacros", type: "StringifyMacro")

// 使用
let (result, code) = #stringify(2 + 3)
print(result) // 5
print(code)   // "2 + 3"
宏实现：


public struct StringifyMacro: ExpressionMacro {
    public static func expansion(
        of node: some FreestandingMacroExpansionSyntax,
        in context: some MacroExpansionContext
    ) -> ExprSyntax {
        guard let argument = node.arguments.first?.expression else {
            fatalError("compiler bug: stringify requires an argument")
        }
        return "(\(argument), \(literal: argument.description))"
    }
}


提示： \(literal:) 插值会自动将字符串转义为字面量形式，确保特殊字符被正确处理。这是 SwiftSyntax 的 StringInterpolation 扩展功能。






实现 #URL 宏
#URL 宏在编译期验证 URL 字符串的合法性，避免运行时崩溃：


@freestanding(expression)
public macro URL(_ string: String) -> URL =
    #externalMacro(module: "MyMacrosMacros", type: "URLMacro")

// 使用
let url = #URL("https://swift.org")           // ✅ 编译通过
// let bad = #URL("not a valid url ://")       // ❌ 编译报错
宏实现中需要在编译期解析参数并验证：


import SwiftDiagnostics

public struct URLMacro: ExpressionMacro {
    public static func expansion(
        of node: some FreestandingMacroExpansionSyntax,
        in context: some MacroExpansionContext
    ) throws -> ExprSyntax {
        guard let argument = node.arguments.first?.expression,
              let segments = argument.as(StringLiteralExprSyntax.self)?.segments,
              segments.count == 1,
              case .stringSegment(let literalSegment)? = segments.first
        else {
            throw MacroError.requiresStaticStringLiteral
        }

        let urlString = literalSegment.content.text

        guard let _ = URL(string: urlString) else {
            let diagnostic = Diagnostic(
                node: argument,
                message: URLDiagnostic.invalidURL(urlString)
            )
            context.diagnose(diagnostic)
            return "URL(string: \(argument))!"
        }

        return "URL(string: \(argument))!"
    }
}


宏参数与泛型参数
表达式宏支持接收多个参数和泛型参数：


@freestanding(expression)
public macro clamp<T: Comparable>(_ value: T, min: T, max: T) -> T =
    #externalMacro(module: "MyMacrosMacros", type: "ClampMacro")

let clamped = #clamp(150, min: 0, max: 100) // 100
解析带标签的参数：


public struct ClampMacro: ExpressionMacro {
    public static func expansion(
        of node: some FreestandingMacroExpansionSyntax,
        in context: some MacroExpansionContext
    ) throws -> ExprSyntax {
        let arguments = node.arguments
        guard arguments.count == 3 else {
            throw MacroError.invalidArgumentCount
        }

        let value = arguments[arguments.startIndex].expression
        let minVal = arguments[arguments.index(after: arguments.startIndex)].expression
        let maxVal = arguments[arguments.index(arguments.startIndex, offsetBy: 2)].expression

        return "Swift.min(Swift.max(\(value), \(minVal)), \(maxVal))"
    }
}

诊断与错误报告
宏可以通过 MacroExpansionContext 发出编译期诊断信息，包括错误、警告和提示：


import SwiftDiagnostics

enum MyMacroDiagnostic: String, DiagnosticMessage {
    case requiresStringLiteral
    case deprecatedUsage

    var severity: DiagnosticSeverity {
        switch self {
        case .requiresStringLiteral: return .error
        case .deprecatedUsage: return .warning
        }
    }

    var message: String {
        switch self {
        case .requiresStringLiteral:
            return "此宏要求传入字符串字面量"
        case .deprecatedUsage:
            return "此用法已弃用，请使用新版 API"
        }
    }

    var diagnosticID: MessageID {
        MessageID(domain: "MyMacros", id: rawValue)
    }
}
在宏实现中使用诊断：


public struct SafeUnwrapMacro: ExpressionMacro {
    public static func expansion(
        of node: some FreestandingMacroExpansionSyntax,
        in context: some MacroExpansionContext
    ) throws -> ExprSyntax {
        guard let argument = node.arguments.first?.expression else {
            let diagnostic = Diagnostic(
                node: Syntax(node),
                message: MyMacroDiagnostic.requiresStringLiteral
            )
            context.diagnose(diagnostic)
            return "Optional<Never>.none!"
        }

        let tempName = context.makeUniqueName("unwrapped")

        return """
        { [_tmp = \(argument)] in
            guard let \(tempName) = _tmp else {
                preconditionFailure("Unexpected nil value")
            }
            return \(tempName)
        }()
        """
    }
}

还可以附加 Fix-It 建议：


let fixIt = FixIt(
    message: MyFixItMessage.useNewAPI,
    changes: [
        .replace(
            oldNode: Syntax(node),
            newNode: Syntax("newAPI(\(argument))" as ExprSyntax)
        )
    ]
)

context.diagnose(
    Diagnostic(node: Syntax(node), message: MyMacroDiagnostic.deprecatedUsage, fixIts: [fixIt])
)


提示： 良好的诊断信息是宏可用性的关键。始终为错误场景提供清晰的消息和 Fix-It 建议，让使用者能快速理解并修复问题。






声明宏
除了表达式宏，独立宏还包括声明宏（Declaration Macros），它们生成新的声明而非表达式：


@freestanding(declaration, names: arbitrary)
public macro generateEnumCases(_ cases: String...) =
    #externalMacro(module: "MyMacrosMacros", type: "EnumCasesMacro")

// 使用
enum Direction {
    #generateEnumCases("north", "south", "east", "west")
    // 展开为:
    // case north
    // case south
    // case east
    // case west
}
声明宏实现 DeclarationMacro 协议，返回 [DeclSyntax] 数组：


public struct EnumCasesMacro: DeclarationMacro {
    public static func expansion(
        of node: some FreestandingMacroExpansionSyntax,
        in context: some MacroExpansionContext
    ) throws -> [DeclSyntax] {
        return node.arguments.compactMap { argument in
            guard let stringLiteral = argument.expression.as(StringLiteralExprSyntax.self),
                  let segment = stringLiteral.segments.first,
                  case .stringSegment(let content) = segment
            else { return nil }
            return "case \(raw: content.content.text)"
        }
    }
}


小结
    

  • 表达式宏通过 #macroName 语法在表达式位置生成代码，遵循 ExpressionMacro 协议
    • 

  • expansion(of:in:) 方法接收语法树节点和展开上下文，返回 ExprSyntax
    • 

  • 宏参数通过 node.arguments 访问，支持多参数和泛型参数
    • 

  • 通过 MacroExpansionContext 可以发出 error/warning/note 诊断，并附加 Fix-It
    • 

  • \(literal:) 插值用于安全地将值转换为字面量语法
    • 

  • 声明宏实现 DeclarationMacro 协议，返回声明数组，可生成任意声明
    •
