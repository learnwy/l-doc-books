Result Builders 结果构建器

概述

Result Builders（结果构建器）是 Swift 5.4 引入的元编程特性（SE-0289），允许开发者通过声明式语法构建复杂的数据结构。SwiftUI 的 @ViewBuilder 是最知名的 Result Builder 应用。通过自定义 Result Builder，可以创建类型安全的领域特定语言（DSL），将命令式代码转换为声明式语法。

基本原理
@resultBuilder 标记的类型需要实现一组静态 build 方法，编译器会将声明式代码块自动转换为对这些方法的调用：


@resultBuilder
struct ArrayBuilder<Element> {
    static func buildBlock(_ components: Element...) -> [Element] {
        components
    }
}

@ArrayBuilder<String>
func greetings() -> [String] {
    "Hello"
    "World"
    "Swift"
}
编译器将 greetings() 的函数体转换为：


func greetings() -> [String] {
    ArrayBuilder<String>.buildBlock("Hello", "World", "Swift")
}


buildBlock：基础组合
buildBlock 是唯一必须实现的方法，它接收闭包中每个语句的结果并组合成最终值：


@resultBuilder
struct HTMLBuilder {
    static func buildBlock(_ components: HTMLNode...) -> [HTMLNode] {
        components
    }
}

protocol HTMLNode {
    func render() -> String
}

struct Text: HTMLNode {
    let content: String
    func render() -> String { content }
}

struct Tag: HTMLNode {
    let name: String
    let children: [HTMLNode]

    func render() -> String {
        let inner = children.map { $0.render() }.joined()
        return "<\(name)>\(inner)</\(name)>"
    }
}

func div(@HTMLBuilder content: () -> [HTMLNode]) -> Tag {
    Tag(name: "div", children: content())
}

func p(@HTMLBuilder content: () -> [HTMLNode]) -> Tag {
    Tag(name: "p", children: content())
}
使用自定义 DSL：


let html = div {
    p {
        Text(content: "第一段")
    }
    p {
        Text(content: "第二段")
    }
}
print(html.render())


buildOptional：支持 if 语句
实现 buildOptional 可以在 Result Builder 中使用不带 else 的 if 语句：


@resultBuilder
struct ArrayBuilder<Element> {
    static func buildBlock(_ components: [Element]...) -> [Element] {
        components.flatMap { $0 }
    }

    static func buildExpression(_ expression: Element) -> [Element] {
        [expression]
    }

    static func buildOptional(_ component: [Element]?) -> [Element] {
        component ?? []
    }
}
@ArrayBuilder<String>
func conditionalGreetings(includeEmoji: Bool) -> [String] {
    "Hello"
    if includeEmoji {
        "👋"
    }
    "World"
}


buildEither：支持 if-else 和 switch
buildEither(first:) 和 buildEither(second:) 支持条件分支：


extension ArrayBuilder {
    static func buildEither(first component: [Element]) -> [Element] {
        component
    }

    static func buildEither(second component: [Element]) -> [Element] {
        component
    }
}
@ArrayBuilder<String>
func branchGreeting(isEnglish: Bool) -> [String] {
    if isEnglish {
        "Hello"
        "World"
    } else {
        "你好"
        "世界"
    }
}

buildArray：支持 for-in 循环
buildArray 方法使 Result Builder 支持循环：


extension ArrayBuilder {
    static func buildArray(_ components: [[Element]]) -> [Element] {
        components.flatMap { $0 }
    }
}

@ArrayBuilder<String>
func numberedList(items: [String]) -> [String] {
    for (index, item) in items.enumerated() {
        "\(index + 1). \(item)"
    }
}

实战：SQL 查询构建器
struct SQLQuery {
    var parts: [String]
    var sql: String { parts.joined(separator: " ") }
}

@resultBuilder
struct QueryBuilder {
    static func buildBlock(_ components: String...) -> SQLQuery {
        SQLQuery(parts: components)
    }

    static func buildOptional(_ component: SQLQuery?) -> String {
        component?.sql ?? ""
    }

    static func buildEither(first component: String) -> String {
        component
    }

    static func buildEither(second component: String) -> String {
        component
    }
}

func query(@QueryBuilder _ build: () -> SQLQuery) -> SQLQuery {
    build()
}

let searchQuery = query {
    "SELECT * FROM users"
    "WHERE active = true"
    "ORDER BY created_at DESC"
    "LIMIT 10"
}
print(searchQuery.sql)


提示： Result Builder 在编译时进行完整的类型检查，任何类型不匹配都会产生编译错误。这使得 DSL 在拥有灵活语法的同时保持类型安全。






小结
    

  • @resultBuilder 允许定义声明式 DSL，编译器自动将代码块转换为 build 方法调用
    • 

  • buildBlock 是唯一必须的方法，用于组合闭包中的多个语句
    • 

  • buildOptional 和 buildEither 分别支持 if 和 if-else/switch 分支
    • 

  • buildArray 支持 for-in 循环
    • 

  • SwiftUI 的 @ViewBuilder 是最广泛使用的 Result Builder 实现
    • 

  • 自定义 Result Builder 适合构建 HTML、SQL、配置文件等声明式 DSL，兼具灵活性和类型安全
