条件遵循

概述

条件遵循（Conditional Conformance）是 Swift 4.1 引入的特性（SE-0143），它允许泛型类型在其类型参数满足特定条件时自动遵循某个协议。这是 Swift 类型系统中最优雅的设计之一，标准库大量依赖此特性——例如 Array 只在其 Element 遵循 Equatable 时才自动遵循 Equatable。

基本语法
条件遵循使用 extension ... where ... 语法声明：


struct Wrapper<Value> {
    let value: Value
}

extension Wrapper: Equatable where Value: Equatable {
    static func == (lhs: Wrapper, rhs: Wrapper) -> Bool {
        lhs.value == rhs.value
    }
}

extension Wrapper: Hashable where Value: Hashable {
    func hash(into hasher: inout Hasher) {
        hasher.combine(value)
    }
}

extension Wrapper: Comparable where Value: Comparable {
    static func < (lhs: Wrapper, rhs: Wrapper) -> Bool {
        lhs.value < rhs.value
    }
}

let a = Wrapper(value: 42)
let b = Wrapper(value: 42)
print(a == b) // true

let c = Wrapper(value: "hello")
let set: Set = [Wrapper(value: "a"), Wrapper(value: "b")]
当 Value 不遵循 Equatable 时，Wrapper 也不遵循，编译器会报错：


struct NonEquatable {}
let x = Wrapper(value: NonEquatable())
// x == x  // ❌ 编译错误：Wrapper<NonEquatable> 不遵循 Equatable


级联遵循（Cascading Conformance）
条件遵循支持级联——当内层类型满足条件时，外层包装自动满足相同条件：


let arrays: [[Int]] = [[1, 2], [3, 4]]
let isEqual = arrays == [[1, 2], [3, 4]] // true

let optionalArrays: [Int]? = [1, 2, 3]
let anotherOptional: [Int]? = [1, 2, 3]
print(optionalArrays == anotherOptional) // true
这之所以可行，是因为：


每一层都通过条件遵循传递 Equatable。

自动合成（Synthesis）
Swift 编译器可以自动合成某些协议的实现，当与条件遵循结合时更加强大：


struct Container<T> {
    let items: [T]
    let label: String
}

extension Container: Equatable where T: Equatable {}
extension Container: Hashable where T: Hashable {}
extension Container: Codable where T: Codable {}
编译器自动生成 ==、hash(into:) 和 encode/decode 方法，无需手写任何实现。




提示： 自动合成要求所有存储属性都满足相应的协议。如果 Container 中有一个属性不是 Equatable，则 Equatable 的自动合成会失败。






条件遵循与协议继承
条件遵循可以应用于协议继承链：


protocol Serializable {
    func serialize() -> Data
}

protocol JSONSerializable: Serializable {
    func toJSON() -> [String: Any]
}

extension Array: Serializable where Element: Serializable {
    func serialize() -> Data {
        let combined = map { $0.serialize() }
        return combined.reduce(Data(), +)
    }
}

extension Array: JSONSerializable where Element: JSONSerializable {
    func toJSON() -> [String: Any] {
        ["items": map { $0.toJSON() }]
    }
}


自定义类型的条件遵循实践
enum Either<Left, Right> {
    case left(Left)
    case right(Right)
}

extension Either: Equatable where Left: Equatable, Right: Equatable {
    static func == (lhs: Either, rhs: Either) -> Bool {
        switch (lhs, rhs) {
        case let (.left(l1), .left(l2)):
            return l1 == l2
        case let (.right(r1), .right(r2)):
            return r1 == r2
        default:
            return false
        }
    }
}

extension Either: CustomStringConvertible
    where Left: CustomStringConvertible, Right: CustomStringConvertible {
    var description: String {
        switch self {
        case .left(let value): return "Left(\(value.description))"
        case .right(let value): return "Right(\(value.description))"
        }
    }
}

let value: Either<Int, String> = .left(42)
print(value) // Left(42)
print(Either.left(1) == Either.left(1)) // true


运行时检查条件遵循
使用条件转换（as?）可以在运行时检查条件遵循是否成立：


func tryEqual<T>(_ a: [T], _ b: [T]) -> Bool? {
    if let eqA = a as? any Equatable,
       let eqB = b as? any Equatable {
        return type(of: eqA) == type(of: eqB)
    }
    return nil
}

  • 级联遵循使嵌套泛型（如 [[Int]]）自动传递协议遵循

  • Swift 标准库中 Array、Optional、Dictionary 等类型广泛使用条件遵循
    •

    • 

  • 条件遵循适用于协议继承链，可以分层级添加能力
    •
