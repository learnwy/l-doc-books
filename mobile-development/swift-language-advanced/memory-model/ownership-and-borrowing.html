所有权与借用

概述

Swift 5.9 引入了所有权（Ownership）相关特性（SE-0366, SE-0377），包括 consuming、borrowing 参数修饰符和 ~Copyable（不可复制类型）。这些特性受 Rust 所有权模型启发，让 Swift 能够表达更精细的值生命周期管理，在避免不必要的拷贝的同时保证内存安全。

consuming 与 borrowing
Swift 函数参数默认使用隐式的所有权约定。新的显式修饰符让开发者表达精确意图：


struct LargeBuffer {
    var data: [UInt8]

    func process(borrowing input: LargeBuffer) -> Int {
        input.data.count
    }

    consuming func transformed() -> LargeBuffer {
        var copy = self
        copy.data.reverse()
        return copy
    }
}

func printInfo(borrowing value: String) {
    print(value)
}

func takeOwnership(consuming value: String) {
    print("Took: \(value)")
}

var name = "Swift"
printInfo(borrowing: name)       // name 仍然有效
takeOwnership(consuming: name)    // name 的所有权转移
// print(name) // ⚠️ 使用 consuming 后，name 可能不再有效


~Copyable：不可复制类型
~Copyable（也称 noncopyable 或 move-only 类型）表示该类型的值不能被隐式复制：


struct FileHandle: ~Copyable {
    private let fd: Int32

    init(path: String) throws {
        fd = open(path, O_RDONLY)
        guard fd >= 0 else { throw FileError.cannotOpen }
    }

    func read(count: Int) -> Data {
        var buffer = [UInt8](repeating: 0, count: count)
        _ = Darwin.read(fd, &buffer, count)
        return Data(buffer)
    }

    deinit {
        close(fd)
    }
}
不可复制类型确保资源只有唯一拥有者：


func processFile() throws {
    let handle = try FileHandle(path: "/tmp/data.txt")
    let data = handle.read(count: 1024)

    // let copy = handle  // ❌ 编译错误：FileHandle 不可复制
    // useFile(handle)    // 编译器确保所有权正确
}


consume 操作符
consume 显式地将值的所有权从一个变量转移到另一个：


struct UniqueToken: ~Copyable {
    let id: UUID

    init() { id = UUID() }

    deinit { print("Token \(id) expired") }
}

func transferToken() {
    let token = UniqueToken()
    print("Created: \(token.id)")

    let newOwner = consume token
    print("Transferred: \(newOwner.id)")

    // print(token.id) // ❌ 编译错误：token 已被消耗
}
对于可复制类型，consume 同样有效——它告诉编译器此后不再需要该值，可以避免不必要的拷贝：


func processLargeArray() {
    var data = [Int](repeating: 0, count: 1_000_000)
    data.append(42)

    let result = consume data
    // data 不再可用，避免了大数组的拷贝
    process(result)
}


不可复制类型的方法
不可复制类型的方法需要注意所有权：


struct Connection: ~Copyable {
    private var isOpen: Bool

    init() { isOpen = true }

    borrowing func query(_ sql: String) -> [Row] {
        guard isOpen else { return [] }
        return executeQuery(sql)
    }

    mutating func reconnect() {
        isOpen = true
    }

    consuming func close() {
        isOpen = false
        print("Connection closed")
    }
}

func useConnection() {
    var conn = Connection()
    let rows = conn.query("SELECT * FROM users") // borrowing
    conn.reconnect()                               // mutating
    conn.close()                                   // consuming：conn 不再可用
}

实际应用场景
RAII 模式
不可复制类型天然适合 RAII（资源获取即初始化）模式：


struct MutexGuard<Value>: ~Copyable {
    private let mutex: UnsafeMutablePointer<pthread_mutex_t>
    var value: Value

    init(mutex: UnsafeMutablePointer<pthread_mutex_t>, value: Value) {
        pthread_mutex_lock(mutex)
        self.mutex = mutex
        self.value = value
    }

    deinit {
        pthread_mutex_unlock(mutex)
    }
}
唯一资源令牌
struct DatabaseTransaction: ~Copyable {
    private let connection: DatabaseConnection

    init(connection: DatabaseConnection) throws {
        self.connection = connection
        try connection.execute("BEGIN")
    }

    consuming func commit() throws {
        try connection.execute("COMMIT")
    }

    consuming func rollback() throws {
        try connection.execute("ROLLBACK")
    }

    deinit {
        try? connection.execute("ROLLBACK")
    }
}


提示： 不可复制类型目前（Swift 5.9+）仍在快速演进中。泛型中使用 ~Copyable 的支持在 Swift 6 中进一步完善。






小结
    

  • borrowing 参数只读借用值，不产生拷贝；consuming 参数转移所有权
    • 

  • ~Copyable 标记不可复制类型，确保值的唯一所有权
    • 

  • consume 操作符显式转移所有权，转移后原变量不可再使用
    • 

  • 不可复制类型的方法可以标记为 borrowing、mutating 或 consuming
    • 

  • 不可复制类型适合建模唯一资源（文件句柄、网络连接、事务）和 RAII 模式
    • 

  • 所有权特性让 Swift 能更精细地控制值的生命周期，减少不必要的拷贝
    •
