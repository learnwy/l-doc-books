不安全指针

概述

Swift 的 UnsafePointer 家族提供了直接操作内存的能力，是与 C/Objective-C 互操作、高性能计算和底层系统编程的基础。虽然 Swift 是一门内存安全的语言，但 Unsafe* API 让开发者在需要时可以绕过安全检查，以获得最大的灵活性和性能。使用不安全指针需要开发者自行保证内存安全。

values.withUnsafeBufferPointer { buffer in
    for i in 0..<buffer.count {
        print(buffer[i])
    }

    let pointer = buffer.baseAddress!
    print(pointer.pointee)           // 10
    print((pointer + 2).pointee)     // 30
}


内存分配与释放
手动分配和管理内存：


let count = 5
let pointer = UnsafeMutablePointer<Int>.allocate(capacity: count)

pointer.initialize(repeating: 0, count: count)

for i in 0..<count {
    pointer[i] = i * 10
}

for i in 0..<count {
    print(pointer[i]) // 0, 10, 20, 30, 40
}

pointer.deinitialize(count: count)
pointer.deallocate()


提示： 内存管理的步骤必须严格按顺序执行：allocate → initialize → use → deinitialize → deallocate。跳过任何步骤都可能导致未定义行为。






withUnsafe* 系列方法
Swift 推荐使用 withUnsafe* 闭包作用域方法来访问指针，确保指针不会逃逸：


var numbers = [1, 2, 3, 4, 5]

let sum = numbers.withUnsafeBufferPointer { buffer -> Int in
    var total = 0
    for element in buffer {
        total += element
    }
    return total
}

numbers.withUnsafeMutableBufferPointer { buffer in
    for i in buffer.indices {
        buffer[i] *= 2
    }
}
print(numbers) // [2, 4, 6, 8, 10]
常用的 withUnsafe* 方法：


// Array
array.withUnsafeBufferPointer { }
array.withUnsafeMutableBufferPointer { }
array.withUnsafeBytes { }

// String
string.withCString { }
string.withUTF8 { }

// Data
data.withUnsafeBytes { }
data.withUnsafeMutableBytes { }

// 手动内存
withUnsafeTemporaryAllocation(byteCount: 1024, alignment: 8) { buffer in
    // 使用临时缓冲区
}


原始指针与内存绑定
原始指针操作字节级内存，需要显式绑定类型：


let rawPointer = UnsafeMutableRawPointer.allocate(
    byteCount: 16,
    alignment: MemoryLayout<Int>.alignment
)

rawPointer.storeBytes(of: 42, as: Int.self)
rawPointer.storeBytes(of: 3.14, toByteOffset: 8, as: Double.self)

let intValue = rawPointer.load(as: Int.self)
let doubleValue = rawPointer.load(fromByteOffset: 8, as: Double.self)

print(intValue)    // 42
print(doubleValue) // 3.14

rawPointer.deallocate()
使用 bindMemory 和 assumingMemoryBound 进行类型转换：


let typedPointer = rawPointer.bindMemory(to: Int.self, capacity: 2)

let assumed = rawPointer.assumingMemoryBound(to: Int.self)

C 互操作
不安全指针是 Swift 与 C 互操作的桥梁：


import Darwin

func readFile(path: String) throws -> Data {
    let fd = open(path, O_RDONLY)
    guard fd >= 0 else { throw POSIXError(.ENOENT) }
    defer { close(fd) }

    var stat = stat()
    fstat(fd, &stat)
    let size = Int(stat.st_size)

    let buffer = UnsafeMutablePointer<UInt8>.allocate(capacity: size)
    defer {
        buffer.deinitialize(count: size)
        buffer.deallocate()
    }

    let bytesRead = read(fd, buffer, size)
    guard bytesRead == size else { throw POSIXError(.EIO) }

    return Data(bytes: buffer, count: size)
}
C 函数参数映射：


// C: void process(const int *data, size_t count)
// Swift: func process(_ data: UnsafePointer<Int32>?, _ count: Int)

let values: [Int32] = [1, 2, 3]
values.withUnsafeBufferPointer { buffer in
    process(buffer.baseAddress, buffer.count)
}


安全准则
// ❌ 危险：指针逃逸
var escapedPointer: UnsafeBufferPointer<Int>?
[1, 2, 3].withUnsafeBufferPointer { buffer in
    escapedPointer = buffer // 指针可能在闭包外失效！
}

// ❌ 危险：use-after-free
let p = UnsafeMutablePointer<Int>.allocate(capacity: 1)
p.initialize(to: 42)
p.deallocate()
// print(p.pointee) // 未定义行为！

// ✅ 安全：作用域内使用
[1, 2, 3].withUnsafeBufferPointer { buffer in
    let sum = buffer.reduce(0, +)
    print(sum) // 6
}


提示： 不安全指针的核心原则——最小化 unsafe 代码的范围，使用 withUnsafe* 方法限制指针生命周期，永远不要让指针逃逸出安全作用域。






小结
    

  • Swift 提供 UnsafePointer、UnsafeMutablePointer、UnsafeRawPointer 等系列指针类型
    • 

  • 内存管理遵循 allocate → initialize → use → deinitialize → deallocate 流程
    • 

  • 推荐使用 withUnsafe* 闭包限制指针生命周期，防止逃逸
    • 

  • 原始指针需要通过 bindMemory 或 assumingMemoryBound 进行类型绑定
    • 

  • 不安全指针是 C 互操作的桥梁，Swift 数组可直接传递给 C 函数
    • 

  • 最小化 unsafe 代码范围，在安全边界内封装不安全操作
    •
