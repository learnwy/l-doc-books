数据仓库 SQL

核心概念

数据仓库 SQL 是面向分析场景的 SQL 编程范式，处理的数据量通常从 GB 到 PB 级别。与 OLTP 数据库侧重单行操作不同，数据仓库（OLAP）关注批量数据的聚合分析、维度建模和 ETL 变换。现代数据仓库（如 BigQuery、Snowflake、ClickHouse）在标准 SQL 基础上提供了大量分析扩展，支持窗口函数、QUALIFY、UNPIVOT 等高级特性。

维度建模与星型模式
数据仓库的核心设计模式是星型模式（Star Schema），由事实表和维度表组成。事实表存储业务事件的度量值，维度表存储描述性属性：


CREATE TABLE fact_orders (
    order_id BIGINT,
    customer_key INT REFERENCES dim_customer(customer_key),
    product_key INT REFERENCES dim_product(product_key),
    date_key INT REFERENCES dim_date(date_key),
    store_key INT REFERENCES dim_store(store_key),
    quantity INT,
    unit_price DECIMAL(10,2),
    discount DECIMAL(5,2),
    total_amount DECIMAL(12,2),
    created_at TIMESTAMP
);

CREATE TABLE dim_date (
    date_key INT PRIMARY KEY,
    full_date DATE,
    year INT,
    quarter INT,
    month INT,
    week INT,
    day_of_week INT,
    is_weekend BOOLEAN,
    is_holiday BOOLEAN,
    fiscal_year INT,
    fiscal_quarter INT
);

CREATE TABLE dim_customer (
    customer_key INT PRIMARY KEY,
    customer_id VARCHAR(50),
    customer_name VARCHAR(200),
    segment VARCHAR(50),
    region VARCHAR(100),
    city VARCHAR(100),
    registration_date DATE,
    scd_start_date DATE,
    scd_end_date DATE,
    is_current BOOLEAN
);

缓慢变化维（SCD）
维度数据会随时间变化（如客户地址变更），缓慢变化维（Slowly Changing Dimensions）追踪这些历史变化：


-- SCD Type 2: 保留完整历史，使用生效/失效日期
MERGE INTO dim_customer target
USING staging_customer source
ON target.customer_id = source.customer_id AND target.is_current = true

WHEN MATCHED AND (
    target.segment <> source.segment OR
    target.region <> source.region
) THEN
    UPDATE SET
        scd_end_date = CURRENT_DATE - INTERVAL '1 day',
        is_current = false

WHEN NOT MATCHED THEN
    INSERT (customer_key, customer_id, customer_name, segment, region,
            scd_start_date, scd_end_date, is_current)
    VALUES (nextval('customer_key_seq'), source.customer_id, source.customer_name,
            source.segment, source.region, CURRENT_DATE, '9999-12-31', true);

-- 查询某历史时刻的维度值
SELECT c.customer_name, c.segment, c.region, SUM(f.total_amount) AS revenue
FROM fact_orders f
JOIN dim_customer c ON f.customer_key = c.customer_key
JOIN dim_date d ON f.date_key = d.date_key
WHERE d.full_date BETWEEN c.scd_start_date AND c.scd_end_date
  AND d.year = 2024
GROUP BY c.customer_name, c.segment, c.region;

增量加载模式
数据仓库的 ETL 流程核心是增量加载——只处理上次加载后的新增和变更数据：


-- 基于时间戳的增量加载
INSERT INTO dwh.fact_orders
SELECT
    o.order_id,
    COALESCE(c.customer_key, -1) AS customer_key,
    COALESCE(p.product_key, -1) AS product_key,
    d.date_key,
    o.quantity,
    o.unit_price,
    o.discount,
    o.total_amount,
    o.created_at
FROM staging.orders o
LEFT JOIN dwh.dim_customer c
    ON o.customer_id = c.customer_id AND c.is_current = true
LEFT JOIN dwh.dim_product p
    ON o.product_id = p.product_id AND p.is_current = true
LEFT JOIN dwh.dim_date d
    ON DATE(o.created_at) = d.full_date
WHERE o.updated_at > (
    SELECT COALESCE(MAX(load_timestamp), '1970-01-01')
    FROM dwh.etl_log
    WHERE table_name = 'fact_orders'
);
interface IncrementalLoadConfig {

分析查询优化
数据仓库查询优化的核心策略与 OLTP 差异很大，重点在于减少扫描数据量和利用列存优势：


-- 分区裁剪：只扫描相关分区
SELECT
    d.month,
    SUM(f.total_amount) AS revenue
FROM fact_orders f
JOIN dim_date d ON f.date_key = d.date_key
WHERE f.created_at >= '2024-01-01'
  AND f.created_at < '2025-01-01'
GROUP BY d.month;

-- 物化视图：预计算常用聚合
CREATE MATERIALIZED VIEW mv_monthly_revenue AS
SELECT
    DATE_TRUNC('month', created_at) AS month,
    customer_segment,
    COUNT(*) AS order_count,
    SUM(total_amount) AS revenue,
    AVG(total_amount) AS avg_order_value
FROM fact_orders f
JOIN dim_customer c ON f.customer_key = c.customer_key
WHERE c.is_current = true
GROUP BY 1, 2;

REFRESH MATERIALIZED VIEW CONCURRENTLY mv_monthly_revenue;

小结
    

  • 数据仓库 SQL 侧重批量聚合分析，与 OLTP 的单行操作范式不同
    • 

  • 星型模式（事实表 + 维度表）是数据仓库的核心建模方法
    • 

  • SCD Type 2 通过新增行和生效日期追踪维度的历史变化
    • 

  • 增量加载基于时间戳水位线或 CDC，只处理新增和变更数据
    • 

  • 分区裁剪、列投影和物化视图是数据仓库查询优化的三大支柱
    • 

  • 现代云数据仓库（BigQuery/Snowflake）通过存算分离实现弹性伸缩
