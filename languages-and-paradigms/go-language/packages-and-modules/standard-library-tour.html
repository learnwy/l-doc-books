标准库导览

核心概念

Go 标准库是语言的核心竞争力之一。它提供了丰富、高质量、开箱即用的包，覆盖了 I/O、网络、加密、编码、测试等几乎所有基础领域。Go 标准库的设计哲学是"足够好"——不追求大而全的框架，而是提供简洁实用的构建块。

格式化占位符
package main

import "fmt"

type User struct {
    Name string
    Age  int
}

func main() {
    u := User{Name: "Alice", Age: 30}

    fmt.Printf("%%v:  %v\n", u)
    fmt.Printf("%%+v: %+v\n", u)
    fmt.Printf("%%#v: %#v\n", u)
    fmt.Printf("%%T:  %T\n", u)

    fmt.Printf("%%d: %d\n", 42)
    fmt.Printf("%%b: %b\n", 42)
    fmt.Printf("%%o: %o\n", 42)
    fmt.Printf("%%x: %x\n", 42)
    fmt.Printf("%%X: %X\n", 42)

    fmt.Printf("%%f:   %f\n", 3.14159)
    fmt.Printf("%%.2f: %.2f\n", 3.14159)
    fmt.Printf("%%e:   %e\n", 3.14159)
    fmt.Printf("%%g:   %g\n", 3.14159)

    fmt.Printf("%%s:  %s\n", "hello")
    fmt.Printf("%%q:  %q\n", "hello")
    fmt.Printf("%%10s:  %10s|\n", "right")
    fmt.Printf("%%-10s: %-10s|\n", "left")

    fmt.Printf("%%p: %p\n", &u)
}

核心接口
package main

import (
    "bytes"
    "fmt"
    "io"
    "strings"
)

func main() {
    r := strings.NewReader("Hello, io!")

    data, err := io.ReadAll(r)
    if err != nil {
        fmt.Println(err)
        return
    }
    fmt.Println(string(data))

    src := strings.NewReader("源数据")
    var dst bytes.Buffer
    n, _ := io.Copy(&dst, src)
    fmt.Printf("复制了 %d 字节: %s\n", n, dst.String())

    r2 := strings.NewReader("Hello, World!")
    limited := io.LimitReader(r2, 5)
    data2, _ := io.ReadAll(limited)
    fmt.Println(string(data2))

    r3 := strings.NewReader("aaaa")
    r4 := strings.NewReader("bbbb")
    multi := io.MultiReader(r3, r4)
    all, _ := io.ReadAll(multi)
    fmt.Println(string(all))
}

os 文件操作
package main

import (
    "bufio"
    "fmt"
    "os"
)

func main() {
    f, err := os.Create("example.txt")
    if err != nil {
        fmt.Println(err)
        return
    }
    defer f.Close()

    f.WriteString("第一行\n")
    f.WriteString("第二行\n")
    f.WriteString("第三行\n")

    content, err := os.ReadFile("example.txt")
    if err != nil {
        fmt.Println(err)
        return
    }
    fmt.Println("文件内容:")
    fmt.Print(string(content))

    f2, _ := os.Open("example.txt")
    defer f2.Close()

    scanner := bufio.NewScanner(f2)
    lineNum := 1
    for scanner.Scan() {
        fmt.Printf("行 %d: %s\n", lineNum, scanner.Text())
        lineNum++
    }

    info, _ := os.Stat("example.txt")
    fmt.Printf("文件名: %s, 大小: %d, 修改时间: %v\n",
        info.Name(), info.Size(), info.ModTime())

    os.Remove("example.txt")
}

HTTP 服务器
package main

import (
    "encoding/json"
    "fmt"
    "log"
    "net/http"
    "time"
)

type Response struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
    Time    string `json:"time"`
}

func jsonResponse(w http.ResponseWriter, code int, data interface{}) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(code)
    json.NewEncoder(w).Encode(data)
}

func healthHandler(w http.ResponseWriter, r *http.Request) {
    jsonResponse(w, http.StatusOK, Response{
        Code:    200,
        Message: "OK",
        Time:    time.Now().Format(time.RFC3339),
    })
}

func helloHandler(w http.ResponseWriter, r *http.Request) {
    name := r.URL.Query().Get("name")
    if name == "" {
        name = "World"
    }
    jsonResponse(w, http.StatusOK, Response{
        Code:    200,
        Message: fmt.Sprintf("Hello, %s!", name),
    })
}

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("GET /health", healthHandler)
    mux.HandleFunc("GET /hello", helloHandler)

    server := &http.Server{
        Addr:         ":8080",
        Handler:      mux,
        ReadTimeout:  10 * time.Second,
        WriteTimeout: 10 * time.Second,
        IdleTimeout:  60 * time.Second,
    }

    log.Printf("服务启动在 :8080")
    log.Fatal(server.ListenAndServe())
}
HTTP 客户端
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "time"
)

func main() {
    client := &http.Client{
        Timeout: 10 * time.Second,
    }

    resp, err := client.Get("https://httpbin.org/get")
    if err != nil {
        fmt.Println("GET 错误:", err)
        return
    }
    defer resp.Body.Close()

    body, _ := io.ReadAll(resp.Body)
    fmt.Printf("状态码: %d\n", resp.StatusCode)
    fmt.Printf("响应体: %s\n", body[:100])

    payload := map[string]string{"name": "Alice"}
    jsonData, _ := json.Marshal(payload)

    resp2, err := client.Post(
        "https://httpbin.org/post",
        "application/json",
        bytes.NewBuffer(jsonData),
    )
    if err != nil {
        fmt.Println("POST 错误:", err)
        return
    }
    defer resp2.Body.Close()
    fmt.Printf("POST 状态码: %d\n", resp2.StatusCode)
}
encoding/json
序列化与反序列化
package main

import (
    "encoding/json"
    "fmt"
    "time"
)

type User struct {
    ID        int       `json:"id"`
    Name      string    `json:"name"`
    Email     string    `json:"email"`
    Password  string    `json:"-"`
    CreatedAt time.Time `json:"created_at"`
    Tags      []string  `json:"tags,omitempty"`
    Score     *float64  `json:"score,omitempty"`
}

func main() {
    u := User{
        ID:        1,
        Name:      "Alice",
        Email:     "alice@example.com",
        Password:  "secret",
        CreatedAt: time.Now(),
        Tags:      []string{"admin", "vip"},
    }

    data, _ := json.Marshal(u)
    fmt.Println("紧凑:", string(data))

    pretty, _ := json.MarshalIndent(u, "", "  ")
    fmt.Println("美化:")
    fmt.Println(string(pretty))

    jsonStr := `{"id":2,"name":"Bob","email":"bob@example.com","tags":["user"]}`
    var u2 User
    json.Unmarshal([]byte(jsonStr), &u2)
    fmt.Printf("解析: %+v\n", u2)

    var raw map[string]interface{}
    json.Unmarshal([]byte(jsonStr), &raw)
    fmt.Println("动态:", raw["name"])
}

context 上下文
package main

import (
    "context"
    "fmt"
    "time"
)

func slowTask(ctx context.Context, name string) error {
    select {
    case <-time.After(2 * time.Second):
        fmt.Printf("[%s] 完成\n", name)
        return nil
    case <-ctx.Done():
        fmt.Printf("[%s] 取消: %v\n", name, ctx.Err())
        return ctx.Err()
    }
}

func main() {
    ctx := context.Background()

    ctx1, cancel1 := context.WithTimeout(ctx, 1*time.Second)
    defer cancel1()
    slowTask(ctx1, "超时任务")

    ctx2 := context.WithValue(ctx, "request_id", "abc-123")
    ctx2 = context.WithValue(ctx2, "user_id", 42)

    reqID := ctx2.Value("request_id").(string)
    userID := ctx2.Value("user_id").(int)
    fmt.Printf("request_id=%s, user_id=%d\n", reqID, userID)

    ctx3, cancel3 := context.WithCancel(ctx)
    go func() {
        time.Sleep(500 * time.Millisecond)
        cancel3()
    }()
    slowTask(ctx3, "取消任务")
}

    fmt.Println(strings.Replace(s, "World", "Gopher", 1))

sort 与 slices
package main

import (
    "cmp"
    "fmt"
    "slices"
    "sort"
)

type Employee struct {
    Name   string
    Salary int
}

func main() {
    nums := []int{5, 3, 8, 1, 9, 2}
    sort.Ints(nums)
    fmt.Println("排序后:", nums)

    strs := []string{"banana", "apple", "cherry"}
    sort.Strings(strs)
    fmt.Println("排序后:", strs)

    employees := []Employee{
        {"Alice", 80000},
        {"Bob", 60000},
        {"Charlie", 90000},
    }
    sort.Slice(employees, func(i, j int) bool {
        return employees[i].Salary > employees[j].Salary
    })
    fmt.Println("按薪资降序:", employees)

    nums2 := []int{5, 3, 8, 1, 9, 2}
    slices.Sort(nums2)
    fmt.Println("slices.Sort:", nums2)

    slices.SortFunc(employees, func(a, b Employee) int {
        return cmp.Compare(a.Name, b.Name)
    })
    fmt.Println("按名字排序:", employees)

    idx, found := slices.BinarySearch(nums2, 5)
    fmt.Printf("查找5: idx=%d, found=%v\n", idx, found)
}
time 时间处理
package main

import (
    "fmt"
    "time"
)

func main() {
    now := time.Now()
    fmt.Println("当前时间:", now)
    fmt.Println("Unix 时间戳:", now.Unix())
    fmt.Println("年月日:", now.Year(), now.Month(), now.Day())
    fmt.Println("时分秒:", now.Hour(), now.Minute(), now.Second())

    fmt.Println("RFC3339:", now.Format(time.RFC3339))
    fmt.Println("自定义:", now.Format("2006-01-02 15:04:05"))

    t, _ := time.Parse("2006-01-02", "2024-03-15")
    fmt.Println("解析:", t)

    d := 2*time.Hour + 30*time.Minute
    fmt.Println("时长:", d)
    future := now.Add(d)
    fmt.Println("2.5小时后:", future.Format("15:04:05"))

    diff := future.Sub(now)
    fmt.Println("时间差:", diff)

    start := time.Now()
    time.Sleep(100 * time.Millisecond)
    elapsed := time.Since(start)
    fmt.Println("耗时:", elapsed)
}

小结
Go 标准库设计精良，以简洁的接口覆盖了绝大多数开发需求。


核心要点回顾：


    

  • fmt 提供格式化 I/O，%v/%+v/%#v 满足不同调试需求
    • 

  • io.Reader/io.Writer 是 Go I/O 抽象的基石
    • 

  • os 提供跨平台文件和系统操作
    • 

  • net/http 同时提供高性能 HTTP 服务器和客户端
    • 

  • encoding/json 通过结构体标签控制序列化行为
    • 

  • context 管理请求生命周期，传播取消信号和超时
    • 

  • strings/strconv 处理字符串操作和类型转换
    • 

  • sort/slices 提供排序和二分查找
    • 

  • time 使用参考时间 2006-01-02 15:04:05 格式化
