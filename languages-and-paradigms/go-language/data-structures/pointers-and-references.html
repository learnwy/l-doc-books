指针与引用

核心概念

Go 的指针提供了对内存的直接操控能力，同时通过禁止指针运算保证了内存安全。理解值传递、指针语义以及逃逸分析机制是编写高效 Go 程序的基础。与 C/C++ 不同，Go 的指针更加安全——不支持指针算术运算，垃圾回收器自动管理内存。

指针基础
声明与使用
package main

import "fmt"

func main() {
    x := 42
    var p *int = &x

    fmt.Printf("x 的值: %d\n", x)
    fmt.Printf("x 的地址: %p\n", &x)
    fmt.Printf("p 的值（地址）: %p\n", p)
    fmt.Printf("p 指向的值: %d\n", *p)

    *p = 100
    fmt.Printf("修改后 x: %d\n", x)

    y := 200
    p = &y
    fmt.Printf("重新指向后 *p: %d\n", *p)
    fmt.Printf("x 未变: %d\n", x)
}

new 与 & 的区别
package main

import "fmt"

    Host string
    Port int

值传递 vs 指针传递
Go 只有值传递
Go 中所有参数传递都是值传递。传递指针时，复制的是指针值（地址），不是指针指向的数据：


package main

import "fmt"

type User struct {
    Name string
    Age  int
}

func modifyByValue(u User) {
    u.Name = "Modified"
    fmt.Printf("函数内（值）: %+v, 地址: %p\n", u, &u)
}

func modifyByPointer(u *User) {
    u.Name = "Modified"
    fmt.Printf("函数内（指针）: %+v, 地址: %p\n", *u, u)
}

func main() {
    u := User{Name: "Alice", Age: 30}
    fmt.Printf("原始: %+v, 地址: %p\n", u, &u)

    modifyByValue(u)
    fmt.Printf("值传递后: %+v\n", u)

    modifyByPointer(&u)
    fmt.Printf("指针传递后: %+v\n", u)
}

切片传递的微妙之处
package main

import "fmt"

func modifySlice(s []int) {
    s[0] = 999
    fmt.Println("函数内修改元素:", s)
}

func appendSlice(s []int) {
    s = append(s, 100)
    fmt.Println("函数内 append:", s)
}

func appendSlicePtr(s *[]int) {
    *s = append(*s, 100)
}

func main() {
    s := []int{1, 2, 3}

    modifySlice(s)
    fmt.Println("修改元素后:", s)

    s = []int{1, 2, 3}
    appendSlice(s)
    fmt.Println("append 后:", s)

    s = []int{1, 2, 3}
    appendSlicePtr(&s)
    fmt.Println("指针 append 后:", s)
}
指针与结构体
自动解引用
Go 在访问结构体字段和调用方法时自动解引用指针：


package main

import "fmt"

type Point struct {
    X, Y int
}

func (p *Point) Move(dx, dy int) {
    p.X += dx
    p.Y += dy
}

func main() {
    p := &Point{X: 1, Y: 2}

    fmt.Println(p.X, p.Y)

    p.Move(10, 20)
    fmt.Println(p.X, p.Y)

    v := Point{X: 5, Y: 6}
    v.Move(1, 1)
    fmt.Println(v.X, v.Y)
}
链表实现
package main

import "fmt"

type Node struct {
    Value int
    Next  *Node
}

type LinkedList struct {
    Head *Node
    Size int
}

func (l *LinkedList) Push(val int) {
    l.Head = &Node{Value: val, Next: l.Head}
    l.Size++
}

func (l *LinkedList) Pop() (int, bool) {
    if l.Head == nil {
        return 0, false
    }
    val := l.Head.Value
    l.Head = l.Head.Next
    l.Size--
    return val, true
}

func (l *LinkedList) String() string {
    result := ""
    current := l.Head
    for current != nil {
        if result != "" {
            result += " -> "
        }
        result += fmt.Sprintf("%d", current.Value)
        current = current.Next
    }
    return result
}

func main() {
    list := &LinkedList{}
    list.Push(1)
    list.Push(2)
    list.Push(3)
    fmt.Println("链表:", list)
    fmt.Println("大小:", list.Size)

    val, ok := list.Pop()
    fmt.Printf("Pop: %d, ok=%v\n", val, ok)
    fmt.Println("链表:", list)
}

逃逸场景
package main

import "fmt"

func noEscape() int {
    x := 42
    return x
}

func escapeToHeap() *int {
    x := 42
    return &x
}

func escapeInterface() {
    x := 42
    fmt.Println(x)
}

func escapeSlice() []int {
    s := make([]int, 10000)
    return s
}

func main() {
    _ = noEscape()
    _ = escapeToHeap()
    escapeInterface()
    _ = escapeSlice()
}

减少逃逸的技巧
package main

import "fmt"

type Result struct {
    Value int
    Err   error
}

func processInPlace(r *Result) {
    r.Value = 42
    r.Err = nil
}

func main() {
    var r Result
    processInPlace(&r)
    fmt.Println(r.Value)
}

unsafe.Pointer
基本用法
unsafe.Pointer 是 Go 中绕过类型系统的底层工具，仅在性能关键的底层代码中使用：


package main

import (
    "fmt"
    "unsafe"
)

func main() {
    var x int64 = 42
    p := unsafe.Pointer(&x)

    ip := (*int64)(p)
    fmt.Println(*ip)

    type SliceHeader struct {
        Data unsafe.Pointer
        Len  int
        Cap  int
    }

    s := []int{10, 20, 30}
    header := (*SliceHeader)(unsafe.Pointer(&s))
    fmt.Printf("Data=%p, Len=%d, Cap=%d\n", header.Data, header.Len, header.Cap)

    firstElement := (*int)(header.Data)
    fmt.Println("第一个元素:", *firstElement)
}

高效字符串与字节切片转换
package main

import (
    "fmt"
    "unsafe"
)

func stringToBytes(s string) []byte {
    return unsafe.Slice(unsafe.StringData(s), len(s))
}

func bytesToString(b []byte) string {
    return unsafe.String(unsafe.SliceData(b), len(b))
}

func main() {
    s := "hello, world"
    b := stringToBytes(s)
    fmt.Println("bytes:", b)

    b2 := []byte{71, 111}
    s2 := bytesToString(b2)
    fmt.Println("string:", s2)
}

小结
Go 的指针在保持安全的前提下提供了足够的底层控制能力。值传递机制清晰简单，逃逸分析让编译器自动优化分配策略。


核心要点回顾：


    

  • Go 只有值传递，指针传递的是地址的副本
    • 

  • & 取地址，* 解引用，Go 自动解引用结构体指针
    • 

  • new(T) 返回零值指针，&T{} 返回可初始化的指针
    • 

  • 切片、map、channel 底层是指针，传递时共享数据
    • 

  • 逃逸分析决定变量在栈还是堆上分配，-gcflags="-m" 查看结果
    • 

  • 返回局部指针、传递给 interface{}、闭包引用会导致逃逸
    • 

  • unsafe.Pointer 绕过类型系统，仅在底层性能关键代码中使用
    • 

  • 优先用值传递小对象，用指针传递大对象或需要修改的对象
