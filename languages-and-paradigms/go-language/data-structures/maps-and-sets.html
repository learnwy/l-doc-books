Map 与 Set

核心概念

Map 是 Go 中的哈希表实现，提供 O(1) 平均时间的键值查找。Go 没有内置 Set 类型，但可以用 map[T]struct{} 高效模拟。理解 Map 的底层实现原理和并发安全机制对于编写正确的 Go 程序至关重要。

Map 基础
创建与初始化
package main

import "fmt"

func main() {
    m1 := make(map[string]int)
    m1["apple"] = 5
    m1["banana"] = 3
    m1["cherry"] = 8
    fmt.Println("make:", m1)

    m2 := map[string]int{
        "x": 10,
        "y": 20,
        "z": 30,
    }
    fmt.Println("字面量:", m2)

    m3 := make(map[string][]int)
    m3["odds"] = []int{1, 3, 5}
    m3["evens"] = []int{2, 4, 6}
    fmt.Println("值为切片:", m3)

    m4 := make(map[string]int, 100)
    fmt.Printf("预分配: len=%d\n", len(m4))
}

遍历
package main

import (
    "fmt"
    "sort"
)

func main() {
    m := map[string]int{
        "banana": 3,
        "apple":  5,
        "cherry": 8,
        "date":   2,
    }

    fmt.Println("无序遍历:")
    for k, v := range m {
        fmt.Printf("  %s: %d\n", k, v)
    }

    fmt.Println("只遍历键:")
    for k := range m {
        fmt.Println(" ", k)
    }

    keys := make([]string, 0, len(m))
    for k := range m {
        keys = append(keys, k)
    }
    sort.Strings(keys)

    fmt.Println("有序遍历:")
    for _, k := range keys {
        fmt.Printf("  %s: %d\n", k, m[k])
    }
}

Map 使用模式
计数器
package main

import "fmt"

func wordCount(text string) map[string]int {
    counts := make(map[string]int)
    word := ""
    for _, ch := range text + " " {
        if ch == ' ' || ch == '\n' {
            if word != "" {
                counts[word]++
                word = ""
            }
        } else {
            word += string(ch)
        }
    }
    return counts
}

func main() {
    text := "the quick brown fox jumps over the lazy dog the fox"
    counts := wordCount(text)
    for word, count := range counts {
        fmt.Printf("%-10s %d\n", word, count)
    }
}
分组
package main

import "fmt"

type Student struct {
    Name  string
    Grade string
}

func groupBy(students []Student) map[string][]Student {
    groups := make(map[string][]Student)
    for _, s := range students {
        groups[s.Grade] = append(groups[s.Grade], s)
    }
    return groups
}

func main() {
    students := []Student{
        {"Alice", "A"},
        {"Bob", "B"},
        {"Charlie", "A"},
        {"David", "C"},
        {"Eve", "B"},
        {"Frank", "A"},
    }

    groups := groupBy(students)
    for grade, members := range groups {
        names := make([]string, len(members))
        for i, m := range members {
            names[i] = m.Name
        }
        fmt.Printf("Grade %s: %v\n", grade, names)
    }
}
缓存/备忘录
package main

import "fmt"

func memoize(fn func(int) int) func(int) int {
    cache := make(map[int]int)
    return func(n int) int {
        if v, ok := cache[n]; ok {
            return v
        }
        result := fn(n)
        cache[n] = result
        return result
    }
}

func main() {
    var fib func(int) int
    fib = memoize(func(n int) int {
        if n <= 1 {
            return n
        }
        return fib(n-1) + fib(n-2)
    })

    for i := 0; i <= 10; i++ {
        fmt.Printf("fib(%d) = %d\n", i, fib(i))
    }
}
Set 实现
基本 Set
package main

import "fmt"

type Set[T comparable] map[T]struct{}

func NewSet[T comparable](items ...T) Set[T] {
    s := make(Set[T], len(items))
    for _, item := range items {
        s[item] = struct{}{}
    }
    return s
}

func (s Set[T]) Add(item T) {
    s[item] = struct{}{}
}

func (s Set[T]) Remove(item T) {
    delete(s, item)
}

func (s Set[T]) Contains(item T) bool {
    _, ok := s[item]
    return ok
}

func (s Set[T]) Size() int {
    return len(s)
}

func (s Set[T]) Items() []T {
    result := make([]T, 0, len(s))
    for item := range s {
        result = append(result, item)
    }
    return result
}

func main() {
    s := NewSet("apple", "banana", "cherry")
    fmt.Println("包含 apple:", s.Contains("apple"))
    fmt.Println("包含 grape:", s.Contains("grape"))

    s.Add("grape")
    s.Remove("banana")
    fmt.Println("元素:", s.Items())
    fmt.Println("大小:", s.Size())
}
Set 运算
package main

import "fmt"

type Set[T comparable] map[T]struct{}

func NewSet[T comparable](items ...T) Set[T] {
    s := make(Set[T], len(items))
    for _, item := range items {
        s[item] = struct{}{}
    }
    return s
}

func (s Set[T]) Union(other Set[T]) Set[T] {
    result := make(Set[T], len(s)+len(other))
    for k := range s {
        result[k] = struct{}{}
    }
    for k := range other {
        result[k] = struct{}{}
    }
    return result
}

func (s Set[T]) Intersect(other Set[T]) Set[T] {
    result := make(Set[T])
    for k := range s {
        if _, ok := other[k]; ok {
            result[k] = struct{}{}
        }
    }
    return result
}

func (s Set[T]) Difference(other Set[T]) Set[T] {
    result := make(Set[T])
    for k := range s {
        if _, ok := other[k]; !ok {
            result[k] = struct{}{}
        }
    }
    return result
}

func items[T comparable](s Set[T]) []T {
    result := make([]T, 0, len(s))
    for k := range s {
        result = append(result, k)
    }
    return result
}

func main() {
    a := NewSet(1, 2, 3, 4, 5)
    b := NewSet(4, 5, 6, 7, 8)

    fmt.Println("A:", items(a))
    fmt.Println("B:", items(b))
    fmt.Println("并集:", items(a.Union(b)))
    fmt.Println("交集:", items(a.Intersect(b)))
    fmt.Println("差集 A-B:", items(a.Difference(b)))
}

并发安全
sync.Map
package main

import (
    "fmt"
    "sync"
)

func main() {
    var m sync.Map

    var wg sync.WaitGroup
    for i := 0; i < 100; i++ {
        wg.Add(1)
        go func(n int) {
            defer wg.Done()
            m.Store(fmt.Sprintf("key_%d", n), n*10)
        }(i)
    }
    wg.Wait()

    m.Range(func(key, value interface{}) bool {
        if key.(string) < "key_10" {
            fmt.Printf("%s: %v\n", key, value)
        }
        return true
    })

    if val, ok := m.Load("key_50"); ok {
        fmt.Println("key_50:", val)
    }

    actual, loaded := m.LoadOrStore("key_200", 2000)
    fmt.Printf("key_200: val=%v, 已存在=%v\n", actual, loaded)

    m.Delete("key_50")
}

Mutex + Map
package main

import (
    "fmt"
    "sync"
)

type SafeMap[K comparable, V any] struct {
    mu   sync.RWMutex
    data map[K]V
}

func NewSafeMap[K comparable, V any]() *SafeMap[K, V] {
    return &SafeMap[K, V]{
        data: make(map[K]V),
    }
}

func (m *SafeMap[K, V]) Get(key K) (V, bool) {
    m.mu.RLock()
    defer m.mu.RUnlock()
    val, ok := m.data[key]
    return val, ok
}

func (m *SafeMap[K, V]) Set(key K, val V) {
    m.mu.Lock()
    defer m.mu.Unlock()
    m.data[key] = val
}

func (m *SafeMap[K, V]) Delete(key K) {
    m.mu.Lock()
    defer m.mu.Unlock()
    delete(m.data, key)
}

func (m *SafeMap[K, V]) Len() int {
    m.mu.RLock()
    defer m.mu.RUnlock()
    return len(m.data)
}

func main() {
    m := NewSafeMap[string, int]()

    var wg sync.WaitGroup
    for i := 0; i < 100; i++ {
        wg.Add(1)
        go func(n int) {
            defer wg.Done()
            m.Set(fmt.Sprintf("key_%d", n), n)
        }(i)
    }
    wg.Wait()

    fmt.Println("长度:", m.Len())
    if v, ok := m.Get("key_42"); ok {
        fmt.Println("key_42:", v)
    }
}

分片 Map
大并发场景用多个 map 分片





package main

import "fmt"

type ShardedMap[V any] struct {
    shards [16]map[string]V
}

func NewShardedMap[V any]() *ShardedMap[V] {
    sm := &ShardedMap[V]{}
    for i := range sm.shards {
        sm.shards[i] = make(map[string]V)
    }
    return sm
}

func (sm *ShardedMap[V]) shard(key string) int {
    h := 0
    for _, c := range key {
        h = h*31 + int(c)
    }
    if h < 0 {
        h = -h
    }
    return h % len(sm.shards)
}

func (sm *ShardedMap[V]) Set(key string, val V) {
    sm.shards[sm.shard(key)][key] = val
}

func (sm *ShardedMap[V]) Get(key string) (V, bool) {
    v, ok := sm.shards[sm.shard(key)][key]
    return v, ok
}

func main() {
    sm := NewShardedMap[int]()
    sm.Set("hello", 42)
    if v, ok := sm.Get("hello"); ok {
        fmt.Println("hello:", v)
    }
}
小结
Map 是 Go 中功能强大的哈希表实现，配合 struct{} 可以高效模拟 Set。并发场景必须使用 sync.Map 或 Mutex + map 的组合。


核心要点回顾：


    

  • Map 使用 make 或字面量创建，nil map 可读不可写
    • 

  • 使用二值检测 val, ok := m[key] 区分键不存在和零值
    • 

  • Map 遍历顺序是随机的，有序遍历需要先排序键
    • 

  • Map 的键必须是可比较类型，切片、map、函数不能做键
    • 

  • struct{} 作为 Set 值零内存占用，是最佳实践
    • 

  • 原生 map 非并发安全，并发场景用 sync.Map 或 RWMutex + map
    • 

  • sync.Map 适合读多写少场景，通用场景更推荐 RWMutex + map
    • 

  • 预分配容量和分片策略可以显著提升 Map 性能
