结构体与方法

核心概念

结构体（Struct）是 Go 中组织数据的核心类型，等同于面向对象语言中的类。Go 没有继承，通过组合（Composition）和接口实现代码复用。方法（Method）是绑定到类型上的函数，通过接收者（Receiver）将行为与数据关联。

结构体定义
基本结构体
package main

import "fmt"

type User struct {
    ID        int
    Name      string
    Email     string
    Age       int
    IsActive  bool
}

func main() {
    u1 := User{
        ID:       1,
        Name:     "Alice",
        Email:    "alice@example.com",
        Age:      30,
        IsActive: true,
    }
    fmt.Printf("%+v\n", u1)

    u2 := User{1, "Bob", "bob@example.com", 25, true}
    fmt.Println(u2)

    var u3 User
    u3.Name = "Charlie"
    u3.Age = 35
    fmt.Printf("%+v\n", u3)

    u4 := new(User)
    u4.Name = "David"
    fmt.Printf("%+v\n", u4)

    u5 := &User{Name: "Eve", Age: 28}
    fmt.Printf("%+v\n", u5)
}
结构体内存布局
字段对齐会影响结构体大小，合理排列字段可以减少内存浪费：


package main

import (
    "fmt"
    "unsafe"
)

type BadLayout struct {
    A bool
    B int64
    C bool
    D int64
}

type GoodLayout struct {
    B int64
    D int64
    A bool
    C bool
}

func main() {
    fmt.Printf("BadLayout:  size=%d\n", unsafe.Sizeof(BadLayout{}))
    fmt.Printf("GoodLayout: size=%d\n", unsafe.Sizeof(GoodLayout{}))
}

方法
值接收者 vs 指针接收者
package main

import "fmt"

type Rectangle struct {
    Width  float64
    Height float64
}

func (r Rectangle) Area() float64 {
    return r.Width * r.Height
}

func (r Rectangle) Perimeter() float64 {
    return 2 * (r.Width + r.Height)
}

func (r *Rectangle) Scale(factor float64) {
    r.Width *= factor
    r.Height *= factor
}

func (r *Rectangle) SetWidth(w float64) {
    r.Width = w
}

func main() {
    r := Rectangle{Width: 10, Height: 5}
    fmt.Printf("面积: %.1f\n", r.Area())
    fmt.Printf("周长: %.1f\n", r.Perimeter())

    r.Scale(2)
    fmt.Printf("缩放后: %+v\n", r)

    p := &r
    fmt.Printf("指针调用: %.1f\n", p.Area())
}
何时用指针接收者




选择
场景





指针接收者 (r *T)
需要修改接收者状态




指针接收者 (r *T)
结构体较大，避免复制开销




指针接收者 (r *T)
一致性——如果某个方法用了指针接收者，其他方法也应该用




值接收者 (r T)
不需要修改状态的小结构体




值接收者 (r T)
基本类型或小型不可变结构





方法集规则




类型
可调用的方法





T（值）
值接收者方法




*T（指针）
值接收者方法 + 指针接收者方法





这在接口实现中非常重要：只有 *T 才实现了带指针接收者方法的接口。


构造函数模式
Go 没有构造函数语法，使用 NewXxx 工厂函数作为惯例：


package main

import (
    "errors"
    "fmt"
    "time"
)

type Server struct {
    host    string
    port    int
    timeout time.Duration
    maxConn int
    tls     bool
}

func NewServer(host string, port int) *Server {
    return &Server{
        host:    host,
        port:    port,
        timeout: 30 * time.Second,
        maxConn: 100,
        tls:     false,
    }
}

type ServerOption func(*Server)

func WithTimeout(d time.Duration) ServerOption {
    return func(s *Server) {
        s.timeout = d
    }
}

func WithMaxConn(n int) ServerOption {
    return func(s *Server) {
        s.maxConn = n
    }
}

func WithTLS(enabled bool) ServerOption {
    return func(s *Server) {
        s.tls = enabled
    }
}

func NewServerWithOptions(host string, port int, opts ...ServerOption) (*Server, error) {
    if host == "" {
        return nil, errors.New("host 不能为空")
    }
    if port <= 0 || port > 65535 {
        return nil, errors.New("端口无效")
    }

    s := &Server{
        host:    host,
        port:    port,
        timeout: 30 * time.Second,
        maxConn: 100,
    }

    for _, opt := range opts {
        opt(s)
    }

    return s, nil
}

func (s *Server) String() string {
    scheme := "http"
    if s.tls {
        scheme = "https"
    }
    return fmt.Sprintf("%s://%s:%d (timeout=%v, maxConn=%d)", scheme, s.host, s.port, s.timeout, s.maxConn)
}

func main() {
    s1 := NewServer("localhost", 8080)
    fmt.Println(s1)

    s2, err := NewServerWithOptions("0.0.0.0", 443,
        WithTLS(true),
        WithTimeout(60*time.Second),
        WithMaxConn(1000),
    )
    if err != nil {
        fmt.Println("错误:", err)
        return
    }
    fmt.Println(s2)
}
嵌入与组合
匿名嵌入
package main

import "fmt"

type Address struct {
    Street string
    City   string
    State  string
}

func (a Address) FullAddress() string {
    return fmt.Sprintf("%s, %s, %s", a.Street, a.City, a.State)
}

type Person struct {
    Name string
    Age  int
    Address
}

type Employee struct {
    Person
    Company  string
    Position string
}

func main() {
    emp := Employee{
        Person: Person{
            Name: "Alice",
            Age:  30,
            Address: Address{
                Street: "123 Main St",
                City:   "Beijing",
                State:  "BJ",
            },
        },
        Company:  "Gopher Inc",
        Position: "Engineer",
    }

    fmt.Println(emp.Name)
    fmt.Println(emp.City)
    fmt.Println(emp.FullAddress())

    fmt.Println(emp.Person.Address.FullAddress())
}

结构体标签
Tag 语法与反射
package main

import (
    "encoding/json"
    "fmt"
    "reflect"
)

type APIResponse struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
    Data    any    `json:"data,omitempty"`
}

type User struct {
    ID        int    `json:"id" db:"user_id" validate:"required"`
    Name      string `json:"name" db:"user_name" validate:"required,min=2"`
    Email     string `json:"email" db:"email" validate:"required,email"`
    Password  string `json:"-" db:"password_hash"`
    CreatedAt string `json:"created_at,omitempty" db:"created_at"`
}

func main() {
    u := User{
        ID:    1,
        Name:  "Alice",
        Email: "alice@example.com",
    }

    data, _ := json.MarshalIndent(u, "", "  ")
    fmt.Println(string(data))

    t := reflect.TypeOf(u)
    for i := 0; i < t.NumField(); i++ {
        field := t.Field(i)
        fmt.Printf("%-10s json=%-20s db=%-15s\n",
            field.Name,
            field.Tag.Get("json"),
            field.Tag.Get("db"),
        )
    }
}

空结构体
package main

import (
    "fmt"
    "unsafe"
)

func main() {
    var empty struct{}
    fmt.Printf("空结构体大小: %d\n", unsafe.Sizeof(empty))

    done := make(chan struct{})
    go func() {
        fmt.Println("工作完成")
        close(done)
    }()
    <-done

    set := make(map[string]struct{})

}
小结
结构体和方法是 Go 程序组织数据和行为的核心机制。Go 通过组合取代继承，通过接口实现多态，形成了独特而强大的类型系统。


核心要点回顾：


    

  • 结构体是字段的集合，使用 type Name struct {} 定义
    • 

  • 值接收者不修改状态，指针接收者可修改且避免大对象复制
    • 

  • *T 的方法集包含值接收者和指针接收者方法，T 只包含值接收者方法
    • 

  • NewXxx 工厂函数 + Option 模式是 Go 推荐的构造模式
    • 

  • 匿名嵌入实现组合，嵌入字段的方法和字段自动提升
    • 

  • 结构体标签通过反射访问，广泛用于序列化和 ORM
    • 

  • 字段排列顺序影响内存对齐，大字段在前可减少填充
    • 

  • 空结构体 struct{} 占零字节，用于信号 channel 和 Set 值
