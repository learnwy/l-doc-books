同步原语

核心概念

Go 的并发模型以 goroutine 和 channel 为核心，但低级同步场景仍需 sync 包提供的原语。sync 包包含互斥锁、读写锁、等待组、条件变量、原子操作等，是构建并发安全数据结构的基础。

Mutex 互斥锁
基本用法
package main

import (
    "fmt"
    "sync"
)

type SafeCounter struct {
    mu    sync.Mutex
    count int
}

func (c *SafeCounter) Inc() {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.count++
}

func (c *SafeCounter) Dec() {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.count--
}

func (c *SafeCounter) Value() int {
    c.mu.Lock()
    defer c.mu.Unlock()
    return c.count
}

func main() {
    counter := &SafeCounter{}
    var wg sync.WaitGroup

    for i := 0; i < 1000; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            counter.Inc()
        }()
    }

    wg.Wait()
    fmt.Println("最终计数:", counter.Value())
}
Mutex 注意事项




注意事项
说明





不可复制
sync.Mutex 使用后不能复制，传递时用指针




不可重入
同一 goroutine 重复 Lock 会死锁




defer Unlock
总是用 defer mu.Unlock() 防止忘记释放




最小化临界区
锁的范围越小越好




零值可用
不需要初始化





死锁示例
package main

import (
    "fmt"
    "sync"
    "time"
)

func main() {
    var mu1, mu2 sync.Mutex

    go func() {
        mu1.Lock()
        defer mu1.Unlock()
        time.Sleep(100 * time.Millisecond)
        mu2.Lock()
        defer mu2.Unlock()
        fmt.Println("goroutine 1")
    }()

    go func() {
        mu2.Lock()
        defer mu2.Unlock()
        time.Sleep(100 * time.Millisecond)
        mu1.Lock()
        defer mu1.Unlock()
        fmt.Println("goroutine 2")
    }()

    time.Sleep(2 * time.Second)
    fmt.Println("可能死锁！")
}
RWMutex 读写锁
读多写少场景
package main

import (
    "fmt"
    "sync"
    "time"
)

type Cache struct {
    mu   sync.RWMutex
    data map[string]string
}

func NewCache() *Cache {
    return &Cache{data: make(map[string]string)}
}

func (c *Cache) Get(key string) (string, bool) {
    c.mu.RLock()
    defer c.mu.RUnlock()
    val, ok := c.data[key]
    return val, ok
}

func (c *Cache) Set(key, value string) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.data[key] = value
}

func (c *Cache) Delete(key string) {
    c.mu.Lock()
    defer c.mu.Unlock()
    delete(c.data, key)
}

func (c *Cache) Len() int {
    c.mu.RLock()
    defer c.mu.RUnlock()
    return len(c.data)
}

func main() {
    cache := NewCache()
    var wg sync.WaitGroup

    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func(n int) {
            defer wg.Done()
            key := fmt.Sprintf("key_%d", n)
            cache.Set(key, fmt.Sprintf("value_%d", n))
        }(i)
    }

    for i := 0; i < 100; i++ {
        wg.Add(1)
        go func(n int) {
            defer wg.Done()
            key := fmt.Sprintf("key_%d", n%10)
            if v, ok := cache.Get(key); ok {
                _ = v
            }
            time.Sleep(time.Millisecond)
        }(i)
    }

    wg.Wait()
    fmt.Println("缓存大小:", cache.Len())
}

WaitGroup 等待组
基本用法
package main

import (
    "fmt"
    "sync"
    "time"
)

func main() {
    var wg sync.WaitGroup
    urls := []string{
        "https://example.com",
        "https://golang.org",
        "https://github.com",
    }

    for _, url := range urls {
        wg.Add(1)
        go func(u string) {
            defer wg.Done()
            time.Sleep(100 * time.Millisecond)
            fmt.Printf("已获取: %s\n", u)
        }(url)
    }

    wg.Wait()
    fmt.Println("所有请求完成")
}
带错误收集的 WaitGroup
package main

import (
    "fmt"
    "sync"
)

type Task struct {
    ID   int
    Name string
}

func processTask(t Task) error {
    if t.ID%3 == 0 {
        return fmt.Errorf("任务 %d 失败", t.ID)
    }
    return nil
}

func main() {
    tasks := []Task{
        {1, "A"}, {2, "B"}, {3, "C"},
        {4, "D"}, {5, "E"}, {6, "F"},
    }

    var wg sync.WaitGroup
    var mu sync.Mutex
    var errs []error

    for _, task := range tasks {
        wg.Add(1)
        go func(t Task) {
            defer wg.Done()
            if err := processTask(t); err != nil {
                mu.Lock()
                errs = append(errs, err)
                mu.Unlock()
            }
        }(task)
    }

    wg.Wait()

    if len(errs) > 0 {
        fmt.Printf("发生 %d 个错误:\n", len(errs))
        for _, err := range errs {
            fmt.Println(" -", err)
        }
    } else {
        fmt.Println("所有任务成功")
    }
}
Once 单次执行
单例模式
package main

import (
    "fmt"
    "sync"
)

type Database struct {
    DSN string
}

var (
    dbInstance *Database
    dbOnce     sync.Once
)

func GetDB() *Database {
    dbOnce.Do(func() {
        fmt.Println("初始化数据库连接...")
        dbInstance = &Database{DSN: "postgres://localhost/mydb"}
    })
    return dbInstance
}

func main() {
    var wg sync.WaitGroup
    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func(n int) {
            defer wg.Done()
            db := GetDB()
            fmt.Printf("goroutine %d: %p\n", n, db)
        }(i)
    }
    wg.Wait()
}

Cond 条件变量
生产者-消费者
package main

import (
    "fmt"
    "sync"
    "time"
)

type Queue struct {
    mu    sync.Mutex
    cond  *sync.Cond
    items []int
    max   int
}

func NewQueue(max int) *Queue {
    q := &Queue{max: max}
    q.cond = sync.NewCond(&q.mu)
    return q
}

func (q *Queue) Put(item int) {
    q.mu.Lock()
    defer q.mu.Unlock()

    for len(q.items) >= q.max {
        q.cond.Wait()
    }

    q.items = append(q.items, item)
    q.cond.Signal()
}

func (q *Queue) Get() int {
    q.mu.Lock()
    defer q.mu.Unlock()

    for len(q.items) == 0 {
        q.cond.Wait()
    }

    item := q.items[0]
    q.items = q.items[1:]
    q.cond.Signal()
    return item
}

func main() {
    q := NewQueue(5)
    var wg sync.WaitGroup

    wg.Add(1)
    go func() {
        defer wg.Done()
        for i := 0; i < 10; i++ {
            q.Put(i)
            fmt.Println("生产:", i)
            time.Sleep(50 * time.Millisecond)
        }
    }()

    wg.Add(1)
    go func() {
        defer wg.Done()
        for i := 0; i < 10; i++ {
            item := q.Get()
            fmt.Println("消费:", item)
            time.Sleep(100 * time.Millisecond)
        }
    }()

    wg.Wait()
}

Atomic 原子操作
基本原子操作
package main

import (
    "fmt"
    "sync"
    "sync/atomic"
)

func main() {
    var counter int64 = 0
    var wg sync.WaitGroup

    for i := 0; i < 1000; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            atomic.AddInt64(&counter, 1)
        }()
    }

    wg.Wait()
    fmt.Println("原子计数:", atomic.LoadInt64(&counter))

    var flag int32 = 0
    atomic.StoreInt32(&flag, 1)
    fmt.Println("flag:", atomic.LoadInt32(&flag))

    old := atomic.SwapInt64(&counter, 0)
    fmt.Println("旧值:", old, "新值:", atomic.LoadInt64(&counter))

    atomic.StoreInt64(&counter, 100)
    swapped := atomic.CompareAndSwapInt64(&counter, 100, 200)
    fmt.Println("CAS 成功:", swapped, "值:", atomic.LoadInt64(&counter))

    swapped = atomic.CompareAndSwapInt64(&counter, 100, 300)
    fmt.Println("CAS 成功:", swapped, "值:", atomic.LoadInt64(&counter))
}
atomic.Value
package main

import (
    "fmt"
    "sync"
    "sync/atomic"
    "time"
)

type Config struct {
    Workers int
    Debug   bool
    Version string
}

var configStore atomic.Value

func init() {
    configStore.Store(&Config{
        Workers: 4,
        Debug:   false,
        Version: "1.0.0",
    })
}

func GetConfig() *Config {
    return configStore.Load().(*Config)
}

func UpdateConfig(cfg *Config) {
    configStore.Store(cfg)
}

func main() {
    var wg sync.WaitGroup

    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            cfg := GetConfig()
            fmt.Printf("Workers=%d, Version=%s\n", cfg.Workers, cfg.Version)
            time.Sleep(50 * time.Millisecond)
        }()
    }

    time.Sleep(20 * time.Millisecond)
    UpdateConfig(&Config{Workers: 8, Debug: true, Version: "2.0.0"})

    wg.Wait()
}

sync.Pool 对象池
减少 GC 压力
package main

import (
    "bytes"
    "fmt"
    "sync"
)

var bufferPool = sync.Pool{
    New: func() interface{} {
        return new(bytes.Buffer)
    },
}

func processRequest(data string) string {
    buf := bufferPool.Get().(*bytes.Buffer)
    defer func() {
        buf.Reset()
        bufferPool.Put(buf)
    }()

    buf.WriteString("处理: ")
    buf.WriteString(data)
    return buf.String()
}

func main() {
    var wg sync.WaitGroup
    for i := 0; i < 100; i++ {
        wg.Add(1)
        go func(n int) {
            defer wg.Done()
            result := processRequest(fmt.Sprintf("请求_%d", n))
            _ = result
        }(i)
    }
    wg.Wait()
    fmt.Println("所有请求处理完成")
}

小结
sync 包提供了 Go 并发编程的低级工具箱。在大多数场景下优先使用 channel，当 channel 不够高效或不适合时再选择 sync 原语。


核心要点回顾：


    

  • Mutex 用于互斥访问，RWMutex 用于读多写少场景
    • 

  • WaitGroup 等待一组 goroutine 完成，Add 在 go 之前调用
    • 

  • Once 保证函数只执行一次，适合单例和延迟初始化
    • 

  • Cond 实现条件等待和通知，用于生产者-消费者模式
    • 

  • atomic 提供无锁操作，性能极高但只适合简单场景
    • 

  • sync.Pool 复用临时对象减少 GC 压力，对象可能被 GC 回收
    • 

  • 所有 sync 类型使用后不可复制，始终传指针
    • 

  • 优先 channel，必要时再用 sync 原语
