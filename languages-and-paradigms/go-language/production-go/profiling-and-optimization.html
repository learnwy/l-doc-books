Profiling And Optimization 性能分析与优化

核心概念

Go 内建了强大的性能分析工具链，以 pprof 为核心，覆盖 CPU、内存、Goroutine、阻塞、互斥锁等多个维度。配合 benchmark、trace 和逃逸分析，形成完整的性能诊断与优化体系。

pprof 工具
CPU Profiling
CPU Profiling 通过定期采样（默认 10ms 间隔）记录调用栈，找出消耗 CPU 最多的函数：


package main

import (
	"crypto/sha256"
	"fmt"
	"log"
	"os"
	"runtime/pprof"
)

func cpuIntensiveWork() {
	data := []byte("hello world")
	for i := 0; i < 1_000_000; i++ {
		sha256.Sum256(data)
	}
}

func main() {
	f, err := os.Create("cpu.pprof")
	if err != nil {
		log.Fatal(err)
	}
	defer f.Close()

	pprof.StartCPUProfile(f)
	defer pprof.StopCPUProfile()

	cpuIntensiveWork()
	fmt.Println("完成")
}
go run main.go

go tool pprof cpu.pprof

# 交互式命令
# (pprof) top10
# (pprof) list cpuIntensiveWork
# (pprof) web

go tool pprof -http=:8080 cpu.pprof
内存 Profiling
package main

import (
	"fmt"
	"log"
	"os"
	"runtime"
	"runtime/pprof"
)

func allocateMemory() [][]byte {
	var data [][]byte
	for i := 0; i < 10000; i++ {
		buf := make([]byte, 1024)
		data = append(data, buf)
	}
	return data
}

func main() {
	data := allocateMemory()
	fmt.Printf("分配了 %d 个缓冲区\n", len(data))

	runtime.GC()

	f, err := os.Create("mem.pprof")
	if err != nil {
		log.Fatal(err)
	}
	defer f.Close()

	pprof.WriteHeapProfile(f)
}
go tool pprof mem.pprof

# 查看内存分配
# (pprof) top10 -cum
# (pprof) list allocateMemory

go tool pprof -alloc_space mem.pprof

go tool pprof -inuse_space mem.pprof

Goroutine Profiling
package main

import (
	"fmt"
	"log"
	"os"
	"runtime/pprof"
	"time"
)

func leakyWorker(ch chan struct{}) {
	<-ch
}

func main() {
	for i := 0; i < 1000; i++ {
		ch := make(chan struct{})
		go leakyWorker(ch)
	}

	time.Sleep(time.Second)

	f, err := os.Create("goroutine.pprof")
	if err != nil {
		log.Fatal(err)
	}
	defer f.Close()

	pprof.Lookup("goroutine").WriteTo(f, 1)
	fmt.Println("Goroutine profile 已写入")
}

Benchmark
Go 的 benchmark 框架内置在 testing 包中，可以精确测量函数性能：


package main

import (
	"fmt"
	"strconv"
	"strings"
	"testing"
)

func ConcatPlus(n int) string {
	s := ""
	for i := 0; i < n; i++ {
		s += strconv.Itoa(i)
	}
	return s
}

func ConcatBuilder(n int) string {
	var b strings.Builder
	for i := 0; i < n; i++ {
		b.WriteString(strconv.Itoa(i))
	}
	return b.String()
}

func ConcatBuilderPrealloc(n int) string {
	var b strings.Builder
	b.Grow(n * 4)
	for i := 0; i < n; i++ {
		b.WriteString(strconv.Itoa(i))
	}
	return b.String()
}

func BenchmarkConcatPlus(b *testing.B) {
	for i := 0; i < b.N; i++ {
		ConcatPlus(1000)
	}
}

func BenchmarkConcatBuilder(b *testing.B) {
	for i := 0; i < b.N; i++ {
		ConcatBuilder(1000)
	}
}

func BenchmarkConcatBuilderPrealloc(b *testing.B) {
	for i := 0; i < b.N; i++ {
		ConcatBuilderPrealloc(1000)
	}
}

func main() {
	fmt.Println("运行 go test -bench=. -benchmem")
}
go test -bench=. -benchmem -count=5

go test -bench=. -benchmem -cpuprofile=cpu.pprof -memprofile=mem.pprof

go install golang.org/x/perf/cmd/benchstat@latest
go test -bench=. -count=10 > old.txt
# 优化后
go test -bench=. -count=10 > new.txt
benchstat old.txt new.txt

Trace
trace 提供了比 pprof 更细粒度的运行时信息，可以可视化 goroutine 调度、GC 事件、系统调用等：


package main

import (
	"fmt"
	"os"
	"runtime/trace"
	"sync"
)

func main() {
	f, err := os.Create("trace.out")
	if err != nil {
		panic(err)
	}
	defer f.Close()

	trace.Start(f)
	defer trace.Stop()

	var wg sync.WaitGroup
	for i := 0; i < 10; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			sum := 0
			for j := 0; j < 1e7; j++ {
				sum += j
			}
			_ = sum
		}()
	}
	wg.Wait()
	fmt.Println("完成")
}
go run main.go
go tool trace trace.out

go test -trace=trace.out ./...
go tool trace trace.out
逃逸分析
逃逸分析是 Go 编译器的重要优化——决定变量分配在栈上还是堆上。栈分配零开销且自动回收，堆分配需要 GC 参与。


go build -gcflags="-m -m" main.go 2>&1 | head -30

go build -gcflags="-m" ./...
package main

import "fmt"

func stackAlloc() int {
	x := 42
	return x
}

func heapAlloc() *int {
	x := 42
	return &x
}

type User struct {
	Name string
	Age  int
}

func newUserStack() User {
	return User{Name: "Alice", Age: 30}
}

func newUserHeap() *User {
	return &User{Name: "Bob", Age: 25}
}

func main() {
	a := stackAlloc()
	b := heapAlloc()
	u1 := newUserStack()
	u2 := newUserHeap()
	fmt.Println(a, *b, u1, u2)
}

常见优化技巧
sync.Pool 复用对象
package main

import (
	"bytes"
	"fmt"
	"sync"
)

var bufPool = sync.Pool{
	New: func() any {
		return new(bytes.Buffer)
	},
}

func processWithPool(data string) string {
	buf := bufPool.Get().(*bytes.Buffer)
	defer func() {
		buf.Reset()
		bufPool.Put(buf)
	}()

	buf.WriteString("processed: ")
	buf.WriteString(data)
	return buf.String()
}

func processWithoutPool(data string) string {
	buf := new(bytes.Buffer)
	buf.WriteString("processed: ")
	buf.WriteString(data)
	return buf.String()
}

func main() {
	fmt.Println(processWithPool("hello"))
	fmt.Println(processWithoutPool("world"))
}
预分配切片容量
package main

import "fmt"

func appendNoPrealloc(n int) []int {
	var s []int
	for i := 0; i < n; i++ {
		s = append(s, i)
	}
	return s
}

func appendPrealloc(n int) []int {
	s := make([]int, 0, n)
	for i := 0; i < n; i++ {
		s = append(s, i)
	}
	return s
}

func main() {
	s1 := appendNoPrealloc(10000)
	s2 := appendPrealloc(10000)
	fmt.Println(len(s1), len(s2))
}
避免不必要的内存拷贝
package main

import (
	"fmt"
	"strings"
)

func processBytes(data []byte) int {
	return len(data)
}

func processBytesConvert(data string) int {
	return len([]byte(data))
}

func joinStrings(parts []string) string {
	return strings.Join(parts, ",")
}

func joinStringsSlow(parts []string) string {
	result := ""
	for i, p := range parts {
		if i > 0 {
			result += ","
		}
		result += p
	}
	return result
}

func main() {
	data := []byte("hello")
	fmt.Println(processBytes(data))

	parts := []string{"a", "b", "c", "d", "e"}
	fmt.Println(joinStrings(parts))
}
减少接口转换
package main

import "fmt"

type Writer interface {
	Write([]byte) (int, error)
}

type ConcreteWriter struct{}

func (w *ConcreteWriter) Write(p []byte) (int, error) {
	return len(p), nil
}

func writeViaInterface(w Writer, data []byte) {
	w.Write(data)
}

func writeDirect(w *ConcreteWriter, data []byte) {
	w.Write(data)
}

func main() {
	w := &ConcreteWriter{}
	data := []byte("hello")
	writeViaInterface(w, data)
	writeDirect(w, data)
	fmt.Println("完成")
}
