错误处理模式

核心概念

Go 的错误处理基于值返回而非异常机制。error 是一个内置接口，只有一个 Error() string 方法。这种设计使得错误处理显式、可控、可组合。Go 1.13 引入的 errors.Is/errors.As 和 %w 错误包装机制进一步完善了错误处理生态。

Error 接口
基本接口
package main

import "fmt"

type error interface {
    Error() string
}

type ValidationError struct {
    Field   string
    Message string
}

func (e *ValidationError) Error() string {
    return fmt.Sprintf("验证失败: 字段 %s - %s", e.Field, e.Message)
}

func validateAge(age int) error {
    if age < 0 {
        return &ValidationError{Field: "age", Message: "不能为负数"}
    }
    if age > 150 {
        return &ValidationError{Field: "age", Message: "超出合理范围"}
    }
    return nil
}

func main() {
    if err := validateAge(-1); err != nil {
        fmt.Println(err)
    }
    if err := validateAge(200); err != nil {
        fmt.Println(err)
    }
    if err := validateAge(25); err != nil {
        fmt.Println(err)
    } else {
        fmt.Println("验证通过")
    }
}

package main

import (
    "errors"
    "fmt"
)

var (
    ErrNotFound     = errors.New("资源未找到")
    ErrUnauthorized = errors.New("未授权")
    ErrForbidden    = errors.New("禁止访问")
)

func findUser(id int) error {
    if id <= 0 {
        return fmt.Errorf("无效用户 ID %d: %w", id, ErrNotFound)
    }
    return nil
}

func main() {
    err := findUser(-1)
    fmt.Println(err)
    fmt.Println("Is ErrNotFound:", errors.Is(err, ErrNotFound))
}
Sentinel Errors
Sentinel errors 是包级别的预定义错误值，用于与 errors.Is 进行比较：


package main

import (
    "errors"
    "fmt"
    "io"
    "os"
)

var (
    ErrInvalidInput = errors.New("无效输入")
    ErrTimeout      = errors.New("操作超时")
    ErrConflict     = errors.New("资源冲突")
)

func readConfig(path string) ([]byte, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        if errors.Is(err, os.ErrNotExist) {
            return nil, fmt.Errorf("配置文件不存在 %s: %w", path, ErrInvalidInput)
        }
        return nil, fmt.Errorf("读取配置失败: %w", err)
    }
    return data, nil
}

func processStream(r io.Reader) error {
    buf := make([]byte, 1024)
    for {
        _, err := r.Read(buf)
        if errors.Is(err, io.EOF) {
            return nil
        }
        if err != nil {
            return fmt.Errorf("读取流失败: %w", err)
        }
    }
}

func main() {
    _, err := readConfig("nonexistent.yaml")
    if err != nil {
        fmt.Println("错误:", err)
        fmt.Println("Is ErrInvalidInput:", errors.Is(err, ErrInvalidInput))
        fmt.Println("Is os.ErrNotExist:", errors.Is(err, os.ErrNotExist))
    }
}
错误包装与解包
fmt.Errorf + %w
package main

import (
    "errors"
    "fmt"
)

type DBError struct {
    Query string
    Err   error
}

func (e *DBError) Error() string {
    return fmt.Sprintf("数据库查询失败 [%s]: %v", e.Query, e.Err)
}

func (e *DBError) Unwrap() error {
    return e.Err
}

var ErrRecordNotFound = errors.New("记录不存在")

func queryUser(id int) error {
    return &DBError{
        Query: fmt.Sprintf("SELECT * FROM users WHERE id = %d", id),
        Err:   ErrRecordNotFound,
    }
}

func getUser(id int) error {
    err := queryUser(id)
    if err != nil {
        return fmt.Errorf("获取用户 %d: %w", id, err)
    }
    return nil
}

func handleRequest(userID int) error {
    err := getUser(userID)
    if err != nil {
        return fmt.Errorf("处理请求: %w", err)
    }
    return nil
}

func main() {
    err := handleRequest(42)
    fmt.Println("错误:", err)

    fmt.Println("\n--- 错误链检查 ---")
    fmt.Println("Is ErrRecordNotFound:", errors.Is(err, ErrRecordNotFound))

    var dbErr *DBError
    if errors.As(err, &dbErr) {
        fmt.Println("DB Query:", dbErr.Query)
    }
}

自定义 Is 和 As
package main

import (
    "errors"
    "fmt"
)

type HTTPError struct {
    Code    int
    Message string
}

func (e *HTTPError) Error() string {
    return fmt.Sprintf("HTTP %d: %s", e.Code, e.Message)
}

func (e *HTTPError) Is(target error) bool {
    t, ok := target.(*HTTPError)
    if !ok {
        return false
    }
    return e.Code == t.Code
}

func main() {
    err := &HTTPError{Code: 404, Message: "用户不存在"}
    target := &HTTPError{Code: 404}

    fmt.Println(errors.Is(err, target))

    err2 := &HTTPError{Code: 500, Message: "内部错误"}
    fmt.Println(errors.Is(err2, target))
}
错误处理模式
模式一：提前返回
package main

import (
    "errors"
    "fmt"
)

type Order struct {
    ID     int
    Amount float64
    Status string
}

func processOrder(o *Order) error {
    if o == nil {
        return errors.New("订单不能为空")
    }

    if o.Amount <= 0 {
        return fmt.Errorf("无效金额: %.2f", o.Amount)
    }

    if o.Status != "pending" {
        return fmt.Errorf("订单状态错误: %s", o.Status)
    }

    o.Status = "processed"
    return nil
}

func main() {
    orders := []*Order{
        nil,
        {ID: 1, Amount: -10, Status: "pending"},
        {ID: 2, Amount: 100, Status: "shipped"},
        {ID: 3, Amount: 200, Status: "pending"},
    }

    for _, o := range orders {
        if err := processOrder(o); err != nil {
            fmt.Println("失败:", err)
            continue
        }
        fmt.Printf("订单 %d 处理成功\n", o.ID)
    }
}
模式二：错误类型层次
package main

import (
    "errors"
    "fmt"
)

type AppError struct {
    Code    string
    Message string
    Err     error
}

func (e *AppError) Error() string {
    if e.Err != nil {
        return fmt.Sprintf("[%s] %s: %v", e.Code, e.Message, e.Err)
    }
    return fmt.Sprintf("[%s] %s", e.Code, e.Message)
}

func (e *AppError) Unwrap() error {
    return e.Err
}

func NewAppError(code, message string, err error) *AppError {
    return &AppError{Code: code, Message: message, Err: err}
}

var (
    ErrValidation   = errors.New("validation_error")
    ErrNotFound     = errors.New("not_found")
    ErrInternal     = errors.New("internal_error")
    ErrUnauthorized = errors.New("unauthorized")
)

func ValidateEmail(email string) error {
    if email == "" {
        return NewAppError("VALIDATION", "邮箱不能为空",
            fmt.Errorf("email is empty: %w", ErrValidation))
    }
    return nil
}

func main() {
    err := ValidateEmail("")
    fmt.Println(err)

    var appErr *AppError
    if errors.As(err, &appErr) {
        fmt.Println("错误码:", appErr.Code)
        fmt.Println("消息:", appErr.Message)
    }

    fmt.Println("Is ErrValidation:", errors.Is(err, ErrValidation))
}
模式三：多错误聚合
package main

import (
    "errors"
    "fmt"
    "strings"
)

type MultiError struct {
    Errors []error
}

func (m *MultiError) Error() string {
    msgs := make([]string, len(m.Errors))
    for i, err := range m.Errors {
        msgs[i] = err.Error()
    }
    return fmt.Sprintf("%d 个错误:\n  - %s", len(m.Errors), strings.Join(msgs, "\n  - "))
}

func (m *MultiError) Unwrap() []error {
    return m.Errors
}

type Validator struct {
    errors []error
}

func (v *Validator) Check(condition bool, msg string) {
    if !condition {
        v.errors = append(v.errors, errors.New(msg))
    }
}

func (v *Validator) Err() error {
    if len(v.errors) == 0 {
        return nil
    }
    return &MultiError{Errors: v.errors}
}

type User struct {
    Name  string
    Email string
    Age   int
}

func validateUser(u User) error {
    v := &Validator{}
    v.Check(u.Name != "", "姓名不能为空")
    v.Check(len(u.Name) >= 2, "姓名至少2个字符")
    v.Check(u.Email != "", "邮箱不能为空")
    v.Check(strings.Contains(u.Email, "@"), "邮箱格式无效")
    v.Check(u.Age > 0, "年龄必须为正数")
    v.Check(u.Age < 150, "年龄超出范围")
    return v.Err()
}

func main() {
    u := User{Name: "", Email: "invalid", Age: -1}
    if err := validateUser(u); err != nil {
        fmt.Println(err)
    }

    u2 := User{Name: "Alice", Email: "alice@example.com", Age: 30}
    if err := validateUser(u2); err != nil {
        fmt.Println(err)
    } else {
        fmt.Println("验证通过")
    }
}
模式四：错误转 HTTP 响应
package main

import (
    "encoding/json"
    "errors"
    "fmt"
    "net/http"
)

type APIError struct {
    StatusCode int    `json:"-"`
    Code       string `json:"code"`
    Message    string `json:"message"`
}

func (e *APIError) Error() string {
    return fmt.Sprintf("[%d] %s: %s", e.StatusCode, e.Code, e.Message)
}

func NewBadRequest(msg string) *APIError {
    return &APIError{StatusCode: 400, Code: "BAD_REQUEST", Message: msg}
}

func NewNotFound(msg string) *APIError {
    return &APIError{StatusCode: 404, Code: "NOT_FOUND", Message: msg}
}

func NewInternal(msg string) *APIError {
    return &APIError{StatusCode: 500, Code: "INTERNAL", Message: msg}
}

func handleError(w http.ResponseWriter, err error) {
    var apiErr *APIError
    if errors.As(err, &apiErr) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(apiErr.StatusCode)
        json.NewEncoder(w).Encode(apiErr)
        return
    }

    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(500)
    json.NewEncoder(w).Encode(NewInternal("未知错误"))
}

func main() {
    err := NewNotFound("用户 42 不存在")
    fmt.Println(err)

    var apiErr *APIError
    if errors.As(err, &apiErr) {
        fmt.Printf("HTTP %d: %s\n", apiErr.StatusCode, apiErr.Message)
    }
}

小结
Go 的错误处理以显式和可组合为核心原则。error 作为值返回让每个错误点都可见可控。


核心要点回顾：


    

  • error 是接口，errors.New 和 fmt.Errorf 创建错误
    • 

  • Sentinel errors 是包级预定义错误，用 errors.Is 比较
    • 

  • %w 包装错误保留原始链，Unwrap() 支持链遍历
    • 

  • errors.Is 做值匹配，errors.As 做类型匹配，都遍历错误链
    • 

  • 提前返回减少嵌套，添加上下文信息方便调试
    • 

  • 多错误聚合使用 Unwrap() []error（Go 1.20+）
    • 

  • 自定义错误类型携带结构化信息（错误码、HTTP 状态码等）
    • 

  • 错误消息小写开头，避免在库代码中 panic
    •
