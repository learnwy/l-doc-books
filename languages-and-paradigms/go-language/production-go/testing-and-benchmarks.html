测试与基准测试

核心概念

Go 内置了强大的测试框架 testing 包，配合 go test 命令，无需任何第三方依赖即可进行单元测试、基准测试、模糊测试和示例测试。测试在 Go 中是一等公民，与源代码共存于同一包中，遵循约定大于配置的设计哲学。

基本测试
package math

func Add(a, b int) int {
    return a + b
}

func Abs(x int) int {
    if x < 0 {
        return -x
    }
    return x
}

func Max(a, b int) int {
    if a > b {
        return a
    }
    return b
}

func Clamp(val, min, max int) int {
    if val < min {
        return min
    }
    if val > max {
        return max
    }
    return val
}
package math

import "testing"

func TestAdd(t *testing.T) {
    got := Add(2, 3)
    want := 5
    if got != want {
        t.Errorf("Add(2, 3) = %d, want %d", got, want)
    }
}

func TestAbs(t *testing.T) {
    if got := Abs(-5); got != 5 {
        t.Errorf("Abs(-5) = %d, want 5", got)
    }
    if got := Abs(5); got != 5 {
        t.Errorf("Abs(5) = %d, want 5", got)
    }
    if got := Abs(0); got != 0 {
        t.Errorf("Abs(0) = %d, want 0", got)
    }
}

表驱动测试
表驱动测试是 Go 社区最推荐的测试模式：


package math

import "testing"

func TestAddTable(t *testing.T) {
    tests := []struct {
        name string
        a, b int
        want int
    }{
        {"正数相加", 2, 3, 5},
        {"负数相加", -1, -2, -3},
        {"零值", 0, 0, 0},
        {"正负混合", -5, 10, 5},
        {"大数", 1000000, 2000000, 3000000},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            got := Add(tt.a, tt.b)
            if got != tt.want {
                t.Errorf("Add(%d, %d) = %d, want %d", tt.a, tt.b, got, tt.want)
            }
        })
    }
}

func TestClampTable(t *testing.T) {
    tests := []struct {
        name           string
        val, min, max  int
        want           int
    }{
        {"在范围内", 5, 0, 10, 5},
        {"低于最小值", -5, 0, 10, 0},
        {"超过最大值", 15, 0, 10, 10},
        {"等于最小值", 0, 0, 10, 0},
        {"等于最大值", 10, 0, 10, 10},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            got := Clamp(tt.val, tt.min, tt.max)
            if got != tt.want {
                t.Errorf("Clamp(%d, %d, %d) = %d, want %d",
                    tt.val, tt.min, tt.max, got, tt.want)
            }
        })
    }
}
测试辅助函数
package math

import "testing"

func assertEqual(t *testing.T, got, want int) {
    t.Helper()
    if got != want {
        t.Errorf("got %d, want %d", got, want)
    }
}

func assertError(t *testing.T, err error) {
    t.Helper()
    if err == nil {
        t.Error("expected error, got nil")
    }
}

func assertNoError(t *testing.T, err error) {
    t.Helper()
    if err != nil {
        t.Errorf("unexpected error: %v", err)
    }
}

func TestWithHelpers(t *testing.T) {
    assertEqual(t, Add(2, 3), 5)
    assertEqual(t, Abs(-10), 10)
    assertEqual(t, Max(3, 7), 7)
}
并行测试
package math

import "testing"

func TestParallel(t *testing.T) {
    tests := []struct {
        name string
        a, b int
        want int
    }{
        {"case1", 1, 2, 3},
        {"case2", 10, 20, 30},
        {"case3", -1, 1, 0},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            t.Parallel()
            got := Add(tt.a, tt.b)
            if got != tt.want {
                t.Errorf("Add(%d, %d) = %d, want %d", tt.a, tt.b, got, tt.want)
            }
        })
    }
}
HTTP 测试
httptest 包
package api

import (
    "encoding/json"
    "fmt"
    "net/http"
)

type Response struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
}

func HealthHandler(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(Response{Code: 200, Message: "OK"})
}

func GreetHandler(w http.ResponseWriter, r *http.Request) {
    name := r.URL.Query().Get("name")
    if name == "" {
        w.WriteHeader(http.StatusBadRequest)
        json.NewEncoder(w).Encode(Response{Code: 400, Message: "name required"})
        return
    }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(Response{
        Code:    200,
        Message: fmt.Sprintf("Hello, %s!", name),
    })
}
package api

import (
    "encoding/json"
    "net/http"
    "net/http/httptest"
    "testing"
)

func TestHealthHandler(t *testing.T) {
    req := httptest.NewRequest(http.MethodGet, "/health", nil)
    rec := httptest.NewRecorder()

    HealthHandler(rec, req)

    if rec.Code != http.StatusOK {
        t.Errorf("status = %d, want %d", rec.Code, http.StatusOK)
    }

    var resp Response
    json.NewDecoder(rec.Body).Decode(&resp)
    if resp.Message != "OK" {
        t.Errorf("message = %q, want %q", resp.Message, "OK")
    }
}

func TestGreetHandler(t *testing.T) {
    tests := []struct {
        name       string
        query      string
        wantCode   int
        wantMsg    string
    }{
        {"有名字", "?name=Alice", 200, "Hello, Alice!"},
        {"无名字", "", 400, "name required"},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            req := httptest.NewRequest(http.MethodGet, "/greet"+tt.query, nil)
            rec := httptest.NewRecorder()

            GreetHandler(rec, req)

            if rec.Code != tt.wantCode {
                t.Errorf("status = %d, want %d", rec.Code, tt.wantCode)
            }

            var resp Response
            json.NewDecoder(rec.Body).Decode(&resp)
            if resp.Message != tt.wantMsg {
                t.Errorf("message = %q, want %q", resp.Message, tt.wantMsg)
            }
        })
    }
}

func TestWithServer(t *testing.T) {
    mux := http.NewServeMux()
    mux.HandleFunc("/health", HealthHandler)

    server := httptest.NewServer(mux)
    defer server.Close()

    resp, err := http.Get(server.URL + "/health")
    if err != nil {
        t.Fatal(err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != 200 {
        t.Errorf("status = %d, want 200", resp.StatusCode)
    }
}
测试替身（Mock）
接口驱动的 Mock
package service

import (
    "errors"
    "testing"
)

type UserRepository interface {
    FindByID(id int) (*User, error)
    Save(user *User) error
}

type User struct {
    ID   int
    Name string
}

type UserService struct {
    repo UserRepository
}

func NewUserService(repo UserRepository) *UserService {
    return &UserService{repo: repo}
}

func (s *UserService) GetUser(id int) (*User, error) {
    if id <= 0 {
        return nil, errors.New("invalid id")
    }
    return s.repo.FindByID(id)
}

type mockRepo struct {
    users map[int]*User
    err   error
}

func (m *mockRepo) FindByID(id int) (*User, error) {
    if m.err != nil {
        return nil, m.err
    }
    if u, ok := m.users[id]; ok {
        return u, nil
    }
    return nil, errors.New("not found")
}

func (m *mockRepo) Save(user *User) error {
    return m.err
}

func TestGetUser(t *testing.T) {
    tests := []struct {
        name    string
        id      int
        repo    *mockRepo
        wantErr bool
    }{
        {
            "正常获取",
            1,
            &mockRepo{users: map[int]*User{1: {ID: 1, Name: "Alice"}}},
            false,
        },
        {
            "无效 ID",
            -1,
            &mockRepo{},
            true,
        },
        {
            "用户不存在",
            99,
            &mockRepo{users: map[int]*User{}},
            true,
        },
        {
            "数据库错误",
            1,
            &mockRepo{err: errors.New("db connection lost")},
            true,
        },
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            svc := NewUserService(tt.repo)
            user, err := svc.GetUser(tt.id)
            if (err != nil) != tt.wantErr {
                t.Errorf("error = %v, wantErr %v", err, tt.wantErr)
            }
            if !tt.wantErr && user == nil {
                t.Error("expected user, got nil")
            }
        })
    }
}

子基准与比较
package bench

import (
    "fmt"
    "strings"
    "testing"
)

func concatPlus(n int) string {
    s := ""
    for i := 0; i < n; i++ {
        s += "a"
    }
    return s
}

func concatBuilder(n int) string {
    var b strings.Builder
    for i := 0; i < n; i++ {
        b.WriteString("a")
    }
    return b.String()
}

func concatSprintf(n int) string {
    s := ""
    for i := 0; i < n; i++ {
        s = fmt.Sprintf("%sa", s)
    }
    return s
}

func BenchmarkConcat(b *testing.B) {
    sizes := []int{10, 100, 1000}

    for _, size := range sizes {
        b.Run(fmt.Sprintf("Plus/%d", size), func(b *testing.B) {
            for i := 0; i < b.N; i++ {
                concatPlus(size)
            }
        })

        b.Run(fmt.Sprintf("Builder/%d", size), func(b *testing.B) {
            for i := 0; i < b.N; i++ {
                concatBuilder(size)
            }
        })

        b.Run(fmt.Sprintf("Sprintf/%d", size), func(b *testing.B) {
            for i := 0; i < b.N; i++ {
                concatSprintf(size)
            }
        })
    }
}

TestMain
TestMain 用于在所有测试前后执行设置和清理：


package db

import (
    "fmt"
    "os"
    "testing"
)

var testDB string

func TestMain(m *testing.M) {
    fmt.Println("设置测试环境...")
    testDB = "test_database"

    code := m.Run()

    fmt.Println("清理测试环境...")
    testDB = ""

    os.Exit(code)
}

func TestQuery(t *testing.T) {
    if testDB == "" {
        t.Fatal("数据库未初始化")
    }
    t.Log("使用数据库:", testDB)
}

小结
Go 的测试框架遵循"简单即强大"的设计理念，内置的工具链覆盖了从单元测试到性能分析的全流程。


核心要点回顾：


    

  • 测试文件以 _test.go 结尾，函数以 Test/Benchmark/Fuzz/Example 开头
    • 

  • 表驱动测试是 Go 社区最佳实践，配合 t.Run 实现子测试
    • 

  • t.Helper() 标记辅助函数，使错误定位到调用者
    • 

  • t.Parallel() 标记可并行执行的测试，提升测试速度
    • 

  • httptest 包提供 HTTP 处理器和服务器的测试工具
    • 

  • 接口驱动 Mock 是 Go 推荐的测试替身方式
    • 

  • 基准测试使用 b.N 自适应迭代，-benchmem 报告内存分配
    • 

  • 避免编译器优化基准测试结果：将结果赋值给包级变量
    • 

  • TestMain 用于全局测试初始化和清理
    • 

  • go test -cover 生成覆盖率报告，目标 80%+ 覆盖率
    •
