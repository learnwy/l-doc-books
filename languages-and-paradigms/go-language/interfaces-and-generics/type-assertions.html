Type Assertions 类型断言

核心概念

类型断言是 Go 语言中从接口值提取具体类型的机制。每个接口值在运行时都包含两部分信息：具体类型（type）和具体值（value），类型断言就是对这两部分进行检查和提取。

类型断言语法
类型断言有两种形式：单值形式和双值（comma-ok）形式。


单值形式——断言失败触发 panic：


package main

import "fmt"

func main() {
	var i interface{} = "hello"

	s := i.(string)
	fmt.Println(s)
}
双值形式——安全的断言，不会 panic：


package main

import "fmt"

func main() {
	var i interface{} = 42

	s, ok := i.(string)
	fmt.Printf("s=%q, ok=%v\n", s, ok)

	n, ok := i.(int)
	fmt.Printf("n=%d, ok=%v\n", n, ok)

	f, ok := i.(float64)
	fmt.Printf("f=%f, ok=%v\n", f, ok)
}
断言到接口类型——检查是否实现了某个接口：


package main

import (
	"fmt"
	"io"
	"os"
	"strings"
)

func tryClose(v any) {
	if closer, ok := v.(io.Closer); ok {
		fmt.Printf("%T 实现了 io.Closer\n", v)
		closer.Close()
	} else {
		fmt.Printf("%T 未实现 io.Closer\n", v)
	}
}

func main() {
	tryClose(os.Stdout)
	tryClose(strings.NewReader("hello"))
}

Type Switch
type switch 是 Go 中优雅地处理多种类型的结构，比链式 if-else 类型断言更清晰：


package main

import (
	"fmt"
	"math"
)

type Circle struct{ Radius float64 }
type Rectangle struct{ Width, Height float64 }
type Triangle struct{ Base, Height float64 }

type Shape interface {
	Area() float64
}

func (c Circle) Area() float64    { return math.Pi * c.Radius * c.Radius }
func (r Rectangle) Area() float64 { return r.Width * r.Height }
func (t Triangle) Area() float64  { return 0.5 * t.Base * t.Height }

func describeShape(s Shape) string {
	switch v := s.(type) {
	case Circle:
		return fmt.Sprintf("圆形，半径=%.2f，面积=%.2f", v.Radius, v.Area())
	case Rectangle:
		return fmt.Sprintf("矩形，%0.f×%.0f，面积=%.2f", v.Width, v.Height, v.Area())
	case Triangle:
		return fmt.Sprintf("三角形，底=%.0f 高=%.0f，面积=%.2f", v.Base, v.Height, v.Area())
	default:
		return fmt.Sprintf("未知形状: %T", v)
	}
}

func main() {
	shapes := []Shape{
		Circle{Radius: 5},
		Rectangle{Width: 10, Height: 3},
		Triangle{Base: 8, Height: 6},
	}

	for _, s := range shapes {
		fmt.Println(describeShape(s))
	}
}
type switch 多类型匹配
package main

import "fmt"

func classify(v any) string {
	switch v.(type) {
	case int, int8, int16, int32, int64:
		return "有符号整数"
	case uint, uint8, uint16, uint32, uint64:
		return "无符号整数"
	case float32, float64:
		return "浮点数"
	case string:
		return "字符串"
	case bool:
		return "布尔值"
	case nil:
		return "nil"
	default:
		return fmt.Sprintf("其他: %T", v)
	}
}

func main() {
	values := []any{42, uint(10), 3.14, "hello", true, nil, []int{1}}
	for _, v := range values {
		fmt.Printf("%-12v -> %s\n", v, classify(v))
	}
}

编译期接口满足检查：


package main

import "fmt"

type Validator interface {
	Validate() error
}

type Email struct {
	Address string
}

func (e Email) Validate() error {
	if len(e.Address) == 0 {
		return fmt.Errorf("邮箱地址不能为空")
	}
	return nil
}

var _ Validator = Email{}
var _ Validator = (*Email)(nil)

func main() {
	e := Email{Address: "user@example.com"}
	if err := e.Validate(); err != nil {
		fmt.Println("验证失败:", err)
	} else {
		fmt.Println("验证通过")
	}
}
运行时反射
reflect 包提供了更强大的运行时类型检查能力，但性能代价更高：


package main

import (
	"fmt"
	"reflect"
)

type User struct {
	Name  string `json:"name" validate:"required"`
	Email string `json:"email" validate:"required,email"`
	Age   int    `json:"age" validate:"gte=0,lte=130"`
}

func inspectStruct(v any) {
	t := reflect.TypeOf(v)
	val := reflect.ValueOf(v)

	if t.Kind() == reflect.Ptr {
		t = t.Elem()
		val = val.Elem()
	}

	fmt.Printf("类型: %s\n", t.Name())
	fmt.Printf("字段数: %d\n\n", t.NumField())

	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		value := val.Field(i)
		jsonTag := field.Tag.Get("json")
		validateTag := field.Tag.Get("validate")

		fmt.Printf("字段: %-10s 类型: %-10s 值: %-20v json:%q validate:%q\n",
			field.Name, field.Type, value.Interface(), jsonTag, validateTag)
	}
}

func main() {
	u := User{Name: "Alice", Email: "alice@go.dev", Age: 30}
	inspectStruct(u)
}
反射 vs 类型断言性能对比
package main

import (
	"fmt"
	"reflect"
	"time"
)

func benchmarkAssertion(v any, n int) time.Duration {
	start := time.Now()
	for i := 0; i < n; i++ {
		_, _ = v.(int)
	}
	return time.Since(start)
}

func benchmarkReflect(v any, n int) time.Duration {
	start := time.Now()
	for i := 0; i < n; i++ {
		_ = reflect.TypeOf(v).Kind() == reflect.Int
	}
	return time.Since(start)
}

func main() {
	const iterations = 10_000_000
	var v any = 42

	d1 := benchmarkAssertion(v, iterations)
	d2 := benchmarkReflect(v, iterations)

	fmt.Printf("类型断言: %v\n", d1)
	fmt.Printf("反射检查: %v\n", d2)
	fmt.Printf("反射/断言 比率: %.1fx\n", float64(d2)/float64(d1))
}
常见模式
错误类型断言
package main

import (
	"errors"
	"fmt"
	"net"
)

type NotFoundError struct {
	Resource string
	ID       string
}

func (e *NotFoundError) Error() string {
	return fmt.Sprintf("%s(id=%s) 未找到", e.Resource, e.ID)
}

type ValidationError struct {
	Field   string
	Message string
}

func (e *ValidationError) Error() string {
	return fmt.Sprintf("字段 %s 验证失败: %s", e.Field, e.Message)
}

func handleError(err error) {
	if err == nil {
		return
	}

	var notFound *NotFoundError
	if errors.As(err, &notFound) {
		fmt.Printf("资源不存在: %s [%s]\n", notFound.Resource, notFound.ID)
		return
	}

	var validErr *ValidationError
	if errors.As(err, &validErr) {
		fmt.Printf("参数错误: %s - %s\n", validErr.Field, validErr.Message)
		return
	}

	var netErr net.Error
	if errors.As(err, &netErr) {
		if netErr.Timeout() {
			fmt.Println("网络超时")
		} else {
			fmt.Println("网络错误:", netErr)
		}
		return
	}

	fmt.Println("未知错误:", err)
}

func main() {
	errs := []error{
		&NotFoundError{Resource: "User", ID: "123"},
		&ValidationError{Field: "email", Message: "格式无效"},
		fmt.Errorf("包装错误: %w", &NotFoundError{Resource: "Order", ID: "456"}),
		errors.New("普通错误"),
	}

	for _, err := range errs {
		handleError(err)
	}
}
接口能力探测
package main

import "fmt"

type BasicLogger interface {
	Log(msg string)
}

type StructuredLogger interface {
	BasicLogger
	LogFields(msg string, fields map[string]any)
}

type LevelLogger interface {
	BasicLogger
	Debug(msg string)
	Info(msg string)
	Error(msg string)
}

type SimpleLogger struct{}

func (l SimpleLogger) Log(msg string) {
	fmt.Println("[LOG]", msg)
}

type AdvancedLogger struct{}

func (l AdvancedLogger) Log(msg string)   { fmt.Println("[LOG]", msg) }
func (l AdvancedLogger) Debug(msg string) { fmt.Println("[DEBUG]", msg) }
func (l AdvancedLogger) Info(msg string)  { fmt.Println("[INFO]", msg) }
func (l AdvancedLogger) Error(msg string) { fmt.Println("[ERROR]", msg) }
func (l AdvancedLogger) LogFields(msg string, fields map[string]any) {
	fmt.Printf("[FIELDS] %s %v\n", msg, fields)
}

func logMessage(logger BasicLogger, msg string) {
	if sl, ok := logger.(StructuredLogger); ok {
		sl.LogFields(msg, map[string]any{"source": "app"})
		return
	}

	if ll, ok := logger.(LevelLogger); ok {
		ll.Info(msg)
		return
	}

	logger.Log(msg)
}

func main() {
	logMessage(SimpleLogger{}, "使用简单日志")
	logMessage(AdvancedLogger{}, "使用高级日志")
}
Stringer 接口实现
package main

import "fmt"

type HTTPMethod int

const (
	GET HTTPMethod = iota
	POST
	PUT
	DELETE
	PATCH
)

func (m HTTPMethod) String() string {
	names := [...]string{"GET", "POST", "PUT", "DELETE", "PATCH"}
	if int(m) < len(names) {
		return names[m]
	}
	return fmt.Sprintf("HTTPMethod(%d)", m)
}

type Endpoint struct {
	Method HTTPMethod
	Path   string
}

func (e Endpoint) String() string {
	return fmt.Sprintf("%s %s", e.Method, e.Path)
}

func main() {
	endpoints := []Endpoint{
		{GET, "/api/users"},
		{POST, "/api/users"},
		{PUT, "/api/users/:id"},
		{DELETE, "/api/users/:id"},
	}

	for _, ep := range endpoints {
		fmt.Println(ep)
	}
}

package main

import "fmt"

func MaxAssert(a, b any) any {
	switch va := a.(type) {
	case int:
		vb := b.(int)
		if va > vb {
			return va
		}
		return vb
	case float64:
		vb := b.(float64)
		if va > vb {
			return va
		}
		return vb
	default:
		panic(fmt.Sprintf("不支持的类型: %T", a))
	}
}

type Ordered interface {
	~int | ~int8 | ~int16 | ~int32 | ~int64 |
		~uint | ~uint8 | ~uint16 | ~uint32 | ~uint64 |
		~float32 | ~float64 | ~string
}

func MaxGeneric[T Ordered](a, b T) T {
	if a > b {
		return a
	}
	return b
}

func main() {
	fmt.Println("断言版:", MaxAssert(3, 5))
	fmt.Println("泛型版:", MaxGeneric(3, 5))
	fmt.Println("泛型版:", MaxGeneric("abc", "xyz"))
}
