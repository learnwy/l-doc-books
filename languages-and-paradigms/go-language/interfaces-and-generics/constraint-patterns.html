Constraint Patterns 约束模式

核心概念

Go 1.18 引入泛型后，类型约束（Type Constraint）成为泛型系统的核心。约束定义了泛型类型参数必须满足的条件，使编译器能在编译期验证类型安全。

comparable 约束
comparable 是 Go 内建的约束，允许使用 == 和 != 操作符：


package main

import "fmt"

func Contains[T comparable](slice []T, target T) bool {
	for _, v := range slice {
		if v == target {
			return true
		}
	}
	return false
}

func IndexOf[T comparable](slice []T, target T) int {
	for i, v := range slice {
		if v == target {
			return i
		}
	}
	return -1
}

func Unique[T comparable](slice []T) []T {
	seen := make(map[T]struct{})
	result := make([]T, 0, len(slice))
	for _, v := range slice {
		if _, ok := seen[v]; !ok {
			seen[v] = struct{}{}
			result = append(result, v)
		}
	}
	return result
}

func main() {
	nums := []int{1, 2, 3, 4, 5, 3, 2, 1}
	fmt.Println("包含 3:", Contains(nums, 3))
	fmt.Println("3 的索引:", IndexOf(nums, 3))
	fmt.Println("去重:", Unique(nums))

	strs := []string{"go", "rust", "go", "python", "rust"}
	fmt.Println("去重:", Unique(strs))
}

类型联合约束
package main

import "fmt"

type Number interface {
	~int | ~int8 | ~int16 | ~int32 | ~int64 |
		~uint | ~uint8 | ~uint16 | ~uint32 | ~uint64 |
		~float32 | ~float64
}

func Sum[T Number](nums []T) T {
	var total T
	for _, n := range nums {
		total += n
	}
	return total
}

func Average[T Number](nums []T) float64 {
	if len(nums) == 0 {
		return 0
	}
	return float64(Sum(nums)) / float64(len(nums))
}

func MinMax[T Number](nums []T) (T, T) {
	if len(nums) == 0 {
		var zero T
		return zero, zero
	}
	min, max := nums[0], nums[0]
	for _, n := range nums[1:] {
		if n < min {
			min = n
		}
		if n > max {
			max = n
		}
	}
	return min, max
}

func main() {
	ints := []int{5, 3, 8, 1, 9, 2}
	fmt.Printf("Sum=%d, Avg=%.2f\n", Sum(ints), Average(ints))
	min, max := MinMax(ints)
	fmt.Printf("Min=%d, Max=%d\n", min, max)

	type Score float64
	scores := []Score{89.5, 92.3, 78.0, 95.7}
	fmt.Printf("Score Sum=%.1f\n", Sum(scores))
}
方法约束
package main

import (
	"fmt"
	"strings"
)

type Stringer interface {
	String() string
}

type Validator interface {
	Validate() error
}

type StringValidator interface {
	Stringer
	Validator
}

func ValidateAll[T Validator](items []T) []error {
	var errs []error
	for _, item := range items {
		if err := item.Validate(); err != nil {
			errs = append(errs, err)
		}
	}
	return errs
}

func JoinStrings[T Stringer](items []T, sep string) string {
	strs := make([]string, len(items))
	for i, item := range items {
		strs[i] = item.String()
	}
	return strings.Join(strs, sep)
}

type Email struct {
	Address string
}

func (e Email) String() string { return e.Address }
func (e Email) Validate() error {
	if !strings.Contains(e.Address, "@") {
		return fmt.Errorf("无效邮箱: %s", e.Address)
	}
	return nil
}

type Phone struct {
	Number string
}

func (p Phone) String() string { return p.Number }
func (p Phone) Validate() error {
	if len(p.Number) < 10 {
		return fmt.Errorf("号码太短: %s", p.Number)
	}
	return nil
}

func main() {
	emails := []Email{
		{Address: "alice@go.dev"},
		{Address: "invalid"},
		{Address: "bob@rust.dev"},
	}
	for _, err := range ValidateAll(emails) {
		fmt.Println("错误:", err)
	}
	fmt.Println("邮箱列表:", JoinStrings(emails, ", "))

	phones := []Phone{{Number: "13800138000"}, {Number: "123"}}
	for _, err := range ValidateAll(phones) {
		fmt.Println("错误:", err)
	}
}
混合约束（方法+类型）
package main

import (
	"encoding/json"
	"fmt"
)

type Serializable interface {
	~struct{} | ~string | ~int | ~float64
	MarshalJSON() ([]byte, error)
}

type JsonNumber interface {
	~int | ~int32 | ~int64 | ~float32 | ~float64
}

type Addable interface {
	JsonNumber
	comparable
}

func SumAndEncode[T Addable](nums []T) (T, string) {
	var total T
	for _, n := range nums {
		total += n
	}
	data, _ := json.Marshal(total)
	return total, string(data)
}

func main() {
	sum, encoded := SumAndEncode([]int{1, 2, 3, 4, 5})
	fmt.Printf("Sum=%d, JSON=%s\n", sum, encoded)

	sumF, encodedF := SumAndEncode([]float64{1.1, 2.2, 3.3})
	fmt.Printf("Sum=%.1f, JSON=%s\n", sumF, encodedF)
}
近似约束（~）
~T 表示底层类型为 T 的所有类型，包括 T 本身和以 T 为底层类型的自定义类型：


package main

import "fmt"

type Meter float64
type Kilogram float64
type Second float64

type PhysicalUnit interface {
	~float64
}

func Convert[T PhysicalUnit](value T, factor float64) T {
	return T(float64(value) * factor)
}

type Stringish interface {
	~string
}

func ToUpper[T Stringish](s T) T {
	bytes := []byte(string(s))
	for i, b := range bytes {
		if b >= 'a' && b <= 'z' {
			bytes[i] = b - 32
		}
	}
	return T(bytes)
}

type Name string
type Code string

func main() {
	distance := Meter(100)
	fmt.Printf("%.1f m = %.1f km\n", distance, Convert(distance, 0.001))

	weight := Kilogram(75.5)
	fmt.Printf("%.1f kg = %.1f g\n", weight, Convert(weight, 1000))

	name := Name("alice")
	fmt.Println(ToUpper(name))

	code := Code("abc123")
	fmt.Println(ToUpper(code))
}

约束组合
通过接口嵌入实现约束的组合：


package main

import (
	"fmt"
	"sort"
)

type Ordered interface {
	~int | ~int8 | ~int16 | ~int32 | ~int64 |
		~uint | ~uint8 | ~uint16 | ~uint32 | ~uint64 |
		~float32 | ~float64 | ~string
}

type OrderedSlice[T Ordered] []T

func (s OrderedSlice[T]) Len() int           { return len(s) }
func (s OrderedSlice[T]) Less(i, j int) bool { return s[i] < s[j] }
func (s OrderedSlice[T]) Swap(i, j int)      { s[i], s[j] = s[j], s[i] }

func Sort[T Ordered](s []T) []T {
	result := make([]T, len(s))
	copy(result, s)
	sort.Sort(OrderedSlice[T](result))
	return result
}

func BinarySearch[T Ordered](sorted []T, target T) int {
	lo, hi := 0, len(sorted)-1
	for lo <= hi {
		mid := lo + (hi-lo)/2
		if sorted[mid] == target {
			return mid
		} else if sorted[mid] < target {
			lo = mid + 1
		} else {
			hi = mid - 1
		}
	}
	return -1
}

func Filter[T any](slice []T, predicate func(T) bool) []T {
	result := make([]T, 0)
	for _, v := range slice {
		if predicate(v) {
			result = append(result, v)
		}
	}
	return result
}

func Map[T any, U any](slice []T, transform func(T) U) []U {
	result := make([]U, len(slice))
	for i, v := range slice {
		result[i] = transform(v)
	}
	return result
}

func Reduce[T any, U any](slice []T, initial U, reducer func(U, T) U) U {
	acc := initial
	for _, v := range slice {
		acc = reducer(acc, v)
	}
	return acc
}

func main() {
	nums := []int{5, 3, 8, 1, 9, 2, 7}
	sorted := Sort(nums)
	fmt.Println("排序:", sorted)
	fmt.Println("查找 8:", BinarySearch(sorted, 8))

	evens := Filter(nums, func(n int) bool { return n%2 == 0 })
	fmt.Println("偶数:", evens)

	doubled := Map(nums, func(n int) int { return n * 2 })
	fmt.Println("翻倍:", doubled)

	strs := Map(nums, func(n int) string { return fmt.Sprintf("#%d", n) })
	fmt.Println("转字符串:", strs)

	sum := Reduce(nums, 0, func(acc, n int) int { return acc + n })
	fmt.Println("求和:", sum)
}

泛型栈
package main

import (
	"errors"
	"fmt"
)

type Stack[T any] struct {
	items []T
}

func NewStack[T any]() *Stack[T] {
	return &Stack[T]{}
}

func (s *Stack[T]) Push(item T) {
	s.items = append(s.items, item)
}

func (s *Stack[T]) Pop() (T, error) {
	if len(s.items) == 0 {
		var zero T
		return zero, errors.New("栈为空")
	}
	item := s.items[len(s.items)-1]
	s.items = s.items[:len(s.items)-1]
	return item, nil
}

func (s *Stack[T]) Peek() (T, error) {
	if len(s.items) == 0 {
		var zero T
		return zero, errors.New("栈为空")
	}
	return s.items[len(s.items)-1], nil
}

func (s *Stack[T]) Len() int {
	return len(s.items)
}

func main() {
	intStack := NewStack[int]()
	intStack.Push(1)
	intStack.Push(2)
	intStack.Push(3)
	fmt.Printf("栈长度: %d\n", intStack.Len())
	for intStack.Len() > 0 {
		v, _ := intStack.Pop()
		fmt.Println("弹出:", v)
	}

	strStack := NewStack[string]()
	strStack.Push("hello")
	strStack.Push("world")
	top, _ := strStack.Peek()
	fmt.Println("栈顶:", top)
}
泛型有序集合
package main

import "fmt"

type OrderedSet[T comparable] struct {
	items  []T
	lookup map[T]struct{}
}

func NewOrderedSet[T comparable]() *OrderedSet[T] {
	return &OrderedSet[T]{
		lookup: make(map[T]struct{}),
	}
}

func (s *OrderedSet[T]) Add(item T) bool {
	if _, ok := s.lookup[item]; ok {
		return false
	}
	s.lookup[item] = struct{}{}
	s.items = append(s.items, item)
	return true
}

func (s *OrderedSet[T]) Contains(item T) bool {
	_, ok := s.lookup[item]
	return ok
}

func (s *OrderedSet[T]) Items() []T {
	result := make([]T, len(s.items))
	copy(result, s.items)
	return result
}

func (s *OrderedSet[T]) Len() int {
	return len(s.items)
}

func main() {
	set := NewOrderedSet[string]()
	set.Add("Go")
	set.Add("Rust")
	set.Add("Go")
	set.Add("Python")

	fmt.Println("元素:", set.Items())
	fmt.Println("长度:", set.Len())
	fmt.Println("包含 Go:", set.Contains("Go"))
	fmt.Println("包含 Java:", set.Contains("Java"))
}
泛型最佳实践




实践
说明





优先使用具体类型
不需要泛型时不要用泛型




使用最严格的约束
约束越紧，类型安全越强




any 适用于容器
不需要操作元素时用 any




comparable 适用于查找
需要 == 比较时




避免过度泛型化
只有 2+ 类型需要时才抽象




约束命名要有意义
Number 优于 T1，语义清晰




用 ~ 支持自定义类型
除非有特殊原因，否则用近似约束





小结
类型约束是 Go 泛型系统的基石，合理使用约束可以在保持类型安全的同时实现代码复用。
