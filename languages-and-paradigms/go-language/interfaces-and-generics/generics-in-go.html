泛型

核心概念

Go 1.18 引入了泛型（Generics），这是 Go 语言自诞生以来最重大的语言特性变化。泛型通过类型参数和约束，让函数和类型可以处理多种数据类型，同时保持编译时类型安全。在泛型之前，Go 开发者需要使用 interface{} + 类型断言或代码生成来实现通用逻辑。

泛型函数
基本语法
package main

import "fmt"

func Max[T int | float64 | string](a, b T) T {
    if a > b {
        return a
    }
    return b
}

func Swap[T any](a, b T) (T, T) {
    return b, a
}

func Contains[T comparable](s []T, target T) bool {
    for _, v := range s {
        if v == target {
            return true
        }
    }
    return false
}

func main() {
    fmt.Println(Max(3, 5))
    fmt.Println(Max(3.14, 2.71))
    fmt.Println(Max("hello", "world"))

    a, b := Swap(10, 20)
    fmt.Println(a, b)

    c, d := Swap("foo", "bar")
    fmt.Println(c, d)

    fmt.Println(Contains([]int{1, 2, 3}, 2))
    fmt.Println(Contains([]string{"a", "b"}, "c"))
}
类型推断
大多数情况下不需要显式指定类型参数，编译器会自动推断：


package main

import "fmt"

func Map[T, U any](s []T, fn func(T) U) []U {
    result := make([]U, len(s))
    for i, v := range s {
        result[i] = fn(v)
    }
    return result
}

func Filter[T any](s []T, fn func(T) bool) []T {
    var result []T
    for _, v := range s {
        if fn(v) {
            result = append(result, v)
        }
    }
    return result
}

func Reduce[T, U any](s []T, init U, fn func(U, T) U) U {
    acc := init
    for _, v := range s {
        acc = fn(acc, v)
    }
    return acc
}

func main() {
    nums := []int{1, 2, 3, 4, 5}

    doubled := Map(nums, func(n int) int { return n * 2 })
    fmt.Println("doubled:", doubled)

    strs := Map(nums, func(n int) string {
        return fmt.Sprintf("#%d", n)
    })
    fmt.Println("strings:", strs)

    evens := Filter(nums, func(n int) bool { return n%2 == 0 })
    fmt.Println("evens:", evens)

    sum := Reduce(nums, 0, func(acc, n int) int { return acc + n })
    fmt.Println("sum:", sum)

    _ = Map[int, string](nums, func(n int) string {
        return fmt.Sprint(n)
    })
}

自定义约束
package main

import (
    "fmt"
    "strings"
)

type Number interface {
    ~int | ~int8 | ~int16 | ~int32 | ~int64 |
    ~uint | ~uint8 | ~uint16 | ~uint32 | ~uint64 |
    ~float32 | ~float64
}

type Ordered interface {
    Number | ~string
}

func Sum[T Number](nums []T) T {
    var total T
    for _, n := range nums {
        total += n
    }
    return total
}

func Min[T Ordered](a, b T) T {
    if a < b {
        return a
    }
    return b
}

type Stringer interface {
    ~string
    String() string
}

type MyString string

func (s MyString) String() string {
    return strings.ToUpper(string(s))
}

func PrintUpper[T Stringer](s T) {
    fmt.Println(s.String())
}

func main() {
    fmt.Println(Sum([]int{1, 2, 3, 4, 5}))
    fmt.Println(Sum([]float64{1.1, 2.2, 3.3}))

    fmt.Println(Min(3, 5))
    fmt.Println(Min("alpha", "beta"))

    PrintUpper(MyString("hello"))
}
波浪号 ~ 的含义
~T 表示底层类型是 T 的所有类型：


package main

import "fmt"

type Celsius float64
type Fahrenheit float64

type Float interface {
    ~float32 | ~float64
}

func Abs[T Float](x T) T {
    if x < 0 {
        return -x
    }
    return x
}

func main() {
    var temp Celsius = -10.5
    fmt.Println(Abs(temp))

    var f Fahrenheit = -20.3
    fmt.Println(Abs(f))

    fmt.Println(Abs(-3.14))
}

package main

import "fmt"

type Formatter interface {
    Format() string
}

type User struct {
    Name string
    Age  int
}

func (u User) Format() string {
    return fmt.Sprintf("%s(%d)", u.Name, u.Age)
}

type Product struct {
    Name  string
    Price float64
}

func (p Product) Format() string {
    return fmt.Sprintf("%s: ¥%.2f", p.Name, p.Price)
}

func FormatAll[T Formatter](items []T) []string {
    result := make([]string, len(items))
    for i, item := range items {
        result[i] = item.Format()
    }
    return result
}

func main() {
    users := []User{{Name: "Alice", Age: 30}, {Name: "Bob", Age: 25}}
    fmt.Println(FormatAll(users))

    products := []Product{{Name: "手机", Price: 4999}, {Name: "笔记本", Price: 8999}}
    fmt.Println(FormatAll(products))
}
泛型类型
泛型切片
package main

import "fmt"

type Stack[T any] struct {
    items []T
}

func NewStack[T any]() *Stack[T] {
    return &Stack[T]{}
}

func (s *Stack[T]) Push(item T) {
    s.items = append(s.items, item)
}

func (s *Stack[T]) Pop() (T, bool) {
    if len(s.items) == 0 {
        var zero T
        return zero, false
    }
    last := len(s.items) - 1
    item := s.items[last]
    s.items = s.items[:last]
    return item, true
}

func (s *Stack[T]) Peek() (T, bool) {
    if len(s.items) == 0 {
        var zero T
        return zero, false
    }
    return s.items[len(s.items)-1], true
}

func (s *Stack[T]) Len() int {
    return len(s.items)
}

func main() {
    intStack := NewStack[int]()
    intStack.Push(1)
    intStack.Push(2)
    intStack.Push(3)
    v, _ := intStack.Pop()
    fmt.Println("Pop:", v)

    strStack := NewStack[string]()
    strStack.Push("hello")
    strStack.Push("world")
    top, _ := strStack.Peek()
    fmt.Println("Peek:", top)
}
泛型 Map（字典）
package main

import "fmt"

type OrderedMap[K comparable, V any] struct {
    keys   []K
    values map[K]V
}

func NewOrderedMap[K comparable, V any]() *OrderedMap[K, V] {
    return &OrderedMap[K, V]{
        values: make(map[K]V),
    }
}

func (m *OrderedMap[K, V]) Set(key K, value V) {
    if _, exists := m.values[key]; !exists {
        m.keys = append(m.keys, key)
    }
    m.values[key] = value
}

func (m *OrderedMap[K, V]) Get(key K) (V, bool) {
    v, ok := m.values[key]
    return v, ok
}

func (m *OrderedMap[K, V]) Range(fn func(K, V) bool) {
    for _, key := range m.keys {
        if !fn(key, m.values[key]) {
            break
        }
    }
}

func main() {
    om := NewOrderedMap[string, int]()
    om.Set("c", 3)
    om.Set("a", 1)
    om.Set("b", 2)

    om.Range(func(k string, v int) bool {
        fmt.Printf("%s=%d\n", k, v)
        return true
    })
}
泛型 Result 类型
package main

import (
    "errors"
    "fmt"
    "strconv"
)

type Result[T any] struct {
    value T
    err   error
}

func Ok[T any](value T) Result[T] {
    return Result[T]{value: value}
}

func Err[T any](err error) Result[T] {
    return Result[T]{err: err}
}

func (r Result[T]) Unwrap() (T, error) {
    return r.value, r.err
}

func (r Result[T]) IsOk() bool {
    return r.err == nil
}

func Then[T, U any](r Result[T], fn func(T) Result[U]) Result[U] {
    if r.err != nil {
        return Err[U](r.err)
    }
    return fn(r.value)
}

func parseAge(s string) Result[int] {
    n, err := strconv.Atoi(s)
    if err != nil {
        return Err[int](fmt.Errorf("无效年龄: %w", err))
    }
    if n < 0 || n > 150 {
        return Err[int](errors.New("年龄超出范围"))
    }
    return Ok(n)
}

func categorize(age int) Result[string] {
    switch {
    case age < 18:
        return Ok("未成年")
    case age < 65:
        return Ok("成年")
    default:
        return Ok("老年")
    }
}

func main() {
    r1 := parseAge("25")
    r2 := Then(r1, categorize)
    if v, err := r2.Unwrap(); err == nil {
        fmt.Println("类别:", v)
    }

    r3 := parseAge("abc")
    r4 := Then(r3, categorize)
    if _, err := r4.Unwrap(); err != nil {
        fmt.Println("错误:", err)
    }
}

泛型最佳实践
package main

import (
    "fmt"
    "sort"
)

type Pair[T, U any] struct {
    First  T
    Second U
}

func NewPair[T, U any](first T, second U) Pair[T, U] {
    return Pair[T, U]{First: first, Second: second}
}

func Keys[K comparable, V any](m map[K]V) []K {
    keys := make([]K, 0, len(m))
    for k := range m {
        keys = append(keys, k)
    }
    return keys
}

func Values[K comparable, V any](m map[K]V) []V {
    values := make([]V, 0, len(m))
    for _, v := range m {
        values = append(values, v)
    }
    return values
}

func GroupBy[T any, K comparable](items []T, keyFn func(T) K) map[K][]T {
    groups := make(map[K][]T)
    for _, item := range items {
        key := keyFn(item)
        groups[key] = append(groups[key], item)
    }
    return groups
}

func main() {
    p := NewPair("name", 42)
    fmt.Printf("Pair: %+v\n", p)

    m := map[string]int{"c": 3, "a": 1, "b": 2}
    keys := Keys(m)
    sort.Strings(keys)
    fmt.Println("Keys:", keys)
    fmt.Println("Values:", Values(m))

    words := []string{"apple", "avocado", "banana", "blueberry", "cherry"}
    groups := GroupBy(words, func(s string) byte { return s[0] })
    for k, v := range groups {
        fmt.Printf("%c: %v\n", k, v)
    }
}
小结
Go 泛型通过类型参数和约束接口，在保持语言简洁性的同时提供了强大的代码复用能力。


核心要点回顾：


    

  • 泛型通过 [T constraint] 语法声明类型参数
    • 

  • any 允许任意类型，comparable 允许可比较类型
    • 

  • ~T 匹配底层类型为 T 的所有类型，用于自定义类型
    • 

  • 类型推断让大多数场景不需要显式指定类型参数
    • 

  • 泛型适合通用数据结构和集合操作，不适合行为差异大的场景
    • 

  • 接口用于行为抽象，泛型用于类型参数化，两者互补
    • 

  • 泛型函数在编译时实例化，无运行时开销
    • 

  • 遵循"先用接口，需要时再用泛型"的原则
