Interface Design 接口设计

接口核心概念

Go 的接口是一组方法签名的集合，任何类型只要实现了接口要求的所有方法，就自动满足（实现）该接口——这就是隐式实现。无需 implements 关键字，编译器在赋值时自动检查。

package main

import (
	"fmt"
	"io"
	"strings"
)

type CountingReader struct {
	reader    io.Reader
	bytesRead int64
}

func NewCountingReader(r io.Reader) *CountingReader {
	return &CountingReader{reader: r}
}

func (cr *CountingReader) Read(p []byte) (int, error) {
	n, err := cr.reader.Read(p)
	cr.bytesRead += int64(n)
	return n, err
}

func (cr *CountingReader) BytesRead() int64 {
	return cr.bytesRead
}

func main() {
	r := strings.NewReader("Hello, Go Interface!")
	cr := NewCountingReader(r)

	buf := make([]byte, 1024)
	n, _ := cr.Read(buf)
	fmt.Printf("读取了 %d 字节，内容: %s\n", cr.BytesRead(), string(buf[:n]))
}
io.Reader 和 io.Writer
io.Reader 和 io.Writer 是 Go 标准库中最重要的两个接口，贯穿整个 I/O 体系。


package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"
	"strings"
)

type LogWriter struct {
	dest io.Writer
}

func (lw *LogWriter) Write(p []byte) (int, error) {
	prefix := []byte("[LOG] ")
	data := append(prefix, p...)
	return lw.dest.Write(data)
}

func main() {
	var readers []io.Reader
	readers = append(readers, strings.NewReader("来自字符串\n"))
	readers = append(readers, bytes.NewBufferString("来自 Buffer\n"))

	multi := io.MultiReader(readers...)
	io.Copy(os.Stdout, multi)

	var buf bytes.Buffer
	lw := &LogWriter{dest: &buf}
	fmt.Fprint(lw, "应用启动成功\n")
	fmt.Fprint(lw, "监听端口 8080\n")
	fmt.Print(buf.String())

	data := map[string]string{"name": "Go", "version": "1.22"}
	var jsonBuf bytes.Buffer
	json.NewEncoder(&jsonBuf).Encode(data)
	fmt.Println(jsonBuf.String())
}

接口组合
Go 通过接口嵌入实现接口的组合，这是 Go 面向组合而非继承的核心体现：


package main

import (
	"fmt"
	"io"
)

type Reader interface {
	Read(p []byte) (n int, err error)
}

type Writer interface {
	Write(p []byte) (n int, err error)
}

type Closer interface {
	Close() error
}

type ReadWriteCloser interface {
	Reader
	Writer
	Closer
}

type Flusher interface {
	Flush() error
}

type BufferedWriteCloser interface {
	Writer
	Closer
	Flusher
}

type MockRWC struct{}

func (m *MockRWC) Read(p []byte) (int, error)  { return 0, io.EOF }
func (m *MockRWC) Write(p []byte) (int, error) { return len(p), nil }
func (m *MockRWC) Close() error                { return nil }

func process(rwc ReadWriteCloser) {
	defer rwc.Close()
	rwc.Write([]byte("hello"))
	buf := make([]byte, 1024)
	rwc.Read(buf)
}

func main() {
	m := &MockRWC{}
	process(m)
	fmt.Println("接口组合示例完成")
}
接口断言
接口断言用于从接口值中提取底层具体类型或判断是否实现了另一个接口：


package main

import (
	"fmt"
	"io"
	"os"
	"strings"
)

func checkCapabilities(r io.Reader) {
	if ws, ok := r.(io.WriterTo); ok {
		fmt.Printf("  类型 %T 支持 WriterTo\n", r)
		_ = ws
	}

	if rs, ok := r.(io.Seeker); ok {
		fmt.Printf("  类型 %T 支持 Seeker\n", r)
		_ = rs
	}

	if rc, ok := r.(io.Closer); ok {
		fmt.Printf("  类型 %T 支持 Closer\n", r)
		_ = rc
	}
}

func main() {
	fmt.Println("strings.Reader:")
	checkCapabilities(strings.NewReader("hello"))

	fmt.Println("os.File:")
	f, _ := os.Open("/dev/null")
	defer f.Close()
	checkCapabilities(f)
}
空接口
interface{}（Go 1.18+ 可使用别名 any）可以持有任何类型的值，是 Go 的"万能类型"：


package main

import "fmt"

type Metadata map[string]any

func describe(val any) string {
	switch v := val.(type) {
	case nil:
		return "nil"
	case int:
		return fmt.Sprintf("整数: %d", v)
	case string:
		return fmt.Sprintf("字符串: %q", v)
	case []any:
		return fmt.Sprintf("切片，长度: %d", len(v))
	case map[string]any:
		return fmt.Sprintf("映射，键数: %d", len(v))
	default:
		return fmt.Sprintf("未知类型: %T", v)
	}
}

func main() {
	values := []any{
		42,
		"hello",
		[]any{1, 2, 3},
		map[string]any{"key": "value"},
		nil,
		3.14,
	}

	for _, v := range values {
		fmt.Println(describe(v))
	}
}

接口与方法集
值接收者和指针接收者对接口实现有不同的影响：


package main

import "fmt"

type Speaker interface {
	Speak() string
}

type Muter interface {
	Mute()
}

type Dog struct {
	Name string
	muted bool
}

func (d Dog) Speak() string {
	if d.muted {
		return "..."
	}
	return fmt.Sprintf("%s: Woof!", d.Name)
}

func (d *Dog) Mute() {
	d.muted = true
}

func main() {
	d := Dog{Name: "Buddy"}

	var s Speaker = d
	fmt.Println(s.Speak())

	var m Muter = &d
	m.Mute()
	fmt.Println(d.Speak())
}

接口最佳实践
在消费者端定义接口
Go 的惯例是在使用接口的包中定义接口，而非在实现者的包中：


package main

import "fmt"

type UserStore interface {
	GetUser(id string) (User, error)
	SaveUser(user User) error
}

type User struct {
	ID   string
	Name string
}

type UserService struct {
	store UserStore
}

func NewUserService(store UserStore) *UserService {
	return &UserService{store: store}
}

func (s *UserService) GetUserName(id string) (string, error) {
	u, err := s.store.GetUser(id)
	if err != nil {
		return "", err
	}
	return u.Name, nil
}

type MemoryStore struct {
	data map[string]User
}

func NewMemoryStore() *MemoryStore {
	return &MemoryStore{data: make(map[string]User)}
}

func (m *MemoryStore) GetUser(id string) (User, error) {
	u, ok := m.data[id]
	if !ok {
		return User{}, fmt.Errorf("用户 %s 不存在", id)
	}
	return u, nil
}

func (m *MemoryStore) SaveUser(user User) error {
	m.data[user.ID] = user
	return nil
}

func main() {
	store := NewMemoryStore()
	store.SaveUser(User{ID: "1", Name: "Alice"})

	svc := NewUserService(store)
	name, err := svc.GetUserName("1")
	if err != nil {
		fmt.Println("错误:", err)
		return
	}
	fmt.Println("用户名:", name)
}
接口用于 Mock 测试
package main

import (
	"fmt"
	"time"
)

type Clock interface {
	Now() time.Time
}

type RealClock struct{}

func (RealClock) Now() time.Time {
	return time.Now()
}

type MockClock struct {
	FixedTime time.Time
}

func (m MockClock) Now() time.Time {
	return m.FixedTime
}

type Scheduler struct {
	clock Clock
}

func (s *Scheduler) IsWorkingHours() bool {
	hour := s.clock.Now().Hour()
	return hour >= 9 && hour < 18
}

func main() {
	realScheduler := &Scheduler{clock: RealClock{}}
	fmt.Println("当前是否工作时间:", realScheduler.IsWorkingHours())

	mockClock := MockClock{FixedTime: time.Date(2024, 1, 1, 10, 0, 0, 0, time.UTC)}
	testScheduler := &Scheduler{clock: mockClock}
	fmt.Println("Mock 10:00 是否工作时间:", testScheduler.IsWorkingHours())

	mockClock2 := MockClock{FixedTime: time.Date(2024, 1, 1, 22, 0, 0, 0, time.UTC)}
	testScheduler2 := &Scheduler{clock: mockClock2}
	fmt.Println("Mock 22:00 是否工作时间:", testScheduler2.IsWorkingHours())
}
