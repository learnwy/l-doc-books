类型与变量

核心概念

Go 是静态强类型语言，所有变量在编译时必须确定类型。Go 的类型系统既保证了类型安全，又通过类型推断降低了声明冗余。理解 Go 的类型体系是掌握这门语言的第一步。

    fmt.Printf("int16:      size=%d, align=%d\n", unsafe.Sizeof(int16(0)), unsafe.Alignof(int16(0)))
    fmt.Printf("int32:      size=%d, align=%d\n", unsafe.Sizeof(int32(0)), unsafe.Alignof(int32(0)))
    fmt.Printf("int64:      size=%d, align=%d\n", unsafe.Sizeof(int64(0)), unsafe.Alignof(int64(0)))
    fmt.Printf("int:        size=%d, align=%d\n", unsafe.Sizeof(int(0)), unsafe.Alignof(int(0)))

无类型常量
Go 的常量可以是"无类型"的，这意味着它们拥有更高的精度，并且可以灵活地参与不同类型的运算：


package main

import "fmt"

const (
    Big   = 1 << 62
    Small = Big >> 60
)

const Pi = 3.14159265358979323846264338327950288

func main() {
    fmt.Println(Small)

    var f32 float32 = Pi
    var f64 float64 = Pi
    fmt.Printf("float32: %.20f\n", f32)
    fmt.Printf("float64: %.20f\n", f64)

    needInt := Small * 10
    needFloat := Small * 0.1
    fmt.Println(needInt, needFloat)
}
iota 常量生成器
iota 是 Go 特有的常量生成器，在 const 块中从 0 开始自增：


package main

import "fmt"

const (
    Sunday = iota
    Monday
    Tuesday
    Wednesday
    Thursday
    Friday
    Saturday
)

const (
    _  = iota
    KB = 1 << (10 * iota)
    MB
    GB
    TB
    PB
)

const (
    ReadPerm   = 1 << iota
    WritePerm
    ExecPerm
)

func main() {
    fmt.Println("Wednesday =", Wednesday)
    fmt.Printf("KB=%d, MB=%d, GB=%d, TB=%d, PB=%d\n", KB, MB, GB, TB, PB)

    perm := ReadPerm | WritePerm
    fmt.Printf("ReadWrite = %03b\n", perm)
    fmt.Println("Has Read:", perm&ReadPerm != 0)
    fmt.Println("Has Exec:", perm&ExecPerm != 0)
}

零值
Go 中所有变量在声明时如果不赋值会自动初始化为"零值"，这保证了变量始终可用：


package main

import "fmt"

type User struct {
    Name   string
    Age    int
    Active bool
}

func main() {
    var i int
    var f float64
    var b bool
    var s string
    var p *int
    var sl []int
    var m map[string]int
    var ch chan int
    var fn func()
    var iface interface{}
    var u User

    fmt.Printf("int:       %v\n", i)
    fmt.Printf("float64:   %v\n", f)
    fmt.Printf("bool:      %v\n", b)
    fmt.Printf("string:    %q\n", s)
    fmt.Printf("pointer:   %v\n", p)
    fmt.Printf("slice:     %v (nil=%v)\n", sl, sl == nil)
    fmt.Printf("map:       %v (nil=%v)\n", m, m == nil)
    fmt.Printf("chan:       %v (nil=%v)\n", ch, ch == nil)
    fmt.Printf("func:      %v (nil=%v)\n", fn, fn == nil)
    fmt.Printf("interface: %v (nil=%v)\n", iface, iface == nil)
    fmt.Printf("struct:    %+v\n", u)
}

类型转换
Go 不支持隐式类型转换，所有转换必须显式进行：


package main

import (
    "fmt"
    "math"
    "strconv"
)

func main() {
    var i int = 42
    var f float64 = float64(i)
    var u uint = uint(f)
    fmt.Println(i, f, u)

    var big float64 = 1e20
    var small int32 = int32(big)
    fmt.Println(small)

    var neg int = -1
    var unsigned uint = uint(neg)
    fmt.Printf("neg=%d, unsigned=%d\n", neg, unsigned)

    s1 := strconv.Itoa(42)
    fmt.Println(s1)

    n1, err := strconv.Atoi("42")
    fmt.Println(n1, err)

    f1, err := strconv.ParseFloat("3.14", 64)
    fmt.Println(f1, err)

    b1, err := strconv.ParseBool("true")
    fmt.Println(b1, err)

    s2 := fmt.Sprintf("%d", 42)
    s3 := fmt.Sprintf("%.2f", 3.14159)
    fmt.Println(s2, s3)

    _ = math.MaxInt32
}

类型别名与类型定义
package main

import "fmt"

type Celsius float64
type Fahrenheit float64

func (c Celsius) ToFahrenheit() Fahrenheit {
    return Fahrenheit(c*9/5 + 32)
}

func (f Fahrenheit) ToCelsius() Celsius {
    return Celsius((f - 32) * 5 / 9)
}

type UserID = int64
type OrderID = int64

func main() {
    var temp Celsius = 100
    fmt.Printf("%v°C = %v°F\n", temp, temp.ToFahrenheit())

    var uid UserID = 1001
    var oid OrderID = 2001
    fmt.Println(uid, oid)

    var n int64 = uid + oid
    fmt.Println(n)
}

字符串深入
string 内部结构
Go 的 string 是不可变的字节序列，内部由指针和长度两个字段组成。


字符串操作
package main

import (
    "fmt"
    "strings"
    "unicode/utf8"
)

func main() {
    s := "Hello, 世界!"

    fmt.Println("len(bytes):", len(s))
    fmt.Println("len(runes):", utf8.RuneCountInString(s))

    for i, r := range s {
        fmt.Printf("index=%d, rune=%c, unicode=%U\n", i, r, r)
    }

    bs := []byte(s)
    rs := []rune(s)
    fmt.Println("bytes:", bs)
    fmt.Println("runes:", rs)

    fmt.Println(strings.Contains(s, "世界"))
    fmt.Println(strings.HasPrefix(s, "Hello"))
    fmt.Println(strings.HasSuffix(s, "!"))
    fmt.Println(strings.Index(s, "世界"))
    fmt.Println(strings.ToUpper("hello"))
    fmt.Println(strings.TrimSpace("  hello  "))
    fmt.Println(strings.Split("a,b,c", ","))
    fmt.Println(strings.Join([]string{"a", "b", "c"}, "-"))
    fmt.Println(strings.Replace("aabaa", "a", "x", 2))
    fmt.Println(strings.ReplaceAll("aabaa", "a", "x"))
}
strings.Builder 高效拼接
package main

import (
    "fmt"
    "strings"
)

func main() {
    var b strings.Builder
    for i := 0; i < 100; i++ {
        fmt.Fprintf(&b, "item_%d ", i)
    }
    result := b.String()
    fmt.Println(len(result))
}

小结
Go 的类型系统追求简洁与安全的平衡。强类型保证了编译时错误检查，类型推断减少了代码冗余，零值机制确保了变量始终可用。


核心要点回顾：


    

  • Go 有 18 种基本类型，int/uint 大小与平台相关
    • 

  • 四种变量声明方式，短变量声明 := 最常用但仅限函数内
    • 

  • 常量支持无类型特性，iota 生成枚举和位掩码
    • 

  • 所有类型都有零值，nil 类型的零值使用需谨慎
    • 

  • 类型转换必须显式进行，strconv 包处理字符串与数值的转换
    • 

  • 类型定义创建新类型可添加方法，类型别名仅是同义词
    • 

  • 字符串是不可变的 UTF-8 字节序列，rune 表示 Unicode 码点
    • 

  • 高性能字符串拼接使用 strings.Builder
    •
