控制流

核心概念

Go 的控制流设计追求极简主义：只有 if、for、switch、select 四种控制结构，没有 while、do-while、三元运算符。这种设计减少了语言的复杂度，使代码更加统一和可读。

If/Else 条件判断
基本语法
Go 的 if 语句不需要括号包围条件表达式，但大括号是必须的：


package main

import "fmt"

func main() {
    score := 85

    if score >= 90 {
        fmt.Println("优秀")
    } else if score >= 80 {
        fmt.Println("良好")
    } else if score >= 60 {
        fmt.Println("及格")
    } else {
        fmt.Println("不及格")
    }
}
If 初始化语句
if 支持在条件前执行一个初始化语句，变量的作用域仅限于 if-else 块：


package main

import (
    "fmt"
    "os"
    "strconv"
)

func main() {
    if f, err := os.Open("config.json"); err != nil {
        fmt.Println("打开文件失败:", err)
    } else {
        fmt.Println("文件大小:", f.Name())
        f.Close()
    }

    if n, err := strconv.Atoi("42"); err == nil {
        fmt.Println("转换成功:", n)
    }

    if v := compute(); v > 100 {
        fmt.Println("超过阈值:", v)
    }
}

func compute() int {
    return 150
}

For 循环
Go 只有 for 一种循环结构，但它可以表达所有循环模式：


三段式 for（经典 C 风格）
package main

import "fmt"

func main() {
    for i := 0; i < 10; i++ {
        fmt.Print(i, " ")
    }
    fmt.Println()

    for i, j := 0, 10; i < j; i, j = i+1, j-1 {
        fmt.Printf("i=%d, j=%d\n", i, j)
    }
}
条件式 for（while 等价）
package main

import "fmt"

func main() {
    n := 1
    for n < 1000 {
        n *= 2
    }
    fmt.Println(n)
}
无限循环
package main

import "fmt"

func main() {
    count := 0
    for {
        if count >= 5 {
            break
        }
        fmt.Println(count)
        count++
    }
}
Range 遍历
range 可以遍历切片、数组、字符串、map 和 channel：


package main

import "fmt"

func main() {
    nums := []int{10, 20, 30, 40, 50}

    for i, v := range nums {
        fmt.Printf("index=%d, value=%d\n", i, v)
    }

    for _, v := range nums {
        fmt.Println(v)
    }

    for i := range nums {
        fmt.Println(i)
    }

    m := map[string]int{"a": 1, "b": 2, "c": 3}
    for k, v := range m {
        fmt.Printf("%s=%d\n", k, v)
    }

    s := "Hello, 世界"
    for i, r := range s {
        fmt.Printf("byte_index=%d, rune=%c\n", i, r)
    }

    ch := make(chan int, 3)
    ch <- 1
    ch <- 2
    ch <- 3
    close(ch)
    for v := range ch {
        fmt.Println(v)
    }
}

import "fmt"

    for i := 0; i < 10; i++ {
        if i%2 == 0 {
            continue
        }
        if i > 7 {
            break
        }
        fmt.Print(i, " ")

标签跳转
标签用于在嵌套循环中跳出外层循环：


package main

import "fmt"

func main() {
outer:
    for i := 0; i < 5; i++ {
        for j := 0; j < 5; j++ {
            if i+j == 6 {
                fmt.Printf("在 i=%d, j=%d 时跳出外层循环\n", i, j)
                break outer
            }
        }
    }

    fmt.Println("查找矩阵中的目标值:")
    matrix := [][]int{{1, 2, 3}, {4, 5, 6}, {7, 8, 9}}
    target := 5
search:
    for i, row := range matrix {
        for j, val := range row {
            if val == target {
                fmt.Printf("找到 %d 在 [%d][%d]\n", target, i, j)
                break search
            }
        }
    }
}
Switch 分支
基本 Switch
Go 的 switch 默认不会 fallthrough，每个 case 自动 break：


package main

import (
    "fmt"
    "runtime"
)

func main() {
    switch runtime.GOOS {
    case "darwin":
        fmt.Println("macOS")
    case "linux":
        fmt.Println("Linux")
    case "windows":
        fmt.Println("Windows")
    default:
        fmt.Println("其他系统:", runtime.GOOS)
    }
}
多值 Case
package main

import "fmt"

func classify(c byte) string {
    switch c {
    case 'a', 'e', 'i', 'o', 'u',
        'A', 'E', 'I', 'O', 'U':
        return "元音"
    case ' ', '\t', '\n':
        return "空白"
    default:
        return "其他"
    }
}

func main() {
    fmt.Println(classify('a'))
    fmt.Println(classify(' '))
    fmt.Println(classify('z'))
}
条件 Switch（无表达式）
package main

import (
    "fmt"
    "time"
)

func main() {
    hour := time.Now().Hour()

    switch {
    case hour < 6:
        fmt.Println("凌晨")
    case hour < 12:
        fmt.Println("上午")
    case hour < 14:
        fmt.Println("中午")
    case hour < 18:
        fmt.Println("下午")
    default:
        fmt.Println("晚上")
    }
}
Switch 初始化语句
package main

import (
    "fmt"
    "math/rand"

func main() {
    switch n := rand.Intn(10); {
    case n < 3:
        fmt.Printf("%d: 小\n", n)
    case n < 7:

Type Switch
类型断言 switch 用于判断接口的动态类型：


package main

import "fmt"

func describe(i interface{}) string {
    switch v := i.(type) {
    case int:
        return fmt.Sprintf("整数: %d", v)
    case float64:
        return fmt.Sprintf("浮点: %.2f", v)
    case string:
        return fmt.Sprintf("字符串: %q, 长度=%d", v, len(v))
    case bool:
        return fmt.Sprintf("布尔: %v", v)
    case []int:
        return fmt.Sprintf("整数切片: %v, 长度=%d", v, len(v))
    case nil:
        return "nil"
    default:
        return fmt.Sprintf("未知类型: %T", v)
    }
}

func main() {
    fmt.Println(describe(42))
    fmt.Println(describe(3.14))
    fmt.Println(describe("hello"))
    fmt.Println(describe(true))
    fmt.Println(describe([]int{1, 2}))
    fmt.Println(describe(nil))
}
Fallthrough
package main

import "fmt"

func main() {
    n := 3
    switch {
    case n > 0:
        fmt.Println("正数")
        fallthrough
    case n > -5:
        fmt.Println("大于 -5")
        fallthrough
    case n > -10:
        fmt.Println("大于 -10")
    }
}

Defer 延迟执行
基本用法
defer 将函数调用推迟到当前函数返回时执行，常用于资源清理：


package main

import (
    "fmt"
    "os"
)

func main() {
    f, err := os.Create("test.txt")
    if err != nil {
        fmt.Println(err)
        return
    }
    defer f.Close()

    f.WriteString("hello, defer!")
    fmt.Println("文件写入完成")
}
LIFO 执行顺序
多个 defer 按照后进先出（LIFO）顺序执行：


package main

import "fmt"

func main() {
    fmt.Println("开始")

    defer fmt.Println("第一个 defer")
    defer fmt.Println("第二个 defer")
    defer fmt.Println("第三个 defer")

    fmt.Println("结束")
}
Defer 参数求值时机
defer 语句的参数在 defer 执行时立即求值，而非延迟到调用时：


package main

import "fmt"

func main() {
    x := 10
    defer fmt.Println("defer 中的 x:", x)
    x = 20
    fmt.Println("当前 x:", x)

    y := 10
    defer func() {
        fmt.Println("闭包中的 y:", y)
    }()
    y = 20
    fmt.Println("当前 y:", y)
}

Panic 与 Recover
Panic
panic 会立即终止当前函数执行，逐层向上展开调用栈，执行每一层的 defer：


package main

import "fmt"

func a() {
    defer fmt.Println("defer in a")
    fmt.Println("执行 a")
    b()
    fmt.Println("a 结束")
}

func b() {
    defer fmt.Println("defer in b")
    fmt.Println("执行 b")
    panic("出错了!")
}

func main() {
    a()
}
Recover
recover 只能在 defer 函数中使用，用于捕获 panic 并恢复正常执行：


package main

import "fmt"

func safeDivide(a, b int) (result int, err error) {
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("recovered: %v", r)
        }
    }()

    return a / b, nil
}

func main() {
    result, err := safeDivide(10, 0)
    if err != nil {
        fmt.Println("错误:", err)
    } else {
        fmt.Println("结果:", result)
    }

    result, err = safeDivide(10, 3)
    fmt.Println("结果:", result, "错误:", err)
}
Panic/Recover 使用原则




原则
说明





优先使用 error
panic 仅用于不可恢复的错误




库不应 panic
库代码应返回 error，由调用者决定




recover 限定边界
在 goroutine 入口或 HTTP handler 中 recover




不要滥用
panic/recover 不是异常处理机制





HTTP 中间件 Recover 示例
package main

import (
    "fmt"
    "log"
    "net/http"
)

func recoveryMiddleware(next http.HandlerFunc) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        defer func() {
            if err := recover(); err != nil {
                log.Printf("panic recovered: %v", err)
                http.Error(w, "Internal Server Error", http.StatusInternalServerError)
            }
        }()
        next(w, r)
    }
}

func riskyHandler(w http.ResponseWriter, r *http.Request) {
    panic("意外错误")
}

func main() {
    http.HandleFunc("/", recoveryMiddleware(riskyHandler))
    fmt.Println("服务启动在 :8080")
    log.Fatal(http.ListenAndServe(":8080", nil))
}

package main

import (
    "errors"
    "fmt"
)

func processItem(id int) error {
    item, err := findItem(id)
    if err != nil {
        return fmt.Errorf("查找失败: %w", err)
    }

    if !item.IsValid() {
        return errors.New("无效的项目")
    }

    if err := item.Process(); err != nil {
        return fmt.Errorf("处理失败: %w", err)
    }

    return nil
}

type Item struct{ ID int }

func (i Item) IsValid() bool    { return i.ID > 0 }
func (i Item) Process() error   { return nil }
func findItem(id int) (Item, error) {
    if id <= 0 {
        return Item{}, errors.New("id must be positive")
    }
    return Item{ID: id}, nil
}

func main() {
    fmt.Println(processItem(1))
    fmt.Println(processItem(-1))
}
小结
Go 的控制流设计体现了"少即是多"的哲学，用最少的关键字覆盖所有常见场景。


核心要点回顾：


    

  • if 支持初始化语句，变量作用域限定在 if-else 块内
    • 

  • for 是唯一的循环结构，三种形式覆盖经典循环、while 循环和无限循环
    • 

  • range 遍历切片、map、字符串、channel，Go 1.22+ 支持整数范围
    • 

  • switch 默认不 fallthrough，支持多值 case、条件 switch 和类型 switch
    • 

  • defer 按 LIFO 顺序执行，参数在 defer 语句时求值
    • 

  • panic 触发栈展开，recover 在 defer 中捕获 panic
    • 

  • 提前返回减少嵌套，是 Go 代码风格的核心原则
