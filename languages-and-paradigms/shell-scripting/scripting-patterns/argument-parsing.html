参数解析

核心概念

命令行参数解析是 Shell 脚本从"一次性工具"升级为"可复用命令"的关键能力。一个设计良好的参数接口不仅让脚本更易使用，还能通过参数验证提升脚本的健壮性。Shell 脚本的参数解析主要有三种方式：手动解析、getopts 内置命令和外部 getopt 程序，它们在灵活性和复杂度之间各有取舍。

Unix/Linux 命令行遵循 POSIX 约定：短选项以单个连字符开头（-v），长选项以双连字符开头（--verbose），-- 标记选项结束。

手动参数解析
手动使用 while 循环和 case 语句是最灵活、最可移植的参数解析方式：


#!/usr/bin/env bash
set -euo pipefail

VERBOSE=false
DRY_RUN=false
OUTPUT=""
CONFIG_FILE="config.yml"
ARGS=()

usage() {
    cat << EOF
用法: $(basename "$0") [选项] <文件...>

选项:
  -v, --verbose         启用详细输出
  -n, --dry-run         模拟执行，不实际操作
  -o, --output <file>   指定输出文件
  -c, --config <file>   指定配置文件 (默认: config.yml)
  -h, --help            显示帮助信息
  --version             显示版本信息

示例:
  $(basename "$0") -v --output result.txt input.txt
  $(basename "$0") --dry-run -c prod.yml *.log
EOF
}

parse_args() {
    while (( $# > 0 )); do
        case "$1" in
            -v|--verbose)
                VERBOSE=true
                shift
                ;;
            -n|--dry-run)
                DRY_RUN=true
                shift
                ;;
            -o|--output)
                [[ -n "${2:-}" ]] || die "选项 $1 需要参数"
                OUTPUT="$2"
                shift 2
                ;;
            -c|--config)
                [[ -n "${2:-}" ]] || die "选项 $1 需要参数"
                CONFIG_FILE="$2"
                shift 2
                ;;
            -h|--help)
                usage
                exit 0
                ;;
            --version)
                echo "v1.0.0"
                exit 0
                ;;
            --)
                shift
                ARGS+=("$@")
                break
                ;;
            -*)
                die "未知选项: $1（使用 --help 查看帮助）"
                ;;
            *)
                ARGS+=("$1")
                shift
                ;;
        esac
    done
}

die() { echo "[ERROR] $*" >&2; exit 1; }

main() {
    parse_args "$@"

    (( ${#ARGS[@]} > 0 )) || die "缺少必要参数（使用 --help 查看帮助）"

    $VERBOSE && echo "配置文件: $CONFIG_FILE"
    $VERBOSE && echo "输出文件: ${OUTPUT:-stdout}"
    $VERBOSE && echo "处理文件: ${ARGS[*]}"

    for file in "${ARGS[@]}"; do
        [[ -f "$file" ]] || die "文件不存在: $file"
        process_file "$file"
    done
}

process_file() {
    local file="$1"
    if $DRY_RUN; then
        echo "[DRY RUN] 将处理: $file"
    else
        echo "处理: $file"
    fi
}

main "$@"
支持合并短选项（如 -vn 等同于 -v -n）：


parse_combined_short_opts() {
    while (( $# > 0 )); do
        case "$1" in
            -[!-]?*)
                local i=1
                while (( i < ${#1} )); do
                    local opt="-${1:i:1}"
                    case "$opt" in
                        -v) VERBOSE=true ;;
                        -n) DRY_RUN=true ;;
                        -o)
                            OUTPUT="${1:i+1}"
                            [[ -n "$OUTPUT" ]] || { shift; OUTPUT="${1:-}"; }
                            break
                            ;;
                        *) die "未知选项: $opt" ;;
                    esac
                    ((i++))
                done
                shift
                ;;
            *) break ;;
        esac
    done
}
getopts 内置命令
getopts 是 Bash 内置的参数解析器，语法简洁但仅支持短选项：


#!/usr/bin/env bash
set -euo pipefail

VERBOSE=false
OUTPUT=""
COUNT=1

usage() {
    echo "用法: $(basename "$0") [-v] [-o output] [-c count] file..."
}

while getopts ":vo:c:h" opt; do
    case "$opt" in
        v) VERBOSE=true ;;
        o) OUTPUT="$OPTARG" ;;
        c)
            if ! [[ "$OPTARG" =~ ^[0-9]+$ ]]; then
                die "选项 -c 的值必须是数字: $OPTARG"
            fi
            COUNT="$OPTARG"
            ;;
        h) usage; exit 0 ;;
        :) die "选项 -${OPTARG} 需要参数" ;;
        \?) die "未知选项: -${OPTARG}" ;;
    esac
done

shift $((OPTIND - 1))

(( $# > 0 )) || { usage; exit 1; }

echo "VERBOSE=$VERBOSE, OUTPUT=$OUTPUT, COUNT=$COUNT"
echo "文件参数: $*"

子命令模式
现代 CLI 工具（如 git、docker）广泛使用子命令模式，Shell 脚本也可以实现类似结构：


#!/usr/bin/env bash
set -euo pipefail

readonly VERSION="2.0.0"

die() { echo "[ERROR] $*" >&2; exit 1; }

usage() {
    cat << 'EOF'
用法: mytool <命令> [选项] [参数]

命令:
  init       初始化项目
  build      构建项目
  deploy     部署应用
  status     查看状态

全局选项:
  -h, --help       显示帮助
  --version        显示版本
  --debug          启用调试模式
EOF
}

cmd_init() {
    local template="default"
    local name=""

    while (( $# > 0 )); do
        case "$1" in
            -t|--template) template="$2"; shift 2 ;;
            *) name="$1"; shift ;;
        esac
    done

    [[ -n "$name" ]] || die "init 需要项目名称"
    echo "初始化项目: $name (模板: $template)"
}

cmd_build() {
    local target="production"
    local parallel=false

    while (( $# > 0 )); do
        case "$1" in
            --target) target="$2"; shift 2 ;;
            -j|--parallel) parallel=true; shift ;;
            *) die "build: 未知参数: $1" ;;
        esac
    done

    echo "构建目标: $target, 并行: $parallel"
}

cmd_deploy() {
    local env="${1:-staging}"
    local force=false

    shift || true
    while (( $# > 0 )); do
        case "$1" in
            -f|--force) force=true; shift ;;
            *) die "deploy: 未知参数: $1" ;;
        esac
    done

    echo "部署到: $env, 强制: $force"
}

main() {
    (( $# > 0 )) || { usage; exit 1; }

    case "$1" in
        -h|--help) usage; exit 0 ;;
        --version) echo "mytool v${VERSION}"; exit 0 ;;
    esac

    local cmd="$1"
    shift

    case "$cmd" in
        init)   cmd_init "$@" ;;
        build)  cmd_build "$@" ;;
        deploy) cmd_deploy "$@" ;;
        status) echo "状态: 正常" ;;
        *)      die "未知命令: $cmd（使用 --help 查看帮助）" ;;
    esac
}

main "$@"
参数验证与类型检查
严格的参数验证是脚本健壮性的保障：


validate_integer() {
    local name="$1" value="$2"
    [[ "$value" =~ ^-?[0-9]+$ ]] || die "$name 必须是整数: $value"
}

validate_positive_integer() {
    local name="$1" value="$2"
    [[ "$value" =~ ^[1-9][0-9]*$ ]] || die "$name 必须是正整数: $value"
}

validate_in_list() {
    local name="$1" value="$2"
    shift 2
    local valid=("$@")

    for v in "${valid[@]}"; do
        [[ "$value" == "$v" ]] && return 0
    done

    die "$name 必须是以下之一: ${valid[*]}，但收到: $value"
}

validate_file_extension() {
    local file="$1"
    shift
    local extensions=("$@")

    local ext="${file##*.}"
    for e in "${extensions[@]}"; do
        [[ "$ext" == "$e" ]] && return 0
    done

    die "不支持的文件格式: .$ext（支持: ${extensions[*]}）"
}

parse_key_value() {
    local input="$1"
    if [[ "$input" == *=* ]]; then
        local key="${input%%=*}"
        local value="${input#*=}"
        echo "$key" "$value"
    else
        die "无效的键值对格式: $input（期望: key=value）"
    fi
}

deploy() {
    local env="$1"
    local replicas="$2"
    local config="$3"

    validate_in_list "env" "$env" "development" "staging" "production"
    validate_positive_integer "replicas" "$replicas"
    [[ -f "$config" ]] || die "配置文件不存在: $config"

    echo "部署: env=$env, replicas=$replicas, config=$config"
}
小结
    

  • 手动 while+case 是最灵活的参数解析方式，支持长选项和短选项，推荐作为默认选择
    • 

  • getopts 内置命令适合简单脚本，语法简洁但仅支持短选项
    • 

  • 子命令模式适合复杂 CLI 工具，通过 case 分发到对应的命令处理函数
    • 

  • 每个选项的参数都应进行类型验证（整数、文件存在、枚举值等）
    • 

  • usage() 函数应包含完整的用法说明、选项列表和使用示例
    • 

  • -- 标记选项结束，其后的所有参数都被视为位置参数
    • 

  • shift 在参数解析中用于消耗已处理的参数，保持参数指针前进
    • 

  • 错误信息应友好且具有指导性，包含期望格式和有效值范围
    •
