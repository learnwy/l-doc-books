日志模式

核心概念

Shell 脚本中的日志系统是调试、监控和审计的基础设施。与应用程序的日志框架不同，Shell 日志需要在没有第三方库支持的情况下，仅依靠内置功能实现结构化输出、级别控制和文件轮转。一个好的日志系统能够在脚本出问题时快速定位根因，在正常运行时提供可观测性。

Shell 日志的设计需要平衡可读性和机器可解析性。人类可读的彩色终端输出适合交互式使用，而结构化的日志格式适合日志聚合系统（如 ELK、Loki）。

基础日志函数
一个最小可用的日志系统只需几个函数：


#!/usr/bin/env bash

LOG_LEVEL="${LOG_LEVEL:-1}"

readonly _LOG_LEVELS=([DEBUG]=0 [INFO]=1 [WARN]=2 [ERROR]=3 [FATAL]=4)

_log() {
    local level="$1"
    shift
    local level_num="${_LOG_LEVELS[$level]}"

    (( level_num >= LOG_LEVEL )) || return 0

    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')

    local caller_info=""
    if (( level_num >= 3 )); then
        caller_info=" (${BASH_SOURCE[2]:-unknown}:${BASH_LINENO[1]:-0})"
    fi

    if (( level_num >= 2 )); then
        echo "[$timestamp] [$level] $*${caller_info}" >&2
    else
        echo "[$timestamp] [$level] $*"
    fi
}

log_debug() { _log DEBUG "$@"; }
log_info()  { _log INFO "$@"; }
log_warn()  { _log WARN "$@"; }
log_error() { _log ERROR "$@"; }
log_fatal() { _log FATAL "$@"; exit 1; }

log_info "应用启动"
log_debug "配置加载完成"
log_warn "配置项 DB_POOL_SIZE 使用默认值"
log_error "连接数据库失败"
彩色终端输出
终端彩色输出显著提升日志可读性，但需要检测是否为终端环境：


if [[ -t 2 ]] && [[ "${NO_COLOR:-}" == "" ]]; then
    readonly _C_RED='\033[0;31m'
    readonly _C_YELLOW='\033[1;33m'
    readonly _C_GREEN='\033[0;32m'
    readonly _C_BLUE='\033[0;34m'
    readonly _C_GRAY='\033[0;90m'
    readonly _C_BOLD='\033[1m'
    readonly _C_RESET='\033[0m'
else
    readonly _C_RED='' _C_YELLOW='' _C_GREEN='' _C_BLUE=''
    readonly _C_GRAY='' _C_BOLD='' _C_RESET=''
fi

_log_colored() {
    local level="$1" color="$2"
    shift 2
    local level_num="${_LOG_LEVELS[$level]}"

    (( level_num >= LOG_LEVEL )) || return 0

    local ts
    ts=$(date '+%H:%M:%S')

    local msg="${color}[${level}]${_C_RESET} ${_C_GRAY}${ts}${_C_RESET} $*"

    if (( level_num >= 2 )); then
        echo -e "$msg" >&2
    else
        echo -e "$msg"
    fi
}

log_debug() { _log_colored DEBUG "$_C_GRAY"   "$@"; }
log_info()  { _log_colored INFO  "$_C_GREEN"  "$@"; }
log_warn()  { _log_colored WARN  "$_C_YELLOW" "$@"; }
log_error() { _log_colored ERROR "$_C_RED"    "$@"; }

log_step() {
    local step_num="$1" total="$2"
    shift 2
    echo -e "${_C_BOLD}[${step_num}/${total}]${_C_RESET} $*"
}

log_step 1 5 "安装依赖"
log_step 2 5 "运行测试"

spinner() {
    local pid=$1
    local message="${2:-处理中}"
    local chars='⠋⠙⠹⠸⠼⠴⠦⠧⠇⠏'
    local i=0

    while kill -0 "$pid" 2>/dev/null; do
        printf "\r${_C_BLUE}%s${_C_RESET} %s" "${chars:i%${#chars}:1}" "$message"
        ((i++))
        sleep 0.1
    done
    printf "\r${_C_GREEN}✓${_C_RESET} %s\n" "$message"
}

long_task() { sleep 3; }
long_task &
spinner $! "正在编译项目"
日志文件管理
生产环境中的日志需要写入文件并进行轮转管理：


setup_logging() {
    local log_dir="${1:-/var/log/myapp}"
    local log_name="${2:-app}"

    mkdir -p "$log_dir"

    LOG_FILE="${log_dir}/${log_name}-$(date +%Y%m%d).log"

    exec 3>&1 4>&2

    exec 1> >(tee -a "$LOG_FILE")
    exec 2> >(tee -a "$LOG_FILE" >&2)

    echo "日志文件: $LOG_FILE"
}

rotate_logs() {
    local log_dir="$1"
    local keep_days="${2:-30}"

    find "$log_dir" -name "*.log" -mtime +"$keep_days" -delete

    find "$log_dir" -name "*.log" -size +100M | while read -r logfile; do
        local rotated="${logfile}.$(date +%Y%m%d%H%M%S)"
        mv "$logfile" "$rotated"
        gzip "$rotated" &
    done

    log_info "日志轮转完成，保留最近 ${keep_days} 天"
}

log_with_rotation() {
    local msg="$1"
    local log_file="${LOG_FILE:-/tmp/app.log}"
    local max_size=$((10 * 1024 * 1024))  # 10MB

    if [[ -f "$log_file" ]] && (( $(stat -f%z "$log_file" 2>/dev/null || stat -c%s "$log_file" 2>/dev/null) > max_size )); then
        mv "$log_file" "${log_file}.1"
        [[ -f "${log_file}.1" ]] && gzip -f "${log_file}.1" &
    fi

    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $msg" >> "$log_file"
}
结构化日志与集成
现代日志系统通常需要 JSON 格式输出以便于日志聚合和分析：


log_json() {
    local level="$1"
    shift
    local message="$*"

    local timestamp
    timestamp=$(date -u '+%Y-%m-%dT%H:%M:%SZ')

    local escaped_msg
    escaped_msg=$(echo "$message" | sed 's/\\/\\\\/g; s/"/\\"/g')

    printf '{"timestamp":"%s","level":"%s","message":"%s","script":"%s","pid":%d}\n' \
        "$timestamp" "$level" "$escaped_msg" "${BASH_SOURCE[1]:-unknown}" "$$"
}

log_json_with_context() {
    local level="$1" message="$2"
    shift 2

    local timestamp
    timestamp=$(date -u '+%Y-%m-%dT%H:%M:%SZ')

    local extra=""
    while (( $# > 0 )); do
        local key="${1%%=*}"
        local value="${1#*=}"
        extra="${extra},\"${key}\":\"${value}\""
        shift
    done

    printf '{"timestamp":"%s","level":"%s","message":"%s"%s}\n' \
        "$timestamp" "$level" "$message" "$extra"
}

log_json_with_context "INFO" "用户登录" "user=admin" "ip=192.168.1.1" "method=password"

to_syslog() {
    local level="$1"
    shift
    local facility="local0"

    case "$level" in
        DEBUG) logger -p "${facility}.debug" -t "myapp" "$*" ;;
        INFO)  logger -p "${facility}.info" -t "myapp" "$*" ;;
        WARN)  logger -p "${facility}.warning" -t "myapp" "$*" ;;
        ERROR) logger -p "${facility}.err" -t "myapp" "$*" ;;
    esac
}

audit_log() {
    local action="$1" target="$2" result="$3"
    local user="${USER:-unknown}"
    local timestamp
    timestamp=$(date -u '+%Y-%m-%dT%H:%M:%SZ')

    printf '[AUDIT] %s user=%s action=%s target=%s result=%s\n' \
        "$timestamp" "$user" "$action" "$target" "$result" >> "${AUDIT_LOG:-/var/log/audit.log}"
}

audit_log "deploy" "production" "success"
audit_log "config_change" "database.yml" "success"
小结
    

  • Shell 日志系统需要实现级别控制、格式化输出和多目标写入三大功能
    • 

  • 日志级别遵循 DEBUG < INFO < WARN < ERROR < FATAL 的严重程度递增顺序
    • 

  • 彩色输出应检测终端环境并尊重 NO_COLOR 环境变量
    • 

  • WARN 及以上级别的日志应输出到 stderr，INFO 和 DEBUG 输出到 stdout
    • 

  • 生产环境需要日志文件轮转，按大小或日期切割并定期清理旧日志
    • 

  • JSON 结构化日志便于接入 ELK、Loki 等日志聚合系统
    • 

  • syslog 集成通过 logger 命令实现，适合系统级日志收集
    • 

  • 审计日志应记录操作者、动作、目标和结果，保存在独立的审计日志文件中
    •
