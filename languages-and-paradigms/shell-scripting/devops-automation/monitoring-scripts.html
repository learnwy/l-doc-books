监控脚本

核心概念

监控脚本是 DevOps 自动化中实现系统可观测性的基础工具。它们持续采集系统指标、检测异常状态并触发告警，是确保服务稳定运行的第一道防线。Shell 监控脚本的优势在于无需额外运行时依赖，能够直接调用系统命令获取底层数据，适合作为轻量级监控方案或专业监控系统的补充。

监控脚本的设计需要关注采集频率、告警阈值、数据存储和通知渠道四个核心维度。一个好的监控脚本应该做到低开销、低误报、高可维护性。

系统资源监控
系统资源监控是最基础的监控需求，覆盖 CPU、内存、磁盘和网络：


#!/usr/bin/env bash
set -euo pipefail

check_cpu() {
    local threshold="${1:-80}"
    local usage

    usage=$(top -l 1 -n 0 2>/dev/null | awk '/CPU usage/ {print $3}' | tr -d '%' || \
            mpstat 1 1 2>/dev/null | awk 'END {print 100-$NF}' || \
            awk '{printf "%.1f", ($1+$2+$3)*100/($1+$2+$3+$4)}' /proc/stat 2>/dev/null)

    local int_usage=${usage%.*}

    if (( int_usage > threshold )); then
        echo "WARNING: CPU 使用率 ${usage}% 超过阈值 ${threshold}%"
        return 1
    fi

    echo "OK: CPU 使用率 ${usage}%"
    return 0
}

check_memory() {
    local threshold="${1:-85}"
    local usage

    if [[ "$(uname)" == "Darwin" ]]; then
        local pages_free pages_active pages_speculative
        pages_free=$(vm_stat | awk '/Pages free/ {print $3}' | tr -d '.')
        pages_active=$(vm_stat | awk '/Pages active/ {print $3}' | tr -d '.')
        pages_speculative=$(vm_stat | awk '/Pages speculative/ {print $3}' | tr -d '.')
        local total_pages=$((pages_free + pages_active + pages_speculative))
        usage=$(( (total_pages - pages_free) * 100 / total_pages ))
    else
        usage=$(free | awk 'NR==2 {printf "%.0f", $3/$2*100}')
    fi

    if (( usage > threshold )); then
        echo "WARNING: 内存使用率 ${usage}% 超过阈值 ${threshold}%"
        return 1
    fi

    echo "OK: 内存使用率 ${usage}%"
    return 0
}

check_disk() {
    local threshold="${1:-80}"
    local issues=()

    while IFS= read -r line; do
        local mount usage_pct
        mount=$(echo "$line" | awk '{print $6}')
        usage_pct=$(echo "$line" | awk '{print $5}' | tr -d '%')

        if (( usage_pct > threshold )); then
            issues+=("${mount}: ${usage_pct}%")
        fi
    done < <(df -h | awk 'NR>1 && $1 !~ /^(tmpfs|devtmpfs|overlay)/')

    if (( ${#issues[@]} > 0 )); then
        echo "WARNING: 磁盘使用率超过阈值:"
        printf "  %s\n" "${issues[@]}"
        return 1
    fi

    echo "OK: 所有磁盘使用率低于 ${threshold}%"
    return 0
}

run_all_checks() {
    local failures=0

    check_cpu 80   || ((failures++))
    check_memory 85 || ((failures++))
    check_disk 80   || ((failures++))

    echo "---"
    echo "检查完成: $failures 个告警"
    return $failures
}
服务健康检查
服务级监控检测应用的可用性和响应质量：


check_http() {
    local url="$1"
    local timeout="${2:-10}"
    local expected_code="${3:-200}"

    local start_time end_time response_code response_time
    start_time=$(date +%s%N)

    response_code=$(curl -s -o /dev/null -w "%{http_code}" \
        --connect-timeout "$timeout" \
        --max-time "$((timeout * 2))" \
        "$url" 2>/dev/null || echo "000")

    end_time=$(date +%s%N)
    response_time=$(( (end_time - start_time) / 1000000 ))

    if [[ "$response_code" != "$expected_code" ]]; then
        echo "CRITICAL: $url 返回 $response_code (期望 $expected_code), 耗时 ${response_time}ms"
        return 2
    fi

    if (( response_time > 2000 )); then
        echo "WARNING: $url 响应缓慢 ${response_time}ms"
        return 1
    fi

    echo "OK: $url → $response_code (${response_time}ms)"
    return 0
}

check_port() {
    local host="$1"
    local port="$2"
    local timeout="${3:-5}"

    if nc -z -w "$timeout" "$host" "$port" 2>/dev/null; then
        echo "OK: ${host}:${port} 可达"
        return 0
    else
        echo "CRITICAL: ${host}:${port} 不可达"
        return 2
    fi
}

check_process() {
    local process_name="$1"
    local min_count="${2:-1}"

    local count
    count=$(pgrep -fc "$process_name" 2>/dev/null || echo 0)

    if (( count < min_count )); then
        echo "CRITICAL: 进程 $process_name 实例数 $count (最低 $min_count)"
        return 2
    fi

    echo "OK: 进程 $process_name 运行 $count 个实例"
    return 0
}

check_service_suite() {
    local -A services=(
        ["Web 服务"]="check_http http://localhost:8080/health"
        ["API 服务"]="check_http http://localhost:3000/api/health"
        ["数据库"]="check_port localhost 5432"
        ["Redis"]="check_port localhost 6379"
        ["Nginx"]="check_process nginx 2"
    )

    local failures=0

    for name in "${!services[@]}"; do
        local result
        if result=$(${services[$name]} 2>&1); then
            echo "[✓] $name: $result"
        else
            echo "[✗] $name: $result"
            ((failures++))
        fi
    done

    return $failures
}
告警通知系统
监控脚本需要在检测到异常时及时通知运维团队：


send_slack() {
    local webhook_url="$1"
    local message="$2"
    local severity="${3:-warning}"

    local color
    case "$severity" in
        critical) color="#ff0000" ;;
        warning)  color="#ffaa00" ;;
        ok)       color="#00ff00" ;;
        *)        color="#cccccc" ;;
    esac

    local hostname
    hostname=$(hostname -f 2>/dev/null || hostname)

    curl -sf -X POST "$webhook_url" \
        -H 'Content-type: application/json' \
        -d "{
            \"attachments\": [{
                \"color\": \"${color}\",
                \"title\": \"监控告警 [${severity^^}]\",
                \"text\": \"${message}\",
                \"fields\": [
                    {\"title\": \"主机\", \"value\": \"${hostname}\", \"short\": true},
                    {\"title\": \"时间\", \"value\": \"$(date '+%Y-%m-%d %H:%M:%S')\", \"short\": true}
                ]
            }]
        }" > /dev/null
}

send_dingtalk() {
    local webhook_url="$1"
    local message="$2"

    curl -sf -X POST "$webhook_url" \
        -H 'Content-Type: application/json' \
        -d "{
            \"msgtype\": \"text\",
            \"text\": {
                \"content\": \"[监控告警] ${message}\"
            }
        }" > /dev/null
}

readonly ALERT_STATE_DIR="/tmp/monitor_alerts"
mkdir -p "$ALERT_STATE_DIR"

should_alert() {
    local key="$1"
    local cooldown="${2:-300}"
    local state_file="${ALERT_STATE_DIR}/${key}"

    if [[ -f "$state_file" ]]; then
        local last_alert
        last_alert=$(cat "$state_file")
        local now
        now=$(date +%s)

        if (( now - last_alert < cooldown )); then
            return 1
        fi
    fi

    date +%s > "$state_file"
    return 0
}

alert() {
    local severity="$1"
    local message="$2"
    local key="${3:-default}"

    if ! should_alert "$key" 300; then
        return 0
    fi

    local webhook="${SLACK_WEBHOOK:-}"
    if [[ -n "$webhook" ]]; then
        send_slack "$webhook" "$message" "$severity"
    fi

    echo "[ALERT] [$severity] $message" >> /var/log/monitor/alerts.log
}
持续监控守护进程
将监控脚本组织为后台持续运行的守护进程：


#!/usr/bin/env bash
set -euo pipefail

readonly CHECK_INTERVAL=30
readonly PID_FILE="/var/run/monitor.pid"

monitor_loop() {
    log_info "监控守护进程启动 (PID: $$, 间隔: ${CHECK_INTERVAL}s)"

    echo $$ > "$PID_FILE"
    trap 'rm -f "$PID_FILE"; log_info "监控守护进程退出"; exit 0' EXIT INT TERM

    while true; do
        local check_start
        check_start=$(date +%s)

        run_checks

        local check_end elapsed sleep_time
        check_end=$(date +%s)
        elapsed=$((check_end - check_start))
        sleep_time=$((CHECK_INTERVAL - elapsed))

        if (( sleep_time > 0 )); then
            sleep "$sleep_time"
        fi
    done
}

run_checks() {
    check_cpu 80 || alert "warning" "CPU 使用率过高" "cpu"
    check_memory 85 || alert "warning" "内存使用率过高" "memory"
    check_disk 90 || alert "critical" "磁盘空间不足" "disk"
    check_http "http://localhost:8080/health" || alert "critical" "Web 服务不可用" "web"
}

log_info() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] [INFO] $*"; }

case "${1:-start}" in
    start)
        if [[ -f "$PID_FILE" ]] && kill -0 "$(cat "$PID_FILE")" 2>/dev/null; then
            echo "监控进程已在运行 (PID: $(cat "$PID_FILE"))"
            exit 1
        fi
        monitor_loop &
        echo "监控守护进程已启动 (PID: $!)"
        ;;
    stop)
        if [[ -f "$PID_FILE" ]]; then
            kill "$(cat "$PID_FILE")" 2>/dev/null
            echo "监控守护进程已停止"
        fi
        ;;
    status)
        if [[ -f "$PID_FILE" ]] && kill -0 "$(cat "$PID_FILE")" 2>/dev/null; then
            echo "运行中 (PID: $(cat "$PID_FILE"))"
        else
            echo "未运行"
        fi
        ;;
esac
小结
    

  • 系统资源监控覆盖 CPU、内存、磁盘和网络四大维度，需要适配不同操作系统（macOS/Linux）
    • 

  • 服务健康检查包括 HTTP 探针、端口检测和进程存活检查三种方式
    • 

  • 告警需要冷却机制（debounce），避免同一问题重复告警造成告警疲劳
    • 

  • Webhook 集成（Slack/钉钉）是最轻量级的告警通知方案
    • 

  • 持续监控应组织为守护进程模式，支持 start/stop/status 子命令
    • 

  • PID 文件用于防止重复启动和支持优雅停止
    • 

  • 监控阈值应根据业务特点和历史数据动态调整，避免固定阈值导致的误报
    • 

  • 对于大规模环境，Shell 监控脚本应作为 Prometheus/Grafana 等专业方案的补充
    •
