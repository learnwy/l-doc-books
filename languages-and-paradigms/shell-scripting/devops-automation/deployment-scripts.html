部署脚本

核心概念

部署脚本是将应用代码从构建产物推送到目标环境的自动化工具。一个生产级部署脚本需要具备原子性（要么完全成功要么完全回滚）、幂等性（多次执行结果一致）、可观测性（清晰的日志和状态报告）三大特性。在容器化和 CI/CD 管道日益普及的今天，Shell 部署脚本仍然是"最后一英里"自动化的核心组件。

部署流程通常包括构建、传输、安装、验证、切换五个阶段。每个阶段都需要完善的错误处理和回滚机制，确保在任何环节失败时系统能够恢复到稳定状态。

符号链接部署模式
符号链接部署是传统服务器环境中最经典的零停机部署方式。通过切换 current 符号链接指向新版本目录实现瞬间切换：


#!/usr/bin/env bash
set -euo pipefail

readonly DEPLOY_ROOT="/opt/app"
readonly RELEASES_DIR="${DEPLOY_ROOT}/releases"
readonly CURRENT_LINK="${DEPLOY_ROOT}/current"
readonly SHARED_DIR="${DEPLOY_ROOT}/shared"
readonly KEEP_RELEASES=5

deploy() {
    local version="$1"
    local artifact="$2"
    local release_dir="${RELEASES_DIR}/${version}"

    log_info "开始部署 v${version}"

    mkdir -p "$release_dir"

    tar xzf "$artifact" -C "$release_dir"

    link_shared_resources "$release_dir"

    run_hooks "$release_dir" "pre-deploy"

    local previous
    previous=$(readlink -f "$CURRENT_LINK" 2>/dev/null || echo "")

    ln -sfn "$release_dir" "${CURRENT_LINK}.new"
    mv -T "${CURRENT_LINK}.new" "$CURRENT_LINK"

    if ! verify_deployment; then
        log_error "部署验证失败，执行回滚"
        if [[ -n "$previous" ]]; then
            ln -sfn "$previous" "${CURRENT_LINK}.new"
            mv -T "${CURRENT_LINK}.new" "$CURRENT_LINK"
            reload_service
        fi
        return 1
    fi

    reload_service
    run_hooks "$release_dir" "post-deploy"
    cleanup_old_releases

    log_info "部署 v${version} 完成"
}

link_shared_resources() {
    local release_dir="$1"

    ln -sfn "${SHARED_DIR}/config" "${release_dir}/config"
    ln -sfn "${SHARED_DIR}/logs" "${release_dir}/logs"
    ln -sfn "${SHARED_DIR}/uploads" "${release_dir}/uploads"
    ln -sfn "${SHARED_DIR}/.env" "${release_dir}/.env"
}

cleanup_old_releases() {
    local releases
    mapfile -t releases < <(ls -1t "$RELEASES_DIR")

    local count=${#releases[@]}
    if (( count > KEEP_RELEASES )); then
        for ((i = KEEP_RELEASES; i < count; i++)); do
            local old="${RELEASES_DIR}/${releases[i]}"
            log_info "删除旧版本: ${releases[i]}"
            rm -rf "$old"
        done
    fi
}
远程部署与 SSH
通过 SSH 执行远程部署是分布式环境中的标准方式：


readonly SSH_OPTS="-o StrictHostKeyChecking=accept-new -o ConnectTimeout=10 -o BatchMode=yes"

deploy_to_remote() {
    local host="$1"
    local artifact="$2"
    local version="$3"

    log_info "部署到 ${host}: v${version}"

    ssh $SSH_OPTS "$host" "mkdir -p /opt/app/releases/${version}"

    rsync -avz --progress \
        -e "ssh ${SSH_OPTS}" \
        "$artifact" \
        "${host}:/opt/app/releases/${version}/"

    ssh $SSH_OPTS "$host" bash -s "$version" << 'REMOTE_SCRIPT'
        set -euo pipefail
        version="$1"
        release_dir="/opt/app/releases/${version}"
        current_link="/opt/app/current"

        ln -sfn "$release_dir" "${current_link}.new"
        mv -T "${current_link}.new" "$current_link"

        sudo systemctl reload app

        sleep 5
        if curl -sf http://localhost:8080/health > /dev/null; then
            echo "部署验证通过"
        else
            echo "部署验证失败" >&2
            exit 1
        fi
REMOTE_SCRIPT

    log_info "${host} 部署完成"
}

deploy_to_fleet() {
    local artifact="$1"
    local version="$2"
    shift 2
    local hosts=("$@")

    local failed=()

    for host in "${hosts[@]}"; do
        if ! deploy_to_remote "$host" "$artifact" "$version"; then
            failed+=("$host")
            log_error "部署失败: $host"
        fi
    done

    if (( ${#failed[@]} > 0 )); then
        log_error "以下主机部署失败: ${failed[*]}"
        return 1
    fi

    log_info "所有 ${#hosts[@]} 台主机部署成功"
}

deploy_to_fleet "app-v1.2.3.tar.gz" "1.2.3" \
    "web1.example.com" \
    "web2.example.com" \
    "web3.example.com"
容器部署脚本
容器环境的部署脚本需要处理镜像构建、推送和编排更新：


deploy_docker() {
    local version="$1"
    local registry="registry.example.com"
    local image="${registry}/myapp"
    local tag="${image}:${version}"

    log_info "构建 Docker 镜像: $tag"
    docker build \
        --build-arg "VERSION=${version}" \
        --build-arg "BUILD_TIME=$(date -u +%Y-%m-%dT%H:%M:%SZ)" \
        -t "$tag" \
        -t "${image}:latest" \
        .

    log_info "推送镜像到仓库"
    docker push "$tag"
    docker push "${image}:latest"

    log_info "更新服务"
    docker service update \
        --image "$tag" \
        --update-parallelism 1 \
        --update-delay 30s \
        --update-failure-action rollback \
        --rollback-parallelism 1 \
        myapp-service

    log_info "等待服务稳定..."
    local attempts=0
    local max_attempts=30
    while (( attempts < max_attempts )); do
        local running
        running=$(docker service ps myapp-service \
            --filter "desired-state=running" \
            --format "{{.CurrentState}}" | grep -c "Running" || true)

        if (( running >= 3 )); then
            log_info "服务已稳定（$running 个实例运行中）"
            return 0
        fi

        ((attempts++))
        sleep 10
    done

    log_error "服务未能在预期时间内稳定"
    return 1
}

deploy_compose() {
    local version="$1"
    local env="${2:-production}"
    local compose_file="docker-compose.${env}.yml"

    export APP_VERSION="$version"

    docker compose -f "$compose_file" pull
    docker compose -f "$compose_file" up -d --remove-orphans

    docker compose -f "$compose_file" ps
}
健康检查与验证
部署后的健康检查是确保服务正常运行的最后防线：


verify_deployment() {
    local url="${1:-http://localhost:8080}"
    local max_attempts="${2:-30}"
    local delay="${3:-2}"

    log_info "执行部署验证..."

    local attempt=0
    while (( attempt < max_attempts )); do
        if curl -sf "${url}/health" > /dev/null 2>&1; then
            log_info "健康检查通过"
            break
        fi
        ((attempt++))
        sleep "$delay"
    done

    if (( attempt >= max_attempts )); then
        log_error "健康检查超时（${max_attempts} 次尝试）"
        return 1
    fi

    local version_response
    version_response=$(curl -sf "${url}/version" 2>/dev/null || echo "")
    if [[ -n "$version_response" ]]; then
        log_info "运行版本: $version_response"
    fi

    local status_code
    status_code=$(curl -s -o /dev/null -w "%{http_code}" "${url}/")
    if (( status_code >= 200 && status_code < 400 )); then
        log_info "HTTP 状态码: $status_code"
    else
        log_error "异常状态码: $status_code"
        return 1
    fi

    log_info "部署验证全部通过"
    return 0
}

smoke_test() {
    local base_url="$1"
    local endpoints=(
        "GET /api/health 200"
        "GET /api/version 200"
        "GET /api/users 200"
        "POST /api/login 401"
    )

    local failed=0
    for spec in "${endpoints[@]}"; do
        local method url expected_code
        read -r method url expected_code <<< "$spec"

        local actual_code
        actual_code=$(curl -s -o /dev/null -w "%{http_code}" -X "$method" "${base_url}${url}")

        if [[ "$actual_code" == "$expected_code" ]]; then
            log_info "✓ $method $url → $actual_code"
        else
            log_error "✗ $method $url → $actual_code (期望 $expected_code)"
            ((failed++))
        fi
    done

    if (( failed > 0 )); then
        log_error "冒烟测试失败: $failed 个端点异常"
        return 1
    fi

    log_info "冒烟测试全部通过"
}
小结
    

  • 部署脚本需要满足原子性、幂等性和可观测性三大特性
    • 

  • 符号链接切换是传统服务器环境中实现零停机部署的经典方式
    • 

  • 远程部署通过 SSH + rsync 组合实现，heredoc 注入远程脚本执行
    • 

  • 容器部署需要处理镜像构建、推送和编排更新的完整流程
    • 

  • 健康检查是部署验证的最后防线，应包括端口检测、HTTP 状态码和业务冒烟测试
    • 

  • 回滚机制应在部署前就准备好，确保失败时能快速恢复
    • 

  • 旧版本清理策略应保留最近 N 个版本，便于快速回滚
    • 

  • 多主机部署应记录每台主机的部署状态，支持部分失败时的定向修复
    •
