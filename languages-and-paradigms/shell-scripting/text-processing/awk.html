awk 编程

核心概念

awk 是一种功能完备的文本处理语言，以其发明者 Aho、Weinberger 和 Kernighan 三人姓氏首字母命名。与 grep 的行匹配和 sed 的流编辑不同，awk 的核心能力在于结构化文本处理——它天然理解字段（列）的概念，将每行输入自动拆分为字段，并提供完整的编程结构（变量、数组、函数、控制流）进行数据处理。

awk 程序由一组"模式-动作"规则组成：pattern { action }。对于输入的每一行，awk 依次检查所有规则，如果行匹配模式，则执行对应的动作。

字段处理与内置变量
awk 将输入行自动按分隔符拆分为字段，这是其最强大的特性：


echo "John 28 Engineer" | awk '{print $1, $3}'

awk -F: '{print $1, $3, $7}' /etc/passwd

awk -F',' '{print $2, $4}' data.csv

awk '{print $NF}' file.txt       # 最后一个字段
awk '{print $(NF-1)}' file.txt   # 倒数第二个字段

echo "a:b:c:d" | awk -F: '{
    for (i = 1; i <= NF; i++)
        print "字段" i ": " $i
}'

echo "hello world" | awk '{$2 = "awk"; print $0}'

ps aux | awk '{printf "%-10s %5s %s\n", $1, $2, $11}'

ls -la | awk 'NR > 1 {total += $5} END {print "总大小:", total, "bytes"}'
awk 的内置变量提供了丰富的上下文信息：


awk '{
    print "行号:", NR
    print "文件行号:", FNR
    print "字段数:", NF
    print "当前文件:", FILENAME
    print "字段分隔符:", FS
    print "输出分隔符:", OFS
    print "记录分隔符:", RS
}' file.txt

awk 'BEGIN {
    FS = ","
    OFS = "\t"
}
{
    print $1, $2, $3
}' data.csv

awk 'BEGIN {
    RS = ""      # 空行作为记录分隔符（段落模式）
    FS = "\n"    # 换行作为字段分隔符
}
{
    print "段落", NR, ":"
    for (i = 1; i <= NF; i++)
        print "  行", i, ":", $i
}' paragraphs.txt
模式匹配与条件过滤
awk 的模式可以是正则表达式、条件表达式或特殊模式（BEGIN/END）：


awk '/error/' log.txt

awk '!/^#/ && !/^$/' config.txt

awk '$3 > 1000' data.txt

awk -F: '$3 >= 1000 && $7 ~ /bash/' /etc/passwd

awk '$0 ~ /pattern/' file.txt
awk '$2 ~ /^[A-Z]/' file.txt

awk '/START/,/END/' file.txt

awk 'NR >= 10 && NR <= 20' file.txt

awk 'NR == 1 || $3 > 100' data.csv

awk '
    length($0) > 120 {
        printf "%s:%d: 行长度 %d 超过 120\n", FILENAME, NR, length($0)
    }
' *.ts

awk '
    BEGIN { FS = "," }
    NR == 1 { header = $0; next }
    $4 == "ERROR" {
        if (!printed_header) {
            print header
            printed_header = 1
        }
        print
    }
' report.csv
编程结构与数据处理
awk 提供了完整的编程能力，包括变量、数组、循环和函数：


awk '
    BEGIN {
        FS = ","
        print "=== 销售统计 ==="
    }
    NR > 1 {
        total += $3
        count++
        if ($3 > max) { max = $3; max_item = $1 }
        if ($3 < min || NR == 2) { min = $3; min_item = $1 }
    }
    END {
        printf "总计: %.2f\n", total
        printf "平均: %.2f\n", total / count
        printf "最高: %s (%.2f)\n", max_item, max
        printf "最低: %s (%.2f)\n", min_item, min
    }
' sales.csv

awk '{
    words[NR] = $0
    for (i = 1; i <= NF; i++) {
        freq[$i]++
    }
}
END {
    for (word in freq)
        if (freq[word] > 1)
            printf "%4d %s\n", freq[word], word
}' text.txt | sort -rn | head -20

awk -F: '{
    shell[$7]++
}
END {
    for (s in shell)
        printf "%4d %s\n", shell[s], s
}' /etc/passwd | sort -rn

access_log_stats() {
    awk '{
        status[$9]++
        ip[$1]++
        url[$7]++
        total++
    }
    END {
        print "=== 状态码统计 ==="
        for (s in status)
            printf "  %s: %d (%.1f%%)\n", s, status[s], status[s]*100/total

        print "\n=== Top 10 IP ==="
        for (i in ip)
            printf "%6d %s\n", ip[i], i | "sort -rn | head -10"

        print "\n=== Top 10 URL ==="
        for (u in url)
            printf "%6d %s\n", url[u], u | "sort -rn | head -10"
    }' "$1"
}
关联数组是 awk 最强大的数据结构：


awk '
    BEGIN { FS = "," }
    NR > 1 {
        dept_total[$2] += $4
        dept_count[$2]++
    }
    END {
        printf "%-15s %10s %8s %10s\n", "部门", "总额", "人数", "平均"
        printf "%-15s %10s %8s %10s\n", "---", "---", "---", "---"
        for (dept in dept_total) {
            avg = dept_total[dept] / dept_count[dept]
            printf "%-15s %10.2f %8d %10.2f\n", dept, dept_total[dept], dept_count[dept], avg
        }
    }
' employees.csv

awk '
    NR == FNR {
        lookup[$1] = $2
        next
    }
    {
        key = $3
        if (key in lookup)
            print $0, lookup[key]
        else
            print $0, "N/A"
    }
' lookup_table.txt main_data.txt
自定义函数与高级技巧
awk 支持自定义函数，可以封装复用逻辑：


awk '
    function trim(s) {
        gsub(/^[[:space:]]+|[[:space:]]+$/, "", s)
        return s
    }
    function human_size(bytes) {
        if (bytes >= 1073741824)
            return sprintf("%.1fG", bytes / 1073741824)
        else if (bytes >= 1048576)
            return sprintf("%.1fM", bytes / 1048576)
        else if (bytes >= 1024)
            return sprintf("%.1fK", bytes / 1024)
        else
            return bytes "B"
    }
    {
        name = trim($1)
        size = $5 + 0
        printf "%-20s %8s\n", name, human_size(size)
    }
' <(ls -la)

awk '
    function parse_json_value(line, key,    re, val) {
        re = "\"" key "\"[[:space:]]*:[[:space:]]*\"([^\"]*)\""
        if (match(line, re)) {
            val = substr(line, RSTART, RLENGTH)
            gsub(/.*": *"/, "", val)
            gsub(/"$/, "", val)
            return val
        }
        return ""
    }
    {
        name = parse_json_value($0, "name")
        version = parse_json_value($0, "version")
        if (name != "")
            printf "%s@%s\n", name, version
    }
' package.json

awk '
    BEGIN {
        srand()
    }
    {
        lines[NR] = $0
    }
    END {
        n = NR
        for (i = n; i > 1; i--) {
            j = int(rand() * i) + 1
            temp = lines[i]
            lines[i] = lines[j]
            lines[j] = temp
        }
        for (i = 1; i <= n; i++)
            print lines[i]
    }
' input.txt

awk -v threshold=100 '
    $3 > threshold {
        printf "\033[31m%s\033[0m\n", $0
    }
    $3 <= threshold {
        print
    }
' data.txt
小结
    

  • awk 是功能完备的文本处理语言，核心优势在于字段自动拆分和结构化数据处理
    • 

  • 程序结构为 BEGIN { 初始化 } pattern { 动作 } END { 收尾 }
    • 

  • $0 表示整行，$1..$NF 表示各字段，NR 为全局行号，FNR 为文件内行号
    • 

  • -F 指定字段分隔符，支持正则表达式分隔符
    • 

  • 关联数组是 awk 最强大的数据结构，适合统计、分组和查找操作
    • 

  • awk 支持自定义函数，可以封装复杂的文本处理逻辑
    • 

  • grep 适合搜索、sed 适合替换、awk 适合结构化数据处理——三者组合使用是 Shell 文本处理的核心
    • 

  • 对于超过 50 行的 awk 程序，建议迁移到 Python 或其他通用编程语言
    •
