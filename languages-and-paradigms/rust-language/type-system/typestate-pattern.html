类型状态模式

核心思想

类型状态模式（Typestate Pattern）利用 Rust 的类型系统在编译期编码状态机，使非法的状态转换成为编译错误而非运行时 bug。

传统方法用枚举或布尔标记状态，状态检查在运行时完成；类型状态模式让每个状态成为独立类型，非法转换直接无法通过编译。




基础实现：文章发布流程
struct Draft;
struct PendingReview;
struct Published;

struct Article<State> {
    title: String,
    content: String,
    state: std::marker::PhantomData<State>,
}

impl Article<Draft> {
    fn new(title: String, content: String) -> Self {
        Article {
            title,
            content,
            state: std::marker::PhantomData,
        }
    }

    fn submit(self) -> Article<PendingReview> {
        println!("文章 '{}' 已提交审核", self.title);
        Article {
            title: self.title,
            content: self.content,
            state: std::marker::PhantomData,
        }
    }
}

impl Article<PendingReview> {
    fn approve(self) -> Article<Published> {
        println!("文章 '{}' 审核通过", self.title);
        Article {
            title: self.title,
            content: self.content,
            state: std::marker::PhantomData,
        }
    }

    fn reject(self) -> Article<Draft> {
        println!("文章 '{}' 被退回", self.title);
        Article {
            title: self.title,
            content: self.content,
            state: std::marker::PhantomData,
        }
    }
}

impl Article<Published> {
    fn view(&self) -> &str {
        &self.content
    }
}

fn main() {
    let article = Article::<Draft>::new(
        "Rust 类型状态".into(),
        "利用类型系统编码状态机...".into(),
    );

    let article = article.submit();
    let article = article.approve();
    println!("已发布内容: {}", article.view());

    // article.submit(); // 编译错误：Published 状态没有 submit 方法
}

建造者模式与类型状态
类型状态模式非常适合实现编译期安全的建造者（Builder），确保必填字段不被遗漏。


struct NoHost;
struct HasHost(String);
struct NoPort;
struct HasPort(u16);

struct ServerConfigBuilder<H, P> {
    host: H,
    port: P,
    max_connections: Option<u32>,
}

impl ServerConfigBuilder<NoHost, NoPort> {
    fn new() -> Self {
        ServerConfigBuilder {
            host: NoHost,
            port: NoPort,
            max_connections: None,
        }
    }
}

impl<P> ServerConfigBuilder<NoHost, P> {
    fn host(self, host: &str) -> ServerConfigBuilder<HasHost, P> {
        ServerConfigBuilder {
            host: HasHost(host.to_string()),
            port: self.port,
            max_connections: self.max_connections,
        }
    }
}

impl<H> ServerConfigBuilder<H, NoPort> {
    fn port(self, port: u16) -> ServerConfigBuilder<H, HasPort> {
        ServerConfigBuilder {
            host: self.host,
            port: HasPort(port),
            max_connections: self.max_connections,
        }
    }
}

impl<H, P> ServerConfigBuilder<H, P> {
    fn max_connections(mut self, max: u32) -> Self {
        self.max_connections = Some(max);
        self
    }
}

impl ServerConfigBuilder<HasHost, HasPort> {
    fn build(self) -> ServerConfig {
        ServerConfig {
            host: self.host.0,
            port: self.port.0,
            max_connections: self.max_connections.unwrap_or(100),
        }
    }
}

struct ServerConfig {
    host: String,
    port: u16,
    max_connections: u32,
}

fn main() {
    let config = ServerConfigBuilder::new()
        .host("0.0.0.0")
        .port(8080)
        .max_connections(500)
        .build();

    // ServerConfigBuilder::new().build(); // 编译错误：缺少 host 和 port

    println!("服务器: {}:{}", config.host, config.port);
}


Sealed Trait 封装状态
使用 Sealed Trait 防止外部代码定义新状态，确保状态转换的完整性。


mod state {
    mod sealed {
        pub trait Sealed {}
    }

    pub trait ConnectionState: sealed::Sealed {}

    pub struct Disconnected;
    pub struct Connected;
    pub struct Authenticated;

    impl sealed::Sealed for Disconnected {}
    impl sealed::Sealed for Connected {}
    impl sealed::Sealed for Authenticated {}

    impl ConnectionState for Disconnected {}
    impl ConnectionState for Connected {}
    impl ConnectionState for Authenticated {}
}

use state::*;

struct Connection<S: ConnectionState> {
    addr: String,
    _state: std::marker::PhantomData<S>,
}

impl Connection<Disconnected> {
    fn connect(self) -> Connection<Connected> {
        Connection {
            addr: self.addr,
            _state: std::marker::PhantomData,
        }
    }
}

impl Connection<Connected> {
    fn authenticate(self, _token: &str) -> Connection<Authenticated> {
        Connection {
            addr: self.addr,
            _state: std::marker::PhantomData,
        }
    }
}

impl Connection<Authenticated> {
    fn query(&self, sql: &str) -> String {
        format!("在 {} 上执行: {}", self.addr, sql)
    }
}
