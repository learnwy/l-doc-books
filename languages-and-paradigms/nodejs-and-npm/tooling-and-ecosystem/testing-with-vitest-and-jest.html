Vitest 与 Jest 测试

为什么需要测试

测试是保障代码质量的基础设施。在 Node.js 项目中，自动化测试不仅能捕获回归 bug，还能作为代码的活文档，帮助新成员理解模块的预期行为。

Vitest：现代测试框架
Vitest 是基于 Vite 的测试框架，原生支持 ESM、TypeScript，与现代开发工具链无缝集成。它的 API 与 Jest 兼容，迁移成本极低。

编写测试
import { describe, it, expect, beforeEach } from 'vitest';

function add(a: number, b: number): number {
  return a + b;
}

function divide(a: number, b: number): number {
  if (b === 0) throw new Error('除数不能为零');
  return a / b;
}

describe('数学工具函数', () => {
  describe('add', () => {
    it('应该正确计算两个正数之和', () => {
      expect(add(1, 2)).toBe(3);
    });

    it('应该正确处理负数', () => {
      expect(add(-1, -2)).toBe(-3);
    });

    it('应该正确处理零', () => {
      expect(add(0, 0)).toBe(0);
    });
  });

  describe('divide', () => {
    it('应该正确计算除法', () => {
      expect(divide(10, 2)).toBe(5);
    });

    it('除以零应该抛出错误', () => {
      expect(() => divide(10, 0)).toThrow('除数不能为零');
    });

    it('应该返回浮点结果', () => {
      expect(divide(1, 3)).toBeCloseTo(0.333, 2);
    });
  });
});
异步测试
import { describe, it, expect } from 'vitest';
import fs from 'node:fs/promises';

async function readJsonFile(path: string): Promise<Record<string, unknown>> {
  const content = await fs.readFile(path, 'utf-8');
  return JSON.parse(content);
}

describe('readJsonFile', () => {
  it('应该正确读取和解析 JSON 文件', async () => {
    const result = await readJsonFile('package.json');
    expect(result).toHaveProperty('name');
    expect(result).toBeTypeOf('object');
  });

  it('文件不存在时应该抛出错误', async () => {
    await expect(readJsonFile('nonexistent.json')).rejects.toThrow();
  });
});


Mock 技术
函数 Mock
import { describe, it, expect, vi } from 'vitest';

interface UserService {
  getUser(id: string): Promise<{ id: string; name: string }>;
}

async function getUserName(service: UserService, id: string): Promise<string> {
  const user = await service.getUser(id);
  return user.name;
}

describe('getUserName', () => {
  it('应该返回用户名称', async () => {
    const mockService: UserService = {
      getUser: vi.fn().mockResolvedValue({ id: '1', name: '张三' }),
    };

    const name = await getUserName(mockService, '1');
    expect(name).toBe('张三');
    expect(mockService.getUser).toHaveBeenCalledWith('1');
    expect(mockService.getUser).toHaveBeenCalledTimes(1);
  });

  it('服务异常时应该抛出错误', async () => {
    const mockService: UserService = {
      getUser: vi.fn().mockRejectedValue(new Error('服务不可用')),
    };

    await expect(getUserName(mockService, '1')).rejects.toThrow('服务不可用');
  });
});
模块 Mock
import { describe, it, expect, vi } from 'vitest';

vi.mock('node:fs/promises', () => ({
  readFile: vi.fn(),
}));

import fs from 'node:fs/promises';

describe('模块 Mock', () => {
  it('应该使用 mock 的 readFile', async () => {
    vi.mocked(fs.readFile).mockResolvedValue('{"key": "value"}');

    const content = await fs.readFile('test.json', 'utf-8');
    expect(JSON.parse(content as string)).toEqual({ key: 'value' });
  });
});
定时器 Mock
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';

function debounce<T extends (...args: unknown[]) => void>(fn: T, delay: number): T {
  let timer: ReturnType<typeof setTimeout>;
  return ((...args: unknown[]) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  }) as T;
}

describe('debounce', () => {
  beforeEach(() => {
    vi.useFakeTimers();
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  it('应该在延迟后执行函数', () => {
    const fn = vi.fn();
    const debounced = debounce(fn, 300);

    debounced();
    expect(fn).not.toHaveBeenCalled();

    vi.advanceTimersByTime(300);
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it('多次调用应该只执行最后一次', () => {
    const fn = vi.fn();
    const debounced = debounce(fn, 300);

    debounced();
    debounced();
    debounced();

    vi.advanceTimersByTime(300);
    expect(fn).toHaveBeenCalledTimes(1);
  });
});


HTTP API 测试
import { describe, it, expect } from 'vitest';
import { Hono } from 'hono';

const app = new Hono();

app.get('/api/health', (c) => c.json({ status: 'ok' }));
app.post('/api/echo', async (c) => {
  const body = await c.req.json();
  return c.json(body);
});

describe('API 测试', () => {
  it('GET /api/health 应该返回 200', async () => {
    const res = await app.request('/api/health');
    expect(res.status).toBe(200);

    const body = await res.json();
    expect(body).toEqual({ status: 'ok' });
  });

  it('POST /api/echo 应该返回请求体', async () => {
    const res = await app.request('/api/echo', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message: 'hello' }),
    });

    expect(res.status).toBe(200);
    const body = await res.json();
    expect(body).toEqual({ message: 'hello' });
  });

  it('不存在的路由应该返回 404', async () => {
    const res = await app.request('/api/nonexistent');
    expect(res.status).toBe(404);
  });
});

测试最佳实践
const testingPrinciples = {
  'AAA 模式': 'Arrange（准备）→ Act（执行）→ Assert（断言）',
  测试行为而非实现: '测试函数的输入输出，而非内部实现细节',
  每个测试独立: '测试之间不应共享状态或依赖执行顺序',
  描述性命名: "测试名称应说明'在什么条件下做什么应该怎样'",
  '避免过度 Mock': '只 Mock 外部依赖（网络、文件系统、数据库）',
  保持测试简短: '每个测试只验证一个行为',
  覆盖边界情况: '空值、错误输入、边界条件是最容易出 bug 的地方',
};


小结
    

  • 测试金字塔：大量单元测试 + 适量集成测试 + 少量端到端测试。
    • 

  • Vitest 原生支持 ESM 和 TypeScript，速度快，推荐新项目使用。
    • 

  • vi.fn() 创建 Mock 函数，vi.mock() Mock 整个模块，vi.useFakeTimers() Mock 定时器。
    • 

  • Hono 的 app.request() 提供了简洁的 API 测试方式，无需启动真实服务器。
    • 

  • Vitest 与 Jest API 兼容，迁移成本低。
    • 

  • 遵循 AAA 模式和"测试行为而非实现"的原则编写高质量测试。
