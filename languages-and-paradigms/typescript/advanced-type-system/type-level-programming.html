类型体操入门

问题背景

TypeScript 的类型系统已经被证明是图灵完备的——理论上可以在类型层面进行任意计算。社区将这种利用泛型、条件类型、映射类型、递归类型等特性构建复杂类型运算的实践称为"类型体操"（Type Gymnastics）。合理使用类型体操可以构建出极其精确的类型约束，但过度使用则会让代码难以理解和维护。

基本语法
递归类型
TypeScript 支持类型的递归定义。最常见的用例是处理嵌套数据结构：


type JSONValue = string | number | boolean | null | JSONValue[] | { [key: string]: JSONValue };

const data: JSONValue = {
  name: '张三',
  age: 25,
  hobbies: ['编程', '阅读'],
  address: {
    city: '北京',
    coordinates: [39.9, 116.4],
  },
};
递归展开嵌套类型：


type DeepFlatten<T> = T extends readonly (infer E)[] ? DeepFlatten<E> : T;

type F1 = DeepFlatten<number[]>; // number
type F2 = DeepFlatten<number[][][]>; // number
type F3 = DeepFlatten<string[][]>; // string
可变元组类型
TypeScript 4.0 引入了可变元组类型，允许在元组类型中使用展开操作：


type Concat<A extends readonly unknown[], B extends readonly unknown[]> = [...A, ...B];

type R1 = Concat<[1, 2], [3, 4]>; // [1, 2, 3, 4]
type R2 = Concat<[string], [number]>; // [string, number]

type Push<T extends readonly unknown[], V> = [...T, V];
type Unshift<T extends readonly unknown[], V> = [V, ...T];

type Pushed = Push<[1, 2], 3>; // [1, 2, 3]
type Unshifted = Unshift<[2, 3], 1>; // [1, 2, 3]
元组的解构：


type First<T extends readonly unknown[]> = T extends readonly [infer F, ...unknown[]] ? F : never;

type Last<T extends readonly unknown[]> = T extends readonly [...unknown[], infer L] ? L : never;

type Tail<T extends readonly unknown[]> = T extends readonly [unknown, ...infer R] ? R : [];

type H = First<[1, 2, 3]>; // 1
type L = Last<[1, 2, 3]>; // 3
type T = Tail<[1, 2, 3]>; // [2, 3]
进阶用法
构建实用工具类型
逐步构建一个 DeepRequired 类型：


type DeepRequired<T> = T extends object
  ? T extends (...args: any[]) => any
    ? T
    : {
        [K in keyof T]-?: DeepRequired<T[K]>;
      }
  : T;

interface FormConfig {
  title?: string;
  fields?: {
    name?: string;
    validation?: {
      required?: boolean;
      pattern?: string;
    };
  };
}

type StrictFormConfig = DeepRequired<FormConfig>;
// {
//   title: string;
//   fields: {
//     name: string;
//     validation: {
//       required: boolean;
//       pattern: string;
//     };
//   };
// }
构建一个路径访问类型：


type Get<T, Path extends string> = Path extends `${infer Key}.${infer Rest}`
  ? Key extends keyof T
    ? Get<T[Key], Rest>
    : never
  : Path extends keyof T
    ? T[Path]
    : never;

interface AppState {
  user: {
    profile: {
      name: string;
      age: number;
    };
    settings: {
      theme: 'light' | 'dark';
    };
  };
  posts: { id: string; title: string }[];
}

type UserName = Get<AppState, 'user.profile.name'>; // string
type Theme = Get<AppState, 'user.settings.theme'>; // "light" | "dark"
type Posts = Get<AppState, 'posts'>; // { id: string; title: string }[]
类型层面的数据变换
type TupleToUnion<T extends readonly unknown[]> = T[number];

type U = TupleToUnion<[string, number, boolean]>;
// string | number | boolean

type TupleToObject<T extends readonly string[]> = {
  [K in T[number]]: K;
};

type Obj = TupleToObject<['name', 'age', 'email']>;
// { name: "name"; age: "age"; email: "email" }

type UnionToTuple<U> = (
  (U extends unknown ? (arg: () => U) => void : never) extends (arg: infer I) => void ? I : never
) extends () => infer Last
  ? [...UnionToTuple<Exclude<U, Last>>, Last]
  : [];

type Tuple = UnionToTuple<'a' | 'b' | 'c'>; // ["a", "b", "c"]
字符串类型运算
type TrimLeft<S extends string> = S extends ` ${infer Rest}` ? TrimLeft<Rest> : S;

type TrimRight<S extends string> = S extends `${infer Rest} ` ? TrimRight<Rest> : S;

type Trim<S extends string> = TrimLeft<TrimRight<S>>;

type Trimmed = Trim<'  hello world  '>; // "hello world"

type Replace<S extends string, From extends string, To extends string> = From extends ''
  ? S
  : S extends `${infer Head}${From}${infer Tail}`
    ? `${Head}${To}${Tail}`
    : S;

type ReplaceAll<S extends string, From extends string, To extends string> = From extends ''
  ? S
  : S extends `${infer Head}${From}${infer Tail}`
    ? ReplaceAll<`${Head}${To}${Tail}`, From, To>
    : S;

type R1 = Replace<'hello world', 'world', 'TS'>; // "hello TS"
type R2 = ReplaceAll<'a-b-c-d', '-', '.'>; // "a.b.c.d"
类型推断与约束
调试类型错误
当类型体操的结果不符合预期时，可以使用以下技巧调试：


type Debug<T> = { [K in keyof T]: T[K] };

type Expect<T extends true> = T;
type Equal<X, Y> = (<T>() => T extends X ? 1 : 2) extends <T>() => T extends Y ? 1 : 2 ? true : false;

type TestCase1 = Expect<Equal<Trim<'  hi  '>, 'hi'>>; // 通过
type TestCase2 = Expect<Equal<First<[1, 2, 3]>, 1>>; // 通过
type TestCase3 = Expect<Equal<Concat<[1], [2]>, [1, 2]>>; // 通过
性能注意事项
// ❌ 避免：过深的递归（可能超过 TypeScript 的递归限制）
type BadRepeat<S extends string, N extends number, Acc extends string = ''> = Acc['length'] extends N
  ? S // 此处逻辑简化示意
  : BadRepeat<S, N, `${Acc}x`>;

// ✅ 推荐：利用分治降低递归深度
type BuildTuple<N extends number, T extends unknown[] = []> = T['length'] extends N
  ? T
  : BuildTuple<N, [...T, unknown]>;

type Length5 = BuildTuple<5>; // [unknown, unknown, unknown, unknown, unknown]
何时不使用类型体操
// ❌ 过度复杂的类型——团队成员难以理解
// type UltraComplexType<T extends ...> = ...（50 行嵌套条件类型）

// ✅ 简单明了的类型——首选方案
interface UserCreateDTO {
  name: string;
  email: string;
}

interface UserUpdateDTO {
  name?: string;
  email?: string;
}

// ✅ 适度使用工具类型
type UserCreateDTO2 = Pick<User, 'name' | 'email'>;
type UserUpdateDTO2 = Partial<Pick<User, 'name' | 'email'>>;

interface User {
  id: string;
  name: string;
  email: string;
  createdAt: Date;
}

实际应用
类型安全的状态机
type Transitions = {
  idle: 'running' | 'error';
  running: 'paused' | 'completed' | 'error';
  paused: 'running' | 'completed';
  completed: 'idle';
  error: 'idle';
};

type State = keyof Transitions;

class StateMachine<T extends Record<string, string>> {
  private current: keyof T;

  constructor(initial: keyof T) {
    this.current = initial;
  }

  getState(): keyof T {
    return this.current;
  }

  transition<S extends keyof T>(from: S, to: T[S] extends string ? T[S] : never): void {
    if (this.current !== from) {
      throw new Error(`当前状态为 ${String(this.current)}，不是 ${String(from)}`);
    }
    this.current = to as keyof T;
  }
}

const machine = new StateMachine<Transitions>('idle');
machine.transition('idle', 'running');
machine.transition('running', 'paused');
machine.transition('paused', 'completed');
类型安全的深层取值函数
type PathSegments<P extends string> = P extends `${infer Head}.${infer Tail}` ? [Head, ...PathSegments<Tail>] : [P];

type GetByPath<T, Segments extends string[]> = Segments extends [
  infer First extends string,
  ...infer Rest extends string[],
]
  ? First extends keyof T
    ? Rest extends []
      ? T[First]
      : GetByPath<T[First], Rest>
    : undefined
  : T;

function get<T extends object, P extends string>(obj: T, path: P): GetByPath<T, PathSegments<P>> {
  const segments = path.split('.') as string[];
  let current: unknown = obj;
  for (const segment of segments) {
    if (current == null || typeof current !== 'object') {
      return undefined as GetByPath<T, PathSegments<P>>;
    }
    current = (current as Record<string, unknown>)[segment];
  }
  return current as GetByPath<T, PathSegments<P>>;
}

const state = {
  user: {
    profile: { name: '张三', age: 25 },
    settings: { theme: 'dark' as const },
  },
};

const name = get(state, 'user.profile.name'); // string
const theme = get(state, 'user.settings.theme'); // "dark"
小结
    

  • TypeScript 的类型系统是图灵完备的，支持递归、条件分支、模式匹配等编程范式
    • 

  • 递归类型可以处理嵌套数据结构，可变元组类型支持类型层面的数组操作
    • 

  • 构建复杂工具类型应采用渐进式方法：先实现简单版本，逐步增加功能
    • 

  • 类型测试（Expect<Equal<...>>）是验证类型正确性的有效手段
    • 

  • 性能方面需注意递归深度限制（通常约 50 层），过深的递归会导致编译错误
    • 

  • 类型体操的使用应遵循"适度"原则——能用简单类型解决的问题不要引入复杂的类型运算
    • 

  • 判断是否使用类型体操的核心标准：它是否让代码更安全且团队成员都能理解
    •
