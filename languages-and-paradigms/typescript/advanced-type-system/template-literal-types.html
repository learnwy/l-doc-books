模板字面量类型

问题背景

在前端开发中，大量场景需要处理特定模式的字符串：CSS 属性名、事件名、路由路径、国际化键名等。传统做法只能用 string 类型标注，丧失了类型安全性。TypeScript 4.1 引入的模板字面量类型（Template Literal Types）允许在类型层面进行字符串拼接和模式匹配，将字符串的结构信息编码到类型系统中。

基本语法
模板字面量类型语法
模板字面量类型使用与 JavaScript 模板字符串相同的反引号语法：


type Greeting = `hello ${string}`;

const a: Greeting = 'hello world'; // 正确
const b: Greeting = 'hello TypeScript'; // 正确
// const c: Greeting = "hi world";     // 错误

type HttpMethod = 'GET' | 'POST' | 'PUT' | 'DELETE';
type ApiPath = '/users' | '/posts' | '/comments';

type Endpoint = `${HttpMethod} ${ApiPath}`;
// "GET /users" | "GET /posts" | "GET /comments"
// | "POST /users" | "POST /posts" | "POST /comments"
// | "PUT /users" | "PUT /posts" | "PUT /comments"
// | "DELETE /users" | "DELETE /posts" | "DELETE /comments"
联合类型的笛卡尔积
当模板中嵌入联合类型时，结果是所有组合的笛卡尔积：


type Size = 'sm' | 'md' | 'lg';
type Color = 'primary' | 'secondary' | 'danger';

type ButtonClass = `btn-${Size}-${Color}`;
// "btn-sm-primary" | "btn-sm-secondary" | "btn-sm-danger"
// | "btn-md-primary" | "btn-md-secondary" | "btn-md-danger"
// | "btn-lg-primary" | "btn-lg-secondary" | "btn-lg-danger"

type Direction = 'top' | 'right' | 'bottom' | 'left';
type CssProperty = 'margin' | 'padding';

type SpacingProp = `${CssProperty}-${Direction}`;
// "margin-top" | "margin-right" | "margin-bottom" | "margin-left"
// | "padding-top" | "padding-right" | "padding-bottom" | "padding-left"
进阶用法
内置字符串操作类型
TypeScript 内置了四个字符串操作类型：


type Upper = Uppercase<'hello'>; // "HELLO"
type Lower = Lowercase<'HELLO'>; // "hello"
type Cap = Capitalize<'hello'>; // "Hello"
type Uncap = Uncapitalize<'Hello'>; // "hello"

type EventName = 'click' | 'focus' | 'blur';
type HandlerName = `on${Capitalize<EventName>}`;
// "onClick" | "onFocus" | "onBlur"
组合使用：


type CSSVariable = `--${string}`;
type ScreamCase<S extends string> = Uppercase<S>;

type EnvKey = 'nodeEnv' | 'apiUrl' | 'debug';
type EnvVar = `VITE_${Uppercase<EnvKey>}`;
// "VITE_NODEENV" | "VITE_APIURL" | "VITE_DEBUG"
模式匹配与类型提取
模板字面量类型可以配合 infer 进行模式匹配：


type ExtractRouteParam<T extends string> = T extends `${string}:${infer Param}` ? Param : never;

type P1 = ExtractRouteParam<'/users/:id'>; // "id"
type P2 = ExtractRouteParam<'/static/page'>; // never

type SplitString<S extends string, D extends string> = S extends `${infer Head}${D}${infer Tail}`
  ? [Head, ...SplitString<Tail, D>]
  : [S];

type Parts = SplitString<'a.b.c', '.'>; // ["a", "b", "c"]
type Words = SplitString<'hello world', ' '>; // ["hello", "world"]
camelCase 到 kebab-case 转换
type CamelToKebab<S extends string> = S extends `${infer Head}${infer Tail}`
  ? Head extends Uppercase<Head>
    ? Head extends Lowercase<Head>
      ? `${Head}${CamelToKebab<Tail>}`
      : `-${Lowercase<Head>}${CamelToKebab<Tail>}`
    : `${Head}${CamelToKebab<Tail>}`
  : S;

type R1 = CamelToKebab<'backgroundColor'>; // "background-color"
type R2 = CamelToKebab<'fontSize'>; // "font-size"
type R3 = CamelToKebab<'borderTopWidth'>; // "border-top-width"
类型推断与约束
模板字面量中的类型约束
type NumericString = `${number}`;

function parseNumeric(value: `${number}`): number {
  return Number(value);
}

parseNumeric('123'); // 正确
parseNumeric('3.14'); // 正确
// parseNumeric("abc"); // 错误

type HexColor = `#${string}`;

function setColor(color: HexColor): void {
  console.log(`设置颜色: ${color}`);
}

setColor('#ff0000'); // 正确
// setColor("red");  // 错误
递归模板类型
type Join<T extends readonly string[], D extends string> = T extends readonly [infer F extends string]
  ? F
  : T extends readonly [infer F extends string, ...infer R extends string[]]
    ? `${F}${D}${Join<R, D>}`
    : '';

type Joined = Join<['a', 'b', 'c'], '-'>; // "a-b-c"
type Dotted = Join<['x', 'y'], '.'>; // "x.y"
type Single = Join<['hello'], ','>; // "hello"
深层路径类型
type PathOf<T, Prefix extends string = ''> = T extends object
  ? {
      [K in keyof T & string]: T[K] extends object ? PathOf<T[K], `${Prefix}${K}.`> | `${Prefix}${K}` : `${Prefix}${K}`;
    }[keyof T & string]
  : never;

interface AppConfig {
  server: {
    host: string;
    port: number;
  };
  database: {
    name: string;
    pool: {
      min: number;
      max: number;
    };
  };
}

type ConfigPath = PathOf<AppConfig>;
// "server" | "server.host" | "server.port"
// | "database" | "database.name" | "database.pool"
// | "database.pool.min" | "database.pool.max"
实际应用
类型安全的事件系统
type EventConfig = {
  user: {
    login: { userId: string; timestamp: number };
    logout: { userId: string };
    update: { userId: string; fields: string[] };
  };
  order: {
    create: { orderId: string; amount: number };
    cancel: { orderId: string; reason: string };
  };
};

type EventName<T> = {
  [K in keyof T & string]: T[K] extends object ? `${K}:${keyof T[K] & string}` : never;
}[keyof T & string];

type AppEvent = EventName<EventConfig>;
// "user:login" | "user:logout" | "user:update" | "order:create" | "order:cancel"

type EventPayload<T, E extends string> = E extends `${infer NS extends string}:${infer Name extends string}`
  ? NS extends keyof T
    ? Name extends keyof T[NS]
      ? T[NS][Name]
      : never
    : never
  : never;

class EventBus<T extends Record<string, Record<string, unknown>>> {
  private handlers = new Map<string, Set<(data: any) => void>>();

  on<E extends EventName<T>>(event: E, handler: (data: EventPayload<T, E>) => void): void {
    if (!this.handlers.has(event)) {
      this.handlers.set(event, new Set());
    }
    this.handlers.get(event)!.add(handler);
  }

  emit<E extends EventName<T>>(event: E, data: EventPayload<T, E>): void {
    this.handlers.get(event)?.forEach((h) => h(data));
  }
}

const bus = new EventBus<EventConfig>();

bus.on('user:login', (data) => {
  console.log(`用户 ${data.userId} 登录于 ${data.timestamp}`);
});

bus.on('order:create', (data) => {
  console.log(`订单 ${data.orderId} 金额 ${data.amount}`);
});

bus.emit('user:login', { userId: 'u-001', timestamp: Date.now() });
CSS-in-TS 类型安全
type CSSUnit = 'px' | 'rem' | 'em' | 'vh' | 'vw' | '%';
type CSSValue = `${number}${CSSUnit}` | 'auto' | '0';

interface TypedStyle {
  width?: CSSValue;
  height?: CSSValue;
  margin?: CSSValue;
  padding?: CSSValue;
  fontSize?: CSSValue;
}

function applyStyle(element: string, style: TypedStyle): void {
  console.log(`应用样式到 ${element}:`, style);
}

applyStyle('header', {
  width: '100%',
  height: '64px',
  padding: '16px',
  fontSize: '1.5rem',
});
小结
    

  • 模板字面量类型使用反引号语法，将字符串的结构信息编码到类型系统中
    • 

  • 嵌入联合类型时会产生笛卡尔积，自动生成所有可能的字符串组合
    • 

  • 四个内置字符串操作类型（Uppercase/Lowercase/Capitalize/Uncapitalize）支持类型层面的大小写转换
    • 

  • 配合 infer 可以从字符串类型中提取子串，实现模式匹配和字符串解析
    • 

  • 递归模板字面量类型可以处理任意深度的字符串拼接和路径生成
    • 

  • 实际应用包括类型安全的事件系统、CSS 值约束、路由参数提取等场景
    • 

  • 注意避免过度使用导致类型组合爆炸，影响编译性能
    •
