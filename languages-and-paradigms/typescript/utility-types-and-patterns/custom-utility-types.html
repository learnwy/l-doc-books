自定义工具类型

问题背景

TypeScript 内置的工具类型覆盖了最常见的类型变换场景，但实际项目中经常需要更复杂的类型操作：深层递归地将对象变为可选或只读、按值类型筛选对象属性、要求至少一个属性存在等。这些需求超出了内置工具类型的能力范围。

自定义工具类型通过组合映射类型、条件类型、递归类型和模板字面量类型，可以构建出任意复杂的类型变换逻辑。理解这些构建技巧是迈向类型级编程的关键一步。



基本语法
DeepPartial<T>
内置的 Partial 只作用于第一层属性。DeepPartial 递归地将所有嵌套属性变为可选：


type DeepPartial<T> = {
  [K in keyof T]?: T[K] extends object
    ? T[K] extends Array<infer U>
      ? Array<DeepPartial<U>>
      : DeepPartial<T[K]>
    : T[K];
};
interface AppConfig {
  server: {
    host: string;
    port: number;
    ssl: {
      enabled: boolean;
      cert: string;
    };
  };
  database: {
    url: string;
    pool: {
      min: number;
      max: number;
    };
  };
}

type PartialConfig = DeepPartial<AppConfig>;

const override: PartialConfig = {
  server: {
    ssl: {
      enabled: true,
    },
  },
};
DeepReadonly<T>
递归地将所有嵌套属性变为只读：


type DeepReadonly<T> = {
  readonly [K in keyof T]: T[K] extends object
    ? T[K] extends Array<infer U>
      ? ReadonlyArray<DeepReadonly<U>>
      : DeepReadonly<T[K]>
    : T[K];
};
interface State {
  user: {
    name: string;
    preferences: {
      theme: string;
      notifications: boolean;
    };
  };
  items: Array<{ id: string; value: number }>;
}

type FrozenState = DeepReadonly<State>;

function getState(): FrozenState {
  return {
    user: {
      name: 'Alice',
      preferences: { theme: 'dark', notifications: true },
    },
    items: [{ id: '1', value: 42 }],
  };
}

const state = getState();
// state.user.preferences.theme = "light";  // ❌ Error: readonly
// state.items.push({ id: "2", value: 0 }); // ❌ Error: readonly


进阶用法
Mutable<T>
移除所有 readonly 修饰符，与 Readonly 相反：


type Mutable<T> = {
  -readonly [K in keyof T]: T[K];
};
interface FrozenConfig {
  readonly host: string;
  readonly port: number;
  readonly debug: boolean;
}

type MutableConfig = Mutable<FrozenConfig>;
// { host: string; port: number; debug: boolean }

function buildConfig(): FrozenConfig {
  const config: MutableConfig = { host: '', port: 0, debug: false };
  config.host = 'localhost';
  config.port = 3000;
  config.debug = true;
  return config;
}
RequireAtLeastOne<T>
要求对象至少提供一个属性，常用于搜索条件或过滤器：


type RequireAtLeastOne<T> = {
  [K in keyof T]-?: Required<Pick<T, K>> & Partial<Pick<T, Exclude<keyof T, K>>>;
}[keyof T];
interface SearchFilters {
  name: string;
  email: string;
  phone: string;
}

type SearchQuery = RequireAtLeastOne<SearchFilters>;

function searchUsers(query: SearchQuery): void {
  console.log('Searching with:', query);
}

searchUsers({ name: 'Alice' });
searchUsers({ email: 'a@b.com', phone: '123' });
// searchUsers({});  // ❌ Error: 至少需要一个属性
PickByType<T, V>
选取值类型匹配 V 的所有属性：


type PickByType<T, V> = {
  [K in keyof T as T[K] extends V ? K : never]: T[K];
};
interface Mixed {
  id: number;
  name: string;
  email: string;
  age: number;
  active: boolean;
}

type StringFields = PickByType<Mixed, string>;
// { name: string; email: string }

type NumberFields = PickByType<Mixed, number>;
// { id: number; age: number }
OmitByType<T, V>
排除值类型匹配 V 的所有属性：


type OmitByType<T, V> = {
  [K in keyof T as T[K] extends V ? never : K]: T[K];
};
type NonStringFields = OmitByType<Mixed, string>;
// { id: number; age: number; active: boolean }


类型推断与约束
构建过程详解：PickByType 的分步推导
理解自定义工具类型的关键在于逐步拆解其构建过程：


// 第一步：基本映射类型
type Step1<T> = {
  [K in keyof T]: T[K];
};

// 第二步：添加条件过滤（键重映射，TypeScript 4.1+）
type Step2<T, V> = {
  [K in keyof T as T[K] extends V ? K : never]: T[K];
};

// 第三步：完成 — 这就是 PickByType
type PickByType<T, V> = {
  [K in keyof T as T[K] extends V ? K : never]: T[K];
};
关键机制：as 子句中返回 never 会使该键被过滤掉。


DeepRequired<T>
递归地将所有嵌套属性变为必需：


type DeepRequired<T> = {
  [K in keyof T]-?: T[K] extends object
    ? T[K] extends Array<infer U>
      ? Array<DeepRequired<U>>
      : DeepRequired<T[K]>
    : T[K];
};
OptionalKeys<T> 与 RequiredKeys<T>
提取可选属性或必需属性的键名：


type OptionalKeys<T> = {
  [K in keyof T]-?: undefined extends T[K] ? K : never;
}[keyof T];

type RequiredKeys<T> = {
  [K in keyof T]-?: undefined extends T[K] ? never : K;
}[keyof T];
interface Form {
  name: string;
  email: string;
  nickname?: string;
  bio?: string;
}

type FormOptional = OptionalKeys<Form>;
// "nickname" | "bio"

type FormRequired = RequiredKeys<Form>;
// "name" | "email"
UnionToIntersection<U>
将联合类型转为交叉类型，这是高级类型编程中的经典技巧：


type UnionToIntersection<U> = (U extends unknown ? (arg: U) => void : never) extends (arg: infer I) => void ? I : never;
type Union = { a: string } | { b: number } | { c: boolean };

type Intersection = UnionToIntersection<Union>;
// { a: string } & { b: number } & { c: boolean }


实际应用
类型安全的配置合并
type DeepPartial<T> = {
  [K in keyof T]?: T[K] extends object
    ? T[K] extends Array<infer U>
      ? Array<DeepPartial<U>>
      : DeepPartial<T[K]>
    : T[K];
};

interface AppSettings {
  theme: {
    primary: string;
    secondary: string;
    font: {
      family: string;
      size: number;
    };
  };
  features: {
    darkMode: boolean;
    analytics: boolean;
  };
}

const defaults: AppSettings = {
  theme: {
    primary: '#3b82f6',
    secondary: '#8b5cf6',
    font: { family: 'Inter', size: 14 },
  },
  features: { darkMode: false, analytics: true },
};

function mergeSettings(base: AppSettings, overrides: DeepPartial<AppSettings>): AppSettings {
  return {
    theme: {
      ...base.theme,
      ...overrides.theme,
      font: {
        ...base.theme.font,
        ...overrides.theme?.font,
      },
    },
    features: {
      ...base.features,
      ...overrides.features,
    },
  };
}

const custom = mergeSettings(defaults, {
  theme: { primary: '#ef4444', font: { size: 16 } },
});
事件处理器类型提取
type EventHandlerMap = {
  click: { x: number; y: number };
  keydown: { key: string; code: string };
  resize: { width: number; height: number };
  scroll: { scrollTop: number; scrollLeft: number };
};

type EventNames = keyof EventHandlerMap;

type EventPayload<E extends EventNames> = EventHandlerMap[E];

type EventHandler<E extends EventNames> = (payload: EventPayload<E>) => void;

function on<E extends EventNames>(event: E, handler: EventHandler<E>): void {
  console.log(`Registered handler for ${event}`);
  handler({} as EventPayload<E>);
}

on('click', (payload) => {
  console.log(payload.x, payload.y);
});

on('keydown', (payload) => {
  console.log(payload.key);
});


小结
    

  • DeepPartial/DeepReadonly/DeepRequired 通过递归 + 条件类型实现深层属性变换，是内置 Partial/Readonly/Required 的增强版。
    • 

  • Mutable 使用 -readonly 修饰符移除只读约束，与 Readonly 互为逆操作。
    • 

  • RequireAtLeastOne 结合映射类型和联合类型，确保对象至少有一个属性被提供。
    • 

  • PickByType/OmitByType 利用键重映射（as 子句）按值类型筛选属性，是 Pick/Omit 的泛化版本。
    • 

  • UnionToIntersection 是高级类型编程的经典工具，利用逆变位置的推断将联合转为交叉。
    • 

  • 构建自定义工具类型的核心方法是分步推导：从简单映射开始，逐步添加条件判断、键过滤和递归逻辑。
