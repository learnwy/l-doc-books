内置工具类型

问题背景

在日常开发中，我们经常需要基于已有类型创建变体：将所有属性变为可选、从对象中挑选部分字段、提取函数返回值类型等。如果每次都手动编写这些类型转换逻辑，不仅冗余且容易出错。

TypeScript 内置了一系列工具类型（Utility Types），它们是预定义的泛型类型别名，利用映射类型和条件类型实现常见的类型变换。掌握这些工具类型可以大幅减少重复代码并增强类型安全性。



基本语法
对象属性变换类
Partial<T>
将 T 的所有属性变为可选：


interface User {
  id: string;
  name: string;
  email: string;
  age: number;
}

type PartialUser = Partial<User>;
// { id?: string; name?: string; email?: string; age?: number }

function updateUser(id: string, updates: Partial<User>): User {
  const existing: User = { id, name: 'Alice', email: 'a@b.com', age: 25 };
  return { ...existing, ...updates };
}

updateUser('1', { name: 'Bob' });
updateUser('1', { email: 'bob@example.com', age: 30 });
Required<T>
将 T 的所有属性变为必需，与 Partial 相反：


interface Config {
  host?: string;
  port?: number;
  debug?: boolean;
}

type RequiredConfig = Required<Config>;
// { host: string; port: number; debug: boolean }

function startServer(config: RequiredConfig): void {
  console.log(`${config.host}:${config.port}`);
}
Readonly<T>
将 T 的所有属性变为只读：


interface State {
  count: number;
  items: string[];
}

type ImmutableState = Readonly<State>;

function reducer(state: ImmutableState): ImmutableState {
  // state.count = 1;  // ❌ Error: Cannot assign to 'count'
  return { ...state, count: state.count + 1 };
}
Record<K, V>
创建键类型为 K、值类型为 V 的对象类型：


type Role = 'admin' | 'editor' | 'viewer';

interface Permission {
  read: boolean;
  write: boolean;
  delete: boolean;
}

const permissions: Record<Role, Permission> = {
  admin: { read: true, write: true, delete: true },
  editor: { read: true, write: true, delete: false },
  viewer: { read: true, write: false, delete: false },
};


进阶用法
对象属性选取类
Pick<T, K>
从 T 中选取属性集合 K：


interface Article {
  id: string;
  title: string;
  content: string;
  author: string;
  createdAt: Date;
  updatedAt: Date;
}

type ArticlePreview = Pick<Article, 'id' | 'title' | 'author'>;
// { id: string; title: string; author: string }

function renderPreview(article: ArticlePreview): string {
  return `${article.title} by ${article.author}`;
}
Omit<T, K>
从 T 中排除属性集合 K，与 Pick 互补：


type CreateArticleInput = Omit<Article, 'id' | 'createdAt' | 'updatedAt'>;
// { title: string; content: string; author: string }

function createArticle(input: CreateArticleInput): Article {
  return {
    ...input,
    id: crypto.randomUUID(),
    createdAt: new Date(),
    updatedAt: new Date(),
  };
}
联合类型过滤类
Exclude<T, U>
从联合类型 T 中排除可赋值给 U 的成员：


type AllEvents = 'click' | 'scroll' | 'mouseover' | 'keydown' | 'keyup';
type MouseEvents = 'click' | 'scroll' | 'mouseover';

type KeyboardEvents = Exclude<AllEvents, MouseEvents>;
// "keydown" | "keyup"
Extract<T, U>
从联合类型 T 中提取可赋值给 U 的成员：


type StringOrNumber = string | number | boolean | null;

type OnlyStringOrNumber = Extract<StringOrNumber, string | number>;
// string | number
NonNullable<T>
从 T 中排除 null 和 undefined：


type MaybeString = string | null | undefined;

type DefiniteString = NonNullable<MaybeString>;
// string

function processValue(value: MaybeString): void {
  const safe: DefiniteString = value ?? 'default';
  console.log(safe.toUpperCase());
}


类型推断与约束
函数相关工具类型
ReturnType<T>
提取函数类型 T 的返回值类型：


function fetchUser(id: string) {
  return { id, name: 'Alice', createdAt: new Date() };
}

type FetchUserResult = ReturnType<typeof fetchUser>;
// { id: string; name: string; createdAt: Date }

async function fetchData() {
  return { items: [1, 2, 3], total: 100 };
}

type FetchDataResult = ReturnType<typeof fetchData>;
// Promise<{ items: number[]; total: number }>
Parameters<T>
提取函数类型 T 的参数类型（元组）：


function createEvent(name: string, timestamp: number, metadata: Record<string, unknown>) {
  return { name, timestamp, metadata };
}

type EventParams = Parameters<typeof createEvent>;
// [name: string, timestamp: number, metadata: Record<string, unknown>]

function wrapCreateEvent(...args: Parameters<typeof createEvent>) {
  console.log('Creating event:', args[0]);
  return createEvent(...args);
}
ConstructorParameters<T>
提取构造函数的参数类型：


class HttpClient {
  constructor(
    private baseURL: string,
    private timeout: number,
    private headers: Record<string, string>,
  ) {}
}

type HttpClientArgs = ConstructorParameters<typeof HttpClient>;
// [baseURL: string, timeout: number, headers: Record<string, string>]

function createClient(...args: HttpClientArgs): HttpClient {
  return new HttpClient(...args);
}
InstanceType<T>
提取构造函数类型 T 的实例类型：


class Database {
  query(sql: string): unknown[] {
    return [];
  }
}

type DbInstance = InstanceType<typeof Database>;
// Database

function initDb(DbClass: typeof Database): DbInstance {
  return new DbClass();
}
异步相关工具类型
Awaited<T>
递归解包 Promise 类型，获取最终的 resolved 值类型：


type A = Awaited<Promise<string>>;
// string

type B = Awaited<Promise<Promise<number>>>;
// number

type C = Awaited<boolean | Promise<string>>;
// boolean | string

async function getUser(): Promise<{ name: string }> {
  return { name: 'Alice' };
}

type UserResult = Awaited<ReturnType<typeof getUser>>;
// { name: string }


实际应用
API 层类型设计
interface ApiResponse<T> {
  code: number;
  message: string;
  data: T;
  timestamp: number;
}

interface Product {
  id: string;
  name: string;
  price: number;
  description: string;
  stock: number;
  createdAt: string;
}

type CreateProductInput = Omit<Product, 'id' | 'createdAt'>;
type UpdateProductInput = Partial<Omit<Product, 'id' | 'createdAt'>>;
type ProductListItem = Pick<Product, 'id' | 'name' | 'price'>;

async function createProduct(input: CreateProductInput): Promise<ApiResponse<Product>> {
  const response = await fetch('/api/products', {
    method: 'POST',
    body: JSON.stringify(input),
  });
  return response.json() as Promise<ApiResponse<Product>>;
}
表单状态管理
interface FormFields {
  username: string;
  email: string;
  password: string;
  confirmPassword: string;
}

type FormErrors = Partial<Record<keyof FormFields, string>>;
type FormTouched = Partial<Record<keyof FormFields, boolean>>;

interface FormState {
  values: FormFields;
  errors: FormErrors;
  touched: FormTouched;
  isSubmitting: boolean;
}

function validateField<K extends keyof FormFields>(field: K, value: FormFields[K]): string | undefined {
  if (field === 'email' && typeof value === 'string' && !value.includes('@')) {
    return '邮箱格式无效';
  }
  return undefined;
}
组合使用多个工具类型
type ImmutableProduct = Readonly<Product>;
type ProductRecord = Record<string, Product>;
type ProductKeys = Extract<keyof Product, 'name' | 'price' | 'description'>;
type NonIdFields = Exclude<keyof Product, 'id'>;


小结
    

  • 属性变换类：Partial 使所有属性可选，Required 使所有属性必需，Readonly 使所有属性只读，Record 构造键值对类型。
    • 

  • 属性选取类：Pick 从类型中选取部分属性，Omit 排除部分属性——两者互补，是 API 类型设计的核心工具。
    • 

  • 联合过滤类：Exclude 排除联合成员，Extract 提取联合成员，NonNullable 排除 null/undefined。
    • 

  • 函数相关类：ReturnType 提取返回类型，Parameters 提取参数类型，ConstructorParameters 和 InstanceType 用于类构造函数。
    • 

  • 异步相关类：Awaited 递归解包 Promise，获取最终值类型。
    • 

  • 组合使用多个工具类型可以从一个基础类型派生出创建、更新、列表等多种变体，避免重复定义。
    •
