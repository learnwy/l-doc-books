import OpenAI from 'openai';
import { PineconeClient } from '@pinecone-database/pinecone';

const openai = new OpenAI();

async function indexDocument(doc: { id: string; content: string }) {
  const embedding = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: doc.content,
  });

  await pinecone.index('docs').upsert([
    {
      id: doc.id,
      values: embedding.data[0].embedding,
      metadata: { content: doc.content },
    },
  ]);
}

async function semanticSearch(query: string, topK = 5) {
  const queryEmbedding = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: query,
  });

  const results = await pinecone.index('docs').query({
    vector: queryEmbedding.data[0].embedding,
    topK,
    includeMetadata: true,
  });

  return results.matches;
}
功能二：内容摘要
将长文本自动生成摘要，适用于新闻聚合、文档预览、会议记录。


async function summarize(content: string, maxLength = 200): Promise<string> {
  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      {
        role: 'system',
        content: `你是一个专业的摘要生成器。请用中文生成不超过${maxLength}字的摘要。
要求：保留核心信息，使用简洁的语言，不添加原文中没有的信息。`,
      },
      { role: 'user', content },
    ],
    temperature: 0.3,
  });

  return response.choices[0].message.content ?? '';
}
功能三：文本分类
自动对用户反馈、工单、内容进行分类。


async function classifyFeedback(text: string): Promise<{ category: string; confidence: number }> {
  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      {
        role: 'system',
        content: `将用户反馈分类为以下类别之一：
- bug_report（Bug 报告）
- feature_request（功能请求）
- question（使用咨询）
- praise（正面反馈）
- complaint（投诉）

返回 JSON 格式：{"category": "类别", "confidence": 0.0-1.0}`,
      },
      { role: 'user', content: text },
    ],
    temperature: 0,
    response_format: { type: 'json_object' },
  });

  return JSON.parse(response.choices[0].message.content ?? '{}');
}

功能五：对话式交互
为产品添加智能对话能力。


interface Message {
  role: 'system' | 'user' | 'assistant';
  content: string;
}

class ConversationManager {
  private messages: Message[] = [];
  private systemPrompt: string;

  constructor(systemPrompt: string) {
    this.systemPrompt = systemPrompt;
    this.messages = [{ role: 'system', content: systemPrompt }];
  }

  async chat(userMessage: string): Promise<string> {
    this.messages.push({ role: 'user', content: userMessage });

    const response = await openai.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: this.messages,
      temperature: 0.7,
    });

    const assistantMessage = response.choices[0].message.content ?? '';
    this.messages.push({ role: 'assistant', content: assistantMessage });

    if (this.messages.length > 20) {
      this.messages = [this.messages[0], ...this.messages.slice(-10)];
    }

    return assistantMessage;
  }
}

function splitIntoChunks(text: string, chunkSize = 500, overlap = 50): string[] {
  const chunks: string[] = [];
  let start = 0;

  while (start < text.length) {
    const end = Math.min(start + chunkSize, text.length);
    chunks.push(text.slice(start, end));
    start = end - overlap;
  }

  return chunks;
}

async function answerQuestion(question: string): Promise<string> {
  const relevantChunks = await semanticSearch(question, 3);

  const context = relevantChunks.map((chunk, i) => `[来源${i + 1}] ${chunk.metadata?.content}`).join('\n\n');

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      {
        role: 'system',
        content: `基于以下参考资料回答用户问题。如果资料中没有相关信息，明确告知用户。
在回答中标注信息来源（如 [来源1]）。

参考资料：
${context}`,
      },
      { role: 'user', content: question },
    ],
  });

  return response.choices[0].message.content ?? '';
}

实践路线
第 1 天： 注册 AI API 账号，在命令行调用一次 API


第 1 周： 实现一个简单的文本分类或摘要功能


第 2 周： 实现语义搜索，接入向量数据库


第 3 周： 构建完整的 RAG 问答系统


第 4 周： 添加监控、缓存、错误处理，使其达到生产质量
