个人数据仓库

概述

你的数字生活散布在数十个应用中：笔记在 Obsidian、书签在 Raindrop、习惯追踪在 Streaks、阅读标注在 Readwise、代码统计在 GitHub。每个应用都是一座数据孤岛。个人数据仓库（Personal Data Warehouse）的理念是：用一个 SQLite 数据库作为统一的本地数据湖，将所有这些异构数据汇聚到一处，用 SQL 的力量进行跨源查询、关联分析和知识发现。这不是要替代专业应用，而是为你的数据建立一个"第二大脑的后端"——一个你完全掌控的、可编程的、持久的数据中心。

核心概念

架构设计

核心原则：


    

  • 单文件：整个数据仓库就是一个 .db 文件，易于备份和迁移
    • 

  • Schema-on-write：导入时将数据归一化为结构化表
    • 

  • 只追加：原始数据只追加不修改，保留完整历史
    • 

  • 衍生视图：分析结论通过 View/CTE 动态计算，不存储冗余结果
    •

-- === 核心维度表 ===

-- 数据源注册
CREATE TABLE sources (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,           -- 'github', 'obsidian', 'browser'
    description TEXT,
    last_sync_at TEXT,
    sync_cursor TEXT,             -- 增量同步游标
    config JSON                   -- 源特定配置
);

-- 统一标签系统
CREATE TABLE tags (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL UNIQUE COLLATE NOCASE
);

-- === 事实表（每源一张） ===

-- 笔记
CREATE TABLE notes (
    id TEXT PRIMARY KEY,
    source_id TEXT NOT NULL REFERENCES sources(id),
    title TEXT NOT NULL,
    content TEXT,
    path TEXT,                    -- 原始文件路径
    word_count INTEGER,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    metadata JSON                 -- 源特有字段
);

-- 书签
CREATE TABLE bookmarks (
    id TEXT PRIMARY KEY,
    source_id TEXT NOT NULL REFERENCES sources(id),
    url TEXT NOT NULL,
    title TEXT NOT NULL,
    description TEXT,
    saved_at TEXT NOT NULL,
    read_at TEXT,                 -- 是否已读
    metadata JSON
);

-- 习惯记录
CREATE TABLE habits (
    id TEXT PRIMARY KEY,
    source_id TEXT NOT NULL REFERENCES sources(id),
    habit_name TEXT NOT NULL,
    completed_at TEXT NOT NULL,
    value REAL,                   -- 量化值（如分钟数）
    note TEXT,
    metadata JSON
);

-- 代码活动
CREATE TABLE code_activity (
    id TEXT PRIMARY KEY,
    source_id TEXT NOT NULL REFERENCES sources(id),
    repo TEXT NOT NULL,
    event_type TEXT NOT NULL,     -- 'commit', 'pr', 'review', 'issue'
    title TEXT,
    additions INTEGER DEFAULT 0,
    deletions INTEGER DEFAULT 0,
    occurred_at TEXT NOT NULL,
    metadata JSON
);

-- 阅读标注
CREATE TABLE highlights (
    id TEXT PRIMARY KEY,
    source_id TEXT NOT NULL REFERENCES sources(id),
    book_title TEXT NOT NULL,
    author TEXT,
    content TEXT NOT NULL,        -- 高亮文本
    note TEXT,                    -- 个人批注
    location TEXT,               -- 位置信息
    highlighted_at TEXT NOT NULL,
    metadata JSON
);

-- === 关联表 ===

CREATE TABLE item_tags (
    item_type TEXT NOT NULL,      -- 'note', 'bookmark', 'highlight'
    item_id TEXT NOT NULL,
    tag_id INTEGER NOT NULL REFERENCES tags(id),
    PRIMARY KEY (item_type, item_id, tag_id)
);

-- === 全文搜索（跨所有文本数据）===

CREATE VIRTUAL TABLE unified_fts USING fts5(
    title, content, source_type,
    tokenize='unicode61 remove_diacritics 2'
);
ETL 管道实现
// GitHub 数据导入示例
async function syncGitHub(db, token) {
  const source = db.prepare('SELECT * FROM sources WHERE id = ?').get('github');
  const cursor = source?.sync_cursor || '2020-01-01T00:00:00Z';

  // 增量拉取
  const events = await fetchGitHubEvents(token, { since: cursor });

  const insertActivity = db.prepare(`
    INSERT OR IGNORE INTO code_activity 
    (id, source_id, repo, event_type, title, additions, deletions, occurred_at, metadata)
    VALUES (?, 'github', ?, ?, ?, ?, ?, ?, ?)
  `);

  const syncBatch = db.transaction((events) => {
    let latestDate = cursor;
    for (const event of events) {
      insertActivity.run(
        event.id,
        event.repo.name,
        event.type,
        event.title || '',
        event.additions || 0,
        event.deletions || 0,
        event.created_at,
        JSON.stringify(event.payload),
      );
      if (event.created_at > latestDate) latestDate = event.created_at;
    }
    // 更新同步游标
    db.prepare('UPDATE sources SET last_sync_at = ?, sync_cursor = ? WHERE id = ?').run(
      new Date().toISOString(),
      latestDate,
      'github',
    );
  });

  syncBatch(events);
}

// Obsidian 笔记导入
function syncObsidianVault(db, vaultPath) {
  const files = glob.sync('**/*.md', { cwd: vaultPath });

  const insertNote = db.prepare(`
    INSERT OR REPLACE INTO notes 
    (id, source_id, title, content, path, word_count, created_at, updated_at)
    VALUES (?, 'obsidian', ?, ?, ?, ?, ?, ?)
  `);

  const batch = db.transaction((files) => {
    for (const file of files) {
      const fullPath = path.join(vaultPath, file);
      const content = fs.readFileSync(fullPath, 'utf-8');
      const stats = fs.statSync(fullPath);
      const title = path.basename(file, '.md');

      insertNote.run(
        `obsidian:${file}`,
        title,
        content,
        file,
        content.split(/\s+/).length,
        stats.birthtime.toISOString(),
        stats.mtime.toISOString(),
      );
    }
  });

  batch(files);
}
跨源查询示例
-- 本周的知识活动摘要
WITH this_week AS (
    SELECT date('now', '-7 days') AS start_date
)
SELECT
    '笔记' AS type, COUNT(*) AS count, SUM(word_count) AS detail
FROM notes, this_week
WHERE updated_at >= start_date
UNION ALL
SELECT
    '书签', COUNT(*), NULL
FROM bookmarks, this_week
WHERE saved_at >= start_date
UNION ALL
SELECT
    '代码提交', COUNT(*), SUM(additions + deletions)
FROM code_activity, this_week
WHERE occurred_at >= start_date AND event_type = 'commit'
UNION ALL
SELECT
    '阅读标注', COUNT(*), NULL
FROM highlights, this_week
WHERE highlighted_at >= start_date;

-- 跨源全文搜索
SELECT source_type, title, snippet(unified_fts, 1, '<b>', '</b>', '...', 32) AS excerpt
FROM unified_fts
WHERE unified_fts MATCH 'distributed systems OR CRDT'
ORDER BY rank
LIMIT 20;

-- 发现知识关联：哪些书签和笔记涉及同一主题
SELECT b.title AS bookmark, n.title AS note, t.name AS shared_tag
FROM bookmarks b
JOIN item_tags bt ON bt.item_type = 'bookmark' AND bt.item_id = b.id
JOIN item_tags nt ON nt.item_type = 'note' AND nt.item_id = nt.item_id AND nt.tag_id = bt.tag_id
JOIN notes n ON n.id = nt.item_id
JOIN tags t ON t.id = bt.tag_id
ORDER BY b.saved_at DESC
LIMIT 50;
最佳实践
数据治理原则
    

  1. 只追加，不修改：原始数据 INSERT 后不 UPDATE，保留完整历史
    2. 

  2. 来源追踪：每条数据关联 source_id，知道它从哪里来
    3. 

  3. 时间戳标准化：统一用 ISO 8601 UTC 格式存储
    4. 

  4. ID 去重：用 源名:原始ID 格式作为主键，避免跨源冲突
    5. 

  5. 增量同步：记录每个源的 sync_cursor，只拉取新数据
    6.

# 每日自动备份（配合 cron/launchd）
sqlite3 personal-warehouse.db "VACUUM INTO 'backups/pdw-$(date +%Y%m%d).db'"

sqlite3 -json personal-warehouse.db "SELECT * FROM notes WHERE updated_at > '2026-01-01'" > notes-export.json

# Litestream 实时复制到 S3
litestream replicate personal-warehouse.db s3://my-bucket/pdw/
JSON 扩展字段查询
-- 从 metadata JSON 中提取字段
SELECT
    title,
    json_extract(metadata, '$.language') AS language,
    json_extract(metadata, '$.stars') AS stars
FROM code_activity
WHERE json_extract(metadata, '$.language') = 'TypeScript'
ORDER BY json_extract(metadata, '$.stars') DESC;

总结
个人数据仓库将 SQLite 的全部能力——事务、索引、FTS5、JSON 函数、CTE——汇聚为一个服务于你个人知识管理的数据平台。它不是一个需要运维的"大数据系统"，而是一个 200MB 的单文件，用一条 SQL 就能回答"上周我学了什么"、"这半年的阅读主题趋势"、"哪些书签和笔记涉及同一个话题"。构建它的投入是：设计好 schema、为每个数据源写一个导入脚本、设置定时同步。回报是：你的所有数字记忆，在一个你完全掌控的、可查询的、不会消失的地方。
