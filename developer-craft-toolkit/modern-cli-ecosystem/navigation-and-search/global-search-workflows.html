全局搜索工作流

概述

单独使用 fd、ripgrep 或 fzf 已经很强大，但将它们组合起来能构建出超越任何单一工具能力上限的搜索体验。本文设计三种核心搜索工作流：交互式代码搜索（rg + fzf 实时重载）、文件名搜索（fd + fzf 预览）、以及多级筛选管道。这些工作流覆盖了日常开发中 95% 的搜索场景，且全部可通过 shell 函数一键触发。

核心概念

搜索场景分类

开发者的搜索需求可以分为三个层次：

层次 场景 工具组合 触发方式
L1 知道文件名片段 fd + fzf Ctrl+T
L2 知道代码内容片段 rg + fzf reload 自定义函数
L3 模糊记忆，需要浏览 fd/rg + fzf preview + bat 多阶段管道

实战指南
工作流一：交互式代码搜索（Live Grep）
这是最强大的搜索模式——每个按键都触发 ripgrep 重新搜索，结果实时刷新：


# 完整的 live grep 函数
rgi() {
  local RG_PREFIX="rg --column --line-number --no-heading --color=always --smart-case"
  local INITIAL_QUERY="${*:-}"
  fzf --ansi --disabled --query "$INITIAL_QUERY" \
      --bind "start:reload:$RG_PREFIX {q}" \
      --bind "change:reload:sleep 0.05; $RG_PREFIX {q} || true" \
      --delimiter : \
      --preview 'bat --color=always {1} --highlight-line {2} --line-range {2}:+30' \
      --preview-window 'up,60%,border-bottom,+{2}+3/3,~3' \
      --bind 'enter:become(${EDITOR:-vim} {1} +{2})' \
      --bind 'ctrl-o:execute(code --goto {1}:{2}:{3})' \
      --header 'ENTER=vim | Ctrl-O=VSCode | Ctrl-F=fzf mode | Ctrl-R=rg mode' \
      --bind "ctrl-f:unbind(change)+clear-query+enable-search" \
      --bind "ctrl-r:rebind(change)+disable-search+clear-query"
}

工作流二：文件名搜索（File Finder）
# 增强版文件搜索（带智能预览）
ff() {
  local file
  file=$(fd --type f --hidden --follow --exclude .git |
    fzf --preview '
      case {} in
        *.md|*.txt|*.py|*.js|*.ts|*.rs|*.go|*.sh|*.yaml|*.yml|*.json|*.toml)
          bat --color=always --line-range :80 {} ;;
        *.png|*.jpg|*.gif)
          echo "Image: $(file {})" ;;
        *)
          file {} ;;
      esac
    ' \
    --preview-window 'right:60%' \
    --bind 'ctrl-/:toggle-preview' \
    --bind 'ctrl-e:become(${EDITOR:-vim} {})' \
    --header 'ENTER=select | Ctrl-E=edit'
  )
  [ -n "$file" ] && echo "$file"
}

# 搜索特定类型文件
fft() {
  local ext="${1:-ts}"
  fd --type f -e "$ext" |
    fzf --preview 'bat --color=always --line-range :80 {}' \
        --header "Searching *.$ext files"
}
工作流三：多级筛选管道
当搜索范围很大时，使用多级管道逐步缩小范围：


# 两阶段搜索：先选目录，再搜内容
deep_search() {
  # Stage 1: 选择搜索目录
  local dir
  dir=$(fd --type d --max-depth 3 |
    fzf --preview 'eza --tree --level=1 --icons --color=always {}' \
        --header 'Stage 1: Select directory to search in')
  [ -z "$dir" ] && return

  # Stage 2: 在选定目录中搜索内容
  local RG_PREFIX="rg --column --line-number --no-heading --color=always --smart-case"
  fzf --ansi --disabled \
      --bind "start:reload:$RG_PREFIX '' $dir" \
      --bind "change:reload:$RG_PREFIX {q} $dir || true" \
      --delimiter : \
      --preview 'bat --color=always {1} --highlight-line {2}' \
      --preview-window 'right:60%' \
      --header "Stage 2: Search in $dir" \
      --bind 'enter:become(${EDITOR:-vim} {1} +{2})'
}

# 搜索并批量操作
search_and_act() {
  local files
  files=$(rg -l "${1:-.}" |
    fzf --multi \
        --preview 'rg --color=always "'"$1"'" {}' \
        --header "Select files containing: $1")
  [ -n "$files" ] && echo "$files"
}
工作流四：Git 集成搜索
# 搜索 Git 提交内容
flog() {
  git log --oneline --color=always |
    fzf --ansi --no-sort \
        --preview 'git show --color=always {1}' \
        --preview-window 'right:60%' \
        --bind 'enter:become(git show {1})'
}

# 搜索 Git 暂存的变更
fstash() {
  git stash list |
    fzf --preview 'git stash show --color=always -p {1}' \
        --delimiter : \
        --bind 'enter:become(git stash pop {1})'  \
        --bind 'ctrl-d:execute(git stash drop {1})+reload(git stash list)'
}

# 搜索变更的文件
fchanged() {
  git diff --name-only |
    fzf --preview 'git diff --color=always -- {}' \
        --preview-window 'right:70%' \
        --bind 'enter:become(${EDITOR:-vim} {})'
}
最佳实践
性能优化
# 大型项目优化：限制搜索范围
export FZF_DEFAULT_COMMAND='fd --type f --hidden --max-depth 8 --exclude .git'

# ripgrep 搜索优化：限制结果数
rg --max-count 1 "pattern"   # 每个文件最多 1 个匹配

# fzf 性能：大数据集使用 --no-sort
very_large_search() {
  rg --files | fzf --no-sort --preview 'bat --color=always --line-range :50 {}'
}

# 延迟预览（避免快速滚动时的 CPU 浪费）
# fzf 默认有 100ms 的预览延迟，可以增加
fzf --preview 'bat --color=always {}' --preview-window '~3'

总结
全局搜索工作流的核心设计思想是「分层搜索、渐进筛选」。不同的搜索场景对应不同的工具组合：知道名字用 fd + fzf，知道内容用 rg + fzf reload，记忆模糊用多级管道。将这些工作流封装为 shell 函数后，搜索操作从「想命令 → 敲参数 → 看结果 → 跳转」简化为「调函数 → 交互式选择 → 直接到位」。这种效率提升是累积性的——每天节省的零碎时间加起来，一年能省下数十小时的有效工作时间。
