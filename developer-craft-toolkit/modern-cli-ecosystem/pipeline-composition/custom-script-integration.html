自定义脚本集成

概述

当工具链复杂度超越 shell 函数的表达能力时，需要将其封装为独立的脚本或小型 CLI 工具。本文讲解如何将现代终端工具集成到可维护、可移植、可分享的脚本中——包括工具可用性检测与优雅降级、跨平台兼容处理、脚本中的交互式组件设计、以及如何将成熟的管道工作流发布为可安装的 CLI 工具。

工具检测与优雅降级
#!/usr/bin/env bash
set -euo pipefail

# === 工具检测框架 ===
declare -A TOOLS=(
  [fd]="fd"
  [rg]="rg"
  [bat]="bat"
  [fzf]="fzf"
  [eza]="eza"
)

# 检测并设置回退
has() { command -v "$1" &>/dev/null; }

# 文件搜索：fd → find
find_files() {
  if has fd; then
    fd --type f "$@"
  else
    find . -type f -name "*${1:-}*" 2>/dev/null
  fi
}

# 内容搜索：rg → grep
search_content() {
  if has rg; then
    rg "$@"
  else
    grep -rn "$@" .
  fi
}

# 文件查看：bat → cat
view_file() {
  if has bat; then
    bat --color=always --paging=never "$@"
  else
    cat "$@"
  fi
}

# 目录列表：eza → ls
list_dir() {
  if has eza; then
    eza --icons --group-directories-first "$@"
  else
    ls -la "$@"
  fi
}

# 交互选择：fzf → select
interactive_select() {
  if has fzf; then
    fzf "$@"
  else
    # 回退到 bash select
    local items=()
    while IFS= read -r line; do items+=("$line"); done
    select item in "${items[@]}"; do
      [ -n "$item" ] && echo "$item" && break
    done
  fi
}
跨平台兼容处理
#!/usr/bin/env bash
# === 平台检测 ===
OS="$(uname -s)"
ARCH="$(uname -m)"

# 剪贴板操作
clipboard_copy() {
  case "$OS" in
    Darwin) pbcopy ;;
    Linux)
      if has xclip; then xclip -selection clipboard
      elif has xsel; then xsel --clipboard --input
      elif has wl-copy; then wl-copy
      else cat > /tmp/clipboard_fallback
      fi ;;
    MINGW*|MSYS*) clip.exe ;;
  esac
}

# 打开文件/URL
open_it() {
  case "$OS" in
    Darwin) open "$@" ;;
    Linux)  xdg-open "$@" ;;
    MINGW*) start "$@" ;;
  esac
}

# fd 的跨平台包名处理
setup_fd() {
  if has fd; then
    FD_CMD="fd"
  elif has fdfind; then  # Debian/Ubuntu
    FD_CMD="fdfind"
  else
    FD_CMD=""
  fi
}

# bat 的跨平台包名处理
setup_bat() {
  if has bat; then
    BAT_CMD="bat"
  elif has batcat; then  # Debian/Ubuntu
    BAT_CMD="batcat"
  else
    BAT_CMD="cat"
  fi
}
脚本中的交互式组件
#!/usr/bin/env bash
# === 交互式项目管理脚本 ===

# 项目选择器
select_project() {
  local project_dirs=("$HOME/code" "$HOME/work" "$HOME/personal")
  local all_projects=""

  for dir in "${project_dirs[@]}"; do
    [ -d "$dir" ] && all_projects+="$(find_files -t d --max-depth 1 . "$dir")\n"
  done

  echo -e "$all_projects" | grep -v '^$' |
    fzf --preview 'eza --tree --level=1 --icons --color=always {}' \
        --header 'Select project'
}

# 确认操作（带颜色）
confirm() {
  local msg="${1:-Continue?}"
  echo -en "\033[33m$msg [y/N] \033[0m"
  read -r answer
  [[ "$answer" =~ ^[Yy]$ ]]
}

# 进度指示器
progress() {
  local current=$1 total=$2 label="${3:-Processing}"
  local pct=$((current * 100 / total))
  local bar_width=40
  local filled=$((pct * bar_width / 100))
  local empty=$((bar_width - filled))
  printf "\r\033[36m%s\033[0m [%s%s] %d%%" \
    "$label" \
    "$(printf '#%.0s' $(seq 1 $filled))" \
    "$(printf '-%.0s' $(seq 1 $empty))" \
    "$pct"
  [ "$current" -eq "$total" ] && echo
}

# 完整的交互式工作流
main() {
  echo "=== Project Manager ==="

  # Step 1: 选择项目
  local project
  project=$(select_project) || exit 0
  echo "Selected: $project"

  # Step 2: 在项目中搜索
  cd "$project" || exit 1
  local file
  file=$(find_files | interactive_select --preview "$(setup_bat; echo "$BAT_CMD") --color=always {}") || exit 0

  # Step 3: 确认操作
  if confirm "Open $file in editor?"; then
    ${EDITOR:-vim} "$file"
  fi
}

main "$@"
封装为可分发的 CLI 工具
#!/usr/bin/env bash
# === smartfind: 智能文件搜索工具 ===
# 安装：cp smartfind ~/.local/bin/ && chmod +x ~/.local/bin/smartfind
set -euo pipefail

VERSION="1.0.0"
SCRIPT_NAME="$(basename "$0")"

# 帮助信息
usage() {
  cat <<EOF
$SCRIPT_NAME v$VERSION - Smart file search with modern tools

Usage: $SCRIPT_NAME [options] [query]

Options:
  -t, --type TYPE    File type filter (ts, py, rs, etc.)
  -d, --dir DIR      Search directory (default: .)
  -e, --edit         Open selected file in editor
  -c, --content      Search file content (not names)
  -m, --multi        Enable multi-select
  -h, --help         Show this help

Examples:
  $SCRIPT_NAME                    # Interactive file picker
  $SCRIPT_NAME -t ts              # Find TypeScript files
  $SCRIPT_NAME -c "useState"      # Search content
  $SCRIPT_NAME -t py -e -d src/   # Find Python in src/, open in editor
EOF
}

# 参数解析
TYPE="" DIR="." EDIT=false CONTENT=false MULTI=""
while [[ $# -gt 0 ]]; do
  case $1 in
    -t|--type) TYPE="$2"; shift 2 ;;
    -d|--dir) DIR="$2"; shift 2 ;;
    -e|--edit) EDIT=true; shift ;;
    -c|--content) CONTENT=true; shift ;;
    -m|--multi) MULTI="--multi"; shift ;;
    -h|--help) usage; exit 0 ;;
    --) shift; break ;;
    -*) echo "Unknown option: $1" >&2; exit 1 ;;
    *) break ;;
  esac
done
QUERY="${*:-}"

# 依赖检查
check_deps() {
  local missing=()
  has fzf || missing+=(fzf)
  if $CONTENT; then
    has rg || missing+=(ripgrep)
  else
    has fd || missing+=(fd)
  fi
  if [ ${#missing[@]} -gt 0 ]; then
    echo "Missing required tools: ${missing[*]}" >&2
    echo "Install: brew install ${missing[*]}" >&2
    exit 1
  fi
}

# 主逻辑
main() {
  check_deps

  local result
  if $CONTENT; then
    # 内容搜索模式
    local rg_cmd="rg --column --line-number --no-heading --color=always --smart-case"
    [ -n "$TYPE" ] && rg_cmd+=" --type $TYPE"
    result=$(fzf --ansi --disabled $MULTI \
      --query "$QUERY" \
      --bind "start:reload:$rg_cmd {q} $DIR" \
      --bind "change:reload:$rg_cmd {q} $DIR || true" \
      --delimiter : \
      --preview 'bat --color=always {1} --highlight-line {2} --line-range {2}:+20')
  else
    # 文件名搜索模式
    local fd_args=(--type f)
    [ -n "$TYPE" ] && fd_args+=(-e "$TYPE")
    result=$(fd "${fd_args[@]}" . "$DIR" |
      fzf $MULTI --query "$QUERY" \
          --preview 'bat --color=always --line-range :80 {}')
  fi

  [ -z "$result" ] && exit 0

  if $EDIT; then
    if $CONTENT; then
      local file=$(echo "$result" | cut -d: -f1)
      local line=$(echo "$result" | cut -d: -f2)
      ${EDITOR:-vim} "$file" +$line
    else
      ${EDITOR:-vim} "$result"
    fi
  else
    echo "$result"
  fi
}

总结
自定义脚本集成是个人工具链成熟度的最高阶段。通过工具检测 + 优雅降级，你的脚本可以在任何环境中运行；通过交互式组件，脚本获得了现代 TUI 的体验；通过参数化和帮助系统，脚本变成了可分享的工具。从 shell 函数到可分发的 CLI 工具，这条路径让你的生产力投资可以惠及团队和社区。记住：最好的工具是你自己打造的工具——因为它完全贴合你的工作流。
