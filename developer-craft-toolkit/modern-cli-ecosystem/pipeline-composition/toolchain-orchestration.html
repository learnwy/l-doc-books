工具链编排

概述

单条管道解决单次问题，工具链编排解决反复出现的问题类型。本文讲解如何从单条管道进化为可复用的工具链——通过 shell 函数封装、环境变量参数化、配置文件外置来实现工具组合的系统化。最终目标是建立个人的「命令词汇表」：每一个高频操作都有一个简洁的命令名，背后是精心设计的工具组合。

工具链设计模式
Chain 模式：线性串联，每步处理前步输出


fd -e ts | fzf | xargs vim     # find → select → open
Fan-out 模式：一个输入分发到多个处理器


fd -e log | tee >(wc -l > /tmp/count) >(gzip > archive.gz)
Gather 模式：多个源汇聚到一个处理器


{ rg -l "TODO" src/; rg -l "FIXME" src/; } | sort -u | fzf

封装为 shell 函数
# 项目搜索函数：结合 rg + fzf + bat 预览
psearch() {
  local query="${1:-}"
  local dir="${2:-.}"
  local rg_cmd="rg --column --line-number --no-heading --color=always --smart-case"

  if [ -n "$query" ]; then
    # 有初始查询：直接搜索
    eval "$rg_cmd '$query' '$dir'" |
      fzf --ansi --delimiter : \
          --preview 'bat --color=always {1} --highlight-line {2} --line-range {2}:+20' \
          --preview-window 'right:60%' \
          --bind 'enter:become(${EDITOR:-vim} {1} +{2})'
  else
    # 无查询：进入 live grep 模式
    fzf --ansi --disabled \
        --bind "start:reload:$rg_cmd '' '$dir'" \
        --bind "change:reload:$rg_cmd {q} '$dir' || true" \
        --delimiter : \
        --preview 'bat --color=always {1} --highlight-line {2} --line-range {2}:+20' \
        --preview-window 'right:60%' \
        --bind 'enter:become(${EDITOR:-vim} {1} +{2})'
  fi
}

# 项目文件导航函数
pnav() {
  local dir="${1:-.}"
  local depth="${2:-5}"
  fd --type f --max-depth "$depth" . "$dir" |
    fzf --preview 'bat --color=always --line-range :80 {}' \
        --preview-window 'right:55%' \
        --bind 'enter:become(${EDITOR:-vim} {})' \
        --bind 'ctrl-d:become(cd $(dirname {}))'
}
参数化工具链
# 通过环境变量实现配置化
export TOOLCHAIN_EDITOR="${EDITOR:-vim}"
export TOOLCHAIN_PREVIEW_LINES=80
export TOOLCHAIN_FZF_HEIGHT="70%"

# 统一的 fzf 选项生成
_fzf_opts() {
  echo "--height $TOOLCHAIN_FZF_HEIGHT --border --info=inline-right"
}

# 可配置的搜索函数
csearch() {
  local opts=$(_fzf_opts)
  local preview_cmd="bat --color=always --line-range :${TOOLCHAIN_PREVIEW_LINES} {}"
  fd --type f | fzf $opts --preview "$preview_cmd" |
    xargs -r "$TOOLCHAIN_EDITOR"
}
工具链组合示例库
# 1. 代码质量检查链
lint_check() {
  local ext="${1:-ts}"
  fd -e "$ext" |
    fzf -m --preview 'bat --color=always {}' \
        --header "Select $ext files to lint" |
    xargs -r npx eslint --fix
}

# 2. 依赖分析链
dep_analyze() {
  rg "^import|^from|require\(" -l |
    fzf --preview 'rg --color=always "import|require" {}' |
    xargs -I {} sh -c 'echo "=== {} ==="; rg "import|require" {}'
}

# 3. 大文件清理链
cleanup_large() {
  local threshold="${1:-10m}"
  fd --type f --size "+$threshold" |
    fzf -m --preview 'du -sh {} && file {}' \
        --header "Files larger than $threshold (TAB to select)" |
    xargs -r -p rm  # -p 要求确认
}

# 4. TODO 管理链
todo_manage() {
  rg "TODO|FIXME|HACK|XXX" --color=always -n |
    fzf --ansi --multi \
        --preview 'bat --color=always {1} --highlight-line {2}' \
        --delimiter : \
        --header 'Select TODOs to address' |
    cut -d: -f1-2 | sort -u
}

# 5. 配置文件编辑链
edit_config() {
  fd -e yaml -e yml -e toml -e json -e conf --hidden |
    fzf --preview 'bat --color=always {}' \
        --header 'Select config file to edit' |
    xargs -r "$TOOLCHAIN_EDITOR"
}
工具链的错误处理
# 健壮的工具链函数模板
robust_chain() {
  # 1. 前置检查
  command -v fzf >/dev/null || { echo "fzf required"; return 1; }
  command -v rg >/dev/null || { echo "ripgrep required"; return 1; }

  # 2. 数据获取（可能失败）
  local data
  data=$(rg -l "$1" 2>/dev/null)
  if [ -z "$data" ]; then
    echo "No files match pattern: $1"
    return 0
  fi

  # 3. 交互选择（用户可取消）
  local selected
  selected=$(echo "$data" | fzf --preview 'bat --color=always {}')
  [ -z "$selected" ] && return 0  # 用户取消，正常退出

  # 4. 执行动作
  ${EDITOR:-vim} "$selected"
}

总结
工具链编排是从「临时管道」到「个人 CLI 框架」的关键跨越。核心方法论是：重复三次以上的管道封装为函数，需要参数化的函数通过环境变量配置，需要分享的函数提取为独立脚本。通过统一的命名约定和文件组织，你可以逐步积累一套个人的「命令词汇表」——这套词汇表就是你的终端生产力基础设施，它会随着使用不断优化，成为你独有的效率优势。
