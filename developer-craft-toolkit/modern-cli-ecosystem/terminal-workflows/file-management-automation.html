文件管理自动化

概述

文件管理是开发者最容易积累「技术债」的领域之一——下载目录堆满杂物、项目中散落临时文件、node_modules 吞噬磁盘空间、照片和文档缺乏组织。本文将 fd、fzf、bat 等现代工具组合为文件管理自动化工作流，覆盖批量重命名、大文件清理、项目归档、重复检测等高频场景。每个工作流都设计为安全优先——先预览再执行，避免误操作。

核心概念

文件管理的安全原则

原则 实现方式
先预览后执行 所有批量操作先 dry-run 显示计划
可撤销 移入回收站而非直接 rm
交互确认 危险操作使用 fzf 多选确认
日志记录 操作日志写入 ~/.local/share/file-ops.log

批量重命名
# 交互式批量重命名（预览 + 确认）
batch_rename() {
  local pattern="${1:?Usage: batch_rename PATTERN [REPLACEMENT]}"
  local replacement="${2:-}"
  local dir="${3:-.}"

  # 列出匹配的文件
  local files
  files=$(fd "$pattern" "$dir" --type f)
  [ -z "$files" ] && { echo "No files match: $pattern"; return 0; }

  # 预览重命名计划
  echo "=== Rename Preview ==="
  echo "$files" | while IFS= read -r f; do
    local new_name
    if [ -n "$replacement" ]; then
      new_name=$(echo "$f" | sed "s/$pattern/$replacement/g")
    else
      new_name=$(echo "$f" | sed "s/$pattern//g")
    fi
    echo "  $f → $new_name"
  done

  # 确认执行
  echo ""
  read -p "Execute rename? [y/N] " -r
  [[ $REPLY =~ ^[Yy]$ ]] || return 0

  echo "$files" | while IFS= read -r f; do
    local new_name
    if [ -n "$replacement" ]; then
      new_name=$(echo "$f" | sed "s/$pattern/$replacement/g")
    else
      new_name=$(echo "$f" | sed "s/$pattern//g")
    fi
    mv "$f" "$new_name"
    echo "Renamed: $f → $new_name"
  done
}

# fzf 交互式选择要重命名的文件
rename_select() {
  local files
  files=$(fd --type f | fzf -m --preview 'bat --color=always {}' \
    --header 'Select files to rename (TAB=multi)')
  [ -z "$files" ] && return 0

  echo "Selected files:"
  echo "$files"
  echo ""
  read -p "New pattern (use {n} for number, {name} for original): " -r pattern

  local n=1
  echo "$files" | while IFS= read -r f; do
    local ext="${f##*.}"
    local base="${f%.*}"
    local new_name=$(echo "$pattern" | sed "s/{n}/$n/g" | sed "s/{name}/$base/g")
    [ -n "$ext" ] && new_name="$new_name.$ext"
    echo "  $f → $new_name"
    n=$((n + 1))
  done
}

# 文件扩展名批量修改
rename_ext() {
  local old_ext="${1:?Usage: rename_ext OLD_EXT NEW_EXT}"
  local new_ext="${2:?Usage: rename_ext OLD_EXT NEW_EXT}"
  fd -e "$old_ext" --exec mv {} {.}."$new_ext" \;
}
大文件清理
# 磁盘空间分析 + 交互式清理
disk_cleanup() {
  local threshold="${1:-50m}"
  echo "Finding files larger than $threshold..."

  fd --type f --size "+$threshold" --exec du -h {} \; |
    sort -rh |
    fzf --multi \
        --preview 'file {2}; echo "---"; head -c 1000 {2} 2>/dev/null | bat --color=always -l txt' \
        --header "Large files (>$threshold) | TAB to select | ENTER to delete" \
        --bind 'ctrl-i:toggle' |
    awk '{print $2}' |
    while IFS= read -r f; do
      echo "Moving to trash: $f"
      # macOS: 移入回收站
      if command -v trash >/dev/null; then
        trash "$f"
      else
        mv "$f" ~/.Trash/ 2>/dev/null || rm -i "$f"
      fi
    done
}

# node_modules 清理器
clean_node_modules() {
  echo "Searching for node_modules directories..."
  fd -t d "node_modules" --prune |
    fzf -m --preview 'du -sh {}' \
        --header 'Select node_modules to delete (TAB=multi)' |
    while IFS= read -r dir; do
      local size=$(du -sh "$dir" | cut -f1)
      echo "Removing: $dir ($size)"
      rm -rf "$dir"
    done
}

# 过期文件清理（超过 N 天未修改）
clean_stale() {
  local days="${1:-90}"
  local dir="${2:-.}"
  fd --type f --changed-before "${days}d" . "$dir" |
    fzf -m --preview 'stat {} && echo "---" && head -5 {}' \
        --header "Files not modified in $days days"
}
项目归档
# 智能项目归档（排除生成文件）
archive_project() {
  local project="${1:-.}"
  local name=$(basename "$(cd "$project" && pwd)")
  local timestamp=$(date +%Y%m%d)
  local archive="${name}_${timestamp}.tar.gz"

  echo "Archiving: $project → $archive"
  echo "Excluding: node_modules, dist, target, .git, __pycache__"

  tar -czf "$archive" \
    --exclude='node_modules' \
    --exclude='dist' \
    --exclude='target' \
    --exclude='.git' \
    --exclude='__pycache__' \
    --exclude='.next' \
    --exclude='build' \
    -C "$(dirname "$project")" "$(basename "$project")"

  local size=$(du -sh "$archive" | cut -f1)
  echo "Created: $archive ($size)"
}

# 批量归档不活跃项目
archive_inactive() {
  local days="${1:-180}"
  local code_dir="${2:-$HOME/code}"

  echo "Projects not modified in $days days:"
  fd --type d --max-depth 1 --changed-before "${days}d" . "$code_dir" |
    fzf -m --preview 'echo "Last modified:"; stat -f "%Sm" {} ; echo "Size:"; du -sh {}' \
        --header "Select projects to archive" |
    while IFS= read -r proj; do
      archive_project "$proj"
      echo "Archived: $proj"
    done
}
重复文件检测
# 基于文件大小 + hash 的重复检测
find_duplicates() {
  local dir="${1:-.}"
  echo "Scanning for duplicates in: $dir"

  # Step 1: 按大小分组
  fd --type f . "$dir" --exec du -b {} \; |
    sort -n |
    awk 'BEGIN{prev_size=0; prev_file=""}
         {if($1==prev_size) {if(prev_file!="") print prev_file; print $2; prev_file=""}
          else {prev_size=$1; prev_file=$2}}' |
    # Step 2: 对同大小文件计算 hash
    xargs -I {} sh -c 'echo "$(md5 -q "{}" 2>/dev/null || md5sum "{}" | cut -d" " -f1) {}"' |
    sort |
    # Step 3: 找出 hash 相同的文件
    awk 'BEGIN{prev=""} {if(substr($0,1,32)==prev) print; prev=substr($0,1,32)}' |
    fzf --multi \
        --preview 'bat --color=always {2}' \
        --header 'Duplicate files (TAB to select for deletion)'
}

# 简化版：使用 fdupes（如果安装了的话）
find_dupes_simple() {
  if command -v fdupes >/dev/null; then
    fdupes -r "${1:-.}" | bat --paging=never -l txt
  else
    echo "Install fdupes: brew install fdupes"
    echo "Or use find_duplicates function above"
  fi
}
文件整理自动化
# 下载目录自动整理
organize_downloads() {
  local dl_dir="${1:-$HOME/Downloads}"
  local doc_dir="$dl_dir/Documents"
  local img_dir="$dl_dir/Images"
  local arc_dir="$dl_dir/Archives"
  local code_dir="$dl_dir/Code"

  mkdir -p "$doc_dir" "$img_dir" "$arc_dir" "$code_dir"

  # 按类型移动
  fd -e pdf -e doc -e docx -e xlsx -e pptx . "$dl_dir" --max-depth 1 --exec mv {} "$doc_dir/" \;
  fd -e png -e jpg -e jpeg -e gif -e svg -e webp . "$dl_dir" --max-depth 1 --exec mv {} "$img_dir/" \;
  fd -e zip -e tar -e gz -e 7z -e rar . "$dl_dir" --max-depth 1 --exec mv {} "$arc_dir/" \;
  fd -e py -e js -e ts -e rs -e go -e sh . "$dl_dir" --max-depth 1 --exec mv {} "$code_dir/" \;

  echo "Organized $(fd --type f . "$dl_dir" --max-depth 1 | wc -l) remaining files"
  eza --icons "$dl_dir"
}

# 空目录清理
clean_empty_dirs() {
  local dir="${1:-.}"
  fd --type d --empty . "$dir" |
    fzf -m --header 'Empty directories (TAB to select for removal)' |
    xargs -r rmdir
}

安全封装模式
# 所有文件操作都通过安全层
safe_rm() {
  if command -v trash >/dev/null; then
    trash "$@"
  elif [ -d ~/.Trash ]; then
    mv "$@" ~/.Trash/
  else
    rm -i "$@"
  fi
}

# 操作日志
log_operation() {
  local logfile="${XDG_DATA_HOME:-$HOME/.local/share}/file-ops.log"
  echo "[$(date -Iseconds)] $*" >> "$logfile"
}

总结
文件管理自动化的核心价值是「把重复性的文件操作从手动变为可复用的工具链」。通过 fd 精确定位目标文件、fzf 提供安全的交互式确认、bat 提供清晰的预览，每个操作都是「看清楚再动手」。将这些组合为 shell 函数后，原本需要 10 分钟手动操作的清理/整理工作缩短为几秒钟。更重要的是，这些函数可以定期运行（通过 cron 或 launchd），让文件管理从被动应对变为主动维护，永远保持工作环境的整洁有序。
