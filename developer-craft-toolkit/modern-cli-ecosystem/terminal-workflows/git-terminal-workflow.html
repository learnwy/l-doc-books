Git 终端工作流

概述

Git 是开发者每天使用频率最高的工具之一，但其 CLI 界面对交互式操作并不友好。现代工具组合——delta（语法高亮 diff）、lazygit（全功能 Git TUI）、fzf（交互式选择器）——将 Git 体验从「记忆命令 + 阅读原始文本」升级为「可视化操作 + 即时预览」。本文构建一套完整的 Git 终端工作流，覆盖从日常 diff 查看到复杂 rebase 操作的所有场景。

delta 配置详解
delta 作为 Git pager，让 git diff、git log -p、git show 的输出从纯文本变为带语法高亮的现代视图：


# ~/.gitconfig — delta 完整配置
[core]
    pager = delta

[interactive]
    diffFilter = delta --color-only

[delta]
    navigate = true           # 用 n/N 跳转 diff 段落
    side-by-side = true       # 并排对比视图
    line-numbers = true       # 显示行号
    syntax-theme = TwoDark    # 语法高亮主题
    file-style = bold yellow ul
    file-decoration-style = none
    hunk-header-decoration-style = cyan box ul
    minus-style = syntax "#3f1919"
    plus-style = syntax "#1a3a1a"
    line-numbers-minus-style = "#b5483f"
    line-numbers-plus-style = "#3fb551"

[merge]
    conflictstyle = diff3

[diff]
    colorMoved = default

fzf Git 函数库
# Git 分支切换（fzf 交互式）
gco() {
  local branch
  branch=$(git branch --all --format='%(refname:short)' |
    grep -v HEAD |
    fzf --preview 'git log --oneline --graph --color=always -20 {}' \
        --header 'Checkout branch') &&
  git checkout "$branch"
}

# Git 提交浏览
glog() {
  git log --oneline --color=always --graph |
    fzf --ansi --no-sort --reverse \
        --preview 'git show --color=always {1}' \
        --preview-window 'right:60%' \
        --bind 'enter:become(git show --color=always {1} | delta)' \
        --bind 'ctrl-y:execute-silent(echo {1} | pbcopy)+abort' \
        --header 'ENTER=show | Ctrl-Y=copy hash'
}

# Git stash 管理
gstash() {
  git stash list |
    fzf --preview 'git stash show -p --color=always {1}' \
        --delimiter : \
        --bind 'enter:become(git stash pop {1})' \
        --bind 'ctrl-d:execute(git stash drop {1})+reload(git stash list)' \
        --bind 'ctrl-a:become(git stash apply {1})' \
        --header 'ENTER=pop | Ctrl-D=drop | Ctrl-A=apply'
}

# Git add 交互式（选择性 stage）
gadd() {
  git diff --name-only --diff-filter=ACMR |
    fzf --multi \
        --preview 'git diff --color=always -- {}' \
        --preview-window 'right:70%' \
        --header 'Select files to stage (TAB=multi)' |
    xargs -r git add
}

# Git 查看变更文件并编辑
gdiff() {
  git diff --name-only |
    fzf --preview 'git diff --color=always -- {}' \
        --preview-window 'right:70%' \
        --bind 'enter:become(${EDITOR:-vim} {})'
}
delta 高级用法
# 单独使用 delta（不通过 git）
diff file1.py file2.py | delta                  # 两文件对比
delta file1.py file2.py                         # 直接对比

# 控制 delta 输出模式
git diff --stat                                  # 文件级变更统计
git diff | delta --side-by-side --width 200     # 临时开宽屏并排
git diff | delta --line-numbers-left-format=""  # 隐藏左侧行号

# delta 与 grep 结合
git log -p --color=always | delta | rg "function"  # 在 diff 中搜索

# 查看特定提交的完整变更
git show abc123 | delta --side-by-side

# 对比两个分支
git diff main..feature | delta
最佳实践
日常 Git 工作流模板
# 早上开始工作
gco          # fzf 选择分支
git pull     # 拉取最新

# 开发过程中
# ... 编码 ...
gadd         # fzf 选择性 stage
lazygit      # 或 lg 进入 TUI 操作

# 代码审查
git diff main..HEAD | delta --side-by-side   # 查看所有变更
gdiff        # fzf 逐文件查看变更

# 处理冲突
lazygit      # 在 TUI 中解决冲突更直观

# 结束工作
glog         # 检查今天的提交历史

总结
Git 终端工作流的核心升级路径：delta 让 diff 可读，lazygit 让操作可视，fzf 函数让选择交互。三者互补而非替代——简单的 git diff 查看用 delta 美化后的输出即可，复杂的 stage/rebase/merge 操作进 lazygit TUI，而分支切换、commit 选择等「选一个」场景用 fzf 函数最快。根据操作的复杂度选择合适的工具层级，你的 Git 体验将从「命令行的痛苦记忆」变为「视觉化的流畅操作」。
