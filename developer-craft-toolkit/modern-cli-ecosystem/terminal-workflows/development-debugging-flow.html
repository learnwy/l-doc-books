开发调试流

概述

开发调试的终端工作流关注一个核心问题：如何从海量信息中快速定位问题根因。传统方式是在多个终端窗口间切换——一个看日志、一个查进程、一个测网络。通过将 ripgrep、fzf、bat 等工具组合成调试管道，我们可以在单个终端内完成「发现异常 → 缩小范围 → 定位根因 → 执行修复」的完整闭环。

日志搜索与实时过滤
# 实时日志过滤（tail + rg + bat 高亮）
log_watch() {
  local pattern="${1:-ERROR|WARN|FATAL}"
  local logfile="${2:-/var/log/app.log}"
  tail -f "$logfile" | rg --line-buffered "$pattern" | bat --paging=never -l log
}

# 历史日志搜索（rg + fzf 交互式选择上下文）
log_search() {
  local pattern="${1:-error}"
  local logdir="${2:-./logs}"
  rg -n --color=always "$pattern" "$logdir" |
    fzf --ansi --delimiter : \
        --preview 'bat --color=always {1} --highlight-line {2} --line-range $(({2}-10)):$(({2}+10))' \
        --preview-window 'right:60%' \
        --header "Searching: $pattern in $logdir"
}

# 多日志文件聚合搜索
log_aggregate() {
  local pattern="${1:-exception}"
  fd -e log -e txt . ./logs |
    xargs rg -l "$pattern" 2>/dev/null |
    fzf --preview "rg --color=always -C 3 '$pattern' {}" \
        --preview-window 'right:70%' \
        --header "Files containing: $pattern"
}

# 按时间范围过滤日志
log_timerange() {
  local start="${1:-2024-01-01}"
  local end="${2:-2099-12-31}"
  rg "\d{4}-\d{2}-\d{2}" --color=always |
    awk -v s="$start" -v e="$end" -F'[^0-9-]' '
      { for(i=1;i<=NF;i++) if($i ~ /^[0-9]{4}-[0-9]{2}-[0-9]{2}$/ && $i >= s && $i <= e) { print; next } }
    ' | fzf --ansi
}
进程诊断工作流
# 交互式进程管理
proc_manage() {
  local selected
  selected=$(ps aux --sort=-%cpu |
    fzf --header-lines=1 \
        --preview 'echo "PID: {2}\nCPU: {3}%\nMEM: {4}%\nSTART: {9}\nCMD: {11..}"' \
        --preview-window 'up:5:wrap' \
        --bind 'ctrl-k:execute(kill -9 {2})+reload(ps aux --sort=-%cpu)' \
        --bind 'ctrl-s:execute(kill -STOP {2})+reload(ps aux --sort=-%cpu)' \
        --bind 'ctrl-c:execute(kill -CONT {2})+reload(ps aux --sort=-%cpu)' \
        --header 'Ctrl-K=kill | Ctrl-S=stop | Ctrl-C=continue')
  [ -n "$selected" ] && echo "Selected PID: $(echo "$selected" | awk '{print $2}')"
}

# 查找内存占用最高的进程
proc_memory() {
  ps aux --sort=-%mem | head -20 |
    fzf --header-lines=1 \
        --preview 'echo "Process Details:"; ps -p {2} -o pid,ppid,user,%cpu,%mem,vsz,rss,stat,start,time,command' \
        --preview-window 'up:5:wrap'
}

# 按名称搜索并管理进程
proc_find() {
  local name="${1:-}"
  ps aux | rg -i "${name:-.}" |
    fzf --header 'Select process' \
        --preview 'lsof -p {2} 2>/dev/null | head -20' \
        --bind 'ctrl-k:execute(kill {2})+abort'
}
端口与网络排查
# 端口占用查询（交互式）
port_check() {
  lsof -iTCP -sTCP:LISTEN -n -P |
    fzf --header-lines=1 \
        --preview 'echo "Full info:"; lsof -p {2} 2>/dev/null | head -30' \
        --bind 'ctrl-k:execute(kill {2})+reload(lsof -iTCP -sTCP:LISTEN -n -P)' \
        --header 'Listening ports | Ctrl-K=kill process'
}

# 查找占用特定端口的进程
port_who() {
  local port="${1:?Usage: port_who PORT}"
  lsof -i ":$port" -n -P
}

# 网络连接概览
net_overview() {
  {
    echo "=== Listening Ports ==="
    lsof -iTCP -sTCP:LISTEN -n -P 2>/dev/null
    echo ""
    echo "=== Established Connections ==="
    lsof -iTCP -sTCP:ESTABLISHED -n -P 2>/dev/null | head -20
  } | bat --paging=never -l yaml
}
环境变量调试
# 交互式环境变量浏览
env_browse() {
  env | sort |
    fzf --preview 'echo {1} | cut -d= -f2-' \
        --delimiter '=' \
        --bind 'ctrl-y:execute-silent(echo {1} | cut -d= -f2- | pbcopy)' \
        --header 'Ctrl-Y=copy value'
}

# 搜索特定环境变量
env_search() {
  local pattern="${1:-PATH}"
  env | rg -i "$pattern" | bat -l ini --paging=never
}

# PATH 可视化
path_show() {
  echo "$PATH" | tr ':' '\n' |
    fzf --preview '[ -d {} ] && eza --icons --color=always {} || echo "NOT FOUND: {}"' \
        --header 'PATH entries'
}

# 对比两个环境的变量差异
env_diff() {
  local env1="${1:-.env.local}"
  local env2="${2:-.env.production}"
  delta "$env1" "$env2"
}
配置文件问题定位
# 查找包含特定配置项的文件
config_find() {
  local key="${1:?Usage: config_find KEY}"
  fd -e yaml -e yml -e toml -e json -e conf -e ini --hidden |
    xargs rg -l "$key" 2>/dev/null |
    fzf --preview "rg --color=always -C 2 '$key' {}" \
        --preview-window 'right:60%' \
        --bind 'enter:become(${EDITOR:-vim} {})'
}

# 配置文件语法检查
config_validate() {
  local file="${1:?Usage: config_validate FILE}"
  case "$file" in
    *.json) jq . "$file" >/dev/null && echo "Valid JSON" || echo "Invalid JSON" ;;
    *.yaml|*.yml) yq . "$file" >/dev/null && echo "Valid YAML" || echo "Invalid YAML" ;;
    *.toml) command -v taplo >/dev/null && taplo check "$file" || echo "Install taplo for TOML validation" ;;
    *) echo "Unknown format" ;;
  esac
}

调试会话管理
# 启动调试会话（开多个相关面板）
debug_session() {
  local project="${1:-.}"
  # 在 tmux 中创建调试布局
  tmux new-window -n debug
  tmux split-window -h -p 50
  tmux split-window -v -p 50
  tmux select-pane -t 0
  tmux send-keys "cd $project && tail -f logs/*.log | rg --line-buffered 'ERROR|WARN'" Enter
  tmux select-pane -t 1
  tmux send-keys "cd $project && lazygit" Enter
  tmux select-pane -t 2
  tmux send-keys "cd $project" Enter
}
异常检测一键脚本
# 快速健康检查
health_check() {
  echo "=== Disk Usage ==="
  df -h | head -5

  echo -e "\n=== High CPU Processes ==="
  ps aux --sort=-%cpu | head -5

  echo -e "\n=== Memory Usage ==="
  free -h 2>/dev/null || vm_stat

  echo -e "\n=== Listening Ports ==="
  lsof -iTCP -sTCP:LISTEN -n -P 2>/dev/null | tail -5

  echo -e "\n=== Recent Errors in Logs ==="
  fd -e log --changed-within 1h | xargs rg -c "ERROR" 2>/dev/null | sort -t: -k2 -rn | head -5
}

总结
开发调试流的核心设计理念是「让信息主动浮现，而非被动搜索」。通过 ripgrep 的模式匹配能力精准提取关键信息，通过 fzf 的交互式界面在大量候选中快速定位目标，通过 bat 的语法高亮让上下文一目了然。将这些组合封装为函数后，调试从「打开多个窗口、敲多条命令、手动比对」简化为「调用一个函数、交互式选择、直接到位」。这种效率差异在紧急排查场景中尤为明显。
