灵敏度分析

核心概念

灵敏度分析研究模型输出对输入参数变化的响应程度。当模型参数存在不确定性时，灵敏度分析帮助我们识别哪些参数对结果影响最大（关键参数），哪些参数的不确定性可以容忍（次要参数）。这对于参数校准的优先级排序和模型鲁棒性评估至关重要。

局部方法在参数基准值附近求偏导数，计算快但只反映局部行为。全局方法在整个参数空间内采样，捕捉非线性效应和参数交互作用，但计算成本高。两者互补使用是最佳实践。

局部灵敏度分析
局部灵敏度分析在参数的基准值处计算模型输出对单个参数的偏导数。最直接的方法是有限差分：给参数一个小扰动 Δp，观察输出变化 Δy，则灵敏度 S ≈ Δy/Δp。


为了比较不同量纲的参数，通常使用弹性系数（归一化灵敏度）：E = (∂y/∂p) × (p₀/y₀)，它衡量参数变化 1% 引起输出变化的百分比。


type ModelFunction = (params: number[]) => number;

interface LocalSensitivity {
  paramIndex: number;
  paramName: string;
  derivative: number;
  elasticity: number;
  rank: number;
}

function localSensitivityAnalysis(
  model: ModelFunction,
  baseParams: number[],
  paramNames: string[],
  perturbation: number = 0.01,
): LocalSensitivity[] {
  const baseOutput = model(baseParams);
  const results: LocalSensitivity[] = [];

  for (let i = 0; i < baseParams.length; i++) {
    const perturbedParams = [...baseParams];
    const delta = baseParams[i] * perturbation;
    perturbedParams[i] += delta;

    const perturbedOutput = model(perturbedParams);
    const derivative = (perturbedOutput - baseOutput) / delta;
    const elasticity = derivative * (baseParams[i] / baseOutput);

    results.push({
      paramIndex: i,
      paramName: paramNames[i],
      derivative,
      elasticity,
      rank: 0,
    });
  }

  results.sort((a, b) => Math.abs(b.elasticity) - Math.abs(a.elasticity));
  results.forEach((r, i) => (r.rank = i + 1));

  return results;
}

function centralDifference(model: ModelFunction, params: number[], index: number, h: number): number {
  const forward = [...params];
  const backward = [...params];
  forward[index] += h;
  backward[index] -= h;
  return (model(forward) - model(backward)) / (2 * h);
}
龙卷风图是展示局部灵敏度分析结果的标准可视化方式。它将每个参数在其不确定性范围内变化导致的输出变化用水平条形表示，按影响大小排序。最长的条形对应最敏感的参数。


interface TornadoBar {
  paramName: string;
  lowValue: number;
  highValue: number;
  lowOutput: number;
  highOutput: number;
  range: number;
}

function generateTornadoData(
  model: ModelFunction,
  baseParams: number[],
  paramNames: string[],
  uncertaintyRanges: { low: number; high: number }[],
): TornadoBar[] {
  const bars: TornadoBar[] = [];

  for (let i = 0; i < baseParams.length; i++) {
    const lowParams = [...baseParams];
    const highParams = [...baseParams];
    lowParams[i] = uncertaintyRanges[i].low;
    highParams[i] = uncertaintyRanges[i].high;

    const lowOutput = model(lowParams);
    const highOutput = model(highParams);

    bars.push({
      paramName: paramNames[i],
      lowValue: uncertaintyRanges[i].low,
      highValue: uncertaintyRanges[i].high,
      lowOutput,
      highOutput,
      range: Math.abs(highOutput - lowOutput),
    });
  }

  return bars.sort((a, b) => b.range - a.range);
}
局部灵敏度分析的主要局限性在于只反映基准点附近的行为。如果模型是高度非线性的，参数在不同区域的灵敏度可能截然不同。此外，局部方法无法捕捉参数间的交互效应——两个参数单独变化影响不大，但同时变化时可能产生显著的协同效应。


Morris 筛选法
Morris 方法是介于局部和全局之间的筛选方法。它通过在参数空间中沿不同路径进行多次单参数扰动，计算每个参数的"基本效应"的均值和标准差。均值反映参数的总体影响，标准差反映参数的非线性程度和交互效应。


interface MorrisResult {
  paramName: string;
  muStar: number;
  sigma: number;
  category: 'linear-important' | 'nonlinear-important' | 'negligible';
}

function morrisScreening(
  model: ModelFunction,
  paramRanges: { min: number; max: number }[],
  paramNames: string[],
  levels: number = 4,
  trajectories: number = 10,
): MorrisResult[] {
  const p = paramRanges.length;
  const delta = 1 / (levels - 1);
  const allEffects: number[][] = Array.from({ length: p }, () => []);

  for (let t = 0; t < trajectories; t++) {
    const baseNorm = paramRanges.map(() => Math.floor(Math.random() * levels) / (levels - 1));
    const base = baseNorm.map((v, i) => paramRanges[i].min + v * (paramRanges[i].max - paramRanges[i].min));

    const order = Array.from({ length: p }, (_, i) => i).sort(() => Math.random() - 0.5);

    let current = [...base];
    let currentOutput = model(current);

    for (const idx of order) {
      const next = [...current];
      const direction = Math.random() > 0.5 ? delta : -delta;
      const newNorm = baseNorm[idx] + direction;

      if (newNorm >= 0 && newNorm <= 1) {
        next[idx] = paramRanges[idx].min + newNorm * (paramRanges[idx].max - paramRanges[idx].min);
        const nextOutput = model(next);
        const ee = (nextOutput - currentOutput) / direction;
        allEffects[idx].push(ee);
        current = next;
        currentOutput = nextOutput;
      }
    }
  }

  return paramNames.map((name, i) => {
    const effects = allEffects[i];
    const muStar = effects.reduce((sum, e) => sum + Math.abs(e), 0) / effects.length;
    const mu = effects.reduce((sum, e) => sum + e, 0) / effects.length;
    const sigma = Math.sqrt(effects.reduce((sum, e) => sum + (e - mu) ** 2, 0) / effects.length);

    let category: MorrisResult['category'];
    if (muStar < 0.1) category = 'negligible';
    else if (sigma / muStar < 0.5) category = 'linear-important';
    else category = 'nonlinear-important';

    return { paramName: name, muStar, sigma, category };
  });
}
Morris 方法的优势在于计算成本远低于全局方法（仅需 r(p+1) 次模型评估，其中 r 为轨迹数，p 为参数数），同时能初步识别非线性和交互效应。它常用作全局灵敏度分析的预筛选步骤，先用 Morris 筛掉不重要参数，再对重要参数进行精确的 Sobol 分析。


Sobol 全局灵敏度分析
Sobol 方法基于方差分解，将模型输出的总方差分解为各参数的独立贡献和交互贡献。一阶 Sobol 指数 Sᵢ 衡量参数 xᵢ 单独对输出方差的贡献比例，总效应指数 STᵢ 则包含了 xᵢ 与其他参数的所有交互贡献。






指数类型
含义
计算公式
解读





一阶 Sᵢ
xᵢ 独立贡献
Var[E(Y|xᵢ)] / Var(Y)
Sᵢ > 0.1 为重要参数




二阶 Sᵢⱼ
xᵢ 与 xⱼ 交互
Sᵢⱼ = Var[E(Y|xᵢ,xⱼ)]/(Var(Y)) - Sᵢ - Sⱼ
衡量两两交互强度




总效应 STᵢ
xᵢ 全部贡献
1 - Var[E(Y|x~ᵢ)] / Var(Y)
STᵢ ≈ 0 可固定该参数





interface SobolResult {
  paramName: string;
  firstOrder: number;
  totalEffect: number;
  interactionStrength: number;
}

function sobolAnalysis(
  model: ModelFunction,
  paramRanges: { min: number; max: number }[],
  paramNames: string[],
  sampleSize: number = 10000,
): SobolResult[] {
  const p = paramRanges.length;

  const matrixA = generateSobolSamples(sampleSize, p, paramRanges);
  const matrixB = generateSobolSamples(sampleSize, p, paramRanges);

  const yA = matrixA.map((row) => model(row));
  const yB = matrixB.map((row) => model(row));

  const totalVar = variance(yA);
  const results: SobolResult[] = [];

  for (let i = 0; i < p; i++) {
    const matrixABi = matrixA.map((rowA, j) => {
      const mixed = [...rowA];
      mixed[i] = matrixB[j][i];
      return mixed;
    });
    const yABi = matrixABi.map((row) => model(row));

    let firstOrderNum = 0;
    let totalEffectNum = 0;
    for (let j = 0; j < sampleSize; j++) {
      firstOrderNum += yB[j] * (yABi[j] - yA[j]);
      totalEffectNum += (yA[j] - yABi[j]) ** 2;
    }

    const firstOrder = firstOrderNum / sampleSize / totalVar;
    const totalEffect = totalEffectNum / (2 * sampleSize) / totalVar;

    results.push({
      paramName: paramNames[i],
      firstOrder: Math.max(0, firstOrder),
      totalEffect: Math.max(0, totalEffect),
      interactionStrength: Math.max(0, totalEffect - firstOrder),
    });
  }

  return results;
}

function generateSobolSamples(n: number, p: number, ranges: { min: number; max: number }[]): number[][] {
  return Array.from({ length: n }, () => ranges.map((r) => r.min + Math.random() * (r.max - r.min)));
}

function variance(values: number[]): number {
  const mean = values.reduce((a, b) => a + b, 0) / values.length;
  return values.reduce((sum, v) => sum + (v - mean) ** 2, 0) / values.length;
}
Sobol 分析的结果可以直接指导参数管理策略：STᵢ ≈ 0 的参数可以固定为任意值（节省校准成本）；Sᵢ 大且交互效应小的参数可以独立校准；交互效应强的参数需要联合优化。


鲁棒设计与参数管理
灵敏度分析的最终目的是指导鲁棒设计——使模型输出对参数不确定性尽可能不敏感。田口方法是鲁棒设计的经典框架，其核心思想是通过调整可控参数来降低噪声参数的影响。


interface RobustDesignResult {
  optimalParams: number[];
  nominalOutput: number;
  outputVariance: number;
  signalToNoise: number;
}

function robustOptimization(
  model: ModelFunction,
  controllableRanges: { min: number; max: number }[],
  noiseParams: { mean: number; std: number }[],
  controlIndices: number[],
  noiseIndices: number[],
  targetOutput: number,
  samples: number = 1000,
): RobustDesignResult {
  let bestResult: RobustDesignResult = {
    optimalParams: [],
    nominalOutput: 0,
    outputVariance: Infinity,
    signalToNoise: -Infinity,
  };

  const gridSize = 10;
  const gridPoints = generateGrid(controllableRanges, gridSize);

  for (const controlValues of gridPoints) {
    const outputs: number[] = [];
    for (let s = 0; s < samples; s++) {
      const params: number[] = [];
      let ci = 0;
      let ni = 0;
      for (let i = 0; i < controlIndices.length + noiseIndices.length; i++) {
        if (controlIndices.includes(i)) {
          params.push(controlValues[ci++]);
        } else {
          const noise = noiseParams[ni];
          params.push(noise.mean + noise.std * gaussianRandom());
          ni++;
        }
      }
      outputs.push(model(params));
    }

    const mean = outputs.reduce((a, b) => a + b, 0) / outputs.length;
    const v = outputs.reduce((sum, o) => sum + (o - mean) ** 2, 0) / outputs.length;
    const snr = -10 * Math.log10((mean - targetOutput) ** 2 + v);

    if (snr > bestResult.signalToNoise) {
      bestResult = {
        optimalParams: controlValues,
        nominalOutput: mean,
        outputVariance: v,
        signalToNoise: snr,
      };
    }
  }

  return bestResult;
}

function generateGrid(ranges: { min: number; max: number }[], size: number): number[][] {
  if (ranges.length === 0) return [[]];
  const rest = generateGrid(ranges.slice(1), size);
  const result: number[][] = [];
  const { min, max } = ranges[0];
  for (let i = 0; i < size; i++) {
    const val = min + (i / (size - 1)) * (max - min);
    for (const r of rest) result.push([val, ...r]);
  }
  return result;
}

function gaussianRandom(): number {
  const u1 = Math.random();
  const u2 = Math.random();
  return Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
}
小结
    

  • 灵敏度分析识别对模型输出影响最大的参数，指导参数校准优先级
    • 

  • 局部方法（偏导数/弹性系数）计算快但只反映基准点附近行为
    • 

  • 龙卷风图是局部灵敏度结果的标准可视化方式
    • 

  • Morris 筛选是高效的预筛选方法，能初步识别参数的非线性和交互效应
    • 

  • Sobol 方法基于方差分解，提供一阶指数（独立贡献）和总效应指数（包含交互）
    • 

  • STᵢ ≈ 0 的参数可以固定，交互效应强的参数需要联合优化
    • 

  • 鲁棒设计的目标是使输出对噪声参数不敏感，通过调整可控参数实现
    • 

  • 实践中常采用"Morris 筛选→Sobol 精确分析→鲁棒优化"的三步策略
