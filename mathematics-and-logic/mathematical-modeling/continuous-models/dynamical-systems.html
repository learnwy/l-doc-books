动力系统

核心概念

动力系统理论研究确定性演化规则驱动的系统的长期行为。其核心问题不是求出精确解轨迹，而是理解系统的定性行为：平衡点在哪里？它们稳定吗？系统是否存在周期振荡？是否可能出现混沌？

动力系统通常表示为 dx/dt = f(x)，其中 x 是状态向量，f 定义了状态空间中每一点的"速度场"。通过分析这个速度场的结构，我们可以不求解方程就理解系统的全局行为。

相平面与平衡点分类
对于二维系统 dx/dt = f(x, y)，dy/dt = g(x, y)，相平面是理解系统行为的核心工具。平衡点是 f(x, y) = 0 且 g(x, y) = 0 的点——系统在此处静止不动。


平衡点的类型由线性化矩阵（雅可比矩阵）的特征值决定。设平衡点为 (x₀, y₀)，雅可比矩阵 J 的特征值 λ₁, λ₂ 的实部和虚部组合确定了附近轨迹的几何形态。


interface Equilibrium {
  x: number;
  y: number;
  jacobian: number[][];
  eigenvalues: { real: number; imag: number }[];
  type: string;
  stability: 'stable' | 'unstable' | 'center';
}

function findEquilibria(
  f: (x: number, y: number) => number,
  g: (x: number, y: number) => number,
  xRange: [number, number],
  yRange: [number, number],
  gridSize: number = 20,
): { x: number; y: number }[] {
  const equilibria: { x: number; y: number }[] = [];
  const dx = (xRange[1] - xRange[0]) / gridSize;
  const dy = (yRange[1] - yRange[0]) / gridSize;

  for (let i = 0; i <= gridSize; i++) {
    for (let j = 0; j <= gridSize; j++) {
      let x = xRange[0] + i * dx;
      let y = yRange[0] + j * dy;

      for (let iter = 0; iter < 50; iter++) {
        const fv = f(x, y);
        const gv = g(x, y);
        if (Math.abs(fv) < 1e-10 && Math.abs(gv) < 1e-10) {
          const isDuplicate = equilibria.some((e) => Math.abs(e.x - x) < 1e-6 && Math.abs(e.y - y) < 1e-6);
          if (!isDuplicate) equilibria.push({ x, y });
          break;
        }

        const h = 1e-8;
        const fx = (f(x + h, y) - f(x - h, y)) / (2 * h);
        const fy = (f(x, y + h) - f(x, y - h)) / (2 * h);
        const gx = (g(x + h, y) - g(x - h, y)) / (2 * h);
        const gy = (g(x, y + h) - g(x, y - h)) / (2 * h);

        const det = fx * gy - fy * gx;
        if (Math.abs(det) < 1e-12) break;

        x -= (gy * fv - fy * gv) / det;
        y -= (fx * gv - gx * fv) / det;
      }
    }
  }

  return equilibria;
}

function classifyEquilibrium(
  f: (x: number, y: number) => number,
  g: (x: number, y: number) => number,
  eq: { x: number; y: number },
): Equilibrium {
  const h = 1e-8;
  const { x, y } = eq;

  const J = [
    [(f(x + h, y) - f(x - h, y)) / (2 * h), (f(x, y + h) - f(x, y - h)) / (2 * h)],
    [(g(x + h, y) - g(x - h, y)) / (2 * h), (g(x, y + h) - g(x, y - h)) / (2 * h)],
  ];

  const trace = J[0][0] + J[1][1];
  const det = J[0][0] * J[1][1] - J[0][1] * J[1][0];
  const disc = trace * trace - 4 * det;

  let eigenvalues: { real: number; imag: number }[];
  if (disc >= 0) {
    eigenvalues = [
      { real: (trace + Math.sqrt(disc)) / 2, imag: 0 },
      { real: (trace - Math.sqrt(disc)) / 2, imag: 0 },
    ];
  } else {
    eigenvalues = [
      { real: trace / 2, imag: Math.sqrt(-disc) / 2 },
      { real: trace / 2, imag: -Math.sqrt(-disc) / 2 },
    ];
  }

  let type: string;
  let stability: Equilibrium['stability'];

  if (det < 0) {
    type = '鞍点';
    stability = 'unstable';
  } else if (disc > 0 && trace < 0) {
    type = '稳定节点';
    stability = 'stable';
  } else if (disc > 0 && trace > 0) {
    type = '不稳定节点';
    stability = 'unstable';
  } else if (disc < 0 && trace < 0) {
    type = '稳定焦点';
    stability = 'stable';
  } else if (disc < 0 && trace > 0) {
    type = '不稳定焦点';
    stability = 'unstable';
  } else {
    type = '中心';
    stability = 'center';
  }

  return { x, y, jacobian: J, eigenvalues, type, stability };
}

Lyapunov 稳定性分析
特征值判据只给出平衡点附近的局部稳定性。Lyapunov 方法提供了一种证明全局稳定性的强大工具。核心思想是构造一个"能量函数" V(x)，如果 V 沿系统轨迹单调递减，则系统必然趋向最低能量状态。


Lyapunov 函数 V(x) 需要满足：V(x₀) = 0（平衡点处为零）、V(x) > 0 对所有 x ≠ x₀（正定）、dV/dt ≤ 0 沿系统轨迹非正（能量不增）。如果 dV/dt < 0（严格递减），则平衡点是渐近稳定的。


interface LyapunovAnalysis {
  isPositiveDefinite: boolean;
  isDerivativeNegative: boolean;
  isAsymptoticallyStable: boolean;
  samplePoints: { x: number[]; V: number; dVdt: number }[];
}

function checkLyapunovCandidate(
  V: (x: number[]) => number,
  system: (x: number[]) => number[],
  equilibrium: number[],
  testRadius: number,
  numSamples: number = 1000,
): LyapunovAnalysis {
  const dim = equilibrium.length;
  const samplePoints: LyapunovAnalysis['samplePoints'] = [];
  let isPositiveDefinite = true;
  let isDerivativeNegative = true;

  for (let i = 0; i < numSamples; i++) {
    const direction = Array.from({ length: dim }, () => gaussRandom());
    const norm = Math.sqrt(direction.reduce((s, d) => s + d * d, 0));
    const r = testRadius * Math.random();
    const x = equilibrium.map((e, j) => e + (r * direction[j]) / norm);

    const v = V(x);
    if (v <= 0 && r > 1e-10) isPositiveDefinite = false;

    const f = system(x);
    const h = 1e-8;
    let dVdt = 0;
    for (let j = 0; j < dim; j++) {
      const xPlus = [...x];
      const xMinus = [...x];
      xPlus[j] += h;
      xMinus[j] -= h;
      const partialV = (V(xPlus) - V(xMinus)) / (2 * h);
      dVdt += partialV * f[j];
    }

    if (dVdt > 1e-10) isDerivativeNegative = false;

    samplePoints.push({ x, V: v, dVdt });
  }

  return {
    isPositiveDefinite,
    isDerivativeNegative,
    isAsymptoticallyStable: isPositiveDefinite && isDerivativeNegative,
    samplePoints,
  };
}

function gaussRandom(): number {
  return Math.sqrt(-2 * Math.log(Math.random())) * Math.cos(2 * Math.PI * Math.random());
}
构造 Lyapunov 函数是一门艺术而非科学——没有通用算法能为任意系统找到 Lyapunov 函数。常见策略包括：物理能量函数（力学系统的动能+势能）、二次形式 V = xᵀPx（线性系统）、和试探函数法。


分岔理论与混沌入门
分岔是指当系统参数平滑变化时，系统的定性行为发生突变。例如，一个稳定的平衡点可能在参数超过临界值时突然变得不稳定，或者分裂为多个平衡点。


function bifurcationDiagram(
  map: (x: number, r: number) => number,
  rRange: [number, number],
  rSteps: number,
  transient: number = 500,
  plotPoints: number = 200,
): { r: number; x: number }[] {
  const points: { r: number; x: number }[] = [];
  const dr = (rRange[1] - rRange[0]) / rSteps;

  for (let i = 0; i <= rSteps; i++) {
    const r = rRange[0] + i * dr;
    let x = 0.5;

    for (let j = 0; j < transient; j++) {
      x = map(x, r);
    }

    for (let j = 0; j < plotPoints; j++) {
      x = map(x, r);
      points.push({ r, x });
    }
  }

  return points;
}

function lyapunovExponent(
  map: (x: number, r: number) => number,
  derivative: (x: number, r: number) => number,
  r: number,
  x0: number = 0.5,
  iterations: number = 10000,
  transient: number = 1000,
): number {
  let x = x0;
  for (let i = 0; i < transient; i++) {
    x = map(x, r);
  }

  let sum = 0;
  for (let i = 0; i < iterations; i++) {
    const df = Math.abs(derivative(x, r));
    if (df > 0) sum += Math.log(df);
    x = map(x, r);
  }

  return sum / iterations;
}
混沌是确定性系统中的看似随机行为，其标志性特征是对初始条件的敏感依赖——两个极其接近的初始状态会在有限时间内产生截然不同的轨迹。Lyapunov 指数量化了这种敏感性：正的最大 Lyapunov 指数表明系统处于混沌状态。

小结
    

  • 动力系统理论关注系统的定性行为（稳定性、周期性、混沌），而非精确解轨迹
    • 

  • 平衡点的类型由雅可比矩阵特征值决定：实部符号决定稳定性，虚部决定是否振荡
    • 

  • 鞍点是不稳定的，节点/焦点的稳定性取决于特征值实部的符号
    • 

  • Lyapunov 方法通过构造"能量函数"证明全局稳定性，但构造函数本身没有通用算法
    • 

  • 分岔是参数变化导致的系统行为质变，常见类型有鞍结、叉形、Hopf 和倍周期分岔
    • 

  • 混沌是确定性系统的复杂行为，以正 Lyapunov 指数为标志
    • 

  • 分岔图是理解参数对系统行为影响的核心可视化工具
