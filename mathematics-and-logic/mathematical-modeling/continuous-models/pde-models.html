偏微分方程模型

核心概念

偏微分方程（PDE）描述多个自变量（通常包括时间和空间）的未知函数的偏导数关系。与 ODE 只关注时间演化不同，PDE 同时捕捉空间分布和时间变化的耦合效应。热传导、波动传播、流体运动等涉及空间扩展的现象都需要 PDE 模型来描述。

PDE 的分类决定了其物理行为和数值求解策略。三种基本类型——椭圆型、抛物型和双曲型——分别对应稳态平衡、扩散传播和波动传播三种物理过程。

热传导方程
一维热传导方程 ∂u/∂t = α·∂²u/∂x² 描述温度 u(x,t) 在空间和时间上的分布变化。热扩散系数 α 决定了热量传播的速度。这个方程的核心物理含义是：温度的时间变化率正比于温度的空间二阶导数——即温度分布越"弯曲"的地方，变化越快。


interface HeatEquationConfig {
  alpha: number;
  L: number;
  nx: number;
  dt: number;
  tEnd: number;
}

function solveHeatEquation1D(
  config: HeatEquationConfig,
  initialCondition: (x: number) => number,
  leftBC: number,
  rightBC: number,
): { x: number[]; t: number[]; u: number[][] } {
  const { alpha, L, nx, dt, tEnd } = config;
  const dx = L / (nx - 1);
  const r = (alpha * dt) / (dx * dx);

  if (r > 0.5) {
    throw new Error(`CFL 条件不满足: r = ${r} > 0.5，请减小 dt 或增大 dx`);
  }

  const x = Array.from({ length: nx }, (_, i) => (i * L) / (nx - 1));
  const nt = Math.ceil(tEnd / dt);
  const t = Array.from({ length: nt + 1 }, (_, i) => i * dt);

  const u: number[][] = [];
  let current = x.map(initialCondition);
  current[0] = leftBC;
  current[nx - 1] = rightBC;
  u.push([...current]);

  for (let n = 0; n < nt; n++) {
    const next = [...current];
    for (let i = 1; i < nx - 1; i++) {
      next[i] = current[i] + r * (current[i + 1] - 2 * current[i] + current[i - 1]);
    }
    next[0] = leftBC;
    next[nx - 1] = rightBC;
    current = next;
    u.push([...current]);
  }

  return { x, t, u };
}
CFL（Courant-Friedrichs-Lewy）条件 r = αΔt/Δx² ≤ 0.5 是显式差分格式稳定的必要条件。违反此条件会导致数值解发散。隐式格式（如 Crank-Nicolson）无条件稳定，但需要在每个时间步求解线性方程组。


function crankNicolson1D(
  config: HeatEquationConfig,
  initialCondition: (x: number) => number,
  leftBC: number,
  rightBC: number,
): { x: number[]; t: number[]; u: number[][] } {
  const { alpha, L, nx, dt, tEnd } = config;
  const dx = L / (nx - 1);
  const r = (alpha * dt) / (2 * dx * dx);

  const x = Array.from({ length: nx }, (_, i) => (i * L) / (nx - 1));
  const nt = Math.ceil(tEnd / dt);
  const t = Array.from({ length: nt + 1 }, (_, i) => i * dt);

  const u: number[][] = [];
  let current = x.map(initialCondition);
  current[0] = leftBC;
  current[nx - 1] = rightBC;
  u.push([...current]);

  for (let n = 0; n < nt; n++) {
    const rhs = new Array(nx).fill(0);
    for (let i = 1; i < nx - 1; i++) {
      rhs[i] = r * current[i - 1] + (1 - 2 * r) * current[i] + r * current[i + 1];
    }
    rhs[1] += r * leftBC;
    rhs[nx - 2] += r * rightBC;

    const a = new Array(nx).fill(-r);
    const b = new Array(nx).fill(1 + 2 * r);
    const c = new Array(nx).fill(-r);

    const interior = thomasAlgorithm(a.slice(1, nx - 1), b.slice(1, nx - 1), c.slice(1, nx - 1), rhs.slice(1, nx - 1));

    const next = [leftBC, ...interior, rightBC];
    current = next;
    u.push([...current]);
  }

  return { x, t, u };
}

function thomasAlgorithm(a: number[], b: number[], c: number[], d: number[]): number[] {
  const n = b.length;
  const cp = new Array(n).fill(0);
  const dp = new Array(n).fill(0);

  cp[0] = c[0] / b[0];
  dp[0] = d[0] / b[0];

  for (let i = 1; i < n; i++) {
    const m = b[i] - a[i] * cp[i - 1];
    cp[i] = c[i] / m;
    dp[i] = (d[i] - a[i] * dp[i - 1]) / m;
  }

  const x = new Array(n).fill(0);
  x[n - 1] = dp[n - 1];
  for (let i = n - 2; i >= 0; i--) {
    x[i] = dp[i] - cp[i] * x[i + 1];
  }

  return x;
}
波动方程
一维波动方程 ∂²u/∂t² = c²·∂²u/∂x² 描述波在介质中的传播。与热传导方程的扩散行为不同，波动方程的解保持波形不变地传播——信息以有限速度 c 传播，不会像热传导那样无限速扩散。


function solveWaveEquation1D(
  c: number,
  L: number,
  nx: number,
  dt: number,
  tEnd: number,
  initialDisplacement: (x: number) => number,
  initialVelocity: (x: number) => number,
): { x: number[]; t: number[]; u: number[][] } {
  const dx = L / (nx - 1);
  const courant = (c * dt) / dx;

  if (courant > 1) {
    throw new Error(`CFL 条件不满足: C = ${courant} > 1`);
  }

  const r2 = courant * courant;
  const x = Array.from({ length: nx }, (_, i) => (i * L) / (nx - 1));
  const nt = Math.ceil(tEnd / dt);
  const t = Array.from({ length: nt + 1 }, (_, i) => i * dt);

  const u: number[][] = [];
  const u0 = x.map(initialDisplacement);
  u0[0] = 0;
  u0[nx - 1] = 0;
  u.push([...u0]);

  const u1 = new Array(nx).fill(0);
  for (let i = 1; i < nx - 1; i++) {
    u1[i] = u0[i] + dt * initialVelocity(x[i]) + (r2 / 2) * (u0[i + 1] - 2 * u0[i] + u0[i - 1]);
  }
  u.push([...u1]);

  let prev = u0;
  let curr = u1;

  for (let n = 1; n < nt; n++) {
    const next = new Array(nx).fill(0);
    for (let i = 1; i < nx - 1; i++) {
      next[i] = 2 * curr[i] - prev[i] + r2 * (curr[i + 1] - 2 * curr[i] + curr[i - 1]);
    }
    prev = curr;
    curr = next;
    u.push([...next]);
  }

  return { x, t, u };
}
波动方程的 CFL 条件是 C = cΔt/Δx ≤ 1，其物理含义是数值信息传播速度不能超过物理波速。这个条件比热传导方程的 CFL 条件更容易理解——如果数值格式的"视野"赶不上波的传播速度，就会丢失信息。

Laplace 方程与边值问题
Laplace 方程 ∇²u = 0 描述没有内部热源的稳态温度分布。它是椭圆型 PDE 的原型，其解完全由边界条件决定——内部没有时间演化，只有空间上的平衡。


function solveLaplace2D(
  nx: number,
  ny: number,
  topBC: number,
  bottomBC: number,
  leftBC: number,
  rightBC: number,
  tolerance: number = 1e-6,
  maxIter: number = 10000,
): { u: number[][]; iterations: number } {
  const u: number[][] = Array.from({ length: ny }, () => new Array(nx).fill(0));

  for (let j = 0; j < ny; j++) {
    u[j][0] = leftBC;
    u[j][nx - 1] = rightBC;
  }
  for (let i = 0; i < nx; i++) {
    u[0][i] = topBC;
    u[ny - 1][i] = bottomBC;
  }

  let iterations = 0;
  for (let iter = 0; iter < maxIter; iter++) {
    let maxDiff = 0;

    for (let j = 1; j < ny - 1; j++) {
      for (let i = 1; i < nx - 1; i++) {
        const oldVal = u[j][i];
        u[j][i] = (u[j - 1][i] + u[j + 1][i] + u[j][i - 1] + u[j][i + 1]) / 4;
        maxDiff = Math.max(maxDiff, Math.abs(u[j][i] - oldVal));
      }
    }

    iterations = iter + 1;
    if (maxDiff < tolerance) break;
  }

  return { u, iterations };
}

function solvePoisson2D(
  nx: number,
  ny: number,
  source: (x: number, y: number) => number,
  Lx: number,
  Ly: number,
  tolerance: number = 1e-6,
  maxIter: number = 10000,
): { u: number[][]; iterations: number } {
  const dx = Lx / (nx - 1);
  const dy = Ly / (ny - 1);
  const u: number[][] = Array.from({ length: ny }, () => new Array(nx).fill(0));

  let iterations = 0;
  for (let iter = 0; iter < maxIter; iter++) {
    let maxDiff = 0;

    for (let j = 1; j < ny - 1; j++) {
      for (let i = 1; i < nx - 1; i++) {
        const x = i * dx;
        const y = j * dy;
        const oldVal = u[j][i];
        u[j][i] =
          (dy * dy * (u[j][i - 1] + u[j][i + 1]) +
            dx * dx * (u[j - 1][i] + u[j + 1][i]) -
            dx * dx * dy * dy * source(x, y)) /
          (2 * (dx * dx + dy * dy));
        maxDiff = Math.max(maxDiff, Math.abs(u[j][i] - oldVal));
      }
    }

    iterations = iter + 1;
    if (maxDiff < tolerance) break;
  }

  return { u, iterations };
}
Laplace 方程的解满足极值原理：内部点的值不会超过边界值的最大值或小于最小值。这个性质对于验证数值解非常有用——如果数值解出现了超过边界极值的内部值，说明计算有误。

小结
    

  • PDE 模型同时描述空间分布和时间演化，是多维连续现象的核心建模工具
    • 

  • 三种基本 PDE 类型（椭圆/抛物/双曲）对应三种物理过程（平衡/扩散/波动）
    • 

  • 热传导方程描述扩散过程，波动方程描述波传播，Laplace 方程描述稳态平衡
    • 

  • CFL 条件是显式差分格式稳定性的核心约束，由数值信息传播速度与物理速度的关系决定
    • 

  • Crank-Nicolson 格式兼具二阶精度和无条件稳定性，是实际应用的推荐选择
    • 

  • 有限差分法将连续 PDE 离散化为代数方程组，是最直观的 PDE 数值方法
    • 

  • 极值原理是验证椭圆型 PDE 数值解正确性的重要工具
    •
