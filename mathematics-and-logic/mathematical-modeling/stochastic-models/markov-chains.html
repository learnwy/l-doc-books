马尔可夫链

核心概念

马尔可夫链是一类特殊的随机过程，具有"无记忆性"——系统的未来状态只取决于当前状态，与过去的历史无关。这个性质称为马尔可夫性。尽管假设简单，马尔可夫链能建模大量现实系统：天气预报、股票价格变动、网页浏览行为、基因序列演化、排队系统等。

马尔可夫链由三个要素定义：状态空间 S（系统可能处于的所有状态的集合）、转移概率矩阵 P（P(i,j) 表示从状态 i 转移到状态 j 的概率）、初始分布 π₀（系统起始时各状态的概率分布）。

转移矩阵与状态演化
转移概率矩阵 P 是马尔可夫链的核心数学对象。P 的第 (i,j) 个元素表示从状态 i 转到状态 j 的概率，每行之和为 1。n 步转移概率等于矩阵 P 的 n 次幂 Pⁿ 的对应元素。


interface MarkovChain {
  states: string[];
  transitionMatrix: number[][];
  initialDistribution: number[];
}

function createMarkovChain(states: string[], transitions: number[][]): MarkovChain {
  const n = states.length;
  for (let i = 0; i < n; i++) {
    const rowSum = transitions[i].reduce((a, b) => a + b, 0);
    if (Math.abs(rowSum - 1) > 1e-10) {
      throw new Error(`第 ${i} 行概率之和 ${rowSum} ≠ 1`);
    }
  }

  return {
    states,
    transitionMatrix: transitions,
    initialDistribution: new Array(n).fill(1 / n),
  };
}

function nStepTransition(P: number[][], n: number): number[][] {
  const size = P.length;
  let result = P.map((row) => [...row]);

  for (let step = 1; step < n; step++) {
    const next: number[][] = Array.from({ length: size }, () => new Array(size).fill(0));
    for (let i = 0; i < size; i++) {
      for (let j = 0; j < size; j++) {
        for (let k = 0; k < size; k++) {
          next[i][j] += result[i][k] * P[k][j];
        }
      }
    }
    result = next;
  }

  return result;
}

function evolveDistribution(chain: MarkovChain, steps: number): number[][] {
  const distributions: number[][] = [chain.initialDistribution];
  let current = [...chain.initialDistribution];
  const n = chain.states.length;

  for (let t = 0; t < steps; t++) {
    const next = new Array(n).fill(0);
    for (let j = 0; j < n; j++) {
      for (let i = 0; i < n; i++) {
        next[j] += current[i] * chain.transitionMatrix[i][j];
      }
    }
    current = next;
    distributions.push([...current]);
  }

  return distributions;
}

function simulate(chain: MarkovChain, startState: number, steps: number): number[] {
  const path: number[] = [startState];
  let current = startState;

  for (let t = 0; t < steps; t++) {
    const r = Math.random();
    let cumulative = 0;
    for (let j = 0; j < chain.states.length; j++) {
      cumulative += chain.transitionMatrix[current][j];
      if (r <= cumulative) {
        current = j;
        break;
      }
    }
    path.push(current);
  }

  return path;
}
平稳分布与收敛性
平稳分布 π 是满足 πP = π 的概率分布——系统到达平稳分布后，状态分布不再随时间变化。对于遍历性马尔可夫链（不可约且非周期），无论初始分布如何，系统都会收敛到唯一的平稳分布。


function stationaryDistribution(P: number[][], tolerance: number = 1e-10, maxIter: number = 10000): number[] {
  const n = P.length;
  let pi = new Array(n).fill(1 / n);

  for (let iter = 0; iter < maxIter; iter++) {
    const next = new Array(n).fill(0);
    for (let j = 0; j < n; j++) {
      for (let i = 0; i < n; i++) {
        next[j] += pi[i] * P[i][j];
      }
    }

    const diff = Math.max(...pi.map((v, i) => Math.abs(v - next[i])));
    pi = next;

    if (diff < tolerance) return pi;
  }

  return pi;
}

function stationaryByEigen(P: number[][]): number[] {
  const n = P.length;
  const A: number[][] = Array.from({ length: n }, (_, i) =>
    P.map((row) => row[i]).map((v, j) => (i === j ? v - 1 : v)),
  );

  A[n - 1] = new Array(n).fill(1);
  const b = new Array(n).fill(0);
  b[n - 1] = 1;

  for (let col = 0; col < n; col++) {
    let maxRow = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(A[row][col]) > Math.abs(A[maxRow][col])) maxRow = row;
    }
    [A[col], A[maxRow]] = [A[maxRow], A[col]];
    [b[col], b[maxRow]] = [b[maxRow], b[col]];

    for (let row = col + 1; row < n; row++) {
      const factor = A[row][col] / A[col][col];
      for (let j = col; j < n; j++) A[row][j] -= factor * A[col][j];
      b[row] -= factor * b[col];
    }
  }

  const pi = new Array(n).fill(0);
  for (let i = n - 1; i >= 0; i--) {
    pi[i] = b[i];
    for (let j = i + 1; j < n; j++) pi[i] -= A[i][j] * pi[j];
    pi[i] /= A[i][i];
  }

  return pi;
}
收敛速度由转移矩阵第二大特征值的模 |λ₂| 决定。|λ₂| 越小，收敛越快。这个量称为谱隙（spectral gap），在 MCMC 方法的效率分析中至关重要。


吸收马尔可夫链
吸收状态是一旦进入就无法离开的状态。吸收马尔可夫链的核心问题是：从每个非吸收状态出发，被各个吸收状态吸收的概率是多少？到达吸收状态的平均时间是多少？


interface AbsorbingChainAnalysis {
  fundamentalMatrix: number[][];
  absorptionProbabilities: number[][];
  expectedSteps: number[];
}

function analyzeAbsorbingChain(P: number[][], absorbingStates: number[]): AbsorbingChainAnalysis {
  const n = P.length;
  const transientStates = Array.from({ length: n }, (_, i) => i).filter((i) => !absorbingStates.includes(i));
  const t = transientStates.length;
  const a = absorbingStates.length;

  const Q: number[][] = Array.from({ length: t }, (_, i) => transientStates.map((j) => P[transientStates[i]][j]));

  const R: number[][] = Array.from({ length: t }, (_, i) => absorbingStates.map((j) => P[transientStates[i]][j]));

  const IminusQ: number[][] = Array.from({ length: t }, (_, i) => Q[i].map((q, j) => (i === j ? 1 - q : -q)));

  const N = invertMatrix(IminusQ);

  const B: number[][] = Array.from({ length: t }, (_, i) => {
    const row = new Array(a).fill(0);
    for (let j = 0; j < a; j++) {
      for (let k = 0; k < t; k++) {
        row[j] += N[i][k] * R[k][j];
      }
    }
    return row;
  });

  const expectedSteps = N.map((row) => row.reduce((sum, v) => sum + v, 0));

  return {
    fundamentalMatrix: N,
    absorptionProbabilities: B,
    expectedSteps,
  };
}

function invertMatrix(A: number[][]): number[][] {
  const n = A.length;
  const aug = A.map((row, i) => [...row, ...Array.from({ length: n }, (_, j) => (i === j ? 1 : 0))]);

  for (let col = 0; col < n; col++) {
    let maxRow = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) maxRow = row;
    }
    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];

    const pivot = aug[col][col];
    for (let j = 0; j < 2 * n; j++) aug[col][j] /= pivot;

    for (let row = 0; row < n; row++) {
      if (row === col) continue;
      const factor = aug[row][col];
      for (let j = 0; j < 2 * n; j++) aug[row][j] -= factor * aug[col][j];
    }
  }

  return aug.map((row) => row.slice(n));
}
PageRank 算法
Google 的 PageRank 算法是马尔可夫链的经典应用。将网页视为状态，超链接视为转移，PageRank 值就是对应马尔可夫链的平稳分布。引入阻尼因子 d（通常 0.85）处理悬挂节点和不连通问题。


function pageRank(
  adjacency: number[][],
  damping: number = 0.85,
  tolerance: number = 1e-8,
  maxIter: number = 100,
): number[] {
  const n = adjacency.length;
  const outDegree = adjacency.map((row) => row.reduce((a, b) => a + b, 0));

  const M: number[][] = Array.from({ length: n }, () => new Array(n).fill(0));

  for (let j = 0; j < n; j++) {
    if (outDegree[j] > 0) {
      for (let i = 0; i < n; i++) {
        M[i][j] = adjacency[j][i] / outDegree[j];
      }
    } else {
      for (let i = 0; i < n; i++) {
        M[i][j] = 1 / n;
      }
    }
  }

  let rank = new Array(n).fill(1 / n);

  for (let iter = 0; iter < maxIter; iter++) {
    const next = new Array(n).fill((1 - damping) / n);
    for (let i = 0; i < n; i++) {
      for (let j = 0; j < n; j++) {
        next[i] += damping * M[i][j] * rank[j];
      }
    }

    const diff = Math.max(...rank.map((v, i) => Math.abs(v - next[i])));
    rank = next;
    if (diff < tolerance) break;
  }

  return rank;
}

小结
    

  • 马尔可夫链的核心假设是无记忆性——未来只取决于现在，不依赖过去
    • 

  • 转移矩阵 P 完全描述了系统的随机演化规律，Pⁿ 给出 n 步转移概率
    • 

  • 遍历性马尔可夫链（不可约+非周期）保证收敛到唯一的平稳分布
    • 

  • 平稳分布可通过迭代法或求解线性方程组 πP = π 获得
    • 

  • 吸收马尔可夫链的基本矩阵 N = (I-Q)⁻¹ 给出到达吸收态的期望步数
    • 

  • PageRank 是马尔可夫链平稳分布的实际应用，阻尼因子保证遍历性
    • 

  • 谱隙（第二大特征值）决定了收敛速度，是 MCMC 效率的关键指标
    • 

  • 马尔可夫链是蒙特卡罗方法（MCMC）和强化学习（MDP）的数学基础
