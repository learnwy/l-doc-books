基于代理的建模

核心概念

基于代理的建模（Agent-Based Modeling, ABM）是一种自底向上的建模方法，通过定义大量自主代理（Agent）的个体行为规则和交互机制，观察系统层面的涌现现象。与方程式模型从宏观描述出发不同，ABM 从微观个体出发，让宏观行为自然涌现——这使得 ABM 特别适合建模异质性强、交互复杂、具有空间结构的系统。

ABM 的核心哲学是"简单规则产生复杂行为"——单个代理的行为规则可能很简单，但大量代理的交互可以产生不可预见的集体现象（如鸟群的队形变换、交通拥堵的自发形成、市场泡沫的涌现）。

代理设计与属性系统
代理（Agent）是 ABM 的基本构建块。每个代理具有内部状态（属性）、感知能力（获取环境和邻居信息）、决策逻辑（根据感知选择行为）和执行能力（改变自身状态或环境）。


interface Agent {
  id: number;
  x: number;
  y: number;
  state: Record<string, number>;
  type: string;
}

interface Environment {
  width: number;
  height: number;
  agents: Agent[];
  grid: (Agent | null)[][];
  time: number;
  globalState: Record<string, number>;
}

function createEnvironment(width: number, height: number): Environment {
  return {
    width,
    height,
    agents: [],
    grid: Array.from({ length: height }, () => new Array(width).fill(null)),
    time: 0,
    globalState: {},
  };
}

function addAgent(env: Environment, type: string, x: number, y: number, state: Record<string, number>): Agent {
  const agent: Agent = {
    id: env.agents.length,
    x,
    y,
    state: { ...state },
    type,
  };
  env.agents.push(agent);
  env.grid[y][x] = agent;
  return agent;
}

function getNeighbors(env: Environment, agent: Agent, radius: number): Agent[] {
  const neighbors: Agent[] = [];
  for (let dy = -radius; dy <= radius; dy++) {
    for (let dx = -radius; dx <= radius; dx++) {
      if (dx === 0 && dy === 0) continue;
      const nx = (agent.x + dx + env.width) % env.width;
      const ny = (agent.y + dy + env.height) % env.height;
      const neighbor = env.grid[ny][nx];
      if (neighbor) neighbors.push(neighbor);
    }
  }
  return neighbors;
}

function moveAgent(env: Environment, agent: Agent, newX: number, newY: number): void {
  env.grid[agent.y][agent.x] = null;
  agent.x = (newX + env.width) % env.width;
  agent.y = (newY + env.height) % env.height;
  env.grid[agent.y][agent.x] = agent;
}
Boids 群体行为模型
Craig Reynolds 的 Boids 模型是 ABM 的经典范例，用三条简单规则模拟鸟群飞行行为：分离（避免碰撞邻居）、对齐（与邻居保持相同方向）、聚合（向邻居中心移动）。


interface Boid {
  x: number;
  y: number;
  vx: number;
  vy: number;
}

interface BoidParams {
  separationWeight: number;
  alignmentWeight: number;
  cohesionWeight: number;
  perceptionRadius: number;
  maxSpeed: number;
  separationRadius: number;
}

function updateBoids(boids: Boid[], params: BoidParams, width: number, height: number): void {
  const forces = boids.map((boid) => {
    const neighbors = boids.filter((other) => {
      if (other === boid) return false;
      const dx = other.x - boid.x;
      const dy = other.y - boid.y;
      return Math.sqrt(dx * dx + dy * dy) < params.perceptionRadius;
    });

    if (neighbors.length === 0) return { fx: 0, fy: 0 };

    let sepX = 0,
      sepY = 0;
    let aliX = 0,
      aliY = 0;
    let cohX = 0,
      cohY = 0;

    for (const other of neighbors) {
      const dx = boid.x - other.x;
      const dy = boid.y - other.y;
      const dist = Math.sqrt(dx * dx + dy * dy);

      if (dist < params.separationRadius && dist > 0) {
        sepX += dx / dist;
        sepY += dy / dist;
      }

      aliX += other.vx;
      aliY += other.vy;
      cohX += other.x;
      cohY += other.y;
    }

    aliX /= neighbors.length;
    aliY /= neighbors.length;
    cohX = cohX / neighbors.length - boid.x;
    cohY = cohY / neighbors.length - boid.y;

    return {
      fx: sepX * params.separationWeight + (aliX - boid.vx) * params.alignmentWeight + cohX * params.cohesionWeight,
      fy: sepY * params.separationWeight + (aliY - boid.vy) * params.alignmentWeight + cohY * params.cohesionWeight,
    };
  });

  boids.forEach((boid, i) => {
    boid.vx += forces[i].fx;
    boid.vy += forces[i].fy;

    const speed = Math.sqrt(boid.vx ** 2 + boid.vy ** 2);
    if (speed > params.maxSpeed) {
      boid.vx = (boid.vx / speed) * params.maxSpeed;
      boid.vy = (boid.vy / speed) * params.maxSpeed;
    }

    boid.x = (boid.x + boid.vx + width) % width;
    boid.y = (boid.y + boid.vy + height) % height;
  });
}

Schelling 分离模型
Thomas Schelling 的分离模型是社会科学中最著名的 ABM 之一。它证明了即使个体只有轻微的同类偏好（例如希望至少 30% 的邻居是同类），群体层面也会产生显著的空间隔离现象——温和的个体偏好导致极端的集体后果。


interface SchellingConfig {
  gridSize: number;
  occupancyRate: number;
  numTypes: number;
  toleranceThreshold: number;
  maxIterations: number;
}

function schellingModel(config: SchellingConfig): {
  initialGrid: number[][];
  finalGrid: number[][];
  iterations: number;
  satisfactionHistory: number[];
} {
  const { gridSize, occupancyRate, numTypes, toleranceThreshold, maxIterations } = config;
  const grid: number[][] = Array.from({ length: gridSize }, () =>
    Array.from({ length: gridSize }, () => {
      if (Math.random() > occupancyRate) return 0;
      return Math.floor(Math.random() * numTypes) + 1;
    }),
  );

  const initialGrid = grid.map((row) => [...row]);
  const satisfactionHistory: number[] = [];

  function isSatisfied(x: number, y: number): boolean {
    const type = grid[y][x];
    if (type === 0) return true;

    let same = 0;
    let total = 0;
    for (let dy = -1; dy <= 1; dy++) {
      for (let dx = -1; dx <= 1; dx++) {
        if (dx === 0 && dy === 0) continue;
        const nx = x + dx;
        const ny = y + dy;
        if (nx >= 0 && nx < gridSize && ny >= 0 && ny < gridSize) {
          if (grid[ny][nx] !== 0) {
            total++;
            if (grid[ny][nx] === type) same++;
          }
        }
      }
    }

    return total === 0 || same / total >= toleranceThreshold;
  }

  let iterations = 0;
  for (let iter = 0; iter < maxIterations; iter++) {
    const unsatisfied: { x: number; y: number }[] = [];
    const empty: { x: number; y: number }[] = [];
    let satisfied = 0;
    let total = 0;

    for (let y = 0; y < gridSize; y++) {
      for (let x = 0; x < gridSize; x++) {
        if (grid[y][x] === 0) {
          empty.push({ x, y });
        } else {
          total++;
          if (isSatisfied(x, y)) satisfied++;
          else unsatisfied.push({ x, y });
        }
      }
    }

    satisfactionHistory.push(total > 0 ? satisfied / total : 1);

    if (unsatisfied.length === 0) {
      iterations = iter;
      break;
    }

    for (let i = unsatisfied.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [unsatisfied[i], unsatisfied[j]] = [unsatisfied[j], unsatisfied[i]];
    }

    for (const agent of unsatisfied) {
      if (empty.length === 0) break;
      const idx = Math.floor(Math.random() * empty.length);
      const target = empty[idx];
      grid[target.y][target.x] = grid[agent.y][agent.x];
      grid[agent.y][agent.x] = 0;
      empty[idx] = agent;
    }

    iterations = iter;
  }

  return {
    initialGrid,
    finalGrid: grid.map((row) => [...row]),
    iterations,
    satisfactionHistory,
  };
}
TypeScript 仿真框架
构建一个通用的 ABM 仿真框架需要解决调度、空间索引和数据收集三个核心问题。


type AgentBehavior = (agent: Agent, env: Environment) => void;

interface ABMConfig {
  width: number;
  height: number;
  agentTypes: {
    type: string;
    count: number;
    initialState: Record<string, number>;
    behavior: AgentBehavior;
  }[];
  steps: number;
  collectInterval: number;
}

interface SimulationResult {
  snapshots: { time: number; agents: Agent[] }[];
  metrics: Record<string, number[]>;
}

function runSimulation(config: ABMConfig): SimulationResult {
  const env = createEnvironment(config.width, config.height);
  const snapshots: SimulationResult['snapshots'] = [];
  const metrics: Record<string, number[]> = {
    totalAgents: [],
    avgState: [],
  };

  for (const agentType of config.agentTypes) {
    for (let i = 0; i < agentType.count; i++) {
      let x: number, y: number;
      do {
        x = Math.floor(Math.random() * config.width);
        y = Math.floor(Math.random() * config.height);
      } while (env.grid[y][x] !== null);

      addAgent(env, agentType.type, x, y, { ...agentType.initialState });
    }
  }

  for (let step = 0; step < config.steps; step++) {
    env.time = step;

    const shuffled = [...env.agents].sort(() => Math.random() - 0.5);
    for (const agent of shuffled) {
      const agentConfig = config.agentTypes.find((t) => t.type === agent.type);
      agentConfig?.behavior(agent, env);
    }

    if (step % config.collectInterval === 0) {
      snapshots.push({
        time: step,
        agents: env.agents.map((a) => ({
          ...a,
          state: { ...a.state },
        })),
      });

      metrics.totalAgents.push(env.agents.length);
      const avg = env.agents.reduce((sum, a) => sum + (a.state.energy ?? 0), 0) / env.agents.length;
      metrics.avgState.push(avg);
    }
  }

  return { snapshots, metrics };
}

小结
    

  • ABM 从微观个体行为出发，让宏观模式自然涌现，是自底向上建模的代表方法
    • 

  • 代理具有状态、感知、决策和执行四个核心能力
    • 

  • Boids 模型用分离/对齐/聚合三条规则产生逼真的群体飞行行为
    • 

  • Schelling 模型证明了温和的个体偏好可以导致极端的集体隔离——涌现的非直觉性
    • 

  • ABM 的核心优势是支持异质性和空间结构，这是方程式模型难以处理的
    • 

  • 仿真框架的三个核心组件：调度器（代理执行顺序）、空间索引（邻居查找）、数据收集器
    • 

  • ABM 广泛应用于流行病传播、城市规划、市场模拟、生态系统和社会科学研究
    • 

  • ABM 的主要挑战是参数校准困难和结果的可重复性需要严格的随机种子管理
    •
