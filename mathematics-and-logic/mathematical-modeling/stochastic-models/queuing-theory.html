排队论

核心概念

排队论（Queueing Theory）研究等待线系统的数学模型，分析顾客到达、等待、服务和离开的随机过程。排队现象无处不在：超市收银台、客服呼叫中心、网络数据包传输、CPU 任务调度、微服务请求处理。排队论为系统容量规划和性能优化提供了严格的数学基础。

排队系统用 Kendall 记号 A/B/c/K/N/D 描述，其中 A=到达过程分布，B=服务时间分布，c=服务台数量，K=系统容量，N=顾客源总数，D=排队规则。常省略后三个参数（默认 K=∞, N=∞, D=FCFS）。

M/M/1 排队模型
M/M/1 是最基本的排队模型：泊松到达（参数 λ）、指数服务（参数 μ）、单服务台。系统稳定条件是流量强度 ρ = λ/μ < 1，即到达率必须小于服务率。


interface MM1Metrics {
  rho: number;
  L: number;
  Lq: number;
  W: number;
  Wq: number;
  P0: number;
  Pn: (n: number) => number;
  isStable: boolean;
}

function mm1Analysis(lambda: number, mu: number): MM1Metrics {
  const rho = lambda / mu;
  const isStable = rho < 1;

  if (!isStable) {
    return {
      rho,
      L: Infinity,
      Lq: Infinity,
      W: Infinity,
      Wq: Infinity,
      P0: 0,
      Pn: () => 0,
      isStable: false,
    };
  }

  const L = rho / (1 - rho);
  const Lq = (rho * rho) / (1 - rho);
  const W = 1 / (mu - lambda);
  const Wq = rho / (mu - lambda);
  const P0 = 1 - rho;

  return {
    rho,
    L,
    Lq,
    W,
    Wq,
    P0,
    Pn: (n: number) => (1 - rho) * Math.pow(rho, n),
    isStable,
  };
}

function mm1Simulation(
  lambda: number,
  mu: number,
  numCustomers: number,
): {
  avgWait: number;
  avgSystem: number;
  avgQueueLength: number;
  serverUtilization: number;
} {
  const arrivals: number[] = [];
  const serviceStarts: number[] = [];
  const departures: number[] = [];

  let clock = 0;
  for (let i = 0; i < numCustomers; i++) {
    const interArrival = -Math.log(1 - Math.random()) / lambda;
    clock += interArrival;
    arrivals.push(clock);
  }

  for (let i = 0; i < numCustomers; i++) {
    const serviceTime = -Math.log(1 - Math.random()) / mu;
    const start = i === 0 ? arrivals[0] : Math.max(arrivals[i], departures[i - 1]);
    serviceStarts.push(start);
    departures.push(start + serviceTime);
  }

  const waits = serviceStarts.map((s, i) => s - arrivals[i]);
  const systemTimes = departures.map((d, i) => d - arrivals[i]);

  const totalTime = departures[numCustomers - 1] - arrivals[0];
  const busyTime = departures.reduce((sum, d, i) => sum + (d - serviceStarts[i]), 0);

  return {
    avgWait: waits.reduce((a, b) => a + b, 0) / numCustomers,
    avgSystem: systemTimes.reduce((a, b) => a + b, 0) / numCustomers,
    avgQueueLength: (waits.reduce((a, b) => a + b, 0) / totalTime) * lambda,
    serverUtilization: busyTime / totalTime,
  };
}
M/M/1 的核心公式揭示了排队系统的非线性特性：当 ρ 接近 1 时，队列长度和等待时间急剧增长。这解释了为什么系统在 80% 负载和 95% 负载下的体验差距如此巨大。

M/M/c 多服务台模型
M/M/c 模型扩展了 M/M/1，允许 c 个并行服务台同时工作。稳定条件变为 ρ = λ/(cμ) < 1。多服务台可以显著降低等待时间，但投入产出比递减——从 1 台增加到 2 台的效果远大于从 10 台增加到 11 台。


interface MMcMetrics {
  rho: number;
  c: number;
  P0: number;
  Lq: number;
  L: number;
  Wq: number;
  W: number;
  Pc: number;
}

function mmcAnalysis(lambda: number, mu: number, c: number): MMcMetrics {
  const a = lambda / mu;
  const rho = a / c;

  if (rho >= 1) {
    throw new Error(`系统不稳定: ρ = ${rho.toFixed(3)} >= 1`);
  }

  let sum = 0;
  for (let k = 0; k < c; k++) {
    sum += Math.pow(a, k) / factorial(k);
  }
  const lastTerm = Math.pow(a, c) / (factorial(c) * (1 - rho));
  const P0 = 1 / (sum + lastTerm);

  const Pc = (Math.pow(a, c) / factorial(c)) * P0;
  const Lq = (Pc * rho) / Math.pow(1 - rho, 2);
  const Wq = Lq / lambda;
  const W = Wq + 1 / mu;
  const L = lambda * W;

  return { rho, c, P0, Lq, L, Wq, W, Pc };
}

function factorial(n: number): number {
  if (n <= 1) return 1;
  let result = 1;
  for (let i = 2; i <= n; i++) result *= i;
  return result;
}

function optimalServerCount(
  lambda: number,
  mu: number,
  maxWaitTime: number,
  serverCost: number,
  waitCostPerUnit: number,
): { optimalC: number; totalCost: number; metrics: MMcMetrics } {
  const minC = Math.ceil(lambda / mu);
  let bestC = minC;
  let bestCost = Infinity;
  let bestMetrics: MMcMetrics | null = null;

  for (let c = minC; c <= minC + 20; c++) {
    try {
      const metrics = mmcAnalysis(lambda, mu, c);
      const totalCost = c * serverCost + lambda * metrics.Wq * waitCostPerUnit;

      if (totalCost < bestCost) {
        bestCost = totalCost;
        bestC = c;
        bestMetrics = metrics;
      }
    } catch {
      continue;
    }
  }

  return {
    optimalC: bestC,
    totalCost: bestCost,
    metrics: bestMetrics!,
  };
}
Little's Law
Little's 定律 L = λW 是排队论中最优雅也最实用的结果——系统中的平均顾客数等于到达率乘以平均逗留时间。这个公式不依赖于到达分布、服务分布或排队规则，是一个普适的守恒律。


interface LittleLawCheck {
  observedL: number;
  observedLambda: number;
  observedW: number;
  expectedL: number;
  isConsistent: boolean;
}

function verifyLittlesLaw(arrivals: number[], departures: number[], observationEnd: number): LittleLawCheck {
  const n = arrivals.length;

  const lambda = n / observationEnd;
  const W = departures.reduce((sum, d, i) => sum + (d - arrivals[i]), 0) / n;

  let areaUnderCurve = 0;
  const events: { time: number; delta: number }[] = [];
  for (let i = 0; i < n; i++) {
    events.push({ time: arrivals[i], delta: 1 });
    events.push({ time: departures[i], delta: -1 });
  }
  events.sort((a, b) => a.time - b.time);

  let currentCount = 0;
  let lastTime = 0;
  for (const event of events) {
    areaUnderCurve += currentCount * (event.time - lastTime);
    currentCount += event.delta;
    lastTime = event.time;
  }
  areaUnderCurve += currentCount * (observationEnd - lastTime);

  const observedL = areaUnderCurve / observationEnd;
  const expectedL = lambda * W;

  return {
    observedL,
    observedLambda: lambda,
    observedW: W,
    expectedL,
    isConsistent: Math.abs(observedL - expectedL) / expectedL < 0.1,
  };
}
Little's 定律在系统设计中的应用极其广泛。例如：如果一个微服务的平均响应时间 W = 200ms，希望系统同时处理 L = 100 个请求，则需要支持的吞吐量 λ = L/W = 500 req/s。

排队论在系统设计中的应用
排队论为微服务架构、数据库连接池、消息队列和负载均衡器的设计提供了理论指导。核心洞察是：系统性能在高负载下急剧恶化，预留足够的空闲容量是保障服务质量的关键。


interface SystemDesign {
  servers: number;
  avgResponseTime: number;
  p95ResponseTime: number;
  throughput: number;
  utilizationTarget: number;
}

function designSystem(
  peakRequestRate: number,
  avgProcessingTime: number,
  slaResponseTime: number,
  slaPercentile: number = 0.95,
): SystemDesign {
  const mu = 1 / avgProcessingTime;
  const lambda = peakRequestRate;

  let c = Math.ceil(lambda / mu);
  let metrics: MMcMetrics;

  while (c < 100) {
    try {
      metrics = mmcAnalysis(lambda, mu, c);
      const p95 = avgProcessingTime + metrics.Wq * 3;
      if (p95 <= slaResponseTime && metrics.rho <= 0.75) {
        return {
          servers: c,
          avgResponseTime: metrics.W,
          p95ResponseTime: p95,
          throughput: lambda,
          utilizationTarget: metrics.rho,
        };
      }
    } catch {
      // pass
    }
    c++;
  }

  return {
    servers: c,
    avgResponseTime: avgProcessingTime,
    p95ResponseTime: slaResponseTime,
    throughput: lambda,
    utilizationTarget: lambda / (c * mu),
  };
}

function connectionPoolSize(concurrentRequests: number, avgQueryTime: number, targetUtilization: number = 0.7): number {
  return Math.ceil((concurrentRequests * avgQueryTime) / targetUtilization);
}

function messageQueueCapacity(
  producerRate: number,
  consumerRate: number,
  maxAcceptableDelay: number,
): { bufferSize: number; consumers: number } {
  const minConsumers = Math.ceil(producerRate / consumerRate);
  const consumers = Math.ceil(minConsumers / 0.75);
  const effectiveRate = consumers * consumerRate;
  const rho = producerRate / effectiveRate;
  const avgQueueLength = rho / (1 - rho);
  const bufferSize = Math.ceil(avgQueueLength * 3);

  return { bufferSize, consumers };
}
小结
    

  • 排队论为等待线系统提供数学分析框架，Kendall 记号系统化描述队列特征
    • 

  • M/M/1 模型揭示了排队系统的非线性特性——负载从 80% 到 95% 时性能急剧恶化
    • 

  • M/M/c 多服务台可显著降低等待时间，但投入产出比递减
    • 

  • Little's 定律 L = λW 是普适的守恒律，不依赖于具体分布假设
    • 

  • 系统设计应保持利用率在 70-80% 以下，为突发流量留有余量
    • 

  • 连接池大小、消息队列容量和服务实例数都可通过排队论公式指导设计
    • 

  • 排队论的核心实践价值：用数学量化"多少资源够用"，避免过度配置或配置不足
