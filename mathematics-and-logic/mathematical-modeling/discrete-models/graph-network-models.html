图与网络模型

核心概念

图论是离散数学中研究对象之间关系的核心工具。一个图 G = (V, E) 由顶点集 V 和边集 E 组成，边连接顶点对，表示它们之间的某种关系。网络模型在图的基础上为边赋予权重（距离、容量、成本），使其能建模物流、通信、社交等现实系统中的优化问题。

图模型的强大之处在于其抽象能力——无论是社交网络中的好友关系、城市间的公路连接、还是程序中的函数调用依赖，都可以统一用图来表示和分析。

图的表示与基本遍历
图的两种主要表示方式是邻接矩阵和邻接表。邻接矩阵适合稠密图（边数接近 V²），邻接表适合稀疏图（边数远小于 V²）。选择哪种表示直接影响算法的时空效率。


interface Graph {
  vertices: number;
  adjacencyList: Map<number, { to: number; weight: number }[]>;
}

function createGraph(vertices: number): Graph {
  return {
    vertices,
    adjacencyList: new Map(Array.from({ length: vertices }, (_, i) => [i, []])),
  };
}

function addEdge(graph: Graph, from: number, to: number, weight: number = 1, directed: boolean = false): void {
  graph.adjacencyList.get(from)?.push({ to, weight });
  if (!directed) {
    graph.adjacencyList.get(to)?.push({ to: from, weight });
  }
}

function bfs(graph: Graph, start: number): number[] {
  const visited = new Set<number>();
  const order: number[] = [];
  const queue: number[] = [start];
  visited.add(start);

  while (queue.length > 0) {
    const node = queue.shift()!;
    order.push(node);

    for (const neighbor of graph.adjacencyList.get(node) ?? []) {
      if (!visited.has(neighbor.to)) {
        visited.add(neighbor.to);
        queue.push(neighbor.to);
      }
    }
  }

  return order;
}

function dfs(graph: Graph, start: number): number[] {
  const visited = new Set<number>();
  const order: number[] = [];

  function visit(node: number): void {
    visited.add(node);
    order.push(node);
    for (const neighbor of graph.adjacencyList.get(node) ?? []) {
      if (!visited.has(neighbor.to)) {
        visit(neighbor.to);
      }
    }
  }

  visit(start);
  return order;
}
BFS（广度优先搜索）按层级遍历，天然求解无权图的最短路径。DFS（深度优先搜索）按深度遍历，适合检测环、拓扑排序、连通分量分析。两者时间复杂度都是 O(V + E)。


最短路径算法
最短路径问题是图论中最经典的优化问题。Dijkstra 算法解决单源最短路径（非负权），Floyd-Warshall 解决全源最短路径，Bellman-Ford 处理负权边。


function dijkstra(graph: Graph, source: number): { distances: number[]; predecessors: (number | null)[] } {
  const n = graph.vertices;
  const distances = new Array(n).fill(Infinity);
  const predecessors: (number | null)[] = new Array(n).fill(null);
  const visited = new Set<number>();

  distances[source] = 0;

  for (let i = 0; i < n; i++) {
    let u = -1;
    let minDist = Infinity;
    for (let v = 0; v < n; v++) {
      if (!visited.has(v) && distances[v] < minDist) {
        u = v;
        minDist = distances[v];
      }
    }

    if (u === -1) break;
    visited.add(u);

    for (const edge of graph.adjacencyList.get(u) ?? []) {
      const alt = distances[u] + edge.weight;
      if (alt < distances[edge.to]) {
        distances[edge.to] = alt;
        predecessors[edge.to] = u;
      }
    }
  }

  return { distances, predecessors };
}

function reconstructPath(predecessors: (number | null)[], target: number): number[] {
  const path: number[] = [];
  let current: number | null = target;
  while (current !== null) {
    path.unshift(current);
    current = predecessors[current];
  }
  return path;
}

function floydWarshall(graph: Graph): { distances: number[][]; next: (number | null)[][] } {
  const n = graph.vertices;
  const dist: number[][] = Array.from({ length: n }, () => new Array(n).fill(Infinity));
  const next: (number | null)[][] = Array.from({ length: n }, () => new Array(n).fill(null));

  for (let i = 0; i < n; i++) dist[i][i] = 0;

  for (let u = 0; u < n; u++) {
    for (const edge of graph.adjacencyList.get(u) ?? []) {
      dist[u][edge.to] = edge.weight;
      next[u][edge.to] = edge.to;
    }
  }

  for (let k = 0; k < n; k++) {
    for (let i = 0; i < n; i++) {
      for (let j = 0; j < n; j++) {
        if (dist[i][k] + dist[k][j] < dist[i][j]) {
          dist[i][j] = dist[i][k] + dist[k][j];
          next[i][j] = next[i][k];
        }
      }
    }
  }

  return { distances: dist, next };
}

最大流问题
最大流问题求解网络中从源点 s 到汇点 t 的最大可行流量。每条边有容量上限，流量守恒约束要求除源汇外每个节点流入等于流出。最大流-最小割定理表明最大流等于最小割的容量。


function fordFulkerson(capacity: number[][], source: number, sink: number): { maxFlow: number; flow: number[][] } {
  const n = capacity.length;
  const flow: number[][] = Array.from({ length: n }, () => new Array(n).fill(0));
  let maxFlow = 0;

  function bfsPath(): number[] | null {
    const visited = new Set<number>();
    const parent = new Array(n).fill(-1);
    const queue: number[] = [source];
    visited.add(source);

    while (queue.length > 0) {
      const u = queue.shift()!;
      for (let v = 0; v < n; v++) {
        if (!visited.has(v) && capacity[u][v] - flow[u][v] > 0) {
          visited.add(v);
          parent[v] = u;
          if (v === sink) {
            const path: number[] = [];
            let curr = sink;
            while (curr !== source) {
              path.unshift(curr);
              curr = parent[curr];
            }
            path.unshift(source);
            return path;
          }
          queue.push(v);
        }
      }
    }

    return null;
  }

  let path = bfsPath();
  while (path !== null) {
    let pathFlow = Infinity;
    for (let i = 0; i < path.length - 1; i++) {
      const u = path[i];
      const v = path[i + 1];
      pathFlow = Math.min(pathFlow, capacity[u][v] - flow[u][v]);
    }

    for (let i = 0; i < path.length - 1; i++) {
      const u = path[i];
      const v = path[i + 1];
      flow[u][v] += pathFlow;
      flow[v][u] -= pathFlow;
    }

    maxFlow += pathFlow;
    path = bfsPath();
  }

  return { maxFlow, flow };
}
最大流问题有广泛的应用：网络带宽规划、任务分配、图像分割、棒球联赛淘汰判定等。最大流-最小割的对偶关系在优化理论中也具有深远意义。


网络优化应用
图与网络模型在实际工程中有广泛应用。最小生成树用于网络铺设成本优化，拓扑排序用于任务依赖调度，中心性分析用于社交网络关键节点识别。


interface MinSpanTreeResult {
  edges: { from: number; to: number; weight: number }[];
  totalWeight: number;
}

function kruskal(vertices: number, edges: { from: number; to: number; weight: number }[]): MinSpanTreeResult {
  const sorted = [...edges].sort((a, b) => a.weight - b.weight);
  const parent = Array.from({ length: vertices }, (_, i) => i);
  const rank = new Array(vertices).fill(0);

  function find(x: number): number {
    if (parent[x] !== x) parent[x] = find(parent[x]);
    return parent[x];
  }

  function union(x: number, y: number): boolean {
    const px = find(x);
    const py = find(y);
    if (px === py) return false;
    if (rank[px] < rank[py]) parent[px] = py;
    else if (rank[px] > rank[py]) parent[py] = px;
    else {
      parent[py] = px;
      rank[px]++;
    }
    return true;
  }

  const mstEdges: MinSpanTreeResult['edges'] = [];
  let totalWeight = 0;

  for (const edge of sorted) {
    if (union(edge.from, edge.to)) {
      mstEdges.push(edge);
      totalWeight += edge.weight;
      if (mstEdges.length === vertices - 1) break;
    }
  }

  return { edges: mstEdges, totalWeight };
}

function topologicalSort(graph: Graph): number[] | null {
  const n = graph.vertices;
  const inDegree = new Array(n).fill(0);

  for (let u = 0; u < n; u++) {
    for (const edge of graph.adjacencyList.get(u) ?? []) {
      inDegree[edge.to]++;
    }
  }

  const queue: number[] = [];
  for (let i = 0; i < n; i++) {
    if (inDegree[i] === 0) queue.push(i);
  }

  const order: number[] = [];
  while (queue.length > 0) {
    const u = queue.shift()!;
    order.push(u);
    for (const edge of graph.adjacencyList.get(u) ?? []) {
      inDegree[edge.to]--;
      if (inDegree[edge.to] === 0) queue.push(edge.to);
    }
  }

  return order.length === n ? order : null;
}

小结
    

  • 图模型通过顶点和边的抽象结构统一表示各种关系网络
    • 

  • 邻接矩阵适合稠密图（O(V²) 空间），邻接表适合稀疏图（O(V+E) 空间）
    • 

  • Dijkstra 算法求解非负权单源最短路径，Floyd-Warshall 求解全源最短路径
    • 

  • 最大流-最小割定理建立了流量与切割的对偶关系
    • 

  • Ford-Fulkerson 通过迭代寻找增广路径来计算最大流
    • 

  • Kruskal 算法用并查集高效求解最小生成树
    • 

  • 拓扑排序处理有向无环图的依赖关系，检测环的存在
    • 

  • 图论算法是路径规划、网络优化、社交分析等领域的基础工具
