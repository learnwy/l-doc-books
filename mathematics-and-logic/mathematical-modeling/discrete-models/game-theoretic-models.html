博弈论模型

核心概念

博弈论研究多个理性决策者之间的策略互动。在博弈中，每个参与者的收益不仅取决于自己的策略选择，还取决于其他参与者的策略。这种相互依赖性使得博弈论成为经济学、政治学、生物学和计算机科学中不可或缺的建模工具。

博弈论的核心概念是纳什均衡：一个策略组合，在该组合下每个参与者都无法通过单方面改变策略来提高自己的收益。纳什均衡不一定是社会最优（如囚徒困境），但它是理性预测的基础。

纳什均衡与策略分析
纳什均衡的求解对于 2×2 博弈可以通过穷举法或线性方程组直接计算。对于更大的博弈，需要支撑集枚举或 Lemke-Howson 算法。


interface PayoffMatrix {
  player1: number[][];
  player2: number[][];
}

interface NashEquilibrium {
  strategy1: number[];
  strategy2: number[];
  payoff1: number;
  payoff2: number;
  type: 'pure' | 'mixed';
}

function findPureNashEquilibria(game: PayoffMatrix): NashEquilibrium[] {
  const m = game.player1.length;
  const n = game.player1[0].length;
  const equilibria: NashEquilibrium[] = [];

  for (let i = 0; i < m; i++) {
    for (let j = 0; j < n; j++) {
      let isBestResponse1 = true;
      for (let k = 0; k < m; k++) {
        if (game.player1[k][j] > game.player1[i][j]) {
          isBestResponse1 = false;
          break;
        }
      }

      let isBestResponse2 = true;
      for (let k = 0; k < n; k++) {
        if (game.player2[i][k] > game.player2[i][j]) {
          isBestResponse2 = false;
          break;
        }
      }

      if (isBestResponse1 && isBestResponse2) {
        const s1 = new Array(m).fill(0);
        const s2 = new Array(n).fill(0);
        s1[i] = 1;
        s2[j] = 1;
        equilibria.push({
          strategy1: s1,
          strategy2: s2,
          payoff1: game.player1[i][j],
          payoff2: game.player2[i][j],
          type: 'pure',
        });
      }
    }
  }

  return equilibria;
}

function findMixedNashEquilibrium2x2(game: PayoffMatrix): NashEquilibrium | null {
  const [[a, b], [c, d]] = game.player2;
  const denom1 = a - b - c + d;
  if (Math.abs(denom1) < 1e-10) return null;
  const q = (a - c) / denom1;

  const [[e, f], [g, h]] = game.player1;
  const denom2 = e - f - g + h;
  if (Math.abs(denom2) < 1e-10) return null;
  const p = (e - g) / denom2;

  if (p < 0 || p > 1 || q < 0 || q > 1) return null;

  const payoff1 = p * (q * e + (1 - q) * f) + (1 - p) * (q * g + (1 - q) * h);
  const payoff2 = q * (p * a + (1 - p) * c) + (1 - q) * (p * b + (1 - p) * d);

  return {
    strategy1: [p, 1 - p],
    strategy2: [q, 1 - q],
    payoff1,
    payoff2,
    type: 'mixed',
  };
}
经典博弈矩阵分析——以囚徒困境为例：两个嫌疑人分别决定合作（沉默）或背叛（揭发）。理性分析下，双方都选择背叛是唯一的纳什均衡，尽管双方合作能获得更好的结果。这个矛盾揭示了个体理性与集体理性的冲突。

零和博弈与极小极大策略
零和博弈中一方的收益恰好等于另一方的损失，总和为零。极小极大定理（冯·诺伊曼）证明了在混合策略下零和博弈总存在一个确定的值——博弈值。


interface ZeroSumGame {
  payoff: number[][];
}

function minimaxValue(game: ZeroSumGame): {
  value: number;
  rowStrategy: number;
  colStrategy: number;
} {
  const m = game.payoff.length;
  const n = game.payoff[0].length;

  const rowMins = game.payoff.map((row) => Math.min(...row));
  const maximin = Math.max(...rowMins);
  const rowStrategy = rowMins.indexOf(maximin);

  const colMaxs: number[] = [];
  for (let j = 0; j < n; j++) {
    colMaxs.push(Math.max(...game.payoff.map((row) => row[j])));
  }
  const minimax = Math.min(...colMaxs);
  const colStrategy = colMaxs.indexOf(minimax);

  return {
    value: maximin === minimax ? maximin : (maximin + minimax) / 2,
    rowStrategy,
    colStrategy,
  };
}

function solveMixedStrategy2x2(payoff: number[][]): { p: number; q: number; value: number } {
  const [[a, b], [c, d]] = payoff;
  const denom = a - b - c + d;

  if (Math.abs(denom) < 1e-10) {
    return { p: 0.5, q: 0.5, value: (a + d) / 2 };
  }

  const p = (d - c) / denom;
  const q = (d - b) / denom;
  const value = (a * d - b * c) / denom;

  return { p, q, value };
}
零和博弈的极小极大策略在 AI 游戏中广泛应用。棋类游戏（国际象棋、围棋）的搜索算法本质上就是在博弈树上执行极小极大搜索，配合 α-β 剪枝减少搜索空间。


演化博弈论
演化博弈论放弃了经典博弈论的理性假设，转而研究策略在群体中通过自然选择（或社会学习）的演化过程。核心工具是复制动态方程：策略的增长率与其相对适应度成正比。


interface EvolutionaryGame {
  payoff: number[][];
  strategies: string[];
}

function replicatorDynamics(
  game: EvolutionaryGame,
  initial: number[],
  tEnd: number,
  dt: number,
): { t: number[]; frequencies: number[][] } {
  const n = game.strategies.length;
  const t: number[] = [];
  const frequencies: number[][] = [];
  let x = [...initial];

  for (let time = 0; time <= tEnd; time += dt) {
    t.push(time);
    frequencies.push([...x]);

    const fitness: number[] = [];
    for (let i = 0; i < n; i++) {
      let fi = 0;
      for (let j = 0; j < n; j++) {
        fi += game.payoff[i][j] * x[j];
      }
      fitness.push(fi);
    }

    const avgFitness = fitness.reduce((sum, f, i) => sum + f * x[i], 0);

    const dx = x.map((xi, i) => xi * (fitness[i] - avgFitness));
    x = x.map((xi, i) => Math.max(0, xi + dx[i] * dt));
    const total = x.reduce((a, b) => a + b, 0);
    x = x.map((xi) => xi / total);
  }

  return { t, frequencies };
}

function isESS(payoff: number[][], strategyIndex: number): boolean {
  const n = payoff.length;
  const e = payoff[strategyIndex][strategyIndex];

  for (let j = 0; j < n; j++) {
    if (j === strategyIndex) continue;

    const mutantVsIncumbent = payoff[j][strategyIndex];
    if (mutantVsIncumbent > e) return false;

    if (mutantVsIncumbent === e && payoff[j][j] >= payoff[strategyIndex][j]) {
      return false;
    }
  }

  return true;
}
演化稳定策略（ESS）是不可被突变策略入侵的策略。一个策略 σ 是 ESS，当且仅当对任何突变策略 τ：要么 σ 对 σ 的收益严格大于 τ 对 σ 的收益，要么两者收益相等但 σ 对 τ 的收益大于 τ 对 τ 的收益。


小结
    

  • 博弈论建模多个理性决策者之间的策略互动，核心概念是纳什均衡
    • 

  • 纳什均衡中每个参与者都在给定对手策略下做出最优反应
    • 

  • 囚徒困境揭示了个体理性与集体理性的冲突
    • 

  • 零和博弈的极小极大定理保证了博弈值的存在，是 AI 游戏搜索的基础
    • 

  • 混合策略纳什均衡在纯策略均衡不存在时仍然存在
    • 

  • 演化博弈论不假设理性，通过复制动态方程描述策略的演化过程
    • 

  • ESS 是不可被突变策略入侵的稳定状态，对应生态和社会系统的长期均衡
    • 

  • 博弈论广泛应用于市场竞争、机制设计、拍卖理论和多智能体系统
