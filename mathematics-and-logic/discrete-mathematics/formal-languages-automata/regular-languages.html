正则语言

核心概念

正则语言（Regular Language）是 Chomsky 层次结构中最底层、也是性质最丰富的一类形式语言。正则语言有三种等价的描述方式：正则表达式（代数描述）、有限自动机（机器模型）和正则文法（生成规则）。这种"三位一体"的等价性是形式语言理论最优美的结论之一。

正则语言的核心魅力在于其判定性——几乎所有关于正则语言的问题（空性、等价性、包含性、有限性）都是可判定的，且有高效算法。这与上下文无关语言和递归可枚举语言形成鲜明对比。在实际工程中，正则语言是词法分析、模式匹配、网络入侵检测和协议验证的理论基础。

正则表达式与 Kleene 定理
正则表达式通过三种基本运算构建：并（|）、连接（·）和 Kleene 闭包（*）。形式化定义：


    

  1. ∅ 和 ε 是正则表达式
    2. 

  2. 对每个 a ∈ Σ，a 是正则表达式
    3. 

  3. 若 R₁, R₂ 是正则表达式，则 R₁|R₂、R₁R₂、R₁* 也是正则表达式
    4. 



Kleene 定理：一个语言是正则的，当且仅当它可以用正则表达式描述。这建立了正则表达式和有限自动机之间的完美桥梁。


type RegexAST =
  | { type: 'empty' }
  | { type: 'epsilon' }
  | { type: 'char'; value: string }
  | { type: 'union'; left: RegexAST; right: RegexAST }
  | { type: 'concat'; left: RegexAST; right: RegexAST }
  | { type: 'star'; child: RegexAST };

function parseRegex(pattern: string): RegexAST {
  let pos = 0;

  function parseUnion(): RegexAST {
    let node = parseConcat();
    while (pos < pattern.length && pattern[pos] === '|') {
      pos++;
      node = { type: 'union', left: node, right: parseConcat() };
    }
    return node;
  }

  function parseConcat(): RegexAST {
    let node = parseStar();
    while (pos < pattern.length && pattern[pos] !== ')' && pattern[pos] !== '|') {
      node = { type: 'concat', left: node, right: parseStar() };
    }
    return node;
  }

  function parseStar(): RegexAST {
    let node = parseAtom();
    while (pos < pattern.length && pattern[pos] === '*') {
      pos++;
      node = { type: 'star', child: node };
    }
    return node;
  }

  function parseAtom(): RegexAST {
    if (pos >= pattern.length) return { type: 'epsilon' };
    if (pattern[pos] === '(') {
      pos++;
      const node = parseUnion();
      pos++;
      return node;
    }
    return { type: 'char', value: pattern[pos++] };
  }

  return parseUnion();
}

function regexToNFA(ast: RegexAST): {
  states: number;
  start: number;
  accept: number;
  transitions: [number, string, number][];
} {
  let stateCount = 0;
  const newState = () => stateCount++;

  function build(node: RegexAST): { start: number; accept: number; transitions: [number, string, number][] } {
    switch (node.type) {
      case 'epsilon': {
        const s = newState();
        const a = newState();
        return { start: s, accept: a, transitions: [[s, 'ε', a]] };
      }
      case 'char': {
        const s = newState();
        const a = newState();
        return { start: s, accept: a, transitions: [[s, node.value, a]] };
      }
      case 'union': {
        const s = newState();
        const a = newState();
        const l = build(node.left);
        const r = build(node.right);
        return {
          start: s,
          accept: a,
          transitions: [
            ...l.transitions,
            ...r.transitions,
            [s, 'ε', l.start],
            [s, 'ε', r.start],
            [l.accept, 'ε', a],
            [r.accept, 'ε', a],
          ],
        };
      }
      case 'concat': {
        const l = build(node.left);
        const r = build(node.right);
        return {
          start: l.start,
          accept: r.accept,
          transitions: [...l.transitions, ...r.transitions, [l.accept, 'ε', r.start]],
        };
      }
      case 'star': {
        const s = newState();
        const a = newState();
        const c = build(node.child);
        return {
          start: s,
          accept: a,
          transitions: [...c.transitions, [s, 'ε', c.start], [s, 'ε', a], [c.accept, 'ε', c.start], [c.accept, 'ε', a]],
        };
      }
      default: {
        const s = newState();
        return { start: s, accept: s, transitions: [] };
      }
    }
  }

  const result = build(ast);
  return { states: stateCount, ...result };
}
Myhill-Nerode 定理与最小化
Myhill-Nerode 定理是正则语言理论中最深刻的结果之一，它给出了一个语言是正则的充要条件，同时确定了最小 DFA 的状态数。


定义右同余关系 ~_L：x ~_L y 当且仅当对所有 z ∈ Σ*，xz ∈ L ⟺ yz ∈ L。


Myhill-Nerode 定理：语言 L 是正则的，当且仅当 ~_L 的等价类数目有限。此时最小 DFA 的状态数恰好等于等价类数目。


function myhillNerodeClasses(
  alphabet: string[],
  accepts: (s: string) => boolean,
  maxLen: number,
): Map<string, string[]> {
  const strings: string[] = [''];
  for (let len = 1; len <= maxLen; len++) {
    for (const s of [...strings]) {
      if (s.length === len - 1) {
        for (const c of alphabet) strings.push(s + c);
      }
    }
  }

  const classes = new Map<string, string[]>();
  const getKey = (s: string): string => {
    const suffixes: string[] = [''];
    for (let len = 1; len <= maxLen - s.length; len++) {
      const generate = (prefix: string, remaining: number) => {
        if (remaining === 0) {
          suffixes.push(prefix);
          return;
        }
        for (const c of alphabet) generate(prefix + c, remaining - 1);
      };
      generate('', len);
    }
    return suffixes.map((z) => (accepts(s + z) ? '1' : '0')).join('');
  };

  for (const s of strings) {
    const key = getKey(s);
    if (!classes.has(key)) classes.set(key, []);
    classes.get(key)!.push(s);
  }

  return classes;
}

function tableFillingMinimization(
  stateCount: number,
  acceptStates: Set<number>,
  transition: (state: number, char: string) => number,
  alphabet: string[],
): Set<string> {
  const distinguishable = new Set<string>();
  const pair = (i: number, j: number): string => (i < j ? `${i},${j}` : `${j},${i}`);

  for (let i = 0; i < stateCount; i++) {
    for (let j = i + 1; j < stateCount; j++) {
      if (acceptStates.has(i) !== acceptStates.has(j)) {
        distinguishable.add(pair(i, j));
      }
    }
  }

  let changed = true;
  while (changed) {
    changed = false;
    for (let i = 0; i < stateCount; i++) {
      for (let j = i + 1; j < stateCount; j++) {
        if (distinguishable.has(pair(i, j))) continue;
        for (const c of alphabet) {
          const ni = transition(i, c);
          const nj = transition(j, c);
          if (ni !== nj && distinguishable.has(pair(ni, nj))) {
            distinguishable.add(pair(i, j));
            changed = true;
            break;
          }
        }
      }
    }
  }
  return distinguishable;
}

function productAutomaton(
  dfa1: { states: number; transition: (s: number, c: string) => number; start: number; accept: Set<number> },
  dfa2: { states: number; transition: (s: number, c: string) => number; start: number; accept: Set<number> },
  alphabet: string[],
  mode: 'intersection' | 'union' | 'difference',
): { states: number; start: number; accept: Set<number>; transition: Map<string, number> } {
  const states = dfa1.states * dfa2.states;
  const encode = (s1: number, s2: number) => s1 * dfa2.states + s2;
  const start = encode(dfa1.start, dfa2.start);
  const accept = new Set<number>();
  const trans = new Map<string, number>();

  for (let s1 = 0; s1 < dfa1.states; s1++) {
    for (let s2 = 0; s2 < dfa2.states; s2++) {
      const s = encode(s1, s2);
      const a1 = dfa1.accept.has(s1);
      const a2 = dfa2.accept.has(s2);

      const isAccept = mode === 'intersection' ? a1 && a2 : mode === 'union' ? a1 || a2 : a1 && !a2;

      if (isAccept) accept.add(s);

      for (const c of alphabet) {
        trans.set(`${s},${c}`, encode(dfa1.transition(s1, c), dfa2.transition(s2, c)));
      }
    }
  }

  return { states, start, accept, transition: trans };
}
小结
    

  • 正则语言有三种等价描述：正则表达式、有限自动机、正则文法，Kleene 定理建立了它们的等价性
    • 

  • Myhill-Nerode 定理给出正则性的充要条件，等价类数目决定了最小 DFA 的状态数
    • 

  • 正则语言在并、交、补、连接、Kleene 星下封闭，几乎所有判定问题都有高效算法
    • 

  • Thompson 构造将正则表达式转为 NFA，子集构造转为 DFA，Hopcroft 算法实现 DFA 最小化
    • 

  • 正则语言理论是词法分析器、正则匹配引擎和协议验证工具的数学基础
