几何编程实践

核心概念

几何编程实践将理论几何算法转化为可靠的工程代码。在实际开发中，几何算法面临三大挑战：浮点精度问题、边界情况处理和性能优化。一个理论上正确的几何算法，在浮点运算下可能产生完全错误的结果——两条看似不相交的线段被判为相交，一个应该在多边形内的点被判为外部。

本文聚焦于四个核心几何编程任务：点在多边形内判定、最近点对问题、几何基元库设计以及浮点精度处理策略。这些是构建任何几何系统（游戏引擎、GIS、CAD）的基础砖块。

点在多边形内判定
点在多边形内判定（Point-in-Polygon）是最常见的几何查询。主要有两种方法：射线法（ray casting）和卷绕数法（winding number）。


射线法从查询点向任意方向发射一条射线，统计与多边形边的交点数：奇数为内部，偶数为外部。这是最简单且高效的方法。


卷绕数法计算多边形边界绕查询点的圈数。圈数非零表示点在内部。这种方法可以正确处理自交多边形。


interface Point {
  x: number;
  y: number;
}

function pointInPolygonRayCast(point: Point, polygon: Point[]): boolean {
  const n = polygon.length;
  let inside = false;

  for (let i = 0, j = n - 1; i < n; j = i++) {
    const xi = polygon[i].x,
      yi = polygon[i].y;
    const xj = polygon[j].x,
      yj = polygon[j].y;

    if (yi > point.y !== yj > point.y && point.x < ((xj - xi) * (point.y - yi)) / (yj - yi) + xi) {
      inside = !inside;
    }
  }

  return inside;
}

function windingNumber(point: Point, polygon: Point[]): number {
  const n = polygon.length;
  let winding = 0;

  for (let i = 0; i < n; i++) {
    const j = (i + 1) % n;
    const yi = polygon[i].y;
    const yj = polygon[j].y;

    if (yi <= point.y) {
      if (yj > point.y) {
        if (isLeft(polygon[i], polygon[j], point) > 0) {
          winding++;
        }
      }
    } else {
      if (yj <= point.y) {
        if (isLeft(polygon[i], polygon[j], point) < 0) {
          winding--;
        }
      }
    }
  }

  return winding;
}

function isLeft(a: Point, b: Point, p: Point): number {
  return (b.x - a.x) * (p.y - a.y) - (p.x - a.x) * (b.y - a.y);
}

function pointOnPolygonEdge(point: Point, polygon: Point[], eps: number = 1e-10): boolean {
  const n = polygon.length;
  for (let i = 0; i < n; i++) {
    const j = (i + 1) % n;
    const a = polygon[i],
      b = polygon[j];
    const cross = (b.x - a.x) * (point.y - a.y) - (b.y - a.y) * (point.x - a.x);
    if (Math.abs(cross) > eps) continue;
    const dotProduct = (point.x - a.x) * (b.x - a.x) + (point.y - a.y) * (b.y - a.y);
    const lenSq = (b.x - a.x) ** 2 + (b.y - a.y) ** 2;
    if (dotProduct >= -eps && dotProduct <= lenSq + eps) return true;
  }
  return false;
}

最近点对问题
最近点对（Closest Pair）问题求 n 个点中距离最近的两个点。暴力方法 O(n²)，分治法可达 O(n log n)。


分治思想：将点集按 x 坐标中位数分为左右两半，分别求解，然后检查跨越分界线的近距点对。关键优化是只需检查距分界线不超过 d（当前最短距离）的点。


function closestPairBrute(points: Point[]): { dist: number; p1: number; p2: number } {
  let minDist = Infinity;
  let p1 = 0,
    p2 = 1;

  for (let i = 0; i < points.length; i++) {
    for (let j = i + 1; j < points.length; j++) {
      const d = Math.hypot(points[i].x - points[j].x, points[i].y - points[j].y);
      if (d < minDist) {
        minDist = d;
        p1 = i;
        p2 = j;
      }
    }
  }

  return { dist: minDist, p1, p2 };
}

function closestPairDivideConquer(points: Point[]): number {
  const sorted = points.map((p, i) => ({ ...p, idx: i })).sort((a, b) => a.x - b.x);

  function solve(pts: typeof sorted): number {
    if (pts.length <= 3) {
      let min = Infinity;
      for (let i = 0; i < pts.length; i++) {
        for (let j = i + 1; j < pts.length; j++) {
          min = Math.min(min, Math.hypot(pts[i].x - pts[j].x, pts[i].y - pts[j].y));
        }
      }
      return min;
    }

    const mid = Math.floor(pts.length / 2);
    const midX = pts[mid].x;
    const left = pts.slice(0, mid);
    const right = pts.slice(mid);

    let d = Math.min(solve(left), solve(right));

    const strip = pts.filter((p) => Math.abs(p.x - midX) < d).sort((a, b) => a.y - b.y);

    for (let i = 0; i < strip.length; i++) {
      for (let j = i + 1; j < strip.length && strip[j].y - strip[i].y < d; j++) {
        d = Math.min(d, Math.hypot(strip[i].x - strip[j].x, strip[i].y - strip[j].y));
      }
    }

    return d;
  }

  return solve(sorted);
}
几何基元库设计
设计一个可靠的几何基元库需要考虑类型安全、不可变性和浮点精度处理。以下是面向游戏和图形应用的基元库设计。


const EPSILON = 1e-10;

function nearlyEqual(a: number, b: number, eps: number = EPSILON): boolean {
  if (a === b) return true;
  const diff = Math.abs(a - b);
  if (a === 0 || b === 0 || diff < Number.MIN_VALUE) {
    return diff < eps;
  }
  return diff / Math.min(Math.abs(a) + Math.abs(b), Number.MAX_VALUE) < eps;
}

interface Vec2 {
  readonly x: number;
  readonly y: number;
}

function vec2(x: number, y: number): Vec2 {
  return { x, y };
}

function add(a: Vec2, b: Vec2): Vec2 {
  return vec2(a.x + b.x, a.y + b.y);
}

function sub(a: Vec2, b: Vec2): Vec2 {
  return vec2(a.x - b.x, a.y - b.y);
}

function scale(v: Vec2, s: number): Vec2 {
  return vec2(v.x * s, v.y * s);
}

function dot(a: Vec2, b: Vec2): number {
  return a.x * b.x + a.y * b.y;
}

function cross(a: Vec2, b: Vec2): number {
  return a.x * b.y - a.y * b.x;
}

function length(v: Vec2): number {
  return Math.sqrt(v.x * v.x + v.y * v.y);
}

function normalize(v: Vec2): Vec2 {
  const len = length(v);
  return len > EPSILON ? scale(v, 1 / len) : vec2(0, 0);
}

function perpendicular(v: Vec2): Vec2 {
  return vec2(-v.y, v.x);
}

function distanceToSegment(p: Vec2, a: Vec2, b: Vec2): number {
  const ab = sub(b, a);
  const ap = sub(p, a);
  const t = Math.max(0, Math.min(1, dot(ap, ab) / dot(ab, ab)));
  const closest = add(a, scale(ab, t));
  return length(sub(p, closest));
}

浮点精度与鲁棒性
浮点精度问题是几何编程中最隐蔽也最致命的陷阱。IEEE 754 双精度浮点数有约 15-17 位有效数字，但几何运算中的减法消去（catastrophic cancellation）可能导致精度急剧下降。

function robustOrientation(a: Vec2, b: Vec2, c: Vec2): number {
  const det = (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);

  const absSum = Math.abs(b.x - a.x) * Math.abs(c.y - a.y) + Math.abs(b.y - a.y) * Math.abs(c.x - a.x);

  if (Math.abs(det) < 1e-14 * absSum) {
    return 0;
  }
  return det > 0 ? 1 : -1;
}

function kahanSum(values: number[]): number {
  let sum = 0;
  let compensation = 0;

  for (const value of values) {
    const y = value - compensation;
    const t = sum + y;
    compensation = t - sum - y;
    sum = t;
  }

  return sum;
}

function robustPolygonArea(vertices: Vec2[]): number {
  const n = vertices.length;
  const terms: number[] = [];

  for (let i = 0; i < n; i++) {
    const j = (i + 1) % n;
    terms.push(vertices[i].x * vertices[j].y);
    terms.push(-vertices[j].x * vertices[i].y);
  }

  return Math.abs(kahanSum(terms)) / 2;
}

function robustSegmentIntersection(a1: Vec2, a2: Vec2, b1: Vec2, b2: Vec2): Vec2 | null {
  const d1 = robustOrientation(b1, b2, a1);
  const d2 = robustOrientation(b1, b2, a2);
  const d3 = robustOrientation(a1, a2, b1);
  const d4 = robustOrientation(a1, a2, b2);

  if (d1 * d2 < 0 && d3 * d4 < 0) {
    const da = sub(a2, a1);
    const db = sub(b2, b1);
    const denom = cross(da, db);
    const t = cross(sub(b1, a1), db) / denom;
    return add(a1, scale(da, t));
  }

  return null;
}
自适应精度策略是现代计算几何库（如 Shewchuk 的 robust predicates）采用的方法：先用快速的浮点运算判断，如果结果的误差界不能确定方位，再切换到精确算术（如扩展精度或有理数算术）。


小结
    

  • 点在多边形内判定：射线法简单高效适用于简单多边形，卷绕数法可处理自交多边形
    • 

  • 最近点对问题：分治法 O(n log n) 是最优算法，关键优化是条带内每点只需检查常数个邻居
    • 

  • 几何基元库设计应遵循不可变性、值语义和集中 epsilon 管理的原则
    • 

  • 浮点精度是几何编程最大的工程挑战，减法消去和累积误差是两大陷阱
    • 

  • 鲁棒方位测试通过误差界判断来避免错误的几何判定
    • 

  • Kahan 求和可以有效减少浮点累积误差，提高面积和距离计算的精度
    • 

  • 自适应精度策略在快速判断和精确算术之间平衡，是工业级几何库的标准做法
