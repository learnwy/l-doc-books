线段相交判定

核心概念

线段相交判定是计算几何中最基本的操作之一。判断两条线段是否相交、求交点坐标，是碰撞检测、地图渲染、路径规划等应用的底层基元。看似简单的问题实际上涉及精妙的数学——叉积法可以优雅地处理所有边界情况，而当需要处理大量线段时，扫描线算法（如 Bentley-Ottmann）将暴力搜索的 O(n²) 优化为 O((n+k) log n)。

两条线段相交判定的核心工具是叉积：通过计算端点相对于另一条线段的方位（左侧、右侧或共线），判断两条线段是否"跨越"彼此。当且仅当两条线段互相跨越时，它们相交。

叉积方位测试法
叉积方位测试是线段相交判定的核心。给定有序三点 P₁P₂P₃，计算叉积 (P₂-P₁) × (P₃-P₁) 的符号：正值表示 P₃ 在 P₁P₂ 的左侧（逆时针），负值表示右侧（顺时针），零表示共线。


两线段 AB 和 CD 相交的充要条件是：A 和 B 在 CD 的两侧，且 C 和 D 在 AB 的两侧。共线情况需要额外的包围盒检测。


interface Point {
  x: number;
  y: number;
}

interface Segment {
  p1: Point;
  p2: Point;
}

function cross(o: Point, a: Point, b: Point): number {
  return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
}

function onSegment(p: Point, q: Point, r: Point): boolean {
  return (
    Math.min(p.x, r.x) <= q.x + 1e-10 &&
    q.x <= Math.max(p.x, r.x) + 1e-10 &&
    Math.min(p.y, r.y) <= q.y + 1e-10 &&
    q.y <= Math.max(p.y, r.y) + 1e-10
  );
}

function segmentsIntersect(s1: Segment, s2: Segment): boolean {
  const d1 = cross(s2.p1, s2.p2, s1.p1);
  const d2 = cross(s2.p1, s2.p2, s1.p2);
  const d3 = cross(s1.p1, s1.p2, s2.p1);
  const d4 = cross(s1.p1, s1.p2, s2.p2);

  if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) && ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))) {
    return true;
  }

  if (Math.abs(d1) < 1e-10 && onSegment(s2.p1, s1.p1, s2.p2)) return true;
  if (Math.abs(d2) < 1e-10 && onSegment(s2.p1, s1.p2, s2.p2)) return true;
  if (Math.abs(d3) < 1e-10 && onSegment(s1.p1, s2.p1, s1.p2)) return true;
  if (Math.abs(d4) < 1e-10 && onSegment(s1.p1, s2.p2, s1.p2)) return true;

  return false;
}

交点坐标计算
当确认两线段相交后，通常需要计算精确的交点坐标。参数方程法将两条线段参数化，联立求解参数 t 和 u，当 0 ≤ t ≤ 1 且 0 ≤ u ≤ 1 时线段相交。


线段 AB：P = A + t(B - A)，t ∈ [0, 1] 线段 CD：P = C + u(D - C)，u ∈ [0, 1]


interface IntersectionResult {
  intersects: boolean;
  point: Point | null;
  t: number;
  u: number;
}

function segmentIntersection(s1: Segment, s2: Segment): IntersectionResult {
  const dx1 = s1.p2.x - s1.p1.x;
  const dy1 = s1.p2.y - s1.p1.y;
  const dx2 = s2.p2.x - s2.p1.x;
  const dy2 = s2.p2.y - s2.p1.y;

  const denom = dx1 * dy2 - dy1 * dx2;

  if (Math.abs(denom) < 1e-10) {
    return { intersects: false, point: null, t: 0, u: 0 };
  }

  const dx3 = s2.p1.x - s1.p1.x;
  const dy3 = s2.p1.y - s1.p1.y;

  const t = (dx3 * dy2 - dy3 * dx2) / denom;
  const u = (dx3 * dy1 - dy3 * dx1) / denom;

  const eps = -1e-10;
  if (t >= eps && t <= 1 + 1e-10 && u >= eps && u <= 1 + 1e-10) {
    return {
      intersects: true,
      point: {
        x: s1.p1.x + t * dx1,
        y: s1.p1.y + t * dy1,
      },
      t,
      u,
    };
  }

  return { intersects: false, point: null, t, u };
}

function raySegmentIntersection(
  origin: Point,
  direction: Point,
  segment: Segment,
): { hit: boolean; t: number; point: Point | null } {
  const dx = segment.p2.x - segment.p1.x;
  const dy = segment.p2.y - segment.p1.y;
  const denom = direction.x * dy - direction.y * dx;

  if (Math.abs(denom) < 1e-10) {
    return { hit: false, t: Infinity, point: null };
  }

  const ox = segment.p1.x - origin.x;
  const oy = segment.p1.y - origin.y;
  const t = (ox * dy - oy * dx) / denom;
  const u = (ox * direction.y - oy * direction.x) / denom;

  if (t >= 0 && u >= -1e-10 && u <= 1 + 1e-10) {
    return {
      hit: true,
      t,
      point: {
        x: origin.x + t * direction.x,
        y: origin.y + t * direction.y,
      },
    };
  }

  return { hit: false, t: Infinity, point: null };
}
扫描线算法
当需要检测 n 条线段中的所有相交对时，暴力方法需要 O(n²) 时间。Bentley-Ottmann 扫描线算法通过维护一条从左到右移动的垂直扫描线，只检测扫描线上相邻的线段对，将复杂度降至 O((n+k) log n)，其中 k 是交点数。


扫描线算法的核心数据结构：


    

  1. 事件队列：存储所有线段端点和交点，按 x 坐标排序
    2. 

  2. 状态结构：存储当前与扫描线相交的线段，按 y 坐标排序
    3. 



type EventType = 'left' | 'right' | 'intersection';

interface SweepEvent {
  x: number;
  y: number;
  type: EventType;
  segments: Segment[];
}

function shamosHoey(segments: Segment[]): boolean {
  const events: SweepEvent[] = [];

  for (const seg of segments) {
    const [left, right] =
      seg.p1.x < seg.p2.x || (seg.p1.x === seg.p2.x && seg.p1.y < seg.p2.y) ? [seg.p1, seg.p2] : [seg.p2, seg.p1];

    events.push({ x: left.x, y: left.y, type: 'left', segments: [seg] });
    events.push({ x: right.x, y: right.y, type: 'right', segments: [seg] });
  }

  events.sort((a, b) => (a.x !== b.x ? a.x - b.x : a.y - b.y));

  const active: Segment[] = [];

  for (const event of events) {
    if (event.type === 'left') {
      active.push(event.segments[0]);
      const idx = active.length - 1;
      if (idx > 0 && segmentsIntersect(active[idx], active[idx - 1])) return true;
      if (idx < active.length - 1 && segmentsIntersect(active[idx], active[idx + 1])) return true;
    } else {
      const idx = active.indexOf(event.segments[0]);
      if (idx > 0 && idx < active.length - 1) {
        if (segmentsIntersect(active[idx - 1], active[idx + 1])) return true;
      }
      active.splice(idx, 1);
    }
  }

  return false;
}

function lineSegmentPolygonIntersection(seg: Segment, polygon: Point[]): Point[] {
  const intersections: Point[] = [];
  const n = polygon.length;

  for (let i = 0; i < n; i++) {
    const j = (i + 1) % n;
    const polySeg: Segment = { p1: polygon[i], p2: polygon[j] };
    const result = segmentIntersection(seg, polySeg);
    if (result.intersects && result.point) {
      intersections.push(result.point);
    }
  }

  return intersections;
}

function visibilityCheck(from: Point, to: Point, obstacles: Segment[]): boolean {
  const sight: Segment = { p1: from, p2: to };
  return !obstacles.some((obs) => segmentsIntersect(sight, obs));
}

function clipLineToRect(seg: Segment, minX: number, minY: number, maxX: number, maxY: number): Segment | null {
  let t0 = 0,
    t1 = 1;
  const dx = seg.p2.x - seg.p1.x;
  const dy = seg.p2.y - seg.p1.y;

  const edges = [
    { p: -dx, q: seg.p1.x - minX },
    { p: dx, q: maxX - seg.p1.x },
    { p: -dy, q: seg.p1.y - minY },
    { p: dy, q: maxY - seg.p1.y },
  ];

  for (const { p, q } of edges) {
    if (Math.abs(p) < 1e-10) {
      if (q < 0) return null;
    } else {
      const r = q / p;
      if (p < 0) t0 = Math.max(t0, r);
      else t1 = Math.min(t1, r);
    }
  }

  if (t0 > t1) return null;

  return {
    p1: { x: seg.p1.x + t0 * dx, y: seg.p1.y + t0 * dy },
    p2: { x: seg.p1.x + t1 * dx, y: seg.p1.y + t1 * dy },
  };
}

小结
    

  • 线段相交判定的核心工具是叉积方位测试：通过判断端点在线段两侧来确定相交
    • 

  • 共线情况需要额外的包围盒检测，这是实现中最容易遗漏的边界情况
    • 

  • 参数方程法同时提供是否相交和交点坐标，参数 t, u ∈ [0,1] 时线段相交
    • 

  • 射线-线段相交是光线投射和可见性判断的基础，t ≥ 0 且 u ∈ [0,1] 时射线命中
    • 

  • Bentley-Ottmann 扫描线将 n 条线段的批量相交检测从 O(n²) 优化到 O((n+k) log n)
    • 

  • Liang-Barsky 裁剪算法高效地将线段裁剪到矩形区域内，是图形渲染管线的基础
    • 

  • 浮点精度问题（epsilon 处理）是线段相交实现中最大的工程挑战
    •
