信道模型

核心概念

信道模型（Channel Models）是信息论分析通信系统的数学抽象。信道模型用转移概率 p(y|x) 描述输入与输出之间的统计关系，将物理传输介质的复杂行为简化为可分析的数学对象。不同的信道模型捕捉了不同类型的传输损伤，从简单的比特翻转到复杂的多径衰落。

选择合适的信道模型是通信系统设计的第一步：模型太简单会低估实际干扰，模型太复杂则难以分析和优化。实际工程中，往往需要在分析可行性和模型精度之间取得平衡。

离散无记忆信道
离散无记忆信道（DMC）是最基本的信道模型，输入和输出都取有限字母表中的值，且每次使用信道的概率转移独立于过去。


interface DMCChannel {
  inputAlphabet: number;
  outputAlphabet: number;
  transition: number[][];
}

function createBSC(p: number): DMCChannel {
  return {
    inputAlphabet: 2,
    outputAlphabet: 2,
    transition: [
      [1 - p, p],
      [p, 1 - p],
    ],
  };
}

function createBEC(epsilon: number): DMCChannel {
  return {
    inputAlphabet: 2,
    outputAlphabet: 3,
    transition: [
      [1 - epsilon, 0, epsilon],
      [0, 1 - epsilon, epsilon],
    ],
  };
}

function createZChannel(p: number): DMCChannel {
  return {
    inputAlphabet: 2,
    outputAlphabet: 2,
    transition: [
      [1, 0],
      [p, 1 - p],
    ],
  };
}

function simulateChannel(channel: DMCChannel, input: number[]): number[] {
  return input.map((x) => {
    const row = channel.transition[x];
    let r = Math.random();
    for (let y = 0; y < row.length; y++) {
      r -= row[y];
      if (r <= 0) return y;
    }
    return row.length - 1;
  });
}

function estimateBER(channel: DMCChannel, nTrials: number = 10000): number {
  let errors = 0;
  for (let i = 0; i < nTrials; i++) {
    const input = Math.round(Math.random());
    const output = simulateChannel(channel, [input]);
    if (output[0] !== input && output[0] < channel.inputAlphabet) {
      errors++;
    }
  }
  return errors / nTrials;
}

const bsc = createBSC(0.1);
const bec = createBEC(0.3);
const zch = createZChannel(0.15);

console.log(`BSC(0.1) 实测 BER: ${estimateBER(bsc).toFixed(4)}`);
console.log(`BEC(0.3) 模拟完成`);
console.log(`Z信道(0.15) 实测 BER: ${estimateBER(zch).toFixed(4)}`);
加性高斯白噪声信道
AWGN 信道是连续信道中最重要的模型，Y = X + N，其中噪声 N 是均值为零、方差为 σ² 的高斯随机变量。AWGN 模型适用于热噪声主导的通信场景。


function awgnChannel(signal: number[], snrDb: number): { received: number[]; noisePower: number } {
  const signalPower = signal.reduce((s, x) => s + x * x, 0) / signal.length;
  const snrLinear = Math.pow(10, snrDb / 10);
  const noisePower = signalPower / snrLinear;
  const noiseStd = Math.sqrt(noisePower);

  const received = signal.map((x) => {
    const u1 = Math.random();
    const u2 = Math.random();
    const noise = noiseStd * Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
    return x + noise;
  });

  return { received, noisePower };
}

function bpskModulate(bits: number[]): number[] {
  return bits.map((b) => 2 * b - 1);
}

function bpskDemodulate(received: number[]): number[] {
  return received.map((r) => (r >= 0 ? 1 : 0));
}

function theoreticalBER_BPSK(snrDb: number): number {
  const snrLinear = Math.pow(10, snrDb / 10);
  return 0.5 * erfc(Math.sqrt(snrLinear));
}

function erfc(x: number): number {
  const t = 1 / (1 + 0.3275911 * Math.abs(x));
  const poly = t * (0.254829592 + t * (-0.284496736 + t * (1.421413741 + t * (-1.453152027 + t * 1.061405429))));
  const result = poly * Math.exp(-x * x);
  return x >= 0 ? result : 2 - result;
}

function simulateBPSK_AWGN(snrDb: number, nBits: number = 100000): number {
  const bits = Array.from({ length: nBits }, () => Math.round(Math.random()));
  const modulated = bpskModulate(bits);
  const { received } = awgnChannel(modulated, snrDb);
  const decoded = bpskDemodulate(received);

  let errors = 0;
  for (let i = 0; i < nBits; i++) {
    if (bits[i] !== decoded[i]) errors++;
  }
  return errors / nBits;
}

for (const snr of [0, 2, 4, 6, 8, 10]) {
  const simBer = simulateBPSK_AWGN(snr);
  const theoBer = theoreticalBER_BPSK(snr);
  console.log(`SNR=${snr}dB: 仿真BER=${simBer.toFixed(6)}, 理论BER=${theoBer.toFixed(6)}`);
}
AWGN 信道的容量为 C = ½ log₂(1 + SNR) bits/channel use。对于带宽为 W Hz 的 AWGN 信道，容量为 C = W log₂(1 + P/(N₀W)) bits/s，其中 P 是信号功率，N₀ 是噪声功率谱密度。


衰落信道与无线传播
无线通信中，信号经过多条路径到达接收端，产生多径效应。衰落信道用随机信道系数 h 建模：Y = hX + N。


function rayleighFading(nSamples: number): number[] {
  return Array.from({ length: nSamples }, () => {
    const u1 = Math.random();
    const u2 = Math.random();
    const real = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
    const imag = Math.sqrt(-2 * Math.log(u1)) * Math.sin(2 * Math.PI * u2);
    return Math.sqrt(real * real + imag * imag);
  });
}

function ricianFading(nSamples: number, kFactor: number): number[] {
  const los = Math.sqrt(kFactor / (kFactor + 1));
  const scatter = Math.sqrt(1 / (2 * (kFactor + 1)));

  return Array.from({ length: nSamples }, () => {
    const u1 = Math.random();
    const u2 = Math.random();
    const real = los + scatter * Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
    const imag = scatter * Math.sqrt(-2 * Math.log(u1)) * Math.sin(2 * Math.PI * u2);
    return Math.sqrt(real * real + imag * imag);
  });
}

function fadingChannelCapacity(snrDb: number, fadingGains: number[], csi: 'none' | 'receiver' | 'both'): number {
  const snr = Math.pow(10, snrDb / 10);
  const n = fadingGains.length;

  if (csi === 'receiver') {
    let sumCap = 0;
    for (const h of fadingGains) {
      sumCap += Math.log2(1 + snr * h * h);
    }
    return sumCap / n;
  }

  if (csi === 'none') {
    const avgGain = fadingGains.reduce((s, h) => s + h * h, 0) / n;
    return Math.log2(1 + snr * avgGain);
  }

  let totalCap = 0;
  const sorted = [...fadingGains].sort((a, b) => b - a);
  for (const h of sorted) {
    const gamma = snr * h * h;
    if (gamma > 1) {
      totalCap += Math.log2(gamma);
    }
  }
  return totalCap / n;
}

const rayleigh = rayleighFading(10000);
const rician = ricianFading(10000, 3);

console.log(`Rayleigh 平均增益: ${(rayleigh.reduce((s, h) => s + h * h, 0) / rayleigh.length).toFixed(4)}`);
console.log(`Rician(K=3) 平均增益: ${(rician.reduce((s, h) => s + h * h, 0) / rician.length).toFixed(4)}`);

const capRx = fadingChannelCapacity(10, rayleigh, 'receiver');
console.log(`Rayleigh 信道遍历容量(接收CSI): ${capRx.toFixed(4)} bits`);

小结
    

  • 信道模型是通信系统分析和设计的数学基础，将物理现象抽象为概率转移
    • 

  • BSC 和 BEC 是最基本的离散信道模型，分别描述翻转错误和擦除丢失
    • 

  • AWGN 是连续信道的标准模型，其容量公式 C = ½ log₂(1+SNR) 是通信理论的基石
    • 

  • 衰落信道通过随机增益建模多径效应，Rayleigh 和 Rician 是最常用的衰落模型
    • 

  • 信道状态信息（CSI）的可用性显著影响衰落信道的容量和系统设计策略
    • 

  • 模型选择需在精确性和可分析性之间取得平衡，过于复杂的模型反而阻碍设计
