率失真理论

核心概念

率失真理论（Rate-Distortion Theory）是有损压缩的信息论基础，由 Shannon 于 1959 年建立。它回答了一个核心问题：在允许一定失真 D 的条件下，最低的压缩码率 R 是多少？率失真函数 R(D) 定义了这个理论极限：R(D) = min_{p(x̂|x): E[d(x,x̂)]≤D} I(X;X̂)，其中 d(x,x̂) 是失真度量。

率失真理论揭示了压缩率与重构质量之间的根本权衡。在实际应用中，图像压缩（JPEG）、视频编码（H.264/H.265）和音频编码（AAC）都在追逼各自信源的率失真极限。

高斯信源的率失真
对于高斯信源 X ~ N(0, σ²)，在均方误差失真下，率失真函数有封闭形式：R(D) = max(0, ½ log₂(σ²/D))。这意味着将失真减半需要增加 0.5 bit 的码率。


function gaussianRD(variance: number, distortion: number): number {
  if (distortion >= variance) return 0;
  return 0.5 * Math.log2(variance / distortion);
}

function gaussianDR(variance: number, rate: number): number {
  if (rate <= 0) return variance;
  return variance * Math.pow(2, -2 * rate);
}

function bernoullRD(p: number, distortion: number): number {
  if (distortion >= Math.min(p, 1 - p)) return 0;
  const h = (x: number): number => {
    if (x <= 0 || x >= 1) return 0;
    return -x * Math.log2(x) - (1 - x) * Math.log2(1 - x);
  };
  return h(p) - h(distortion);
}

function vectorGaussianRD(variances: number[], totalRate: number): { allocations: number[]; distortions: number[] } {
  const n = variances.length;
  const sorted = variances.map((v, i) => ({ v, i })).sort((a, b) => b.v - a.v);

  let lambda =
    Math.pow(
      sorted.reduce((p, { v }) => p * v, 1),
      1 / n,
    ) * Math.pow(2, (-2 * totalRate) / n);

  const allocations = Array(n).fill(0);
  const distortions = Array(n).fill(0);

  for (const { v, i } of sorted) {
    const rate = Math.max(0, 0.5 * Math.log2(v / lambda));
    allocations[i] = rate;
    distortions[i] = rate > 0 ? lambda : v;
  }

  return { allocations, distortions };
}

const sigma2 = 4;
for (const D of [0.1, 0.5, 1, 2, 4]) {
  console.log(`高斯 σ²=${sigma2}, D=${D}: R(D)=${gaussianRD(sigma2, D).toFixed(4)} bits`);
}

for (const R of [0, 0.5, 1, 2, 3]) {
  console.log(`高斯 σ²=${sigma2}, R=${R}: D(R)=${gaussianDR(sigma2, R).toFixed(4)}`);
}

const variances = [8, 4, 2, 1];
const result = vectorGaussianRD(variances, 4);
console.log(
  `注水法: 分配=[${result.allocations.map((r) => r.toFixed(3))}], 失真=[${result.distortions.map((d) => d.toFixed(3))}]`,
);
量化与率失真逼近
标量量化和矢量量化是逼近率失真极限的实际方法。最优量化器设计遵循 Lloyd-Max 算法（标量）或 LBG 算法（矢量）。


function uniformQuantize(value: number, bits: number, min: number, max: number): { quantized: number; index: number } {
  const levels = 1 << bits;
  const step = (max - min) / levels;
  const index = Math.max(0, Math.min(levels - 1, Math.floor((value - min) / step)));
  const quantized = min + (index + 0.5) * step;
  return { quantized, index };
}

function lloydMaxQuantizer(
  samples: number[],
  levels: number,
  iterations: number = 100,
): { boundaries: number[]; centroids: number[] } {
  const sorted = [...samples].sort((a, b) => a - b);
  let centroids: number[] = Array.from(
    { length: levels },
    (_, i) => sorted[Math.floor(((i + 0.5) * sorted.length) / levels)],
  );

  for (let iter = 0; iter < iterations; iter++) {
    const boundaries: number[] = [];
    for (let i = 0; i < levels - 1; i++) {
      boundaries.push((centroids[i] + centroids[i + 1]) / 2);
    }

    const clusters: number[][] = Array.from({ length: levels }, () => []);
    for (const s of samples) {
      let idx = 0;
      for (let i = 0; i < boundaries.length; i++) {
        if (s > boundaries[i]) idx = i + 1;
      }
      clusters[idx].push(s);
    }

    centroids = clusters.map((cluster) =>
      cluster.length > 0 ? cluster.reduce((a, b) => a + b, 0) / cluster.length : centroids[0],
    );
  }

  const boundaries: number[] = [];
  for (let i = 0; i < levels - 1; i++) {
    boundaries.push((centroids[i] + centroids[i + 1]) / 2);
  }

  return { boundaries, centroids };
}

function computeDistortion(samples: number[], boundaries: number[], centroids: number[]): number {
  let totalDist = 0;
  for (const s of samples) {
    let idx = 0;
    for (let i = 0; i < boundaries.length; i++) {
      if (s > boundaries[i]) idx = i + 1;
    }
    const diff = s - centroids[idx];
    totalDist += diff * diff;
  }
  return totalDist / samples.length;
}

const gaussSamples = Array.from({ length: 10000 }, () => {
  const u1 = Math.random();
  const u2 = Math.random();
  return 2 * Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
});

for (const bits of [1, 2, 3, 4, 6, 8]) {
  const levels = 1 << bits;
  const { boundaries, centroids } = lloydMaxQuantizer(gaussSamples, levels);
  const dist = computeDistortion(gaussSamples, boundaries, centroids);
  const theoreticalDist = gaussianDR(4, bits);
  console.log(
    `${bits} bits: Lloyd-Max D=${dist.toFixed(4)}, 理论 D=${theoreticalDist.toFixed(4)}, 差距=${(10 * Math.log10(dist / theoreticalDist)).toFixed(2)} dB`,
  );
}

变换编码与率失真优化
实际的有损压缩系统通常采用变换编码：先将信号变换到频域（如 DCT、小波变换），然后对变换系数分别量化。变换编码的优势在于去相关，使各分量独立，从而可以独立应用率失真优化。


function dct1d(signal: number[]): number[] {
  const N = signal.length;
  const result: number[] = Array(N).fill(0);
  for (let k = 0; k < N; k++) {
    let sum = 0;
    for (let n = 0; n < N; n++) {
      sum += signal[n] * Math.cos((Math.PI * (2 * n + 1) * k) / (2 * N));
    }
    result[k] = sum * Math.sqrt(2 / N) * (k === 0 ? 1 / Math.sqrt(2) : 1);
  }
  return result;
}

function rdOptimalBitAllocation(variances: number[], totalBits: number): number[] {
  const n = variances.length;
  const logVar = variances.map((v) => Math.log2(Math.max(v, 1e-10)));
  const avgLogVar = logVar.reduce((a, b) => a + b, 0) / n;
  const bitsPerComponent = totalBits / n;

  return variances.map((_, i) => {
    const bits = bitsPerComponent + 0.5 * (logVar[i] - avgLogVar);
    return Math.max(0, bits);
  });
}

const signal = Array.from(
  { length: 8 },
  (_, i) => 100 + 50 * Math.cos((2 * Math.PI * i) / 8) + 20 * Math.cos((4 * Math.PI * i) / 8),
);

const dctCoeffs = dct1d(signal);
const coeffVar = dctCoeffs.map((c) => c * c);
const allocation = rdOptimalBitAllocation(coeffVar, 24);

console.log(`DCT 系数: [${dctCoeffs.map((c) => c.toFixed(2))}]`);
console.log(`系数方差: [${coeffVar.map((v) => v.toFixed(2))}]`);
console.log(`比特分配: [${allocation.map((b) => b.toFixed(2))}]`);
小结
    

  • 率失真函数 R(D) 定义了有损压缩的理论极限，是压缩率与质量的最优权衡
    • 

  • 高斯信源的率失真函数为 R(D) = ½ log₂(σ²/D)，每多 0.5 bit 失真减半
    • 

  • 多维信源使用注水法分配比特，高方差分量获得更多比特
    • 

  • Lloyd-Max 量化器通过迭代优化逼近最优标量量化，比均匀量化好约 0.5 dB
    • 

  • 变换编码通过去相关将信号分解为独立分量，使率失真优化可按分量独立进行
    • 

  • 现代视频编码（H.265/VVC）通过精细的率失真优化已非常接近理论极限
    •
