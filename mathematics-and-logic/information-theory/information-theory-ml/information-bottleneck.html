信息瓶颈

核心概念

信息瓶颈（Information Bottleneck, IB）方法由 Tishby 等人于 1999 年提出，是一种利用信息论原理进行表示学习的理论框架。其核心思想是寻找输入变量 X 的压缩表示 T，使得 T 保留尽可能多的关于目标变量 Y 的信息，同时丢弃与 Y 无关的冗余信息。

信息瓶颈优化目标：min_{p(t|x)} I(X;T) - β I(T;Y)，其中 β 是拉格朗日乘子，控制压缩与预测之间的权衡。当 β → ∞ 时，T 尽可能保留 Y 的信息；当 β → 0 时，T 被最大压缩。

IB 方法的求解算法
信息瓶颈问题可以通过迭代 Blahut-Arimoto 类型的算法求解。对于离散变量，交替优化编码分布 p(t|x)、边际分布 p(t) 和解码分布 p(y|t)。


function ibIteration(
  pXY: number[][],
  nT: number,
  beta: number,
  iterations: number = 100,
): {
  pTgivenX: number[][];
  iXT: number;
  iTY: number;
} {
  const nX = pXY.length;
  const nY = pXY[0].length;

  const pX: number[] = Array(nX).fill(0);
  for (let x = 0; x < nX; x++) {
    for (let y = 0; y < nY; y++) {
      pX[x] += pXY[x][y];
    }
  }

  let pTgivenX: number[][] = Array.from({ length: nX }, () => {
    const row = Array.from({ length: nT }, () => Math.random());
    const sum = row.reduce((a, b) => a + b, 0);
    return row.map((v) => v / sum);
  });

  for (let iter = 0; iter < iterations; iter++) {
    const pT: number[] = Array(nT).fill(0);
    for (let t = 0; t < nT; t++) {
      for (let x = 0; x < nX; x++) {
        pT[t] += pX[x] * pTgivenX[x][t];
      }
    }

    const pYgivenT: number[][] = Array.from({ length: nT }, () => Array(nY).fill(0));
    for (let t = 0; t < nT; t++) {
      if (pT[t] > 0) {
        for (let y = 0; y < nY; y++) {
          for (let x = 0; x < nX; x++) {
            pYgivenT[t][y] += (pXY[x][y] * pTgivenX[x][t]) / pT[t];
          }
        }
      }
    }

    const pYgivenX: number[][] = pXY.map((row, x) => row.map((p) => (pX[x] > 0 ? p / pX[x] : 0)));

    const newPTgivenX: number[][] = Array.from({ length: nX }, () => Array(nT).fill(0));
    for (let x = 0; x < nX; x++) {
      for (let t = 0; t < nT; t++) {
        let klDiv = 0;
        for (let y = 0; y < nY; y++) {
          if (pYgivenX[x][y] > 0 && pYgivenT[t][y] > 0) {
            klDiv += pYgivenX[x][y] * Math.log(pYgivenX[x][y] / pYgivenT[t][y]);
          }
        }
        newPTgivenX[x][t] = pT[t] * Math.exp(-beta * klDiv);
      }
      const sum = newPTgivenX[x].reduce((a, b) => a + b, 0);
      if (sum > 0) {
        newPTgivenX[x] = newPTgivenX[x].map((v) => v / sum);
      }
    }

    pTgivenX = newPTgivenX;
  }

  const pT: number[] = Array(nT).fill(0);
  for (let t = 0; t < nT; t++) {
    for (let x = 0; x < nX; x++) {
      pT[t] += pX[x] * pTgivenX[x][t];
    }
  }

  let iXT = 0;
  for (let x = 0; x < nX; x++) {
    for (let t = 0; t < nT; t++) {
      const joint = pX[x] * pTgivenX[x][t];
      if (joint > 0 && pT[t] > 0) {
        iXT += joint * Math.log2(pTgivenX[x][t] / pT[t]);
      }
    }
  }

  let iTY = 0;
  const pTY: number[][] = Array.from({ length: nT }, () => Array(nY).fill(0));
  for (let t = 0; t < nT; t++) {
    for (let x = 0; x < nX; x++) {
      for (let y = 0; y < nY; y++) {
        pTY[t][y] += pX[x] * pTgivenX[x][t] * (pX[x] > 0 ? pXY[x][y] / pX[x] : 0);
      }
    }
  }
  const pY: number[] = Array(nY).fill(0);
  for (let y = 0; y < nY; y++) {
    for (let t = 0; t < nT; t++) {
      pY[y] += pTY[t][y];
    }
  }
  for (let t = 0; t < nT; t++) {
    for (let y = 0; y < nY; y++) {
      if (pTY[t][y] > 0 && pT[t] > 0 && pY[y] > 0) {
        iTY += pTY[t][y] * Math.log2(pTY[t][y] / (pT[t] * pY[y]));
      }
    }
  }

  return { pTgivenX, iXT, iTY };
}

const pXY = [
  [0.2, 0.05],
  [0.05, 0.15],
  [0.15, 0.05],
  [0.05, 0.3],
];

for (const beta of [0.5, 1.0, 2.0, 5.0, 10.0]) {
  const result = ibIteration(pXY, 2, beta);
  console.log(`β=${beta}: I(X;T)=${result.iXT.toFixed(4)}, I(T;Y)=${result.iTY.toFixed(4)}`);
}
深度学习中的信息瓶颈
Shwartz-Ziv 和 Tishby 提出深度学习可以在信息平面上理解：深度网络的每一层构成一个马尔可夫链 X → T₁ → T₂ → ... → Tₗ → Y。训练过程分为两个阶段：拟合阶段（I(Tₗ;Y) 快速增长）和压缩阶段（I(X;Tₗ) 逐渐减小）。


function computeLayerMI(
  inputs: number[][],
  activations: number[][],
  labels: number[],
  bins: number = 30,
): { ixT: number; iTy: number } {
  const n = inputs.length;
  const inputDim = inputs[0].length;
  const actDim = activations[0].length;

  const discretizeVec = (vec: number[], nBins: number): number => {
    let hash = 0;
    for (const v of vec) {
      const bin = Math.min(Math.floor(((v + 5) / 10) * nBins), nBins - 1);
      hash = hash * nBins + Math.max(0, bin);
    }
    return hash;
  };

  const actBins = activations.map((a) => discretizeVec(a, bins));
  const inputBins = inputs.map((inp) => discretizeVec(inp, Math.min(bins, 10)));

  const computeMI = (x: number[], y: number[]): number => {
    const jointCounts = new Map<string, number>();
    const xCounts = new Map<number, number>();
    const yCounts = new Map<number, number>();

    for (let i = 0; i < n; i++) {
      const key = `${x[i]},${y[i]}`;
      jointCounts.set(key, (jointCounts.get(key) ?? 0) + 1);
      xCounts.set(x[i], (xCounts.get(x[i]) ?? 0) + 1);
      yCounts.set(y[i], (yCounts.get(y[i]) ?? 0) + 1);
    }

    let mi = 0;
    for (const [key, count] of jointCounts) {
      const [xi, yi] = key.split(',').map(Number);
      const pxy = count / n;
      const px = (xCounts.get(xi) ?? 0) / n;
      const py = (yCounts.get(yi) ?? 0) / n;
      if (pxy > 0 && px > 0 && py > 0) {
        mi += pxy * Math.log2(pxy / (px * py));
      }
    }
    return Math.max(0, mi);
  };

  return {
    ixT: computeMI(inputBins, actBins),
    iTy: computeMI(actBins, labels),
  };
}

const nSamples = 500;
const mockInputs = Array.from({ length: nSamples }, () => Array.from({ length: 2 }, () => Math.random() * 10 - 5));
const mockLabels = mockInputs.map((inp) => (inp[0] + inp[1] > 0 ? 1 : 0));
const mockActivations = mockInputs.map((inp) => [
  Math.tanh(inp[0] * 0.5 + inp[1] * 0.3),
  Math.tanh(-inp[0] * 0.2 + inp[1] * 0.7),
]);

const mi = computeLayerMI(mockInputs, mockActivations, mockLabels, 10);
console.log(`I(X;T) = ${mi.ixT.toFixed(4)} bits`);
console.log(`I(T;Y) = ${mi.iTy.toFixed(4)} bits`);

变分信息瓶颈
变分信息瓶颈（Variational Information Bottleneck, VIB）将信息瓶颈原理与变分推断结合，提供了一种可以端到端训练的深度学习方法。VIB 通过参数化编码器 pθ(t|x) 和解码器 pφ(y|t)，用变分下界近似 IB 目标。


function vibLoss(
  encoderMean: number[],
  encoderLogVar: number[],
  decoderLogits: number[],
  target: number,
  beta: number = 0.01,
): { totalLoss: number; reconLoss: number; klLoss: number } {
  const logProbs: number[] = [];
  const maxLogit = Math.max(...decoderLogits);
  const shifted = decoderLogits.map((l) => l - maxLogit);
  const logSumExp = Math.log(shifted.reduce((s, l) => s + Math.exp(l), 0));
  for (const l of shifted) {
    logProbs.push(l - logSumExp);
  }
  const reconLoss = -logProbs[target];

  let klLoss = 0;
  for (let i = 0; i < encoderMean.length; i++) {
    const mu = encoderMean[i];
    const logVar = encoderLogVar[i];
    klLoss += -0.5 * (1 + logVar - mu * mu - Math.exp(logVar));
  }

  const totalLoss = reconLoss + beta * klLoss;
  return { totalLoss, reconLoss, klLoss };
}

const mean = [0.5, -0.3, 0.8];
const logVar = [-1.0, -0.5, -1.5];
const decLogits = [2.0, 0.5, -1.0];
const target2 = 0;

for (const beta of [0.001, 0.01, 0.1, 1.0]) {
  const loss = vibLoss(mean, logVar, decLogits, target2, beta);
  console.log(
    `β=${beta}: total=${loss.totalLoss.toFixed(4)}, recon=${loss.reconLoss.toFixed(4)}, KL=${loss.klLoss.toFixed(4)}`,
  );
}
小结
    

  • 信息瓶颈寻找最优压缩表示，在压缩 I(X;T) 和预测 I(T;Y) 之间权衡
    • 

  • IB 迭代算法类似 Blahut-Arimoto，交替优化编码、解码和边际分布
    • 

  • 深度学习训练可以在信息平面上理解为先拟合后压缩的两阶段过程
    • 

  • 变分信息瓶颈（VIB）将 IB 原理与深度学习结合，实现端到端训练
    • 

  • β 参数控制正则化强度：大 β 保留更多信息，小 β 产生更紧凑的表示
    • 

  • 信息瓶颈理论为理解深度学习的泛化能力提供了信息论视角
