算术编码

核心概念

算术编码（Arithmetic Coding）是一种熵编码方法，通过将整个消息映射到 [0, 1) 区间内的一个子区间来实现压缩。与 Huffman 编码不同，算术编码不需要将每个符号映射到整数个比特，因此可以更精确地逼近信源熵，尤其是当符号概率高度不均匀时。

算术编码的核心思想：每个符号根据其概率划分当前区间，编码过程就是不断缩小区间的过程。最终用区间内最短的二进制分数表示整个消息。理论上，算术编码的平均码长可以达到 H(X) + 2/n 比特每符号，其中 n 是消息长度。

区间编码过程
算术编码的编码过程维护一个当前区间 [low, high)，每处理一个符号就根据该符号的概率缩小区间。


interface ArithmeticModel {
  symbols: string[];
  cumProbs: number[];
}

function createModel(symbols: string[], probs: number[]): ArithmeticModel {
  const cumProbs: number[] = [0];
  for (let i = 0; i < probs.length; i++) {
    cumProbs.push(cumProbs[i] + probs[i]);
  }
  return { symbols, cumProbs };
}

function arithmeticEncode(
  message: string,
  model: ArithmeticModel,
): {
  low: number;
  high: number;
  bits: number;
} {
  let low = 0;
  let high = 1;

  for (const ch of message) {
    const idx = model.symbols.indexOf(ch);
    if (idx === -1) throw new Error(`Unknown symbol: ${ch}`);

    const range = high - low;
    high = low + range * model.cumProbs[idx + 1];
    low = low + range * model.cumProbs[idx];
  }

  const bitsNeeded = Math.ceil(-Math.log2(high - low)) + 1;

  return { low, high, bits: bitsNeeded };
}

function arithmeticDecode(value: number, model: ArithmeticModel, length: number): string {
  let low = 0;
  let high = 1;
  let result = '';

  for (let i = 0; i < length; i++) {
    const range = high - low;
    const scaled = (value - low) / range;

    let idx = 0;
    for (let j = 0; j < model.symbols.length; j++) {
      if (scaled >= model.cumProbs[j] && scaled < model.cumProbs[j + 1]) {
        idx = j;
        break;
      }
    }

    result += model.symbols[idx];
    high = low + range * model.cumProbs[idx + 1];
    low = low + range * model.cumProbs[idx];
  }

  return result;
}

const model = createModel(['A', 'B', 'C'], [0.6, 0.3, 0.1]);
const msg = 'AABCA';
const encoded = arithmeticEncode(msg, model);
console.log(`消息: "${msg}"`);
console.log(`区间: [${encoded.low.toFixed(10)}, ${encoded.high.toFixed(10)})`);
console.log(`需要比特数: ${encoded.bits}`);

const midpoint = (encoded.low + encoded.high) / 2;
const decoded = arithmeticDecode(midpoint, model, msg.length);
console.log(`解码: "${decoded}"`);
console.log(`正确: ${decoded === msg}`);

const entropy = -[0.6, 0.3, 0.1].reduce((s, p) => s + p * Math.log2(p), 0);
console.log(`熵: ${entropy.toFixed(4)} bits/symbol`);
console.log(`实际: ${(encoded.bits / msg.length).toFixed(4)} bits/symbol`);
整数算术编码
实际实现中，浮点数精度不足以处理长消息。整数算术编码通过定点数运算和渐进式输出避免精度问题。


class IntegerArithmeticEncoder {
  private low: number = 0;
  private high: number = 0xffffffff;
  private pending: number = 0;
  private output: number[] = [];
  private model: ArithmeticModel;
  private precision: number = 32;

  constructor(model: ArithmeticModel) {
    this.model = model;
  }

  encode(symbol: string): void {
    const idx = this.model.symbols.indexOf(symbol);
    const range = this.high - this.low + 1;
    const totalFreq = 10000;
    const cumLow = Math.round(this.model.cumProbs[idx] * totalFreq);
    const cumHigh = Math.round(this.model.cumProbs[idx + 1] * totalFreq);

    this.high = this.low + Math.floor((range * cumHigh) / totalFreq) - 1;
    this.low = this.low + Math.floor((range * cumLow) / totalFreq);

    this.normalize();
  }

  private normalize(): void {
    const half = 1 << (this.precision - 1);
    const quarter = half >>> 1;

    while (true) {
      if (this.high < half) {
        this.outputBit(0);
        while (this.pending > 0) {
          this.output.push(1);
          this.pending--;
        }
      } else if (this.low >= half) {
        this.outputBit(1);
        while (this.pending > 0) {
          this.output.push(0);
          this.pending--;
        }
        this.low -= half;
        this.high -= half;
      } else if (this.low >= quarter && this.high < 3 * quarter) {
        this.pending++;
        this.low -= quarter;
        this.high -= quarter;
      } else {
        break;
      }

      this.low = this.low << 1;
      this.high = (this.high << 1) | 1;
      this.low &= 0xffffffff;
      this.high &= 0xffffffff;
    }
  }

  private outputBit(bit: number): void {
    this.output.push(bit);
  }

  finish(): number[] {
    this.pending++;
    if (this.low < 1 << (this.precision - 2)) {
      this.outputBit(0);
      while (this.pending > 0) {
        this.output.push(1);
        this.pending--;
      }
    } else {
      this.outputBit(1);
      while (this.pending > 0) {
        this.output.push(0);
        this.pending--;
      }
    }
    return this.output;
  }

  getOutput(): number[] {
    return [...this.output];
  }
}

const intModel = createModel(['A', 'B', 'C', 'D'], [0.5, 0.25, 0.15, 0.1]);
const encoder = new IntegerArithmeticEncoder(intModel);
const message = 'ABACDA';
for (const ch of message) {
  encoder.encode(ch);
}
const bits = encoder.finish();
console.log(`整数编码: ${bits.length} bits for "${message}"`);
console.log(`每符号: ${(bits.length / message.length).toFixed(4)} bits`);

自适应算术编码与上下文建模
自适应算术编码在编码过程中动态更新概率模型，无需预先传输概率表。结合上下文建模，可以利用符号间的统计相关性进一步提高压缩率。


class AdaptiveArithmeticModel {
  private counts: number[];
  private total: number;
  readonly symbols: string[];

  constructor(symbols: string[]) {
    this.symbols = symbols;
    this.counts = Array(symbols.length).fill(1);
    this.total = symbols.length;
  }

  getCumProbs(): number[] {
    const cumProbs: number[] = [0];
    for (let i = 0; i < this.counts.length; i++) {
      cumProbs.push(cumProbs[i] + this.counts[i] / this.total);
    }
    return cumProbs;
  }

  update(symbol: string): void {
    const idx = this.symbols.indexOf(symbol);
    if (idx >= 0) {
      this.counts[idx]++;
      this.total++;
    }
  }

  getModel(): ArithmeticModel {
    return {
      symbols: this.symbols,
      cumProbs: this.getCumProbs(),
    };
  }
}

class ContextModel {
  private contexts: Map<string, AdaptiveArithmeticModel>;
  private order: number;
  private symbols: string[];

  constructor(symbols: string[], order: number = 1) {
    this.symbols = symbols;
    this.order = order;
    this.contexts = new Map();
  }

  getModel(context: string): ArithmeticModel {
    const ctx = context.slice(-this.order);
    if (!this.contexts.has(ctx)) {
      this.contexts.set(ctx, new AdaptiveArithmeticModel(this.symbols));
    }
    return this.contexts.get(ctx)!.getModel();
  }

  update(context: string, symbol: string): void {
    const ctx = context.slice(-this.order);
    if (!this.contexts.has(ctx)) {
      this.contexts.set(ctx, new AdaptiveArithmeticModel(this.symbols));
    }
    this.contexts.get(ctx)!.update(symbol);
  }
}

function adaptiveCompress(text: string, symbols: string[]): number {
  const ctxModel = new ContextModel(symbols, 2);
  let totalBits = 0;

  for (let i = 0; i < text.length; i++) {
    const ctx = i >= 2 ? text.slice(i - 2, i) : text.slice(0, i);
    const model = ctxModel.getModel(ctx);
    const idx = model.symbols.indexOf(text[i]);
    const prob = model.cumProbs[idx + 1] - model.cumProbs[idx];
    totalBits += -Math.log2(prob);
    ctxModel.update(ctx, text[i]);
  }

  return totalBits;
}

const testText = 'ABABABABABCACACACBDBDBDBD';
const syms = ['A', 'B', 'C', 'D'];
const compressedBits = adaptiveCompress(testText, syms);
const naiveEntropy = Math.log2(syms.length) * testText.length;
console.log(`自适应压缩: ${compressedBits.toFixed(2)} bits`);
console.log(`等概率编码: ${naiveEntropy.toFixed(2)} bits`);
console.log(`压缩比: ${((compressedBits / naiveEntropy) * 100).toFixed(1)}%`);
小结
    

  • 算术编码将整个消息映射为区间内的一个数，突破了每符号至少 1 比特的限制
    • 

  • 整数算术编码通过定点运算和归一化解决浮点精度问题，适用于任意长度消息
    • 

  • 自适应算术编码结合上下文建模，可以利用符号间相关性实现更高压缩率
    • 

  • 算术编码的压缩率理论上可以达到 H(X) + 2/n，比 Huffman 编码更接近理论极限
    • 

  • 现代压缩标准（JPEG 2000、H.265、LZMA）广泛使用算术编码或其变体
    • 

  • ANS（Asymmetric Numeral Systems）是算术编码的现代替代，在 Zstandard 等工具中使用
    •
