最大似然估计

核心概念

最大似然估计（Maximum Likelihood Estimation, MLE）是参数统计中最重要的估计方法。其核心思想是：选择使观测数据出现概率最大的参数值。给定独立同分布样本 x₁,...,xₙ，似然函数 L(θ) = Π f(xᵢ|θ)，MLE 求解 θ̂ = argmax L(θ)。实际计算中取对数：ℓ(θ) = Σ log f(xᵢ|θ)，将乘积变为求和。

MLE 在机器学习中无处不在：逻辑回归的交叉熵损失就是负对数似然、朴素贝叶斯的参数学习是 MLE、神经网络的 softmax + cross-entropy 损失等价于类别分布的 MLE。理解 MLE 的原理和性质，是连接统计理论与机器学习实践的桥梁。

常见分布的解析 MLE
许多常见分布的 MLE 可以通过求导方程的解析解直接得到。


const normalMLE = (data: number[]): { mu: number; sigma2: number } => {
  const n = data.length;
  const mu = data.reduce((s, x) => s + x, 0) / n;
  const sigma2 = data.reduce((s, x) => s + (x - mu) ** 2, 0) / n;
  return { mu, sigma2 };
};

const exponentialMLE = (data: number[]): { lambda: number; logLik: number } => {
  const mean = data.reduce((s, x) => s + x, 0) / data.length;
  const lambda = 1 / mean;
  const logLik = data.length * Math.log(lambda) - lambda * data.reduce((s, x) => s + x, 0);
  return { lambda, logLik };
};

const poissonMLE = (data: number[]): { lambda: number; logLik: number } => {
  const lambda = data.reduce((s, x) => s + x, 0) / data.length;
  const logLik = data.reduce((s, x) => s + x * Math.log(lambda) - lambda - Math.log(factorial(x)), 0);
  return { lambda, logLik };
};

const factorial = (n: number): number => {
  let r = 1;
  for (let i = 2; i <= n; i++) r *= i;
  return r;
};

const uniformMLE = (data: number[]): { a: number; b: number } => ({
  a: Math.min(...data),
  b: Math.max(...data),
});

const gammaMLE = (data: number[], maxIter: number = 100): { alpha: number; beta: number } => {
  const n = data.length;
  const mean = data.reduce((s, x) => s + x, 0) / n;
  const logMean = data.reduce((s, x) => s + Math.log(x), 0) / n;
  let alpha = 0.5 / (Math.log(mean) - logMean);

  for (let i = 0; i < maxIter; i++) {
    const psiAlpha = digamma(alpha);
    const psiPrimeAlpha = trigamma(alpha);
    const update = (Math.log(alpha) - psiAlpha - Math.log(mean) + logMean) / (1 / alpha - psiPrimeAlpha);
    alpha = alpha - update;
    if (Math.abs(update) < 1e-10) break;
  }

  return { alpha, beta: alpha / mean };
};

const digamma = (x: number): number => {
  let result = 0;
  let xx = x;
  while (xx < 6) {
    result -= 1 / xx;
    xx += 1;
  }
  result += Math.log(xx) - 1 / (2 * xx);
  const xx2 = 1 / (xx * xx);
  result -= xx2 * (1 / 12 - xx2 * (1 / 120 - xx2 / 252));
  return result;
};

const trigamma = (x: number): number => {
  let result = 0;
  let xx = x;
  while (xx < 6) {
    result += 1 / (xx * xx);
    xx += 1;
  }
  result += 1 / xx + 1 / (2 * xx * xx);
  const xx2 = 1 / (xx * xx);
  result += (xx2 / xx) * (1 / 6 - xx2 * (1 / 30 - xx2 / 42));
  return result;
};

const normalData = Array.from({ length: 500 }, () => {
  const u1 = Math.random(),
    u2 = Math.random();
  return 10 + 3 * Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
});
const nMLE = normalMLE(normalData);
console.log(`正态 MLE: μ̂=${nMLE.mu.toFixed(3)}, σ̂²=${nMLE.sigma2.toFixed(3)}`);
注意正态分布 MLE 中 σ̂² 除以 n 而非 n-1，因此是有偏的（偏向偏小）。无偏估计 S² 除以 n-1，但 MSE 意义下 MLE 可能更优。


数值最大似然
当似然方程没有解析解时，需要数值优化。梯度上升（或等价地最小化负对数似然的梯度下降）是最常用的方法。


const numericalMLE = (
  negLogLik: (params: number[], data: number[]) => number,
  gradNegLogLik: (params: number[], data: number[]) => number[],
  data: number[],
  initial: number[],
  options: { lr?: number; maxIter?: number; tol?: number } = {},
): { params: number[]; logLik: number; iterations: number } => {
  const { lr = 0.01, maxIter = 5000, tol = 1e-8 } = options;
  let params = [...initial];

  for (let iter = 0; iter < maxIter; iter++) {
    const grad = gradNegLogLik(params, data);
    const gradNorm = Math.sqrt(grad.reduce((s, g) => s + g * g, 0));
    if (gradNorm < tol) return { params, logLik: -negLogLik(params, data), iterations: iter };
    params = params.map((p, i) => p - lr * grad[i]);
  }

  return { params, logLik: -negLogLik(params, data), iterations: maxIter };
};

const logisticNLL = (params: number[], data: number[]): number => {
  const [beta0, beta1] = params;
  let nll = 0;
  for (let i = 0; i < data.length; i += 2) {
    const x = data[i],
      y = data[i + 1];
    const logit = beta0 + beta1 * x;
    const logSigmoid = logit > 0 ? -Math.log(1 + Math.exp(-logit)) : logit - Math.log(1 + Math.exp(logit));
    const logOneMinusSigmoid = logit > 0 ? -logit - Math.log(1 + Math.exp(-logit)) : -Math.log(1 + Math.exp(logit));
    nll -= y * logSigmoid + (1 - y) * logOneMinusSigmoid;
  }
  return nll;
};

const logisticGradNLL = (params: number[], data: number[]): number[] => {
  const [beta0, beta1] = params;
  let g0 = 0,
    g1 = 0;
  for (let i = 0; i < data.length; i += 2) {
    const x = data[i],
      y = data[i + 1];
    const p = 1 / (1 + Math.exp(-(beta0 + beta1 * x)));
    g0 += p - y;
    g1 += (p - y) * x;
  }
  return [g0, g1];
};

const logisticData: number[] = [];
for (let i = 0; i < 200; i++) {
  const x = (Math.random() - 0.5) * 6;
  const p = 1 / (1 + Math.exp(-(1.5 + 2 * x)));
  const y = Math.random() < p ? 1 : 0;
  logisticData.push(x, y);
}
const logisticResult = numericalMLE(logisticNLL, logisticGradNLL, logisticData, [0, 0], { lr: 0.01 });
console.log(`逻辑回归 MLE: β₀=${logisticResult.params[0].toFixed(3)}, β₁=${logisticResult.params[1].toFixed(3)}`);
EM 算法
当数据包含缺失值或隐变量时，直接最大化似然函数困难。EM 算法通过交替执行 E 步（计算隐变量的期望）和 M 步（最大化完整数据似然）来迭代求解。EM 保证每步似然不减，收敛到局部最大值。


const gaussianMixtureEM = (
  data: number[],
  K: number,
  maxIter: number = 100,
): { weights: number[]; means: number[]; variances: number[] } => {
  const n = data.length;
  let weights = Array(K).fill(1 / K);
  let means = Array.from({ length: K }, (_, k) => data[Math.floor(Math.random() * n)]);
  let variances = Array(K).fill(1);

  for (let iter = 0; iter < maxIter; iter++) {
    const resp = Array.from({ length: n }, () => Array(K).fill(0));
    for (let i = 0; i < n; i++) {
      let total = 0;
      for (let k = 0; k < K; k++) {
        const diff = data[i] - means[k];
        resp[i][k] = (weights[k] * Math.exp((-0.5 * diff * diff) / variances[k])) / Math.sqrt(variances[k]);
        total += resp[i][k];
      }
      for (let k = 0; k < K; k++) resp[i][k] /= total;
    }

    for (let k = 0; k < K; k++) {
      const nk = resp.reduce((s, r) => s + r[k], 0);
      weights[k] = nk / n;
      means[k] = resp.reduce((s, r, i) => s + r[k] * data[i], 0) / nk;
      variances[k] = resp.reduce((s, r, i) => s + r[k] * (data[i] - means[k]) ** 2, 0) / nk;
      variances[k] = Math.max(variances[k], 1e-6);
    }
  }

  return { weights, means, variances };
};

const mixData: number[] = [];
for (let i = 0; i < 300; i++) {
  const u1 = Math.random(),
    u2 = Math.random();
  const z = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
  mixData.push(Math.random() < 0.4 ? -2 + z : 3 + 0.5 * z);
}
const em = gaussianMixtureEM(mixData, 2);
console.log(`GMM: μ=[${em.means.map((m) => m.toFixed(2))}], w=[${em.weights.map((w) => w.toFixed(2))}]`);
似然比与模型选择
似然比检验比较嵌套模型的拟合优度：Λ = -2[ℓ(θ₀) - ℓ(θ̂)]，渐近服从 χ² 分布。AIC = -2ℓ(θ̂) + 2k 和 BIC = -2ℓ(θ̂) + k·ln(n) 在模型选择中平衡拟合度和复杂度。


const aic = (logLik: number, k: number): number => -2 * logLik + 2 * k;
const bic = (logLik: number, k: number, n: number): number => -2 * logLik + k * Math.log(n);

const models = [
  { name: 'Exponential', k: 1, logLik: -450 },
  { name: 'Gamma', k: 2, logLik: -440 },
  { name: 'LogNormal', k: 2, logLik: -438 },
  { name: 'Weibull', k: 2, logLik: -435 },
];

const n = 200;
for (const m of models) {
  console.log(`${m.name}: AIC=${aic(m.logLik, m.k).toFixed(1)}, BIC=${bic(m.logLik, m.k, n).toFixed(1)}`);
}
小结
    

  • MLE 选择使数据出现概率最大的参数值，是最广泛使用的估计方法
    • 

  • MLE 在正则条件下具有一致性、渐近正态性和渐近有效性
    • 

  • 逻辑回归、朴素贝叶斯、GMM 的训练本质上都是 MLE
    • 

  • EM 算法处理含隐变量的 MLE 问题，保证似然单调不减
    • 

  • AIC 和 BIC 在拟合度与复杂度之间权衡，用于模型选择
    • 

  • MLE 的不变性使参数变换（如 σ² → σ）时无需重新估计
