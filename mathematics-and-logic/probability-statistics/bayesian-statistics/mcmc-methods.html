马尔可夫链蒙特卡洛方法

核心概念

马尔可夫链蒙特卡洛（Markov Chain Monte Carlo, MCMC）是贝叶斯推断的计算引擎。当后验分布没有解析解时（非共轭模型、复杂层次模型），MCMC 通过构造一条以后验分布为平稳分布的马尔可夫链，从中采样以近似后验分布。MCMC 的出现使得贝叶斯方法从理论走向实用，能够处理任意复杂的概率模型。

MCMC 的核心思想：构造一条马尔可夫链 θ₁, θ₂, ..., θₜ，使得其平稳分布恰好是目标后验分布 P(θ|X)。经过充分的"燃烧期"（Burn-in）后，链上的样本可以视为后验分布的近似样本。

Metropolis-Hastings 算法
Metropolis-Hastings (MH) 算法是最基本的 MCMC 算法。从当前状态 θ 出发，按提议分布 q(θ'|θ) 生成候选状态 θ'，然后以接受概率 α = min(1, [P(θ'|X)q(θ|θ')] / [P(θ|X)q(θ'|θ)]) 决定是否移动到新状态。


关键洞察：计算接受概率时，后验分布中的归一化常数（边际似然）被约去了，因此只需要知道后验分布的核（未归一化密度）即可。


interface MCMCSample {
  chain: number[][];
  acceptanceRate: number;
  logPosteriors: number[];
}

const metropolisHastings = (
  logPosterior: (theta: number[]) => number,
  initialTheta: number[],
  proposalSD: number[],
  iterations: number,
  burnIn: number,
): MCMCSample => {
  const dim = initialTheta.length;
  const chain: number[][] = [];
  const logPosts: number[] = [];
  let theta = [...initialTheta];
  let logPost = logPosterior(theta);
  let accepted = 0;

  for (let i = 0; i < iterations + burnIn; i++) {
    const thetaPrime = theta.map((t, d) => t + proposalSD[d] * gaussianRandom());
    const logPostPrime = logPosterior(thetaPrime);
    const logAlpha = logPostPrime - logPost;

    if (Math.log(Math.random()) < logAlpha) {
      theta = thetaPrime;
      logPost = logPostPrime;
      accepted++;
    }

    if (i >= burnIn) {
      chain.push([...theta]);
      logPosts.push(logPost);
    }
  }

  return {
    chain,
    acceptanceRate: accepted / (iterations + burnIn),
    logPosteriors: logPosts,
  };
};

const adaptiveMetropolis = (
  logPosterior: (theta: number[]) => number,
  initialTheta: number[],
  iterations: number,
  burnIn: number,
  adaptInterval: number = 100,
): MCMCSample => {
  const dim = initialTheta.length;
  let proposalSD = new Array(dim).fill(1);
  const chain: number[][] = [];
  const logPosts: number[] = [];
  let theta = [...initialTheta];
  let logPost = logPosterior(theta);
  let accepted = 0;
  let intervalAccepted = 0;

  for (let i = 0; i < iterations + burnIn; i++) {
    const thetaPrime = theta.map((t, d) => t + proposalSD[d] * gaussianRandom());
    const logPostPrime = logPosterior(thetaPrime);
    const logAlpha = logPostPrime - logPost;

    if (Math.log(Math.random()) < logAlpha) {
      theta = thetaPrime;
      logPost = logPostPrime;
      accepted++;
      intervalAccepted++;
    }

    if (i > 0 && i % adaptInterval === 0) {
      const rate = intervalAccepted / adaptInterval;
      const factor = rate > 0.44 ? 1.1 : rate < 0.23 ? 0.9 : 1.0;
      proposalSD = proposalSD.map((s) => s * factor);
      intervalAccepted = 0;
    }

    if (i >= burnIn) {
      chain.push([...theta]);
      logPosts.push(logPost);
    }
  }

  return {
    chain,
    acceptanceRate: accepted / (iterations + burnIn),
    logPosteriors: logPosts,
  };
};

const gaussianRandom = (): number => {
  const u1 = Math.random();
  const u2 = Math.random();
  return Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
};
Gibbs 采样
Gibbs 采样是 MH 算法的特例，适用于多维参数可以分块、且每个条件分布可以直接采样的情形。在每一步中，依次固定其他参数，从当前参数的条件后验分布中采样。由于每步都是从精确的条件分布采样，接受率为 100%。


const gibbsSamplerNormalMixture = (
  data: number[],
  K: number,
  iterations: number,
  burnIn: number,
): { means: number[][]; variances: number[][]; weights: number[][] } => {
  const n = data.length;
  let means = Array.from({ length: K }, (_, k) => data[Math.floor(Math.random() * n)]);
  let variances = new Array(K).fill(1);
  let weights = new Array(K).fill(1 / K);
  const meansChain: number[][] = [];
  const variancesChain: number[][] = [];
  const weightsChain: number[][] = [];

  for (let iter = 0; iter < iterations + burnIn; iter++) {
    const assignments = data.map((x) => {
      const logProbs = means.map((mu, k) => {
        const sd = Math.sqrt(variances[k]);
        return Math.log(weights[k]) - 0.5 * ((x - mu) / sd) ** 2 - Math.log(sd);
      });
      const maxLP = Math.max(...logProbs);
      const probs = logProbs.map((lp) => Math.exp(lp - maxLP));
      const sum = probs.reduce((a, b) => a + b, 0);
      const normalized = probs.map((p) => p / sum);
      const u = Math.random();
      let cumSum = 0;
      for (let k = 0; k < K; k++) {
        cumSum += normalized[k];
        if (u <= cumSum) return k;
      }
      return K - 1;
    });

    for (let k = 0; k < K; k++) {
      const clusterData = data.filter((_, i) => assignments[i] === k);
      const nk = clusterData.length;
      if (nk === 0) continue;
      const xbar = clusterData.reduce((a, b) => a + b, 0) / nk;
      const priorPrec = 0.01;
      const dataPrec = nk / variances[k];
      means[k] =
        (priorPrec * 0 + dataPrec * xbar) / (priorPrec + dataPrec) + gaussianRandom() / Math.sqrt(priorPrec + dataPrec);
      const ss = clusterData.reduce((a, x) => a + (x - means[k]) ** 2, 0);
      variances[k] = (1 + ss) / gammaRandom((1 + nk) / 2);
      weights[k] = nk;
    }
    const wSum = weights.reduce((a, b) => a + b, 0);
    weights = weights.map((w) => w / wSum);

    if (iter >= burnIn) {
      meansChain.push([...means]);
      variancesChain.push([...variances]);
      weightsChain.push([...weights]);
    }
  }

  return { means: meansChain, variances: variancesChain, weights: weightsChain };
};

const gammaRandom = (shape: number): number => {
  if (shape < 1) return gammaRandom(shape + 1) * Math.random() ** (1 / shape);
  const d = shape - 1 / 3;
  const c = 1 / Math.sqrt(9 * d);
  while (true) {
    let x: number, v: number;
    do {
      x = gaussianRandom();
      v = 1 + c * x;
    } while (v <= 0);
    v = v ** 3;
    const u = Math.random();
    if (u < 1 - 0.0331 * x ** 4) return d * v;
    if (Math.log(u) < 0.5 * x * x + d * (1 - v + Math.log(v))) return d * v;
  }
};
收敛诊断
MCMC 的一个关键挑战是判断链是否已经收敛到平稳分布。常用的收敛诊断方法包括轨迹图目视检查、Gelman-Rubin 统计量（R̂）、有效样本量（ESS）等。


const effectiveSampleSize = (chain: number[]): number => {
  const n = chain.length;
  const mean = chain.reduce((a, b) => a + b, 0) / n;
  const variance = chain.reduce((a, x) => a + (x - mean) ** 2, 0) / n;
  if (variance === 0) return n;

  let rhoSum = 0;
  for (let lag = 1; lag < n; lag++) {
    let autoCorr = 0;
    for (let i = 0; i < n - lag; i++) {
      autoCorr += (chain[i] - mean) * (chain[i + lag] - mean);
    }
    autoCorr /= (n - lag) * variance;
    if (autoCorr < 0.05) break;
    rhoSum += autoCorr;
  }

  return Math.max(1, n / (1 + 2 * rhoSum));
};

const gelmanRubin = (chains: number[][]): number => {
  const m = chains.length;
  const n = chains[0].length;
  const chainMeans = chains.map((c) => c.reduce((a, b) => a + b, 0) / n);
  const overallMean = chainMeans.reduce((a, b) => a + b, 0) / m;
  const B = (n / (m - 1)) * chainMeans.reduce((a, mu) => a + (mu - overallMean) ** 2, 0);
  const W =
    chains.reduce((a, chain, i) => {
      const mu = chainMeans[i];
      return a + chain.reduce((s, x) => s + (x - mu) ** 2, 0) / (n - 1);
    }, 0) / m;
  const varPlus = ((n - 1) / n) * W + (1 / n) * B;
  return Math.sqrt(varPlus / W);
};

interface MCMCDiagnostics {
  ess: number;
  rhat: number;
  acceptRate: number;
  converged: boolean;
}

const diagnoseChain = (chains: number[][][], paramIndex: number): MCMCDiagnostics => {
  const paramChains = chains.map((c) => c.map((s) => s[paramIndex]));
  const ess = Math.min(...paramChains.map((c) => effectiveSampleSize(c)));
  const rhat = gelmanRubin(paramChains);
  return {
    ess,
    rhat,
    acceptRate: 0,
    converged: rhat < 1.01 && ess > 400,
  };
};
哈密顿蒙特卡洛
哈密顿蒙特卡洛（Hamiltonian Monte Carlo, HMC）通过引入物理学中的哈密顿动力学，利用后验分布的梯度信息来高效探索参数空间。HMC 在每一步中模拟一个"粒子"在后验分布对应的势能面上的运动，从而产生远距离的提议点，同时保持高接受率。


const hmc = (
  logPosterior: (theta: number[]) => number,
  gradLogPosterior: (theta: number[]) => number[],
  initialTheta: number[],
  stepSize: number,
  numLeapfrog: number,
  iterations: number,
  burnIn: number,
): MCMCSample => {
  const dim = initialTheta.length;
  const chain: number[][] = [];
  const logPosts: number[] = [];
  let theta = [...initialTheta];
  let logPost = logPosterior(theta);
  let accepted = 0;

  for (let i = 0; i < iterations + burnIn; i++) {
    let p = Array.from({ length: dim }, () => gaussianRandom());
    const currentH = -logPost + 0.5 * p.reduce((a, pi) => a + pi * pi, 0);
    let thetaPrime = [...theta];
    let pPrime = [...p];
    let grad = gradLogPosterior(thetaPrime);
    pPrime = pPrime.map((pi, d) => pi + 0.5 * stepSize * grad[d]);
    for (let l = 0; l < numLeapfrog - 1; l++) {
      thetaPrime = thetaPrime.map((t, d) => t + stepSize * pPrime[d]);
      grad = gradLogPosterior(thetaPrime);
      pPrime = pPrime.map((pi, d) => pi + stepSize * grad[d]);
    }
    thetaPrime = thetaPrime.map((t, d) => t + stepSize * pPrime[d]);
    grad = gradLogPosterior(thetaPrime);
    pPrime = pPrime.map((pi, d) => pi + 0.5 * stepSize * grad[d]);

    const logPostPrime = logPosterior(thetaPrime);
    const proposedH = -logPostPrime + 0.5 * pPrime.reduce((a, pi) => a + pi * pi, 0);

    if (Math.log(Math.random()) < currentH - proposedH) {
      theta = thetaPrime;
      logPost = logPostPrime;
      accepted++;
    }

    if (i >= burnIn) {
      chain.push([...theta]);
      logPosts.push(logPost);
    }
  }

  return {
    chain,
    acceptanceRate: accepted / (iterations + burnIn),
    logPosteriors: logPosts,
  };
};
小结
    

  • MCMC 通过构造马尔可夫链来近似采样后验分布，是非共轭贝叶斯模型的计算基础
    • 

  • Metropolis-Hastings 是最通用的 MCMC 算法，自适应版本自动调节提议分布
    • 

  • Gibbs 采样利用条件分布实现 100% 接受率，适合层次模型
    • 

  • HMC 利用梯度信息在高维空间中高效探索，是现代贝叶斯计算的首选
    • 

  • 收敛诊断（R̂ < 1.01, ESS > 400）是保证 MCMC 结果可靠性的必要步骤
    •
