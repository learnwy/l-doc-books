先验分布与后验更新

核心概念

先验分布（Prior Distribution）和后验更新（Posterior Update）是贝叶斯推断的两个核心环节。先验分布编码了在观测数据之前对参数的信念，后验更新则通过贝叶斯定理将先验与数据证据结合，产生更新后的信念。这一过程可以迭代进行——上一轮的后验成为下一轮的先验，实现持续学习。

先验的选择既是贝叶斯方法的最大优势（可以融合领域知识），也是最大争议来源（主观性）。现代贝叶斯实践发展出了一套系统化的先验选择方法论，在灵活性和客观性之间取得平衡。

共轭先验体系
共轭先验是贝叶斯分析中最优雅的构造。如果先验分布 P(θ) 与似然函数 P(X|θ) 的乘积仍属于与先验相同的分布族，则称该先验为似然的共轭先验。共轭性保证了后验更新可以通过简单的参数变换完成，无需数值积分。


interface ConjugateFamily<Params> {
  priorParams: Params;
  update: (data: number[], params: Params) => Params;
  posteriorMean: (params: Params) => number;
  posteriorVariance: (params: Params) => number;
  posteriorMode: (params: Params) => number;
  predictive: (params: Params, x: number) => number;
}

const betaBernoulli: ConjugateFamily<{ alpha: number; beta: number }> = {
  priorParams: { alpha: 1, beta: 1 },
  update: (data, { alpha, beta }) => ({
    alpha: alpha + data.filter((x) => x === 1).length,
    beta: beta + data.filter((x) => x === 0).length,
  }),
  posteriorMean: ({ alpha, beta }) => alpha / (alpha + beta),
  posteriorVariance: ({ alpha, beta }) => (alpha * beta) / ((alpha + beta) ** 2 * (alpha + beta + 1)),
  posteriorMode: ({ alpha, beta }) =>
    alpha > 1 && beta > 1 ? (alpha - 1) / (alpha + beta - 2) : alpha / (alpha + beta),
  predictive: ({ alpha, beta }) => alpha / (alpha + beta),
};

const normalNormal: ConjugateFamily<{ mu: number; tau2: number }> = {
  priorParams: { mu: 0, tau2: 100 },
  update: (data, { mu: mu0, tau2 }) => {
    const sigma2 = 1;
    const n = data.length;
    const xbar = data.reduce((a, b) => a + b, 0) / n;
    const prec0 = 1 / tau2;
    const precData = n / sigma2;
    const postPrec = prec0 + precData;
    return {
      mu: (prec0 * mu0 + precData * xbar) / postPrec,
      tau2: 1 / postPrec,
    };
  },
  posteriorMean: ({ mu }) => mu,
  posteriorVariance: ({ tau2 }) => tau2,
  posteriorMode: ({ mu }) => mu,
  predictive: ({ mu, tau2 }) => mu,
};

const gammaPoisson: ConjugateFamily<{ alpha: number; beta: number }> = {
  priorParams: { alpha: 1, beta: 1 },
  update: (data, { alpha, beta }) => ({
    alpha: alpha + data.reduce((a, b) => a + b, 0),
    beta: beta + data.length,
  }),
  posteriorMean: ({ alpha, beta }) => alpha / beta,
  posteriorVariance: ({ alpha, beta }) => alpha / (beta * beta),
  posteriorMode: ({ alpha, beta }) => (alpha >= 1 ? (alpha - 1) / beta : 0),
  predictive: ({ alpha, beta }, x) => {
    const p = beta / (beta + 1);
    return Math.exp(lgamma(alpha + x) - lgamma(alpha) - lgamma(x + 1) + alpha * Math.log(p) + x * Math.log(1 - p));
  },
};

const lgamma = (x: number): number => {
  const c = [
    76.18009172947146, -86.50532032941677, 24.01409824083091, -1.231739572450155, 0.1208650973866179e-2,
    -0.5395239384953e-5,
  ];
  let y = x,
    tmp = x + 5.5;
  tmp -= (x + 0.5) * Math.log(tmp);
  let ser = 1.000000000190015;
  for (let j = 0; j < 6; j++) ser += c[j] / ++y;
  return -tmp + Math.log((2.5066282746310005 * ser) / x);
};
无信息与弱信息先验
无信息先验（Non-informative Prior）的目标是让后验主要由数据决定。经典的无信息先验包括均匀先验和 Jeffreys 先验。


Jeffreys 先验 π(θ) ∝ √det(I(θ)) 基于 Fisher 信息矩阵，具有参数变换不变性——对 θ 的 Jeffreys 先验经过变换后自动成为 g(θ) 的 Jeffreys 先验。


弱信息先验（Weakly Informative Prior）在现代贝叶斯实践中更受推荐，由 Andrew Gelman 等人倡导。其核心思想是：先验不需要精确，但应排除明显不合理的参数值，提供适度的正则化。

const jeffreysPrior = (
  distribution: 'normal-mean' | 'normal-variance' | 'binomial' | 'poisson',
): ((theta: number) => number) => {
  switch (distribution) {
    case 'normal-mean':
      return () => 1;
    case 'normal-variance':
      return (sigma2) => 1 / sigma2;
    case 'binomial':
      return (p) => 1 / Math.sqrt(p * (1 - p));
    case 'poisson':
      return (lambda) => 1 / Math.sqrt(lambda);
  }
};

const weaklyInformativePriors = {
  locationParam: (dataSD: number) => ({
    distribution: 'Normal',
    mean: 0,

    distribution: 'Half-Cauchy',

    distribution: 'Beta',
    alpha,
    beta,
    logDensity: (p: number) => (p > 0 && p < 1 ? (alpha - 1) * Math.log(p) + (beta - 1) * Math.log(1 - p) : -Infinity),
  }),
};
序贯更新与在线学习
贝叶斯框架天然支持序贯更新（Sequential Update）：观测到新数据后，将当前后验作为新的先验，与新数据结合得到更新的后验。这种增量学习方式在流式数据处理和在线系统中极为实用。


对于共轭模型，序贯更新只需更新超参数，计算成本恒定（与历史数据量无关）。这是贝叶斯方法在实时系统中的核心优势。


class OnlineBayesianEstimator {
  private alpha: number;
  private beta: number;
  private observations: number = 0;

  constructor(priorAlpha: number = 1, priorBeta: number = 1) {
    this.alpha = priorAlpha;
    this.beta = priorBeta;
  }

  update(success: boolean): void {
    if (success) this.alpha += 1;
    else this.beta += 1;
    this.observations += 1;
  }

  batchUpdate(successes: number, failures: number): void {
    this.alpha += successes;
    this.beta += failures;
    this.observations += successes + failures;
  }

  getMean(): number {
    return this.alpha / (this.alpha + this.beta);
  }

  getCI(credMass: number = 0.95): [number, number] {
    const a = this.alpha,
      b = this.beta;
    const mean = a / (a + b);
    const variance = (a * b) / ((a + b) ** 2 * (a + b + 1));
    const z = credMass === 0.95 ? 1.96 : 2.576;
    const sd = Math.sqrt(variance);
    return [Math.max(0, mean - z * sd), Math.min(1, mean + z * sd)];
  }

  thompsonSample(): number {
    return betaSample(this.alpha, this.beta);
  }

  getState(): { alpha: number; beta: number; n: number } {
    return { alpha: this.alpha, beta: this.beta, n: this.observations };
  }
}

const betaSample = (alpha: number, beta: number): number => {
  const x = gammaSample(alpha);
  const y = gammaSample(beta);
  return x / (x + y);
};

const gammaSample = (shape: number): number => {
  if (shape < 1) return gammaSample(shape + 1) * Math.random() ** (1 / shape);
  const d = shape - 1 / 3;
  const c = 1 / Math.sqrt(9 * d);
  while (true) {
    let x: number, v: number;
    do {
      x = gaussianRandom();
      v = 1 + c * x;
    } while (v <= 0);
    v = v ** 3;
    const u = Math.random();
    if (u < 1 - 0.0331 * x ** 4) return d * v;
    if (Math.log(u) < 0.5 * x * x + d * (1 - v + Math.log(v))) return d * v;
  }
};

const gaussianRandom = (): number => {
  const u1 = Math.random();
  const u2 = Math.random();
  return Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
};

class ThompsonSamplingBandit {
  private arms: OnlineBayesianEstimator[];

  constructor(numArms: number) {
    this.arms = Array.from({ length: numArms }, () => new OnlineBayesianEstimator(1, 1));
  }

  selectArm(): number {
    const samples = this.arms.map((arm) => arm.thompsonSample());
    return samples.indexOf(Math.max(...samples));
  }

  updateArm(armIndex: number, reward: boolean): void {
    this.arms[armIndex].update(reward);
  }

  getStats(): Array<{ mean: number; ci: [number, number]; n: number }> {
    return this.arms.map((arm) => ({
      mean: arm.getMean(),
      ci: arm.getCI(),
      n: arm.getState().n,
    }));
  }
}
先验敏感性分析
先验敏感性分析（Prior Sensitivity Analysis）是贝叶斯实践中的重要步骤，检验推断结论对先验选择的依赖程度。如果结论对先验高度敏感，说明数据信息不足以主导推断，需要更多数据或更仔细地论证先验。


const priorSensitivityAnalysis = (
  data: number[],
  priorSpecs: Array<{ name: string; alpha: number; beta: number }>,
): Array<{ prior: string; posteriorMean: number; ci: [number, number] }> => {
  const successes = data.filter((x) => x === 1).length;
  const failures = data.length - successes;

  return priorSpecs.map(({ name, alpha, beta }) => {
    const postAlpha = alpha + successes;
    const postBeta = beta + failures;
    const mean = postAlpha / (postAlpha + postBeta);
    const variance = (postAlpha * postBeta) / ((postAlpha + postBeta) ** 2 * (postAlpha + postBeta + 1));
    const sd = Math.sqrt(variance);
    return {
      prior: name,
      posteriorMean: mean,
      ci: [Math.max(0, mean - 1.96 * sd), Math.min(1, mean + 1.96 * sd)] as [number, number],
    };
  });
};
