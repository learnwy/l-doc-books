贝叶斯方法实际应用

核心概念

贝叶斯方法在现代工程和数据科学中有广泛的实际应用。从推荐系统的冷启动到 A/B 测试的提前终止，从异常检测到超参数优化，贝叶斯思维提供了一种系统化地处理不确定性的框架。相比频率学派方法，贝叶斯方法的优势在实际系统中尤为突出：自然的在线更新、小样本下的稳健推断、以及对不确定性的精确量化。

贝叶斯 A/B 测试
贝叶斯 A/B 测试相比传统频率学派的假设检验有三大优势：（1）可以在实验过程中随时查看结果而不膨胀假阳性率；（2）直接给出"B 优于 A 的概率"，解释更直观；（3）可以设置实际意义阈值（ROPE），区分统计显著和实际显著。


class BayesianABTest {
  private controlAlpha: number;
  private controlBeta: number;
  private treatmentAlpha: number;
  private treatmentBeta: number;

  constructor(priorAlpha: number = 1, priorBeta: number = 1) {
    this.controlAlpha = priorAlpha;
    this.controlBeta = priorBeta;
    this.treatmentAlpha = priorAlpha;
    this.treatmentBeta = priorBeta;
  }

  addControlData(successes: number, failures: number): void {
    this.controlAlpha += successes;
    this.controlBeta += failures;
  }

  addTreatmentData(successes: number, failures: number): void {
    this.treatmentAlpha += successes;
    this.treatmentBeta += failures;
  }

  probabilityBBeatsA(simulations: number = 50000): number {
    let wins = 0;
    for (let i = 0; i < simulations; i++) {
      const a = betaSample(this.controlAlpha, this.controlBeta);
      const b = betaSample(this.treatmentAlpha, this.treatmentBeta);
      if (b > a) wins++;
    }
    return wins / simulations;
  }

  expectedLoss(simulations: number = 50000): { lossA: number; lossB: number } {
    let lossA = 0,
      lossB = 0;
    for (let i = 0; i < simulations; i++) {
      const a = betaSample(this.controlAlpha, this.controlBeta);
      const b = betaSample(this.treatmentAlpha, this.treatmentBeta);
      lossA += Math.max(0, b - a);
      lossB += Math.max(0, a - b);
    }
    return { lossA: lossA / simulations, lossB: lossB / simulations };
  }

  credibleInterval(variant: 'control' | 'treatment', mass: number = 0.95): [number, number] {
    const [alpha, beta] =
      variant === 'control' ? [this.controlAlpha, this.controlBeta] : [this.treatmentAlpha, this.treatmentBeta];
    const samples = Array.from({ length: 10000 }, () => betaSample(alpha, beta));
    samples.sort((a, b) => a - b);
    const lo = Math.floor(((1 - mass) / 2) * 10000);
    const hi = Math.floor((1 - (1 - mass) / 2) * 10000);
    return [samples[lo], samples[hi]];
  }

  shouldStop(threshold: number = 0.95): {
    stop: boolean;
    winner: 'control' | 'treatment' | 'undecided';
    probability: number;
  } {
    const pBWins = this.probabilityBBeatsA();
    if (pBWins >= threshold) {
      return { stop: true, winner: 'treatment', probability: pBWins };
    }
    if (1 - pBWins >= threshold) {
      return { stop: true, winner: 'control', probability: 1 - pBWins };
    }
    return { stop: false, winner: 'undecided', probability: pBWins };
  }
}

const betaSample = (alpha: number, beta: number): number => {
  const x = gammaSample(alpha);
  const y = gammaSample(beta);
  return x / (x + y);
};

const gammaSample = (shape: number): number => {
  if (shape < 1) return gammaSample(shape + 1) * Math.random() ** (1 / shape);
  const d = shape - 1 / 3;
  const c = 1 / Math.sqrt(9 * d);
  while (true) {
    let x: number, v: number;
    do {
      x = gaussianRandom();
      v = 1 + c * x;
    } while (v <= 0);
    v = v ** 3;
    const u = Math.random();
    if (u < 1 - 0.0331 * x ** 4) return d * v;
    if (Math.log(u) < 0.5 * x * x + d * (1 - v + Math.log(v))) return d * v;
  }
};

const gaussianRandom = (): number => {
  const u1 = Math.random();
  const u2 = Math.random();
  return Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
};
贝叶斯优化
贝叶斯优化（Bayesian Optimization）是一种样本高效的全局优化方法，特别适合评估代价昂贵的黑盒函数（如神经网络超参数调优、A/B 测试配置）。它用高斯过程（GP）作为目标函数的代理模型，通过采集函数（Acquisition Function）选择下一个评估点。


interface BayesOptPoint {
  x: number[];
  y: number;
}

class SimpleBayesianOptimizer {
  private points: BayesOptPoint[] = [];
  private bounds: Array<[number, number]>;
  private kernel: (x1: number[], x2: number[]) => number;

  constructor(bounds: Array<[number, number]>) {
    this.bounds = bounds;
    this.kernel = (x1, x2) => {
      const lengthScale = 1.0;
      const dist = x1.reduce((a, v, i) => a + ((v - x2[i]) / lengthScale) ** 2, 0);
      return Math.exp(-0.5 * dist);
    };
  }

  addObservation(x: number[], y: number): void {
    this.points.push({ x, y });
  }

  predict(x: number[]): { mean: number; variance: number } {
    if (this.points.length === 0) return { mean: 0, variance: 1 };
    const n = this.points.length;
    const K = Array.from({ length: n }, (_, i) =>
      Array.from({ length: n }, (_, j) => this.kernel(this.points[i].x, this.points[j].x) + (i === j ? 1e-6 : 0)),
    );
    const kStar = this.points.map((p) => this.kernel(x, p.x));
    const y = this.points.map((p) => p.y);
    const alpha = solveLinearSystem(K, y);
    const mean = kStar.reduce((a, k, i) => a + k * alpha[i], 0);
    const v = solveLinearSystem(K, kStar);
    const kStarStar = this.kernel(x, x);
    const variance = kStarStar - kStar.reduce((a, k, i) => a + k * v[i], 0);
    return { mean, variance: Math.max(0, variance) };
  }

  expectedImprovement(x: number[]): number {
    const { mean, variance } = this.predict(x);
    if (variance <= 0) return 0;
    const bestY = Math.max(...this.points.map((p) => p.y));
    const sigma = Math.sqrt(variance);
    const z = (mean - bestY) / sigma;
    return sigma * (z * normalCDF(z) + normalPDF(z));
  }

  suggestNext(nCandidates: number = 1000): number[] {
    let bestX = this.bounds.map(([lo, hi]) => lo + Math.random() * (hi - lo));
    let bestEI = -Infinity;
    for (let i = 0; i < nCandidates; i++) {
      const x = this.bounds.map(([lo, hi]) => lo + Math.random() * (hi - lo));
      const ei = this.expectedImprovement(x);
      if (ei > bestEI) {
        bestEI = ei;
        bestX = x;
      }
    }
    return bestX;
  }
}

const solveLinearSystem = (A: number[][], b: number[]): number[] => {
  const n = A.length;
  const aug = A.map((row, i) => [...row, b[i]]);
  for (let col = 0; col < n; col++) {
    let maxRow = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) maxRow = row;
    }
    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
    for (let row = col + 1; row < n; row++) {
      const factor = aug[row][col] / aug[col][col];
      for (let j = col; j <= n; j++) aug[row][j] -= factor * aug[col][j];
    }
  }
  const x = new Array(n);
  for (let i = n - 1; i >= 0; i--) {
    x[i] = aug[i][n];
    for (let j = i + 1; j < n; j++) x[i] -= aug[i][j] * x[j];
    x[i] /= aug[i][i];
  }
  return x;
};

const normalCDF = (x: number): number => {
  const a1 = 0.254829592,
    a2 = -0.284496736,
    a3 = 1.421413741;
  const a4 = -1.453152027,
    a5 = 1.061405429,
    p = 0.3275911;
  const sign = x < 0 ? -1 : 1;
  const absX = Math.abs(x) / Math.sqrt(2);
  const t = 1.0 / (1.0 + p * absX);
  const y = 1 - ((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX);
  return 0.5 * (1.0 + sign * y);
};

const normalPDF = (x: number): number => Math.exp(-0.5 * x * x) / Math.sqrt(2 * Math.PI);
贝叶斯变点检测
贝叶斯变点检测（Bayesian Change Point Detection）用于在时间序列中识别数据生成过程发生变化的时刻。贝叶斯方法的优势在于可以在线运行，给出每个时间点是变点的后验概率。


class BayesianChangePointDetector {
  private runLength: number[] = [1];
  private mu0 = 0;
  private kappa0 = 1;
  private alpha0 = 1;
  private beta0 = 1;
  private muN: number[] = [0];
  private kappaN: number[] = [1];
  private alphaN: number[] = [1];
  private betaN: number[] = [1];
  private hazardRate: number;

  constructor(hazardRate: number = 1 / 200) {
    this.hazardRate = hazardRate;
  }

  update(x: number): { changePointProb: number; runLengthDist: number[] } {
    const n = this.runLength.length;
    const predProbs = new Array(n);
    for (let i = 0; i < n; i++) {
      const mu = this.muN[i];
      const scale = (this.betaN[i] * (this.kappaN[i] + 1)) / (this.alphaN[i] * this.kappaN[i]);
      const df = 2 * this.alphaN[i];
      predProbs[i] = studentTPDF((x - mu) / Math.sqrt(scale), df) / Math.sqrt(scale);
    }

    const growthProbs = this.runLength.map((r, i) => r * predProbs[i] * (1 - this.hazardRate));
    const changeProb = this.runLength.reduce((a, r, i) => a + r * predProbs[i] * this.hazardRate, 0);

    const newRL = [changeProb, ...growthProbs];
    const evidence = newRL.reduce((a, b) => a + b, 0);
    this.runLength = newRL.map((r) => r / evidence);

    const newMu = [this.mu0];
    const newKappa = [this.kappa0];
    const newAlpha = [this.alpha0];
    const newBeta = [this.beta0];

    for (let i = 0; i < n; i++) {
      const k = this.kappaN[i];
      newMu.push((k * this.muN[i] + x) / (k + 1));
      newKappa.push(k + 1);
      newAlpha.push(this.alphaN[i] + 0.5);
      newBeta.push(this.betaN[i] + (k * (x - this.muN[i]) ** 2) / (2 * (k + 1)));
    }

    this.muN = newMu;
    this.kappaN = newKappa;
    this.alphaN = newAlpha;
    this.betaN = newBeta;

    return {
      changePointProb: this.runLength[0],
      runLengthDist: [...this.runLength],
    };
  }
}

const studentTPDF = (x: number, df: number): number => {
  const coeff = Math.exp(lgamma((df + 1) / 2) - lgamma(df / 2) - 0.5 * Math.log(df * Math.PI));
  return coeff * (1 + (x * x) / df) ** (-(df + 1) / 2);
};

const lgamma = (x: number): number => {
  const c = [
    76.18009172947146, -86.50532032941677, 24.01409824083091, -1.231739572450155, 0.1208650973866179e-2,
    -0.5395239384953e-5,
  ];
  let y = x,
    tmp = x + 5.5;
  tmp -= (x + 0.5) * Math.log(tmp);
  let ser = 1.000000000190015;
  for (let j = 0; j < 6; j++) ser += c[j] / ++y;
  return -tmp + Math.log((2.5066282746310005 * ser) / x);
};

class BayesianRecommender {
  private items: Map<string, { alpha: number; beta: number }> = new Map();

  addItem(itemId: string, priorAlpha: number = 1, priorBeta: number = 1): void {
    this.items.set(itemId, { alpha: priorAlpha, beta: priorBeta });
  }

  recordInteraction(itemId: string, clicked: boolean): void {
    const item = this.items.get(itemId);
    if (!item) return;
    if (clicked) item.alpha += 1;
    else item.beta += 1;
  }

  rank(method: 'thompson' | 'ucb' | 'mean' = 'thompson'): string[] {
    const scored = Array.from(this.items.entries()).map(([id, { alpha, beta }]) => {
      let score: number;
      switch (method) {
        case 'thompson':
          score = betaSample(alpha, beta);
          break;
        case 'ucb': {
          const mean = alpha / (alpha + beta);
          const uncertainty = 1.96 * Math.sqrt((alpha * beta) / ((alpha + beta) ** 2 * (alpha + beta + 1)));
          score = mean + uncertainty;
          break;
        }
        case 'mean':
          score = alpha / (alpha + beta);
          break;
      }
      return { id, score };
    });
    scored.sort((a, b) => b.score - a.score);
    return scored.map((s) => s.id);
  }
}
