非参数检验方法

核心概念

非参数检验（Nonparametric Tests）是一类不依赖于总体分布形式假设的统计检验方法。与参数检验假设数据来自正态分布不同，非参数检验仅对数据的某些一般性质（如连续性、对称性）做出弱假设，因此也被称为"分布自由"（Distribution-free）检验。

非参数检验的核心优势在于鲁棒性：当数据严重偏斜、存在离群值、或样本量太小无法验证正态性时，非参数检验提供了可靠的替代方案。代价是在正态数据上，非参数检验的功效通常低于对应的参数检验。

秩检验的基本原理
非参数检验中最重要的技术是秩变换——将原始数据替换为它们在样本中的排名（秩）。秩变换消除了数据的分布形状，只保留了大小次序信息，从而实现了分布自由的性质。


秩变换的关键是处理结（Ties）：当多个观测值相同时，采用平均秩。


const assignRanks = (data: number[]): number[] => {
  const indexed = data.map((val, idx) => ({ val, idx }));
  indexed.sort((a, b) => a.val - b.val);
  const ranks = new Array(data.length);
  let i = 0;
  while (i < indexed.length) {
    let j = i;
    while (j < indexed.length && indexed[j].val === indexed[i].val) j++;
    const avgRank = (i + 1 + j) / 2;
    for (let k = i; k < j; k++) {
      ranks[indexed[k].idx] = avgRank;
    }
    i = j;
  }
  return ranks;
};

const rankTransform = (groups: number[][]): number[][] => {
  const allData = groups.flatMap((g, gi) => g.map((val) => ({ val, group: gi })));
  allData.sort((a, b) => a.val - b.val);
  const ranks = new Array(allData.length);
  let i = 0;
  while (i < allData.length) {
    let j = i;
    while (j < allData.length && allData[j].val === allData[i].val) j++;
    const avgRank = (i + 1 + j) / 2;
    for (let k = i; k < j; k++) ranks[k] = avgRank;
    i = j;
  }
  const result: number[][] = groups.map(() => []);
  let pos = 0;
  for (const item of allData) {
    result[item.group].push(ranks[pos++]);
  }
  return result;
};
Mann-Whitney U 检验
Mann-Whitney U 检验（也称 Wilcoxon 秩和检验）是独立两样本 t 检验的非参数替代，用于检验两个独立样本是否来自相同分布。检验统计量 U 基于两组数据的秩和计算。


const mannWhitneyU = (sample1: number[], sample2: number[]): { u: number; z: number; pValue: number } => {
  const n1 = sample1.length,
    n2 = sample2.length;
  const combined = [...sample1.map((v) => ({ val: v, group: 1 })), ...sample2.map((v) => ({ val: v, group: 2 }))];
  combined.sort((a, b) => a.val - b.val);

  const ranks = new Array(combined.length);
  let i = 0;
  while (i < combined.length) {
    let j = i;
    while (j < combined.length && combined[j].val === combined[i].val) j++;
    const avg = (i + 1 + j) / 2;
    for (let k = i; k < j; k++) ranks[k] = avg;
    i = j;
  }

  let r1 = 0;
  for (let k = 0; k < combined.length; k++) {
    if (combined[k].group === 1) r1 += ranks[k];
  }

  const u1 = r1 - (n1 * (n1 + 1)) / 2;
  const u2 = n1 * n2 - u1;
  const u = Math.min(u1, u2);
  const mu = (n1 * n2) / 2;
  const sigma = Math.sqrt((n1 * n2 * (n1 + n2 + 1)) / 12);
  const z = (u - mu) / sigma;
  const pValue = 2 * (1 - normalCDF(Math.abs(z)));

  return { u, z, pValue };
};

const normalCDF = (x: number): number => {
  const a1 = 0.254829592,
    a2 = -0.284496736,
    a3 = 1.421413741;
  const a4 = -1.453152027,
    a5 = 1.061405429,
    p = 0.3275911;
  const sign = x < 0 ? -1 : 1;
  x = Math.abs(x) / Math.sqrt(2);
  const t = 1.0 / (1.0 + p * x);
  const y = 1 - ((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);
  return 0.5 * (1.0 + sign * y);
};
Wilcoxon 符号秩与 Kruskal-Wallis 检验
Wilcoxon 符号秩检验是配对 t 检验的非参数替代，适用于配对观测或单样本中位数检验。它基于差值的绝对值排秩和符号信息。


Kruskal-Wallis 检验是单因素 ANOVA 的非参数替代，用于三组及以上的比较，基于混合样本的秩。


const wilcoxonSignedRank = (differences: number[]): { wPlus: number; wMinus: number; z: number; pValue: number } => {
  const nonZero = differences.filter((d) => d !== 0);
  const n = nonZero.length;
  const absVals = nonZero.map((d) => ({ abs: Math.abs(d), sign: d > 0 ? 1 : -1 }));
  absVals.sort((a, b) => a.abs - b.abs);

  const ranks = new Array(n);
  let i = 0;
  while (i < n) {
    let j = i;
    while (j < n && absVals[j].abs === absVals[i].abs) j++;
    const avg = (i + 1 + j) / 2;
    for (let k = i; k < j; k++) ranks[k] = avg;
    i = j;
  }

  let wPlus = 0,
    wMinus = 0;
  for (let k = 0; k < n; k++) {
    if (absVals[k].sign > 0) wPlus += ranks[k];
    else wMinus += ranks[k];
  }

  const mu = (n * (n + 1)) / 4;
  const sigma = Math.sqrt((n * (n + 1) * (2 * n + 1)) / 24);
  const w = Math.min(wPlus, wMinus);
  const z = (w - mu) / sigma;
  const pValue = 2 * (1 - normalCDF(Math.abs(z)));

  return { wPlus, wMinus, z, pValue };
};

const kruskalWallis = (groups: number[][]): { h: number; df: number; pValue: number } => {
  const N = groups.reduce((a, g) => a + g.length, 0);
  const allData: Array<{ val: number; group: number }> = [];
  groups.forEach((g, gi) => g.forEach((v) => allData.push({ val: v, group: gi })));
  allData.sort((a, b) => a.val - b.val);

  const ranks = new Array(N);
  let i = 0;
  while (i < N) {
    let j = i;
    while (j < N && allData[j].val === allData[i].val) j++;
    const avg = (i + 1 + j) / 2;
    for (let k = i; k < j; k++) ranks[k] = avg;
    i = j;
  }

  const groupRankSums = groups.map(() => 0);
  for (let k = 0; k < N; k++) {
    groupRankSums[allData[k].group] += ranks[k];
  }

  let h = 0;
  for (let g = 0; g < groups.length; g++) {
    const ni = groups[g].length;
    const ri = groupRankSums[g];
    h += (ri * ri) / ni;
  }
  h = (12 / (N * (N + 1))) * h - 3 * (N + 1);

  const df = groups.length - 1;
  const pValue = 1 - chiSquareCDF(h, df);
  return { h, df, pValue };
};

const chiSquareCDF = (x: number, k: number): number => {
  if (x <= 0) return 0;
  return regularizedGammaP(k / 2, x / 2);
};

const regularizedGammaP = (a: number, x: number): number => {
  let sum = 0,
    term = 1 / a;
  for (let n = 1; n < 200; n++) {
    sum += term;
    term *= x / (a + n);
    if (Math.abs(term) < 1e-12) break;
  }
  return sum * Math.exp(-x + a * Math.log(x) - lgamma(a));
};

const lgamma = (x: number): number => {
  const c = [
    76.18009172947146, -86.50532032941677, 24.01409824083091, -1.231739572450155, 0.1208650973866179e-2,
    -0.5395239384953e-5,
  ];
  let y = x,
    tmp = x + 5.5;
  tmp -= (x + 0.5) * Math.log(tmp);
  let ser = 1.000000000190015;
  for (let j = 0; j < 6; j++) ser += c[j] / ++y;
  return -tmp + Math.log((2.5066282746310005 * ser) / x);
};
置换检验与 Bootstrap 检验
置换检验（Permutation Test）是非参数检验的现代方法，直接模拟零假设下统计量的分布，无需任何分布假设。基本思想是：如果 H₀ 为真，那么数据的分组标签是可以互换的。通过随机重排标签并计算统计量的分布来得到 p 值。


Bootstrap 检验通过有放回重采样来估计检验统计量的分布，适用于更复杂的统计量和检验问题。


const permutationTest = (
  group1: number[],
  group2: number[],
  iterations: number = 10000,
  statFn: (g1: number[], g2: number[]) => number = (g1, g2) =>
    g1.reduce((a, b) => a + b, 0) / g1.length - g2.reduce((a, b) => a + b, 0) / g2.length,
): { observedStat: number; pValue: number; distribution: number[] } => {
  const combined = [...group1, ...group2];
  const n1 = group1.length;
  const observedStat = statFn(group1, group2);
  const permStats: number[] = [];

  for (let i = 0; i < iterations; i++) {
    const shuffled = [...combined];
    for (let j = shuffled.length - 1; j > 0; j--) {
      const k = Math.floor(Math.random() * (j + 1));
      [shuffled[j], shuffled[k]] = [shuffled[k], shuffled[j]];
    }
    const perm1 = shuffled.slice(0, n1);
    const perm2 = shuffled.slice(n1);
    permStats.push(statFn(perm1, perm2));
  }

  const extremeCount = permStats.filter((s) => Math.abs(s) >= Math.abs(observedStat)).length;

  return {
    observedStat,
    pValue: extremeCount / iterations,
    distribution: permStats,
  };
};

const bootstrapTest = (
  data: number[],
  mu0: number,
  iterations: number = 10000,
): { observedMean: number; pValue: number; ci: [number, number] } => {
  const n = data.length;
  const observedMean = data.reduce((a, b) => a + b, 0) / n;
  const centered = data.map((x) => x - observedMean + mu0);
  let extremeCount = 0;
  const bootMeans: number[] = [];

  for (let i = 0; i < iterations; i++) {
    let sum = 0;
    for (let j = 0; j < n; j++) {
      sum += centered[Math.floor(Math.random() * n)];
    }
    const bootMean = sum / n;
    bootMeans.push(bootMean);
    if (Math.abs(bootMean - mu0) >= Math.abs(observedMean - mu0)) {
      extremeCount++;
    }
  }

  bootMeans.sort((a, b) => a - b);
  const ci: [number, number] = [bootMeans[Math.floor(0.025 * iterations)], bootMeans[Math.floor(0.975 * iterations)]];

  return { observedMean, pValue: extremeCount / iterations, ci };
};
小结
    

  • 非参数检验不依赖分布形式假设，通过秩变换或重采样实现分布自由性
    • 

  • Mann-Whitney U 和 Wilcoxon 符号秩是两样本和配对检验的标准非参数替代
    • 

  • Kruskal-Wallis 检验将秩方法推广到多组比较场景
    • 

  • 置换检验直接模拟零假设分布，概念简单且适用范围广
    • 

  • 非参数检验在正态数据上的效率约为参数检验的 95.5%，在非正态数据上优势明显
    •
