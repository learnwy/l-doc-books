真值表与范式

核心概念

真值表（Truth Table）是命题逻辑中确定公式语义的基本工具，通过穷举所有可能的赋值来计算公式的真值。范式（Normal Form）是命题公式的标准化表示形式，使得每个公式都有唯一的规范表达。合取范式（CNF）和析取范式（DNF）是最重要的两种范式，它们分别是 SAT 求解器和电路综合的理论基础。

从计算的角度看，真值表方法的复杂度为 O(2ⁿ)（n 为变元数），这在变元较少时是精确可行的。但当变元数增大时，需要更高效的方法——范式变换、BDD（二元决策图）和 SAT 求解器应运而生。理解范式变换是掌握自动推理和形式验证工具的基础。

真值表的构造与分析

给定含 n 个命题变元的公式，真值表有 2ⁿ 行，每行对应一组赋值及该赋值下公式的真值。真值表是判定永真式、矛盾式和可满足性的完备方法。

type PropFormula =
  | { type: 'var'; name: string }
  | { type: 'not'; child: PropFormula }
  | { type: 'and'; left: PropFormula; right: PropFormula }
  | { type: 'or'; left: PropFormula; right: PropFormula }
  | { type: 'implies'; left: PropFormula; right: PropFormula }
  | { type: 'iff'; left: PropFormula; right: PropFormula };

function evaluate(f: PropFormula, v: Map<string, boolean>): boolean {
  switch (f.type) {
    case 'var':
      return v.get(f.name) ?? false;
    case 'not':
      return !evaluate(f.child, v);
    case 'and':
      return evaluate(f.left, v) && evaluate(f.right, v);
    case 'or':
      return evaluate(f.left, v) || evaluate(f.right, v);
    case 'implies':
      return !evaluate(f.left, v) || evaluate(f.right, v);
    case 'iff':
      return evaluate(f.left, v) === evaluate(f.right, v);
  }
}

interface TruthTableRow {
  valuation: Map<string, boolean>;
  result: boolean;
}

function buildTruthTable(formula: PropFormula): TruthTableRow[] {
  const vars = [...getVars(formula)].sort();
  const n = vars.length;
  const rows: TruthTableRow[] = [];

  for (let mask = 0; mask < 1 << n; mask++) {
    const v = new Map<string, boolean>();
    for (let i = 0; i < n; i++) {
      v.set(vars[i], Boolean(mask & (1 << (n - 1 - i))));
    }
    rows.push({ valuation: v, result: evaluate(formula, v) });
  }
  return rows;
}

function getVars(f: PropFormula): Set<string> {
  switch (f.type) {
    case 'var':
      return new Set([f.name]);
    case 'not':
      return getVars(f.child);
    default: {
      const l = 'left' in f ? getVars(f.left) : new Set<string>();
      const r = 'right' in f ? getVars(f.right) : new Set<string>();
      return new Set([...l, ...r]);
    }
  }
}

function analyzeTruthTable(rows: TruthTableRow[]): {
  isTautology: boolean;
  isContradiction: boolean;
  isSatisfiable: boolean;
  satisfyingCount: number;
} {
  const trueCount = rows.filter((r) => r.result).length;
  return {
    isTautology: trueCount === rows.length,
    isContradiction: trueCount === 0,
    isSatisfiable: trueCount > 0,
    satisfyingCount: trueCount,
  };
}

CNF 与 DNF 变换

合取范式（CNF）：子句的合取，每个子句是文字的析取。例如：(p ∨ ¬q) ∧ (¬p ∨ r)。

析取范式（DNF）：最小项的析取，每个最小项是文字的合取。例如：(p ∧ q) ∨ (¬p ∧ r)。

变换步骤：

  1. 消除蕴涵和等价：p→q 替换为 ¬p∨q，p↔q 替换为 (¬p∨q)∧(p∨¬q)
  2. 否定内移（De Morgan + 双重否定消除）：得到否定范式 NNF
  3. 分配律展开：∧ 对 ∨ 分配得 CNF，∨ 对 ∧ 分配得 DNF
function eliminateImplication(f: PropFormula): PropFormula {
  switch (f.type) {
    case 'var':
      return f;
    case 'not':
      return { type: 'not', child: eliminateImplication(f.child) };
    case 'implies':
      return {
        type: 'or',
        left: { type: 'not', child: eliminateImplication(f.left) },
        right: eliminateImplication(f.right),
      };
    case 'iff': {
      const left = eliminateImplication(f.left);
      const right = eliminateImplication(f.right);
      return {
        type: 'and',
        left: { type: 'or', left: { type: 'not', child: left }, right },
        right: { type: 'or', left, right: { type: 'not', child: right } },
      };
    }
    default:
      return {
        ...f,
        left: eliminateImplication((f as { left: PropFormula }).left),
        right: eliminateImplication((f as { right: PropFormula }).right),
      } as PropFormula;
  }
}

function toNNF(f: PropFormula): PropFormula {
  const noImpl = eliminateImplication(f);
  return pushNegation(noImpl);
}

function pushNegation(f: PropFormula): PropFormula {
  switch (f.type) {
    case 'var':
      return f;
    case 'not':
      switch (f.child.type) {
        case 'var':
          return f;
        case 'not':
          return pushNegation(f.child.child);
        case 'and':
          return {
            type: 'or',
            left: pushNegation({ type: 'not', child: f.child.left }),
            right: pushNegation({ type: 'not', child: f.child.right }),
          };
        case 'or':
          return {
            type: 'and',
            left: pushNegation({ type: 'not', child: f.child.left }),
            right: pushNegation({ type: 'not', child: f.child.right }),
          };
        default:
          return f;
      }
    case 'and':
      return { type: 'and', left: pushNegation(f.left), right: pushNegation(f.right) };
    case 'or':
      return { type: 'or', left: pushNegation(f.left), right: pushNegation(f.right) };
    default:
      return f;
  }
}

function toCNF(f: PropFormula): PropFormula {
  const nnf = toNNF(f);
  return distributeCNF(nnf);
}

function distributeCNF(f: PropFormula): PropFormula {
  switch (f.type) {
    case 'var':
    case 'not':
      return f;
    case 'and':
      return { type: 'and', left: distributeCNF(f.left), right: distributeCNF(f.right) };
    case 'or': {
      const left = distributeCNF(f.left);
      const right = distributeCNF(f.right);
      if (left.type === 'and')
        return {
          type: 'and',
          left: distributeCNF({ type: 'or', left: left.left, right }),
          right: distributeCNF({ type: 'or', left: left.right, right }),
        };
      if (right.type === 'and')
        return {
          type: 'and',
          left: distributeCNF({ type: 'or', left, right: right.left }),
          right: distributeCNF({ type: 'or', left, right: right.right }),
        };
      return { type: 'or', left, right };
    }
    default:
      return f;
  }
}

主范式与规范表示

主合取范式（Principal CNF）：每个子句包含所有变元的极大项（Maxterm）的合取。主析取范式类似，使用极小项（Minterm）。主范式的唯一性使其成为判定逻辑等价的有效工具。

极小项 mᵢ：第 i 个赋值为真的合取项。例如 m₅ = p ∧ ¬q ∧ r（对应二进制 101）。

极大项 Mᵢ：第 i 个赋值为假的析取项。Mᵢ = ¬mᵢ。

type Literal = { variable: string; negated: boolean };
type Clause = Literal[];

function toPrincipalDNF(formula: PropFormula): Clause[] {
  const vars = [...getVars(formula)].sort();
  const table = buildTruthTable(formula);
  const minterms: Clause[] = [];

  for (const row of table) {
    if (row.result) {
      const minterm: Clause = vars.map((v) => ({
        variable: v,
        negated: !row.valuation.get(v),
      }));
      minterms.push(minterm);
    }
  }
  return minterms;
}

function toPrincipalCNF(formula: PropFormula): Clause[] {
  const vars = [...getVars(formula)].sort();
  const table = buildTruthTable(formula);
  const maxterms: Clause[] = [];

  for (const row of table) {
    if (!row.result) {
      const maxterm: Clause = vars.map((v) => ({
        variable: v,
        negated: row.valuation.get(v)!,
      }));
      maxterms.push(maxterm);
    }
  }
  return maxterms;
}

function quineMcCluskey(minterms: number[], numVars: number): Clause[] {
  const primeImplicants: Set<string> = new Set();
  let current = minterms.map((m) => ({
    binary: m.toString(2).padStart(numVars, '0'),
    covered: new Set([m]),
    used: false,
  }));

  while (current.length > 0) {
    const next: typeof current = [];
    const usedInMerge = new Set<number>();

    for (let i = 0; i < current.length; i++) {
      for (let j = i + 1; j < current.length; j++) {
        const diff = findSingleDiff(current[i].binary, current[j].binary);
        if (diff >= 0) {
          usedInMerge.add(i);
          usedInMerge.add(j);
          const merged = current[i].binary.substring(0, diff) + '-' + current[i].binary.substring(diff + 1);
          const covered = new Set([...current[i].covered, ...current[j].covered]);
          next.push({ binary: merged, covered, used: false });
        }
      }
    }

    for (let i = 0; i < current.length; i++) {
      if (!usedInMerge.has(i)) primeImplicants.add(current[i].binary);
    }
    current = next;
  }

  return [...primeImplicants].map((pi) =>
    pi
      .split('')
      .map((bit, i) => (bit === '-' ? null : { variable: `p${i + 1}`, negated: bit === '0' }))
      .filter((l): l is Literal => l !== null),
  );
}

function findSingleDiff(a: string, b: string): number {
  let diffPos = -1;
  for (let i = 0; i < a.length; i++) {
    if (a[i] !== b[i]) {
      if (diffPos >= 0) return -1;
      diffPos = i;
    }
  }
  return diffPos;
}

  • 真值表通过穷举所有 2ⁿ 个赋值完备地判定公式的语义性质，但复杂度呈指数增长
