常微分方程应用

核心概念

常微分方程（ODE）数值方法的价值最终体现在实际问题的求解能力上。从天体力学到生物动力学，从电路模拟到化学反应，ODE 模型是科学与工程中描述动态系统的基本语言。本节将数值方法应用于典型问题，展示方法选择、精度控制和结果验证的完整实践流程。

将实际问题转化为 ODE 数值求解的标准流程包含：建模（建立方程）、选方法（根据刚性、精度需求选择求解器）、设参数（步长/容差设定）、求解、验证（守恒量检查、收敛性测试、与已知解对比）。方法选择的关键判据是问题是否具有刚性、解是否需要保持某种结构（如能量守恒）、以及精度要求的高低。

天体力学与辛积分
天体力学中的 N 体问题是 Hamilton 系统 q˙=∂H/∂p,  p˙=−∂H/∂q\dot{q} = \partial H / \partial p, \; \dot{p} = -\partial H / \partial qq˙​=∂H/∂p,p˙​=−∂H/∂q，其解在相空间中保持辛结构（面积守恒）。通用 RK 方法虽然短期精度高，但长时间积分会导致能量漂移。辛积分方法（如 Störmer-Verlet、蛙跳法）天然保持辛结构，确保能量误差有界振荡而非单调增长。


import numpy as np

def stormer_verlet(grad_V, grad_T, q0, p0, h, N):
    d = len(q0)
    q = np.zeros((N + 1, d))
    p = np.zeros((N + 1, d))
    q[0] = q0
    p[0] = p0

    for i in range(N):
        p_half = p[i] - h/2 * grad_V(q[i])
        q[i+1] = q[i] + h * grad_T(p_half)
        p[i+1] = p_half - h/2 * grad_V(q[i+1])

    return q, p

def kepler_problem():
    grad_V = lambda q: q / np.linalg.norm(q)**3
    grad_T = lambda p: p
    H = lambda q, p: 0.5 * np.dot(p, p) - 1.0 / np.linalg.norm(q)

    e = 0.6
    q0 = np.array([1 - e, 0.0])
    p0 = np.array([0.0, np.sqrt((1 + e) / (1 - e))])

    T = 2 * np.pi
    h = 0.01
    N_periods = 100
    N_steps = int(N_periods * T / h)

    q, p = stormer_verlet(grad_V, grad_T, q0, p0, h, N_steps)

    H0 = H(q0, p0)
    energies = np.array([H(q[i], p[i]) for i in range(0, N_steps + 1, N_steps // 100)])
    energy_error = np.abs(energies - H0)

    print("Kepler 问题 (e=0.6, 100 周期):")
    print(f"  步长: h = {h}")
    print(f"  总步数: {N_steps}")
    print(f"  初始能量: {H0:.10f}")
    print(f"  最大能量误差: {np.max(energy_error):.6e}")
    print(f"  终态能量误差: {abs(H(q[-1], p[-1]) - H0):.6e}")
    print(f"  轨道闭合误差: {np.linalg.norm(q[-1] - q0):.6e}")

    return q, p

q, p = kepler_problem()

def rk4_step(f, t, y, h):
    k1 = f(t, y)
    k2 = f(t + h/2, y + h/2 * k1)
    k3 = f(t + h/2, y + h/2 * k2)
    k4 = f(t + h, y + h * k3)
    return y + h/6 * (k1 + 2*k2 + 2*k3 + k4)

def kepler_rk4():
    def f(t, state):
        q, p = state[:2], state[2:]
        r = np.linalg.norm(q)
        return np.array([p[0], p[1], -q[0]/r**3, -q[1]/r**3])

    e = 0.6
    state0 = np.array([1-e, 0, 0, np.sqrt((1+e)/(1-e))])
    H = lambda s: 0.5*(s[2]**2 + s[3]**2) - 1/np.sqrt(s[0]**2 + s[1]**2)
    H0 = H(state0)

    h = 0.01
    T = 2 * np.pi * 100
    state = state0.copy()
    t = 0
    max_err = 0

    while t < T:
        state = rk4_step(f, t, state, h)
        t += h
        max_err = max(max_err, abs(H(state) - H0))

    print(f"\nRK4 对比 (同步长 h={h}):")
    print(f"  最大能量误差: {max_err:.6e}")
    print(f"  终态能量误差: {abs(H(state) - H0):.6e}")

流行病学模型
SIR/SEIR 模型是流行病动力学的基础。SIR 模型将人群分为易感者（S）、感染者（I）、康复者（R），通过微分方程描述三者之间的转化动力学。这类问题通常非刚性，但解的结构（如 S+I+R=NS + I + R = NS+I+R=N 守恒）提供了重要的验证手段。


import numpy as np
from scipy.integrate import solve_ivp

def sir_model(t, y, beta, gamma, N):
    S, I, R = y
    dS = -beta * S * I / N
    dI = beta * S * I / N - gamma * I
    dR = gamma * I
    return [dS, dI, dR]

def seir_model(t, y, beta, sigma, gamma, N):
    S, E, I, R = y
    dS = -beta * S * I / N
    dE = beta * S * I / N - sigma * E
    dI = sigma * E - gamma * I
    dR = gamma * I
    return [dS, dE, dI, dR]

N = 1e6
beta = 0.3
gamma = 0.1
I0 = 100
S0 = N - I0
R0_init = 0

R0_number = beta / gamma
print(f"SIR 模型参数:")
print(f"  基本传染数 R0 = {R0_number:.2f}")
print(f"  预测最终感染比例 (通过超越方程)")

sol = solve_ivp(
    lambda t, y: sir_model(t, y, beta, gamma, N),
    (0, 300), [S0, I0, R0_init],
    method='RK45', rtol=1e-10, atol=1e-12,
    dense_output=True
)

t_eval = np.linspace(0, 300, 1000)
y_eval = sol.sol(t_eval)
S, I, R = y_eval

peak_idx = np.argmax(I)
print(f"  感染峰值: {I[peak_idx]:.0f} 人, 出现在第 {t_eval[peak_idx]:.1f} 天")
print(f"  最终感染比例: {R[-1]/N*100:.1f}%")
print(f"  群体免疫阈值: {(1 - 1/R0_number)*100:.1f}%")
print(f"  总人口守恒误差: {abs(S[-1]+I[-1]+R[-1]-N):.2e}")

print("\nSEIR 模型 (含潜伏期):")
sigma = 0.2
sol_seir = solve_ivp(
    lambda t, y: seir_model(t, y, beta, sigma, gamma, N),
    (0, 300), [S0, 0, I0, R0_init],
    method='RK45', rtol=1e-10, atol=1e-12,
    dense_output=True
)
y_seir = sol_seir.sol(t_eval)
peak_seir = np.argmax(y_seir[2])
print(f"  感染峰值: {y_seir[2][peak_seir]:.0f} 人, 出现在第 {t_eval[peak_seir]:.1f} 天")
print(f"  峰值延迟 (vs SIR): {t_eval[peak_seir] - t_eval[peak_idx]:.1f} 天")
方法的线弹性与半离散 PDE
方法线（Method of Lines, MOL）是求解偏微分方程的标准数值策略：先将空间离散化（有限差分、有限元），得到 ODE 系统，再用 ODE 求解器推进时间。MOL 的优势在于可以独立选择空间离散和时间推进方案。


以一维对流-扩散方程 ut+aux=νuxxu_t + a u_x = \nu u_{xx}ut​+aux​=νuxx​ 为例，空间中心差分离散化后得到的 ODE 系统同时包含纯虚特征值（对流项）和负实特征值（扩散项），方法选择需要兼顾两者。


import numpy as np
from scipy.integrate import solve_ivp

def advection_diffusion_mol(N, a=1.0, nu=0.01, L=2*np.pi):
    dx = L / N
    x = np.linspace(0, L - dx, N)

    def rhs(t, u):
        dudx = (np.roll(u, -1) - np.roll(u, 1)) / (2 * dx)
        d2udx2 = (np.roll(u, -1) - 2*u + np.roll(u, 1)) / dx**2
        return -a * dudx + nu * d2udx2

    u0 = np.exp(-50 * (x - np.pi)**2)

    return x, u0, rhs

N = 100
a = 1.0
nu_values = [0.1, 0.01, 0.001]

print("对流-扩散方程 (MOL):")
for nu in nu_values:
    x, u0, rhs = advection_diffusion_mol(N, a=a, nu=nu)

    sol = solve_ivp(rhs, (0, 2*np.pi), u0, method='RK45', rtol=1e-6, atol=1e-8)

    mass_init = np.sum(u0)
    mass_final = np.sum(sol.y[:, -1])
    mass_err = abs(mass_final - mass_init) / abs(mass_init)

    peclet = a * (2*np.pi/N) / nu
    print(f"  ν={nu}: Peclet={peclet:.1f}, 步数={sol.t.size}, "
          f"质量守恒误差={mass_err:.2e}")

print("\n热方程刚性分析:")
for N in [20, 50, 100, 200]:
    dx = 2*np.pi / N
    lambda_max = 4 * 0.01 / dx**2
    h_explicit = 2 / lambda_max
    print(f"  N={N}: λ_max={lambda_max:.2e}, 显式步长限制={h_explicit:.2e}")
对于强扩散主导的问题（小 ν\nuν、高 Peclet 数），空间离散后的 ODE 系统是刚性的，需要隐式时间推进。IMEX（隐式-显式）方法将扩散项隐式处理、对流项显式处理，兼顾两者的特性。


混沌系统与长期行为
混沌系统对初始条件极度敏感——相邻轨道以指数速度发散（Lyapunov 指数为正）。这意味着任何数值方法都无法给出长期的逐点准确解，但可以正确捕捉系统的统计特性（吸引子结构、Lyapunov 指数、功率谱）。Lorenz 系统是最经典的混沌系统之一。


import numpy as np
from scipy.integrate import solve_ivp

def lorenz(t, state, sigma=10, rho=28, beta=8/3):
    x, y, z = state
    return [sigma*(y - x), x*(rho - z) - y, x*y - beta*z]

state0 = np.array([1.0, 1.0, 1.0])

sol = solve_ivp(lorenz, (0, 50), state0, method='RK45',
                rtol=1e-12, atol=1e-14, dense_output=True)

t_eval = np.linspace(0, 50, 50000)
trajectory = sol.sol(t_eval)

state0_perturbed = state0 + np.array([1e-10, 0, 0])
sol_perturbed = solve_ivp(lorenz, (0, 50), state0_perturbed, method='RK45',
                          rtol=1e-12, atol=1e-14, dense_output=True)
traj_perturbed = sol_perturbed.sol(t_eval)

distance = np.sqrt(np.sum((trajectory - traj_perturbed)**2, axis=0))
diverge_idx = np.argmax(distance > 1.0)

print("Lorenz 系统混沌分析:")
print(f"  初始扰动: 1e-10")
print(f"  轨道发散时间 (距离>1): t = {t_eval[diverge_idx]:.2f}")
print(f"  t=50 时轨道距离: {distance[-1]:.2f}")

log_dist = np.log(distance[distance > 0][:5000])
t_short = t_eval[distance > 0][:5000]
if len(t_short) > 100:
    lyapunov_est = np.polyfit(t_short[:500], log_dist[:500], 1)[0]
    print(f"  估计最大 Lyapunov 指数: {lyapunov_est:.2f}")
    print(f"  理论最大 Lyapunov 指数: ≈ 0.91")

print(f"\n吸引子统计特征:")
x, y, z = trajectory
print(f"  x 范围: [{np.min(x):.2f}, {np.max(x):.2f}]")
print(f"  z 均值: {np.mean(z):.2f} (理论: {28-1:.0f})")

小结
    

  • 天体力学的 Hamilton 系统需要辛积分方法保持长期能量有界，Störmer-Verlet 是二阶辛方法
    • 

  • 流行病学 SIR/SEIR 模型是非刚性系统，总人口守恒 S+I+R=NS+I+R=NS+I+R=N 是关键验证指标
    • 

  • Method of Lines 将 PDE 空间离散化为 ODE 系统，方法选择需考虑对流与扩散的平衡
    • 

  • 混沌系统数值解的长期逐点精度不可保证，但统计特性（吸引子、Lyapunov 指数）可靠
    • 

  • 方法选择的核心判据是刚性程度、结构保持需求和精度要求的综合权衡
    • 

  • 结果验证应多管齐下：守恒量检查、收敛性测试、物理直觉和已知解对比
