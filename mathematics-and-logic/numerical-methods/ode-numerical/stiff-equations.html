刚性方程

核心概念

刚性（stiffness）是常微分方程数值求解中最重要的概念之一。一个方程组是刚性的，当且仅当其 Jacobian 矩阵的特征值跨越极大的量级范围——既有快速衰减的分量（对应大负特征值），又有缓慢变化的分量（对应小特征值）。刚性的本质困难在于：显式方法的步长被最快分量的稳定性约束，但解的有意义变化由最慢分量决定，导致巨大的计算浪费。

刚性比（stiffness ratio）ρ=∣λmax⁡∣/∣λmin⁡∣\rho = |\lambda_{\max}| / |\lambda_{\min}|ρ=∣λmax​∣/∣λmin​∣ 量化了这种矛盾。ρ\rhoρ 越大，显式方法越低效——步长由 1/∣λmax⁡∣1/|\lambda_{\max}|1/∣λmax​∣ 决定，但需要积分的时间尺度由 1/∣λmin⁡∣1/|\lambda_{\min}|1/∣λmin​∣ 决定。隐式方法通过无条件稳定性（A-稳定性）或 L-稳定性绕过这一约束，以每步求解非线性方程组的代价换取对步长的自由选择。

刚性方程的识别
刚性方程在化学动力学、电路模拟、控制系统、半离散偏微分方程等领域广泛出现。一个经典的示例是 Robertson 化学动力学方程组：


y1′=−0.04y1+104y2y3y_1' = -0.04y_1 + 10^4 y_2 y_3y1′​=−0.04y1​+104y2​y3​
 


y2′=0.04y1−104y2y3−3×107y22y_2' = 0.04y_1 - 10^4 y_2 y_3 - 3\times10^7 y_2^2y2′​=0.04y1​−104y2​y3​−3×107y22​
 


y3′=3×107y22y_3' = 3\times10^7 y_2^2y3′​=3×107y22​
反应速率常数跨越 11 个数量级（0.040.040.04 到 3×1073\times10^73×107），是经典的刚性问题。


import numpy as np

def robertson(t, y):
    y1, y2, y3 = y
    dy1 = -0.04 * y1 + 1e4 * y2 * y3
    dy2 = 0.04 * y1 - 1e4 * y2 * y3 - 3e7 * y2**2
    dy3 = 3e7 * y2**2
    return np.array([dy1, dy2, dy3])

def robertson_jacobian(t, y):
    y1, y2, y3 = y
    return np.array([
        [-0.04, 1e4 * y3, 1e4 * y2],
        [0.04, -1e4 * y3 - 6e7 * y2, -1e4 * y2],
        [0, 6e7 * y2, 0]
    ])

y0 = np.array([1.0, 0.0, 0.0])
J0 = robertson_jacobian(0, y0)
eigenvalues = np.linalg.eigvals(J0)
print("Robertson 方程初始 Jacobian 特征值:")
for ev in eigenvalues:
    print(f"  λ = {ev:.6e}")
print(f"刚性比: {max(abs(eigenvalues)) / min(abs(eigenvalues[eigenvalues != 0])):.2e}")

from scipy.integrate import solve_ivp

sol = solve_ivp(robertson, (0, 1e6), y0, method='BDF',
                jac=robertson_jacobian, rtol=1e-8, atol=1e-10,
                dense_output=True)

t_eval = np.logspace(-5, 6, 100)
y_eval = sol.sol(t_eval)
print(f"\nBDF 求解器统计:")
print(f"  总步数: {sol.t.size}")

热传导方程的半离散化是另一类典型的刚性问题。将 ut=αuxxu_t = \alpha u_{xx}ut​=αuxx​ 在空间上用中心差分离散化后，得到 ODE 系统 u′=α(Δx)2Au\mathbf{u}' = \frac{\alpha}{(\Delta x)^2} A \mathbf{u}u′=(Δx)2α​Au，其中 AAA 是三对角矩阵，特征值为 λk=−4sin⁡2(kπh/2)\lambda_k = -4\sin^2(k\pi h/2)λk​=−4sin2(kπh/2)，刚性比随网格加细呈 O(1/h2)O(1/h^2)O(1/h2) 增长。


import numpy as np

def heat_equation_semidiscrete(N, alpha=1.0):
    dx = 1.0 / (N + 1)
    diag = -2 * np.ones(N)
    off_diag = np.ones(N - 1)
    A = (np.diag(diag) + np.diag(off_diag, 1) + np.diag(off_diag, -1)) * alpha / dx**2
    return A, dx

print("热传导方程半离散化的刚性分析:")
for N in [10, 20, 50, 100, 200]:
    A, dx = heat_equation_semidiscrete(N)
    eigenvalues = np.linalg.eigvals(A)
    lambda_max = max(abs(eigenvalues))
    lambda_min = min(abs(eigenvalues))
    ratio = lambda_max / lambda_min
    h_explicit = 2.0 / lambda_max
    print(f"  N={N:3d}: |λ_max|={lambda_max:.2e}, |λ_min|={lambda_min:.2e}, "
          f"刚性比={ratio:.1f}, 显式步长上界={h_explicit:.2e}")
隐式 Runge-Kutta 方法
隐式 RK（IRK）方法是刚性问题最强大的求解器。与显式 RK 不同，IRK 的 Butcher 表中 AAA 矩阵是满的（或至少不是严格下三角），每个阶段的 kik_iki​ 需要同时求解一个耦合的非线性方程组。


基于 Gauss-Legendre 节点的 IRK 方法具有最高的精度阶：sss 阶段达到 2s2s2s 阶精度。基于 Radau IIA 节点的方法具有 L-稳定性（A-稳定且当 z→−∞z \to -\inftyz→−∞ 时 R(z)→0R(z) \to 0R(z)→0），更适合刚性问题。


import numpy as np
from scipy.optimize import fsolve

def implicit_midpoint(f, t_span, y0, h):
    t0, tf = t_span
    t_values = np.arange(t0, tf + h/2, h)
    n = len(t_values)
    dim = len(y0)
    y = np.zeros((n, dim))
    y[0] = y0

    for i in range(n - 1):
        def residual(k):
            y_mid = y[i] + h/2 * k
            return k - f(t_values[i] + h/2, y_mid)

        k0 = f(t_values[i], y[i])
        k_sol = fsolve(residual, k0, full_output=False)
        y[i + 1] = y[i] + h * k_sol

    return t_values, y

def sdirk2(f, jac, t_span, y0, h):
    gamma = 1 - np.sqrt(2)/2
    t0, tf = t_span
    t_values = np.arange(t0, tf + h/2, h)
    n = len(t_values)
    dim = len(y0)
    y = np.zeros((n, dim))
    y[0] = y0

    for i in range(n - 1):
        tn = t_values[i]
        J = jac(tn, y[i])
        M = np.eye(dim) - h * gamma * J

        def stage1_residual(k1):
            return k1 - f(tn + gamma*h, y[i] + h*gamma*k1)
        k1 = fsolve(stage1_residual, f(tn, y[i]))

        def stage2_residual(k2):
            return k2 - f(tn + h, y[i] + h*(1-gamma)*k1 + h*gamma*k2)
        k2 = fsolve(stage2_residual, k1)

        y[i + 1] = y[i] + h * ((1-gamma)*k1 + gamma*k2)

    return t_values, y

f_stiff = lambda t, y: np.array([-1000*y[0] + y[1], y[0] - y[1]])

y0 = np.array([1.0, 0.0])
t_im, y_im = implicit_midpoint(f_stiff, (0, 0.1), y0, 0.001)
print(f"隐式中点法求解刚性系统:")
print(f"  t=0.1: y = [{y_im[-1,0]:.6e}, {y_im[-1,1]:.6e}]")
BDF 多步法
后向差分公式（Backward Differentiation Formulas, BDF）是刚性 ODE 求解中另一大类方法。kkk 步 BDF 方法使用前 kkk 步的解值和当前步的导数：


∑j=0kαjyn+1−j=hβ0f(tn+1,yn+1)\sum_{j=0}^{k} \alpha_j y_{n+1-j} = h \beta_0 f(t_{n+1}, y_{n+1})j=0∑k​αj​yn+1−j​=hβ0​f(tn+1​,yn+1​)
BDF 方法的优势在于每步只需求解一个 n×nn \times nn×n 的非线性系统（而非 IRK 的 ns×nsns \times nsns×ns 系统），计算量更小。BDF1 就是隐式欧拉，BDF2 是二阶 A-稳定方法，BDF3-5 是 A(α)-稳定的。BDF6 及更高阶是不稳定的。


import numpy as np
from scipy.optimize import fsolve

BDF_COEFFICIENTS = {
    1: {"alpha": [1, -1], "beta": 1},
    2: {"alpha": [3/2, -2, 1/2], "beta": 1},
    3: {"alpha": [11/6, -3, 3/2, -1/3], "beta": 1},
    4: {"alpha": [25/12, -4, 3, -4/3, 1/4], "beta": 1},
    5: {"alpha": [137/60, -5, 5, -10/3, 5/4, -1/5], "beta": 1},
}

def bdf_solve(f, jac_f, t_span, y0, h, order=2):
    from scipy.integrate import solve_ivp
    t0, tf = t_span
    dim = len(y0)
    coeffs = BDF_COEFFICIENTS[order]
    alpha = coeffs["alpha"]
    beta = coeffs["beta"]

    rk_sol = solve_ivp(f, (t0, t0 + order*h), y0, method='RK45',
                        t_eval=np.linspace(t0, t0 + order*h, order + 1),
                        rtol=1e-12)

    t_values = list(rk_sol.t)
    y_values = [rk_sol.y[:, i] for i in range(order + 1)]

    t = t0 + order * h
    while t < tf - h/2:
        t_new = t + h

        rhs = -sum(alpha[j+1] * y_values[-(j+1)] for j in range(order))

        def residual(y_new):
            return alpha[0] * y_new - h * beta * f(t_new, y_new) - rhs

        y_guess = y_values[-1] + h * f(t, y_values[-1])
        y_new = fsolve(residual, y_guess)

        t_values.append(t_new)
        y_values.append(y_new)
        t = t_new

    return np.array(t_values), np.array(y_values)

f = lambda t, y: np.array([-100*y[0] + y[1] + 1, y[0] - 100*y[1] + 1])
jac = lambda t, y: np.array([[-100, 1], [1, -100]])
y0 = np.array([0.0, 0.0])

for order in [1, 2, 3, 4]:
    t, y = bdf_solve(f, jac, (0, 1), y0, 0.01, order=order)
    y_exact = np.array([1/99, 1/99])
    err = np.linalg.norm(y[-1] - y_exact)
    print(f"BDF{order}: t=1 误差 = {err:.6e}")

Newton 迭代与 Jacobian 策略
隐式方法每步需要求解非线性方程组 G(yn+1)=0G(y_{n+1}) = 0G(yn+1​)=0，Newton 迭代是标准方法。Newton 迭代的核心代价在于 Jacobian 矩阵 J=∂f/∂yJ = \partial f / \partial yJ=∂f/∂y 的计算和线性系统 (I−hγJ)δ=−G(I - h\gamma J)\delta = -G(I−hγJ)δ=−G 的求解。


实际实现中采用多种策略降低代价：冻结 Jacobian（多步共用一个 JJJ）、LU 分解复用、有限差分近似 Jacobian、稀疏 Jacobian 利用等。


import numpy as np

def simplified_newton(f, jac, t_new, y_old, h, gamma, y_guess, tol=1e-10, max_iter=10):
    J = jac(t_new, y_old)
    n = len(y_old)
    M = np.eye(n) - h * gamma * J

    from scipy.linalg import lu_factor, lu_solve
    lu, piv = lu_factor(M)

    y = y_guess.copy()
    for iteration in range(max_iter):
        G = y - y_old - h * gamma * f(t_new, y)
        delta = lu_solve((lu, piv), -G)
        y += delta

        if np.linalg.norm(delta) < tol:
            return y, iteration + 1

    return y, max_iter

f = lambda t, y: np.array([-1000*(y[0] - np.cos(t)), -y[1] + np.sin(t)])
jac = lambda t, y: np.array([[-1000, 0], [0, -1]])

y0 = np.array([1.0, 0.0])
h = 0.1
y_guess = y0 + h * f(0, y0)

y_new, iters = simplified_newton(f, jac, h, y0, h, 1.0, y_guess)
print(f"简化 Newton: {iters} 次迭代收敛")
print(f"  解: y = [{y_new[0]:.10f}, {y_new[1]:.10f}]")

print("\nJacobian 更新策略比较:")
strategies = {
    "每步更新": 1,
    "每 5 步更新": 5,
    "每 10 步更新": 10,
}

for name, freq in strategies.items():
    total_jac_evals = 0
    total_newton_iters = 0
    y = y0.copy()
    t = 0
    J_cached = None

    for step in range(100):
        t_new = t + h
        if step % freq == 0:
            J_cached = jac(t_new, y)
            total_jac_evals += 1

        y_guess = y + h * f(t, y)
        y_new, iters = simplified_newton(f, lambda t, y: J_cached, t_new, y, h, 1.0, y_guess)
        total_newton_iters += iters
        y = y_new
        t = t_new

    print(f"  {name}: Jac求值={total_jac_evals}, 平均Newton迭代={total_newton_iters/100:.1f}")
小结
    

  • 刚性方程的 Jacobian 特征值跨越巨大量级范围，显式方法步长受限于最快分量
    • 

  • 刚性比 ρ=∣λmax⁡∣/∣λmin⁡∣\rho = |\lambda_{\max}|/|\lambda_{\min}|ρ=∣λmax​∣/∣λmin​∣ 量化刚性程度，ρ≫1\rho \gg 1ρ≫1 表明需要隐式方法
    • 

  • 隐式 RK 方法（特别是 Radau IIA）具有高精度和 L-稳定性，适合强刚性问题
    • 

  • BDF 多步法每步只需解 n×nn \times nn×n 系统，计算效率高，BDF2 是通用推荐
    • 

  • Newton 迭代是隐式方法的核心，Jacobian 冻结和 LU 复用是关键优化手段
    • 

  • 现代刚性求解器（如 MATLAB ode15s、SciPy BDF）结合变步长变阶策略实现全自动求解
    •
