龙格-库塔方法

核心概念

龙格-库塔（Runge-Kutta, RK）方法是求解常微分方程初值问题最广泛使用的单步方法族。其核心思想是在一个时间步内计算多个"阶段"（stages）的斜率，通过加权组合这些斜率来逼近 Taylor 展开的高阶项，从而在不显式计算高阶导数的情况下达到高阶精度。

一个 sss 阶段的显式 RK 方法可以用 Butcher 表（Butcher tableau）紧凑表示。经典四阶 Runge-Kutta 方法（RK4）使用 4 个阶段的斜率加权平均，达到四阶精度，是最著名的 ODE 数值方法之一。现代自适应 RK 方法（如 RK45、Dormand-Prince）通过嵌入式公式对实现自动步长控制，成为工业级 ODE 求解器的标准。

import numpy as np

class RungeKutta:
    def __init__(self, A, b, c):
        self.A = np.array(A, dtype=np.float64)
        self.b = np.array(b, dtype=np.float64)
        self.c = np.array(c, dtype=np.float64)
        self.s = len(b)

    def step(self, f, t, y, h):
        k = np.zeros((self.s,) + np.shape(y))
        for i in range(self.s):
            y_stage = y + h * np.dot(self.A[i, :i], k[:i]) if i > 0 else y
            k[i] = f(t + self.c[i] * h, y_stage)
        return y + h * np.dot(self.b, k)

    def solve(self, f, t_span, y0, h):
        t0, tf = t_span
        t = np.arange(t0, tf + h/2, h)
        y = np.zeros((len(t),) + np.shape(y0))
        y[0] = y0
        for i in range(len(t) - 1):
            y[i+1] = self.step(f, t[i], y[i], h)
        return t, y

经典四阶 RK4
经典 RK4 是应用最广泛的固定步长 RK 方法。它在每步计算 4 个斜率：起点、两个中点、终点，然后按 Simpson 权重（1:2:2:1）/6 组合。这种"一头一尾两个中间"的采样策略使得 4 次函数求值就能达到四阶精度。

yn+1=yn+h6(k1+2k2+2k3+k4)y_{n+1} = y_n + \frac{h}{6}(k_1 + 2k_2 + 2k_3 + k_4)yn+1​=yn​+6h​(k1​+2k2​+2k3​+k4​)
import numpy as np

def rk4_step(f, t, y, h):
    k1 = f(t, y)
    k2 = f(t + h/2, y + h/2 * k1)
    k3 = f(t + h/2, y + h/2 * k2)
    k4 = f(t + h, y + h * k3)
    return y + h/6 * (k1 + 2*k2 + 2*k3 + k4)

def three_body_restricted(t, state):
    mu = 0.012277471
    x, y, vx, vy = state
    r1 = np.sqrt((x + mu)**2 + y**2)
    r2 = np.sqrt((x - 1 + mu)**2 + y**2)
    ax = 2*vy + x - (1-mu)*(x+mu)/r1**3 - mu*(x-1+mu)/r2**3
    ay = -2*vx + y - (1-mu)*y/r1**3 - mu*y/r2**3
    return np.array([vx, vy, ax, ay])

自适应步长与嵌入式公式对
固定步长方法的根本缺陷是无法自动适应解的局部变化——在解变化剧烈的区域步长过大导致精度不足，在解平坦的区域步长过小浪费计算。自适应步长 RK 方法通过嵌入式公式对（embedded pair）解决这一问题。


嵌入式公式对使用相同的阶段 kik_iki​，但用两组不同的权重 bib_ibi​ 和 b^i\hat{b}_ib^i​ 得到两个不同精度阶的近似 yn+1y_{n+1}yn+1​ 和 y^n+1\hat{y}_{n+1}y^​n+1​。两者之差提供误差估计，用于自动调整步长。


Dormand-Prince 5(4) 方法（DOPRI5）是最常用的自适应 RK 方法，它使用 7 个阶段产生 5 阶和 4 阶两个近似，且具有 FSAL（First Same As Last）性质——第 7 个阶段的 k7k_7k7​ 等于下一步的 k1k_1k1​，实际每步只需 6 次新的 fff 求值。


import numpy as np

DOPRI5_A = np.array([
    [0, 0, 0, 0, 0, 0, 0],
    [1/5, 0, 0, 0, 0, 0, 0],
    [3/40, 9/40, 0, 0, 0, 0, 0],
    [44/45, -56/15, 32/9, 0, 0, 0, 0],
    [19372/6561, -25360/2187, 64448/6561, -212/729, 0, 0, 0],
    [9017/3168, -355/33, 46732/5247, 49/176, -5103/18656, 0, 0],
    [35/384, 0, 500/1113, 125/192, -2187/6784, 11/84, 0]
])
DOPRI5_b5 = np.array([35/384, 0, 500/1113, 125/192, -2187/6784, 11/84, 0])
DOPRI5_b4 = np.array([5179/57600, 0, 7571/16695, 393/640, -92097/339200, 187/2100, 1/40])
DOPRI5_c = np.array([0, 1/5, 3/10, 4/5, 8/9, 1, 1])

def dopri5_solve(f, t_span, y0, tol=1e-8, h_init=0.01, h_min=1e-12, h_max=1.0):
    t0, tf = t_span
    t = t0
    y = np.array(y0, dtype=np.float64)
    h = h_init

    ts = [t]
    ys = [y.copy()]
    n_accept = 0
    n_reject = 0

    while t < tf:
        h = min(h, tf - t)
        k = np.zeros((7,) + y.shape)

        for i in range(7):
            y_stage = y + h * np.dot(DOPRI5_A[i, :i], k[:i]) if i > 0 else y
            k[i] = f(t + DOPRI5_c[i] * h, y_stage)

        y5 = y + h * np.dot(DOPRI5_b5, k)
        y4 = y + h * np.dot(DOPRI5_b4, k)

        err = np.linalg.norm(y5 - y4)
        err_ratio = tol / max(err, 1e-30)

        if err <= tol:
            t += h
            y = y5
            ts.append(t)
            ys.append(y.copy())
            n_accept += 1
        else:
            n_reject += 1

        h_new = h * min(5.0, max(0.2, 0.9 * err_ratio**0.2))
        h = max(h_min, min(h_max, h_new))

    return np.array(ts), np.array(ys), n_accept, n_reject

def van_der_pol(t, y):
    mu = 20
    return np.array([y[1], mu * (1 - y[0]**2) * y[1] - y[0]])

y0 = np.array([2.0, 0.0])
ts, ys, n_acc, n_rej = dopri5_solve(van_der_pol, (0, 40), y0, tol=1e-8)

print(f"Van der Pol 振荡器 (μ=20):")
print(f"  接受步数: {n_acc}")
print(f"  拒绝步数: {n_rej}")
print(f"  步长范围: [{np.min(np.diff(ts)):.6e}, {np.max(np.diff(ts)):.6e}]")
print(f"  终态: y = [{ys[-1,0]:.6f}, {ys[-1,1]:.6f}]")
步长控制公式 hnew=h⋅min⁡(fmax⁡,max⁡(fmin⁡,S⋅(tol/err)1/(p+1)))h_{\text{new}} = h \cdot \min(f_{\max}, \max(f_{\min}, S \cdot (\text{tol}/\text{err})^{1/(p+1)}))hnew​=h⋅min(fmax​,max(fmin​,S⋅(tol/err)1/(p+1))) 中，S≈0.9S \approx 0.9S≈0.9 是安全系数，fmax⁡f_{\max}fmax​ 和 fmin⁡f_{\min}fmin​ 限制步长变化幅度，ppp 是低阶方法的阶数。


稠密输出与事件检测
实际应用中，常需要在非网格点处获取数值解（稠密输出），或精确找到解满足某一条件的时刻（事件检测）。现代 RK 求解器通过内插多项式（continuous extension）提供稠密输出能力。


Dormand-Prince 方法的稠密输出利用已计算的阶段 kik_iki​ 构造一个 4 阶 Hermite 插值多项式，在 [tn,tn+1][t_n, t_{n+1}][tn​,tn+1​] 内任意点提供一致精度的近似值，无需额外的函数求值。


import numpy as np

def dopri5_dense_coeff(h, k, y):
    d1 = y
    d2 = h * k[0]
    d3 = (h * (k[0] - k[6]) * 2 +
          h * (DOPRI5_b5[0]*k[0] + DOPRI5_b5[2]*k[2] + DOPRI5_b5[3]*k[3] +
               DOPRI5_b5[4]*k[4] + DOPRI5_b5[5]*k[5]) * (-2) +
          h * k[6] * 2)
    return d1, d2, d3

def event_detection_example():
    g = 9.81
    f = lambda t, y: np.array([y[1], -g])
    y0 = np.array([0.0, 20.0])

    ts, ys, _, _ = dopri5_solve(f, (0, 5), y0, tol=1e-10)

    for i in range(1, len(ts)):
        if ys[i, 0] < 0 and ys[i-1, 0] >= 0:
            t_low, t_high = ts[i-1], ts[i]
            for _ in range(50):
                t_mid = (t_low + t_high) / 2
                idx = i - 1
                frac = (t_mid - ts[idx]) / (ts[idx+1] - ts[idx])
                y_mid = (1-frac) * ys[idx, 0] + frac * ys[idx+1, 0]
                if y_mid > 0:
                    t_low = t_mid
                else:
                    t_high = t_mid

            t_impact = (t_low + t_high) / 2
            t_exact = 2 * y0[1] / g
            print(f"抛体落地事件检测:")
            print(f"  检测落地时间: {t_impact:.10f}")
            print(f"  精确落地时间: {t_exact:.10f}")
            print(f"  误差: {abs(t_impact - t_exact):.2e}")
            break

小结
    

  • Runge-Kutta 方法通过多阶段斜率加权达到高阶精度，Butcher 表提供统一描述框架
    • 

  • 经典 RK4 以 4 次函数求值达到四阶精度，是固定步长 ODE 求解的首选
    • 

  • 嵌入式公式对利用同一组阶段产生不同精度估计，实现"免费"误差估计
    • 

  • Dormand-Prince 5(4) 方法具有 FSAL 性质，是 MATLAB ode45 和 SciPy solve_ivp 的默认方法
    • 

  • 自适应步长控制在解变化剧烈区域自动缩小步长，平坦区域放大步长
    • 

  • 稠密输出和事件检测是现代 RK 求解器的标准功能，支持非网格点插值和精确事件定位
