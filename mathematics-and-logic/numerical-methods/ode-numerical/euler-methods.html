欧拉方法

核心概念

欧拉方法是求解常微分方程初值问题（IVP）y′=f(t,y),  y(t0)=y0y' = f(t, y), \; y(t_0) = y_0y′=f(t,y),y(t0​)=y0​ 最基本的数值方法。它直接源于导数的有限差分近似：用差商 yn+1−ynh\frac{y_{n+1} - y_n}{h}hyn+1​−yn​​ 替代 y′(tn)y'(t_n)y′(tn​)，得到显式欧拉公式 yn+1=yn+hf(tn,yn)y_{n+1} = y_n + h f(t_n, y_n)yn+1​=yn​+hf(tn​,yn​)。虽然欧拉方法精度不高（一阶），但它是理解所有高阶方法（如 Runge-Kutta、多步法）的概念基础。

欧拉方法的几何意义是在每一步沿当前点的切线方向前进一个步长 hhh。由于真实解的曲率使得切线逐渐偏离真实轨迹，每一步都会引入局部截断误差。这些局部误差的累积决定了全局精度。显式欧拉方法是条件稳定的——步长过大会导致数值解发散，稳定性对步长的约束在实际计算中至关重要。

显式欧拉方法
显式欧拉方法是最直接的 ODE 数值解法。在每个时间步 tnt_ntn​，用当前状态 (tn,yn)(t_n, y_n)(tn​,yn​) 处的斜率 f(tn,yn)f(t_n, y_n)f(tn​,yn​) 外推下一时刻的值。对于标量方程和向量方程组，形式完全相同。


import numpy as np

def explicit_euler(f, t_span, y0, h):
    t0, tf = t_span
    t = np.arange(t0, tf + h/2, h)
    n = len(t)
    y = np.zeros((n, len(y0))) if hasattr(y0, '__len__') else np.zeros(n)

    y[0] = y0
    for i in range(n - 1):
        y[i + 1] = y[i] + h * f(t[i], y[i])

    return t, y

f = lambda t, y: -2 * y + np.exp(-t)
y_exact = lambda t: np.exp(-t) - np.exp(-2*t)

y0 = 0.0
t_span = (0, 5)

print(f"{'步长 h':<10} {'t=1 误差':<15} {'t=5 误差':<15} {'收敛阶':<10}")
prev_err = None
for h in [0.1, 0.05, 0.025, 0.0125]:
    t, y = explicit_euler(f, t_span, np.array([y0]), h)
    idx1 = int(1.0 / h)
    idx5 = int(5.0 / h)
    err1 = abs(y[idx1, 0] - y_exact(1.0))
    err5 = abs(y[idx5, 0] - y_exact(5.0))
    rate = "" if prev_err is None else f"{np.log2(prev_err / err1):.2f}"
    print(f"{h:<10} {err1:<15.6e} {err5:<15.6e} {rate:<10}")
    prev_err = err1

print("\n向量方程组示例: Lotka-Volterra 捕食者-猎物模型")
def lotka_volterra(t, y):
    alpha, beta, delta, gamma = 1.5, 1.0, 0.3, 1.0
    dydt = np.zeros(2)
    dydt[0] = alpha * y[0] - beta * y[0] * y[1]
    dydt[1] = delta * y[0] * y[1] - gamma * y[1]
    return dydt

t, y = explicit_euler(lotka_volterra, (0, 20), np.array([10.0, 5.0]), 0.001)
print(f"t=20: 猎物={y[-1,0]:.4f}, 捕食者={y[-1,1]:.4f}")

隐式欧拉方法
隐式欧拉方法使用下一时刻的斜率 f(tn+1,yn+1)f(t_{n+1}, y_{n+1})f(tn+1​,yn+1​) 来更新：yn+1=yn+hf(tn+1,yn+1)y_{n+1} = y_n + hf(t_{n+1}, y_{n+1})yn+1​=yn​+hf(tn+1​,yn+1​)。由于 yn+1y_{n+1}yn+1​ 同时出现在等式两端，需要在每一步求解一个（通常是非线性的）方程。


隐式欧拉方法的核心优势是无条件稳定性：对于测试方程 y′=λyy' = \lambda yy′=λy（Re(λ)<0\text{Re}(\lambda) < 0Re(λ)<0），隐式欧拉的增长因子为 11−hλ\frac{1}{1 - h\lambda}1−hλ1​，对任意步长 h>0h > 0h>0 其模都小于 1。这使得隐式欧拉特别适合刚性方程。


import numpy as np

def implicit_euler(f, df_dy, t_span, y0, h, newton_tol=1e-12, max_iter=50):
    t0, tf = t_span
    t = np.arange(t0, tf + h/2, h)
    n = len(t)
    y = np.zeros(n)
    y[0] = y0

    for i in range(n - 1):
        y_guess = y[i] + h * f(t[i], y[i])

        for _ in range(max_iter):
            F = y_guess - y[i] - h * f(t[i+1], y_guess)
            J = 1 - h * df_dy(t[i+1], y_guess)
            delta = -F / J
            y_guess += delta
            if abs(delta) < newton_tol:
                break

        y[i + 1] = y_guess

    return t, y

f = lambda t, y: -50 * (y - np.cos(t))
df_dy = lambda t, y: -50.0
y_exact = lambda t: (50*np.cos(t) + np.sin(t)) / 2501 + (2500/2501) * np.exp(-50*t)

print("显式 vs 隐式欧拉 (刚性方程 y' = -50(y - cos(t))):")
for h in [0.05, 0.04, 0.03, 0.02, 0.01]:
    t_ex, y_ex = explicit_euler(f, (0, 1), np.array([1.0]), h)
    t_im, y_im = implicit_euler(f, df_dy, (0, 1), 1.0, h)

    err_ex = np.max(np.abs(y_ex[:, 0] - y_exact(t_ex)))
    err_im = np.max(np.abs(y_im - y_exact(t_im)))
    stable_ex = "稳定" if err_ex < 10 else "发散"
    print(f"h={h:.2f}: 显式最大误差={err_ex:.2e}({stable_ex}), 隐式最大误差={err_im:.2e}")
改进欧拉方法与预测-校正
改进欧拉方法（也称 Heun 方法或梯形法）结合了显式和隐式欧拉的思想。先用显式欧拉做预测 y~n+1=yn+hfn\tilde{y}_{n+1} = y_n + hf_ny~​n+1​=yn​+hfn​，然后用梯形公式做校正 yn+1=yn+h2[fn+f(tn+1,y~n+1)]y_{n+1} = y_n + \frac{h}{2}[f_n + f(t_{n+1}, \tilde{y}_{n+1})]yn+1​=yn​+2h​[fn​+f(tn+1​,y~​n+1​)]，达到二阶精度。


这是预测-校正（predictor-corrector）框架的最简形式：预测步提供初始猜测，校正步利用更多信息提高精度。改进欧拉方法实质上是二阶显式 Runge-Kutta 方法（RK2）。


import numpy as np

def improved_euler(f, t_span, y0, h):
    t0, tf = t_span
    t = np.arange(t0, tf + h/2, h)
    n = len(t)
    y = np.zeros((n, len(y0))) if hasattr(y0, '__len__') else np.zeros(n)
    y[0] = y0

    for i in range(n - 1):
        k1 = f(t[i], y[i])
        y_predict = y[i] + h * k1
        k2 = f(t[i+1], y_predict)
        y[i + 1] = y[i] + h/2 * (k1 + k2)

    return t, y

def midpoint_method(f, t_span, y0, h):
    t0, tf = t_span
    t = np.arange(t0, tf + h/2, h)
    n = len(t)
    y = np.zeros((n, len(y0))) if hasattr(y0, '__len__') else np.zeros(n)
    y[0] = y0

    for i in range(n - 1):
        k1 = f(t[i], y[i])
        y_mid = y[i] + h/2 * k1
        k2 = f(t[i] + h/2, y_mid)
        y[i + 1] = y[i] + h * k2

    return t, y

f = lambda t, y: np.array([-y[1], y[0]])
y_exact = lambda t: np.array([np.cos(t), np.sin(t)])

y0 = np.array([1.0, 0.0])
t_span = (0, 10)

print(f"{'方法':<15} {'步长':<10} {'t=10 误差':<15} {'收敛阶':<10}")
for name, method in [("显式欧拉", explicit_euler), ("改进欧拉", improved_euler), ("中点法", midpoint_method)]:
    prev_err = None
    for h in [0.1, 0.05, 0.025]:
        t, y = method(f, t_span, y0, h)
        err = np.linalg.norm(y[-1] - y_exact(t[-1]))
        rate = "" if prev_err is None else f"{np.log2(prev_err / err):.2f}"
        print(f"{name:<15} {h:<10} {err:<15.6e} {rate:<10}")
        prev_err = err

稳定性分析
稳定性分析是 ODE 数值方法理论的核心。标准方法是将数值方法应用于 Dahlquist 测试方程 y′=λyy' = \lambda yy′=λy，其中 λ∈C\lambda \in \mathbb{C}λ∈C。定义 z=hλz = h\lambdaz=hλ，数值方法的增长因子 R(z)R(z)R(z) 满足 yn+1=R(z)yny_{n+1} = R(z) y_nyn+1​=R(z)yn​。绝对稳定域是使 ∣R(z)∣≤1|R(z)| \leq 1∣R(z)∣≤1 的 zzz 的集合。


显式欧拉的增长因子为 R(z)=1+zR(z) = 1 + zR(z)=1+z，其稳定域是以 (−1,0)(-1, 0)(−1,0) 为圆心、半径为 1 的圆盘。这意味着对于实负特征值 λ<0\lambda < 0λ<0，步长必须满足 h<2/∣λ∣h < 2/|\lambda|h<2/∣λ∣。


import numpy as np

def stability_region_euler_explicit(N=200):
    theta = np.linspace(0, 2*np.pi, N)
    z = -1 + np.exp(1j * theta)
    return z.real, z.imag

def stability_region_euler_implicit(N=200):
    theta = np.linspace(0, 2*np.pi, N)
    z = 1 + np.exp(1j * theta)
    return z.real, z.imag

lambda_val = -10.0
print("稳定性约束 (λ = -10):")
print(f"  显式欧拉: h < {2/abs(lambda_val):.4f}")
print(f"  隐式欧拉: 无约束")

print("\n稳定性实验:")
for h in [0.25, 0.20, 0.15, 0.10, 0.05]:
    z = h * lambda_val
    R_explicit = abs(1 + z)
    R_implicit = abs(1 / (1 - z))
    stable_ex = "稳定" if R_explicit <= 1 else "不稳定"
    print(f"  h={h:.2f}: |R_显式|={R_explicit:.4f}({stable_ex}), |R_隐式|={R_implicit:.4f}(稳定)")

print("\n多步长精度-稳定性权衡:")
f = lambda t, y: np.array([-10 * y[0]])
y_exact = lambda t: np.exp(-10 * t)
y0 = np.array([1.0])

for h in [0.21, 0.20, 0.19, 0.10, 0.05]:
    t, y = explicit_euler(f, (0, 2), y0, h)
    final_err = abs(y[-1, 0] - y_exact(t[-1]))
    max_val = np.max(np.abs(y[:, 0]))
    status = "发散" if max_val > 10 else "稳定"
    print(f"  h={h:.2f}: 终值误差={final_err:.2e}, 最大值={max_val:.2e} ({status})")
小结
    

  • 显式欧拉方法是最基本的 ODE 数值解法，一阶精度、条件稳定、实现简单
    • 

  • 隐式欧拉以无条件稳定性换取每步需求解非线性方程的代价，适合刚性问题
    • 

  • 改进欧拉（Heun 方法）通过预测-校正达到二阶精度，是 RK2 的特例
    • 

  • 稳定性分析通过 Dahlquist 测试方程确定方法的稳定域，指导步长选择
    • 

  • 显式欧拉的稳定域是单位圆盘，刚性问题中步长严格受限于 h<2/∣λmax⁡∣h < 2/|\lambda_{\max}|h<2/∣λmax​∣
    • 

  • 欧拉方法虽然精度有限，但为理解高阶 Runge-Kutta 和多步法奠定了概念基础
