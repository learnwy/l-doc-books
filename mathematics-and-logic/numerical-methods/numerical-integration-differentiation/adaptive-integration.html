自适应积分

核心概念

自适应积分是一种智能化的数值积分策略，它根据被积函数的局部特征动态调整子区间的划分密度。与等距复合求积不同，自适应积分在函数变化剧烈的区域使用更多节点，在平坦区域使用较少节点，从而以最少的函数求值次数达到预设精度。

自适应积分的核心机制是误差估计与区间细分的反馈循环：在每个子区间上，通过比较不同阶求积公式的结果来估计局部误差；如果误差超过容差，就将该子区间一分为二继续处理。这种策略使得计算资源自动集中在"困难"区域，是工业级数值积分库（如 MATLAB 的 integral、SciPy 的 quad）的基础。

自适应 Simpson 算法
自适应 Simpson 是最直观的自适应积分算法。其基本思想是：在区间 [a,b][a,b][a,b] 上同时计算 Simpson 公式 S1S_1S1​ 和将区间对半分后的两段 Simpson 公式之和 S2=SL+SRS_2 = S_L + S_RS2​=SL​+SR​。利用 Simpson 公式的 O(h5)O(h^5)O(h5) 误差特征，可以证明 ∣S2−S1∣/15|S_2 - S_1|/15∣S2​−S1​∣/15 是 S2S_2S2​ 的误差估计。


import numpy as np

def adaptive_simpson(f, a, b, tol, max_depth=50):
    def _simpson(f, a, b):
        m = (a + b) / 2
        return (b - a) / 6 * (f(a) + 4*f(m) + f(b))

    def _recursive(f, a, b, tol, S, fa, fm, fb, depth):
        m = (a + b) / 2
        fl = f((a + m) / 2)
        fr = f((m + b) / 2)
        S_left = (m - a) / 6 * (fa + 4*fl + fm)
        S_right = (b - m) / 6 * (fm + 4*fr + fb)
        S2 = S_left + S_right
        error = (S2 - S) / 15

        if depth >= max_depth or abs(error) < tol:
            return S2 + error

        return (_recursive(f, a, m, tol/2, S_left, fa, fl, fm, depth+1) +
                _recursive(f, m, b, tol/2, S_right, fm, fr, fb, depth+1))

    m = (a + b) / 2
    fa, fm, fb = f(a), f(m), f(b)
    S = _simpson(f, a, b)
    return _recursive(f, a, b, tol, S, fa, fm, fb, 0)

f = lambda x: np.sin(10 * x) * np.exp(-x)
a, b = 0, np.pi
from scipy.integrate import quad
exact, _ = quad(f, a, b)

for tol in [1e-4, 1e-6, 1e-8, 1e-10, 1e-12]:
    result = adaptive_simpson(f, a, b, tol)
    actual_err = abs(result - exact)
    print(f"容差={tol:.0e}  结果={result:.12f}  实际误差={actual_err:.2e}")
自适应 Simpson 的关键优化是函数值复用：每次区间二分时，已有的三个函数值（端点和中点）可以直接使用，只需额外计算两个新的四分点值。这使得每次细分只需 2 次新的函数求值。


Gauss-Kronrod 自适应策略
Gauss-Kronrod 方法是现代高性能积分库的核心。其思想是构造一对嵌套的求积公式：nnn 点 Gauss 公式 GnG_nGn​ 和 2n+12n+12n+1 点 Kronrod 扩展 K2n+1K_{2n+1}K2n+1​，使得 Kronrod 公式包含所有 Gauss 节点。两个公式之差 ∣K2n+1−Gn∣|K_{2n+1} - G_n|∣K2n+1​−Gn​∣ 用作误差估计。


这种嵌套设计的优势在于：Kronrod 公式的 2n+12n+12n+1 个函数值全部有效利用（Gauss 节点的函数值不需重新计算），误差估计几乎"免费"。最常用的是 G7/K15G_7/K_{15}G7​/K15​ 配对（即 SciPy quad 的默认策略）。


import numpy as np

GK15_NODES = np.array([
    -0.991455371120813, -0.949107912342759, -0.864864423359769,
    -0.741531185599394, -0.586087235467691, -0.405845151377397,
    -0.207784955007898, 0.0,
    0.207784955007898, 0.405845151377397, 0.586087235467691,
    0.741531185599394, 0.864864423359769, 0.949107912342759,
    0.991455371120813
])

GK15_WEIGHTS = np.array([
    0.022935322010529, 0.063092092629979, 0.104790010322250,
    0.140653259715525, 0.169004726639268, 0.190350578064785,
    0.204432940075298, 0.209482141084728,
    0.204432940075298, 0.190350578064785, 0.169004726639268,
    0.140653259715525, 0.104790010322250, 0.063092092629979,
    0.022935322010529
])

G7_WEIGHTS = np.array([
    0.129484966168870, 0.279705391489277, 0.381830050505119,
    0.417959183673469,
    0.381830050505119, 0.279705391489277, 0.129484966168870
])

G7_INDICES = [1, 3, 5, 7, 9, 11, 13]

def gauss_kronrod_single(f, a, b):
    mid = (a + b) / 2
    half = (b - a) / 2

    fvals = f(mid + half * GK15_NODES)
    K15 = half * np.dot(GK15_WEIGHTS, fvals)
    G7 = half * np.dot(G7_WEIGHTS, fvals[G7_INDICES])

    error_est = abs(K15 - G7)
    return K15, error_est

def adaptive_gauss_kronrod(f, a, b, tol, max_intervals=1000):
    import heapq

    result, error = gauss_kronrod_single(f, a, b)
    total_result = result
    total_error = error

    heap = [(-error, a, b, result)]
    n_intervals = 1

    while total_error > tol and n_intervals < max_intervals:
        neg_err, ai, bi, old_result = heapq.heappop(heap)

        mid = (ai + bi) / 2
        result_l, error_l = gauss_kronrod_single(f, ai, mid)
        result_r, error_r = gauss_kronrod_single(f, mid, bi)

        total_result = total_result - old_result + result_l + result_r
        total_error = total_error + neg_err + error_l + error_r

        heapq.heappush(heap, (-error_l, ai, mid, result_l))
        heapq.heappush(heap, (-error_r, mid, bi, result_r))
        n_intervals += 1

    return total_result, total_error, n_intervals

f = lambda x: np.log(x) * np.sqrt(x)
a, b = 0.001, 1

from scipy.integrate import quad
exact, _ = quad(f, a, b)

result, error_est, n_int = adaptive_gauss_kronrod(f, a, b, 1e-10)
print(f"Gauss-Kronrod 结果: {result:.12f}")
print(f"误差估计: {error_est:.2e}")
print(f"实际误差: {abs(result - exact):.2e}")
print(f"子区间数: {n_int}")
全局自适应策略使用优先队列维护所有子区间，每次选取误差最大的区间进行细分。这种贪心策略确保了计算资源始终投向"回报"最大的区域。


奇异积分处理
许多实际问题涉及被积函数在端点或内部存在奇异性的积分，如 ∫01x−1/2sin⁡(x)dx\int_0^1 x^{-1/2} \sin(x) dx∫01​x−1/2sin(x)dx 或 ∫01ln⁡(x)f(x)dx\int_0^1 \ln(x) f(x) dx∫01​ln(x)f(x)dx。标准求积公式在奇异点附近收敛缓慢甚至发散，需要专门的处理技术。


变量替换是处理端点奇异的经典方法。IMT 变换 x=ϕ(t)x = \phi(t)x=ϕ(t) 将奇异点映射到变换后函数的零点，使得被积函数变得光滑。双指数（DE）变换 x=tanh⁡(π2sinh⁡(t))x = \tanh(\frac{\pi}{2}\sinh(t))x=tanh(2π​sinh(t)) 能将 [−1,1][-1,1][−1,1] 上的端点奇异积分转化为全轴上快速衰减的积分，然后用等距梯形公式（具有指数收敛性）求解。


import numpy as np

def de_transform(f, a, b, N, h=0.1):
    t = np.arange(-N, N+1) * h

    phi = np.tanh(np.pi/2 * np.sinh(t))
    dphi = np.pi/2 * np.cosh(t) / np.cosh(np.pi/2 * np.sinh(t))**2

    x = (b - a) / 2 * phi + (a + b) / 2
    mask = (x > a) & (x < b)

    integrand = np.zeros_like(t)
    integrand[mask] = f(x[mask]) * dphi[mask]

    return (b - a) / 2 * h * np.sum(integrand)

f_singular = lambda x: np.sin(x) / np.sqrt(x)
a, b = 0, 1

from scipy.integrate import quad
exact, _ = quad(f_singular, a, b)

print("双指数变换处理端点奇异积分:")
for N in [10, 20, 40, 80]:
    result = de_transform(f_singular, a, b, N)
    err = abs(result - exact)
    print(f"  N={N:3d}: 结果={result:.12f}, 误差={err:.2e}")

f_log = lambda x: np.log(x) * np.cos(x) if x > 0 else 0
exact_log, _ = quad(f_log, 0, 1)

print("\n对数奇异积分:")
for N in [10, 20, 40, 80]:
    result = de_transform(f_log, 0, 1, N)
    err = abs(result - exact_log)
    print(f"  N={N:3d}: 结果={result:.12f}, 误差={err:.2e}")

多维积分与 Monte Carlo
高维积分是数值积分中最具挑战性的领域。一维求积公式的张量积扩展到 ddd 维时，计算量呈指数增长（"维数灾难"）：nnn 点一维公式的 ddd 维张量积需要 ndn^dnd 个节点。当 d≥4d \geq 4d≥4 时，Monte Carlo 方法成为更实用的选择。


Monte Carlo 积分的基本思想是将积分视为期望值：∫Ωf(x)dx=∣Ω∣⋅E[f(X)]\int_{\Omega} f(x)dx = |\Omega| \cdot E[f(X)]∫Ω​f(x)dx=∣Ω∣⋅E[f(X)]，其中 XXX 在 Ω\OmegaΩ 上均匀分布。用 NNN 个随机样本的均值估计期望值，误差为 O(1/N)O(1/\sqrt{N})O(1/N​)，与维度无关。


import numpy as np

def monte_carlo_integrate(f, a, b, N, d=1):
    rng = np.random.default_rng(42)
    samples = rng.uniform(a, b, size=(N, d))
    volume = (b - a)**d
    fvals = np.array([f(x) for x in samples])
    mean = np.mean(fvals)
    std_err = np.std(fvals) / np.sqrt(N)
    return volume * mean, volume * std_err

def sphere_volume_mc(d, N):
    rng = np.random.default_rng(42)
    points = rng.uniform(-1, 1, size=(N, d))
    inside = np.sum(points**2, axis=1) <= 1
    volume = 2**d * np.mean(inside)
    std_err = 2**d * np.std(inside) / np.sqrt(N)
    return volume, std_err

from scipy.special import gamma as gamma_func

print("Monte Carlo 计算高维球体积:")
for d in [2, 3, 5, 8, 10]:
    exact_vol = np.pi**(d/2) / gamma_func(d/2 + 1)
    vol, err = sphere_volume_mc(d, 100000)
    rel_err = abs(vol - exact_vol) / exact_vol
    print(f"  d={d:2d}: MC={vol:.4f}, 精确={exact_vol:.4f}, 相对误差={rel_err:.4f}")

print("\n一维对比 (sin(x), [0, π]):")
f_1d = lambda x: np.sin(x[0])
from scipy.integrate import quad
exact_1d, _ = quad(np.sin, 0, np.pi)

for N in [100, 1000, 10000, 100000]:
    val, err = monte_carlo_integrate(f_1d, 0, np.pi, N, d=1)
    print(f"  N={N:6d}: MC={val:.6f}, 误差={abs(val-exact_1d):.4e}, 标准误差={err:.4e}")
小结
    

  • 自适应积分根据函数局部特征动态分配计算资源，是现代数值积分库的基础
    • 

  • 自适应 Simpson 通过粗细 Simpson 差估计误差，实现简单且函数值可复用
    • 

  • Gauss-Kronrod 嵌套公式对提供高效误差估计，G7/K15G_7/K_{15}G7​/K15​ 是工业标准配置
    • 

  • 全局自适应策略使用优先队列选取最大误差区间细分，确保资源最优分配
    • 

  • 奇异积分需要变量替换（如 DE 变换）或奇异性减除等专门处理
    • 

  • 高维积分面临维数灾难，Monte Carlo 方法以 O(1/N)O(1/\sqrt{N})O(1/N​) 收敛率成为高维积分的主要工具
