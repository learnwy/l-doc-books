多重积分

核心概念

多重积分将定积分从一维推广到多维空间。二重积分 ∬_D f(x,y) dA 计算函数在平面区域 D 上的"体积"，三重积分 ∭_V f(x,y,z) dV 则在空间区域上进行累加。从程序员角度看，多重积分就是嵌套循环的连续版本——将离散的网格求和推广为无穷细分的极限。在机器学习中，多重积分用于计算多维概率分布的边缘分布和期望值；在物理仿真中，它用于质量、重心和转动惯量的计算。

累次积分与积分次序
二重积分的计算通过 Fubini 定理转化为累次积分：∬D f(x,y) dA = ∫_a^b [∫{g₁(x)}^{g₂(x)} f(x,y) dy] dx。交换积分次序有时可以显著简化计算，但需要重新确定积分限。


const doubleIntegral = (
  f: (x: number, y: number) => number,
  xRange: [number, number],
  yRange: (x: number) => [number, number],
  nx: number = 100,
  ny: number = 100,
): number => {
  const [xa, xb] = xRange;
  const hx = (xb - xa) / nx;
  let sum = 0;
  for (let i = 0; i <= nx; i++) {
    const x = xa + i * hx;
    const wx = i === 0 || i === nx ? 0.5 : 1;
    const [ya, yb] = yRange(x);
    const hy = (yb - ya) / ny;
    let innerSum = 0;
    for (let j = 0; j <= ny; j++) {
      const y = ya + j * hy;
      const wy = j === 0 || j === ny ? 0.5 : 1;
      innerSum += wy * f(x, y);
    }
    sum += wx * innerSum * hy;
  }
  return sum * hx;
};

const simpsonDouble = (
  f: (x: number, y: number) => number,
  xRange: [number, number],
  yRange: [number, number],
  nx: number = 50,
  ny: number = 50,
): number => {
  const [xa, xb] = xRange;
  const [ya, yb] = yRange;
  const hx = (xb - xa) / (2 * nx);
  const hy = (yb - ya) / (2 * ny);
  let total = 0;
  for (let i = 0; i <= 2 * nx; i++) {
    const x = xa + i * hx;
    const wx = i === 0 || i === 2 * nx ? 1 : i % 2 === 1 ? 4 : 2;
    for (let j = 0; j <= 2 * ny; j++) {
      const y = ya + j * hy;
      const wy = j === 0 || j === 2 * ny ? 1 : j % 2 === 1 ? 4 : 2;
      total += wx * wy * f(x, y);
    }
  }
  return (total * hx * hy) / 9;
};
交换积分次序的核心是将 X-型区域描述转化为 Y-型区域描述。例如 ∫₀¹ ∫ₓ¹ f(x,y) dy dx 交换后变为 ∫₀¹ ∫₀ʸ f(x,y) dx dy。在数值计算中，选择使内层积分限更简单的次序可以提高精度和速度。


坐标变换与 Jacobian
当积分区域具有特定对称性时，坐标变换可以大幅简化计算。极坐标变换 x = r cosθ, y = r sinθ 的 Jacobian 行列式为 r；球坐标变换的 Jacobian 为 r² sinφ。一般坐标变换 (u,v) → (x,y) 的面积元素变为 |∂(x,y)/∂(u,v)| du dv。


const polarDoubleIntegral = (
  f: (x: number, y: number) => number,
  rRange: [number, number],
  thetaRange: [number, number],
  nr: number = 100,
  nTheta: number = 100,
): number => {
  const [ra, rb] = rRange;
  const [ta, tb] = thetaRange;
  const hr = (rb - ra) / nr;
  const ht = (tb - ta) / nTheta;
  let sum = 0;
  for (let i = 0; i <= nr; i++) {
    const r = ra + i * hr;
    const wr = i === 0 || i === nr ? 0.5 : 1;
    for (let j = 0; j <= nTheta; j++) {
      const theta = ta + j * ht;
      const wt = j === 0 || j === nTheta ? 0.5 : 1;
      const x = r * Math.cos(theta);
      const y = r * Math.sin(theta);
      sum += wr * wt * f(x, y) * r;
    }
  }
  return sum * hr * ht;
};

const sphericalTripleIntegral = (
  f: (x: number, y: number, z: number) => number,
  rRange: [number, number],
  phiRange: [number, number],
  thetaRange: [number, number],
  nr: number = 40,
  nPhi: number = 40,
  nTheta: number = 40,
): number => {
  const [ra, rb] = rRange;
  const [pa, pb] = phiRange;
  const [ta, tb] = thetaRange;
  const hr = (rb - ra) / nr;
  const hp = (pb - pa) / nPhi;
  const ht = (tb - ta) / nTheta;
  let sum = 0;
  for (let i = 0; i <= nr; i++) {
    const r = ra + i * hr;
    const wr = i === 0 || i === nr ? 0.5 : 1;
    for (let j = 0; j <= nPhi; j++) {
      const phi = pa + j * hp;
      const wp = j === 0 || j === nPhi ? 0.5 : 1;
      for (let k = 0; k <= nTheta; k++) {
        const theta = ta + k * ht;
        const wt = k === 0 || k === nTheta ? 0.5 : 1;
        const x = r * Math.sin(phi) * Math.cos(theta);
        const y = r * Math.sin(phi) * Math.sin(theta);
        const z = r * Math.cos(phi);
        sum += wr * wp * wt * f(x, y, z) * r * r * Math.sin(phi);
      }
    }
  }
  return sum * hr * hp * ht;
};

const jacobianDeterminant = (
  transform: (uv: [number, number]) => [number, number],
  uv: [number, number],
  h: number = 1e-6,
): number => {
  const [u, v] = uv;
  const [xpu, ypu] = transform([u + h, v]);
  const [xmu, ymu] = transform([u - h, v]);
  const [xpv, ypv] = transform([u, v + h]);
  const [xmv, ymv] = transform([u, v - h]);
  const dxdu = (xpu - xmu) / (2 * h);
  const dydu = (ypu - ymu) / (2 * h);
  const dxdv = (xpv - xmv) / (2 * h);
  const dydv = (ypv - ymv) / (2 * h);
  return Math.abs(dxdu * dydv - dxdv * dydu);
};
蒙特卡洛积分
当维度升高时，传统网格法的计算量呈指数增长（维数灾难），蒙特卡洛方法通过随机采样避免了这个问题。其误差收敛率为 O(1/√N)，与维度无关。在贝叶斯推断、路径积分和高维期望值计算中，蒙特卡洛积分是唯一可行的方案。


const monteCarloIntegral = (
  f: (...x: number[]) => number,
  bounds: Array<[number, number]>,
  numSamples: number = 100000,
): { estimate: number; stdError: number } => {
  const dim = bounds.length;
  const volume = bounds.reduce((v, [a, b]) => v * (b - a), 1);
  let sum = 0;
  let sumSq = 0;
  for (let i = 0; i < numSamples; i++) {
    const point = bounds.map(([a, b]) => a + Math.random() * (b - a));
    const val = f(...point);
    sum += val;
    sumSq += val * val;
  }
  const mean = sum / numSamples;
  const variance = sumSq / numSamples - mean * mean;
  return {
    estimate: volume * mean,
    stdError: volume * Math.sqrt(variance / numSamples),
  };
};

const importanceSampling = (
  f: (...x: number[]) => number,
  proposal: {
    sample: () => number[];
    pdf: (x: number[]) => number;
  },
  numSamples: number = 100000,
): { estimate: number; stdError: number } => {
  let sum = 0;
  let sumSq = 0;
  for (let i = 0; i < numSamples; i++) {
    const x = proposal.sample();
    const w = f(...x) / proposal.pdf(x);
    sum += w;
    sumSq += w * w;
  }
  const mean = sum / numSamples;
  const variance = sumSq / numSamples - mean * mean;
  return {
    estimate: mean,
    stdError: Math.sqrt(variance / numSamples),
  };
};

const stratifiedSampling = (
  f: (x: number, y: number) => number,
  bounds: [[number, number], [number, number]],
  strata: number = 10,
  samplesPerStratum: number = 100,
): number => {
  const [[xa, xb], [ya, yb]] = bounds;
  const hx = (xb - xa) / strata;
  const hy = (yb - ya) / strata;
  const stratumArea = hx * hy;
  let total = 0;
  for (let i = 0; i < strata; i++) {
    for (let j = 0; j < strata; j++) {
      let stratumSum = 0;
      for (let k = 0; k < samplesPerStratum; k++) {
        const x = xa + (i + Math.random()) * hx;
        const y = ya + (j + Math.random()) * hy;
        stratumSum += f(x, y);
      }
      total += (stratumSum / samplesPerStratum) * stratumArea;
    }
  }
  return total;
};

物理与工程应用
多重积分在物理和工程计算中无处不在。质量、重心、转动惯量等物理量都需要对密度函数进行区域积分。在概率论中，联合分布的边缘化和期望值计算也是多重积分问题。


const computeMass = (
  density: (x: number, y: number) => number,
  xRange: [number, number],
  yRange: (x: number) => [number, number],
  n: number = 200,
): number => doubleIntegral(density, xRange, yRange, n, n);

const computeCenterOfMass = (
  density: (x: number, y: number) => number,
  xRange: [number, number],
  yRange: (x: number) => [number, number],
  n: number = 200,
): { cx: number; cy: number } => {
  const mass = doubleIntegral(density, xRange, yRange, n, n);
  const mx = doubleIntegral((x, y) => x * density(x, y), xRange, yRange, n, n);
  const my = doubleIntegral((x, y) => y * density(x, y), xRange, yRange, n, n);
  return { cx: mx / mass, cy: my / mass };
};

const momentOfInertia = (
  density: (x: number, y: number) => number,
  xRange: [number, number],
  yRange: (x: number) => [number, number],
  axis: 'x' | 'y' | 'origin',
  n: number = 200,
): number => {
  const distSq = (x: number, y: number): number => {
    if (axis === 'x') return y * y;
    if (axis === 'y') return x * x;
    return x * x + y * y;
  };
  return doubleIntegral((x, y) => distSq(x, y) * density(x, y), xRange, yRange, n, n);
};

const marginalDistribution = (
  joint: (x: number, y: number) => number,
  yRange: [number, number],
  x: number,
  n: number = 1000,
): number => {
  const [ya, yb] = yRange;
  const h = (yb - ya) / n;
  let sum = 0;
  for (let i = 0; i <= n; i++) {
    const y = ya + i * h;
    const w = i === 0 || i === n ? 0.5 : 1;
    sum += w * joint(x, y);
  }
  return sum * h;
};
小结
    

  • 多重积分通过 Fubini 定理转化为累次积分，交换积分次序可以简化计算
    • 

  • 坐标变换利用区域对称性简化积分，Jacobian 行列式是变换的关键
    • 

  • 高维积分面临维数灾难，蒙特卡洛方法以 O(1/√N) 的维度无关收敛率成为唯一选择
    • 

  • 重要性采样和分层采样通过减小方差改善蒙特卡洛积分的精度
    • 

  • 物理量（质量、重心、转动惯量）的计算本质上都是多重积分问题
