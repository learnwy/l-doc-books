向量微积分

核心概念

向量微积分研究向量场的微分与积分运算。标量场 f: ℝⁿ → ℝ 在每个点赋予一个数值（如温度场），向量场 F: ℝⁿ → ℝⁿ 在每个点赋予一个向量（如速度场、力场）。三个核心微分算子——梯度（grad）、散度（div）、旋度（curl）——构成向量微积分的基础。它们通过 Green、Stokes 和散度定理与线积分、面积分联系起来。在程序中，向量场的计算广泛用于流体模拟、电磁场仿真和图形学中的法向量计算。

梯度、散度与旋度
梯度将标量场变为向量场：∇f = (∂f/∂x, ∂f/∂y, ∂f/∂z)。散度将向量场变为标量场：∇·F = ∂Fx/∂x + ∂Fy/∂y + ∂Fz/∂z，衡量向量场在某点的"发散"程度——正散度表示该点是源，负散度表示是汇。旋度将向量场变为向量场：∇×F，衡量向量场的旋转倾向。


type ScalarField = (x: number, y: number, z: number) => number;
type VectorField = (x: number, y: number, z: number) => [number, number, number];

const gradient = (f: ScalarField, x: number, y: number, z: number, h: number = 1e-6): [number, number, number] => [
  (f(x + h, y, z) - f(x - h, y, z)) / (2 * h),
  (f(x, y + h, z) - f(x, y - h, z)) / (2 * h),
  (f(x, y, z + h) - f(x, y, z - h)) / (2 * h),
];

const divergence = (F: VectorField, x: number, y: number, z: number, h: number = 1e-6): number => {
  const dFxdx = (F(x + h, y, z)[0] - F(x - h, y, z)[0]) / (2 * h);
  const dFydy = (F(x, y + h, z)[1] - F(x, y - h, z)[1]) / (2 * h);
  const dFzdz = (F(x, y, z + h)[2] - F(x, y, z - h)[2]) / (2 * h);
  return dFxdx + dFydy + dFzdz;
};

const curl = (F: VectorField, x: number, y: number, z: number, h: number = 1e-6): [number, number, number] => {
  const dFzdy = (F(x, y + h, z)[2] - F(x, y - h, z)[2]) / (2 * h);
  const dFydz = (F(x, y, z + h)[1] - F(x, y, z - h)[1]) / (2 * h);
  const dFxdz = (F(x, y, z + h)[0] - F(x, y, z - h)[0]) / (2 * h);
  const dFzdx = (F(x + h, y, z)[2] - F(x - h, y, z)[2]) / (2 * h);
  const dFydx = (F(x + h, y, z)[1] - F(x - h, y, z)[1]) / (2 * h);
  const dFxdy = (F(x, y + h, z)[0] - F(x, y - h, z)[0]) / (2 * h);
  return [dFzdy - dFydz, dFxdz - dFzdx, dFydx - dFxdy];
};

const laplacian = (f: ScalarField, x: number, y: number, z: number, h: number = 1e-6): number => {
  const d2x = (f(x + h, y, z) - 2 * f(x, y, z) + f(x - h, y, z)) / (h * h);
  const d2y = (f(x, y + h, z) - 2 * f(x, y, z) + f(x, y - h, z)) / (h * h);
  const d2z = (f(x, y, z + h) - 2 * f(x, y, z) + f(x, y, z - h)) / (h * h);
  return d2x + d2y + d2z;
};
这些算子之间存在重要的恒等关系：∇×(∇f) = 0（梯度的旋度为零）和 ∇·(∇×F) = 0（旋度的散度为零），在物理和数值验证中非常有用。


线积分与面积分
线积分沿曲线对标量场或向量场积分。标量线积分 ∫_C f ds 计算曲线上函数值的累加，向量线积分 ∫_C F·dr 计算力沿路径做的功。面积分 ∬_S F·dS 计算向量场通过曲面的通量。


type Curve = (t: number) => [number, number, number];
type CurveDerivative = (t: number) => [number, number, number];

const lineIntegralScalar = (
  f: ScalarField,
  curve: Curve,
  curveDeriv: CurveDerivative,
  tRange: [number, number],
  n: number = 1000,
): number => {
  const [ta, tb] = tRange;
  const dt = (tb - ta) / n;
  let sum = 0;
  for (let i = 0; i <= n; i++) {
    const t = ta + i * dt;
    const w = i === 0 || i === n ? 0.5 : 1;
    const [x, y, z] = curve(t);
    const [dx, dy, dz] = curveDeriv(t);
    const speed = Math.sqrt(dx * dx + dy * dy + dz * dz);
    sum += w * f(x, y, z) * speed;
  }
  return sum * dt;
};

const lineIntegralVector = (
  F: VectorField,
  curve: Curve,
  curveDeriv: CurveDerivative,
  tRange: [number, number],
  n: number = 1000,
): number => {
  const [ta, tb] = tRange;
  const dt = (tb - ta) / n;
  let sum = 0;
  for (let i = 0; i <= n; i++) {
    const t = ta + i * dt;
    const w = i === 0 || i === n ? 0.5 : 1;
    const [x, y, z] = curve(t);
    const [Fx, Fy, Fz] = F(x, y, z);
    const [dx, dy, dz] = curveDeriv(t);
    sum += w * (Fx * dx + Fy * dy + Fz * dz);
  }
  return sum * dt;
};

const surfaceIntegral = (
  F: VectorField,
  surface: (u: number, v: number) => [number, number, number],
  surfacePartialU: (u: number, v: number) => [number, number, number],
  surfacePartialV: (u: number, v: number) => [number, number, number],
  uRange: [number, number],
  vRange: [number, number],
  nu: number = 100,
  nv: number = 100,
): number => {
  const [ua, ub] = uRange;
  const [va, vb] = vRange;
  const hu = (ub - ua) / nu;
  const hv = (vb - va) / nv;
  let sum = 0;
  for (let i = 0; i <= nu; i++) {
    const u = ua + i * hu;
    const wu = i === 0 || i === nu ? 0.5 : 1;
    for (let j = 0; j <= nv; j++) {
      const v = va + j * hv;
      const wv = j === 0 || j === nv ? 0.5 : 1;
      const [x, y, z] = surface(u, v);
      const [Fx, Fy, Fz] = F(x, y, z);
      const [rux, ruy, ruz] = surfacePartialU(u, v);
      const [rvx, rvy, rvz] = surfacePartialV(u, v);
      const nx = ruy * rvz - ruz * rvy;
      const ny = ruz * rvx - rux * rvz;
      const nz = rux * rvy - ruy * rvx;
      sum += wu * wv * (Fx * nx + Fy * ny + Fz * nz);
    }
  }
  return sum * hu * hv;
};
三大积分定理
Green 定理、Stokes 定理和散度定理是向量微积分的三大核心定理，它们将区域上的积分与边界上的积分联系起来。Green 定理是 Stokes 定理在二维的特例，散度定理（Gauss 定理）将体积分与面积分联系起来。

const verifyGreenTheorem = (
  P: (x: number, y: number) => number,
  Q: (x: number, y: number) => number,
  curve: (t: number) => [number, number],
  curveDeriv: (t: number) => [number, number],
  xRange: [number, number],
  yRange: (x: number) => [number, number],
  n: number = 500,
): { lineIntegral: number; areaIntegral: number; error: number } => {
  const dt = (2 * Math.PI) / n;
  let lineInt = 0;
  for (let i = 0; i <= n; i++) {
    const t = i * dt;
    const w = i === 0 || i === n ? 0.5 : 1;
    const [x, y] = curve(t);
    const [dx, dy] = curveDeriv(t);
    lineInt += w * (P(x, y) * dx + Q(x, y) * dy);
  }
  lineInt *= dt;

  const h = 1e-6;
  const integrand = (x: number, y: number): number => {
    const dQdx = (Q(x + h, y) - Q(x - h, y)) / (2 * h);
    const dPdy = (P(x, y + h) - P(x, y - h)) / (2 * h);
    return dQdx - dPdy;
  };
  const [xa, xb] = xRange;
  const hx = (xb - xa) / n;
  let areaInt = 0;
  for (let i = 0; i <= n; i++) {
    const x = xa + i * hx;
    const wx = i === 0 || i === n ? 0.5 : 1;
    const [ya, yb] = yRange(x);
    const hy = (yb - ya) / n;
    let inner = 0;
    for (let j = 0; j <= n; j++) {
      const y = ya + j * hy;
      const wy = j === 0 || j === n ? 0.5 : 1;
      inner += wy * integrand(x, y);
    }
    areaInt += wx * inner * hy;
  }
  areaInt *= hx;

  return { lineIntegral: lineInt, areaIntegral: areaInt, error: Math.abs(lineInt - areaInt) };
};

const verifyDivergenceTheorem = (
  F: VectorField,
  bounds: Array<[number, number]>,
  n: number = 50,
): { volumeIntegral: number; surfaceIntegral: number; error: number } => {
  const [[xa, xb], [ya, yb], [za, zb]] = bounds;
  const hx = (xb - xa) / n;
  const hy = (yb - ya) / n;
  const hz = (zb - za) / n;
  let volInt = 0;
  for (let i = 0; i <= n; i++) {
    const x = xa + i * hx;
    const wx = i === 0 || i === n ? 0.5 : 1;
    for (let j = 0; j <= n; j++) {
      const y = ya + j * hy;
      const wy = j === 0 || j === n ? 0.5 : 1;
      for (let k = 0; k <= n; k++) {
        const z = za + k * hz;
        const wz = k === 0 || k === n ? 0.5 : 1;
        volInt += wx * wy * wz * divergence(F, x, y, z);
      }
    }
  }
  volInt *= hx * hy * hz;

  let surfInt = 0;
  for (let i = 0; i <= n; i++) {
    for (let j = 0; j <= n; j++) {
      const u = xa + i * hx;
      const v = ya + j * hy;
      const wu = i === 0 || i === n ? 0.5 : 1;
      const wv = j === 0 || j === n ? 0.5 : 1;
      surfInt += wu * wv * (F(u, v, zb)[2] - F(u, v, za)[2]);
      surfInt += wu * wv * (F(u, yb, za + j * hz)[1] - F(u, ya, za + j * hz)[1]);
      surfInt += wu * wv * (F(xb, ya + j * hy, za + i * hz)[0] - F(xa, ya + j * hy, za + i * hz)[0]);
    }
  }
  surfInt *= hx * hy;

  return { volumeIntegral: volInt, surfaceIntegral: surfInt, error: Math.abs(volInt - surfInt) };
};
小结
    

  • 梯度、散度、旋度是向量场的三个基本微分算子，连接标量场与向量场
    • 

  • 线积分计算沿路径的累积量（功、环量），面积分计算通过曲面的通量
    • 

  • Green、Stokes、散度定理将区域积分与边界积分联系起来，是数值验证的基础
    • 

  • ∇×(∇f) = 0 和 ∇·(∇×F) = 0 是两个核心恒等关系
    • 

  • 数值实现中，所有微分算子都可用中心差分近似，积分用梯形法则离散化
