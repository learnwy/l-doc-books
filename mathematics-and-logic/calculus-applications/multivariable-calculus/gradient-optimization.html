梯度与优化方法

核心概念

梯度优化是机器学习的核心计算范式。给定目标函数 f(x)，梯度下降通过 x_{k+1} = x_k - α∇f(x_k) 迭代地向极小值移动。α 是学习率，∇f 是梯度。不同的优化算法本质上是在如何利用梯度信息（一阶或二阶）和如何选择步长上做出不同的设计选择。

梯度下降变体
基本梯度下降的收敛速度受条件数影响。当目标函数的 Hessian 矩阵条件数很大时，梯度方向与最优方向偏差很大，导致锯齿形路径和缓慢收敛。动量法通过累积历史梯度信息来缓解这个问题。


const gradientDescent = (
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  lr: number = 0.01,
  maxIter: number = 10000,
  tol: number = 1e-8,
): { x: number[]; fVal: number; history: number[] } => {
  let x = [...x0];
  const history: number[] = [];
  for (let i = 0; i < maxIter; i++) {
    const fVal = f(x);
    history.push(fVal);
    const g = grad(x);
    const gradNorm = Math.sqrt(g.reduce((a, gi) => a + gi * gi, 0));
    if (gradNorm < tol) return { x, fVal, history };
    x = x.map((xi, j) => xi - lr * g[j]);
  }
  return { x, fVal: f(x), history };
};

const momentumGD = (
  grad: (x: number[]) => number[],
  x0: number[],
  lr: number = 0.01,
  beta: number = 0.9,
  maxIter: number = 10000,
): { x: number[]; iterations: number } => {
  let x = [...x0];
  let v = new Array(x.length).fill(0);
  for (let i = 0; i < maxIter; i++) {
    const g = grad(x);
    v = v.map((vi, j) => beta * vi + g[j]);
    x = x.map((xi, j) => xi - lr * v[j]);
    const gradNorm = Math.sqrt(g.reduce((a, gi) => a + gi * gi, 0));
    if (gradNorm < 1e-8) return { x, iterations: i };
  }
  return { x, iterations: maxIter };
};

const adam = (
  grad: (x: number[]) => number[],
  x0: number[],
  lr: number = 0.001,
  beta1: number = 0.9,
  beta2: number = 0.999,
  epsilon: number = 1e-8,
  maxIter: number = 10000,
): { x: number[]; iterations: number } => {
  let x = [...x0];
  let m = new Array(x.length).fill(0);
  let v = new Array(x.length).fill(0);
  for (let t = 1; t <= maxIter; t++) {
    const g = grad(x);
    m = m.map((mi, j) => beta1 * mi + (1 - beta1) * g[j]);
    v = v.map((vi, j) => beta2 * vi + (1 - beta2) * g[j] * g[j]);
    const mHat = m.map((mi) => mi / (1 - beta1 ** t));
    const vHat = v.map((vi) => vi / (1 - beta2 ** t));
    x = x.map((xi, j) => xi - (lr * mHat[j]) / (Math.sqrt(vHat[j]) + epsilon));
    const gradNorm = Math.sqrt(g.reduce((a, gi) => a + gi * gi, 0));
    if (gradNorm < 1e-8) return { x, iterations: t };
  }
  return { x, iterations: maxIter };
};
牛顿法与拟牛顿法
牛顿法利用二阶信息（Hessian 矩阵）来选择更好的搜索方向：x_{k+1} = x_k - H⁻¹∇f。在极值点附近具有二次收敛，但每步需要 O(n³) 的 Hessian 求逆。拟牛顿法（BFGS、L-BFGS）用低秩更新近似 Hessian 逆，在精度和效率之间取得平衡。


const newtonOptimization = (
  grad: (x: number[]) => number[],
  hessian: (x: number[]) => number[][],
  x0: number[],
  maxIter: number = 100,
  tol: number = 1e-10,
): { x: number[]; iterations: number } => {
  let x = [...x0];
  for (let i = 0; i < maxIter; i++) {
    const g = grad(x);
    const gradNorm = Math.sqrt(g.reduce((a, gi) => a + gi * gi, 0));
    if (gradNorm < tol) return { x, iterations: i };
    const H = hessian(x);
    const direction = solveLinearSystem(
      H,
      g.map((gi) => -gi),
    );
    x = x.map((xi, j) => xi + direction[j]);
  }
  return { x, iterations: maxIter };
};

const solveLinearSystem = (A: number[][], b: number[]): number[] => {
  const n = A.length;
  const aug = A.map((row, i) => [...row, b[i]]);
  for (let col = 0; col < n; col++) {
    let maxRow = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) maxRow = row;
    }
    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
    for (let row = col + 1; row < n; row++) {
      const factor = aug[row][col] / aug[col][col];
      for (let j = col; j <= n; j++) aug[row][j] -= factor * aug[col][j];
    }
  }
  const x = new Array(n);
  for (let i = n - 1; i >= 0; i--) {
    x[i] = aug[i][n];
    for (let j = i + 1; j < n; j++) x[i] -= aug[i][j] * x[j];
    x[i] /= aug[i][i];
  }
  return x;
};

const bfgs = (
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  maxIter: number = 1000,
  tol: number = 1e-8,
): { x: number[]; fVal: number; iterations: number } => {
  const n = x0.length;
  let x = [...x0];
  let H = Array.from({ length: n }, (_, i) => Array.from({ length: n }, (_, j) => (i === j ? 1 : 0)));
  let g = grad(x);

  for (let k = 0; k < maxIter; k++) {
    const gradNorm = Math.sqrt(g.reduce((a, gi) => a + gi * gi, 0));
    if (gradNorm < tol) return { x, fVal: f(x), iterations: k };
    const d = H.map((row) => -row.reduce((a, h, j) => a + h * g[j], 0));
    let alpha = 1;
    const fx = f(x);
    while (f(x.map((xi, j) => xi + alpha * d[j])) > fx + 1e-4 * alpha * g.reduce((a, gi, j) => a + gi * d[j], 0)) {
      alpha *= 0.5;
      if (alpha < 1e-20) break;
    }
    const xNew = x.map((xi, j) => xi + alpha * d[j]);
    const gNew = grad(xNew);
    const s = xNew.map((xi, j) => xi - x[j]);
    const y = gNew.map((gi, j) => gi - g[j]);
    const sy = s.reduce((a, si, j) => a + si * y[j], 0);
    if (sy > 1e-10) {
      const Hy = H.map((row) => row.reduce((a, h, j) => a + h * y[j], 0));
      const yHy = y.reduce((a, yi, j) => a + yi * Hy[j], 0);
      for (let i = 0; i < n; i++) {
        for (let j = 0; j < n; j++) {
          H[i][j] += ((sy + yHy) * s[i] * s[j]) / (sy * sy) - (Hy[i] * s[j] + s[i] * Hy[j]) / sy;
        }
      }
    }
    x = xNew;
    g = gNew;
  }
  return { x, fVal: f(x), iterations: maxIter };
};
线搜索与收敛条件
线搜索确定每步的步长 α，常用 Wolfe 条件确保足够的下降和曲率条件。Armijo 回溯是最简单实用的线搜索策略。


const armijoBacktrack = (
  f: (x: number[]) => number,
  grad: number[],
  x: number[],
  direction: number[],
  c1: number = 1e-4,
  rho: number = 0.5,
): number => {
  let alpha = 1;
  const fx = f(x);
  const slope = grad.reduce((a, g, i) => a + g * direction[i], 0);
  while (f(x.map((xi, i) => xi + alpha * direction[i])) > fx + c1 * alpha * slope) {
    alpha *= rho;
    if (alpha < 1e-20) break;
  }
  return alpha;
};
小结
    

  • 梯度下降是最基本的一阶优化方法，收敛速度受条件数影响
    • 

  • 动量法和 Adam 通过历史信息加速收敛，是深度学习的标准选择
    • 

  • 牛顿法具有二次收敛但计算量大，L-BFGS 在精度和效率间取得平衡
    • 

  • 线搜索（Armijo 回溯）确保每步有足够的目标函数下降
    • 

  • 方法选择取决于问题规模、精度需求和计算预算
