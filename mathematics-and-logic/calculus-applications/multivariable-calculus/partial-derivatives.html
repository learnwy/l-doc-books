偏导数与多元函数

核心概念

偏导数将单变量导数推广到多元函数：对多元函数 f(x₁,...,xₙ)，偏导数 ∂f/∂xᵢ 衡量固定其他变量时 f 对 xᵢ 的变化率。偏导数是机器学习中梯度计算的基础——损失函数通常是成千上万个参数的函数，梯度就是所有偏导数组成的向量。

梯度与方向导数
梯度 ∇f 是多元函数最重要的微分量。它指向函数增长最快的方向，其模等于最大方向导数。方向导数 D_u f = ∇f · u 给出了沿单位向量 u 方向的变化率。


const partialDerivative = (f: (x: number[]) => number, x: number[], i: number, h?: number): number => {
  h = h ?? Math.max(Math.abs(x[i]), 1) * Math.cbrt(Number.EPSILON);
  const xp = [...x];
  xp[i] += h;
  const xm = [...x];
  xm[i] -= h;
  return (f(xp) - f(xm)) / (2 * h);
};

const computeGradient = (f: (x: number[]) => number, x: number[]): number[] =>
  x.map((_, i) => partialDerivative(f, x, i));

const directionalDerivative = (f: (x: number[]) => number, x: number[], direction: number[]): number => {
  const grad = computeGradient(f, x);
  const norm = Math.sqrt(direction.reduce((a, d) => a + d * d, 0));
  const unit = direction.map((d) => d / norm);
  return grad.reduce((a, g, i) => a + g * unit[i], 0);
};

const computeJacobian = (f: (x: number[]) => number[], x: number[]): number[][] => {
  const y = f(x);
  return y.map((_, i) => {
    const fi = (x: number[]) => f(x)[i];
    return computeGradient(fi, x);
  });
};

const computeHessian = (f: (x: number[]) => number, x: number[]): number[][] => {
  const n = x.length;
  const H: number[][] = Array.from({ length: n }, () => new Array(n));
  const h = Math.pow(Number.EPSILON, 0.25);
  for (let i = 0; i < n; i++) {
    for (let j = i; j < n; j++) {
      const xpp = [...x];
      xpp[i] += h;
      xpp[j] += h;
      const xpm = [...x];
      xpm[i] += h;
      xpm[j] -= h;
      const xmp = [...x];
      xmp[i] -= h;
      xmp[j] += h;
      const xmm = [...x];
      xmm[i] -= h;
      xmm[j] -= h;
      H[i][j] = (f(xpp) - f(xpm) - f(xmp) + f(xmm)) / (4 * h * h);
      H[j][i] = H[i][j];
    }
  }
  return H;
};
多元极值与二阶条件
多元函数极值的一阶必要条件是 ∇f = 0（临界点）。二阶充分条件通过 Hessian 矩阵判定：正定则为极小值，负定则为极大值，不定则为鞍点。


type CriticalPointType = 'minimum' | 'maximum' | 'saddle' | 'degenerate';

const classifyCriticalPoint = (H: number[][]): CriticalPointType => {
  const eigenvalues = computeEigenvalues2x2(H);
  if (eigenvalues.every((e) => e > 1e-10)) return 'minimum';
  if (eigenvalues.every((e) => e < -1e-10)) return 'maximum';
  if (eigenvalues.some((e) => e > 1e-10) && eigenvalues.some((e) => e < -1e-10)) return 'saddle';
  return 'degenerate';
};

const computeEigenvalues2x2 = (H: number[][]): number[] => {
  if (H.length !== 2) {
    const trace = H.reduce((a, row, i) => a + row[i], 0);
    return [trace / H.length];
  }
  const a = H[0][0],
    b = H[0][1],
    c = H[1][0],
    d = H[1][1];
  const trace = a + d;
  const det = a * d - b * c;
  const disc = trace * trace - 4 * det;
  if (disc < 0) return [trace / 2, trace / 2];
  return [(trace + Math.sqrt(disc)) / 2, (trace - Math.sqrt(disc)) / 2];
};

const findCriticalPoints = (
  grad: (x: number[]) => number[],
  bounds: Array<[number, number]>,
  gridSize: number = 20,
): number[][] => {
  const criticalPoints: number[][] = [];
  const dim = bounds.length;
  const step = bounds.map(([a, b]) => (b - a) / gridSize);

  const search = (start: number[]): number[] | null => {
    let x = [...start];
    for (let iter = 0; iter < 100; iter++) {
      const g = grad(x);
      const norm = Math.sqrt(g.reduce((a, gi) => a + gi * gi, 0));
      if (norm < 1e-8) return x;
      x = x.map((xi, i) => xi - 0.01 * g[i]);
    }
    return null;
  };

  const indices = Array.from({ length: gridSize ** dim }, (_, i) => {
    const idx = [];
    let rem = i;
    for (let d = 0; d < dim; d++) {
      idx.push(rem % gridSize);
      rem = Math.floor(rem / gridSize);
    }
    return idx;
  });

  for (const idx of indices.slice(0, 1000)) {
    const start = idx.map((k, d) => bounds[d][0] + (k + 0.5) * step[d]);
    const cp = search(start);
    if (cp && !criticalPoints.some((p) => p.every((v, i) => Math.abs(v - cp[i]) < 1e-4))) {
      criticalPoints.push(cp);
    }
  }
  return criticalPoints;
};
链式法则的多元推广
多元链式法则描述了复合函数的偏导数。如果 z = f(u, v)，u = u(x, y)，v = v(x, y)，则 ∂z/∂x = (∂f/∂u)(∂u/∂x) + (∂f/∂v)(∂v/∂x)。这就是反向传播中的链式法则在多元情形下的推广。


const multiChainRule = (outerGrad: number[], jacobian: number[][]): number[] => {
  const inputDim = jacobian[0].length;
  const result = new Array(inputDim).fill(0);
  for (let j = 0; j < inputDim; j++) {
    for (let i = 0; i < outerGrad.length; i++) {
      result[j] += outerGrad[i] * jacobian[i][j];
    }
  }
  return result;
};

class ComputationGraph {
  private nodes: Array<{
    value: number[];
    grad: number[];
    backward: (grad: number[]) => void;
  }> = [];

  variable(value: number[]): number {
    const idx = this.nodes.length;
    this.nodes.push({ value, grad: new Array(value.length).fill(0), backward: () => {} });
    return idx;
  }

  add(a: number, b: number): number {
    const va = this.nodes[a].value;
    const vb = this.nodes[b].value;
    const value = va.map((v, i) => v + vb[i]);
    const idx = this.nodes.length;
    this.nodes.push({
      value,
      grad: new Array(value.length).fill(0),
      backward: (grad) => {
        this.nodes[a].grad = this.nodes[a].grad.map((g, i) => g + grad[i]);
        this.nodes[b].grad = this.nodes[b].grad.map((g, i) => g + grad[i]);
        this.nodes[a].backward(this.nodes[a].grad);
        this.nodes[b].backward(this.nodes[b].grad);
      },
    });
    return idx;
  }

  getGrad(idx: number): number[] {
    return this.nodes[idx].grad;
  }
}
