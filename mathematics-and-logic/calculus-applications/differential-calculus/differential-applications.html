微分学应用

核心概念

微分学的应用涵盖了优化（极值问题）、曲线分析（凹凸性、拐点）、运动分析（速度、加速度）和近似计算（线性化、微分近似）。这些应用构成了机器学习优化、计算机图形学、物理模拟和金融工程的数学基础。

对程序员而言，微分学最重要的应用是优化——梯度下降、牛顿法、拟牛顿法都建立在微分学之上。理解极值条件（一阶必要条件和二阶充分条件）是正确使用和调试优化算法的关键。

极值分析与优化
函数极值的判定基于微分条件：一阶必要条件 f'(x*) = 0（驻点），二阶充分条件 f''(x*) > 0（极小值）或 f''(x*) < 0（极大值）。多元函数的极值条件涉及梯度为零和 Hessian 矩阵的正定性。


interface CriticalPoint {
  x: number;
  y: number;
  type: 'minimum' | 'maximum' | 'inflection' | 'unknown';
  secondDerivative: number;
}

const findCriticalPoints = (
  f: (x: number) => number,
  a: number,
  b: number,
  resolution: number = 1000,
): CriticalPoint[] => {
  const h = (b - a) / resolution;
  const hd = Math.cbrt(Number.EPSILON);
  const points: CriticalPoint[] = [];

  for (let i = 1; i < resolution; i++) {
    const x = a + i * h;
    const fPrev = (f(x - hd) - f(x + hd)) / (-2 * hd);
    const fCurr = (f(x + hd) - f(x - hd)) / (2 * hd);

    if (Math.abs(fCurr) < 1e-6 || fPrev * fCurr < 0) {
      const d2 = (f(x + hd) - 2 * f(x) + f(x - hd)) / (hd * hd);
      let type: CriticalPoint['type'];
      if (d2 > 1e-6) type = 'minimum';
      else if (d2 < -1e-6) type = 'maximum';
      else type = 'inflection';
      points.push({ x, y: f(x), type, secondDerivative: d2 });
    }
  }
  return points;
};

const goldenSectionSearch = (
  f: (x: number) => number,
  a: number,
  b: number,
  tolerance: number = 1e-10,
): { x: number; fx: number; iterations: number } => {
  const phi = (1 + Math.sqrt(5)) / 2;
  const resphi = 2 - phi;
  let x1 = a + resphi * (b - a);
  let x2 = b - resphi * (b - a);
  let f1 = f(x1),
    f2 = f(x2);
  let iterations = 0;

  while (Math.abs(b - a) > tolerance) {
    iterations++;
    if (f1 < f2) {
      b = x2;
      x2 = x1;
      f2 = f1;
      x1 = a + resphi * (b - a);
      f1 = f(x1);
    } else {
      a = x1;
      x1 = x2;
      f1 = f2;
      x2 = b - resphi * (b - a);
      f2 = f(x2);
    }
  }
  const xMin = (a + b) / 2;
  return { x: xMin, fx: f(xMin), iterations };
};

const gradientDescent = (
  grad: (x: number[]) => number[],
  x0: number[],
  learningRate: number = 0.01,
  maxIter: number = 10000,
  tolerance: number = 1e-8,
): { x: number[]; iterations: number; converged: boolean } => {
  let x = [...x0];
  for (let i = 0; i < maxIter; i++) {
    const g = grad(x);
    const normG = Math.sqrt(g.reduce((a, gi) => a + gi * gi, 0));
    if (normG < tolerance) return { x, iterations: i, converged: true };
    x = x.map((xi, j) => xi - learningRate * g[j]);
  }
  return { x, iterations: maxIter, converged: false };
};
曲率与曲线分析
曲率 κ = |f''(x)| / (1 + f'(x)²)^(3/2) 描述了曲线弯曲的程度。曲率在计算机图形学（光滑路径生成）、机器人路径规划和数据可视化中有重要应用。


const curvature = (f: (x: number) => number, x: number, h: number = 1e-5): number => {
  const fp = (f(x + h) - f(x - h)) / (2 * h);
  const fpp = (f(x + h) - 2 * f(x) + f(x - h)) / (h * h);
  return Math.abs(fpp) / Math.pow(1 + fp * fp, 1.5);
};

const findInflectionPoints = (f: (x: number) => number, a: number, b: number, resolution: number = 1000): number[] => {
  const h = (b - a) / resolution;
  const hd = Math.pow(Number.EPSILON, 0.25);
  const inflections: number[] = [];

  for (let i = 1; i < resolution; i++) {
    const x = a + i * h;
    const d2Prev = (f(x - h + hd) - 2 * f(x - h) + f(x - h - hd)) / (hd * hd);
    const d2Curr = (f(x + hd) - 2 * f(x) + f(x - hd)) / (hd * hd);
    if (d2Prev * d2Curr < 0) inflections.push(x);
  }
  return inflections;
};

interface CurveAnalysis {
  criticalPoints: CriticalPoint[];
  inflectionPoints: number[];
  increasing: Array<[number, number]>;
  decreasing: Array<[number, number]>;
  concaveUp: Array<[number, number]>;
  concaveDown: Array<[number, number]>;
}

const analyzeCurve = (f: (x: number) => number, a: number, b: number): CurveAnalysis => {
  const cp = findCriticalPoints(f, a, b);
  const ip = findInflectionPoints(f, a, b);
  return {
    criticalPoints: cp,
    inflectionPoints: ip,
    increasing: [],
    decreasing: [],
    concaveUp: [],
    concaveDown: [],
  };
};
微分在金融中的应用
金融中的 Greeks 是期权价格对各参数的偏导数：Delta (∂V/∂S) 衡量标的价格变化的敏感度，Gamma (∂²V/∂S²) 衡量 Delta 的变化率，Theta (∂V/∂t) 衡量时间衰减，Vega (∂V/∂σ) 衡量波动率敏感度。


const blackScholesCall = (S: number, K: number, T: number, r: number, sigma: number): number => {
  const d1 = (Math.log(S / K) + (r + (sigma * sigma) / 2) * T) / (sigma * Math.sqrt(T));
  const d2 = d1 - sigma * Math.sqrt(T);
  return S * normalCDF(d1) - K * Math.exp(-r * T) * normalCDF(d2);
};

const greeks = (S: number, K: number, T: number, r: number, sigma: number) => {
  const h = S * 1e-5;
  const price = blackScholesCall(S, K, T, r, sigma);
  const delta = (blackScholesCall(S + h, K, T, r, sigma) - blackScholesCall(S - h, K, T, r, sigma)) / (2 * h);
  const gamma =
    (blackScholesCall(S + h, K, T, r, sigma) - 2 * price + blackScholesCall(S - h, K, T, r, sigma)) / (h * h);
  const theta = (blackScholesCall(S, K, T - 1 / 365, r, sigma) - price) * 365;
  const vega = (blackScholesCall(S, K, T, r, sigma + 0.01) - price) / 0.01;
  return { price, delta, gamma, theta, vega };
};

const normalCDF = (x: number): number => {
  const a1 = 0.254829592,
    a2 = -0.284496736,
    a3 = 1.421413741;
  const a4 = -1.453152027,
    a5 = 1.061405429,
    p = 0.3275911;
  const sign = x < 0 ? -1 : 1;
  x = Math.abs(x) / Math.sqrt(2);
  const t = 1.0 / (1.0 + p * x);
  const y = 1 - ((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);
  return 0.5 * (1.0 + sign * y);
};
