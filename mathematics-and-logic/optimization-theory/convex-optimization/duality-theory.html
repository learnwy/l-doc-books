对偶理论

核心概念

对偶理论（Duality Theory）是凸优化中最深刻的理论框架，它将每个优化问题（原始问题）与一个对偶问题关联起来。通过拉格朗日函数 L(x, λ, ν) = f₀(x) + ∑λᵢfᵢ(x) + ∑νᵢhᵢ(x)，原始问题的最优值 p* 与对偶问题的最优值 d* 之间满足弱对偶性 d* ≤ p*。当 p* = d* 时称为强对偶性，此时可以通过求解对偶问题来解决原始问题。

对偶理论的实用价值在于：对偶问题往往比原始问题更容易求解（例如对偶问题总是凸的），对偶变量具有明确的经济学解释（影子价格），且对偶间隙可作为最优性的终止准则。

拉格朗日对偶
对于约束优化问题 min f₀(x) s.t. fᵢ(x) ≤ 0, hⱼ(x) = 0，拉格朗日对偶是构造对偶问题的标准方法。


function lagrangian(
  f0: (x: number[]) => number,
  constraints: Array<(x: number[]) => number>,
  eqConstraints: Array<(x: number[]) => number>,
  x: number[],
  lambda: number[],
  nu: number[],
): number {
  let L = f0(x);
  for (let i = 0; i < constraints.length; i++) {
    L += lambda[i] * constraints[i](x);
  }
  for (let i = 0; i < eqConstraints.length; i++) {
    L += nu[i] * eqConstraints[i](x);
  }
  return L;
}

function dualFunction(
  f0: (x: number[]) => number,
  grad: (x: number[]) => number[],
  constraints: Array<(x: number[]) => number>,
  constraintGrads: Array<(x: number[]) => number[]>,
  lambda: number[],
  x0: number[],
  lr: number = 0.01,
  steps: number = 100,
): number {
  let x = [...x0];

  for (let step = 0; step < steps; step++) {
    const g0 = grad(x);
    const totalGrad = [...g0];

    for (let i = 0; i < constraints.length; i++) {
      const cg = constraintGrads[i](x);
      for (let j = 0; j < totalGrad.length; j++) {
        totalGrad[j] += lambda[i] * cg[j];
      }
    }

    x = x.map((xi, j) => xi - lr * totalGrad[j]);
  }

  return lagrangian(f0, constraints, [], x, lambda, []);
}

function dualAscent(
  f0: (x: number[]) => number,
  constraints: Array<(x: number[]) => number>,
  x0: number[],
  nIter: number = 200,
): { lambda: number[]; dualValue: number } {
  const m = constraints.length;
  const lambda = Array(m).fill(0);
  let x = [...x0];
  const lr = 0.1;
  const dualLr = 0.05;

  for (let iter = 0; iter < nIter; iter++) {
    const h = 1e-5;
    const grad = x.map((_, j) => {
      const xp = [...x];
      xp[j] += h;
      const xm = [...x];
      xm[j] -= h;
      let dL = (f0(xp) - f0(xm)) / (2 * h);
      for (let i = 0; i < m; i++) {
        dL += (lambda[i] * (constraints[i](xp) - constraints[i](xm))) / (2 * h);
      }
      return dL;
    });

    x = x.map((xi, j) => xi - lr * grad[j]);

    for (let i = 0; i < m; i++) {
      lambda[i] = Math.max(0, lambda[i] + dualLr * constraints[i](x));
    }
  }

  const dualVal = f0(x) + lambda.reduce((s, li, i) => s + li * constraints[i](x), 0);
  return { lambda, dualValue: dualVal };
}

const f0 = (x: number[]) => x[0] * x[0] + x[1] * x[1];
const constraint = [(x: number[]) => x[0] + x[1] - 1];

const result = dualAscent(f0, constraint, [2, 2]);
console.log(`对偶变量: [${result.lambda.map((l) => l.toFixed(4))}]`);
console.log(`对偶值: ${result.dualValue.toFixed(4)}`);
强对偶与 Slater 条件
强对偶性（p* = d*）不总是成立，但在凸优化中有广泛的充分条件。Slater 条件是最常用的约束资格条件：如果存在一个严格可行点 x 使得所有不等式约束严格满足 fᵢ(x) < 0，则强对偶成立。


function checkSlaterCondition(
  constraints: Array<(x: number[]) => number>,
  searchPoints: number[][],
): { slaterSatisfied: boolean; slaterPoint: number[] | null } {
  for (const x of searchPoints) {
    const allStrictlyFeasible = constraints.every((c) => c(x) < -1e-10);
    if (allStrictlyFeasible) {
      return { slaterSatisfied: true, slaterPoint: x };
    }
  }
  return { slaterSatisfied: false, slaterPoint: null };
}

function dualityGap(
  primalValue: number,
  dualValue: number,
): {
  gap: number;
  relativeGap: number;
} {
  const gap = primalValue - dualValue;
  const relativeGap = Math.abs(gap) / Math.max(1, Math.abs(primalValue));
  return { gap, relativeGap };
}

const searchGrid = Array.from({ length: 1000 }, () => Array.from({ length: 2 }, () => Math.random() * 4 - 2));

const ineqConstraints = [(x: number[]) => x[0] * x[0] + x[1] * x[1] - 1, (x: number[]) => -x[0]];

const slater = checkSlaterCondition(ineqConstraints, searchGrid);
console.log(`Slater 条件满足: ${slater.slaterSatisfied}`);
if (slater.slaterPoint) {
  console.log(`Slater 点: [${slater.slaterPoint.map((x) => x.toFixed(4))}]`);
}

const gap = dualityGap(0.5, 0.48);
console.log(`对偶间隙: ${gap.gap.toFixed(4)}, 相对间隙: ${gap.relativeGap.toFixed(6)}`);

对偶问题的应用
对偶理论在支持向量机（SVM）、最优传输和资源分配等问题中有重要应用。SVM 的对偶形式使得核方法成为可能。


function svmDual(
  X: number[][],
  y: number[],
  C: number = 1.0,
  lr: number = 0.001,
  iterations: number = 1000,
): { alpha: number[]; w: number[]; b: number } {
  const n = X.length;
  const dim = X[0].length;
  const alpha = Array(n).fill(0);

  const kernel = (i: number, j: number): number => X[i].reduce((s, xi, k) => s + xi * X[j][k], 0);

  for (let iter = 0; iter < iterations; iter++) {
    for (let i = 0; i < n; i++) {
      let sum = 0;
      for (let j = 0; j < n; j++) {
        sum += alpha[j] * y[j] * kernel(i, j);
      }
      const grad = 1 - y[i] * sum;
      alpha[i] = Math.max(0, Math.min(C, alpha[i] + lr * grad));
    }
  }

  const w = Array(dim).fill(0);
  for (let i = 0; i < n; i++) {
    for (let k = 0; k < dim; k++) {
      w[k] += alpha[i] * y[i] * X[i][k];
    }
  }

  let b = 0;
  let count = 0;
  for (let i = 0; i < n; i++) {
    if (alpha[i] > 1e-6 && alpha[i] < C - 1e-6) {
      b += y[i] - X[i].reduce((s, xi, k) => s + w[k] * xi, 0);
      count++;
    }
  }
  b = count > 0 ? b / count : 0;

  return { alpha, w, b };
}

const X = [
  [1, 2],
  [2, 3],
  [3, 1],
  [4, 3],
  [5, 2],
  [6, 4],
];
const y = [-1, -1, -1, 1, 1, 1];
const svm = svmDual(X, y);
console.log(`SVM w: [${svm.w.map((w) => w.toFixed(4))}], b: ${svm.b.toFixed(4)}`);
console.log(`支持向量 α: [${svm.alpha.map((a) => a.toFixed(4))}]`);
小结
    

  • 对偶理论通过拉格朗日函数将约束问题转化为无约束的极小极大问题
    • 

  • 弱对偶性恒成立，强对偶性需要约束资格（如 Slater 条件）
    • 

  • 对偶问题总是凸的，即使原始问题非凸，且对偶间隙可作为终止准则
    • 

  • SVM 的对偶形式 使核方法成为可能，只需计算内积而非显式映射
    • 

  • 互补松弛条件连接原始和对偶最优解，在灵敏度分析中至关重要
    • 

  • 对偶分解方法将大规模问题分解为可并行求解的子问题
