Adam 优化器

核心概念

Adam（Adaptive Moment Estimation）是结合了动量法和自适应学习率的优化算法，由 Kingma 和 Ba 于 2014 年提出。Adam 维护梯度的一阶矩估计 m（动量）和二阶矩估计 v（自适应缩放），更新规则为：mt = β₁ m{t-1} + (1-β₁)gt，v_t = β₂ v{t-1} + (1-β₂)gt²，然后用偏差校正后的估计 m̂_t = m_t/(1-β₁ᵗ)，v̂_t = v_t/(1-β₂ᵗ) 来更新参数：x_t = x{t-1} - α m̂_t/(√v̂_t + ε)。

Adam 的核心优势是每个参数维度拥有独立的自适应学习率。对于稀疏梯度（如 NLP 中的 embedding 层），不经常更新的参数会累积较小的 v，从而获得较大的有效学习率。偏差校正机制修正了指数移动平均在初始阶段的低估偏差。Adam 几乎不需要调节学习率（默认 α=0.001, β₁=0.9, β₂=0.999 在大多数任务上表现良好），是深度学习中使用最广泛的优化器。

从 AdaGrad 到 Adam 的演进
AdaGrad 使用梯度平方的累积和 vt = Σ{i=1}^{t} g_i² 作为缩放因子，为稀疏梯度问题提供了自适应学习率。但 v_t 单调递增导致学习率单调衰减，在训练后期过度缩小步长。RMSProp 用指数移动平均替代累积和来解决这个问题。Adam 进一步加入了一阶矩（动量）和偏差校正。


interface OptimizerState {
  m: number[];
  v: number[];
  t: number;
}

function adagrad(
  grad: number[],
  state: { v: number[] },
  lr: number = 0.01,
  eps: number = 1e-8,
): { update: number[]; state: { v: number[] } } {
  const v = state.v.map((vi, i) => vi + grad[i] * grad[i]);
  const update = grad.map((gi, i) => (-lr * gi) / (Math.sqrt(v[i]) + eps));
  return { update, state: { v } };
}

function rmsprop(
  grad: number[],
  state: { v: number[] },
  lr: number = 0.001,
  beta: number = 0.9,
  eps: number = 1e-8,
): { update: number[]; state: { v: number[] } } {
  const v = state.v.map((vi, i) => beta * vi + (1 - beta) * grad[i] * grad[i]);
  const update = grad.map((gi, i) => (-lr * gi) / (Math.sqrt(v[i]) + eps));
  return { update, state: { v } };
}

function adam(
  grad: number[],
  state: OptimizerState,
  lr: number = 0.001,
  beta1: number = 0.9,
  beta2: number = 0.999,
  eps: number = 1e-8,
): { update: number[]; state: OptimizerState } {
  const t = state.t + 1;
  const m = state.m.map((mi, i) => beta1 * mi + (1 - beta1) * grad[i]);
  const v = state.v.map((vi, i) => beta2 * vi + (1 - beta2) * grad[i] * grad[i]);

  const mHat = m.map((mi) => mi / (1 - Math.pow(beta1, t)));
  const vHat = v.map((vi) => vi / (1 - Math.pow(beta2, t)));

  const update = mHat.map((mh, i) => (-lr * mh) / (Math.sqrt(vHat[i]) + eps));
  return { update, state: { m, v, t } };
}

function compareOptimizers(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  steps: number = 1000,
): void {
  const d = x0.length;
  const optimizers = [
    { name: 'AdaGrad', state: { v: new Array(d).fill(0) }, lr: 0.1 },
    { name: 'RMSProp', state: { v: new Array(d).fill(0) }, lr: 0.01 },
    { name: 'Adam', state: { m: new Array(d).fill(0), v: new Array(d).fill(0), t: 0 }, lr: 0.01 },
  ];

  for (const opt of optimizers) {
    let x = [...x0];
    for (let i = 0; i < steps; i++) {
      const g = grad(x);
      if (opt.name === 'Adam') {
        const res = adam(g, opt.state as OptimizerState, opt.lr);
        x = x.map((xi, j) => xi + res.update[j]);
        opt.state = res.state;
      } else if (opt.name === 'RMSProp') {
        const res = rmsprop(g, opt.state as { v: number[] }, opt.lr);
        x = x.map((xi, j) => xi + res.update[j]);
        opt.state = res.state;
      } else {
        const res = adagrad(g, opt.state as { v: number[] }, opt.lr);
        x = x.map((xi, j) => xi + res.update[j]);
        opt.state = res.state;
      }
    }
    console.log(`${opt.name}: f*=${f(x).toFixed(6)}`);
  }
}
AdamW 与解耦权重衰减
Adam 中的 L2 正则化（在损失函数中加入 λ‖w‖²）与权重衰减（直接在更新中减去 λw）在 Adam 中并不等价——这与 SGD 不同。Loshchilov 和 Hutter 指出，Adam 的自适应缩放会削弱 L2 正则化的效果，因此提出了 AdamW：将权重衰减从梯度计算中解耦，在参数更新后独立执行 wt = (1 - λα)w{t-1} - α m̂_t/(√v̂_t + ε)。


AdamW 已成为大型语言模型和视觉 Transformer 训练的标准优化器。在 GPT、BERT、ViT 等模型中，AdamW 配合余弦学习率调度和线性 warmup 是最常见的训练配方。


function adamW(
  grad: number[],
  w: number[],
  state: OptimizerState,
  lr: number = 0.001,
  beta1: number = 0.9,
  beta2: number = 0.999,
  eps: number = 1e-8,
  weightDecay: number = 0.01,
): { w: number[]; state: OptimizerState } {
  const t = state.t + 1;
  const m = state.m.map((mi, i) => beta1 * mi + (1 - beta1) * grad[i]);
  const v = state.v.map((vi, i) => beta2 * vi + (1 - beta2) * grad[i] * grad[i]);

  const mHat = m.map((mi) => mi / (1 - Math.pow(beta1, t)));
  const vHat = v.map((vi) => vi / (1 - Math.pow(beta2, t)));

  const wNew = w.map((wi, i) => wi * (1 - lr * weightDecay) - (lr * mHat[i]) / (Math.sqrt(vHat[i]) + eps));

  return { w: wNew, state: { m, v, t } };
}

function adamL2(
  grad: number[],
  w: number[],
  state: OptimizerState,
  lr: number = 0.001,
  beta1: number = 0.9,
  beta2: number = 0.999,
  eps: number = 1e-8,
  l2Lambda: number = 0.01,
): { w: number[]; state: OptimizerState } {
  const regGrad = grad.map((gi, i) => gi + 2 * l2Lambda * w[i]);
  return {
    ...adamW(regGrad, w, state, lr, beta1, beta2, eps, 0),
    state: adam(regGrad, state, lr, beta1, beta2, eps).state,
  };
}

function compareDecoupledVsL2(
  data: { x: number[]; y: number }[],
  gradLoss: (w: number[], xi: number[], yi: number) => number[],
  d: number,
  epochs: number = 100,
): void {
  const w0 = Array.from({ length: d }, () => Math.random() - 0.5);
  const configs = [
    { name: 'Adam + L2', wd: 0, l2: 0.01 },
    { name: 'AdamW (解耦)', wd: 0.01, l2: 0 },
  ];

  for (const cfg of configs) {
    let w = [...w0];
    let state: OptimizerState = { m: new Array(d).fill(0), v: new Array(d).fill(0), t: 0 };

    for (let e = 0; e < epochs; e++) {
      for (const sample of data) {
        const g = gradLoss(w, sample.x, sample.y);
        const regG = g.map((gi, i) => gi + 2 * cfg.l2 * w[i]);
        const res = adam(regG, state);
        state = res.state;
        w = w.map((wi, i) => wi * (1 - 0.001 * cfg.wd) + res.update[i]);
      }
    }

    const norm = Math.sqrt(w.reduce((s, wi) => s + wi * wi, 0));
    console.log(`${cfg.name}: |w|=${norm.toFixed(4)}`);
  }
}
超参数选择与调优
Adam 的默认超参数 (α=0.001, β₁=0.9, β₂=0.999, ε=1e-8) 在大多数场景下表现良好，但针对特定任务的调优可以带来显著提升。β₁ 控制动量的衰减速度，较大的 β₁ 使动量更平滑但响应变化更慢；β₂ 控制二阶矩的衰减速度，较大的 β₂ 使缩放因子更稳定但适应新梯度分布更慢。


ε 在数值上防止除以零，但在稀疏梯度场景下对实际行为有显著影响。较大的 ε（如 1e-4）使 Adam 的行为更接近 SGD with momentum。对于大语言模型训练，常用的配置是 β₁=0.9, β₂=0.95, ε=1e-8, weight_decay=0.1。

function clipGradNorm(grad: number[], maxNorm: number): number[] {
  const norm = Math.sqrt(grad.reduce((s, g) => s + g * g, 0));
  if (norm > maxNorm) {
    const scale = maxNorm / norm;
    return grad.map((g) => g * scale);
  }
  return grad;
}
小结
    

  • Adam 结合一阶矩（动量）和二阶矩（自适应缩放）实现高效优化

  • AdamW 解耦权重衰减避免了自适应缩放对正则化效果的削弱
