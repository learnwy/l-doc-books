进化算法

核心概念

进化算法（Evolutionary Algorithms, EA）是受自然进化过程启发的一类元启发式优化方法。其核心框架是：维护一个候选解的种群（Population），通过选择（Selection）、交叉（Crossover）和变异（Mutation）等算子产生新个体，经过适应度评估和环境选择实现种群的逐代进化。进化算法不需要梯度信息，适用于目标函数不可微、非凸、多模态甚至黑箱优化问题。

进化算法的主要分支包括遗传算法（GA）、差分进化（DE）、进化策略（ES）和粒子群优化（PSO）。这些方法在搜索策略上各有侧重：GA 强调组合式搜索，DE 利用个体差异向量生成新解，ES 通过协方差矩阵自适应（CMA-ES）逼近最优搜索分布，PSO 模拟群体智能的信息共享机制。

遗传算法
遗传算法（Genetic Algorithm, GA）是最经典的进化算法。每个候选解编码为"染色体"，通过锦标赛选择或轮盘赌选择挑选父代，交叉算子混合父代基因产生子代，变异算子引入随机扰动维持种群多样性。精英保留策略确保最优个体不因随机操作而丢失。


interface Individual {
  genes: number[];
  fitness: number;
}

function geneticAlgorithm(
  fitnessFunc: (x: number[]) => number,
  dim: number,
  bounds: { min: number; max: number }[],
  options: {
    popSize?: number;
    generations?: number;
    crossoverRate?: number;
    mutationRate?: number;
    eliteCount?: number;
  } = {},
): { best: Individual; history: number[] } {
  const { popSize = 100, generations = 200, crossoverRate = 0.8, mutationRate = 0.1, eliteCount = 2 } = options;

  let population: Individual[] = Array.from({ length: popSize }, () => {
    const genes = bounds.map((b) => b.min + Math.random() * (b.max - b.min));
    return { genes, fitness: fitnessFunc(genes) };
  });

  const history: number[] = [];

  for (let gen = 0; gen < generations; gen++) {
    population.sort((a, b) => a.fitness - b.fitness);
    history.push(population[0].fitness);

    const newPop: Individual[] = population.slice(0, eliteCount);

    while (newPop.length < popSize) {
      const p1 = tournamentSelect(population, 3);
      const p2 = tournamentSelect(population, 3);

      let childGenes: number[];
      if (Math.random() < crossoverRate) {
        childGenes = sbxCrossover(p1.genes, p2.genes, bounds);
      } else {
        childGenes = [...p1.genes];
      }

      if (Math.random() < mutationRate) {
        polynomialMutation(childGenes, bounds, 20);
      }

      newPop.push({ genes: childGenes, fitness: fitnessFunc(childGenes) });
    }

    population = newPop;
  }

  population.sort((a, b) => a.fitness - b.fitness);
  return { best: population[0], history };
}

function tournamentSelect(pop: Individual[], k: number): Individual {
  let best: Individual = pop[Math.floor(Math.random() * pop.length)];
  for (let i = 1; i < k; i++) {
    const challenger = pop[Math.floor(Math.random() * pop.length)];
    if (challenger.fitness < best.fitness) best = challenger;
  }
  return best;
}

function sbxCrossover(p1: number[], p2: number[], bounds: { min: number; max: number }[]): number[] {
  const eta = 2;
  return p1.map((g1, i) => {
    if (Math.random() > 0.5) return g1;
    const g2 = p2[i];
    if (Math.abs(g1 - g2) < 1e-14) return g1;

    const u = Math.random();
    const beta = u <= 0.5 ? Math.pow(2 * u, 1 / (eta + 1)) : Math.pow(1 / (2 * (1 - u)), 1 / (eta + 1));

    const child = 0.5 * ((1 + beta) * g1 + (1 - beta) * g2);
    return Math.max(bounds[i].min, Math.min(bounds[i].max, child));
  });
}

function polynomialMutation(genes: number[], bounds: { min: number; max: number }[], eta: number): void {
  for (let i = 0; i < genes.length; i++) {
    if (Math.random() > 1 / genes.length) continue;
    const u = Math.random();
    const delta = u < 0.5 ? Math.pow(2 * u, 1 / (eta + 1)) - 1 : 1 - Math.pow(2 * (1 - u), 1 / (eta + 1));
    genes[i] += delta * (bounds[i].max - bounds[i].min);
    genes[i] = Math.max(bounds[i].min, Math.min(bounds[i].max, genes[i]));
  }
}
差分进化
差分进化（Differential Evolution, DE）是连续优化中最简洁高效的进化算法。DE 的核心操作是差异向量：随机选取三个不同个体 r1, r2, r3，生成变异向量 v = r1 + F(r2 - r3)，其中 F 是缩放因子。然后通过二项交叉将变异向量与目标向量混合生成试验向量，如果试验向量优于目标向量则替换之。


DE 的优势在于：算法框架极其简洁，仅有 F（缩放因子）和 CR（交叉概率）两个关键超参数，自适应版本（如 JADE、SHADE）可以动态调节这些参数。DE 在 CEC 竞赛中常年表现优异。


function differentialEvolution(
  objective: (x: number[]) => number,
  dim: number,
  bounds: { min: number; max: number }[],
  options: {
    popSize?: number;
    maxGen?: number;
    F?: number;
    CR?: number;
    strategy?: 'rand1' | 'best1' | 'currentToBest1';
  } = {},
): { best: number[]; bestFitness: number; history: number[] } {
  const { popSize = 50, maxGen = 500, F = 0.8, CR = 0.9, strategy = 'rand1' } = options;

  let pop = Array.from({ length: popSize }, () => bounds.map((b) => b.min + Math.random() * (b.max - b.min)));
  let fitness = pop.map((x) => objective(x));
  const history: number[] = [];

  for (let gen = 0; gen < maxGen; gen++) {
    const bestIdx = fitness.indexOf(Math.min(...fitness));
    history.push(fitness[bestIdx]);

    for (let i = 0; i < popSize; i++) {
      const indices = new Set<number>([i]);
      while (indices.size < 4) indices.add(Math.floor(Math.random() * popSize));
      const pool = [...indices].filter((idx) => idx !== i);

      let mutant: number[];
      if (strategy === 'best1') {
        mutant = pop[bestIdx].map((xb, d) => xb + F * (pop[pool[0]][d] - pop[pool[1]][d]));
      } else if (strategy === 'currentToBest1') {
        mutant = pop[i].map((xi, d) => xi + F * (pop[bestIdx][d] - xi) + F * (pop[pool[0]][d] - pop[pool[1]][d]));
      } else {
        mutant = pop[pool[0]].map((xr, d) => xr + F * (pop[pool[1]][d] - pop[pool[2]][d]));
      }

      const jRand = Math.floor(Math.random() * dim);
      const trial = pop[i].map((xi, d) => {
        const val = Math.random() < CR || d === jRand ? mutant[d] : xi;
        return Math.max(bounds[d].min, Math.min(bounds[d].max, val));
      });

      const trialFit = objective(trial);
      if (trialFit <= fitness[i]) {
        pop[i] = trial;
        fitness[i] = trialFit;
      }
    }
  }

  const bestIdx = fitness.indexOf(Math.min(...fitness));
  return { best: pop[bestIdx], bestFitness: fitness[bestIdx], history };
}

const rastrigin = (x: number[]) =>
  10 * x.length + x.reduce((s, xi) => s + xi * xi - 10 * Math.cos(2 * Math.PI * xi), 0);

const deResult = differentialEvolution(
  rastrigin,
  10,
  Array.from({ length: 10 }, () => ({ min: -5.12, max: 5.12 })),
  { popSize: 100, maxGen: 1000 },
);

CMA-ES 协方差矩阵自适应
CMA-ES（Covariance Matrix Adaptation Evolution Strategy）是连续优化中最先进的进化算法之一。CMA-ES 维护一个多元正态分布 N(m, σ²C)，通过采样、评估和更新分布参数来逼近最优解的分布。协方差矩阵 C 的自适应使算法能够学习目标函数的局部曲率，自动调整搜索方向和步长。


CMA-ES 不需要任何超参数调优（所有内部参数都有理论推导的默认值），在 10-100 维的黑箱优化中表现卓越。


function simplifiedCMAES(
  objective: (x: number[]) => number,
  dim: number,
  options: {
    sigma0?: number;
    maxIter?: number;
    popSize?: number;
  } = {},
): { best: number[]; bestFitness: number; iterations: number } {
  const { sigma0 = 0.5, maxIter = 1000 } = options;
  const lambda = options.popSize ?? 4 + Math.floor(3 * Math.log(dim));
  const mu = Math.floor(lambda / 2);

  const weights = Array.from({ length: mu }, (_, i) => Math.log(mu + 0.5) - Math.log(i + 1));
  const wSum = weights.reduce((s, w) => s + w, 0);
  const normalizedW = weights.map((w) => w / wSum);
  const muEff = 1 / normalizedW.reduce((s, w) => s + w * w, 0);

  let mean = new Array(dim).fill(0);
  let sigma = sigma0;
  const diagC = new Array(dim).fill(1);
  const pc = new Array(dim).fill(0);
  const ps = new Array(dim).fill(0);

  const cc = 4 / (dim + 4);
  const cs = (muEff + 2) / (dim + muEff + 5);
  const c1 = 2 / ((dim + 1.3) ** 2 + muEff);
  const cmu_val = Math.min(1 - c1, (2 * (muEff - 2 + 1 / muEff)) / ((dim + 2) ** 2 + muEff));
  const damps = 1 + 2 * Math.max(0, Math.sqrt((muEff - 1) / (dim + 1)) - 1) + cs;

  let bestX = [...mean];
  let bestF = Infinity;

  for (let iter = 0; iter < maxIter; iter++) {
    const samples: { x: number[]; f: number }[] = [];

    for (let k = 0; k < lambda; k++) {
      const z = Array.from({ length: dim }, () => {
        let u = 0,
          v = 0,
          s: number;
        do {
          u = Math.random() * 2 - 1;
          v = Math.random() * 2 - 1;
          s = u * u + v * v;
        } while (s >= 1 || s === 0);
        return u * Math.sqrt((-2 * Math.log(s)) / s);
      });

      const x = mean.map((mi, i) => mi + sigma * Math.sqrt(diagC[i]) * z[i]);
      const f = objective(x);
      samples.push({ x, f });

      if (f < bestF) {
        bestF = f;
        bestX = [...x];
      }
    }

    samples.sort((a, b) => a.f - b.f);

    const newMean = new Array(dim).fill(0);
    for (let i = 0; i < mu; i++) {
      for (let d = 0; d < dim; d++) {
        newMean[d] += normalizedW[i] * samples[i].x[d];
      }
    }

    const diff = newMean.map((m, i) => (m - mean[i]) / sigma);
    for (let d = 0; d < dim; d++) {
      ps[d] = (1 - cs) * ps[d] + (Math.sqrt(cs * (2 - cs) * muEff) * diff[d]) / Math.sqrt(diagC[d]);
      pc[d] = (1 - cc) * pc[d] + Math.sqrt(cc * (2 - cc) * muEff) * diff[d];
      diagC[d] = (1 - c1 - cmu_val) * diagC[d] + c1 * pc[d] * pc[d];
    }

    const psNorm = Math.sqrt(ps.reduce((s, p) => s + p * p, 0));
    const expectedNorm = Math.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim * dim));
    sigma *= Math.exp((cs / damps) * (psNorm / expectedNorm - 1));

    mean = newMean;

    if (sigma < 1e-12) break;
  }

  return { best: bestX, bestFitness: bestF, iterations: maxIter };
}

粒子群优化
粒子群优化（Particle Swarm Optimization, PSO）模拟鸟群或鱼群的集体搜索行为。每个粒子维护位置 x 和速度 v，根据个体历史最优位置 pBest 和全局最优位置 gBest 更新速度：v = w·v + c₁r₁(pBest - x) + c₂r₂(gBest - x)，然后更新位置 x = x + v。


function particleSwarm(
  objective: (x: number[]) => number,
  dim: number,
  bounds: { min: number; max: number }[],
  options: {
    swarmSize?: number;
    maxIter?: number;
    w?: number;
    c1?: number;
    c2?: number;
  } = {},
): { best: number[]; bestFitness: number; history: number[] } {
  const { swarmSize = 40, maxIter = 500, w = 0.7, c1 = 1.5, c2 = 1.5 } = options;

  const particles = Array.from({ length: swarmSize }, () => {
    const x = bounds.map((b) => b.min + Math.random() * (b.max - b.min));
    const v = bounds.map((b) => (Math.random() - 0.5) * (b.max - b.min) * 0.1);
    return { x, v, pBest: [...x], pBestFit: objective(x) };
  });

  let gBest = [...particles[0].pBest];
  let gBestFit = particles[0].pBestFit;
  for (const p of particles) {
    if (p.pBestFit < gBestFit) {
      gBestFit = p.pBestFit;
      gBest = [...p.pBest];
    }
  }

  const history: number[] = [gBestFit];

  for (let iter = 0; iter < maxIter; iter++) {
    for (const p of particles) {
      for (let d = 0; d < dim; d++) {
        const r1 = Math.random(),
          r2 = Math.random();
        p.v[d] = w * p.v[d] + c1 * r1 * (p.pBest[d] - p.x[d]) + c2 * r2 * (gBest[d] - p.x[d]);
        p.x[d] = Math.max(bounds[d].min, Math.min(bounds[d].max, p.x[d] + p.v[d]));
      }

      const fit = objective(p.x);
      if (fit < p.pBestFit) {
        p.pBestFit = fit;
        p.pBest = [...p.x];
        if (fit < gBestFit) {
          gBestFit = fit;
          gBest = [...p.x];
        }
      }
    }
    history.push(gBestFit);
  }

  return { best: gBest, bestFitness: gBestFit, history };
}

  • 进化算法不依赖梯度信息，适用于不可微、多模态和黑箱优化问题
    • 

  • 遗传算法通过交叉和变异进行组合式搜索，擅长离散和混合优化
    • 

  • 差分进化利用种群差异向量生成变异个体，实现简单且性能优异
    • 

  • CMA-ES 通过协方差矩阵自适应学习搜索分布，是连续黑箱优化的标杆
    • 

  • PSO 模拟群体信息共享，实现简单且天然适合并行化
    • 

  • 进化算法在超参数优化、神经架构搜索和强化学习策略搜索中广泛应用
