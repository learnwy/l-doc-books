深度学习中的优化

核心概念

深度学习中的优化面临独特的挑战：损失景观高度非凸、参数空间维度达数十亿级、训练数据规模庞大且存在噪声。与传统优化理论追求精确收敛不同，深度学习优化的核心目标是找到具有良好泛化性能的解，而非损失函数的全局最小值。过拟合意味着训练损失最小化反而损害了测试性能。

深度学习优化的核心议题包括：权重初始化如何影响训练动态、批归一化如何平滑损失景观、梯度裁剪如何防止梯度爆炸、混合精度如何加速训练、分布式训练如何扩展到多 GPU/多节点。这些技术共同构成了现代深度学习训练的完整工具箱。

权重初始化策略
权重初始化决定了训练的起点和初始梯度的分布。错误的初始化会导致前向传播中激活值指数级增长（爆炸）或衰减到零（消失），使得梯度无法有效传播。Xavier 初始化和 He 初始化通过分析方差传播来确保每层输出的方差保持稳定。


Xavier 初始化适用于 tanh/sigmoid 激活函数：W ~ N(0, 2/(n_in + n_out))。He 初始化适用于 ReLU 系列：W ~ N(0, 2/n_in)。对于 Transformer，通常使用经过缩放的初始化：残差连接层的权重按 1/√(2N) 缩放（N 是层数）。


type InitMethod = (shape: [number, number]) => number[][];

const xavierNormal: InitMethod = ([fanIn, fanOut]) => {
  const std = Math.sqrt(2 / (fanIn + fanOut));
  return Array.from({ length: fanIn }, () => Array.from({ length: fanOut }, () => randn() * std));
};

const heNormal: InitMethod = ([fanIn, fanOut]) => {
  const std = Math.sqrt(2 / fanIn);
  return Array.from({ length: fanIn }, () => Array.from({ length: fanOut }, () => randn() * std));
};

const lecunNormal: InitMethod = ([fanIn, fanOut]) => {
  const std = Math.sqrt(1 / fanIn);
  return Array.from({ length: fanIn }, () => Array.from({ length: fanOut }, () => randn() * std));
};

function randn(): number {
  let u = 0,
    v = 0,
    s: number;
  do {
    u = Math.random() * 2 - 1;
    v = Math.random() * 2 - 1;
    s = u * u + v * v;
  } while (s >= 1 || s === 0);
  return u * Math.sqrt((-2 * Math.log(s)) / s);
}

function analyzeVariancePropagation(
  layers: number[],
  initMethod: InitMethod,
  activation: (x: number) => number,
  inputDim: number = 100,
  samples: number = 1000,
): { layerVariances: number[]; gradientVariances: number[] } {
  const weights: number[][][] = [];
  for (let l = 0; l < layers.length - 1; l++) {
    weights.push(initMethod([layers[l], layers[l + 1]]));
  }

  const layerVariances: number[] = [];
  let x = Array.from({ length: samples }, () => Array.from({ length: layers[0] }, () => randn()));

  for (let l = 0; l < weights.length; l++) {
    const newX = x.map((sample) => {
      const out = new Array(layers[l + 1]).fill(0);
      for (let j = 0; j < layers[l + 1]; j++) {
        for (let i = 0; i < layers[l]; i++) {
          out[j] += sample[i] * weights[l][i][j];
        }
        out[j] = activation(out[j]);
      }
      return out;
    });

    const allValues = newX.flat();
    const mean = allValues.reduce((s, v) => s + v, 0) / allValues.length;
    const variance = allValues.reduce((s, v) => s + (v - mean) ** 2, 0) / allValues.length;
    layerVariances.push(variance);
    x = newX;
  }

  return { layerVariances, gradientVariances: [] };
}

const relu = (x: number) => Math.max(0, x);
const layers = [256, 256, 256, 256, 256, 1];

const xavierResult = analyzeVariancePropagation(layers, xavierNormal, relu);
const heResult = analyzeVariancePropagation(layers, heNormal, relu);
console.log(`Xavier+ReLU 方差: [${xavierResult.layerVariances.map((v) => v.toFixed(4))}]`);
console.log(`He+ReLU 方差: [${heResult.layerVariances.map((v) => v.toFixed(4))}]`);
归一化技术
Batch Normalization（BN）对每个 mini-batch 内的激活值进行标准化：y = γ(x - μ_B)/√(σ²_B + ε) + β，其中 μ_B 和 σ²_B 是批次统计量，γ 和 β 是可学习参数。BN 不仅加速收敛，还起到正则化效果，被认为是通过平滑损失景观来改善优化的。


Layer Normalization（LN）沿特征维度归一化，不依赖批次大小，是 Transformer 架构中的标准归一化方式。RMS Normalization 省略了均值中心化步骤，计算更高效，在 LLaMA 等现代大语言模型中广泛采用。


function batchNorm(
  x: number[][],
  gamma: number[],
  beta: number[],
  eps: number = 1e-5,
  runningMean?: number[],
  runningVar?: number[],
  momentum: number = 0.1,
  training: boolean = true,
): { output: number[][]; mean: number[]; var: number[] } {
  const batchSize = x.length;
  const features = x[0].length;

  let mean: number[], variance: number[];

  if (training) {
    mean = new Array(features).fill(0);
    variance = new Array(features).fill(0);

    for (let j = 0; j < features; j++) {
      for (let i = 0; i < batchSize; i++) mean[j] += x[i][j];
      mean[j] /= batchSize;

      for (let i = 0; i < batchSize; i++) {
        variance[j] += (x[i][j] - mean[j]) ** 2;
      }
      variance[j] /= batchSize;
    }

    if (runningMean && runningVar) {
      for (let j = 0; j < features; j++) {
        runningMean[j] = (1 - momentum) * runningMean[j] + momentum * mean[j];
        runningVar[j] = (1 - momentum) * runningVar[j] + momentum * variance[j];
      }
    }
  } else {
    mean = runningMean ?? new Array(features).fill(0);
    variance = runningVar ?? new Array(features).fill(1);
  }

  const output = x.map((row) =>
    row.map((val, j) => (gamma[j] * (val - mean[j])) / Math.sqrt(variance[j] + eps) + beta[j]),
  );

  return { output, mean, var: variance };
}

function layerNorm(x: number[], gamma: number[], beta: number[], eps: number = 1e-5): number[] {
  const n = x.length;
  const mean = x.reduce((s, v) => s + v, 0) / n;
  const variance = x.reduce((s, v) => s + (v - mean) ** 2, 0) / n;

  return x.map((val, i) => (gamma[i] * (val - mean)) / Math.sqrt(variance + eps) + beta[i]);
}

function rmsNorm(x: number[], gamma: number[], eps: number = 1e-6): number[] {
  const rms = Math.sqrt(x.reduce((s, v) => s + v * v, 0) / x.length + eps);
  return x.map((val, i) => (gamma[i] * val) / rms);
}
梯度裁剪与梯度累积
梯度裁剪（Gradient Clipping）是防止梯度爆炸的关键技术。全局范数裁剪将整个模型的梯度向量缩放到指定的最大范数：if ‖g‖ > max_norm, g ← g · max_norm/‖g‖。这保持了梯度方向不变，仅限制了步长大小。


梯度累积（Gradient Accumulation）在显存受限时模拟大批量训练。将大批量拆分为多个微批量，累积梯度后才执行一次参数更新：效果等价于使用大批量但不需要一次性将所有数据载入显存。


function gradientClipByNorm(
  grads: number[][],
  maxNorm: number,
): { clipped: number[][]; originalNorm: number; wasClipped: boolean } {
  let totalNormSq = 0;
  for (const g of grads) {
    for (const gi of g) totalNormSq += gi * gi;
  }
  const totalNorm = Math.sqrt(totalNormSq);

  if (totalNorm <= maxNorm) {
    return { clipped: grads, originalNorm: totalNorm, wasClipped: false };
  }

  const scale = maxNorm / totalNorm;
  const clipped = grads.map((g) => g.map((gi) => gi * scale));
  return { clipped, originalNorm: totalNorm, wasClipped: true };
}

function gradientClipByValue(grads: number[][], clipValue: number): number[][] {
  return grads.map((g) => g.map((gi) => Math.max(-clipValue, Math.min(clipValue, gi))));
}

interface AccumulationState {
  accumGrads: number[][];
  steps: number;
}

function gradientAccumulation(
  microBatchGrads: number[][],
  state: AccumulationState,
  accumSteps: number,
): { shouldUpdate: boolean; grads: number[][]; state: AccumulationState } {
  const newAccum = state.accumGrads.map((ag, i) => ag.map((v, j) => v + microBatchGrads[i][j]));
  const newSteps = state.steps + 1;

  if (newSteps >= accumSteps) {
    const avgGrads = newAccum.map((ag) => ag.map((v) => v / accumSteps));
    return {
      shouldUpdate: true,
      grads: avgGrads,
      state: {
        accumGrads: newAccum.map((ag) => ag.map(() => 0)),
        steps: 0,
      },
    };
  }

  return {
    shouldUpdate: false,
    grads: [],
    state: { accumGrads: newAccum, steps: newSteps },
  };
}

混合精度与分布式训练
混合精度训练（Mixed Precision Training）使用 FP16 进行前向和反向传播以加速计算和节省显存，同时用 FP32 维护参数的主副本以保持数值精度。损失缩放（Loss Scaling）通过放大损失值来防止小梯度在 FP16 下发生下溢。


分布式训练将大规模模型的训练扩展到多个 GPU 或多个节点。数据并行（Data Parallelism）将数据分到多个 GPU，每个 GPU 维护完整模型副本，通过 AllReduce 同步梯度。模型并行（Model Parallelism）将模型分层到不同 GPU，适用于单个 GPU 无法容纳的超大模型。ZeRO 优化将优化器状态、梯度和参数分片到不同 GPU，极大减少了显存需求。


interface MixedPrecisionConfig {
  initScale: number;
  scaleGrowthFactor: number;
  scaleBackoffFactor: number;
  growthInterval: number;
}

function mixedPrecisionStep(
  params: number[],
  grads: number[],
  optimizer: (p: number[], g: number[]) => number[],
  config: MixedPrecisionConfig,
  state: { scale: number; stepsSinceGrowth: number },
): { params: number[]; state: { scale: number; stepsSinceGrowth: number } } {
  const fp16Grads = grads.map((g) => g * state.scale);

  const hasInf = fp16Grads.some((g) => !isFinite(g));
  if (hasInf) {
    return {
      params,
      state: {
        scale: state.scale * config.scaleBackoffFactor,
        stepsSinceGrowth: 0,
      },
    };
  }

  const unscaledGrads = fp16Grads.map((g) => g / state.scale);
  const newParams = optimizer(params, unscaledGrads);

  let newScale = state.scale;
  let steps = state.stepsSinceGrowth + 1;
  if (steps >= config.growthInterval) {
    newScale *= config.scaleGrowthFactor;
    steps = 0;
  }

  return { params: newParams, state: { scale: newScale, stepsSinceGrowth: steps } };
}

function simulateDataParallel(localGrads: number[][], worldSize: number): number[] {
  const d = localGrads[0].length;
  const allReduced = new Array(d).fill(0);

  for (let rank = 0; rank < worldSize; rank++) {
    for (let j = 0; j < d; j++) {
      allReduced[j] += localGrads[rank][j];
    }
  }

  return allReduced.map((g) => g / worldSize);
}

function linearScalingRule(baseLr: number, baseBatchSize: number, actualBatchSize: number): number {
  return (baseLr * actualBatchSize) / baseBatchSize;
}

const scaledLr = linearScalingRule(0.1, 256, 8192);
console.log(`基础 LR: 0.1 → 缩放 LR: ${scaledLr.toFixed(4)} (32 GPU)`);
小结
    

  • 权重初始化通过方差分析确保各层激活值和梯度的数值稳定性
    • 

  • 归一化技术平滑损失景观、加速收敛并提供正则化效果
    • 

  • 梯度裁剪防止梯度爆炸，梯度累积在显存受限时模拟大批量训练
    • 

  • 混合精度训练通过 FP16 加速计算并减半显存需求，损失缩放确保数值精度
    • 

  • 分布式训练通过数据并行和模型并行将训练扩展到多 GPU，线性缩放规则指导学习率调整
    • 

  • 这些技术共同构成了从 BERT 到 GPT-4 等大型模型训练的标准工具链
