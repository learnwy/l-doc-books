线性规划

核心概念

线性规划（Linear Programming, LP）是目标函数和约束条件都是线性的优化问题，标准形式为：min cᵀx s.t. Ax = b, x ≥ 0。线性规划是运筹学和组合优化的基石，拥有完备的理论体系和高效的求解算法。LP 的可行域是凸多面体（Polyhedron），最优解必然出现在顶点（极点）上——这一性质是单纯形法的理论基础。

线性规划在资源分配、网络流、调度排程、物流运输等领域有着广泛应用。任何 LP 问题都有一个对偶问题，原始问题的最优值等于对偶问题的最优值（强对偶定理）。LP 的对偶理论不仅提供了经济学解释（影子价格），还为灵敏度分析和参数规划奠定了基础。

单纯形法
单纯形法（Simplex Method）由 Dantzig 于 1947 年提出，通过在可行域的顶点之间移动来搜索最优解。在每次迭代中，选择一个非基变量进入基（entering），同时选择一个基变量离开基（leaving），从而移动到相邻的顶点。当所有非基变量对应的缩减成本都非负时，当前顶点即为最优解。


虽然单纯形法的最坏情况复杂度是指数级的（Klee-Minty 反例），但在实际问题中通常只需 O(m) 到 O(3m) 次迭代，其中 m 是约束数。改进的选主元规则（如 Bland 规则、最陡边规则）可以避免循环并提升实际性能。


interface SimplexResult {
  x: number[];
  objective: number;
  status: 'optimal' | 'unbounded' | 'infeasible';
  iterations: number;
  basis: number[];
}

function simplex(c: number[], A: number[][], b: number[], options: { maxIter?: number } = {}): SimplexResult {
  const { maxIter = 1000 } = options;
  const m = A.length;
  const n = c.length;

  const tableau: number[][] = [];
  for (let i = 0; i < m; i++) {
    const row = [...A[i]];
    for (let j = 0; j < m; j++) row.push(i === j ? 1 : 0);
    row.push(b[i]);
    tableau.push(row);
  }

  const objRow = [...c.map((ci) => -ci)];
  for (let j = 0; j < m; j++) objRow.push(0);
  objRow.push(0);
  tableau.push(objRow);

  const basis = Array.from({ length: m }, (_, i) => n + i);
  const totalVars = n + m;

  for (let iter = 0; iter < maxIter; iter++) {
    let pivotCol = -1;
    let minRC = -1e-10;
    for (let j = 0; j < totalVars; j++) {
      if (tableau[m][j] < minRC) {
        minRC = tableau[m][j];
        pivotCol = j;
      }
    }

    if (pivotCol === -1) {
      const x = new Array(n).fill(0);
      for (let i = 0; i < m; i++) {
        if (basis[i] < n) x[basis[i]] = tableau[i][totalVars];
      }
      return {
        x,
        objective: tableau[m][totalVars],
        status: 'optimal',
        iterations: iter,
        basis: [...basis],
      };
    }

    let pivotRow = -1;
    let minRatio = Infinity;
    for (let i = 0; i < m; i++) {
      if (tableau[i][pivotCol] > 1e-10) {
        const ratio = tableau[i][totalVars] / tableau[i][pivotCol];
        if (ratio < minRatio) {
          minRatio = ratio;
          pivotRow = i;
        }
      }
    }

    if (pivotRow === -1) {
      return { x: [], objective: -Infinity, status: 'unbounded', iterations: iter, basis };
    }

    const pivotVal = tableau[pivotRow][pivotCol];
    for (let j = 0; j <= totalVars; j++) {
      tableau[pivotRow][j] /= pivotVal;
    }

    for (let i = 0; i <= m; i++) {
      if (i !== pivotRow) {
        const factor = tableau[i][pivotCol];
        for (let j = 0; j <= totalVars; j++) {
          tableau[i][j] -= factor * tableau[pivotRow][j];
        }
      }
    }

    basis[pivotRow] = pivotCol;
  }

  return { x: [], objective: 0, status: 'infeasible', iterations: maxIter, basis };
}

const lpResult = simplex(
  [3, 5],
  [
    [1, 0],
    [0, 2],
    [3, 2],
  ],
  [4, 12, 18],
);
console.log(`最优解: x=[${lpResult.x.map((v) => v.toFixed(2))}], z=${lpResult.objective.toFixed(2)}`);
对偶线性规划
每个 LP 问题（原始问题）都有一个对偶问题。原始问题 min cᵀx s.t. Ax ≥ b, x ≥ 0 的对偶是 max bᵀy s.t. Aᵀy ≤ c, y ≥ 0。对偶理论建立了原始和对偶问题之间的深刻联系。


弱对偶定理指出：任何原始可行解的目标值都不小于任何对偶可行解的目标值，即 cᵀx ≥ bᵀy。强对偶定理进一步指出：如果原始问题有最优解，则对偶问题也有最优解，且两者的最优值相等。互补松弛条件 x_j(c_j - Aᵀ_j y) = 0 和 y_i(A_i x - b_i) = 0 刻画了原始-对偶最优解对的关系。


interface DualResult {
  primal: SimplexResult;
  dualValues: number[];
  reducedCosts: number[];
  complementarySlackness: boolean;
}

function solveDual(c: number[], A: number[][], b: number[]): DualResult {
  const m = A.length;
  const n = c.length;

  const primal = simplex(c, A, b);

  const dualValues: number[] = [];
  const slackStart = n;
  for (let i = 0; i < m; i++) {
    const idx = primal.basis.indexOf(slackStart + i);
    dualValues.push(idx >= 0 ? 0 : 1);
  }

  const reducedCosts = c.map((cj, j) => {
    let rc = cj;
    for (let i = 0; i < m; i++) {
      rc -= dualValues[i] * A[i][j];
    }
    return rc;
  });

  let csHolds = true;
  for (let j = 0; j < n; j++) {
    if (primal.x[j] > 1e-8 && Math.abs(reducedCosts[j]) > 1e-8) {
      csHolds = false;
    }
  }

  return { primal, dualValues, reducedCosts, complementarySlackness: csHolds };
}

const dualResult = solveDual(
  [3, 5],
  [
    [1, 0],
    [0, 2],
    [3, 2],
  ],
  [4, 12, 18],
);
console.log(`对偶值: [${dualResult.dualValues}], 互补松弛: ${dualResult.complementarySlackness}`);

灵敏度分析与参数规划
灵敏度分析研究当 LP 的参数（目标系数 c、约束右端 b、约束矩阵 A）发生微小变化时，最优解和最优值如何变化。这对于管理决策至关重要——比如"再多 1 单位资源能多创造多少利润？"或者"原料涨价多少之前当前方案仍然最优？"


对偶值 y_i* 直接给出了第 i 个约束右端 b_i 的灵敏度：Δz* = y_i* · Δb_i（在基不变的范围内）。缩减成本给出了目标系数的灵敏度：当非基变量 x_j 的目标系数变化量不超过其缩减成本时，当前基仍然最优。


interface SensitivityReport {
  objectiveRanges: { variable: number; lower: number; upper: number }[];
  rhsRanges: { constraint: number; lower: number; upper: number }[];
  shadowPrices: number[];
}

function sensitivityReport(c: number[], A: number[][], b: number[]): SensitivityReport {
  const n = c.length;
  const m = A.length;
  const base = simplex(c, A, b);

  const objectiveRanges = c.map((cj, j) => {
    let lower = -Infinity,
      upper = Infinity;
    const delta = 0.001;

    for (let d = 1; d <= 1000; d++) {
      const cMod = [...c];
      cMod[j] = cj - d * delta;
      const res = simplex(cMod, A, b);
      const sameBasis = res.basis.every((v, i) => v === base.basis[i]);
      if (!sameBasis) {
        lower = cj - d * delta;
        break;
      }
    }

    for (let d = 1; d <= 1000; d++) {
      const cMod = [...c];
      cMod[j] = cj + d * delta;
      const res = simplex(cMod, A, b);
      const sameBasis = res.basis.every((v, i) => v === base.basis[i]);
      if (!sameBasis) {
        upper = cj + d * delta;
        break;
      }
    }

    return { variable: j, lower, upper };
  });

  const shadowPrices: number[] = [];
  const rhsRanges = b.map((bi, i) => {
    const bPlus = [...b];
    bPlus[i] += 0.001;
    const bMinus = [...b];
    bMinus[i] -= 0.001;
    const resPlus = simplex(c, A, bPlus);
    const resMinus = simplex(c, A, bMinus);

    const shadow = (resPlus.objective - resMinus.objective) / 0.002;
    shadowPrices.push(shadow);

    let lower = -Infinity,
      upper = Infinity;
    for (let d = 1; d <= 1000; d++) {
      const bMod = [...b];
      bMod[i] = bi + d * 0.01;
      const res = simplex(c, A, bMod);
      if (!res.basis.every((v, k) => v === base.basis[k])) {
        upper = bi + d * 0.01;
        break;
      }
    }
    for (let d = 1; d <= 1000; d++) {
      const bMod = [...b];
      bMod[i] = bi - d * 0.01;
      const res = simplex(c, A, bMod);
      if (!res.basis.every((v, k) => v === base.basis[k])) {
        lower = bi - d * 0.01;
        break;
      }
    }

    return { constraint: i, lower, upper };
  });

  return { objectiveRanges, rhsRanges, shadowPrices };
}
整数线性规划与 LP 松弛
整数线性规划（Integer Linear Programming, ILP）要求部分或全部变量取整数值。ILP 是 NP 难问题，但 LP 松弛（去掉整数约束）提供了一个重要的下界。分支定界法（Branch and Bound）通过系统地分裂和求解 LP 松弛来搜索整数最优解。


LP 松弛的强度决定了分支定界的效率。当 LP 松弛的最优解恰好是整数解时（如完全单模矩阵对应的 LP），ILP 可以在多项式时间内求解。切割平面法通过添加有效不等式来收紧 LP 松弛，现代 ILP 求解器结合了分支定界和切割平面（Branch and Cut）。


interface ILPResult {
  x: number[];
  objective: number;
  lpRelaxation: number;
  integrality_gap: number;
  nodes_explored: number;
}

function branchAndBound(
  c: number[],
  A: number[][],
  b: number[],
  integerVars: number[],
  options: { maxNodes?: number } = {},
): ILPResult {
  const { maxNodes = 10000 } = options;
  const n = c.length;

  let bestObj = Infinity;
  let bestX: number[] = [];
  let nodesExplored = 0;
  const lpRelax = simplex(c, A, b);

  interface Node {
    A: number[][];
    b: number[];
    depth: number;
  }
  const queue: Node[] = [{ A: [...A.map((r) => [...r])], b: [...b], depth: 0 }];

  while (queue.length > 0 && nodesExplored < maxNodes) {
    const node = queue.pop()!;
    nodesExplored++;

    const lp = simplex(c, node.A, node.b);
    if (lp.status !== 'optimal' || lp.objective >= bestObj) continue;

    let fractionalVar = -1;
    let maxFrac = 0;
    for (const j of integerVars) {
      const frac = lp.x[j] - Math.floor(lp.x[j]);
      if (frac > 1e-6 && frac < 1 - 1e-6 && frac > maxFrac) {
        maxFrac = frac;
        fractionalVar = j;
      }
    }

    if (fractionalVar === -1) {
      if (lp.objective < bestObj) {
        bestObj = lp.objective;
        bestX = [...lp.x];
      }
      continue;
    }

    const floorVal = Math.floor(lp.x[fractionalVar]);

    const leftRow = new Array(n).fill(0);
    leftRow[fractionalVar] = 1;
    queue.push({
      A: [...node.A.map((r) => [...r]), leftRow],
      b: [...node.b, floorVal],
      depth: node.depth + 1,
    });

    const rightRow = new Array(n).fill(0);
    rightRow[fractionalVar] = -1;
    queue.push({
      A: [...node.A.map((r) => [...r]), rightRow],
      b: [...node.b, -(floorVal + 1)],
      depth: node.depth + 1,
    });
  }

  return {
    x: bestX,
    objective: bestObj,
    lpRelaxation: lpRelax.objective,
    integrality_gap: bestObj > 0 ? (bestObj - lpRelax.objective) / bestObj : 0,
    nodes_explored: nodesExplored,
  };
}

小结
    

  • 线性规划是最基本的约束优化问题类型，最优解存在于可行域多面体的顶点上
    • 

  • 单纯形法沿多面体边搜索最优顶点，实践中通常在 O(m) 次迭代内收敛
    • 

  • 每个 LP 都有对偶问题，强对偶定理保证原始和对偶最优值相等
    • 

  • 对偶值是约束的影子价格，灵敏度分析为管理决策提供定量依据
    • 

  • 整数线性规划是 NP 难问题，分支定界通过 LP 松弛提供下界来剪枝搜索
    • 

  • 线性规划是运筹学、经济学和工程优化的核心工具
