内点法

核心概念

内点法（Interior Point Methods, IPM）是现代约束优化中最强大的算法之一。与单纯形法沿可行域边界搜索不同，内点法从可行域内部出发，沿中心路径（Central Path）逼近最优解。内点法将障碍函数思想与牛顿法结合，每次迭代通过求解修正的 KKT 系统来确定搜索方向，实现了多项式时间复杂度的收敛保证。

内点法在 1984 年由 Karmarkar 提出后引发了优化领域的革命。对于含有 n 个变量和 m 个约束的线性规划问题，内点法的迭代复杂度为 O(√m log(1/ε))，每次迭代需要 O(n²m) 的计算量。在大规模稀疏问题上，内点法通常显著优于单纯形法。

路径跟踪算法

路径跟踪法（Path-Following Method）是内点法最实用的形式。对于标准形式的约束优化 min f(x) s.t. h_i(x) ≤ 0，引入参数 t > 0 定义中心路径上的点 x*(t) = argmin { t·f(x) - Σ log(-h_i(x)) }。当 t → ∞ 时，x*(t) → x*（原问题最优解）。

算法框架是：外层循环逐步增大 t（通常乘以 μ > 1），内层循环用牛顿法求解当前 t 对应的中心方程。对偶间隙为 m/t（m 是约束数），因此达到 ε 精度需要 O(√m log(m/ε)) 次外迭代。

interface PathFollowResult {
  x: number[];
  fVal: number;
  dualityGap: number;
  centralPath: number[][];
  iterations: number;
}

function pathFollowingIPM(
  f: (x: number[]) => number,
  gradF: (x: number[]) => number[],
  hessF: (x: number[]) => number[][],
  constraints: {
    h: (x: number[]) => number;
    grad: (x: number[]) => number[];
    hess: (x: number[]) => number[][];
  }[],
  x0: number[],
  options: { t0?: number; mu?: number; tol?: number; maxOuter?: number } = {},
): PathFollowResult {
  const { t0 = 1, mu = 10, tol = 1e-8, maxOuter = 100 } = options;
  const m = constraints.length;
  const n = x0.length;

  let x = [...x0];
  let t = t0;
  const centralPath: number[][] = [[...x]];

  for (let outer = 0; outer < maxOuter; outer++) {
    for (let newton = 0; newton < 50; newton++) {
      const grad = gradF(x).map((g) => t * g);
      const hess = hessF(x).map((row) => row.map((v) => t * v));

      for (const c of constraints) {
        const hv = c.h(x);
        const cg = c.grad(x);
        const ch = c.hess(x);

        for (let i = 0; i < n; i++) {
          grad[i] -= cg[i] / hv;
          for (let j = 0; j < n; j++) {
            hess[i][j] += (cg[i] * cg[j]) / (hv * hv) - ch[i][j] / hv;
          }
        }
      }

      const dx = solveLinearSystem(
        hess,
        grad.map((g) => -g),
      );
      const gradNorm = Math.sqrt(grad.reduce((s, g) => s + g * g, 0));
      if (gradNorm < tol) break;

      let step = 1.0;
      while (step > 1e-12) {
        const xNew = x.map((xi, i) => xi + step * dx[i]);
        let feasible = true;
        for (const c of constraints) {
          if (c.h(xNew) >= 0) {
            feasible = false;
            break;
          }
        }
        if (feasible) {
          x = xNew;
          break;
        }
        step *= 0.5;
      }
    }

    centralPath.push([...x]);
    const gap = m / t;
    if (gap < tol) break;
    t *= mu;
  }

  return {
    x,
    fVal: f(x),
    dualityGap: m / t,
    centralPath,
    iterations: centralPath.length - 1,
  };
}

function solveLinearSystem(A: number[][], b: number[]): number[] {
  const n = b.length;
  const aug = A.map((row, i) => [...row, b[i]]);

  for (let col = 0; col < n; col++) {
    let maxRow = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) maxRow = row;
    }
    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];

    for (let row = col + 1; row < n; row++) {
      const factor = aug[row][col] / aug[col][col];
      for (let j = col; j <= n; j++) aug[row][j] -= factor * aug[col][j];
    }
  }

  const x = new Array(n);
  for (let i = n - 1; i >= 0; i--) {
    x[i] = aug[i][n];
    for (let j = i + 1; j < n; j++) x[i] -= aug[i][j] * x[j];
    x[i] /= aug[i][i];
  }
  return x;
}

原始-对偶内点法

原始-对偶内点法（Primal-Dual IPM）同时维护原始变量 x、松弛变量 s 和对偶变量 (λ, μ)，通过求解修正的 KKT 系统一步更新所有变量。这是工业级内点法求解器（如 MOSEK、Gurobi 的 barrier solver）的核心算法。

修正的 KKT 系统为：∇f + Aᵀλ + μ = 0（驻点条件），Ax - b = 0（原始可行），Sμ = σμ_avg · e（互补松弛的松弛版），s ≥ 0, μ ≥ 0。其中 σ ∈ (0, 1) 是中心化参数，μ_avg 是当前互补间隙的平均值。

interface PrimalDualState {
  x: number[];
  s: number[];
  lambda: number[];
  mu: number[];
  gap: number;
}

function primalDualIPM(
  c: number[],
  A: number[][],
  b: number[],
  options: { sigma?: number; tol?: number; maxIter?: number } = {},
): PrimalDualState {
  const { sigma = 0.3, tol = 1e-10, maxIter = 100 } = options;
  const m = A.length;
  const n = c.length;

  let x = new Array(n).fill(1);
  let s = new Array(n).fill(1);
  let lambda = new Array(m).fill(0);
  let mu = new Array(n).fill(1);

  for (let iter = 0; iter < maxIter; iter++) {
    const muAvg = s.reduce((sum, si, i) => sum + si * mu[i], 0) / n;
    if (muAvg < tol) break;

    const rd = c.map((ci, j) => {
      let sum = ci - mu[j];
      for (let i = 0; i < m; i++) sum += A[i][j] * lambda[i];
      return sum;
    });

    const rp = b.map((bi, i) => {
      let sum = -bi;
      for (let j = 0; j < n; j++) sum += A[i][j] * x[j];
      return sum;
    });

    const target = sigma * muAvg;
    const rc = s.map((si, i) => si * mu[i] - target);

    const dx = new Array(n).fill(0);
    const ds = new Array(n).fill(0);
    const dmu = new Array(n).fill(0);

    for (let j = 0; j < n; j++) {
      const diag = mu[j] / s[j];
      dx[j] = -(rd[j] + (rc[j] - mu[j] * ds[j]) / s[j]) / (1 + diag);
      ds[j] = -rp.reduce((sum, rpi, i) => sum + A[i][j] * rpi, 0) * 0.01;
      dmu[j] = -(rc[j] + mu[j] * ds[j]) / s[j];
    }

    let alphaP = 1.0;
    let alphaD = 1.0;
    const eta = 0.995;
    for (let j = 0; j < n; j++) {
      if (ds[j] < 0) alphaP = Math.min(alphaP, (-eta * s[j]) / ds[j]);
      if (dmu[j] < 0) alphaD = Math.min(alphaD, (-eta * mu[j]) / dmu[j]);
    }

    x = x.map((xi, j) => xi + alphaP * dx[j]);
    s = s.map((si, j) => si + alphaP * ds[j]);
    lambda = lambda.map((li, i) => li + alphaD * rp[i] * 0.01);
    mu = mu.map((mi, j) => mi + alphaD * dmu[j]);
  }

  return {
    x,
    s,
    lambda,
    mu,
    gap: s.reduce((sum, si, i) => sum + si * mu[i], 0),
  };
}

内点法在二次规划中的应用

二次规划（QP）问题 min 1/2 xᵀQx + cᵀx s.t. Ax ≤ b 是内点法的重要应用场景。QP 的 KKT 系统是线性的，每次牛顿迭代只需求解一个线性方程组。对于稀疏 Q 和 A，利用稀疏 Cholesky 分解可以高效求解大规模 QP。

SVM（支持向量机）的训练问题就是一个典型的 QP：核矩阵定义了 Q，约束来自分类间隔和松弛变量。内点法在中等规模（n ≤ 10⁴）的 SVM 训练中表现优异，但对于大规模问题通常使用 SMO 等分解方法。

interface QPResult {
  x: number[];
  objective: number;
  multipliers: number[];
  iterations: number;
}

function solveQP(
  Q: number[][],
  c: number[],
  A: number[][],
  b: number[],
  options: { tol?: number; maxIter?: number } = {},
): QPResult {
  const { tol = 1e-8, maxIter = 100 } = options;
  const n = c.length;
  const m = A.length;

  let x = new Array(n).fill(0);
  let s = b.map((bi) => bi - A.reduce((_, row) => 0, 0) + 1);
  let mu = new Array(m).fill(1);

  for (let iter = 0; iter < maxIter; iter++) {
    const Qx = Q.map((row) => row.reduce((sum, qij, j) => sum + qij * x[j], 0));
    const gap = s.reduce((sum, si, i) => sum + si * mu[i], 0);
    if (gap / m < tol) break;

    const sigma = 0.2;
    const target = (sigma * gap) / m;

    const grad = c.map((ci, j) => ci + Qx[j]);
    for (let i = 0; i < m; i++) {
      for (let j = 0; j < n; j++) {
        grad[j] -= A[i][j] * mu[i];
      }
    }

    const lr = 0.01 / (1 + iter);
    x = x.map((xi, j) => xi - lr * grad[j]);

    for (let i = 0; i < m; i++) {
      const ax = A[i].reduce((sum, aij, j) => sum + aij * x[j], 0);
      s[i] = Math.max(0.001, b[i] - ax);
      mu[i] = Math.max(0.001, target / s[i]);
    }
  }

  const Qx = Q.map((row) => row.reduce((sum, qij, j) => sum + qij * x[j], 0));
  const obj = 0.5 * x.reduce((sum, xi, i) => sum + xi * Qx[i], 0) + c.reduce((sum, ci, i) => sum + ci * x[i], 0);

  return { x, objective: obj, multipliers: mu, iterations: maxIter };
}

const qpResult = solveQP(
  [
    [2, 0],
    [0, 2],
  ],
  [-2, -5],
  [
    [1, 1],
    [-1, 0],
    [0, -1],
  ],
  [4, 0, 0],
);
console.log(`QP解: x=[${qpResult.x.map((v) => v.toFixed(4))}], obj=${qpResult.objective.toFixed(4)}`);
