牛顿法

核心概念

牛顿法（Newton's Method）是最重要的二阶优化算法，通过利用目标函数的二阶导数（Hessian 矩阵）信息来加速收敛。牛顿法的迭代格式为：x_{k+1} = x_k - [∇²f(x_k)]⁻¹∇f(x_k)。与梯度下降的线性收敛不同，牛顿法在最优解附近具有二次收敛速率，即误差以指数的指数速度减小。

牛顿法的几何直觉是：在当前点用二次函数近似目标函数，然后直接跳到近似函数的极小点。这种二阶近似比梯度下降的线性近似更精确，因此收敛更快。

牛顿法的实现与全局化
纯牛顿法只在初始点足够接近最优解时收敛。阻尼牛顿法（Damped Newton's Method）通过线搜索确定步长，保证全局收敛性。


function newtonsMethod(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  hessian: (x: number[]) => number[][],
  x0: number[],
  options: { maxIter?: number; tol?: number; damped?: boolean } = {},
): { x: number[]; history: number[]; iterations: number } {
  const { maxIter = 100, tol = 1e-10, damped = true } = options;
  let x = [...x0];
  const history: number[] = [f(x)];

  for (let k = 0; k < maxIter; k++) {
    const g = grad(x);
    const H = hessian(x);
    const gradNorm = Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0));
    if (gradNorm < tol) return { x, history, iterations: k };

    const d = solveLinearSystem(
      H,
      g.map((gi) => -gi),
    );
    if (!d) {
      const fallback = g.map((gi) => -gi);
      x = x.map((xi, i) => xi + 0.001 * fallback[i]);
      history.push(f(x));
      continue;
    }

    let alpha = 1.0;
    if (damped) {
      const fx = f(x);
      const dg = d.reduce((s, di, i) => s + di * g[i], 0);
      while (alpha > 1e-12) {
        const xNew = x.map((xi, i) => xi + alpha * d[i]);
        if (f(xNew) <= fx + 0.3 * alpha * dg) break;
        alpha *= 0.5;
      }
    }

    x = x.map((xi, i) => xi + alpha * d[i]);
    history.push(f(x));
  }

  return { x, history, iterations: maxIter };
}

function solveLinearSystem(A: number[][], b: number[]): number[] | null {
  const n = A.length;
  const aug = A.map((row, i) => [...row, b[i]]);

  for (let col = 0; col < n; col++) {
    let maxRow = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) maxRow = row;
    }
    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
    if (Math.abs(aug[col][col]) < 1e-12) return null;

    for (let row = col + 1; row < n; row++) {
      const factor = aug[row][col] / aug[col][col];
      for (let j = col; j <= n; j++) {
        aug[row][j] -= factor * aug[col][j];
      }
    }
  }

  const x = Array(n).fill(0);
  for (let i = n - 1; i >= 0; i--) {
    x[i] = aug[i][n];
    for (let j = i + 1; j < n; j++) {
      x[i] -= aug[i][j] * x[j];
    }
    x[i] /= aug[i][i];
  }
  return x;
}

const rosenbrock = (x: number[]) => (1 - x[0]) ** 2 + 100 * (x[1] - x[0] ** 2) ** 2;
const rosenGrad = (x: number[]) => [-2 * (1 - x[0]) - 400 * x[0] * (x[1] - x[0] ** 2), 200 * (x[1] - x[0] ** 2)];
const rosenHess = (x: number[]) => [
  [2 + 1200 * x[0] ** 2 - 400 * x[1], -400 * x[0]],
  [-400 * x[0], 200],
];

const result = newtonsMethod(rosenbrock, rosenGrad, rosenHess, [-1, 1]);
console.log(`牛顿法: x*=[${result.x.map((x) => x.toFixed(6))}], f*=${rosenbrock(result.x).toFixed(10)}`);
console.log(`迭代次数: ${result.iterations}`);
拟牛顿法
拟牛顿法（Quasi-Newton Methods）用低秩更新近似 Hessian 矩阵（或其逆），避免了每步 O(n³) 的 Hessian 计算和线性方程组求解。BFGS 是最成功的拟牛顿方法。


function bfgs(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  maxIter: number = 200,
): { x: number[]; history: number[] } {
  const n = x0.length;
  let x = [...x0];
  let H = Array.from({ length: n }, (_, i) => Array.from({ length: n }, (_, j) => (i === j ? 1 : 0)));
  let g = grad(x);
  const history: number[] = [f(x)];

  for (let k = 0; k < maxIter; k++) {
    const gradNorm = Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0));
    if (gradNorm < 1e-8) break;

    const d = H.map((row) => -row.reduce((s, hij, j) => s + hij * g[j], 0));

    let alpha = 1.0;
    const fx = f(x);
    const dg = d.reduce((s, di, i) => s + di * g[i], 0);
    while (alpha > 1e-12) {
      const xNew = x.map((xi, i) => xi + alpha * d[i]);
      if (f(xNew) <= fx + 1e-4 * alpha * dg) break;
      alpha *= 0.5;
    }

    const xNew = x.map((xi, i) => xi + alpha * d[i]);
    const gNew = grad(xNew);
    const s = xNew.map((xi, i) => xi - x[i]);
    const y = gNew.map((gi, i) => gi - g[i]);
    const sy = s.reduce((sum, si, i) => sum + si * y[i], 0);

    if (sy > 1e-10) {
      const Hy = H.map((row) => row.reduce((sum, hij, j) => sum + hij * y[j], 0));
      const yHy = y.reduce((sum, yi, i) => sum + yi * Hy[i], 0);

      for (let i = 0; i < n; i++) {
        for (let j = 0; j < n; j++) {
          H[i][j] += ((sy + yHy) * s[i] * s[j]) / (sy * sy) - (Hy[i] * s[j] + s[i] * Hy[j]) / sy;
        }
      }
    }

    x = xNew;
    g = gNew;
    history.push(f(x));
  }

  return { x, history };
}

function lbfgs(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  m: number = 10,
  maxIter: number = 200,
): { x: number[]; history: number[] } {
  const n = x0.length;
  let x = [...x0];
  const history: number[] = [f(x)];
  const sHistory: number[][] = [];
  const yHistory: number[][] = [];

  for (let k = 0; k < maxIter; k++) {
    const g = grad(x);
    const gradNorm = Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0));
    if (gradNorm < 1e-8) break;

    let q = [...g];
    const alphas: number[] = [];
    for (let i = sHistory.length - 1; i >= 0; i--) {
      const rho = 1 / yHistory[i].reduce((s, yi, j) => s + yi * sHistory[i][j], 0);
      const alpha = rho * sHistory[i].reduce((s, si, j) => s + si * q[j], 0);
      alphas.unshift(alpha);
      q = q.map((qi, j) => qi - alpha * yHistory[i][j]);
    }

    let gamma = 1;
    if (sHistory.length > 0) {
      const last = sHistory.length - 1;
      const sy = sHistory[last].reduce((s, si, j) => s + si * yHistory[last][j], 0);
      const yy = yHistory[last].reduce((s, yi) => s + yi * yi, 0);
      gamma = sy / yy;
    }
    let r = q.map((qi) => gamma * qi);

    for (let i = 0; i < sHistory.length; i++) {
      const rho = 1 / yHistory[i].reduce((s, yi, j) => s + yi * sHistory[i][j], 0);
      const beta = rho * yHistory[i].reduce((s, yi, j) => s + yi * r[j], 0);
      r = r.map((ri, j) => ri + sHistory[i][j] * (alphas[i] - beta));
    }

    const d = r.map((ri) => -ri);
    let alpha = 1.0;
    const fx = f(x);
    while (alpha > 1e-12) {
      const xNew = x.map((xi, i) => xi + alpha * d[i]);
      if (f(xNew) <= fx + 1e-4 * alpha * g.reduce((s, gi, i) => s + gi * d[i], 0)) break;
      alpha *= 0.5;
    }

    const xNew = x.map((xi, i) => xi + alpha * d[i]);
    const gNew = grad(xNew);
    sHistory.push(xNew.map((xi, i) => xi - x[i]));
    yHistory.push(gNew.map((gi, i) => gi - g[i]));
    if (sHistory.length > m) {
      sHistory.shift();
      yHistory.shift();
    }

    x = xNew;
    history.push(f(x));
  }

  return { x, history };
}

);
console.log(
  `L-BFGS: ${lbfgsResult.history.length} 迭代, f*=${lbfgsResult.history[lbfgsResult.history.length - 1].toFixed(10)}`,
);

信赖域方法
信赖域方法（Trust Region Methods）是牛顿法全局化的另一种策略。不同于线搜索先确定方向再确定步长，信赖域方法在一个有限区域内最小化二次模型。


function trustRegion(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  hessian: (x: number[]) => number[][],
  x0: number[],
  maxIter: number = 100,
): { x: number[]; history: number[] } {
  let x = [...x0];
  let delta = 1.0;
  const deltaMax = 10.0;
  const eta = 0.1;
  const history: number[] = [f(x)];

  for (let k = 0; k < maxIter; k++) {
    const g = grad(x);
    const H = hessian(x);
    const gradNorm = Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0));
    if (gradNorm < 1e-8) break;

    const d = cauchyPoint(g, H, delta);

    const fx = f(x);
    const xNew = x.map((xi, i) => xi + d[i]);
    const fNew = f(xNew);

    const modelReduction = -(
      g.reduce((s, gi, i) => s + gi * d[i], 0) +
      0.5 * d.reduce((s, di, i) => s + di * H[i].reduce((ss, hij, j) => ss + hij * d[j], 0), 0)
    );
    const actualReduction = fx - fNew;
    const rho = modelReduction > 1e-12 ? actualReduction / modelReduction : 0;

    if (rho > eta) {
      x = xNew;
    }

    if (rho < 0.25) {
      delta *= 0.25;
    } else if (rho > 0.75) {
      delta = Math.min(2 * delta, deltaMax);
    }

    history.push(f(x));
  }

  return { x, history };
}

function cauchyPoint(g: number[], H: number[][], delta: number): number[] {
  const gNorm = Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0));
  const gHg = g.reduce((s, gi, i) => s + gi * H[i].reduce((ss, hij, j) => ss + hij * g[j], 0), 0);

  let tau: number;
  if (gHg <= 0) {
    tau = 1;
  } else {
    tau = Math.min((gNorm * gNorm * gNorm) / (delta * gHg), 1);
  }

  const scale = (-tau * delta) / gNorm;
  return g.map((gi) => scale * gi);
}

const trResult = trustRegion(rosenbrock, rosenGrad, rosenHess, [-1, 1]);
console.log(`信赖域: ${trResult.history.length} 迭代, f*=${trResult.history[trResult.history.length - 1].toFixed(10)}`);
小结
    

  • 牛顿法利用二阶信息实现二次收敛，在最优解附近迭代次数极少
    • 

  • 阻尼牛顿法和信赖域方法保证全局收敛，然后在最优解附近切换到二次收敛
    • 

  • BFGS 通过秩二更新近似逆 Hessian，L-BFGS 仅存储 O(mn) 适用于大规模问题
    • 

  • 信赖域方法通过控制步长范围避免发散，比线搜索更稳健
    • 

  • 牛顿法每步需要 O(n³) 计算，L-BFGS 每步仅需 O(mn) 适用于深度学习
    • 

  • 选择方法取决于问题规模：小规模用牛顿法，大规模用 L-BFGS
    •
