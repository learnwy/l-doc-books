梯度下降法

核心概念

梯度下降法（Gradient Descent）是最基本的一阶优化算法，通过沿负梯度方向迭代更新来最小化目标函数：x_{k+1} = x_k - α∇f(x_k)，其中 α 是步长（学习率）。梯度下降的几何直觉是：在当前点的邻域内，负梯度方向是函数下降最快的方向。

梯度下降的收敛速率取决于函数的性质：对于 L-光滑凸函数，收敛速率为 O(1/k)；对于 L-光滑且 μ-强凸函数，收敛速率为 O((1-μ/L)^k)，即线性收敛。条件数 κ = L/μ 决定了收敛的快慢。

步长选择策略
步长（学习率）的选择是梯度下降法最关键的超参数。步长太大导致发散，步长太小收敛极慢。理论最优步长为 1/L，但实际中 L 通常未知。


function gradientDescent(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  options: {
    lr?: number;
    maxIter?: number;
    tol?: number;
    lineSearch?: boolean;
  } = {},
): { x: number[]; history: number[]; iterations: number } {
  const { lr = 0.01, maxIter = 1000, tol = 1e-8, lineSearch = false } = options;
  let x = [...x0];
  const history: number[] = [f(x)];

  for (let k = 0; k < maxIter; k++) {
    const g = grad(x);
    const gradNorm = Math.sqrt(g.reduce((s, gi) => s + gi * gi, 0));
    if (gradNorm < tol) return { x, history, iterations: k };

    let alpha = lr;
    if (lineSearch) {
      alpha = backtrackingLineSearch(f, x, g, 0.3, 0.8);
    }

    x = x.map((xi, i) => xi - alpha * g[i]);
    history.push(f(x));
  }

  return { x, history, iterations: maxIter };
}

function backtrackingLineSearch(
  f: (x: number[]) => number,
  x: number[],
  grad: number[],
  c: number = 0.3,
  rho: number = 0.8,
): number {
  let alpha = 1.0;
  const fx = f(x);
  const gradNormSq = grad.reduce((s, g) => s + g * g, 0);

  while (true) {
    const xNew = x.map((xi, i) => xi - alpha * grad[i]);
    if (f(xNew) <= fx - c * alpha * gradNormSq) break;
    alpha *= rho;
    if (alpha < 1e-12) break;
  }

  return alpha;
}

function exactLineSearch(f: (x: number[]) => number, x: number[], direction: number[], maxAlpha: number = 1.0): number {
  let best = 0;
  let bestF = f(x);
  const steps = 100;

  for (let i = 1; i <= steps; i++) {
    const alpha = (maxAlpha * i) / steps;
    const xNew = x.map((xi, j) => xi + alpha * direction[j]);
    const fNew = f(xNew);
    if (fNew < bestF) {
      bestF = fNew;
      best = alpha;
    }
  }
  return best;
}

const quadF = (x: number[]) => x[0] ** 2 + 10 * x[1] ** 2;
const quadG = (x: number[]) => [2 * x[0], 20 * x[1]];

const res1 = gradientDescent(quadF, quadG, [10, 10], { lr: 0.01, maxIter: 500 });
const res2 = gradientDescent(quadF, quadG, [10, 10], { lineSearch: true, maxIter: 500 });

console.log(`固定步长: ${res1.iterations} 迭代, f*=${res1.history[res1.history.length - 1].toFixed(6)}`);
console.log(`回溯线搜索: ${res2.iterations} 迭代, f*=${res2.history[res2.history.length - 1].toFixed(6)}`);
梯度下降的变体
标准梯度下降有多种重要变体，针对不同的问题结构进行优化。坐标下降每次只更新一个坐标，近端梯度下降处理复合目标函数，投影梯度下降处理约束优化。


function coordinateDescent(
  f: (x: number[]) => number,
  n: number,
  x0: number[],
  maxIter: number = 100,
): { x: number[]; history: number[] } {
  let x = [...x0];
  const history: number[] = [f(x)];
  const h = 1e-6;

  for (let iter = 0; iter < maxIter; iter++) {
    for (let i = 0; i < n; i++) {
      const xp = [...x];
      xp[i] += h;
      const xm = [...x];
      xm[i] -= h;
      const gi = (f(xp) - f(xm)) / (2 * h);

      let alpha = 0.1;
      const fx = f(x);
      while (true) {
        const xNew = [...x];
        xNew[i] -= alpha * gi;
        if (f(xNew) < fx - 0.3 * alpha * gi * gi) break;
        alpha *= 0.5;
        if (alpha < 1e-12) break;
      }
      x[i] -= alpha * gi;
    }
    history.push(f(x));
  }

  return { x, history };
}

function proximalGradient(
  smoothGrad: (x: number[]) => number[],
  prox: (x: number[], t: number) => number[],
  f: (x: number[]) => number,
  x0: number[],
  lr: number = 0.01,
  maxIter: number = 500,
): { x: number[]; history: number[] } {
  let x = [...x0];
  const history: number[] = [f(x)];

  for (let k = 0; k < maxIter; k++) {
    const g = smoothGrad(x);
    const y = x.map((xi, i) => xi - lr * g[i]);
    x = prox(y, lr);
    history.push(f(x));
  }

  return { x, history };
}

function softThreshold(x: number[], lambda: number): number[] {
  return x.map((xi) => Math.sign(xi) * Math.max(0, Math.abs(xi) - lambda));
}

const lsGrad = (x: number[]) => [2 * (x[0] - 3), 2 * (x[1] - 1)];
const lassoObj = (x: number[]) => (x[0] - 3) ** 2 + (x[1] - 1) ** 2 + 0.5 * (Math.abs(x[0]) + Math.abs(x[1]));
const prox = (x: number[], t: number) => softThreshold(x, 0.5 * t);

const proxResult = proximalGradient(lsGrad, prox, lassoObj, [5, 5], 0.1);
console.log(`近端梯度: x*=[${proxResult.x.map((x) => x.toFixed(4))}]`);

加速梯度法
Nesterov 加速梯度法（Accelerated Gradient Method）通过引入动量项达到了凸优化的最优一阶收敛速率 O(1/k²)，比标准梯度下降的 O(1/k) 快一个量级。


function nesterovAccelerated(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  lr: number = 0.01,
  maxIter: number = 500,
): { x: number[]; history: number[] } {
  let x = [...x0];
  let y = [...x0];
  let xPrev = [...x0];
  const history: number[] = [f(x)];

  for (let k = 1; k <= maxIter; k++) {
    const momentum = (k - 1) / (k + 2);
    y = x.map((xi, i) => xi + momentum * (xi - xPrev[i]));

    const g = grad(y);
    xPrev = [...x];
    x = y.map((yi, i) => yi - lr * g[i]);
    history.push(f(x));
  }

  return { x, history };
}

const highCondF = (x: number[]) => x[0] ** 2 + 100 * x[1] ** 2;
const highCondG = (x: number[]) => [2 * x[0], 200 * x[1]];
const start = [10, 10];

const gdResult = gradientDescent(highCondF, highCondG, start, { lr: 0.005, maxIter: 200 });
const nagResult = nesterovAccelerated(highCondF, highCondG, start, 0.005, 200);

console.log(`GD (200 iter): f=${gdResult.history[gdResult.history.length - 1].toFixed(6)}`);
console.log(`NAG (200 iter): f=${nagResult.history[nagResult.history.length - 1].toFixed(6)}`);
小结
    

  • 梯度下降是最基础的一阶优化方法，收敛速率由函数的条件数 κ 决定
    • 

  • 回溯线搜索通过 Armijo 条件自适应选择步长，避免了估计 L 的需要
    • 

  • 近端梯度法处理复合目标函数 f + g，通过近端算子处理非光滑部分
    • 

  • Nesterov 加速达到一阶方法的最优收敛速率 O(1/k²)，比标准梯度下降快一个量级
    • 

  • 坐标下降适用于可分离问题，是 Lasso 等大规模稀疏问题的标准求解方法
    • 

  • 选择合适的梯度下降变体取决于问题结构：光滑性、约束类型和维度
    •
