动量方法

核心概念

动量方法（Momentum Methods）是一类通过累积历史梯度信息来加速优化的算法。经典动量法（Heavy Ball Method）由 Polyak 于 1964 年提出，通过引入速度项实现惯性效果：v*{k+1} = βv_k - α∇f(x_k)，x*{k+1} = xk + v{k+1}。动量不仅加速了沿低曲率方向的移动，还减少了沿高曲率方向的振荡。

Nesterov 动量（NAG）进一步改进了经典动量：先根据动量"预测"未来位置，再在预测位置计算梯度。NAG 在凸优化中达到了一阶方法的最优收敛速率 O(1/k²)。

经典动量与 Nesterov 加速
经典动量（Heavy Ball）和 Nesterov 加速梯度（NAG）是两种最基本的动量方法。它们的关键区别在于梯度计算的位置。


function heavyBall(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  alpha: number = 0.01,
  beta: number = 0.9,
  maxIter: number = 500,
): { x: number[]; history: number[] } {
  let x = [...x0];
  let v = Array(x0.length).fill(0);
  const history: number[] = [f(x)];

  for (let k = 0; k < maxIter; k++) {
    const g = grad(x);
    v = v.map((vi, i) => beta * vi - alpha * g[i]);
    x = x.map((xi, i) => xi + v[i]);
    history.push(f(x));
  }

  return { x, history };
}

function nesterovMomentum(
  f: (x: number[]) => number,
  grad: (x: number[]) => number[],
  x0: number[],
  alpha: number = 0.01,
  beta: number = 0.9,
  maxIter: number = 500,
): { x: number[]; history: number[] } {
  let x = [...x0];
  let v = Array(x0.length).fill(0);
  const history: number[] = [f(x)];

  for (let k = 0; k < maxIter; k++) {
    const lookAhead = x.map((xi, i) => xi + beta * v[i]);
    const g = grad(lookAhead);
    v = v.map((vi, i) => beta * vi - alpha * g[i]);
    x = x.map((xi, i) => xi + v[i]);
    history.push(f(x));
  }

  return { x, history };
}

function conjugateGradient(
  A: number[][],
  b: number[],
  x0: number[],
  maxIter?: number,
): { x: number[]; iterations: number } {
  const n = b.length;
  let x = [...x0];
  const matVec = (M: number[][], v: number[]): number[] => M.map((row) => row.reduce((s, mij, j) => s + mij * v[j], 0));

  let r = b.map((bi, i) => bi - matVec(A, x)[i]);
  let p = [...r];
  let rsOld = r.reduce((s, ri) => s + ri * ri, 0);
  const iter = maxIter ?? n;

  for (let k = 0; k < iter; k++) {
    const Ap = matVec(A, p);
    const pAp = p.reduce((s, pi, i) => s + pi * Ap[i], 0);
    if (Math.abs(pAp) < 1e-12) break;

    const alpha2 = rsOld / pAp;
    x = x.map((xi, i) => xi + alpha2 * p[i]);
    r = r.map((ri, i) => ri - alpha2 * Ap[i]);

    const rsNew = r.reduce((s, ri) => s + ri * ri, 0);
    if (rsNew < 1e-16) return { x, iterations: k + 1 };

    p = r.map((ri, i) => ri + (rsNew / rsOld) * p[i]);
    rsOld = rsNew;
  }

  return { x, iterations: iter };
}

const quadF = (x: number[]) => 0.5 * (x[0] ** 2 + 100 * x[1] ** 2);
const quadG = (x: number[]) => [x[0], 100 * x[1]];
const start = [10, 10];

const hbResult = heavyBall(quadF, quadG, start, 0.005, 0.8, 300);
const nagResult = nesterovMomentum(quadF, quadG, start, 0.005, 0.8, 300);
console.log(`Heavy Ball: f=${hbResult.history[hbResult.history.length - 1].toFixed(6)}`);
console.log(`Nesterov: f=${nagResult.history[nagResult.history.length - 1].toFixed(6)}`);

const A = [
  [2, 0],
  [0, 200],
];
const b_vec = [0, 0];
const cgResult = conjugateGradient(A, b_vec, [10, 10]);
console.log(`CG: ${cgResult.iterations} 迭代, x=[${cgResult.x.map((x) => x.toFixed(6))}]`);
自适应学习率方法
AdaGrad、RMSProp 和 Adam 等自适应方法为每个参数维护不同的学习率，根据历史梯度信息自动调整。这些方法在深度学习中广泛使用。


function adagrad(
  grad: (x: number[]) => number[],
  x0: number[],
  lr: number = 0.1,
  maxIter: number = 500,
  f?: (x: number[]) => number,
): { x: number[]; history: number[] } {
  let x = [...x0];
  const accum = Array(x0.length).fill(0);
  const eps = 1e-8;
  const history: number[] = f ? [f(x)] : [];

  for (let k = 0; k < maxIter; k++) {
    const g = grad(x);
    for (let i = 0; i < x.length; i++) {
      accum[i] += g[i] * g[i];
      x[i] -= (lr * g[i]) / (Math.sqrt(accum[i]) + eps);
    }
    if (f) history.push(f(x));
  }

  return { x, history };
}

function rmsprop(
  grad: (x: number[]) => number[],
  x0: number[],
  lr: number = 0.01,
  gamma: number = 0.9,
  maxIter: number = 500,
  f?: (x: number[]) => number,
): { x: number[]; history: number[] } {
  let x = [...x0];
  const accum = Array(x0.length).fill(0);
  const eps = 1e-8;
  const history: number[] = f ? [f(x)] : [];

  for (let k = 0; k < maxIter; k++) {
    const g = grad(x);
    for (let i = 0; i < x.length; i++) {
      accum[i] = gamma * accum[i] + (1 - gamma) * g[i] * g[i];
      x[i] -= (lr * g[i]) / (Math.sqrt(accum[i]) + eps);
    }
    if (f) history.push(f(x));
  }

  return { x, history };
}

function adam(
  grad: (x: number[]) => number[],
  x0: number[],
  lr: number = 0.001,
  beta1: number = 0.9,
  beta2: number = 0.999,
  maxIter: number = 500,
  f?: (x: number[]) => number,
): { x: number[]; history: number[] } {
  let x = [...x0];
  const m = Array(x0.length).fill(0);
  const v = Array(x0.length).fill(0);
  const eps = 1e-8;
  const history: number[] = f ? [f(x)] : [];

  for (let k = 1; k <= maxIter; k++) {
    const g = grad(x);
    for (let i = 0; i < x.length; i++) {
      m[i] = beta1 * m[i] + (1 - beta1) * g[i];
      v[i] = beta2 * v[i] + (1 - beta2) * g[i] * g[i];
      const mHat = m[i] / (1 - Math.pow(beta1, k));
      const vHat = v[i] / (1 - Math.pow(beta2, k));
      x[i] -= (lr * mHat) / (Math.sqrt(vHat) + eps);
    }
    if (f) history.push(f(x));
  }

  return { x, history };
}

const adamResult = adam(quadG, [10, 10], 0.1, 0.9, 0.999, 300, quadF);
const rmspropResult = rmsprop(quadG, [10, 10], 0.01, 0.9, 300, quadF);
console.log(`Adam: f=${adamResult.history[adamResult.history.length - 1].toFixed(6)}`);
console.log(`RMSProp: f=${rmspropResult.history[rmspropResult.history.length - 1].toFixed(6)}`);

动量方法的理论分析
动量方法的加速效果可以通过多项式近似理论精确分析。Nesterov 加速的最优性来自于 Chebyshev 多项式在一致范数下的最优近似性质。


function analyzeConvergence(conditionNumber: number, maxIter: number): { gd: number[]; hb: number[]; nag: number[] } {
  const kappa = conditionNumber;
  const gd: number[] = [];
  const hb: number[] = [];
  const nag: number[] = [];

  for (let k = 0; k <= maxIter; k++) {
    gd.push(Math.pow((kappa - 1) / (kappa + 1), k));
    hb.push(Math.pow((Math.sqrt(kappa) - 1) / (Math.sqrt(kappa) + 1), k));
    nag.push(Math.pow((Math.sqrt(kappa) - 1) / (Math.sqrt(kappa) + 1), k));
  }

  return { gd, hb, nag };
}

const analysis = analyzeConvergence(100, 50);
console.log(`条件数 κ=100:`);
console.log(`GD 50步残差: ${analysis.gd[50].toExponential(4)}`);
console.log(`HB 50步残差: ${analysis.hb[50].toExponential(4)}`);
console.log(`NAG 50步残差: ${analysis.nag[50].toExponential(4)}`);

for (const kappa of [10, 100, 1000]) {
  const gdRate = (kappa - 1) / (kappa + 1);
  const nagRate = (Math.sqrt(kappa) - 1) / (Math.sqrt(kappa) + 1);
  console.log(`κ=${kappa}: GD rate=${gdRate.toFixed(4)}, NAG rate=${nagRate.toFixed(4)}`);
}
小结
    

  • 经典动量通过累积速度加速优化，减少高曲率方向的振荡
    • 

  • Nesterov 加速在凸函数上达到一阶方法的最优收敛速率 O(1/k²)
    • 

  • 共轭梯度法对 n 维二次函数最多 n 步精确收敛，无需调参
    • 

  • Adam 结合一阶动量和自适应学习率，是深度学习的默认优化器
    • 

  • 条件数 κ 决定动量方法的加速效果：κ 越大，动量加速越显著
    • 

  • 选择优化器取决于问题性质：凸优化用 Nesterov/CG，深度学习用 Adam/AdamW
    •
