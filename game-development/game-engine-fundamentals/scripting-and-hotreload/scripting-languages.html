脚本语言集成

核心概念

脚本语言集成允许在游戏引擎中使用高级脚本语言编写游戏逻辑，而无需重新编译引擎核心。脚本语言提供了快速迭代、热重载和降低开发门槛的能力，是现代游戏引擎不可或缺的功能。

脚本虚拟机与绑定
脚本语言通过虚拟机（VM）在引擎中运行。绑定层负责将引擎的 C++/TypeScript API 暴露给脚本，并处理两种语言之间的类型转换和内存管理。


interface ScriptFunction {
  name: string;
  argTypes: string[];
  returnType: string;
  handler: (...args: unknown[]) => unknown;
}

interface ScriptClass {
  name: string;
  methods: ScriptFunction[];
  properties: Array<{ name: string; type: string; getter: () => unknown; setter: (v: unknown) => void }>;
}

class ScriptBindings {
  private functions = new Map<string, ScriptFunction>();
  private classes = new Map<string, ScriptClass>();

  registerFunction(fn: ScriptFunction): void {
    this.functions.set(fn.name, fn);
  }

  registerClass(cls: ScriptClass): void {
    this.classes.set(cls.name, cls);
  }

  callFunction(name: string, ...args: unknown[]): unknown {
    const fn = this.functions.get(name);
    if (!fn) throw new Error(`Unregistered function: ${name}`);
    return fn.handler(...args);
  }

  getExposedAPI(): { functions: string[]; classes: string[] } {
    return {
      functions: [...this.functions.keys()],
      classes: [...this.classes.keys()],
    };
  }
}

class ScriptVM {
  private bindings = new ScriptBindings();
  private scripts = new Map<string, CompiledScript>();
  private context: ScriptContext;

  constructor() {
    this.context = { globals: new Map(), bindings: this.bindings };
    this.registerCoreBindings();
  }

  private registerCoreBindings(): void {
    this.bindings.registerFunction({
      name: 'log',
      argTypes: ['string'],
      returnType: 'void',
      handler: (msg: unknown) => console.log(`[Script] ${msg}`),
    });

    this.bindings.registerFunction({
      name: 'getEntity',
      argTypes: ['number'],
      returnType: 'object',
      handler: (id: unknown) => ({ id, type: 'entity' }),
    });

    this.bindings.registerFunction({
      name: 'setPosition',
      argTypes: ['number', 'number', 'number', 'number'],
      returnType: 'void',
      handler: (_id: unknown, _x: unknown, _y: unknown, _z: unknown) => {},
    });
  }

  loadScript(name: string, source: string): void {
    const compiled = this.compile(source);
    this.scripts.set(name, compiled);
  }

  executeScript(name: string): unknown {
    const script = this.scripts.get(name);
    if (!script) throw new Error(`Script not found: ${name}`);
    return this.execute(script, this.context);
  }

  getBindings(): ScriptBindings {
    return this.bindings;
  }

  private compile(_source: string): CompiledScript {
    return { bytecode: new Uint8Array() };
  }
  private execute(_script: CompiledScript, _ctx: ScriptContext): unknown {
    return undefined;
  }
}

interface CompiledScript {
  bytecode: Uint8Array;
}
interface ScriptContext {
  globals: Map<string, unknown>;
  bindings: ScriptBindings;
}
脚本生命周期回调
游戏脚本需要在特定时机执行特定逻辑。引擎通过生命周期回调机制，在初始化、每帧更新、物理更新、销毁等时刻调用脚本中的对应函数。


interface ScriptComponent {
  scriptName: string;
  instance: ScriptInstance | null;
  enabled: boolean;
}

interface ScriptInstance {
  onInit?(): void;
  onEnable?(): void;
  onUpdate?(dt: number): void;
  onFixedUpdate?(fixedDt: number): void;
  onLateUpdate?(dt: number): void;
  onDisable?(): void;
  onDestroy?(): void;
  onCollisionEnter?(other: unknown): void;
  onCollisionExit?(other: unknown): void;
}

class ScriptSystem {
  private components: ScriptComponent[] = [];
  private vm: ScriptVM;

  constructor(vm: ScriptVM) {
    this.vm = vm;
  }

  addScript(entityId: number, scriptName: string): ScriptComponent {
    const component: ScriptComponent = {
      scriptName,
      instance: null,
      enabled: true,
    };
    this.components.push(component);
    this.initializeScript(component);
    return component;
  }

  update(dt: number): void {
    for (const comp of this.components) {
      if (comp.enabled && comp.instance?.onUpdate) {
        try {
          comp.instance.onUpdate(dt);
        } catch (error) {
          console.error(`Script error in ${comp.scriptName}:`, error);
          comp.enabled = false;
        }
      }
    }
  }

  fixedUpdate(fixedDt: number): void {
    for (const comp of this.components) {
      if (comp.enabled && comp.instance?.onFixedUpdate) {
        try {
          comp.instance.onFixedUpdate(fixedDt);
        } catch (error) {
          console.error(`Script error in ${comp.scriptName}:`, error);
        }
      }
    }
  }

  private initializeScript(component: ScriptComponent): void {
    try {
      this.vm.loadScript(component.scriptName, '');
      component.instance = this.vm.executeScript(component.scriptName) as ScriptInstance;
      component.instance?.onInit?.();
      component.instance?.onEnable?.();
    } catch (error) {
      console.error(`Failed to init script ${component.scriptName}:`, error);
    }
  }
}
脚本沙箱与安全
脚本安全在支持用户生成内容（UGC）和模组的游戏中至关重要。沙箱限制脚本可访问的 API 和资源，防止恶意代码破坏系统。


interface SandboxConfig {
  allowedAPIs: Set<string>;
  maxExecutionTime: number;
  maxMemory: number;
  allowFileAccess: boolean;
  allowNetworkAccess: boolean;
}

class ScriptSandbox {
  private config: SandboxConfig;
  private executionStart = 0;

  constructor(config: SandboxConfig) {
    this.config = config;
  }

  createSafeBindings(fullBindings: ScriptBindings): ScriptBindings {
    const safe = new ScriptBindings();
    const exposed = fullBindings.getExposedAPI();

    for (const fnName of exposed.functions) {
      if (this.config.allowedAPIs.has(fnName)) {
        safe.registerFunction({
          name: fnName,
          argTypes: [],
          returnType: 'unknown',
          handler: (...args: unknown[]) => {
            this.checkTimeLimit();
            return fullBindings.callFunction(fnName, ...args);
          },
        });
      }
    }
    return safe;
  }

  private checkTimeLimit(): void {
    const elapsed = performance.now() - this.executionStart;
    if (elapsed > this.config.maxExecutionTime) {
      throw new Error('Script execution timeout');
    }
  }

  beginExecution(): void {
    this.executionStart = performance.now();
  }
}
脚本调试与性能分析
脚本调试器提供断点、单步执行和变量查看等功能，帮助开发者快速定位脚本中的问题。性能分析器追踪每个脚本函数的执行时间，识别性能瓶颈。


interface ScriptProfile {
  functionName: string;
  callCount: number;
  totalTime: number;
  averageTime: number;
  maxTime: number;
}

class ScriptProfiler {
  private profiles = new Map<string, ScriptProfile>();
  private enabled = false;

  enable(): void {
    this.enabled = true;
  }
  disable(): void {
    this.enabled = false;
  }

  beginProfile(functionName: string): () => void {
    if (!this.enabled) return () => {};
    const start = performance.now();
    return () => {
      const elapsed = performance.now() - start;
      const profile = this.profiles.get(functionName) ?? {
        functionName,
        callCount: 0,
        totalTime: 0,
        averageTime: 0,
        maxTime: 0,
      };
      profile.callCount++;
      profile.totalTime += elapsed;
      profile.averageTime = profile.totalTime / profile.callCount;
      profile.maxTime = Math.max(profile.maxTime, elapsed);
      this.profiles.set(functionName, profile);
    };
  }

  getTopFunctions(count: number): ScriptProfile[] {
    return [...this.profiles.values()].sort((a, b) => b.totalTime - a.totalTime).slice(0, count);
  }

  reset(): void {
    this.profiles.clear();
  }
}
