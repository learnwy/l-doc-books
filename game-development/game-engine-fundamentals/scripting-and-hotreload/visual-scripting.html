可视化脚本

核心概念

可视化脚本（Visual Scripting）通过图形化节点和连线替代传统文本代码，使非程序员（策划、美术）也能定义游戏逻辑。每个节点代表一个操作或数据，连线表示数据流或执行流，整个脚本呈现为一张有向图。

节点系统设计
可视化脚本的核心是节点系统。每个节点有输入端口（Input Pin）和输出端口（Output Pin），端口分为执行端口（控制流程）和数据端口（传递值）。


type PinType = 'exec' | 'boolean' | 'number' | 'string' | 'vector3' | 'entity' | 'any';
type PinDirection = 'input' | 'output';

interface Pin {
  id: string;
  name: string;
  direction: PinDirection;
  type: PinType;
  connectedTo: string | null;
  defaultValue?: unknown;
}

interface VisualNode {
  id: string;
  type: string;
  title: string;
  position: { x: number; y: number };
  inputPins: Pin[];
  outputPins: Pin[];
  properties: Record<string, unknown>;
}

interface Connection {
  id: string;
  sourceNodeId: string;
  sourcePinId: string;
  targetNodeId: string;
  targetPinId: string;
}

interface VisualGraph {
  id: string;
  name: string;
  nodes: VisualNode[];
  connections: Connection[];
  variables: Array<{ name: string; type: PinType; defaultValue: unknown }>;
}

class NodeRegistry {
  private definitions = new Map<string, NodeDefinition>();

  register(def: NodeDefinition): void {
    this.definitions.set(def.type, def);
  }

  createNode(type: string, x: number, y: number): VisualNode {
    const def = this.definitions.get(type);
    if (!def) throw new Error(`Unknown node type: ${type}`);

    return {
      id: crypto.randomUUID(),
      type: def.type,
      title: def.title,
      position: { x, y },
      inputPins: def.inputs.map((p) => ({ ...p, id: crypto.randomUUID(), connectedTo: null })),
      outputPins: def.outputs.map((p) => ({ ...p, id: crypto.randomUUID(), connectedTo: null })),
      properties: { ...def.defaultProperties },
    };
  }

  getCategories(): Map<string, NodeDefinition[]> {
    const categories = new Map<string, NodeDefinition[]>();
    for (const def of this.definitions.values()) {
      const list = categories.get(def.category) ?? [];
      list.push(def);
      categories.set(def.category, list);
    }
    return categories;
  }
}

interface NodeDefinition {
  type: string;
  title: string;
  category: string;
  inputs: Omit<Pin, 'id' | 'connectedTo'>[];
  outputs: Omit<Pin, 'id' | 'connectedTo'>[];
  defaultProperties: Record<string, unknown>;
}
图执行引擎
可视化脚本图需要一个执行引擎来驱动节点的计算。执行引擎沿着执行流连线遍历节点，在需要数据时沿数据流回溯计算输入值。


type NodeExecutor = (inputs: Map<string, unknown>, context: ExecutionContext) => Map<string, unknown>;

interface ExecutionContext {
  variables: Map<string, unknown>;
  deltaTime: number;
  entityId: number;
  getNodeExecutor(type: string): NodeExecutor;
}

class GraphExecutor {
  private executors = new Map<string, NodeExecutor>();

  registerExecutor(nodeType: string, executor: NodeExecutor): void {
    this.executors.set(nodeType, executor);
  }

  execute(graph: VisualGraph, entryNodeId: string, context: ExecutionContext): void {
    context.getNodeExecutor = (type: string) => this.executors.get(type)!;
    let currentNodeId: string | null = entryNodeId;

    while (currentNodeId) {
      const node = graph.nodes.find((n) => n.id === currentNodeId);
      if (!node) break;

      const inputs = this.resolveInputs(graph, node, context);
      const executor = this.executors.get(node.type);
      if (!executor) break;

      const outputs = executor(inputs, context);
      currentNodeId = this.getNextExecNode(graph, node, outputs);
    }
  }

  private resolveInputs(graph: VisualGraph, node: VisualNode, context: ExecutionContext): Map<string, unknown> {
    const inputs = new Map<string, unknown>();

    for (const pin of node.inputPins) {
      if (pin.type === 'exec') continue;

      if (pin.connectedTo) {
        const connection = graph.connections.find((c) => c.targetPinId === pin.id);
        if (connection) {
          const sourceNode = graph.nodes.find((n) => n.id === connection.sourceNodeId);
          if (sourceNode) {
            const sourceInputs = this.resolveInputs(graph, sourceNode, context);
            const executor = this.executors.get(sourceNode.type);
            if (executor) {
              const sourceOutputs = executor(sourceInputs, context);
              const sourcePin = sourceNode.outputPins.find((p) => p.id === connection.sourcePinId);
              if (sourcePin) {
                inputs.set(pin.name, sourceOutputs.get(sourcePin.name));
              }
            }
          }
        }
      } else {
        inputs.set(pin.name, pin.defaultValue);
      }
    }
    return inputs;
  }

  private getNextExecNode(graph: VisualGraph, node: VisualNode, outputs: Map<string, unknown>): string | null {
    const execOut = node.outputPins.find((p) => p.type === 'exec');
    if (!execOut) return null;

    const nextPin = outputs.get('__exec_out') as string | undefined;
    if (nextPin) {
      const conn = graph.connections.find((c) => c.sourcePinId === nextPin);
      return conn?.targetNodeId ?? null;
    }

    const conn = graph.connections.find((c) => c.sourceNodeId === node.id && c.sourcePinId === execOut.id);
    return conn?.targetNodeId ?? null;
  }
}
图编译与优化
解释执行的可视化脚本性能较低。通过将图编译为字节码或源代码，可以显著提升运行时性能。编译阶段还可以执行常量折叠、死节点消除等优化。


interface BytecodeInstruction {
  opcode: number;
  operands: number[];
}

class GraphCompiler {
  private instructions: BytecodeInstruction[] = [];
  private constants: unknown[] = [];

  compile(graph: VisualGraph): CompiledGraph {
    this.instructions = [];
    this.constants = [];
    const entryNode = graph.nodes.find((n) => n.type === 'event_start');
    if (entryNode) {
      this.compileNode(graph, entryNode);
    }
    return {
      instructions: [...this.instructions],
      constants: [...this.constants],
      variableCount: graph.variables.length,
    };
  }

  private compileNode(graph: VisualGraph, node: VisualNode): void {
    for (const pin of node.inputPins) {
      if (pin.type !== 'exec' && pin.defaultValue !== undefined) {
        const constIndex = this.constants.length;
        this.constants.push(pin.defaultValue);
        this.emit(OpCode.LOAD_CONST, constIndex);
      }
    }

    switch (node.type) {
      case 'math_add':
        this.emit(OpCode.ADD);
        break;
      case 'math_multiply':
        this.emit(OpCode.MUL);
        break;
      case 'branch':
        this.emit(OpCode.BRANCH, 0);
        break;
      case 'set_variable':
        this.emit(OpCode.STORE_VAR, node.properties.varIndex as number);
        break;
      default:
        this.emit(OpCode.CALL_NODE, this.getNodeTypeId(node.type));
    }

    const execConn = graph.connections.find((c) => c.sourceNodeId === node.id);
    if (execConn) {
      const nextNode = graph.nodes.find((n) => n.id === execConn.targetNodeId);
      if (nextNode) this.compileNode(graph, nextNode);
    }
  }

  private emit(opcode: number, ...operands: number[]): void {
    this.instructions.push({ opcode, operands });
  }

  private getNodeTypeId(_type: string): number {
    return 0;
  }
}

enum OpCode {
  LOAD_CONST = 0x01,
  ADD = 0x10,
  MUL = 0x11,
  BRANCH = 0x20,
  STORE_VAR = 0x30,
  CALL_NODE = 0x40,
}

interface CompiledGraph {
  instructions: BytecodeInstruction[];
  constants: unknown[];
  variableCount: number;
}
与文本脚本的互操作
可视化脚本不应该是封闭系统。它需要与文本脚本双向互操作：可视化图可以调用文本函数，文本脚本可以触发可视化图中的事件。


class VisualTextBridge {
  private textFunctions = new Map<string, (...args: unknown[]) => unknown>();
  private graphTriggers = new Map<string, (args: Map<string, unknown>) => void>();

  registerTextFunction(name: string, fn: (...args: unknown[]) => unknown): void {
    this.textFunctions.set(name, fn);
  }

  registerGraphTrigger(eventName: string, trigger: (args: Map<string, unknown>) => void): void {
    this.graphTriggers.set(eventName, trigger);
  }

  callTextFromGraph(functionName: string, ...args: unknown[]): unknown {
    const fn = this.textFunctions.get(functionName);
    if (!fn) throw new Error(`Text function not found: ${functionName}`);
    return fn(...args);
  }

  triggerGraphFromText(eventName: string, args: Map<string, unknown>): void {
    const trigger = this.graphTriggers.get(eventName);
    if (trigger) trigger(args);
  }

  generateTypeStubs(): string {
    const lines: string[] = [];
    for (const [name] of this.textFunctions) {
      lines.push(`declare function ${name}(...args: unknown[]): unknown;`);
    }
    return lines.join('\n');
  }
}
小结
    

  • 可视化脚本通过节点和连线的图形化方式降低了编程门槛
    • 

  • 节点系统分离执行流和数据流，支持灵活的逻辑表达
    • 

  • 图执行引擎沿执行流驱动，按需通过数据流回溯获取输入
    • 

  • 编译为字节码可显著提升运行时性能
    • 

  • 与文本脚本的互操作确保了可视化脚本的扩展能力
