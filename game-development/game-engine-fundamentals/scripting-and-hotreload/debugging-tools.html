调试工具

核心概念

游戏调试工具是开发者定位和修复问题的核心手段。与普通应用不同，游戏的实时性、状态复杂性和视觉交互性对调试工具提出了更高要求。优秀的调试工具能将问题定位时间从数小时缩短到数分钟。

调试绘制系统

调试绘制（Debug Draw）在游戏画面上叠加渲染线条、形状和文本，用于可视化碰撞体、射线、路径和 AI 感知范围等不可见的游戏数据。

interface DebugLine {
  from: { x: number; y: number; z: number };
  to: { x: number; y: number; z: number };
  color: number;
  duration: number;
  timeRemaining: number;
}

interface DebugText {
  position: { x: number; y: number; z: number };
  text: string;
  color: number;
  size: number;
  duration: number;
  timeRemaining: number;
}

class DebugDrawSystem {
  private lines: DebugLine[] = [];
  private texts: DebugText[] = [];
  private enabled = true;
  private maxLines = 10000;
  private maxTexts = 1000;

  drawLine(
    from: { x: number; y: number; z: number },
    to: { x: number; y: number; z: number },
    color = 0x00ff00,
    duration = 0,
  ): void {
    if (!this.enabled || this.lines.length >= this.maxLines) return;
    this.lines.push({ from, to, color, duration, timeRemaining: duration });
  }

  drawBox(
    center: { x: number; y: number; z: number },
    size: { x: number; y: number; z: number },
    color = 0x00ff00,
    duration = 0,
  ): void {
    const hx = size.x / 2,
      hy = size.y / 2,
      hz = size.z / 2;
    const corners = [
      { x: center.x - hx, y: center.y - hy, z: center.z - hz },
      { x: center.x + hx, y: center.y - hy, z: center.z - hz },
      { x: center.x + hx, y: center.y + hy, z: center.z - hz },
      { x: center.x - hx, y: center.y + hy, z: center.z - hz },
      { x: center.x - hx, y: center.y - hy, z: center.z + hz },
      { x: center.x + hx, y: center.y - hy, z: center.z + hz },
      { x: center.x + hx, y: center.y + hy, z: center.z + hz },
      { x: center.x - hx, y: center.y + hy, z: center.z + hz },
    ];
    const edges = [
      [0, 1],
      [1, 2],
      [2, 3],
      [3, 0],
      [4, 5],
      [5, 6],
      [6, 7],
      [7, 4],
      [0, 4],
      [1, 5],
      [2, 6],
      [3, 7],
    ];
    for (const [a, b] of edges) {
      this.drawLine(corners[a], corners[b], color, duration);
    }
  }

  drawSphere(
    center: { x: number; y: number; z: number },
    radius: number,
    color = 0x00ff00,
    segments = 16,
    duration = 0,
  ): void {
    for (let i = 0; i < segments; i++) {
      const a1 = (i / segments) * Math.PI * 2;
      const a2 = ((i + 1) / segments) * Math.PI * 2;
      this.drawLine(
        { x: center.x + Math.cos(a1) * radius, y: center.y, z: center.z + Math.sin(a1) * radius },
        { x: center.x + Math.cos(a2) * radius, y: center.y, z: center.z + Math.sin(a2) * radius },
        color,
        duration,
      );
    }
  }

  drawText(
    position: { x: number; y: number; z: number },
    text: string,
    color = 0xffffff,
    size = 14,
    duration = 0,
  ): void {
    if (!this.enabled || this.texts.length >= this.maxTexts) return;
    this.texts.push({ position, text, color, size, duration, timeRemaining: duration });
  }

  update(dt: number): void {
    this.lines = this.lines.filter((line) => {
      if (line.duration === 0) return false;
      line.timeRemaining -= dt;
      return line.timeRemaining > 0;
    });
    this.texts = this.texts.filter((text) => {
      if (text.duration === 0) return false;
      text.timeRemaining -= dt;
      return text.timeRemaining > 0;
    });
  }

  getLines(): ReadonlyArray<DebugLine> {
    return this.lines;
  }
  getTexts(): ReadonlyArray<DebugText> {
    return this.texts;
  }
  setEnabled(enabled: boolean): void {
    this.enabled = enabled;
  }
  clear(): void {
    this.lines = [];
    this.texts = [];
  }
}

游戏控制台

游戏控制台（Game Console）允许开发者在运行时输入命令来查询和修改游戏状态。它是快速测试和诊断的利器——无需重新编译即可修改参数、触发事件或切换功能开关。

type CommandHandler = (args: string[]) => string;

class GameConsole {
  private commands = new Map<string, { handler: CommandHandler; help: string }>();
  private history: string[] = [];
  private output: string[] = [];
  private maxHistory = 100;
  private maxOutput = 500;
  private variables = new Map<string, string>();

  registerCommand(name: string, handler: CommandHandler, help: string): void {
    this.commands.set(name, { handler, help });
  }

  execute(input: string): string {
    this.history.push(input);
    if (this.history.length > this.maxHistory) this.history.shift();

    const parts = this.parseInput(input);
    if (parts.length === 0) return '';

    const [commandName, ...args] = parts;
    const command = this.commands.get(commandName);

    if (!command) {
      const result = `Unknown command: ${commandName}`;
      this.log(result);
      return result;
    }

    try {
      const result = command.handler(args);
      this.log(result);
      return result;
    } catch (error) {
      const msg = `Error: ${error instanceof Error ? error.message : String(error)}`;
      this.log(msg);
      return msg;
    }
  }

  private parseInput(input: string): string[] {
    const parts: string[] = [];
    let current = '';
    let inQuotes = false;
    for (const char of input) {
      if (char === '"') {
        inQuotes = !inQuotes;
        continue;
      }
      if (char === ' ' && !inQuotes) {
        if (current) parts.push(current);
        current = '';
        continue;
      }
      current += char;
    }
    if (current) parts.push(current);
    return parts;
  }

  private log(message: string): void {
    this.output.push(message);
    if (this.output.length > this.maxOutput) this.output.shift();
  }

  getSuggestions(partial: string): string[] {
    return [...this.commands.keys()].filter((name) => name.startsWith(partial));
  }

  getOutput(): ReadonlyArray<string> {
    return this.output;
  }
  getHistory(): ReadonlyArray<string> {
    return this.history;
  }
}

性能分析器

性能分析器（Profiler）追踪各系统的 CPU/GPU 耗时，帮助开发者识别性能瓶颈。帧分析器显示每帧中各阶段的耗时占比，支持按热点排序和历史趋势对比。

interface ProfileScope {
  name: string;
  startTime: number;
  endTime: number;
  children: ProfileScope[];
  parent: ProfileScope | null;
}

class FrameProfiler {
  private currentFrame: ProfileScope | null = null;
  private scopeStack: ProfileScope[] = [];
  private frameHistory: ProfileScope[] = [];
  private maxFrames = 300;

  beginFrame(): void {
    this.currentFrame = { name: 'Frame', startTime: performance.now(), endTime: 0, children: [], parent: null };
    this.scopeStack = [this.currentFrame];
  }

  beginScope(name: string): void {
    const scope: ProfileScope = { name, startTime: performance.now(), endTime: 0, children: [], parent: null };
    const parent = this.scopeStack[this.scopeStack.length - 1];
    if (parent) {
      scope.parent = parent;
      parent.children.push(scope);
    }
    this.scopeStack.push(scope);
  }

  endScope(): void {
    const scope = this.scopeStack.pop();
    if (scope) scope.endTime = performance.now();
  }

  endFrame(): void {
    if (this.currentFrame) {
      this.currentFrame.endTime = performance.now();
      this.frameHistory.push(this.currentFrame);
      if (this.frameHistory.length > this.maxFrames) this.frameHistory.shift();
    }
    this.currentFrame = null;
    this.scopeStack = [];
  }

  getLastFrame(): ProfileScope | null {
    return this.frameHistory[this.frameHistory.length - 1] ?? null;
  }

  getAverageFrameTime(): number {
    if (this.frameHistory.length === 0) return 0;
    const total = this.frameHistory.reduce((sum, f) => sum + (f.endTime - f.startTime), 0);
    return total / this.frameHistory.length;
  }

  getHotSpots(count: number): Array<{ name: string; totalTime: number; callCount: number }> {
    const stats = new Map<string, { totalTime: number; callCount: number }>();
    for (const frame of this.frameHistory) {
      this.collectStats(frame, stats);
    }
    return [...stats.entries()]
      .map(([name, s]) => ({ name, ...s }))
      .sort((a, b) => b.totalTime - a.totalTime)
      .slice(0, count);
  }

  private collectStats(scope: ProfileScope, stats: Map<string, { totalTime: number; callCount: number }>): void {
    const existing = stats.get(scope.name) ?? { totalTime: 0, callCount: 0 };
    existing.totalTime += scope.endTime - scope.startTime;
    existing.callCount++;
    stats.set(scope.name, existing);
    for (const child of scope.children) {
      this.collectStats(child, stats);
    }
  }
}

即时调试 UI

即时调试 UI（Immediate Mode Debug UI）在游戏画面上叠加面板，显示实时数据并支持参数调节。ImGui 风格的即时模式 UI 不保持状态，每帧根据当前数据重新生成。

interface DebugPanel {
  name: string;
  visible: boolean;
  render: () => DebugWidget[];
}

type DebugWidget =
  | { type: 'text'; label: string; value: string }
  | { type: 'slider'; label: string; value: number; min: number; max: number; onChange: (v: number) => void }
  | { type: 'checkbox'; label: string; value: boolean; onChange: (v: boolean) => void }
  | { type: 'button'; label: string; onClick: () => void }
  | { type: 'graph'; label: string; values: number[]; min: number; max: number };

class DebugUI {
  private panels: DebugPanel[] = [];
  private globalVisible = false;

  registerPanel(panel: DebugPanel): void {
    this.panels.push(panel);
  }

  toggle(): void {
    this.globalVisible = !this.globalVisible;
  }

  render(): DebugWidget[][] {
    if (!this.globalVisible) return [];
    return this.panels.filter((p) => p.visible).map((p) => p.render());
  }

  getPanels(): ReadonlyArray<DebugPanel> {
    return this.panels;
  }
  isVisible(): boolean {
    return this.globalVisible;
  }
}
