空间分区

核心概念

空间分区（Spatial Partitioning）是将游戏世界划分为更小区域的技术，用于加速空间查询——如碰撞检测、视野裁剪和邻近搜索。没有空间分区，每次查询都需要遍历所有对象（O(n²)），在大规模场景中性能无法接受。

均匀网格
均匀网格将空间划分为等大小的格子，每个格子维护一个对象列表。查询时只需检查目标所在格子及相邻格子中的对象。


class SpatialGrid<T extends SpatialObject> {
  private cells = new Map<string, T[]>();
  private cellSize: number;

  constructor(cellSize: number) {
    this.cellSize = cellSize;
  }

  private getKey(cx: number, cy: number): string {
    return `${cx},${cy}`;
  }

  private getCellCoords(x: number, y: number): [number, number] {
    return [Math.floor(x / this.cellSize), Math.floor(y / this.cellSize)];
  }

  insert(obj: T): void {
    const [cx, cy] = this.getCellCoords(obj.x, obj.y);
    const key = this.getKey(cx, cy);
    if (!this.cells.has(key)) {
      this.cells.set(key, []);
    }
    this.cells.get(key)!.push(obj);
  }

  remove(obj: T): void {
    const [cx, cy] = this.getCellCoords(obj.x, obj.y);
    const key = this.getKey(cx, cy);
    const cell = this.cells.get(key);
    if (cell) {
      const index = cell.indexOf(obj);
      if (index >= 0) cell.splice(index, 1);
    }
  }

  queryRadius(x: number, y: number, radius: number): T[] {
    const results: T[] = [];
    const [minCx, minCy] = this.getCellCoords(x - radius, y - radius);
    const [maxCx, maxCy] = this.getCellCoords(x + radius, y + radius);
    const radiusSq = radius * radius;

    for (let cx = minCx; cx <= maxCx; cx++) {
      for (let cy = minCy; cy <= maxCy; cy++) {
        const cell = this.cells.get(this.getKey(cx, cy));
        if (!cell) continue;
        for (const obj of cell) {
          const dx = obj.x - x;
          const dy = obj.y - y;
          if (dx * dx + dy * dy <= radiusSq) {
            results.push(obj);
          }
        }
      }
    }
    return results;
  }

  clear(): void {
    this.cells.clear();
  }
}

interface SpatialObject {
  x: number;
  y: number;
}
四叉树
四叉树递归地将 2D 空间分割为四个象限。当一个节点中的对象数量超过阈值时，该节点分裂为四个子节点。这种自适应分割使得密集区域获得更精细的划分。


interface AABB {
  x: number;
  y: number;
  width: number;
  height: number;
}

class QuadTree<T extends SpatialObject> {
  private objects: T[] = [];
  private children: QuadTree<T>[] | null = null;
  private bounds: AABB;
  private readonly maxObjects: number;
  private readonly maxDepth: number;
  private depth: number;

  constructor(bounds: AABB, maxObjects = 8, maxDepth = 6, depth = 0) {
    this.bounds = bounds;
    this.maxObjects = maxObjects;
    this.maxDepth = maxDepth;
    this.depth = depth;
  }

  insert(obj: T): void {
    if (!this.contains(obj)) return;

    if (this.children) {
      for (const child of this.children) {
        child.insert(obj);
      }
      return;
    }

    this.objects.push(obj);

    if (this.objects.length > this.maxObjects && this.depth < this.maxDepth) {
      this.subdivide();
    }
  }

  queryRange(range: AABB): T[] {
    const results: T[] = [];
    if (!this.intersects(range)) return results;

    for (const obj of this.objects) {
      if (obj.x >= range.x && obj.x <= range.x + range.width && obj.y >= range.y && obj.y <= range.y + range.height) {
        results.push(obj);
      }
    }

    if (this.children) {
      for (const child of this.children) {
        results.push(...child.queryRange(range));
      }
    }

    return results;
  }

  private subdivide(): void {
    const { x, y, width, height } = this.bounds;
    const hw = width / 2;
    const hh = height / 2;

    this.children = [
      new QuadTree({ x, y, width: hw, height: hh }, this.maxObjects, this.maxDepth, this.depth + 1),
      new QuadTree({ x: x + hw, y, width: hw, height: hh }, this.maxObjects, this.maxDepth, this.depth + 1),
      new QuadTree({ x, y: y + hh, width: hw, height: hh }, this.maxObjects, this.maxDepth, this.depth + 1),
      new QuadTree({ x: x + hw, y: y + hh, width: hw, height: hh }, this.maxObjects, this.maxDepth, this.depth + 1),
    ];

    for (const obj of this.objects) {
      for (const child of this.children) {
        child.insert(obj);
      }
    }
    this.objects = [];
  }

  private contains(obj: SpatialObject): boolean {
    return (
      obj.x >= this.bounds.x &&
      obj.x <= this.bounds.x + this.bounds.width &&
      obj.y >= this.bounds.y &&
      obj.y <= this.bounds.y + this.bounds.height
    );
  }

  private intersects(range: AABB): boolean {
    return !(
      range.x > this.bounds.x + this.bounds.width ||
      range.x + range.width < this.bounds.x ||
      range.y > this.bounds.y + this.bounds.height ||
      range.y + range.height < this.bounds.y
    );
  }
}
BVH 层次包围体
BVH（Bounding Volume Hierarchy）将对象的包围体组织为二叉树，每个内部节点的包围体恰好包含其两个子节点的包围体。BVH 特别适合射线查询和视锥裁剪。


interface BoundingSphere {
  cx: number;
  cy: number;
  cz: number;
  radius: number;
}

interface BVHNode<T> {
  bounds: BoundingSphere;
  left: BVHNode<T> | null;
  right: BVHNode<T> | null;
  object: T | null;
}

class BVHTree<T extends { bounds: BoundingSphere }> {
  root: BVHNode<T> | null = null;

  build(objects: T[]): void {
    this.root = this.buildRecursive(objects);
  }

  private buildRecursive(objects: T[]): BVHNode<T> | null {
    if (objects.length === 0) return null;
    if (objects.length === 1) {
      return { bounds: objects[0].bounds, left: null, right: null, object: objects[0] };
    }

    const axis = this.longestAxis(objects);
    objects.sort((a, b) => this.getCenter(a.bounds, axis) - this.getCenter(b.bounds, axis));

    const mid = Math.floor(objects.length / 2);
    const left = this.buildRecursive(objects.slice(0, mid));
    const right = this.buildRecursive(objects.slice(mid));

    return {
      bounds: this.mergeBounds(left?.bounds, right?.bounds),
      left,
      right,
      object: null,
    };
  }

  queryRay(origin: Vec3, direction: Vec3): T[] {
    const results: T[] = [];
    if (this.root) {
      this.rayTraverse(this.root, origin, direction, results);
    }
    return results;
  }

  private rayTraverse(node: BVHNode<T>, origin: Vec3, dir: Vec3, results: T[]): void {
    if (!this.raySphereIntersect(origin, dir, node.bounds)) return;
    if (node.object) {
      results.push(node.object);
      return;
    }
    if (node.left) this.rayTraverse(node.left, origin, dir, results);
    if (node.right) this.rayTraverse(node.right, origin, dir, results);
  }

  private raySphereIntersect(origin: Vec3, _dir: Vec3, sphere: BoundingSphere): boolean {
    const dx = origin.x - sphere.cx;
    const dy = origin.y - sphere.cy;
    const dz = origin.z - sphere.cz;
    return dx * dx + dy * dy + dz * dz <= sphere.radius * sphere.radius * 4;
  }

  private longestAxis(objects: T[]): 'x' | 'y' | 'z' {
    let minX = Infinity,
      maxX = -Infinity;
    let minY = Infinity,
      maxY = -Infinity;
    let minZ = Infinity,
      maxZ = -Infinity;
    for (const obj of objects) {
      const b = obj.bounds;
      minX = Math.min(minX, b.cx);
      maxX = Math.max(maxX, b.cx);
      minY = Math.min(minY, b.cy);
      maxY = Math.max(maxY, b.cy);
      minZ = Math.min(minZ, b.cz);
      maxZ = Math.max(maxZ, b.cz);
    }
    const dx = maxX - minX,
      dy = maxY - minY,
      dz = maxZ - minZ;
    if (dx >= dy && dx >= dz) return 'x';
    if (dy >= dz) return 'y';
    return 'z';
  }

  private getCenter(b: BoundingSphere, axis: 'x' | 'y' | 'z'): number {
    if (axis === 'x') return b.cx;
    if (axis === 'y') return b.cy;
    return b.cz;
  }

  private mergeBounds(a?: BoundingSphere, b?: BoundingSphere): BoundingSphere {
    if (!a && !b) return { cx: 0, cy: 0, cz: 0, radius: 0 };
    if (!a) return b!;
    if (!b) return a;
    const cx = (a.cx + b.cx) / 2;
    const cy = (a.cy + b.cy) / 2;
    const cz = (a.cz + b.cz) / 2;
    const r = Math.max(
      Math.sqrt((a.cx - cx) ** 2 + (a.cy - cy) ** 2 + (a.cz - cz) ** 2) + a.radius,
      Math.sqrt((b.cx - cx) ** 2 + (b.cy - cy) ** 2 + (b.cz - cz) ** 2) + b.radius,
    );
    return { cx, cy, cz, radius: r };
  }
}

interface Vec3 {
  x: number;
  y: number;
  z: number;
}
动态空间分区
对于移动的对象，空间分区结构需要频繁更新。常见的优化策略包括松散四叉树（Loose Quadtree）、延迟更新和双缓冲分区。


class LooseGrid<T extends SpatialObject & { id: number }> {
  private cellSize: number;
  private looseness: number;
  private cells = new Map<string, Set<T>>();
  private objectCells = new Map<number, string>();

  constructor(cellSize: number, looseness = 2.0) {
    this.cellSize = cellSize;
    this.looseness = looseness;
  }

  update(obj: T): void {
    const oldKey = this.objectCells.get(obj.id);
    const newKey = this.getCellKey(obj.x, obj.y);

    if (oldKey === newKey) return;

    if (oldKey) {
      this.cells.get(oldKey)?.delete(obj);
    }

    if (!this.cells.has(newKey)) {
      this.cells.set(newKey, new Set());
    }
    this.cells.get(newKey)!.add(obj);
    this.objectCells.set(obj.id, newKey);
  }

  queryArea(x: number, y: number, width: number, height: number): T[] {
    const results: T[] = [];
    const looseSize = this.cellSize * this.looseness;
    const minCx = Math.floor((x - looseSize) / this.cellSize);
    const minCy = Math.floor((y - looseSize) / this.cellSize);
    const maxCx = Math.floor((x + width + looseSize) / this.cellSize);
    const maxCy = Math.floor((y + height + looseSize) / this.cellSize);

    for (let cx = minCx; cx <= maxCx; cx++) {
      for (let cy = minCy; cy <= maxCy; cy++) {
        const cell = this.cells.get(`${cx},${cy}`);
        if (cell) {
          for (const obj of cell) {
            if (obj.x >= x && obj.x <= x + width && obj.y >= y && obj.y <= y + height) {
              results.push(obj);
            }
          }
        }
      }
    }
    return results;
  }

  private getCellKey(x: number, y: number): string {
    return `${Math.floor(x / this.cellSize)},${Math.floor(y / this.cellSize)}`;
  }
}
小结
    

  • 空间分区将 O(n²) 的暴力查询优化为 O(n log n) 或 O(1)
    • 

  • 均匀网格最简单高效，适合密度均匀的场景
    • 

  • 四叉树/八叉树自适应处理不均匀分布的对象
    • 

  • BVH 特别适合射线查询和静态几何体
    • 

  • 动态对象需要松散分区或延迟更新策略来平衡更新成本
