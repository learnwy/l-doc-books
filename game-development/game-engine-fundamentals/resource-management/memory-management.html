内存管理

核心概念

游戏引擎的内存管理直接影响性能和稳定性。与通用应用不同，游戏对内存分配的延迟和碎片化极为敏感——一次垃圾回收暂停就可能导致掉帧。因此，游戏引擎通常实现自定义的内存分配器，绕过标准库的通用分配策略。

线性分配器
线性分配器（Linear Allocator）是最简单高效的分配策略。它维护一个大块内存和一个偏移指针，分配时只需移动指针。所有分配在同一批次结束时统一释放（重置偏移指针为 0）。


class LinearAllocator {
  private buffer: ArrayBuffer;
  private offset = 0;
  private readonly capacity: number;

  constructor(sizeBytes: number) {
    this.buffer = new ArrayBuffer(sizeBytes);
    this.capacity = sizeBytes;
  }

  allocate(size: number, alignment = 8): DataView | null {
    const alignedOffset = this.align(this.offset, alignment);
    if (alignedOffset + size > this.capacity) return null;

    const view = new DataView(this.buffer, alignedOffset, size);
    this.offset = alignedOffset + size;
    return view;
  }

  reset(): void {
    this.offset = 0;
  }

  getUsed(): number {
    return this.offset;
  }
  getCapacity(): number {
    return this.capacity;
  }
  getUtilization(): number {
    return this.offset / this.capacity;
  }

  private align(offset: number, alignment: number): number {
    return (offset + alignment - 1) & ~(alignment - 1);
  }
}

class DoubleBufferedAllocator {
  private buffers: [LinearAllocator, LinearAllocator];
  private currentIndex = 0;

  constructor(sizeBytes: number) {
    this.buffers = [new LinearAllocator(sizeBytes), new LinearAllocator(sizeBytes)];
  }

  allocate(size: number, alignment = 8): DataView | null {
    return this.buffers[this.currentIndex].allocate(size, alignment);
  }

  swap(): void {
    this.currentIndex = 1 - this.currentIndex;
    this.buffers[this.currentIndex].reset();
  }

  getCurrent(): LinearAllocator {
    return this.buffers[this.currentIndex];
  }
}
双缓冲分配器（Double Buffered Allocator）是线性分配器的扩展，使用两个交替的缓冲区，在一帧中一个用于分配、另一个被重置，确保上一帧的临时数据在当前帧仍可访问。


池分配器
池分配器（Pool Allocator）将内存预分割为等大小的块，维护一个空闲链表。每次分配和释放都是 O(1) 操作，且完全没有碎片问题。


class PoolAllocator<T> {
  private pool: T[];
  private freeStack: number[];
  private inUse: boolean[];

  constructor(factory: () => T, size: number) {
    this.pool = Array.from({ length: size }, factory);
    this.freeStack = Array.from({ length: size }, (_, i) => size - 1 - i);
    this.inUse = new Array(size).fill(false);
  }

  allocate(): { index: number; value: T } | null {
    if (this.freeStack.length === 0) return null;
    const index = this.freeStack.pop()!;
    this.inUse[index] = true;
    return { index, value: this.pool[index] };
  }

  free(index: number): void {
    if (!this.inUse[index]) return;
    this.inUse[index] = false;
    this.freeStack.push(index);
  }

  getActiveCount(): number {
    return this.inUse.filter(Boolean).length;
  }

  getFreeCount(): number {
    return this.freeStack.length;
  }
}

class TypedPoolAllocator {
  private buffer: ArrayBuffer;
  private blockSize: number;
  private blockCount: number;
  private freeList: number[];

  constructor(blockSize: number, blockCount: number) {
    this.blockSize = this.alignTo(blockSize, 8);
    this.blockCount = blockCount;
    this.buffer = new ArrayBuffer(this.blockSize * blockCount);
    this.freeList = Array.from({ length: blockCount }, (_, i) => blockCount - 1 - i);
  }

  allocate(): { offset: number; view: DataView } | null {
    if (this.freeList.length === 0) return null;
    const index = this.freeList.pop()!;
    const offset = index * this.blockSize;
    return { offset, view: new DataView(this.buffer, offset, this.blockSize) };
  }

  free(offset: number): void {
    const index = offset / this.blockSize;
    this.freeList.push(index);
  }

  private alignTo(size: number, alignment: number): number {
    return (size + alignment - 1) & ~(alignment - 1);
  }
}
内存碎片与整理
长时间运行后，频繁的分配和释放会导致内存碎片——虽然总空闲内存足够，但没有单个连续块能满足分配请求。碎片整理（Defragmentation）通过移动活跃对象来合并空闲块。


interface MemoryBlock {
  offset: number;
  size: number;
  inUse: boolean;
  handle: number;
}

class DefragmentableAllocator {
  private blocks: MemoryBlock[] = [];
  private handleMap = new Map<number, MemoryBlock>();
  private nextHandle = 0;
  private totalSize: number;

  constructor(totalSize: number) {
    this.totalSize = totalSize;
    this.blocks.push({ offset: 0, size: totalSize, inUse: false, handle: -1 });
  }

  allocate(size: number): number | null {
    for (const block of this.blocks) {
      if (!block.inUse && block.size >= size) {
        const handle = this.nextHandle++;
        if (block.size > size) {
          const remainder: MemoryBlock = {
            offset: block.offset + size,
            size: block.size - size,
            inUse: false,
            handle: -1,
          };
          block.size = size;
          const idx = this.blocks.indexOf(block);
          this.blocks.splice(idx + 1, 0, remainder);
        }
        block.inUse = true;
        block.handle = handle;
        this.handleMap.set(handle, block);
        return handle;
      }
    }
    return null;
  }

  free(handle: number): void {
    const block = this.handleMap.get(handle);
    if (!block) return;
    block.inUse = false;
    block.handle = -1;
    this.handleMap.delete(handle);
    this.mergeAdjacentFree();
  }

  defragment(moveCallback: (handle: number, oldOffset: number, newOffset: number) => void): void {
    let writeOffset = 0;
    for (const block of this.blocks) {
      if (block.inUse) {
        if (block.offset !== writeOffset) {
          moveCallback(block.handle, block.offset, writeOffset);
          block.offset = writeOffset;
        }
        writeOffset += block.size;
      }
    }
    this.blocks = this.blocks.filter((b) => b.inUse);
    if (writeOffset < this.totalSize) {
      this.blocks.push({ offset: writeOffset, size: this.totalSize - writeOffset, inUse: false, handle: -1 });
    }
  }

  getFragmentation(): number {
    const freeBlocks = this.blocks.filter((b) => !b.inUse);
    if (freeBlocks.length <= 1) return 0;
    const totalFree = freeBlocks.reduce((sum, b) => sum + b.size, 0);
    const largestFree = Math.max(...freeBlocks.map((b) => b.size));
    return 1 - largestFree / totalFree;
  }

  private mergeAdjacentFree(): void {
    for (let i = 0; i < this.blocks.length - 1; i++) {
      if (!this.blocks[i].inUse && !this.blocks[i + 1].inUse) {
        this.blocks[i].size += this.blocks[i + 1].size;
        this.blocks.splice(i + 1, 1);
        i--;
      }
    }
  }
}
内存预算与监控
游戏引擎需要在有限的内存预算内运行，尤其是移动平台和主机。内存监控系统追踪各子系统的内存使用，在接近预算上限时触发预警和降级策略。


interface MemoryBudget {
  system: string;
  allocated: number;
  limit: number;
  peak: number;
}

class MemoryTracker {
  private budgets = new Map<string, MemoryBudget>();
  private onWarning: (budget: MemoryBudget) => void;
  private warningThreshold = 0.85;

  constructor(onWarning: (budget: MemoryBudget) => void) {
    this.onWarning = onWarning;
  }

  setBudget(system: string, limitBytes: number): void {
    this.budgets.set(system, { system, allocated: 0, limit: limitBytes, peak: 0 });
  }

  allocate(system: string, bytes: number): boolean {
    const budget = this.budgets.get(system);
    if (!budget) return false;
    if (budget.allocated + bytes > budget.limit) return false;
    budget.allocated += bytes;
    budget.peak = Math.max(budget.peak, budget.allocated);
    if (budget.allocated / budget.limit > this.warningThreshold) {
      this.onWarning(budget);
    }
    return true;
  }

  free(system: string, bytes: number): void {
    const budget = this.budgets.get(system);
    if (budget) {
      budget.allocated = Math.max(0, budget.allocated - bytes);
    }
  }

  getReport(): MemoryBudget[] {
    return [...this.budgets.values()];
  }

  getTotalUsed(): number {
    return [...this.budgets.values()].reduce((sum, b) => sum + b.allocated, 0);
  }
}
