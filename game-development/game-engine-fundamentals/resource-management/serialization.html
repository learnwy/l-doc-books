序列化系统

核心概念

序列化是将游戏对象的运行时状态转换为字节流的过程，反序列化则是逆过程。序列化系统是存档、关卡保存、网络同步和资源管线的基础设施。一个好的序列化系统需要在速度、体积、版本兼容性和可读性之间取得平衡。

模式定义与反射
序列化系统需要知道对象的结构才能正确编码。模式（Schema）定义了每个可序列化类型的字段名、类型和默认值。反射机制允许在运行时查询对象结构。


type FieldType = 'int8' | 'int16' | 'int32' | 'float32' | 'float64' | 'string' | 'boolean' | 'array' | 'object';

interface FieldSchema {
  name: string;
  type: FieldType;
  elementType?: FieldType;
  objectSchema?: string;
  defaultValue?: unknown;
  version: number;
}

interface TypeSchema {
  name: string;
  version: number;
  fields: FieldSchema[];
}

class SchemaRegistry {
  private schemas = new Map<string, TypeSchema>();

  register(schema: TypeSchema): void {
    this.schemas.set(schema.name, schema);
  }

  get(name: string): TypeSchema | undefined {
    return this.schemas.get(name);
  }

  validate(name: string, data: Record<string, unknown>): string[] {
    const schema = this.schemas.get(name);
    if (!schema) return [`Unknown schema: ${name}`];
    const errors: string[] = [];
    for (const field of schema.fields) {
      if (!(field.name in data) && field.defaultValue === undefined) {
        errors.push(`Missing required field: ${field.name}`);
      }
    }
    return errors;
  }
}

const schemaRegistry = new SchemaRegistry();
schemaRegistry.register({
  name: 'Transform',
  version: 2,
  fields: [
    { name: 'posX', type: 'float32', defaultValue: 0, version: 1 },
    { name: 'posY', type: 'float32', defaultValue: 0, version: 1 },
    { name: 'posZ', type: 'float32', defaultValue: 0, version: 1 },

二进制编码器
二进制序列化追求最小的存储体积和最快的编解码速度。通过将数据紧凑排列并使用固定宽度或变长编码，二进制格式比文本格式高效数倍。


class BinaryWriter {
  private buffer: ArrayBuffer;
  private view: DataView;
  private offset = 0;

  constructor(initialSize = 1024) {
    this.buffer = new ArrayBuffer(initialSize);
    this.view = new DataView(this.buffer);
  }

  writeInt8(value: number): void {
    this.ensureCapacity(1);
    this.view.setInt8(this.offset++, value);
  }
  writeInt16(value: number): void {
    this.ensureCapacity(2);
    this.view.setInt16(this.offset, value, true);
    this.offset += 2;
  }
  writeInt32(value: number): void {
    this.ensureCapacity(4);
    this.view.setInt32(this.offset, value, true);
    this.offset += 4;
  }
  writeFloat32(value: number): void {
    this.ensureCapacity(4);
    this.view.setFloat32(this.offset, value, true);
    this.offset += 4;
  }
  writeFloat64(value: number): void {
    this.ensureCapacity(8);
    this.view.setFloat64(this.offset, value, true);
    this.offset += 8;
  }

  writeString(value: string): void {
    const encoded = new TextEncoder().encode(value);
    this.writeInt32(encoded.byteLength);
    this.ensureCapacity(encoded.byteLength);
    new Uint8Array(this.buffer, this.offset, encoded.byteLength).set(encoded);
    this.offset += encoded.byteLength;
  }

  writeBoolean(value: boolean): void {
    this.writeInt8(value ? 1 : 0);
  }

  toBytes(): Uint8Array {
    return new Uint8Array(this.buffer, 0, this.offset);
  }

  private ensureCapacity(needed: number): void {
    while (this.offset + needed > this.buffer.byteLength) {
      const newBuffer = new ArrayBuffer(this.buffer.byteLength * 2);
      new Uint8Array(newBuffer).set(new Uint8Array(this.buffer));
      this.buffer = newBuffer;
      this.view = new DataView(this.buffer);
    }
  }
}

class BinaryReader {
  private view: DataView;
  private offset = 0;

  constructor(data: Uint8Array) {
    this.view = new DataView(data.buffer, data.byteOffset, data.byteLength);
  }

  readInt8(): number {
    return this.view.getInt8(this.offset++);
  }
  readInt16(): number {
    const v = this.view.getInt16(this.offset, true);
    this.offset += 2;
    return v;
  }
  readInt32(): number {
    const v = this.view.getInt32(this.offset, true);
    this.offset += 4;
    return v;
  }
  readFloat32(): number {
    const v = this.view.getFloat32(this.offset, true);
    this.offset += 4;
    return v;
  }
  readFloat64(): number {
    const v = this.view.getFloat64(this.offset, true);
    this.offset += 8;
    return v;
  }

  readString(): string {
    const length = this.readInt32();
    const bytes = new Uint8Array(this.view.buffer, this.view.byteOffset + this.offset, length);
    this.offset += length;
    return new TextDecoder().decode(bytes);
  }

  readBoolean(): boolean {
    return this.readInt8() !== 0;
  }
}
版本迁移与兼容性
游戏在迭代过程中数据结构不断变化，序列化系统必须能够加载旧版本的数据。版本迁移器（Migrator）定义了从旧版本到新版本的数据转换规则。


type MigrationFn = (data: Record<string, unknown>) => Record<string, unknown>;

class VersionMigrator {
  private migrations = new Map<string, Map<number, MigrationFn>>();

  registerMigration(typeName: string, fromVersion: number, migration: MigrationFn): void {
    if (!this.migrations.has(typeName)) {
      this.migrations.set(typeName, new Map());
    }
    this.migrations.get(typeName)!.set(fromVersion, migration);
  }

  migrate(
    typeName: string,
    data: Record<string, unknown>,
    fromVersion: number,
    toVersion: number,
  ): Record<string, unknown> {
    let current = data;
    const typeMigrations = this.migrations.get(typeName);
    if (!typeMigrations) return current;

    for (let v = fromVersion; v < toVersion; v++) {
      const migration = typeMigrations.get(v);
      if (migration) {
        current = migration(current);
      }
    }
    return current;
  }
}

const migrator = new VersionMigrator();
migrator.registerMigration('PlayerData', 1, (data) => ({
  ...data,
  inventory: data.items ?? [],
  items: undefined,
}));
migrator.registerMigration('PlayerData', 2, (data) => ({
  ...data,
  settings: { volume: 1.0, sensitivity: 0.5 },
}));
对象图与循环引用
复杂的游戏状态中对象之间可能存在循环引用（A 引用 B，B 引用 A）。序列化系统需要识别循环引用并使用引用 ID 替代重复序列化。


class ObjectGraphSerializer {
  private objectMap = new Map<object, number>();
  private nextId = 0;

  serialize(root: Record<string, unknown>): string {
    this.objectMap.clear();
    this.nextId = 0;
    return JSON.stringify(this.processValue(root));
  }

  private processValue(value: unknown): unknown {
    if (value === null || value === undefined) return value;
    if (typeof value !== 'object') return value;

    const obj = value as Record<string, unknown>;
    if (this.objectMap.has(obj)) {
      return { __ref: this.objectMap.get(obj) };
    }

    const id = this.nextId++;
    this.objectMap.set(obj, id);

    if (Array.isArray(obj)) {
      return { __id: id, __type: 'array', items: obj.map((item) => this.processValue(item)) };
    }

    const serialized: Record<string, unknown> = { __id: id };
    for (const [key, val] of Object.entries(obj)) {
      serialized[key] = this.processValue(val);
    }
    return serialized;
  }
}
