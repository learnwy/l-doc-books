资源管线

核心概念

资源管线（Asset Pipeline）是将原始美术资源（模型、纹理、音频、动画）转化为引擎可高效加载的运行时格式的完整流程。它涵盖资源导入、转换、压缩、打包和版本管理等环节，是连接内容创作工具和游戏引擎的桥梁。

资源类型与导入器
每种资源类型需要专门的导入器来解析其原始格式。导入器将异构的源文件统一转换为引擎内部的中间表示（Intermediate Representation）。


interface AssetMetadata {
  id: string;
  type: AssetType;
  sourcePath: string;
  lastModified: number;
  dependencies: string[];
  settings: Record<string, unknown>;
}

type AssetType = 'texture' | 'mesh' | 'audio' | 'shader' | 'material' | 'animation' | 'prefab';

interface ImportResult {
  metadata: AssetMetadata;
  data: Uint8Array;
  subAssets: ImportResult[];
}

interface AssetImporter {
  supportedExtensions: string[];
  import(sourcePath: string, data: Uint8Array, settings: Record<string, unknown>): Promise<ImportResult>;
  getDefaultSettings(): Record<string, unknown>;
}

class TextureImporter implements AssetImporter {
  supportedExtensions = ['.png', '.jpg', '.tga', '.psd', '.hdr'];

  async import(sourcePath: string, data: Uint8Array, settings: Record<string, unknown>): Promise<ImportResult> {
    const imageData = await this.decodeImage(data);
    const processed = this.processTexture(imageData, settings);

    return {
      metadata: {
        id: this.generateId(sourcePath),
        type: 'texture',
        sourcePath,
        lastModified: Date.now(),
        dependencies: [],
        settings,
      },
      data: processed,
      subAssets: [],
    };
  }

  getDefaultSettings(): Record<string, unknown> {
    return {
      maxSize: 2048,
      format: 'rgba8',
      generateMips: true,
      sRGB: true,
      filterMode: 'bilinear',
      wrapMode: 'repeat',
    };
  }

  private async decodeImage(_data: Uint8Array): Promise<ImageData> {
    return { width: 0, height: 0, data: new Uint8ClampedArray(0), colorSpace: 'srgb' };
  }
  private processTexture(_img: ImageData, _settings: Record<string, unknown>): Uint8Array {
    return new Uint8Array(0);
  }
  private generateId(path: string): string {
    return path.replace(/[^a-z0-9]/gi, '_');
  }
}
增量构建与依赖追踪
大型项目可能包含数万个资源文件，全量构建耗时过长。增量构建系统追踪资源之间的依赖关系，只重新处理发生变化的资源及其依赖方。


interface DependencyNode {
  assetId: string;
  hash: string;
  dependencies: string[];
  dependents: string[];
  lastBuildHash: string;
}

class IncrementalBuilder {
  private graph = new Map<string, DependencyNode>();
  private buildQueue: string[] = [];

  registerAsset(id: string, hash: string, dependencies: string[]): void {
    const node: DependencyNode = {
      assetId: id,
      hash,
      dependencies,
      dependents: [],
      lastBuildHash: '',
    };
    this.graph.set(id, node);
    for (const dep of dependencies) {
      const depNode = this.graph.get(dep);
      if (depNode) {
        depNode.dependents.push(id);
      }
    }
  }

  detectChanges(currentHashes: Map<string, string>): string[] {
    const dirty = new Set<string>();

    for (const [id, hash] of currentHashes) {
      const node = this.graph.get(id);
      if (!node || node.hash !== hash) {
        this.markDirty(id, dirty);
      }
    }

    return this.topologicalSort([...dirty]);
  }

  private markDirty(id: string, dirty: Set<string>): void {
    if (dirty.has(id)) return;
    dirty.add(id);
    const node = this.graph.get(id);
    if (node) {
      for (const dependent of node.dependents) {
        this.markDirty(dependent, dirty);
      }
    }
  }

  private topologicalSort(ids: string[]): string[] {
    const visited = new Set<string>();
    const result: string[] = [];
    const visit = (id: string): void => {
      if (visited.has(id)) return;
      visited.add(id);
      const node = this.graph.get(id);
      if (node) {
        for (const dep of node.dependencies) {
          if (ids.includes(dep)) visit(dep);
        }
      }
      result.push(id);
    };
    for (const id of ids) visit(id);
    return result;
  }
}
资源打包与压缩
资源打包将多个相关资源合并到一个文件中，减少 I/O 操作次数。结合压缩算法可以显著减小安装包体积和加载时间。


interface BundleEntry {
  id: string;
  offset: number;
  size: number;
  compressedSize: number;
  compressionType: 'none' | 'lz4' | 'zstd';
}

interface BundleHeader {
  magic: number;
  version: number;
  entryCount: number;
  entries: BundleEntry[];
}

class AssetBundleWriter {
  private entries: Array<{ id: string; data: Uint8Array; compression: 'none' | 'lz4' | 'zstd' }> = [];

  addAsset(id: string, data: Uint8Array, compression: 'none' | 'lz4' | 'zstd' = 'lz4'): void {
    this.entries.push({ id, data, compression });
  }

  async build(): Promise<Uint8Array> {
    const bundleEntries: BundleEntry[] = [];
    const chunks: Uint8Array[] = [];
    let offset = 0;

    for (const entry of this.entries) {
      const compressed = await this.compress(entry.data, entry.compression);
      bundleEntries.push({
        id: entry.id,
        offset,
        size: entry.data.byteLength,
        compressedSize: compressed.byteLength,
        compressionType: entry.compression,
      });
      chunks.push(compressed);
      offset += compressed.byteLength;
    }

    const header = this.serializeHeader({
      magic: 0x42554e44,
      version: 1,
      entryCount: bundleEntries.length,
      entries: bundleEntries,
    });
    const totalSize = header.byteLength + chunks.reduce((sum, c) => sum + c.byteLength, 0);
    const result = new Uint8Array(totalSize);
    result.set(header, 0);
    let pos = header.byteLength;
    for (const chunk of chunks) {
      result.set(chunk, pos);
      pos += chunk.byteLength;
    }
    return result;
  }

  private async compress(data: Uint8Array, _type: string): Promise<Uint8Array> {
    return data;
  }
  private serializeHeader(_header: BundleHeader): Uint8Array {
    return new Uint8Array(0);
  }
}
热重载与版本管理
开发阶段的资源热重载允许在不重启游戏的情况下更新资源。引擎通过文件监视器检测源文件变化，自动触发重新导入和更新内存中的资源引用。


class AssetWatcher {
  private watchers = new Map<string, number>();
  private lastModified = new Map<string, number>();
  private onChanged: (path: string) => void;

  constructor(onChanged: (path: string) => void) {
    this.onChanged = onChanged;
  }

  watch(directory: string): void {
    const intervalId = window.setInterval(() => {
      this.pollDirectory(directory);
    }, 500);
    this.watchers.set(directory, intervalId);
  }

  unwatch(directory: string): void {
    const intervalId = this.watchers.get(directory);
    if (intervalId) {
      clearInterval(intervalId);
      this.watchers.delete(directory);
    }
  }

  private async pollDirectory(_directory: string): Promise<void> {
    const files = await this.listFiles(_directory);
    for (const file of files) {
      const modified = await this.getLastModified(file);
      const prev = this.lastModified.get(file);
      if (prev !== undefined && prev !== modified) {
        this.onChanged(file);
      }
      this.lastModified.set(file, modified);
    }
  }

  private async listFiles(_dir: string): Promise<string[]> {
    return [];
  }
  private async getLastModified(_path: string): Promise<number> {
    return 0;
  }
}
