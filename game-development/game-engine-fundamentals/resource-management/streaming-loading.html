流式加载

核心概念

流式加载（Streaming Loading）是在游戏运行过程中异步加载和卸载资源的技术。与传统的"全部预加载"不同，流式加载根据玩家位置和视角动态决定哪些资源需要加载，哪些可以卸载，使得游戏能够呈现远超内存容量的世界。

优先级调度系统
流式加载需要一个优先级系统来决定加载顺序。影响优先级的因素包括距离、可见性、使用频率和资源大小。


interface LoadRequest {
  assetId: string;
  priority: number;
  distance: number;
  visible: boolean;
  estimatedSize: number;
  callback: (data: Uint8Array) => void;
  timestamp: number;
}

class StreamingScheduler {
  private queue: LoadRequest[] = [];
  private activeLoads = 0;
  private readonly maxConcurrent: number;
  private readonly bandwidthBudget: number;
  private currentBandwidth = 0;

  constructor(maxConcurrent = 4, bandwidthBudgetMB = 50) {
    this.maxConcurrent = maxConcurrent;
    this.bandwidthBudget = bandwidthBudgetMB * 1024 * 1024;
  }

  enqueue(request: LoadRequest): void {
    const index = this.queue.findIndex((r) => r.priority < request.priority);
    if (index === -1) {
      this.queue.push(request);
    } else {
      this.queue.splice(index, 0, request);
    }
  }

  cancel(assetId: string): void {
    this.queue = this.queue.filter((r) => r.assetId !== assetId);
  }

  update(): void {
    while (this.activeLoads < this.maxConcurrent && this.queue.length > 0) {
      const request = this.queue.shift()!;
      if (this.currentBandwidth + request.estimatedSize > this.bandwidthBudget) {
        this.queue.unshift(request);
        break;
      }
      this.startLoad(request);
    }
  }

  private async startLoad(request: LoadRequest): Promise<void> {
    this.activeLoads++;
    this.currentBandwidth += request.estimatedSize;
    try {
      const data = await this.loadAsset(request.assetId);
      request.callback(data);
    } finally {
      this.activeLoads--;
      this.currentBandwidth -= request.estimatedSize;
    }
  }

  private async loadAsset(_id: string): Promise<Uint8Array> {
    return new Uint8Array(0);
  }
}

class PriorityCalculator {
  static calculate(distance: number, visible: boolean, screenSize: number, lastUsedTime: number): number {
    let priority = 0;
    priority += visible ? 100 : 0;
    priority += Math.max(0, 50 - distance * 0.5);
    priority += screenSize * 20;
    const timeSinceUse = (Date.now() - lastUsedTime) / 1000;
    priority -= Math.min(30, timeSinceUse * 0.1);
    return priority;
  }
}
世界分块与区域管理
开放世界游戏将世界划分为规则的块（Chunk），每个块包含该区域的所有资源。流式系统根据玩家位置维护一个活跃块的网格，动态加载进入网格的块并卸载离开的块。


interface WorldChunk {
  x: number;
  z: number;
  state: 'unloaded' | 'loading' | 'loaded' | 'active' | 'unloading';
  lod: number;
  assets: string[];
  lastAccessTime: number;
}

class ChunkStreamingManager {
  private chunks = new Map<string, WorldChunk>();
  private chunkSize: number;
  private loadRadius: number;
  private activeRadius: number;
  private unloadRadius: number;

  constructor(chunkSize: number, loadRadius: number, activeRadius: number, unloadRadius: number) {
    this.chunkSize = chunkSize;
    this.loadRadius = loadRadius;
    this.activeRadius = activeRadius;
    this.unloadRadius = unloadRadius;
  }

  update(playerX: number, playerZ: number): { toLoad: WorldChunk[]; toUnload: WorldChunk[] } {
    const centerX = Math.floor(playerX / this.chunkSize);
    const centerZ = Math.floor(playerZ / this.chunkSize);
    const toLoad: WorldChunk[] = [];
    const toUnload: WorldChunk[] = [];

    for (let dx = -this.loadRadius; dx <= this.loadRadius; dx++) {
      for (let dz = -this.loadRadius; dz <= this.loadRadius; dz++) {
        const cx = centerX + dx;
        const cz = centerZ + dz;
        const key = `${cx},${cz}`;
        const distance = Math.sqrt(dx * dx + dz * dz);

        if (distance <= this.loadRadius) {
          if (!this.chunks.has(key)) {
            const chunk: WorldChunk = {
              x: cx,
              z: cz,
              state: 'unloaded',
              lod: this.calculateLOD(distance),
              assets: [],
              lastAccessTime: Date.now(),
            };
            this.chunks.set(key, chunk);
            toLoad.push(chunk);
          }
        }
      }
    }

    for (const [key, chunk] of this.chunks) {
      const dx = chunk.x - centerX;
      const dz = chunk.z - centerZ;
      const distance = Math.sqrt(dx * dx + dz * dz);
      if (distance > this.unloadRadius && chunk.state !== 'unloading') {
        toUnload.push(chunk);
        this.chunks.delete(key);
      } else {
        chunk.lod = this.calculateLOD(distance);
        chunk.state = distance <= this.activeRadius ? 'active' : 'loaded';
      }
    }

    return { toLoad, toUnload };
  }

  private calculateLOD(distance: number): number {
    if (distance <= 2) return 0;
    if (distance <= 5) return 1;
    if (distance <= 10) return 2;
    return 3;
  }
}
LOD 与 MIP 流式
纹理和网格的 LOD（Level of Detail）流式按需加载不同精度的资源版本。近处对象使用高精度模型和高分辨率纹理，远处使用低精度版本，中间区域平滑过渡。


interface LODLevel {
  distance: number;
  meshId: string;
  vertexCount: number;
  textureResolution: number;
}

class LODManager {
  private lodConfigs = new Map<string, LODLevel[]>();
  private currentLODs = new Map<string, number>();

  registerLOD(entityId: string, levels: LODLevel[]): void {
    const sorted = [...levels].sort((a, b) => a.distance - b.distance);
    this.lodConfigs.set(entityId, sorted);
    this.currentLODs.set(entityId, 0);
  }

  update(entityId: string, distance: number): { changed: boolean; level: number; mesh: string } {
    const levels = this.lodConfigs.get(entityId);
    if (!levels) return { changed: false, level: 0, mesh: '' };

    let targetLevel = levels.length - 1;
    for (let i = 0; i < levels.length; i++) {
      if (distance <= levels[i].distance) {
        targetLevel = i;
        break;
      }
    }

    const currentLevel = this.currentLODs.get(entityId) ?? 0;
    const changed = targetLevel !== currentLevel;
    if (changed) {
      this.currentLODs.set(entityId, targetLevel);
    }

    return { changed, level: targetLevel, mesh: levels[targetLevel].meshId };
  }

  getTotalVertexCount(): number {
    let total = 0;
    for (const [entityId, level] of this.currentLODs) {
      const levels = this.lodConfigs.get(entityId);
      if (levels && levels[level]) {
        total += levels[level].vertexCount;
      }
    }
    return total;
  }
}

class TextureStreamingManager {
  private textures = new Map<string, { maxMip: number; loadedMip: number; priority: number }>();

  registerTexture(id: string, maxMip: number): void {
    this.textures.set(id, { maxMip, loadedMip: maxMip, priority: 0 });
  }

  updatePriority(id: string, screenSize: number, distance: number): void {
    const tex = this.textures.get(id);
    if (!tex) return;
    tex.priority = screenSize / Math.max(1, distance);
    const desiredMip = Math.max(0, Math.floor(Math.log2(1 / tex.priority)));
    if (desiredMip < tex.loadedMip) {
      this.requestMipLoad(id, desiredMip);
    }
  }

  private requestMipLoad(_id: string, _mipLevel: number): void {}
}
加载屏幕与过渡策略
当流式加载无法完全避免等待时，需要加载屏幕或过渡策略来掩盖延迟。常见策略包括异步加载屏幕、走廊过渡和渐进式呈现。


type TransitionState = 'idle' | 'fadeOut' | 'loading' | 'fadeIn';

class LoadingTransition {
  private state: TransitionState = 'idle';
  private progress = 0;
  private fadeAlpha = 0;
  private fadeDuration = 0.5;
  private fadeTimer = 0;

  startTransition(loadFn: () => Promise<void>): void {
    this.state = 'fadeOut';
    this.fadeTimer = 0;
    this.executeTransition(loadFn);
  }

  private async executeTransition(loadFn: () => Promise<void>): Promise<void> {
    await this.waitForFade('fadeOut');
    this.state = 'loading';
    await loadFn();
    this.state = 'fadeIn';
    this.fadeTimer = 0;
    await this.waitForFade('fadeIn');
    this.state = 'idle';
  }

  update(dt: number): void {
    if (this.state === 'fadeOut') {
      this.fadeTimer += dt;
      this.fadeAlpha = Math.min(1, this.fadeTimer / this.fadeDuration);
    } else if (this.state === 'fadeIn') {
      this.fadeTimer += dt;
      this.fadeAlpha = Math.max(0, 1 - this.fadeTimer / this.fadeDuration);
    }
  }

  setProgress(value: number): void {
    this.progress = Math.min(1, Math.max(0, value));
  }
  getProgress(): number {
    return this.progress;
  }
  getFadeAlpha(): number {
    return this.fadeAlpha;
  }
  getState(): TransitionState {
    return this.state;
  }

  private waitForFade(_phase: string): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, this.fadeDuration * 1000));
  }
}
