延迟补偿技术

核心概念

延迟补偿（Lag Compensation）是服务器端的一种技术，用于解决网络延迟导致的射击判定不公平问题。在权威服务器架构中，当玩家在客户端开枪时，他看到的是过去某个时刻的游戏状态（受快照延迟和网络延迟影响）。如果服务器直接使用当前时刻的玩家位置来判定射击命中，射击者几乎永远无法命中正在移动的目标。

延迟补偿通过让服务器在处理射击判定时"回到过去"来解决这个问题。服务器维护一段时间窗口内的世界状态历史，收到玩家的射击输入后，根据该玩家的网络延迟回退到对应时刻的世界状态进行命中判定。

世界状态历史缓冲
延迟补偿的基础是服务器维护的世界状态历史缓冲区。每个 Tick 结束后，服务器保存一份当前所有可命中实体（Hitbox）的位置快照，形成一个时间轴上的状态序列。


interface HitboxSnapshot {
  entityId: string;
  position: Vector3;
  bounds: BoundingBox;
  isAlive: boolean;
  tick: number;
  timestamp: number;
}

interface Vector3 {
  x: number;
  y: number;
  z: number;
}

interface BoundingBox {
  minX: number;
  minY: number;
  minZ: number;
  maxX: number;
  maxY: number;
  maxZ: number;
}

interface WorldHistoryFrame {
  tick: number;
  timestamp: number;
  hitboxes: Map<string, HitboxSnapshot>;
}

class WorldStateHistory {
  private frames: WorldHistoryFrame[] = [];
  private maxFrames: number;

  constructor(maxFrames: number = 64) {
    this.maxFrames = maxFrames;
  }

  saveFrame(tick: number, timestamp: number, entities: Map<string, HitboxSnapshot>): void {
    this.frames.push({
      tick,
      timestamp,
      hitboxes: new Map(entities),
    });

    while (this.frames.length > this.maxFrames) {
      this.frames.shift();
    }
  }

  getFrameAtTime(targetTime: number): WorldHistoryFrame | null {
    if (this.frames.length === 0) return null;

    let closest: WorldHistoryFrame | null = null;
    let minDiff = Infinity;

    for (const frame of this.frames) {
      const diff = Math.abs(frame.timestamp - targetTime);
      if (diff < minDiff) {
        minDiff = diff;
        closest = frame;
      }
    }

    return closest;
  }

  getInterpolatedFrame(targetTime: number): WorldHistoryFrame | null {
    let before: WorldHistoryFrame | null = null;
    let after: WorldHistoryFrame | null = null;

    for (let i = 0; i < this.frames.length - 1; i++) {
      if (this.frames[i].timestamp <= targetTime && this.frames[i + 1].timestamp >= targetTime) {
        before = this.frames[i];
        after = this.frames[i + 1];
        break;
      }
    }

    if (!before || !after) return this.getFrameAtTime(targetTime);

    const alpha = (targetTime - before.timestamp) / (after.timestamp - before.timestamp);
    const interpolated: WorldHistoryFrame = {
      tick: before.tick,
      timestamp: targetTime,
      hitboxes: new Map(),
    };

    for (const [id, beforeHitbox] of before.hitboxes) {
      const afterHitbox = after.hitboxes.get(id);
      if (!afterHitbox) continue;

      interpolated.hitboxes.set(id, {
        entityId: id,
        position: lerpVector3(beforeHitbox.position, afterHitbox.position, alpha),
        bounds: lerpBounds(beforeHitbox.bounds, afterHitbox.bounds, alpha),
        isAlive: beforeHitbox.isAlive && afterHitbox.isAlive,
        tick: before.tick,
        timestamp: targetTime,
      });
    }

    return interpolated;
  }
}

const lerpVector3 = (a: Vector3, b: Vector3, t: number): Vector3 => ({
  x: a.x + (b.x - a.x) * t,
  y: a.y + (b.y - a.y) * t,
  z: a.z + (b.z - a.z) * t,
});

const lerpBounds = (a: BoundingBox, b: BoundingBox, t: number): BoundingBox => ({
  minX: a.minX + (b.minX - a.minX) * t,
  minY: a.minY + (b.minY - a.minY) * t,
  minZ: a.minZ + (b.minZ - a.minZ) * t,
  maxX: a.maxX + (b.maxX - a.maxX) * t,
  maxY: a.maxY + (b.maxY - a.maxY) * t,
  maxZ: a.maxZ + (b.maxZ - a.maxZ) * t,
});
历史缓冲区的深度决定了可补偿的最大延迟。以 64 Tick/s 计算，64 帧的缓冲区可以覆盖 1 秒的延迟。超出这个范围的射击将不进行延迟补偿，使用当前帧的位置判定。


射线检测与命中判定
当服务器需要处理一次射击时，先根据射击者的 RTT 计算目标时间点，回退到该时刻的世界状态，然后执行射线检测（Raycast）来判定是否命中。


interface ShotRequest {
  shooterId: string;
  origin: Vector3;
  direction: Vector3;
  weaponId: string;
  clientTimestamp: number;
  sequence: number;
}

interface HitResult {
  hit: boolean;
  targetId?: string;
  hitPoint?: Vector3;
  damage?: number;
  headshot?: boolean;
}

class LagCompensation {
  private worldHistory: WorldStateHistory;
  private maxCompensationMs: number;
  private clientRtts: Map<string, number>;

  constructor(worldHistory: WorldStateHistory, maxCompensation: number = 250) {
    this.worldHistory = worldHistory;
    this.maxCompensationMs = maxCompensation;
    this.clientRtts = new Map();
  }

  updateClientRtt(clientId: string, rtt: number): void {
    this.clientRtts.set(clientId, rtt);
  }

  processShot(shot: ShotRequest, currentTime: number): HitResult {
    const rtt = this.clientRtts.get(shot.shooterId) ?? 0;
    const compensationTime = Math.min(rtt / 2, this.maxCompensationMs);
    const targetTime = currentTime - compensationTime;

    const historicalFrame = this.worldHistory.getInterpolatedFrame(targetTime);
    if (!historicalFrame) {
      return { hit: false };
    }

    return this.performRaycast(shot.origin, shot.direction, historicalFrame, shot.shooterId);
  }

  private performRaycast(origin: Vector3, direction: Vector3, frame: WorldHistoryFrame, shooterId: string): HitResult {
    const maxDistance = 200;
    let closestHit: HitResult = { hit: false };
    let closestDist = maxDistance;

    for (const [entityId, hitbox] of frame.hitboxes) {
      if (entityId === shooterId || !hitbox.isAlive) continue;

      const dist = this.raycastAABB(origin, direction, hitbox.bounds);
      if (dist !== null && dist < closestDist) {
        closestDist = dist;
        closestHit = {
          hit: true,
          targetId: entityId,
          hitPoint: {
            x: origin.x + direction.x * dist,
            y: origin.y + direction.y * dist,
            z: origin.z + direction.z * dist,
          },
          damage: 25,
          headshot: false,
        };
      }
    }

    return closestHit;
  }

  private raycastAABB(origin: Vector3, dir: Vector3, box: BoundingBox): number | null {
    const invDirX = dir.x !== 0 ? 1 / dir.x : Infinity;
    const invDirY = dir.y !== 0 ? 1 / dir.y : Infinity;
    const invDirZ = dir.z !== 0 ? 1 / dir.z : Infinity;

    const t1 = (box.minX - origin.x) * invDirX;
    const t2 = (box.maxX - origin.x) * invDirX;
    const t3 = (box.minY - origin.y) * invDirY;
    const t4 = (box.maxY - origin.y) * invDirY;
    const t5 = (box.minZ - origin.z) * invDirZ;
    const t6 = (box.maxZ - origin.z) * invDirZ;

    const tMin = Math.max(Math.min(t1, t2), Math.min(t3, t4), Math.min(t5, t6));
    const tMax = Math.min(Math.max(t1, t2), Math.max(t3, t4), Math.max(t5, t6));

    if (tMax < 0 || tMin > tMax) return null;
    return tMin >= 0 ? tMin : tMax;
  }
}
补偿上限与公平性调节
无限制的延迟补偿会导致"掩体后被击杀"的不良体验。当一个高延迟玩家（如 300ms）射击时，服务器需要回退到 150ms 前的世界状态。在这 150ms 内，被射击的玩家可能已经移动到掩体后面，但在射击者的视角中对方仍然暴露。


interface CompensationPolicy {
  maxCompensationMs: number;
  compensationCurve: 'linear' | 'capped' | 'degraded';
  degradationStart: number;
  degradationEnd: number;
}

const calculateEffectiveCompensation = (rtt: number, policy: CompensationPolicy): number => {
  const halfRtt = rtt / 2;

  switch (policy.compensationCurve) {
    case 'linear':
      return Math.min(halfRtt, policy.maxCompensationMs);

    case 'capped':
      return Math.min(halfRtt, policy.maxCompensationMs);

    case 'degraded': {
      if (halfRtt <= policy.degradationStart) return halfRtt;
      if (halfRtt >= policy.degradationEnd) return policy.maxCompensationMs;

      const range = policy.degradationEnd - policy.degradationStart;
      const progress = (halfRtt - policy.degradationStart) / range;
      const maxRange = policy.maxCompensationMs - policy.degradationStart;
      return policy.degradationStart + maxRange * progress;
    }
  }
};

大多数竞技游戏使用渐进衰减策略：低延迟玩家获得完全补偿，高延迟玩家的补偿量逐渐减少。这平衡了射击者的命中体验和被射击者的躲避有效性。


服务端验证与防作弊
延迟补偿系统容易被利用来进行作弊。作弊客户端可能伪造较高的延迟值来获取更大的补偿窗口，或者发送不一致的时间戳。


interface ShotValidation {
  isValid: boolean;
  reason?: string;
}

class ShotValidator {
  private shotHistory: Map<string, number[]>;
  private maxShotsPerSecond: number;
  private rttTolerance: number;

  constructor(maxRps: number = 30, rttTolerance: number = 50) {
    this.shotHistory = new Map();
    this.maxShotsPerSecond = maxRps;
    this.rttTolerance = rttTolerance;
  }

  validate(shot: ShotRequest, serverTime: number, knownRtt: number): ShotValidation {
    const timeDiff = serverTime - shot.clientTimestamp;
    const expectedDelay = knownRtt / 2;
    if (Math.abs(timeDiff - expectedDelay) > this.rttTolerance) {
      return { isValid: false, reason: 'timestamp_mismatch' };
    }

    const history = this.shotHistory.get(shot.shooterId) ?? [];
    const recentShots = history.filter((t) => serverTime - t < 1000);
    if (recentShots.length >= this.maxShotsPerSecond) {
      return { isValid: false, reason: 'rate_exceeded' };
    }

    recentShots.push(serverTime);
    this.shotHistory.set(shot.shooterId, recentShots);

    return { isValid: true };
  }
}
服务器应使用自己测量的 RTT（而非客户端声明的延迟）来计算补偿量。同时对射击频率、射线角度变化速率和命中率进行统计分析，检测异常行为。


小结
    

  • 延迟补偿通过让服务器回退到射击者视角对应的时间点来判定命中，实现"所见即所得"
    • 

  • 世界状态历史缓冲区保存每 Tick 的 Hitbox 快照，支持精确的时间回退和帧间插值
    • 

  • 补偿上限和渐进衰减策略平衡了射击者命中体验和被射击者躲避有效性
    • 

  • 服务器应使用自身测量的 RTT 计算补偿量，防止客户端伪造延迟
    • 

  • 射击频率和时间戳一致性检查是防止作弊利用补偿系统的必要手段
