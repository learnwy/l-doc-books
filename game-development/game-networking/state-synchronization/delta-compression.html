增量压缩与带宽优化

核心概念

增量压缩（Delta Compression）是游戏网络中最重要的带宽优化技术之一。其核心思想是：不发送实体的完整状态，而只发送相对于某个基准状态的差异（Delta）。由于游戏世界中大多数实体在相邻帧之间只有微小变化，增量数据通常比全量数据小一个数量级。

增量压缩与位打包、量化等技术结合使用，可以将每个快照的数据量从数 KB 降低到几百字节，这对于支持 64 人甚至更大规模的在线对战至关重要。

基准快照与增量计算
增量压缩系统需要维护客户端已确认接收的基准快照。服务器以该基准为参照，计算当前状态与基准之间的差异。客户端收到增量数据后，将其应用到本地基准快照上还原出完整状态。


interface SnapshotState {
  tick: number;
  entities: Map<string, EntityState>;
}

interface EntityState {
  id: string;
  type: string;
  posX: number;
  posY: number;
  posZ: number;
  rotY: number;
  health: number;
  flags: number;
}

interface DeltaSnapshot {
  baseTick: number;
  currentTick: number;
  created: string[];
  destroyed: string[];
  modified: EntityDelta[];
}

interface EntityDelta {
  id: string;
  changedFields: number;
  posX?: number;
  posY?: number;
  posZ?: number;
  rotY?: number;
  health?: number;
  flags?: number;
}

const FIELD_POS_X = 1 << 0;
const FIELD_POS_Y = 1 << 1;
const FIELD_POS_Z = 1 << 2;
const FIELD_ROT_Y = 1 << 3;
const FIELD_HEALTH = 1 << 4;
const FIELD_FLAGS = 1 << 5;

const computeDelta = (base: SnapshotState, current: SnapshotState): DeltaSnapshot => {
  const delta: DeltaSnapshot = {
    baseTick: base.tick,
    currentTick: current.tick,
    created: [],
    destroyed: [],
    modified: [],
  };

  for (const [id] of current.entities) {
    if (!base.entities.has(id)) {
      delta.created.push(id);
    }
  }

  for (const [id] of base.entities) {
    if (!current.entities.has(id)) {
      delta.destroyed.push(id);
    }
  }

  for (const [id, curr] of current.entities) {
    const prev = base.entities.get(id);
    if (!prev) continue;

    let changedFields = 0;
    const entityDelta: EntityDelta = { id, changedFields: 0 };

    if (curr.posX !== prev.posX) {
      changedFields |= FIELD_POS_X;
      entityDelta.posX = curr.posX;
    }
    if (curr.posY !== prev.posY) {
      changedFields |= FIELD_POS_Y;
      entityDelta.posY = curr.posY;
    }
    if (curr.posZ !== prev.posZ) {
      changedFields |= FIELD_POS_Z;
      entityDelta.posZ = curr.posZ;
    }
    if (curr.rotY !== prev.rotY) {
      changedFields |= FIELD_ROT_Y;
      entityDelta.rotY = curr.rotY;
    }
    if (curr.health !== prev.health) {
      changedFields |= FIELD_HEALTH;
      entityDelta.health = curr.health;
    }
    if (curr.flags !== prev.flags) {
      changedFields |= FIELD_FLAGS;
      entityDelta.flags = curr.flags;
    }

    if (changedFields !== 0) {
      entityDelta.changedFields = changedFields;
      delta.modified.push(entityDelta);
    }
  }

  return delta;
};

const applyDelta = (
  base: SnapshotState,
  delta: DeltaSnapshot,
  fullEntities: Map<string, EntityState>,
): SnapshotState => {
  const result: SnapshotState = {
    tick: delta.currentTick,
    entities: new Map(base.entities),
  };

  for (const id of delta.destroyed) {
    result.entities.delete(id);
  }

  for (const id of delta.created) {
    const entity = fullEntities.get(id);
    if (entity) result.entities.set(id, { ...entity });
  }

  for (const mod of delta.modified) {
    const entity = result.entities.get(mod.id);
    if (!entity) continue;

    if (mod.changedFields & FIELD_POS_X && mod.posX !== undefined) entity.posX = mod.posX;
    if (mod.changedFields & FIELD_POS_Y && mod.posY !== undefined) entity.posY = mod.posY;
    if (mod.changedFields & FIELD_POS_Z && mod.posZ !== undefined) entity.posZ = mod.posZ;
    if (mod.changedFields & FIELD_ROT_Y && mod.rotY !== undefined) entity.rotY = mod.rotY;
    if (mod.changedFields & FIELD_HEALTH && mod.health !== undefined) entity.health = mod.health;
    if (mod.changedFields & FIELD_FLAGS && mod.flags !== undefined) entity.flags = mod.flags;
  }

  return result;
};
基准快照的选择通过确认机制（Ack）实现。客户端收到快照后向服务器回报最新确认的 Tick 号，服务器将该 Tick 对应的快照作为下一次增量计算的基准。如果客户端长时间未确认（如丢包），服务器需要回退到更早的基准或发送全量快照。


量化与位打包
量化（Quantization）将连续的浮点值映射为离散的整数值，位打包（Bit Packing）则将这些整数以精确的位数写入比特流，最大限度地压缩每个字段占用的空间。


class BitWriter {
  private buffer: Uint8Array;
  private bitPosition: number = 0;

  constructor(maxBytes: number) {
    this.buffer = new Uint8Array(maxBytes);
  }

  writeBits(value: number, bits: number): void {
    for (let i = 0; i < bits; i++) {
      const bit = (value >> i) & 1;
      const byteIndex = Math.floor(this.bitPosition / 8);
      const bitIndex = this.bitPosition % 8;
      this.buffer[byteIndex] |= bit << bitIndex;
      this.bitPosition++;
    }
  }

  writeBool(value: boolean): void {
    this.writeBits(value ? 1 : 0, 1);
  }

  writeQuantized(value: number, min: number, max: number, bits: number): void {
    const maxInt = (1 << bits) - 1;
    const normalized = Math.max(0, Math.min(1, (value - min) / (max - min)));
    this.writeBits(Math.round(normalized * maxInt), bits);
  }

  getBytes(): Uint8Array {
    const byteCount = Math.ceil(this.bitPosition / 8);
    return this.buffer.slice(0, byteCount);
  }

  getBitCount(): number {
    return this.bitPosition;
  }
}

class BitReader {
  private buffer: Uint8Array;
  private bitPosition: number = 0;

  constructor(buffer: Uint8Array) {
    this.buffer = buffer;
  }

  readBits(bits: number): number {
    let value = 0;
    for (let i = 0; i < bits; i++) {
      const byteIndex = Math.floor(this.bitPosition / 8);
      const bitIndex = this.bitPosition % 8;
      const bit = (this.buffer[byteIndex] >> bitIndex) & 1;
      value |= bit << i;
      this.bitPosition++;
    }
    return value;
  }

  readBool(): boolean {
    return this.readBits(1) === 1;
  }

  readQuantized(min: number, max: number, bits: number): number {
    const maxInt = (1 << bits) - 1;
    const intValue = this.readBits(bits);
    return min + (intValue / maxInt) * (max - min);
  }
}

变长编码与通用压缩
变长编码（Variable-Length Encoding）对小数值使用更少的字节。游戏网络中的很多数据具有偏斜分布——大多数增量值都很小，偶尔出现大值。变长编码完美匹配这种分布。


const encodeVarint = (value: number, buffer: Uint8Array, offset: number): number => {
  let v = value;
  let pos = offset;
  while (v >= 0x80) {
    buffer[pos++] = (v & 0x7f) | 0x80;
    v >>>= 7;
  }
  buffer[pos++] = v;
  return pos - offset;
};

const decodeVarint = (buffer: Uint8Array, offset: number): { value: number; bytesRead: number } => {
  let value = 0;
  let shift = 0;
  let pos = offset;
  let byte: number;
  do {
    byte = buffer[pos++];
    value |= (byte & 0x7f) << shift;
    shift += 7;
  } while (byte & 0x80);
  return { value, bytesRead: pos - offset };
};

const zigzagEncode = (value: number): number => (value << 1) ^ (value >> 31);

const zigzagDecode = (value: number): number => (value >>> 1) ^ -(value & 1);
ZigZag 编码将有符号整数映射为无符号整数，使得绝对值小的数值编码后同样小。例如 0→0，-1→1，1→2，-2→3。这在增量数据中特别有用，因为位置的增量值通常是接近零的小数。


在所有针对性优化之后，还可以对最终的数据包施加通用压缩算法（如 LZ4）进一步减小体积。LZ4 是一种以解压速度优先的压缩算法，其解压速度可达每秒数 GB，非常适合实时游戏场景。


带宽监控与自适应策略
带宽管理系统需要持续监控实际的网络使用情况，并根据反馈动态调整压缩策略和发送频率。


interface BandwidthStats {
  bytesSent: number;
  bytesReceived: number;
  packetsSent: number;
  packetsReceived: number;
  windowStart: number;
}

class BandwidthMonitor {
  private stats: BandwidthStats;
  private history: { timestamp: number; bytesSent: number }[] = [];
  private budgetBytesPerSecond: number;

  constructor(budgetKbps: number) {
    this.budgetBytesPerSecond = (budgetKbps * 1000) / 8;
    this.stats = {
      bytesSent: 0,
      bytesReceived: 0,
      packetsSent: 0,
      packetsReceived: 0,
      windowStart: Date.now(),
    };
  }

  recordSent(bytes: number): void {
    this.stats.bytesSent += bytes;
    this.stats.packetsSent++;
    this.history.push({ timestamp: Date.now(), bytesSent: bytes });
  }

  getCurrentUsage(): number {
    const now = Date.now();
    const windowMs = 1000;
    const recent = this.history.filter((h) => now - h.timestamp < windowMs);
    return recent.reduce((sum, h) => sum + h.bytesSent, 0);
  }

  getBudgetRemaining(): number {
    return Math.max(0, this.budgetBytesPerSecond - this.getCurrentUsage());
  }

  getUtilizationPercent(): number {
    return (this.getCurrentUsage() / this.budgetBytesPerSecond) * 100;
  }

  shouldThrottle(): boolean {
    return this.getUtilizationPercent() > 90;
  }

  getRecommendedTickRate(baseTick: number): number {
    const utilization = this.getUtilizationPercent();
    if (utilization < 70) return baseTick;
    if (utilization < 90) return Math.floor(baseTick * 0.75);
    return Math.floor(baseTick * 0.5);
  }
}
自适应策略在带宽紧张时采取以下措施：降低远距离实体的更新频率、提高量化压缩的激进程度、跳过低优先级属性的复制、增大快照发送间隔。当带宽恢复充足时，逐步恢复到正常水平。


小结
    

  • 增量压缩通过只发送基准快照与当前状态的差异，实现 5-20 倍的数据压缩
    • 

  • 量化和位打包在可接受的精度损失下进一步压缩每个属性的存储空间
    • 

  • 变长编码和 ZigZag 编码利用数据的偏斜分布特性，对小值实现高效压缩
    • 

  • 基准快照通过客户端 Ack 确认机制管理，丢包时回退到更早的基准
    • 

  • 带宽监控系统持续追踪网络使用情况，在带宽紧张时自动降级
    • 

  • 多种压缩技术的组合使用是大规模游戏网络的标准实践
