权威服务器模型

核心概念

权威服务器（Authoritative Server）是现代在线游戏最广泛采用的网络架构模式。在这种模型中，服务器是游戏世界的唯一真相来源（Single Source of Truth），所有游戏逻辑的最终判定权都在服务器端。客户端只负责采集输入和渲染画面，不做任何权威性的游戏逻辑计算。

这种设计的核心目的是防止作弊。如果客户端拥有游戏逻辑的最终决定权，作弊者就可以通过修改客户端来获得不正当优势。权威服务器确保了即使客户端被完全修改，也无法影响服务器上的游戏结果。

服务器 Tick 循环
权威服务器的核心是一个固定频率的 Tick 循环。每个 Tick 中，服务器收集所有客户端输入、执行游戏逻辑、更新世界状态，然后将结果广播给所有客户端。


Tick Rate（每秒 Tick 数）直接影响游戏的响应精度。FPS 游戏通常使用 64-128 Tick/s，MMO 使用 20-30 Tick/s，回合制游戏可以低至 10 Tick/s。更高的 Tick Rate 意味着更精确的模拟和更低的感知延迟，但也带来更高的 CPU 和带宽消耗。


interface ServerTickConfig {
  tickRate: number;
  maxInputsPerTick: number;
  snapshotInterval: number;
}

interface WorldState {
  tick: number;
  timestamp: number;
  players: Map<string, PlayerEntity>;
  projectiles: Map<string, ProjectileEntity>;
  pickups: Map<string, PickupEntity>;
}

interface PlayerEntity {
  id: string;
  position: Vector3;
  velocity: Vector3;
  rotation: number;
  health: number;
  armor: number;
  weapon: string;
  alive: boolean;
}

interface ProjectileEntity {
  id: string;
  ownerId: string;
  position: Vector3;
  velocity: Vector3;
  damage: number;
  spawnTick: number;
  lifetime: number;
}

interface PickupEntity {
  id: string;
  type: string;
  position: Vector3;
  active: boolean;
  respawnTick: number;
}

interface Vector3 {
  x: number;
  y: number;
  z: number;
}

class AuthoritativeServer {
  private world: WorldState;
  private config: ServerTickConfig;
  private inputBuffer: Map<string, ClientInput[]>;
  private tickTimer: ReturnType<typeof setInterval> | null = null;

  constructor(config: ServerTickConfig) {
    this.config = config;
    this.world = {
      tick: 0,
      timestamp: Date.now(),
      players: new Map(),
      projectiles: new Map(),
      pickups: new Map(),
    };
    this.inputBuffer = new Map();
  }

  start(): void {
    const tickInterval = 1000 / this.config.tickRate;
    this.tickTimer = setInterval(() => this.tick(), tickInterval);
  }

  stop(): void {
    if (this.tickTimer) clearInterval(this.tickTimer);
  }

  receiveInput(clientId: string, input: ClientInput): void {
    const buffer = this.inputBuffer.get(clientId) ?? [];
    if (buffer.length < this.config.maxInputsPerTick) {
      buffer.push(input);
      this.inputBuffer.set(clientId, buffer);
    }
  }

  private tick(): void {
    this.processInputs();
    this.simulatePhysics();
    this.resolveCollisions();
    this.updateProjectiles();
    this.checkPickups();
    this.world.tick++;
    this.world.timestamp = Date.now();

    if (this.world.tick % this.config.snapshotInterval === 0) {
      this.broadcastSnapshot();
    }
  }

  private processInputs(): void {
    for (const [clientId, inputs] of this.inputBuffer) {
      const player = this.world.players.get(clientId);
      if (!player || !player.alive) continue;

      for (const input of inputs) {
        if (!this.validateInput(player, input)) continue;
        this.applyInput(player, input);
      }
    }
    this.inputBuffer.clear();
  }

  private validateInput(player: PlayerEntity, input: ClientInput): boolean {
    const maxSpeed = 10;
    const moveLength = Math.sqrt(input.moveX ** 2 + input.moveZ ** 2);
    if (moveLength > maxSpeed) return false;
    if (input.shoot && !player.alive) return false;
    return true;
  }

  private applyInput(player: PlayerEntity, input: ClientInput): void {
    player.velocity.x = input.moveX;
    player.velocity.z = input.moveZ;
    player.rotation = input.rotation;

    if (input.shoot) {
      this.spawnProjectile(player);
    }
  }

  private spawnProjectile(player: PlayerEntity): void {
    const speed = 50;
    const rad = (player.rotation * Math.PI) / 180;
    const projectile: ProjectileEntity = {
      id: crypto.randomUUID(),
      ownerId: player.id,
      position: { ...player.position },
      velocity: { x: Math.sin(rad) * speed, y: 0, z: Math.cos(rad) * speed },
      damage: 25,
      spawnTick: this.world.tick,
      lifetime: 120,
    };
    this.world.projectiles.set(projectile.id, projectile);
  }

  private simulatePhysics(): void {
    const dt = 1 / this.config.tickRate;
    for (const [, player] of this.world.players) {
      player.position.x += player.velocity.x * dt;
      player.position.y += player.velocity.y * dt;
      player.position.z += player.velocity.z * dt;
    }
  }

  private resolveCollisions(): void {
    for (const [, proj] of this.world.projectiles) {
      for (const [, player] of this.world.players) {
        if (proj.ownerId === player.id || !player.alive) continue;
        const dx = proj.position.x - player.position.x;
        const dz = proj.position.z - player.position.z;
        if (Math.sqrt(dx * dx + dz * dz) < 1.0) {
          player.health -= proj.damage;
          if (player.health <= 0) player.alive = false;
          this.world.projectiles.delete(proj.id);
          break;
        }
      }
    }
  }

  private updateProjectiles(): void {
    const dt = 1 / this.config.tickRate;
    for (const [id, proj] of this.world.projectiles) {
      proj.position.x += proj.velocity.x * dt;
      proj.position.z += proj.velocity.z * dt;
      if (this.world.tick - proj.spawnTick > proj.lifetime) {
        this.world.projectiles.delete(id);
      }
    }
  }

  private checkPickups(): void {
    for (const [, pickup] of this.world.pickups) {
      if (!pickup.active && this.world.tick >= pickup.respawnTick) {
        pickup.active = true;
      }
    }
  }

  private broadcastSnapshot(): void {
    const snapshot = this.createSnapshot();
    for (const [clientId] of this.world.players) {
      this.sendToClient(clientId, snapshot);
    }
  }

  private createSnapshot(): WorldSnapshot {
    return {
      tick: this.world.tick,
      timestamp: this.world.timestamp,
      players: Array.from(this.world.players.values()),
      projectiles: Array.from(this.world.projectiles.values()),
    };
  }

  private sendToClient(_clientId: string, _snapshot: WorldSnapshot): void {}
}

interface ClientInput {
  sequence: number;
  tick: number;
  moveX: number;
  moveZ: number;
  rotation: number;
  shoot: boolean;
}

interface WorldSnapshot {
  tick: number;
  timestamp: number;
  players: PlayerEntity[];
  projectiles: ProjectileEntity[];
}

服务端应该对可疑行为采用渐进式处罚。首先记录日志用于后续分析，然后在可疑度累积到一定阈值后自动限制玩家能力（如降低移动速度），最后对确认的作弊者进行封禁处理。


快照生成与分发
权威服务器需要定期将世界状态以快照（Snapshot）的形式发送给所有客户端。快照的生成和分发策略直接影响带宽消耗和客户端的渲染平滑度。


全量快照包含世界中所有实体的完整状态，数据量大但可以独立解码。增量快照只包含自上一个快照以来发生变化的部分，数据量小但依赖前一个快照。实际应用中通常以较低频率发送全量快照作为基准，中间用增量快照补充。


interface SnapshotConfig {
  fullSnapshotInterval: number;
  deltaSnapshotInterval: number;
  maxEntitiesPerPacket: number;
  priorityRadius: number;
}

class SnapshotManager {
  private config: SnapshotConfig;
  private lastFullSnapshot: Map<string, WorldSnapshot>;
  private entityPriority: Map<string, Map<string, number>>;

  constructor(config: SnapshotConfig) {
    this.config = config;
    this.lastFullSnapshot = new Map();
    this.entityPriority = new Map();
  }

  generateForClient(clientId: string, world: WorldState, clientPosition: Vector3): Uint8Array {
    const priorities = this.calculatePriorities(clientId, world, clientPosition);
    const sortedEntities = this.sortByPriority(priorities);
    const topEntities = sortedEntities.slice(0, this.config.maxEntitiesPerPacket);

    if (world.tick % this.config.fullSnapshotInterval === 0) {
      return this.encodeFullSnapshot(world, topEntities);
    }
    return this.encodeDeltaSnapshot(clientId, world, topEntities);
  }

  private calculatePriorities(clientId: string, world: WorldState, viewPosition: Vector3): Map<string, number> {
    const priorities = new Map<string, number>();
    for (const [id, player] of world.players) {
      if (id === clientId) {
        priorities.set(id, 1000);
        continue;
      }
      const dist = this.distance(viewPosition, player.position);
      const priority = Math.max(0, 100 - dist);
      priorities.set(id, priority);
    }
    return priorities;
  }

  private distance(a: Vector3, b: Vector3): number {
    const dx = a.x - b.x;
    const dy = a.y - b.y;
    const dz = a.z - b.z;
    return Math.sqrt(dx * dx + dy * dy + dz * dz);
  }

  private sortByPriority(priorities: Map<string, number>): string[] {
    return Array.from(priorities.entries())
      .sort((a, b) => b[1] - a[1])
      .map(([id]) => id);
  }

  private encodeFullSnapshot(_world: WorldState, _entities: string[]): Uint8Array {
    return new Uint8Array(0);
  }

  private encodeDeltaSnapshot(_clientId: string, _world: WorldState, _entities: string[]): Uint8Array {
    return new Uint8Array(0);
  }
}
基于优先级的快照分发是大规模游戏的关键优化。距离玩家较近的实体以更高频率、更完整的数据同步；远处的实体则降低更新频率，甚至只同步关键属性。这种策略可以在不影响玩家感知的前提下大幅降低带宽消耗。


小结
    

  • 权威服务器是游戏世界的唯一真相来源，从架构层面杜绝客户端作弊
    • 

  • Tick 循环以固定频率运行游戏逻辑，Tick Rate 根据游戏类型选择（FPS 64-128，MMO 20-30）
    • 

  • 输入验证应覆盖数值范围、频率限制和物理合理性，采用渐进式处罚策略
    • 

  • 快照分发需要结合全量/增量策略和基于优先级的实体过滤来优化带宽
    • 

  • 权威服务器的代价是增加了输入延迟，需要配合客户端预测技术来改善手感
