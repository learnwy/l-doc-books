连接管理与会话

核心概念

连接管理是游戏网络系统中负责建立、维护和终止客户端与服务器之间通信链路的子系统。会话（Session）则是对连接状态的高层抽象，它跟踪玩家的身份认证状态、网络质量指标和游戏进度信息。

在基于 UDP 的游戏网络中，由于 UDP 本身无连接，连接管理需要在应用层实现虚拟连接的概念。这包括自定义握手流程、心跳检测、超时判定和优雅断开等机制。良好的连接管理直接影响玩家体验——它决定了玩家能否快速进入游戏、网络波动时是否能保持连接以及断线后能否无缝重连。

自定义握手协议
基于 UDP 的游戏通常需要实现自定义的握手协议来建立虚拟连接。握手过程需要完成身份验证、协议版本协商和会话密钥交换等任务，同时防止连接洪泛等拒绝服务攻击。


const enum ConnectionState {
  Disconnected = 0,
  Connecting = 1,
  Handshaking = 2,
  Connected = 3,
  Playing = 4,
  Reconnecting = 5,
  Disconnecting = 6,
}

interface ConnectionConfig {
  serverAddress: string;
  connectTimeout: number;
  heartbeatInterval: number;
  disconnectTimeout: number;
  maxReconnectAttempts: number;
  reconnectWindow: number;
}

interface SessionInfo {
  sessionId: string;
  clientId: string;
  state: ConnectionState;
  connectTime: number;
  lastHeartbeatSent: number;
  lastHeartbeatReceived: number;
  rtt: number;
  jitter: number;
  packetLoss: number;
  reconnectAttempts: number;
}

const DEFAULT_CONFIG: ConnectionConfig = {
  serverAddress: '',
  connectTimeout: 10000,
  heartbeatInterval: 1000,
  disconnectTimeout: 15000,
  maxReconnectAttempts: 5,
  reconnectWindow: 60000,
};

class ConnectionManager {
  private config: ConnectionConfig;
  private session: SessionInfo | null = null;
  private heartbeatTimer: ReturnType<typeof setInterval> | null = null;
  private rttSamples: number[] = [];

  constructor(config: Partial<ConnectionConfig>) {
    this.config = { ...DEFAULT_CONFIG, ...config };
  }

  async connect(clientId: string, token: string): Promise<SessionInfo> {
    this.session = {
      sessionId: '',
      clientId,
      state: ConnectionState.Connecting,
      connectTime: Date.now(),
      lastHeartbeatSent: 0,
      lastHeartbeatReceived: 0,
      rtt: 0,
      jitter: 0,
      packetLoss: 0,
      reconnectAttempts: 0,
    };

    const challenge = await this.sendConnectionRequest(clientId, token);
    const response = this.computeChallengeResponse(challenge);
    const accepted = await this.sendChallengeResponse(response);

    this.session.sessionId = accepted.sessionId;
    this.session.state = ConnectionState.Connected;
    this.startHeartbeat();
    return this.session;
  }

  private startHeartbeat(): void {
    this.heartbeatTimer = setInterval(() => {
      if (!this.session) return;

      const now = Date.now();
      if (now - this.session.lastHeartbeatReceived > this.config.disconnectTimeout) {
        this.onDisconnectDetected();
        return;
      }

      this.sendHeartbeat(now);
      this.session.lastHeartbeatSent = now;
    }, this.config.heartbeatInterval);
  }

  private onDisconnectDetected(): void {
    if (!this.session) return;
    this.session.state = ConnectionState.Reconnecting;
    this.attemptReconnect();
  }

  private async attemptReconnect(): Promise<void> {
    if (!this.session) return;

    while (
      this.session.reconnectAttempts < this.config.maxReconnectAttempts &&
      this.session.state === ConnectionState.Reconnecting
    ) {
      this.session.reconnectAttempts++;
      const delay = Math.min(1000 * 2 ** this.session.reconnectAttempts, 30000);
      await this.sleep(delay);

      const success = await this.sendReconnectRequest(this.session.sessionId);
      if (success) {
        this.session.state = ConnectionState.Playing;
        this.session.reconnectAttempts = 0;
        return;
      }
    }

    this.disconnect();
  }

  disconnect(): void {
    if (this.heartbeatTimer) clearInterval(this.heartbeatTimer);
    if (this.session) this.session.state = ConnectionState.Disconnected;
    this.session = null;
  }

  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }

  private async sendConnectionRequest(
    _clientId: string,
    _token: string,
  ): Promise<{ challenge: Uint8Array; sessionId: string }> {
    return { challenge: new Uint8Array(32), sessionId: '' };
  }

  private computeChallengeResponse(_challenge: { challenge: Uint8Array; sessionId: string }): Uint8Array {
    return new Uint8Array(32);
  }

  private async sendChallengeResponse(_response: Uint8Array): Promise<{ sessionId: string }> {
    return { sessionId: crypto.randomUUID() };
  }

  private sendHeartbeat(_timestamp: number): void {}
  private async sendReconnectRequest(_sessionId: string): Promise<boolean> {
    return false;
  }
}
Challenge-Response 机制是防止连接洪泛攻击的关键手段。服务器收到连接请求后不会立即分配资源，而是发送一个 Challenge 给客户端。只有客户端正确响应 Challenge 后，服务器才分配完整的会话资源。这迫使攻击者必须维护真实的网络地址，无法使用伪造 IP 进行攻击。


心跳与网络质量监测
心跳机制是连接管理的核心组成部分。它不仅用于检测连接是否存活，还是测量 RTT、计算抖动（Jitter）和估算丢包率的基础。


interface NetworkQuality {
  rtt: number;
  jitter: number;
  packetLoss: number;
  bandwidth: number;
  quality: 'excellent' | 'good' | 'fair' | 'poor';
}

class NetworkMonitor {
  private rttHistory: number[] = [];
  private sentPackets: number = 0;
  private ackedPackets: number = 0;
  private windowSize: number;

  constructor(windowSize: number = 60) {
    this.windowSize = windowSize;
  }

  recordRtt(rtt: number): void {
    this.rttHistory.push(rtt);
    if (this.rttHistory.length > this.windowSize) {
      this.rttHistory.shift();
    }
  }

  recordSent(): void {
    this.sentPackets++;
  }

  recordAck(): void {
    this.ackedPackets++;
  }

  getQuality(): NetworkQuality {
    const rtt = this.getAverageRtt();
    const jitter = this.getJitter();
    const packetLoss = this.getPacketLoss();

    let quality: NetworkQuality['quality'];
    if (rtt < 50 && packetLoss < 1) quality = 'excellent';
    else if (rtt < 100 && packetLoss < 3) quality = 'good';
    else if (rtt < 200 && packetLoss < 5) quality = 'fair';
    else quality = 'poor';

    return { rtt, jitter, packetLoss, bandwidth: 0, quality };
  }

  private getAverageRtt(): number {
    if (this.rttHistory.length === 0) return 0;
    return this.rttHistory.reduce((a, b) => a + b, 0) / this.rttHistory.length;
  }

  private getJitter(): number {
    if (this.rttHistory.length < 2) return 0;
    let sum = 0;
    for (let i = 1; i < this.rttHistory.length; i++) {
      sum += Math.abs(this.rttHistory[i] - this.rttHistory[i - 1]);
    }
    return sum / (this.rttHistory.length - 1);
  }

  private getPacketLoss(): number {
    if (this.sentPackets === 0) return 0;
    return ((this.sentPackets - this.ackedPackets) / this.sentPackets) * 100;
  }
}
网络质量信息不仅用于连接管理决策，还可以反馈给游戏逻辑层进行自适应调整。例如，当检测到网络质量下降时，可以降低服务器的快照发送频率、启用更激进的增量压缩或切换到低带宽模式。


断线重连策略
断线重连是保证玩家体验的关键机制。好的重连策略应该对玩家透明——短暂的网络中断不应导致游戏进度丢失或被踢出房间。


重连流程的核心挑战是状态同步。客户端断线期间，服务器上的游戏状态持续演进。重连后，客户端需要快速同步到最新状态，同时处理可能的冲突。


interface ReconnectState {
  sessionId: string;
  lastConfirmedTick: number;
  pendingInputs: PlayerInputPayload[];
  disconnectTime: number;
}

interface PlayerInputPayload {
  sequence: number;
  tick: number;
  moveX: number;
  moveZ: number;
  actions: number;
}

interface ReconnectResult {
  success: boolean;
  currentTick: number;
  fullSnapshot: Uint8Array;
  missedEvents: GameEvent[];
}

interface GameEvent {
  tick: number;
  type: string;
  data: Record<string, unknown>;
}

class ReconnectHandler {
  private reconnectState: ReconnectState | null = null;
  private maxReconnectWindow: number = 60000;

  saveReconnectState(sessionId: string, lastTick: number, pendingInputs: PlayerInputPayload[]): void {
    this.reconnectState = {
      sessionId,
      lastConfirmedTick: lastTick,
      pendingInputs: [...pendingInputs],
      disconnectTime: Date.now(),
    };
  }

  canReconnect(): boolean {
    if (!this.reconnectState) return false;
    return Date.now() - this.reconnectState.disconnectTime < this.maxReconnectWindow;
  }

  async performReconnect(): Promise<ReconnectResult | null> {
    if (!this.reconnectState || !this.canReconnect()) return null;

    const result = await this.sendReconnectRequest({
      sessionId: this.reconnectState.sessionId,
      lastConfirmedTick: this.reconnectState.lastConfirmedTick,
    });

    if (result.success) {
      this.replayPendingInputs(result.currentTick);
    }

    return result;
  }

  private replayPendingInputs(serverTick: number): void {
    if (!this.reconnectState) return;
    const validInputs = this.reconnectState.pendingInputs.filter((input) => input.tick > serverTick);
    for (const input of validInputs) {
      this.resendInput(input);
    }
  }

  private async sendReconnectRequest(_params: {
    sessionId: string;
    lastConfirmedTick: number;
  }): Promise<ReconnectResult> {
    return {
      success: true,
      currentTick: 0,
      fullSnapshot: new Uint8Array(0),
      missedEvents: [],
    };
  }

  private resendInput(_input: PlayerInputPayload): void {}
}

指数退避（Exponential Backoff）是重连重试的标准策略。每次重试的等待时间按 2 的幂次递增，并加入随机抖动（Jitter），避免大量客户端同时重连造成服务器压力突增。
