兴趣区域管理

核心概念

兴趣区域（Area of Interest, AOI）管理是 MMO 游戏网络优化的核心机制。在一个拥有数千名在线玩家的世界中，如果每个玩家都接收所有其他玩家的状态更新，带宽和计算开销将呈 O(n²) 增长，这在实际中是不可承受的。

AOI 系统解决的核心问题是：对于每个玩家，高效地计算出哪些游戏实体在其"关注范围"内，只将这些实体的状态更新发送给该玩家。这将每个玩家的带宽消耗从 O(n) 降低到 O(k)，其中 k 是视野内的实体数量。

九宫格算法
九宫格（Grid-based AOI）是 MMO 中最经典且广泛使用的 AOI 算法。它将游戏世界划分为固定大小的网格，每个实体根据其位置归属到对应的格子中。查询某个实体的 AOI 时，只需检查它所在的格子及周围的 8 个格子中的实体。


interface GridEntity {
  id: string;
  x: number;
  z: number;
  type: 'player' | 'npc' | 'item';
  cellX: number;
  cellZ: number;
}

interface AOIEvent {
  type: 'enter' | 'leave' | 'move';
  entityId: string;
  observerId: string;
}

class GridAOI {
  private cellSize: number;
  private cells: Map<string, Set<string>>;
  private entities: Map<string, GridEntity>;
  private observers: Map<string, Set<string>>;

  constructor(cellSize: number) {
    this.cellSize = cellSize;
    this.cells = new Map();
    this.entities = new Map();
    this.observers = new Map();
  }

  private getCellKey(cx: number, cz: number): string {
    return `${cx},${cz}`;
  }

  private worldToCell(x: number): number {
    return Math.floor(x / this.cellSize);
  }

  addEntity(entity: GridEntity): AOIEvent[] {
    const cx = this.worldToCell(entity.x);
    const cz = this.worldToCell(entity.z);
    entity.cellX = cx;
    entity.cellZ = cz;

    this.entities.set(entity.id, entity);

    const cellKey = this.getCellKey(cx, cz);
    if (!this.cells.has(cellKey)) this.cells.set(cellKey, new Set());
    this.cells.get(cellKey)!.add(entity.id);

    const events: AOIEvent[] = [];
    const nearbyIds = this.getNearbyEntityIds(cx, cz);

    for (const observerId of nearbyIds) {
      if (observerId === entity.id) continue;
      events.push({ type: 'enter', entityId: entity.id, observerId });
      events.push({ type: 'enter', entityId: observerId, observerId: entity.id });
    }

    this.observers.set(entity.id, new Set(nearbyIds));
    return events;
  }

  moveEntity(entityId: string, newX: number, newZ: number): AOIEvent[] {
    const entity = this.entities.get(entityId);
    if (!entity) return [];

    const newCX = this.worldToCell(newX);
    const newCZ = this.worldToCell(newZ);

    entity.x = newX;
    entity.z = newZ;

    if (newCX === entity.cellX && newCZ === entity.cellZ) return [];

    const oldKey = this.getCellKey(entity.cellX, entity.cellZ);
    this.cells.get(oldKey)?.delete(entityId);

    entity.cellX = newCX;
    entity.cellZ = newCZ;
    const newKey = this.getCellKey(newCX, newCZ);
    if (!this.cells.has(newKey)) this.cells.set(newKey, new Set());
    this.cells.get(newKey)!.add(entityId);

    return this.computeAOIChanges(entityId, newCX, newCZ);
  }

  private computeAOIChanges(entityId: string, cx: number, cz: number): AOIEvent[] {
    const events: AOIEvent[] = [];
    const newNearby = new Set(this.getNearbyEntityIds(cx, cz));
    const oldNearby = this.observers.get(entityId) ?? new Set();

    for (const id of newNearby) {
      if (id === entityId) continue;
      if (!oldNearby.has(id)) {
        events.push({ type: 'enter', entityId: id, observerId: entityId });
        events.push({ type: 'enter', entityId: entityId, observerId: id });
      }
    }

    for (const id of oldNearby) {
      if (id === entityId) continue;
      if (!newNearby.has(id)) {
        events.push({ type: 'leave', entityId: id, observerId: entityId });
        events.push({ type: 'leave', entityId: entityId, observerId: id });
      }
    }

    this.observers.set(entityId, newNearby);
    return events;
  }

  private getNearbyEntityIds(cx: number, cz: number): string[] {
    const result: string[] = [];
    for (let dx = -1; dx <= 1; dx++) {
      for (let dz = -1; dz <= 1; dz++) {
        const key = this.getCellKey(cx + dx, cz + dz);
        const cell = this.cells.get(key);
        if (cell) {
          for (const id of cell) result.push(id);
        }
      }
    }
    return result;
  }

  removeEntity(entityId: string): AOIEvent[] {
    const entity = this.entities.get(entityId);
    if (!entity) return [];

    const events: AOIEvent[] = [];
    const nearby = this.observers.get(entityId) ?? new Set();

    for (const observerId of nearby) {
      if (observerId === entityId) continue;
      events.push({ type: 'leave', entityId, observerId });
    }

    const cellKey = this.getCellKey(entity.cellX, entity.cellZ);
    this.cells.get(cellKey)?.delete(entityId);
    this.entities.delete(entityId);
    this.observers.delete(entityId);

    return events;
  }
}
九宫格的格子大小选择需要权衡。格子太大会包含过多实体，增加带宽消耗；格子太小会导致实体频繁跨格，增加事件处理开销。通常将格子大小设为 AOI 半径的 1-2 倍是一个好的平衡点。


四叉树空间索引
当游戏世界中实体分布极不均匀时（如城镇区域人口密集、荒野区域稀疏），四叉树（Quadtree）提供了更好的查询效率。四叉树递归地将空间四等分，直到每个叶子节点包含的实体数不超过阈值。


interface Rect {
  x: number;
  z: number;
  width: number;
  height: number;
}

interface QuadEntity {
  id: string;
  x: number;
  z: number;
}

class QuadTree {
  private boundary: Rect;
  private capacity: number;
  private entities: QuadEntity[] = [];
  private divided: boolean = false;
  private children: QuadTree[] = [];

  constructor(boundary: Rect, capacity: number = 8) {
    this.boundary = boundary;
    this.capacity = capacity;
  }

  insert(entity: QuadEntity): boolean {
    if (!this.contains(entity.x, entity.z)) return false;

    if (this.entities.length < this.capacity && !this.divided) {
      this.entities.push(entity);
      return true;
    }

    if (!this.divided) this.subdivide();

    for (const child of this.children) {
      if (child.insert(entity)) return true;
    }
    return false;
  }

  queryRange(range: Rect): QuadEntity[] {
    const found: QuadEntity[] = [];

    if (!this.intersects(range)) return found;

    for (const entity of this.entities) {
      if (this.pointInRect(entity.x, entity.z, range)) {
        found.push(entity);
      }
    }

    if (this.divided) {
      for (const child of this.children) {
        found.push(...child.queryRange(range));
      }
    }

    return found;
  }

  queryCircle(cx: number, cz: number, radius: number): QuadEntity[] {
    const range: Rect = {
      x: cx - radius,
      z: cz - radius,
      width: radius * 2,
      height: radius * 2,
    };
    const candidates = this.queryRange(range);
    return candidates.filter((e) => {
      const dx = e.x - cx;
      const dz = e.z - cz;
      return dx * dx + dz * dz <= radius * radius;
    });
  }

  private subdivide(): void {
    const { x, z, width, height } = this.boundary;
    const hw = width / 2;
    const hh = height / 2;

    this.children = [
      new QuadTree({ x, z, width: hw, height: hh }, this.capacity),
      new QuadTree({ x: x + hw, z, width: hw, height: hh }, this.capacity),
      new QuadTree({ x, z: z + hh, width: hw, height: hh }, this.capacity),
      new QuadTree({ x: x + hw, z: z + hh, width: hw, height: hh }, this.capacity),
    ];
    this.divided = true;

    for (const entity of this.entities) {
      for (const child of this.children) {
        if (child.insert(entity)) break;
      }
    }
    this.entities = [];
  }

  private contains(px: number, pz: number): boolean {
    return this.pointInRect(px, pz, this.boundary);
  }

  private pointInRect(px: number, pz: number, rect: Rect): boolean {
    return px >= rect.x && px < rect.x + rect.width && pz >= rect.z && pz < rect.z + rect.height;
  }

  private intersects(range: Rect): boolean {
    return !(
      range.x > this.boundary.x + this.boundary.width ||
      range.x + range.width < this.boundary.x ||
      range.z > this.boundary.z + this.boundary.height ||
      range.z + range.height < this.boundary.z
    );
  }
}
四叉树在实体密集区域会进行更深层的细分，而在稀疏区域保持较大的节点，自适应地匹配实体分布。但实体频繁移动时需要持续更新树结构，这在高频更新的游戏中可能成为性能瓶颈。


动态 AOI 与优先级分层
静态的固定半径 AOI 在某些场景下不够灵活。动态 AOI 根据实体的类型、重要性和玩家的当前状态调整观察范围。


interface DynamicAOIConfig {
  baseRadius: number;
  maxRadius: number;
  minRadius: number;
  priorityLayers: PriorityLayer[];
}

interface PriorityLayer {
  name: string;
  radius: number;
  updateInterval: number;
  entityFilter: (entity: GridEntity) => boolean;
}

class DynamicAOISystem {
  private config: DynamicAOIConfig;
  private spatialIndex: GridAOI;
  private playerRadii: Map<string, number> = new Map();

  constructor(config: DynamicAOIConfig, spatialIndex: GridAOI) {
    this.config = config;
    this.spatialIndex = spatialIndex;
  }

  getVisibleEntities(
    playerId: string,
    playerX: number,
    playerZ: number,
    allEntities: Map<string, GridEntity>,
  ): Map<string, { entity: GridEntity; priority: number }> {
    const result = new Map<string, { entity: GridEntity; priority: number }>();

    for (const layer of this.config.priorityLayers) {
      for (const [id, entity] of allEntities) {
        if (id === playerId) continue;
        if (!layer.entityFilter(entity)) continue;

        const dx = entity.x - playerX;
        const dz = entity.z - playerZ;
        const dist = Math.sqrt(dx * dx + dz * dz);

        if (dist <= layer.radius) {
          const existing = result.get(id);
          const priority = (1 - dist / layer.radius) * 100;
          if (!existing || existing.priority < priority) {
            result.set(id, { entity, priority });
          }
        }
      }
    }

    return result;
  }

  adjustRadius(playerId: string, playerCount: number): void {
    let radius = this.config.baseRadius;

    if (playerCount > 100) {
      radius *= 0.7;
    } else if (playerCount < 20) {
      radius *= 1.3;
    }

    radius = Math.max(this.config.minRadius, Math.min(this.config.maxRadius, radius));
    this.playerRadii.set(playerId, radius);
  }
}

在人口密集的城镇区域，动态缩小 AOI 半径可以限制每个玩家的可见实体数量，防止服务器和客户端过载。在空旷的野外区域，扩大 AOI 半径让玩家能看到更远的实体，提升游戏体验。


小结
    

  • AOI 系统将每个玩家的状态同步范围从 O(n) 降至 O(k)，是 MMO 网络优化的基石
    • 

  • 九宫格算法实现简单、查询高效，适用于大多数 MMO 游戏
    • 

  • 四叉树在实体分布不均匀时提供更好的查询效率，但更新成本较高
    • 

  • 动态 AOI 根据区域密度和实体类型自适应调整半径和更新频率
    • 

  • 优先级分层确保近距重要实体获得最高更新频率
    • 

  • 进出事件驱动客户端的实体创建和销毁，减少不必要的网络流量
