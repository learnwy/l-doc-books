坐标系统与 Transform

核心概念

坐标系统是 Unity 空间定位的基础，Transform 组件是每个 GameObject 必备的核心组件，负责管理对象在三维空间中的位置（Position）、旋转（Rotation）和缩放（Scale）。Unity 使用左手坐标系，Y 轴朝上，Z 轴朝前。理解世界空间、本地空间和屏幕空间之间的转换关系，是实现精确空间操作的关键。

位置与移动
Transform 的 position 属性表示世界空间坐标，localPosition 表示相对于父对象的本地坐标。当对象没有父对象时，两者相同。移动对象有多种方式，选择取决于是否需要物理交互。


using UnityEngine;

public class MovementController : MonoBehaviour
{
    [SerializeField] private float moveSpeed = 5f;
    [SerializeField] private bool usePhysics = true;

    private Rigidbody rb;

    private void Awake()
    {
        rb = GetComponent<Rigidbody>();
    }

    private void Update()
    {
        if (!usePhysics)
        {
            float h = Input.GetAxis("Horizontal");
            float v = Input.GetAxis("Vertical");
            Vector3 direction = new Vector3(h, 0, v).normalized;
            transform.Translate(direction * moveSpeed * Time.deltaTime, Space.World);
        }
    }

    private void FixedUpdate()
    {
        if (usePhysics && rb != null)
        {
            float h = Input.GetAxis("Horizontal");
            float v = Input.GetAxis("Vertical");
            Vector3 velocity = new Vector3(h, rb.linearVelocity.y, v) * moveSpeed;
            rb.linearVelocity = velocity;
        }
    }
}

使用 Transform 直接移动的对象不会触发物理系统，穿透碰撞体时不会产生碰撞事件。需要物理交互的对象必须通过 Rigidbody API 移动。


旋转与四元数
Unity 内部使用四元数（Quaternion）表示旋转，避免了欧拉角的万向节锁（Gimbal Lock）问题。Inspector 面板显示的欧拉角只是四元数的可读表示。开发中应优先使用四元数 API 进行旋转操作。


using UnityEngine;

public class RotationController : MonoBehaviour
{
    [SerializeField] private float rotateSpeed = 90f;
    [SerializeField] private Transform target;

    private void Update()
    {
        transform.Rotate(Vector3.up, rotateSpeed * Time.deltaTime, Space.World);
    }

    public void SmoothLookAt(float smoothTime)
    {
        if (target == null) return;

        Vector3 direction = (target.position - transform.position).normalized;
        if (direction == Vector3.zero) return;

        Quaternion targetRotation = Quaternion.LookRotation(direction);
        transform.rotation = Quaternion.Slerp(
            transform.rotation,
            targetRotation,
            smoothTime * Time.deltaTime
        );
    }

    public void RotateAround(Vector3 pivot, float angle)
    {
        transform.RotateAround(pivot, Vector3.up, angle * Time.deltaTime);
    }
}

避免直接修改 transform.eulerAngles 的单个分量（如 eulerAngles.y = 90），因为欧拉角的 getter 每次可能返回不同的等效表示，应使用 Quaternion.Euler 构建完整旋转。


空间变换与坐标转换
不同坐标空间之间的转换在游戏开发中无处不在：将鼠标屏幕坐标转为世界坐标以实现点击定位、将世界坐标转为本地坐标进行局部判断、将世界坐标转为屏幕坐标显示 UI 标记等。


using UnityEngine;

public class CoordinateConverter : MonoBehaviour
{
    [SerializeField] private Camera mainCamera;
    [SerializeField] private LayerMask groundLayer;

    public Vector3 ScreenToWorldOnPlane(Vector3 screenPos, float planeY = 0f)
    {
        Ray ray = mainCamera.ScreenPointToRay(screenPos);
        Plane ground = new Plane(Vector3.up, new Vector3(0, planeY, 0));

        if (ground.Raycast(ray, out float distance))
            return ray.GetPoint(distance);

        return Vector3.zero;
    }

    public Vector3 WorldToLocal(Vector3 worldPos)
    {
        return transform.InverseTransformPoint(worldPos);
    }

    public Vector3 LocalToWorld(Vector3 localPos)
    {
        return transform.TransformPoint(localPos);
    }

    public Vector3 WorldDirectionToLocal(Vector3 worldDir)
    {
        return transform.InverseTransformDirection(worldDir);
    }
}

TransformDirection 不受缩放影响，适合方向转换；TransformVector 包含缩放，适合向量转换；TransformPoint 包含位置偏移、旋转和缩放，适合点的转换。


缩放与层级陷阱
缩放在 Unity 层级系统中具有级联效应：父对象的缩放会影响所有子对象的最终缩放。非均匀缩放（X/Y/Z 不等）可能导致物理碰撞异常、法线扭曲和性能下降。


using UnityEngine;

public class ScaleManager : MonoBehaviour
{
    public Vector3 GetLossyScale(Transform target)
    {
        return target.lossyScale;
    }

    public void ResetLocalScale(Transform target)
    {
        target.localScale = Vector3.one;
    }

    public void ScaleUniform(Transform target, float factor)
    {
        target.localScale = Vector3.one * factor;
    }

    public bool HasNonUniformScale(Transform target)
    {
        Vector3 s = target.lossyScale;
        return !Mathf.Approximately(s.x, s.y) || !Mathf.Approximately(s.y, s.z);
    }
}
非均匀缩放的常见问题包括：BoxCollider 形变不匹配、SphereCollider 变椭球但物理仍按球计算、法线贴图方向错误、骨骼动画变形。最佳实践是在建模阶段确定正确尺寸，运行时仅使用均匀缩放。


lossyScale 返回的是世界空间的近似缩放值（当存在旋转+非均匀缩放时不精确），localScale 是相对于父对象的精确缩放值。


小结
    

  • Unity 使用左手坐标系，Transform 是每个 GameObject 的必备组件
    • 

  • 世界空间用于全局定位，本地空间用于层级相对定位，屏幕空间用于 UI 交互
    • 

  • 移动对象需区分物理移动（Rigidbody）和非物理移动（Transform），两者不可混用
    • 

  • 旋转应优先使用四元数 API，避免欧拉角的万向节锁问题
    • 

  • 空间转换使用 TransformPoint/InverseTransformPoint 系列方法，注意方向、向量和点的区别
    • 

  • 运行时应避免非均匀缩放，lossyScale 在复杂层级下可能不精确
    •
