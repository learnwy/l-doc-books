事件系统

核心概念

事件系统是 Unity 开发中实现模块间松耦合通信的核心机制。在游戏开发中，角色受伤需要通知 UI 更新血条、音效系统播放受击音、相机执行震动效果——这些系统之间不应直接引用彼此。事件系统通过"发布-订阅"模式让发送方和接收方完全解耦，提升代码的可维护性和可扩展性。

C# 委托与 event 关键字
C# 的 delegate 和 event 关键字是 Unity 事件系统的底层基础。delegate 定义方法签名类型，event 关键字在此基础上添加了访问限制——外部只能订阅/取消订阅，不能直接调用或赋值，防止意外覆盖其他订阅者。


using UnityEngine;
using System;

public class HealthSystem : MonoBehaviour
{
    public event Action<float, float> OnHealthChanged;
    public event Action OnDeath;

    [SerializeField] private float maxHealth = 100f;
    private float currentHealth;

    public float CurrentHealth => currentHealth;
    public float MaxHealth => maxHealth;
    public float HealthPercent => currentHealth / maxHealth;

    private void Awake()
    {
        currentHealth = maxHealth;
    }

    public void TakeDamage(float damage)
    {
        if (currentHealth <= 0f) return;

        currentHealth = Mathf.Max(0f, currentHealth - damage);
        OnHealthChanged?.Invoke(currentHealth, maxHealth);

        if (currentHealth <= 0f)
        {
            OnDeath?.Invoke();
        }
    }

    public void Heal(float amount)
    {
        if (currentHealth <= 0f) return;

        currentHealth = Mathf.Min(maxHealth, currentHealth + amount);
        OnHealthChanged?.Invoke(currentHealth, maxHealth);
    }
}
订阅者通过 += 和 -= 操作符注册和注销回调。必须在 OnDisable 或 OnDestroy 中取消订阅，否则会因为引用残留导致空引用异常或内存泄漏：


using UnityEngine;
using UnityEngine.UI;

public class HealthBarUI : MonoBehaviour
{
    [SerializeField] private HealthSystem healthSystem;
    [SerializeField] private Slider healthSlider;
    [SerializeField] private Image fillImage;
    [SerializeField] private Color healthyColor = Color.green;
    [SerializeField] private Color criticalColor = Color.red;

    private void OnEnable()
    {
        healthSystem.OnHealthChanged += UpdateHealthBar;
        healthSystem.OnDeath += HandleDeath;
    }

    private void OnDisable()
    {
        healthSystem.OnHealthChanged -= UpdateHealthBar;
        healthSystem.OnDeath -= HandleDeath;
    }

    private void UpdateHealthBar(float current, float max)
    {
        float percent = current / max;
        healthSlider.value = percent;
        fillImage.color = Color.Lerp(criticalColor, healthyColor, percent);
    }

    private void HandleDeath()
    {
        gameObject.SetActive(false);
    }
}
UnityEvent 与 Inspector 绑定
UnityEvent 是 Unity 对 C# 事件的封装，最大的优势在于可以在 Inspector 面板中直接配置事件响应，让设计师无需修改代码即可调整事件行为。UnityEvent 支持序列化，可以保存在预制体和场景中。


using UnityEngine;
using UnityEngine.Events;

[System.Serializable]
public class DamageEvent : UnityEvent<float, Vector3> { }

public class InteractableObject : MonoBehaviour
{
    [Header("交互事件")]
    [SerializeField] private UnityEvent onInteractStart;
    [SerializeField] private UnityEvent onInteractEnd;

    [Header("伤害事件")]
    [SerializeField] private DamageEvent onDamageReceived;

    [Header("交互设置")]
    [SerializeField] private float interactRange = 2f;
    [SerializeField] private KeyCode interactKey = KeyCode.E;

    private bool isInteracting;

    public void OnPlayerEnterRange()
    {
        if (Input.GetKeyDown(interactKey) && !isInteracting)
        {
            isInteracting = true;
            onInteractStart?.Invoke();
        }
    }

    public void ReceiveDamage(float damage, Vector3 hitPoint)
    {
        onDamageReceived?.Invoke(damage, hitPoint);
    }

    public void EndInteraction()
    {
        if (isInteracting)
        {
            isInteracting = false;
            onInteractEnd?.Invoke();
        }
    }
}

静态事件总线架构
静态事件总线是大型项目中最常用的全局事件通信方案。通过静态类持有事件实例，任何脚本都可以发布和订阅事件，无需持有对方引用。结合泛型可以构建类型安全的事件系统。


using System;
using System.Collections.Generic;

public static class EventBus<T> where T : struct
{
    private static readonly HashSet<Action<T>> listeners = new();

    public static void Register(Action<T> handler)
    {
        listeners.Add(handler);
    }

    public static void Unregister(Action<T> handler)
    {
        listeners.Remove(handler);
    }

    public static void Raise(T eventData)
    {
        foreach (var listener in listeners)
        {
            listener.Invoke(eventData);
        }
    }

    public static void Clear()
    {
        listeners.Clear();
    }
}

public struct DamageDealtEvent
{
    public float Damage;
    public UnityEngine.Vector3 HitPoint;
    public UnityEngine.GameObject Source;
    public UnityEngine.GameObject Target;
}

public struct EnemyKilledEvent
{
    public UnityEngine.GameObject Enemy;
    public int ScoreValue;
}
使用事件总线时，各系统只需订阅关心的事件类型：


using UnityEngine;

public class DamageNumberSpawner : MonoBehaviour
{
    [SerializeField] private GameObject damageNumberPrefab;

    private void OnEnable()
    {
        EventBus<DamageDealtEvent>.Register(OnDamageDealt);
    }

    private void OnDisable()
    {
        EventBus<DamageDealtEvent>.Unregister(OnDamageDealt);
    }

    private void OnDamageDealt(DamageDealtEvent evt)
    {
        var go = Instantiate(damageNumberPrefab, evt.HitPoint, Quaternion.identity);
        go.GetComponent<DamageNumber>().Setup(evt.Damage);
    }
}

public class ScoreManager : MonoBehaviour
{
    private int totalScore;

    private void OnEnable()
    {
        EventBus<EnemyKilledEvent>.Register(OnEnemyKilled);
    }

    private void OnDisable()
    {
        EventBus<EnemyKilledEvent>.Unregister(OnEnemyKilled);
    }

    private void OnEnemyKilled(EnemyKilledEvent evt)
    {
        totalScore += evt.ScoreValue;
    }
}
ScriptableObject 事件通道
ScriptableObject 事件通道（Event Channel）将事件定义为可持久化的资源文件，结合了静态事件总线的解耦能力和 UnityEvent 的可视化优势。事件通道在 Inspector 中可见，支持跨场景引用，是 Unity 官方推荐的事件架构模式。


using UnityEngine;
using System.Collections.Generic;
using System;

[CreateAssetMenu(menuName = "Events/Void Event Channel")]
public class VoidEventChannel : ScriptableObject
{
    private readonly HashSet<Action> listeners = new();

    public void Register(Action handler)
    {
        listeners.Add(handler);
    }

    public void Unregister(Action handler)
    {
        listeners.Remove(handler);
    }

    public void Raise()
    {
        foreach (var listener in listeners)
        {
            listener.Invoke();
        }
    }
}

[CreateAssetMenu(menuName = "Events/Float Event Channel")]
public class FloatEventChannel : ScriptableObject
{
    private readonly HashSet<Action<float>> listeners = new();

    public void Register(Action<float> handler)
    {
        listeners.Add(handler);
    }

    public void Unregister(Action<float> handler)
    {
        listeners.Remove(handler);
    }

    public void Raise(float value)
    {
        foreach (var listener in listeners)
        {
            listener.Invoke(value);
        }
    }
}
发布者和订阅者通过 Inspector 拖拽同一个 ScriptableObject 资源来建立连接：


using UnityEngine;

public class PlayerHealth : MonoBehaviour
{
    [SerializeField] private FloatEventChannel onHealthChanged;
    [SerializeField] private VoidEventChannel onPlayerDeath;
    [SerializeField] private float maxHealth = 100f;

    private float currentHealth;

    private void Awake()
    {
        currentHealth = maxHealth;
    }

    public void TakeDamage(float damage)
    {
        currentHealth = Mathf.Max(0f, currentHealth - damage);
        onHealthChanged.Raise(currentHealth / maxHealth);

        if (currentHealth <= 0f)
        {
            onPlayerDeath.Raise();
        }
    }
}

public class GameOverScreen : MonoBehaviour
{
    [SerializeField] private VoidEventChannel onPlayerDeath;
    [SerializeField] private GameObject gameOverPanel;

    private void OnEnable()
    {
        onPlayerDeath.Register(ShowGameOver);
    }

    private void OnDisable()
    {
        onPlayerDeath.Unregister(ShowGameOver);
    }

    private void ShowGameOver()
    {
        gameOverPanel.SetActive(true);
        Time.timeScale = 0f;
    }
}
小结
    

  • C# delegate/event 是事件系统的基础，提供强类型安全和高性能的组件级通信
    • 

  • UnityEvent 支持 Inspector 序列化绑定，适合设计师配置驱动的事件响应
    • 

  • 静态事件总线通过泛型实现类型安全的全局事件广播，适合大型项目跨系统通信
    • 

  • ScriptableObject 事件通道结合了解耦能力和可视化优势，是官方推荐的事件架构
    • 

  • 所有事件订阅必须在 OnDisable/OnDestroy 中取消，防止空引用异常和内存泄漏
    • 

  • 根据项目规模和团队组成选择合适的事件方案，小项目用 delegate，大项目用事件总线或 SO 通道
    •
