UI 动画

核心概念

UI 动画是提升游戏交互体验的关键要素，包括面板滑入滑出、按钮悬停反馈、数值滚动变化、弹窗缩放弹出等视觉效果。Unity 提供了多种实现 UI 动画的方案，从内置的 Animator 组件到代码驱动的 Tween 动画，每种方案在灵活性、性能和开发效率上各有侧重。

Animator 驱动的 UI 动画
Animator 组件可以直接驱动 UI 元素的 RectTransform、CanvasGroup 等属性。通过 Animation 窗口录制关键帧，在 Animator Controller 中组织状态机，实现复杂的 UI 动画序列。这种方式对设计师友好，动画调节无需修改代码。


using UnityEngine;

public class PanelAnimator : MonoBehaviour
{
    [SerializeField] private Animator animator;
    [SerializeField] private CanvasGroup canvasGroup;

    private static readonly int ShowHash = Animator.StringToHash("Show");
    private static readonly int HideHash = Animator.StringToHash("Hide");
    private static readonly int IsVisibleHash = Animator.StringToHash("IsVisible");

    private bool isVisible;

    public void Show()
    {
        if (isVisible) return;
        isVisible = true;

        gameObject.SetActive(true);
        canvasGroup.interactable = true;
        canvasGroup.blocksRaycasts = true;
        animator.SetBool(IsVisibleHash, true);
        animator.SetTrigger(ShowHash);
    }

    public void Hide()
    {
        if (!isVisible) return;
        isVisible = false;

        canvasGroup.interactable = false;
        canvasGroup.blocksRaycasts = false;
        animator.SetBool(IsVisibleHash, false);
        animator.SetTrigger(HideHash);
    }

    public void OnHideAnimationComplete()
    {
        gameObject.SetActive(false);
    }
}
Animator 控制 UI 动画时，建议将 Animator 的 Update Mode 设置为 Unscaled Time，这样在游戏暂停（Time.timeScale = 0）时 UI 动画仍然能正常播放，常见于暂停菜单和设置界面。

代码驱动的补间动画
补间动画（Tween）通过代码在两个值之间进行平滑插值，是 UI 动画最灵活的实现方式。无论使用第三方库还是自己实现，核心原理都是在 Update 或协程中对属性进行插值计算。


using UnityEngine;
using System;
using System.Collections;

public class UITweenEngine : MonoBehaviour
{
    public static UITweenEngine Instance { get; private set; }

    private void Awake()
    {
        Instance = this;
    }

    public Coroutine TweenFloat(float from, float to, float duration,
        Action<float> onUpdate, Action onComplete = null,
        Func<float, float> easeFunc = null)
    {
        return StartCoroutine(TweenRoutine(from, to, duration, onUpdate, onComplete, easeFunc));
    }

    private IEnumerator TweenRoutine(float from, float to, float duration,
        Action<float> onUpdate, Action onComplete,
        Func<float, float> easeFunc)
    {
        float elapsed = 0f;
        easeFunc ??= EaseInOutCubic;

        while (elapsed < duration)
        {
            elapsed += Time.unscaledDeltaTime;
            float t = Mathf.Clamp01(elapsed / duration);
            float easedT = easeFunc(t);
            float value = Mathf.LerpUnclamped(from, to, easedT);
            onUpdate(value);
            yield return null;
        }

        onUpdate(to);
        onComplete?.Invoke();
    }

    public static float EaseInOutCubic(float t)
    {
        return t < 0.5f ? 4f * t * t * t : 1f - Mathf.Pow(-2f * t + 2f, 3f) / 2f;
    }

    public static float EaseOutBack(float t)
    {
        const float c1 = 1.70158f;
        const float c3 = c1 + 1f;
        return 1f + c3 * Mathf.Pow(t - 1f, 3f) + c1 * Mathf.Pow(t - 1f, 2f);
    }

    public static float EaseOutElastic(float t)
    {
        if (t <= 0f) return 0f;
        if (t >= 1f) return 1f;
        const float c4 = 2f * Mathf.PI / 3f;
        return Mathf.Pow(2f, -10f * t) * Mathf.Sin((t * 10f - 0.75f) * c4) + 1f;
    }
}
使用补间引擎实现常见的 UI 动画效果：


using UnityEngine;
using UnityEngine.UI;

public class UIAnimations : MonoBehaviour
{
    [SerializeField] private RectTransform panel;
    [SerializeField] private CanvasGroup canvasGroup;
    [SerializeField] private Text scoreText;

    public void SlideIn()
    {
        panel.anchoredPosition = new Vector2(-Screen.width, 0);
        canvasGroup.alpha = 0f;

        UITweenEngine.Instance.TweenFloat(
            -Screen.width, 0f, 0.4f,
            v => panel.anchoredPosition = new Vector2(v, 0f),
            easeFunc: UITweenEngine.EaseOutBack
        );

        UITweenEngine.Instance.TweenFloat(
            0f, 1f, 0.3f,
            v => canvasGroup.alpha = v
        );
    }

    public void PopIn()
    {
        panel.localScale = Vector3.zero;

        UITweenEngine.Instance.TweenFloat(
            0f, 1f, 0.5f,
            v => panel.localScale = Vector3.one * v,
            easeFunc: UITweenEngine.EaseOutElastic
        );
    }

    public void AnimateScore(int from, int to)
    {
        UITweenEngine.Instance.TweenFloat(
            from, to, 1f,
            v => scoreText.text = Mathf.RoundToInt(v).ToString("N0"),
            easeFunc: UITweenEngine.EaseInOutCubic
        );
    }
}
CanvasGroup 与批量控制
CanvasGroup 组件可以批量控制其下所有子元素的透明度、交互性和射线检测。它是实现面板淡入淡出的最简单方式，不需要遍历修改每个子元素的 alpha 值。


using UnityEngine;
using System.Collections;

public class FadePanel : MonoBehaviour
{
    [SerializeField] private CanvasGroup canvasGroup;
    [SerializeField] private float fadeDuration = 0.3f;

    private Coroutine fadeCoroutine;

    public void FadeIn()
    {
        gameObject.SetActive(true);
        StartFade(0f, 1f, () =>
        {
            canvasGroup.interactable = true;
            canvasGroup.blocksRaycasts = true;
        });
    }

    public void FadeOut()
    {
        canvasGroup.interactable = false;
        canvasGroup.blocksRaycasts = false;
        StartFade(1f, 0f, () => gameObject.SetActive(false));
    }

    private void StartFade(float from, float to, System.Action onComplete)
    {
        if (fadeCoroutine != null) StopCoroutine(fadeCoroutine);
        fadeCoroutine = StartCoroutine(FadeRoutine(from, to, onComplete));
    }

    private IEnumerator FadeRoutine(float from, float to, System.Action onComplete)
    {
        float elapsed = 0f;
        canvasGroup.alpha = from;

        while (elapsed < fadeDuration)
        {
            elapsed += Time.unscaledDeltaTime;
            canvasGroup.alpha = Mathf.Lerp(from, to, elapsed / fadeDuration);
            yield return null;
        }

        canvasGroup.alpha = to;
        onComplete?.Invoke();
    }
}
CanvasGroup 的嵌套规则：子 CanvasGroup 的最终 alpha 是父子 alpha 值的乘积。利用这一特性可以实现层级化的透明度控制，如整个面板淡出时内部元素同步变化。

UI 状态机与动画编排
复杂的 UI 系统通常需要管理多个面板之间的切换动画。通过状态机模式编排动画序列，确保切换过程中新面板的进入动画和旧面板的退出动画正确衔接。


using UnityEngine;
using System.Collections;
using System.Collections.Generic;

public class UIPageManager : MonoBehaviour
{
    [System.Serializable]
    public class PageConfig
    {
        public string pageName;
        public CanvasGroup canvasGroup;
        public RectTransform rectTransform;
    }

    [SerializeField] private List<PageConfig> pages;
    [SerializeField] private float transitionDuration = 0.3f;
    [SerializeField] private float slideOffset = 300f;

    private int currentPageIndex = -1;
    private bool isTransitioning;

    public void SwitchToPage(int index)
    {
        if (isTransitioning || index == currentPageIndex) return;
        if (index < 0 || index >= pages.Count) return;

        StartCoroutine(TransitionRoutine(currentPageIndex, index));
    }

    private IEnumerator TransitionRoutine(int fromIndex, int toIndex)
    {
        isTransitioning = true;

        var toPage = pages[toIndex];
        toPage.canvasGroup.gameObject.SetActive(true);
        toPage.canvasGroup.alpha = 0f;
        toPage.rectTransform.anchoredPosition = new Vector2(slideOffset, 0f);

        if (fromIndex >= 0)
        {
            var fromPage = pages[fromIndex];
            yield return StartCoroutine(AnimatePages(fromPage, toPage));
            fromPage.canvasGroup.gameObject.SetActive(false);
        }
        else
        {
            yield return StartCoroutine(AnimateIn(toPage));
        }

        toPage.canvasGroup.interactable = true;
        toPage.canvasGroup.blocksRaycasts = true;
        currentPageIndex = toIndex;
        isTransitioning = false;
    }

    private IEnumerator AnimatePages(PageConfig from, PageConfig to)
    {
        from.canvasGroup.interactable = false;
        from.canvasGroup.blocksRaycasts = false;

        float elapsed = 0f;

        while (elapsed < transitionDuration)
        {
            elapsed += Time.unscaledDeltaTime;
            float t = Mathf.Clamp01(elapsed / transitionDuration);
            float smooth = t * t * (3f - 2f * t);

            from.canvasGroup.alpha = 1f - smooth;
            from.rectTransform.anchoredPosition = new Vector2(-slideOffset * smooth, 0f);

            to.canvasGroup.alpha = smooth;
            to.rectTransform.anchoredPosition = new Vector2(slideOffset * (1f - smooth), 0f);

            yield return null;
        }

        from.canvasGroup.alpha = 0f;
        to.canvasGroup.alpha = 1f;
        to.rectTransform.anchoredPosition = Vector2.zero;
    }

    private IEnumerator AnimateIn(PageConfig page)
    {
        float elapsed = 0f;

        while (elapsed < transitionDuration)
        {
            elapsed += Time.unscaledDeltaTime;
            float t = Mathf.Clamp01(elapsed / transitionDuration);
            float smooth = t * t * (3f - 2f * t);

            page.canvasGroup.alpha = smooth;
            page.rectTransform.anchoredPosition = new Vector2(slideOffset * (1f - smooth), 0f);

            yield return null;
        }

        page.canvasGroup.alpha = 1f;
        page.rectTransform.anchoredPosition = Vector2.zero;
    }
}
小结
    

  • Animator 适合复杂的多步骤 UI 动画序列，设计师可通过 Animation 窗口直接编辑
    • 

  • 代码驱动的补间动画灵活性最高，缓动函数决定动画的运动质感
    • 

  • CanvasGroup 是批量控制 UI 透明度和交互性的核心组件，实现淡入淡出最简单
    • 

  • UI 状态机模式确保多面板切换动画的正确衔接，防止快速切换导致的状态混乱
    • 

  • 暂停菜单等 UI 动画需使用 Unscaled Time 模式，不受 Time.timeScale 影响
    • 

  • 动画缓存和复用策略可以降低 GC 压力，避免在动画过程中频繁分配内存
