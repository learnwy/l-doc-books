UI Toolkit (Unity 现代 UI 框架)

核心概念

UI Toolkit 是 Unity 推出的新一代 UI 系统，采用类似 Web 前端的开发模式，使用 UXML（类似 HTML）定义结构、USS（类似 CSS）定义样式，并通过 C# 代码处理逻辑。与 UGUI 的 GameObject 驱动方式不同，UI Toolkit 使用轻量级的 VisualElement 树结构，不依赖场景对象，渲染效率更高。UI Toolkit 目前同时支持编辑器扩展和运行时 UI。

UXML 文档与元素结构
UXML 是 UI Toolkit 的标记语言，类似于 HTML 但使用 Unity 的元素类型系统。每个 UXML 文件定义一个 UI 片段的结构，可以嵌套引用其他 UXML 模板实现复用。


UXML 文档中可以使用内置控件（Button、Label、TextField 等）和自定义 VisualElement 类型。通过 name 属性给元素命名，通过 class 属性添加 USS 样式类：


using UnityEngine;
using UnityEngine.UIElements;

public class MainMenuController : MonoBehaviour
{
    [SerializeField] private UIDocument uiDocument;

    private Button startButton;
    private Button settingsButton;
    private Button quitButton;
    private Label titleLabel;

    private void OnEnable()
    {
        var root = uiDocument.rootVisualElement;

        titleLabel = root.Q<Label>("title-label");
        startButton = root.Q<Button>("start-button");
        settingsButton = root.Q<Button>("settings-button");
        quitButton = root.Q<Button>("quit-button");

        startButton.clicked += OnStartClicked;
        settingsButton.clicked += OnSettingsClicked;
        quitButton.clicked += OnQuitClicked;
    }

    private void OnDisable()
    {
        startButton.clicked -= OnStartClicked;
        settingsButton.clicked -= OnSettingsClicked;
        quitButton.clicked -= OnQuitClicked;
    }

    private void OnStartClicked()
    {
        Debug.Log("开始游戏");
    }

    private void OnSettingsClicked()
    {
        Debug.Log("打开设置");
    }

    private void OnQuitClicked()
    {
        Application.Quit();
    }
}
元素查询 API 提供了强大的选择能力。Q<T>() 按名称查找单个元素，Query<T>() 返回查询构建器支持链式过滤，类似 CSS 选择器的查询方式：


using UnityEngine.UIElements;

public static class UIQueryExtensions
{
    public static void SetupListView(VisualElement root, string listName)
    {
        var list = root.Q<ListView>(listName);
        var allButtons = root.Query<Button>().ToList();
        var activeItems = root.Query<VisualElement>(className: "item-active").ToList();
        var firstLabel = root.Q<Label>();
        var namedContainer = root.Q<VisualElement>("main-container");
    }
}
USS 样式系统
USS（Unity Style Sheets）是 UI Toolkit 的样式语言，语法高度接近 CSS，支持选择器、伪类和变量。USS 将视觉表现与结构分离，让同一套 UXML 可以通过切换 USS 实现完全不同的视觉主题。


USS 支持的选择器类型包括类型选择器、类选择器、名称选择器和伪类选择器：

USS 变量系统允许定义全局样式变量，方便统一管理颜色、字体等设计令牌，实现主题切换：


using UnityEngine;
using UnityEngine.UIElements;

public class ThemeManager : MonoBehaviour
{
    [SerializeField] private UIDocument uiDocument;
    [SerializeField] private StyleSheet darkTheme;
    [SerializeField] private StyleSheet lightTheme;

    private bool isDarkMode = true;

    public void ToggleTheme()
    {
        var root = uiDocument.rootVisualElement;

        if (isDarkMode)
        {
            root.styleSheets.Remove(darkTheme);
            root.styleSheets.Add(lightTheme);
        }
        else
        {
            root.styleSheets.Remove(lightTheme);
            root.styleSheets.Add(darkTheme);
        }

        isDarkMode = !isDarkMode;
    }
}
通过 C# 也可以直接操作元素的内联样式，适合动态计算的样式值：


using UnityEngine;
using UnityEngine.UIElements;

public class DynamicStyleController : MonoBehaviour
{
    public void AnimateElement(VisualElement element, float progress)
    {
        element.style.opacity = progress;
        element.style.translate = new Translate(0, Length.Percent((1f - progress) * 20f));
        element.style.scale = new Scale(new Vector2(progress, progress));
    }

    public void SetHealthBarColor(VisualElement bar, float healthPercent)
    {
        Color color = healthPercent > 0.5f
            ? Color.Lerp(Color.yellow, Color.green, (healthPercent - 0.5f) * 2f)
            : Color.Lerp(Color.red, Color.yellow, healthPercent * 2f);

        bar.style.backgroundColor = color;
        bar.style.width = Length.Percent(healthPercent * 100f);
    }
}
Flexbox 布局模型
UI Toolkit 采用 Flexbox 布局模型，与 Web CSS Flexbox 基本一致。通过设置容器的 flex-direction、justify-content、align-items 等属性，子元素自动排列和分配空间，无需手动计算位置。

利用 Flexbox 构建响应式布局的 C# 示例：


using UnityEngine;
using UnityEngine.UIElements;

public class FlexLayoutBuilder : MonoBehaviour
{
    [SerializeField] private UIDocument uiDocument;

    private void OnEnable()
    {
        var root = uiDocument.rootVisualElement;
        BuildInventoryGrid(root);
    }

    private void BuildInventoryGrid(VisualElement root)
    {
        var container = new VisualElement();
        container.style.flexDirection = FlexDirection.Row;
        container.style.flexWrap = Wrap.Wrap;
        container.style.justifyContent = Justify.FlexStart;
        container.style.alignItems = Align.FlexStart;
        container.style.paddingTop = 10;
        container.style.paddingLeft = 10;

        for (int i = 0; i < 24; i++)
        {
            var slot = CreateSlot(i);
            container.Add(slot);
        }

        root.Add(container);
    }

    private VisualElement CreateSlot(int index)
    {
        var slot = new VisualElement();
        slot.style.width = 80;
        slot.style.height = 80;
        slot.style.marginRight = 8;
        slot.style.marginBottom = 8;
        slot.style.borderBottomWidth = 2;
        slot.style.borderTopWidth = 2;
        slot.style.borderLeftWidth = 2;
        slot.style.borderRightWidth = 2;
        slot.style.borderBottomColor = Color.gray;
        slot.style.borderTopColor = Color.gray;
        slot.style.borderLeftColor = Color.gray;
        slot.style.borderRightColor = Color.gray;

        var label = new Label($"{index + 1}");
        label.style.unityTextAlign = TextAnchor.MiddleCenter;
        label.style.fontSize = 14;
        slot.Add(label);

        return slot;
    }
}
ListView 与数据绑定
ListView 是 UI Toolkit 中处理大量数据的核心控件，内置了虚拟化（Virtualization）机制——只为可见区域创建 VisualElement，滚动时复用元素并重新绑定数据，极大降低了大列表的内存和性能开销。


using UnityEngine;
using UnityEngine.UIElements;
using System.Collections.Generic;

public class ItemListController : MonoBehaviour
{
    [SerializeField] private UIDocument uiDocument;

    private ListView listView;
    private List<ItemData> items;

    private void OnEnable()
    {
        items = GenerateTestItems(1000);

        var root = uiDocument.rootVisualElement;
        listView = root.Q<ListView>("item-list");

        listView.itemsSource = items;
        listView.fixedItemHeight = 60;
        listView.makeItem = MakeItem;
        listView.bindItem = BindItem;
        listView.selectionType = SelectionType.Single;
        listView.selectionChanged += OnSelectionChanged;
    }

    private VisualElement MakeItem()
    {
        var container = new VisualElement();
        container.style.flexDirection = FlexDirection.Row;
        container.style.alignItems = Align.Center;
        container.style.paddingLeft = 10;
        container.style.paddingRight = 10;
        container.style.height = 60;

        var icon = new VisualElement { name = "icon" };
        icon.style.width = 40;
        icon.style.height = 40;
        icon.style.marginRight = 10;

        var info = new VisualElement();
        info.style.flexGrow = 1;

        var nameLabel = new Label { name = "item-name" };
        nameLabel.style.fontSize = 16;

        var descLabel = new Label { name = "item-desc" };
        descLabel.style.fontSize = 12;

        info.Add(nameLabel);
        info.Add(descLabel);
        container.Add(icon);
        container.Add(info);

        return container;
    }

    private void BindItem(VisualElement element, int index)
    {
        var item = items[index];
        element.Q<Label>("item-name").text = item.Name;
        element.Q<Label>("item-desc").text = item.Description;
    }

    private void OnSelectionChanged(IEnumerable<object> selectedItems)
    {
        foreach (var item in selectedItems)
        {
            Debug.Log($"选中: {((ItemData)item).Name}");
        }
    }

    private List<ItemData> GenerateTestItems(int count)
    {
        var result = new List<ItemData>(count);
        for (int i = 0; i < count; i++)
        {
            result.Add(new ItemData($"物品 {i + 1}", $"这是第 {i + 1} 个物品的描述"));
        }
        return result;
    }
}

public class ItemData
{
    public string Name { get; }
    public string Description { get; }

    public ItemData(string name, string description)
    {
        Name = name;
        Description = description;
    }
}
小结
    

  • UI Toolkit 采用 UXML + USS + C# 三层分离架构，类似 Web 前端的开发模式
    • 

  • VisualElement 树结构比 UGUI 的 GameObject 方式更轻量，适合复杂 UI 场景
    • 

  • USS 样式系统支持选择器、伪类和变量，可实现主题切换和样式复用
    • 

  • Flexbox 布局模型提供了强大的自动排列能力，无需手动计算位置
    • 

  • ListView 内置虚拟化机制，通过 makeItem/bindItem 模式高效处理大量数据
    • 

  • UI Toolkit 适合 2D 界面和编辑器扩展，3D 世界 UI 仍推荐使用 UGUI
    •
