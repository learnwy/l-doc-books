多平台发布

核心概念

Unity 的核心价值主张之一是"一次开发，多平台发布"。通过统一的 API 抽象层和平台适配编译器，同一套 C# 代码可以部署到 iOS、Android、PC、主机、WebGL 等 20+ 个平台。但"多平台"不等于"零适配"——不同平台在图形 API、输入方式、文件系统、性能特征和商店要求上存在显著差异，需要针对性的配置和优化。

平台预编译指令
Unity 通过平台预编译指令（Platform Define Symbols）实现平台特定代码的条件编译。这是处理平台差异最直接的方式，编译时不满足条件的代码会被完全排除，不产生任何运行时开销。


常用的内置平台定义包括 UNITY_IOS、UNITY_ANDROID、UNITY_WEBGL、UNITY_STANDALONE 等。此外可以通过 PlayerSettings 设置自定义脚本定义符号，用于区分发行渠道、服务器环境等。


public static class PlatformAdapter
{
    public static string GetPersistentPath(string filename)
    {
#if UNITY_IOS
        return Path.Combine(Application.persistentDataPath, filename);
#elif UNITY_ANDROID
        return Path.Combine(Application.persistentDataPath, filename);
#elif UNITY_WEBGL
        return $"/idbfs/{filename}";
#else
        return Path.Combine(Application.persistentDataPath, filename);
#endif
    }

    public static void InitializeAds()
    {
#if UNITY_IOS
        ATTrackingStatusBinding.RequestAuthorizationTracking();
        Advertisement.Initialize("ios_game_id", false);
#elif UNITY_ANDROID
        Advertisement.Initialize("android_game_id", false);
#endif
    }

    public static int GetTargetFrameRate()
    {
#if UNITY_IOS || UNITY_ANDROID
        return 60;
#elif UNITY_WEBGL
        return -1;
#else
        return 120;
#endif
    }

    public static void ConfigureQuality()
    {
#if UNITY_IOS || UNITY_ANDROID
        QualitySettings.vSyncCount = 0;
        QualitySettings.antiAliasing = 2;
        QualitySettings.shadows = ShadowQuality.HardOnly;
        QualitySettings.shadowResolution = ShadowResolution.Low;
#else
        QualitySettings.vSyncCount = 1;
        QualitySettings.antiAliasing = 4;
        QualitySettings.shadows = ShadowQuality.All;
        QualitySettings.shadowResolution = ShadowResolution.High;
#endif
    }
}
过度使用预编译指令会导致代码可读性下降。推荐的做法是将平台差异封装在适配器层（Adapter Pattern），上层业务代码通过接口调用，不直接依赖平台宏。


图形 API 适配
不同平台支持不同的图形 API，Unity 允许为每个平台配置首选图形 API 列表。选择正确的图形 API 对渲染性能和功能支持至关重要。Metal 在 iOS/macOS 上是唯一选择；Android 推荐 Vulkan（性能优于 GLES3，但兼容性略差）；PC 平台 DX12 支持更先进的功能但部分旧显卡不兼容。


URP（Universal Render Pipeline）是多平台项目的推荐选择，它在所有平台上提供一致的渲染效果和性能。通过 URP Asset 可以按平台配置不同的渲染参数。


using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

public static class GraphicsAdaptation
{
    public static void ApplyPlatformGraphics(UniversalRenderPipelineAsset urpAsset)
    {
        if (SystemInfo.graphicsDeviceType == GraphicsDeviceType.Metal)
        {
            urpAsset.renderScale = 1.0f;
            urpAsset.msaaSampleCount = 4;
            urpAsset.supportsCameraDepthTexture = true;
        }
        else if (SystemInfo.graphicsDeviceType == GraphicsDeviceType.Vulkan)
        {
            urpAsset.renderScale = 0.85f;
            urpAsset.msaaSampleCount = 2;
            urpAsset.supportsCameraDepthTexture = true;
        }
        else if (SystemInfo.graphicsDeviceType == GraphicsDeviceType.OpenGLES3)
        {
            urpAsset.renderScale = 0.75f;
            urpAsset.msaaSampleCount = 0;
            urpAsset.supportsCameraDepthTexture = false;
        }
    }

    public static bool SupportsComputeShaders()
    {
        return SystemInfo.supportsComputeShaders &&
               SystemInfo.graphicsDeviceType != GraphicsDeviceType.OpenGLES3;
    }

    public static int GetMaxTextureSize()
    {
        int maxSize = SystemInfo.maxTextureSize;
        long memoryMB = SystemInfo.systemMemorySize;

        if (memoryMB < 3072)
            return Mathf.Min(maxSize, 1024);
        if (memoryMB < 6144)
            return Mathf.Min(maxSize, 2048);
        return Mathf.Min(maxSize, 4096);
    }
}
输入系统跨平台处理
不同平台的输入方式差异巨大：移动端是触摸，PC 是键鼠和手柄，主机是手柄，WebGL 可能是键鼠或触摸。Unity 的 New Input System 提供了统一的输入抽象，通过 Input Action Assets 定义逻辑动作，在运行时自动映射到当前平台的物理输入设备。


using UnityEngine;
using UnityEngine.InputSystem;
using UnityEngine.InputSystem.EnhancedTouch;

public class CrossPlatformInput : MonoBehaviour
{
    private PlayerControls _controls;

    private void OnEnable()
    {
        _controls = new PlayerControls();
        _controls.Enable();

#if UNITY_IOS || UNITY_ANDROID
        EnhancedTouchSupport.Enable();
        TouchSimulation.Enable();
#endif

        _controls.Gameplay.Fire.performed += OnFire;
        _controls.Gameplay.Move.performed += OnMove;
    }

    private void OnDisable()
    {
        _controls.Gameplay.Fire.performed -= OnFire;
        _controls.Gameplay.Move.performed -= OnMove;
        _controls.Disable();

#if UNITY_IOS || UNITY_ANDROID
        EnhancedTouchSupport.Disable();
#endif
    }

    private void OnFire(InputAction.CallbackContext ctx)
    {
        var device = ctx.control.device;
        if (device is Touchscreen)
            HandleTouchFire();
        else if (device is Mouse)
            HandleMouseFire();
        else if (device is Gamepad)
            HandleGamepadFire();
    }

    private void OnMove(InputAction.CallbackContext ctx)
    {
        Vector2 input = ctx.ReadValue<Vector2>();
        MoveCharacter(input);
    }

    private void HandleTouchFire() { }
    private void HandleMouseFire() { }
    private void HandleGamepadFire() { }
    private void MoveCharacter(Vector2 direction) { }
}

平台特定配置自动化
在多平台项目中，手动在 Player Settings 面板切换配置容易出错且效率低下。通过 BuildPreprocessor 和 EditorScript 实现配置自动化，确保每次构建使用正确的平台参数。


using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;

public class PlatformPreprocessor : IPreprocessBuildWithReport
{
    public int callbackOrder => 0;

    public void OnPreprocessBuild(BuildReport report)
    {
        var target = report.summary.platform;

        switch (target)
        {
            case BuildTarget.iOS:
                PlayerSettings.iOS.targetOSVersionString = "14.0";
                PlayerSettings.iOS.buildNumber =
                    GetBuildNumber().ToString();
                PlayerSettings.SetArchitecture(
                    BuildTargetGroup.iOS, 1);
                break;

            case BuildTarget.Android:
                PlayerSettings.Android.minSdkVersion =
                    AndroidSdkVersions.AndroidApiLevel24;
                PlayerSettings.Android.targetSdkVersion =
                    AndroidSdkVersions.AndroidApiLevel34;
                PlayerSettings.Android.bundleVersionCode =
                    GetBuildNumber();
                break;

            case BuildTarget.WebGL:
                PlayerSettings.WebGL.compressionFormat =
                    WebGLCompressionFormat.Brotli;
                PlayerSettings.WebGL.linkerTarget =
                    WebGLLinkerTarget.Wasm;
                PlayerSettings.WebGL.memorySize = 512;
                break;
        }

        PlayerSettings.bundleVersion = GetVersionString();
    }

    private static int GetBuildNumber() => 1;
    private static string GetVersionString() => "1.0.0";
}
小结
    

  • Unity 支持 20+ 平台，核心差异在图形 API、输入方式、性能特征和商店要求
    • 

  • 平台预编译指令实现条件编译，但应封装在适配器层避免代码污染
    • 

  • URP 是多平台项目的推荐渲染管线，配合分级 URP Asset 适配不同性能设备
    • 

  • New Input System 通过 Input Action Assets 统一抽象，自动映射到各平台输入设备
    • 

  • BuildPreprocessor 实现平台配置自动化，消除手动切换的出错风险
    • 

  • WebGL 平台有特殊限制：无多线程、内存受限、需 Brotli 压缩
    •
