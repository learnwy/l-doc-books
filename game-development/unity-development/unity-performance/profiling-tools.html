性能分析工具

核心概念

性能优化的第一原则是"先测量，后优化"。Unity 提供了一套完整的性能分析工具链，从 CPU/GPU 耗时分析到内存快照、从帧调试到物理诊断，覆盖了游戏开发中所有需要关注的性能维度。正确使用这些工具定位瓶颈，比盲目优化代码要高效得多。

Profiler CPU 模块
CPU Profiler 是最常用的性能分析工具，它以时间轴和层级树两种视图展示每帧中所有方法的执行耗时。通过 Profiler 可以精确定位哪个脚本的哪个方法消耗了最多的 CPU 时间。


Profiler 支持编辑器内分析和真机远程分析。编辑器内分析存在额外开销，真机分析才能反映真实性能。通过 Build Settings 中启用 Development Build 和 Autoconnect Profiler，可以在真机启动时自动连接 Profiler。


using UnityEngine;
using UnityEngine.Profiling;

public class ProfilerExample : MonoBehaviour
{
    private void Update()
    {
        Profiler.BeginSample("AI Decision");
        MakeAIDecision();
        Profiler.EndSample();

        Profiler.BeginSample("Path Finding");
        CalculatePath();
        Profiler.EndSample();

        Profiler.BeginSample("Animation Update");
        UpdateAnimations();
        Profiler.EndSample();
    }

    private void MakeAIDecision()
    {
        System.Threading.Thread.Sleep(1);
    }

    private void CalculatePath()
    {
        System.Threading.Thread.Sleep(2);
    }

    private void UpdateAnimations()
    {
        System.Threading.Thread.Sleep(1);
    }
}
Profiler.BeginSample / Profiler.EndSample 在 Profiler 的层级树中创建自定义标记，让开发者可以清楚看到每段逻辑的耗时。在发布版本中这些调用会被自动剥离，不产生运行时开销。

重点关注的指标：Total（含子调用的总耗时）和 Self（排除子调用的自身耗时）。Self 高说明方法本身有性能问题，Total 高但 Self 低说明问题在子方法中。


Profiler Memory 与 GC 分析
Memory 模块追踪堆内存的分配和回收。Unity 使用增量式垃圾回收（Incremental GC），但 GC 仍然会造成帧率波动。Memory 模块帮助识别哪些代码在运行时频繁分配堆内存。


常见的 GC 分配陷阱和优化方案：


using UnityEngine;
using System.Collections.Generic;
using System.Text;

public class GCOptimizationExamples : MonoBehaviour
{
    private readonly List<Collider> hitResults = new(32);
    private readonly Collider[] overlapBuffer = new Collider[64];
    private readonly StringBuilder textBuilder = new(128);
    private readonly RaycastHit[] raycastBuffer = new RaycastHit[16];

    private void Update()
    {
        CheckNearbyEnemies();
        UpdateDisplayText();
        PerformRaycast();
    }

    private void CheckNearbyEnemies()
    {
        int count = Physics.OverlapSphereNonAlloc(
            transform.position, 10f, overlapBuffer
        );

        hitResults.Clear();
        for (int i = 0; i < count; i++)
        {
            if (overlapBuffer[i].CompareTag("Enemy"))
            {
                hitResults.Add(overlapBuffer[i]);
            }
        }
    }

    private void UpdateDisplayText()
    {
        textBuilder.Clear();
        textBuilder.Append("HP: ");
        textBuilder.Append(100);
        textBuilder.Append(" / ");
        textBuilder.Append(100);
    }

    private void PerformRaycast()
    {
        int hits = Physics.RaycastNonAlloc(
            transform.position, transform.forward,
            raycastBuffer, 100f
        );

        for (int i = 0; i < hits; i++)
        {
            ProcessHit(raycastBuffer[i]);
        }
    }

    private void ProcessHit(RaycastHit hit)
    {
        Debug.Log(hit.collider.name);
    }
}

Frame Debugger 渲染分析
Frame Debugger 以逐步回放的方式展示当前帧的所有 Draw Call，让开发者清楚看到每个渲染批次绘制了什么、为什么与前一批次无法合并。这是优化 Draw Call 数量的核心工具。


using UnityEngine;
using UnityEngine.Rendering;

public class RenderStatsMonitor : MonoBehaviour
{
    [SerializeField] private bool enableStatsOverlay;

    private int lastBatchCount;
    private int lastSetPassCalls;
    private int lastTriangleCount;

    private void Update()
    {
        if (!enableStatsOverlay) return;

        int currentBatches = UnityEngine.Rendering.RenderPipelineManager
            .currentPipeline != null ? 0 : 0;

        RecordStats();
    }

    private void RecordStats()
    {
        Profiler.BeginSample("Render Stats Collection");

        int triangles = UnityStats.triangles;
        int vertices = UnityStats.vertices;
        int batches = UnityStats.batches;
        int setPassCalls = UnityStats.setPassCalls;

        if (batches != lastBatchCount || setPassCalls != lastSetPassCalls)
        {
            lastBatchCount = batches;
            lastSetPassCalls = setPassCalls;
            lastTriangleCount = triangles;
        }

        Profiler.EndSample();
    }

    public string GetStatsReport()
    {
        return $"Batches: {lastBatchCount}, " +
               $"SetPass: {lastSetPassCalls}, " +
               $"Tris: {lastTriangleCount}";
    }
}

自定义性能监控面板
在开发阶段创建一个实时性能监控面板，可以在运行时随时观察关键性能指标，无需每次打开 Profiler：


using UnityEngine;
using System.Text;

public class PerformanceOverlay : MonoBehaviour
{
    [SerializeField] private bool showOverlay = true;
    [SerializeField] private int fontSize = 14;

    private float deltaTime;
    private float updateInterval = 0.5f;
    private float lastUpdateTime;
    private int frameCount;
    private float currentFPS;
    private float minFPS = float.MaxValue;
    private float maxFPS;
    private readonly StringBuilder sb = new(256);

    private GUIStyle style;

    private void Update()
    {
        deltaTime += (Time.unscaledDeltaTime - deltaTime) * 0.1f;
        frameCount++;

        if (Time.unscaledTime - lastUpdateTime >= updateInterval)
        {
            currentFPS = frameCount / (Time.unscaledTime - lastUpdateTime);
            frameCount = 0;
            lastUpdateTime = Time.unscaledTime;

            if (currentFPS < minFPS && currentFPS > 0) minFPS = currentFPS;
            if (currentFPS > maxFPS) maxFPS = currentFPS;
        }
    }

    private void OnGUI()
    {
        if (!showOverlay) return;

        style ??= new GUIStyle(GUI.skin.label)
        {
            fontSize = fontSize,
            fontStyle = FontStyle.Bold
        };
        style.normal.textColor = currentFPS < 30 ? Color.red :
                                  currentFPS < 60 ? Color.yellow : Color.green;

        float msec = deltaTime * 1000f;
        long totalMemory = UnityEngine.Profiling.Profiler.GetTotalAllocatedMemoryLong();
        long usedMemory = UnityEngine.Profiling.Profiler.GetTotalReservedMemoryLong();

        sb.Clear();
        sb.AppendLine($"FPS: {currentFPS:F1} ({msec:F1}ms)");
        sb.AppendLine($"Min: {minFPS:F1} / Max: {maxFPS:F1}");
        sb.AppendLine($"Alloc: {totalMemory / (1024 * 1024)}MB");
        sb.AppendLine($"Reserved: {usedMemory / (1024 * 1024)}MB");

        GUI.Label(new Rect(10, 10, 300, 100), sb.ToString(), style);
    }

    public void ResetStats()
    {
        minFPS = float.MaxValue;
        maxFPS = 0;
    }
}
小结
    

  • 性能优化的第一原则是"先测量后优化"，盲目优化往往事倍功半
    • 

  • CPU Profiler 的 Self 和 Total 时间区分方法自身和子调用的耗时
    • 

  • Memory 模块配合 NonAlloc API 和 StringBuilder 消除运行时 GC 分配
    • 

  • Frame Debugger 逐步展示 Draw Call，精准定位批处理断裂原因
    • 

  • 自定义性能监控面板提供运行时实时诊断，减少对 Profiler 窗口的依赖
    • 

  • 始终在真机上进行最终性能验证，编辑器内的分析结果存在额外开销误差
