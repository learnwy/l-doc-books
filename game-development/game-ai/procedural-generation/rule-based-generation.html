基于规则的生成

核心概念

基于规则的生成通过形式化的规则系统控制程序化内容的生成过程。与纯随机方法不同，规则系统允许设计师精确定义生成结果必须满足的约束条件和结构模式，从而在保持多样性的同时确保内容质量可控。这类方法将生成问题转化为规则推导或约束求解问题，是连接人工设计与自动生成的桥梁。

L-System 植被与分形生成
L-System（Lindenmayer System）是一种基于字符串重写的形式文法系统。它通过一组起始公理和重写规则，迭代生成越来越复杂的字符串序列，然后由解释器将字符串转化为几何图形。L-System 在模拟植物生长和分形结构方面表现出色。


一个基本的 L-System 由三部分组成：字母表（可用符号集合）、公理（初始字符串）和产生式规则（符号替换规则）。通过引入随机产生式和参数化符号，可以生成更自然的变体。


interface LSystemRule {
  predecessor: string;
  successors: Array<{ weight: number; replacement: string }>;
}

interface LSystem {
  axiom: string;
  rules: LSystemRule[];
  iterations: number;
}

function expandLSystem(system: LSystem, rng: () => number): string {
  let current = system.axiom;

  for (let i = 0; i < system.iterations; i++) {
    let next = '';
    for (const char of current) {
      const rule = system.rules.find((r) => r.predecessor === char);
      if (rule) {
        const totalWeight = rule.successors.reduce((s, e) => s + e.weight, 0);
        let roll = rng() * totalWeight;
        for (const successor of rule.successors) {
          roll -= successor.weight;
          if (roll <= 0) {
            next += successor.replacement;
            break;
          }
        }
      } else {
        next += char;
      }
    }
    current = next;
  }

  return current;
}

interface TurtleState {
  x: number;
  y: number;
  angle: number;
}

interface LineSegment {
  x1: number;
  y1: number;
  x2: number;
  y2: number;
  depth: number;
}

function interpretLSystem(sequence: string, stepLength: number, turnAngle: number): LineSegment[] {
  const segments: LineSegment[] = [];
  const stack: TurtleState[] = [];
  let state: TurtleState = { x: 0, y: 0, angle: -90 };
  let depth = 0;

  for (const char of sequence) {
    switch (char) {
      case 'F': {
        const rad = (state.angle * Math.PI) / 180;
        const nx = state.x + Math.cos(rad) * stepLength;
        const ny = state.y + Math.sin(rad) * stepLength;
        segments.push({
          x1: state.x,
          y1: state.y,
          x2: nx,
          y2: ny,
          depth,
        });
        state.x = nx;
        state.y = ny;
        break;
      }
      case '+':
        state.angle += turnAngle;
        break;
      case '-':
        state.angle -= turnAngle;
        break;
      case '[':
        stack.push({ ...state });
        depth++;
        break;
      case ']':
        state = stack.pop()!;
        depth--;
        break;
    }
  }

  return segments;
}
经典的植物 L-System 规则示例：公理为 F，规则 F → F[+F]F[-F]F，转角 25°，迭代 4 次即可生成逼真的灌木形态。通过调整转角、添加随机偏移和缩短深层枝干的长度，可以生成从草本到乔木的多种植物外观。


波函数坍缩算法
波函数坍缩（Wave Function Collapse，WFC）算法借鉴了量子力学的概念，通过局部约束传播实现全局一致的内容生成。WFC 特别适合基于瓦片的地图生成，能够从少量示例中学习邻接规则并生成无限大的兼容图案。


WFC 的核心流程分为三步：首先从所有位置中选择熵最低的位置（可选瓦片种类最少的位置），随机坍缩为一种具体瓦片，然后将这个选择的影响传播到相邻位置，移除不兼容的可能性。


interface WFCTile {
  id: number;
  weight: number;
}

interface AdjacencyRule {
  tileA: number;
  tileB: number;
  direction: 'up' | 'down' | 'left' | 'right';
}

interface WFCCell {
  possibilities: Set<number>;
  collapsed: boolean;
  tileId: number;
}

function initializeGrid(width: number, height: number, tileCount: number): WFCCell[][] {
  const grid: WFCCell[][] = [];
  for (let y = 0; y < height; y++) {
    grid[y] = [];
    for (let x = 0; x < width; x++) {
      const possibilities = new Set<number>();
      for (let t = 0; t < tileCount; t++) possibilities.add(t);
      grid[y][x] = { possibilities, collapsed: false, tileId: -1 };
    }
  }
  return grid;
}

function findLowestEntropy(grid: WFCCell[][]): [number, number] | null {
  let minEntropy = Infinity;
  let candidates: Array<[number, number]> = [];

  for (let y = 0; y < grid.length; y++) {
    for (let x = 0; x < grid[y].length; x++) {
      const cell = grid[y][x];
      if (cell.collapsed) continue;
      if (cell.possibilities.size < minEntropy) {
        minEntropy = cell.possibilities.size;
        candidates = [[x, y]];
      } else if (cell.possibilities.size === minEntropy) {
        candidates.push([x, y]);
      }
    }
  }

  if (candidates.length === 0) return null;
  return candidates[Math.floor(Math.random() * candidates.length)];
}

function collapse(grid: WFCCell[][], x: number, y: number, tiles: WFCTile[], rng: () => number): void {
  const cell = grid[y][x];
  const available = [...cell.possibilities];
  const totalWeight = available.reduce((s, id) => s + tiles[id].weight, 0);

  let roll = rng() * totalWeight;
  for (const id of available) {
    roll -= tiles[id].weight;
    if (roll <= 0) {
      cell.tileId = id;
      cell.possibilities = new Set([id]);
      cell.collapsed = true;
      return;
    }
  }
}

function propagate(grid: WFCCell[][], startX: number, startY: number, rules: AdjacencyRule[]): boolean {
  const stack: Array<[number, number]> = [[startX, startY]];
  const dirMap = {
    up: [0, -1],
    down: [0, 1],
    left: [-1, 0],
    right: [1, 0],
  } as const;
  const opposite = {
    up: 'down',
    down: 'up',
    left: 'right',
    right: 'left',
  } as const;

  while (stack.length > 0) {
    const [cx, cy] = stack.pop()!;
    const currentPossibilities = grid[cy][cx].possibilities;

    for (const [dir, [dx, dy]] of Object.entries(dirMap)) {
      const nx = cx + dx;
      const ny = cy + dy;
      if (ny < 0 || ny >= grid.length || nx < 0 || nx >= grid[0].length) continue;

      const neighbor = grid[ny][nx];
      if (neighbor.collapsed) continue;

      const validNeighbors = new Set<number>();
      for (const myTile of currentPossibilities) {
        for (const rule of rules) {
          if (rule.tileA === myTile && rule.direction === dir) {
            validNeighbors.add(rule.tileB);
          }
          if (rule.tileB === myTile && rule.direction === opposite[dir as keyof typeof opposite]) {
            validNeighbors.add(rule.tileA);
          }
        }
      }

      const before = neighbor.possibilities.size;
      const intersection = new Set([...neighbor.possibilities].filter((p) => validNeighbors.has(p)));

      if (intersection.size === 0) return false;

      if (intersection.size < before) {
        neighbor.possibilities = intersection;
        stack.push([nx, ny]);
      }
    }
  }

  return true;
}
WFC 的优势在于只需定义局部约束就能涌现出全局一致的结果。设计师提供一小块示例地图，算法自动提取邻接规则，然后可以生成任意大小的兼容地图。在实际项目中常用的 WFC 变体包括重叠模型（Overlapping Model）和简单瓦片模型（Simple Tiled Model）。


约束满足与关卡结构
约束满足问题（CSP）将关卡生成建模为一组变量在各自域中取值、同时满足所有约束的过程。CSP 方法特别适合生成具有严格逻辑关系的内容，如谜题关卡、锁钥结构和任务依赖图。


锁钥结构是 Metroidvania 和塞尔达类游戏的核心设计模式。通过 CSP 建模，可以自动生成满足依赖关系的锁钥布局：钥匙必须在锁之前可达、Boss 钥匙必须在其他所有钥匙之后获得、某些区域需要特定能力才能进入。


interface CSPVariable {
  name: string;
  domain: string[];
  assigned: string | null;
}

interface CSPConstraint {
  variables: string[];
  check: (assignments: Map<string, string>) => boolean;
}

function backtrackSolve(
  variables: CSPVariable[],
  constraints: CSPConstraint[],
  assignments: Map<string, string>,
  rng: () => number,
): boolean {
  const unassigned = variables.find((v) => v.assigned === null);
  if (!unassigned) return true;

  const shuffledDomain = [...unassigned.domain].sort(() => rng() - 0.5);

  for (const value of shuffledDomain) {
    assignments.set(unassigned.name, value);
    unassigned.assigned = value;

    const consistent = constraints
      .filter((c) => c.variables.includes(unassigned.name))
      .every((c) => {
        const allAssigned = c.variables.every((v) => assignments.has(v));
        return !allAssigned || c.check(assignments);
      });

    if (consistent && backtrackSolve(variables, constraints, assignments, rng)) {
      return true;
    }

    assignments.delete(unassigned.name);
    unassigned.assigned = null;
  }

  return false;
}

interface LockKeyLayout {
  rooms: Array<{ id: string; type: 'normal' | 'key' | 'lock' | 'boss' }>;
  connections: Array<[string, string]>;
}

function generateLockKeyLevel(roomCount: number, keyCount: number, rng: () => number): LockKeyLayout {
  const rooms: CSPVariable[] = [];
  for (let i = 0; i < roomCount; i++) {
    rooms.push({
      name: `room_${i}`,
      domain: ['normal', 'key', 'lock', 'boss'],
      assigned: null,
    });
  }
  rooms[0].domain = ['normal'];
  rooms[0].assigned = 'normal';
  rooms[roomCount - 1].domain = ['boss'];
  rooms[roomCount - 1].assigned = 'boss';

  const constraints: CSPConstraint[] = [
    {
      variables: rooms.map((r) => r.name),
      check: (a) => {
        const keys = [...a.values()].filter((v) => v === 'key').length;
        const locks = [...a.values()].filter((v) => v === 'lock').length;
        return keys === keyCount && locks === keyCount;
      },
    },
  ];

  const assignments = new Map<string, string>();
  assignments.set(rooms[0].name, 'normal');
  assignments.set(rooms[roomCount - 1].name, 'boss');

  backtrackSolve(rooms, constraints, assignments, rng);

  const layout: LockKeyLayout = {
    rooms: rooms.map((r) => ({
      id: r.name,
      type: (r.assigned ?? 'normal') as 'normal' | 'key' | 'lock' | 'boss',
    })),
    connections: [],
  };

  for (let i = 0; i < roomCount - 1; i++) {
    layout.connections.push([rooms[i].name, rooms[i + 1].name]);
  }

  return layout;
}
图文法（Graph Grammar）是另一种基于规则的关卡结构生成方法。它在图结构上定义替换规则——将一个子图模式替换为另一个更复杂的子图。通过迭代应用图文法规则，可以从简单的起点-终点图逐步演化出复杂的关卡拓扑。图文法特别适合生成具有明确入口、出口、分支和汇合点的非线性关卡结构。


小结
    

  • L-System 通过字符串重写规则和海龟图形解释器生成逼真的植物和分形结构
    • 

  • 波函数坍缩算法从局部邻接约束出发，通过约束传播实现全局一致的瓦片地图生成
    • 

  • CSP 方法将关卡生成建模为约束求解，特别适合锁钥结构和谜题关卡的自动生成
    • 

  • 图文法在图结构上定义替换规则，适合生成非线性关卡拓扑
    • 

  • 基于规则的方法的核心优势是可控性——设计师通过规则精确定义生成空间的边界
    • 

  • 实际项目中通常组合使用多种方法，例如用 WFC 生成地图瓦片、用 CSP 布置锁钥
