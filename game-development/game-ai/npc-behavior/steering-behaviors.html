转向行为

核心概念

转向行为（Steering Behaviors）是 Craig Reynolds 在 1999 年提出的一套自主角色运动控制方法。它将角色的运动分解为三个层次：行动选择（做什么）、转向（如何移动）和运动（物理实现）。每种转向行为计算一个"期望速度"，与当前速度的差值就是转向力，多种转向力组合后驱动角色移动。这套方法能用极简的规则产生复杂、自然的群体运动效果。

转向行为的核心公式：转向力 = 期望速度 - 当前速度。每种行为只需计算不同的"期望速度"，框架层面完全一致。


基础转向行为实现
所有转向行为共享同一个角色模型——具有位置、速度、最大速度和最大力三个属性的运动实体。每种行为产生一个转向力向量，经过截断和叠加后驱动角色运动。


interface Vector2 {
  x: number;
  y: number;
}

function vecAdd(a: Vector2, b: Vector2): Vector2 {
  return { x: a.x + b.x, y: a.y + b.y };
}
function vecSub(a: Vector2, b: Vector2): Vector2 {
  return { x: a.x - b.x, y: a.y - b.y };
}
function vecScale(v: Vector2, s: number): Vector2 {
  return { x: v.x * s, y: v.y * s };
}
function vecLength(v: Vector2): number {
  return Math.sqrt(v.x * v.x + v.y * v.y);
}
function vecNormalize(v: Vector2): Vector2 {
  const len = vecLength(v);
  return len > 0 ? { x: v.x / len, y: v.y / len } : { x: 0, y: 0 };
}
function vecTruncate(v: Vector2, max: number): Vector2 {
  const len = vecLength(v);
  return len > max ? vecScale(vecNormalize(v), max) : v;
}

class SteeringAgent {
  position: Vector2;
  velocity: Vector2;
  maxSpeed: number;
  maxForce: number;
  mass: number;

  constructor(pos: Vector2, maxSpeed: number, maxForce: number) {
    this.position = pos;
    this.velocity = { x: 0, y: 0 };
    this.maxSpeed = maxSpeed;
    this.maxForce = maxForce;
    this.mass = 1;
  }

  applyForce(steeringForce: Vector2, dt: number): void {
    const truncated = vecTruncate(steeringForce, this.maxForce);
    const acceleration = vecScale(truncated, 1 / this.mass);
    this.velocity = vecTruncate(vecAdd(this.velocity, vecScale(acceleration, dt)), this.maxSpeed);
    this.position = vecAdd(this.position, vecScale(this.velocity, dt));
  }
}

function seek(agent: SteeringAgent, target: Vector2): Vector2 {
  const desired = vecScale(vecNormalize(vecSub(target, agent.position)), agent.maxSpeed);
  return vecSub(desired, agent.velocity);
}

function flee(agent: SteeringAgent, threat: Vector2, panicRadius: number): Vector2 {
  const offset = vecSub(agent.position, threat);
  if (vecLength(offset) > panicRadius) return { x: 0, y: 0 };
  const desired = vecScale(vecNormalize(offset), agent.maxSpeed);
  return vecSub(desired, agent.velocity);
}

function arrive(agent: SteeringAgent, target: Vector2, slowingRadius: number): Vector2 {
  const offset = vecSub(target, agent.position);
  const distance = vecLength(offset);
  if (distance < 0.1) return vecScale(agent.velocity, -1);
  const rampedSpeed = agent.maxSpeed * Math.min(1, distance / slowingRadius);
  const desired = vecScale(vecNormalize(offset), rampedSpeed);
  return vecSub(desired, agent.velocity);
}

function pursue(agent: SteeringAgent, target: SteeringAgent): Vector2 {
  const offset = vecSub(target.position, agent.position);
  const distance = vecLength(offset);
  const lookAhead = distance / agent.maxSpeed;
  const futurePos = vecAdd(target.position, vecScale(target.velocity, lookAhead));
  return seek(agent, futurePos);
}

function evade(agent: SteeringAgent, pursuer: SteeringAgent): Vector2 {
  const offset = vecSub(pursuer.position, agent.position);
  const distance = vecLength(offset);
  const lookAhead = distance / agent.maxSpeed;
  const futurePos = vecAdd(pursuer.position, vecScale(pursuer.velocity, lookAhead));
  return flee(agent, futurePos, Infinity);
}
arrive 行为使用减速半径实现平滑到达——当角色进入减速区域后，期望速度线性降低，避免了 seek 在目标点附近来回抖动的问题。pursue 和 evade 通过预测目标未来位置实现更智能的截击和闪避。


漫游与障碍避让
Wander 行为在角色前方投射一个虚拟圆，在圆上选择随机点作为方向目标，产生自然流畅的随机运动。障碍避让则通过前方射线检测障碍物，提前产生转向力绕过障碍。


class WanderBehavior {
  private wanderAngle = 0;
  private circleDistance: number;
  private circleRadius: number;
  private angleJitter: number;

  constructor(circleDistance: number, circleRadius: number, angleJitter: number) {
    this.circleDistance = circleDistance;
    this.circleRadius = circleRadius;
    this.angleJitter = angleJitter;
  }

  calculate(agent: SteeringAgent): Vector2 {
    const heading = vecLength(agent.velocity) > 0.01 ? vecNormalize(agent.velocity) : { x: 1, y: 0 };

    const circleCenter = vecAdd(agent.position, vecScale(heading, this.circleDistance));
    this.wanderAngle += (Math.random() - 0.5) * this.angleJitter;
    const wanderPoint: Vector2 = {
      x: circleCenter.x + Math.cos(this.wanderAngle) * this.circleRadius,
      y: circleCenter.y + Math.sin(this.wanderAngle) * this.circleRadius,
    };
    return seek(agent, wanderPoint);
  }
}

interface Obstacle {
  position: Vector2;
  radius: number;
}

function obstacleAvoidance(agent: SteeringAgent, obstacles: Obstacle[], detectionLength: number): Vector2 {
  const heading = vecLength(agent.velocity) > 0.01 ? vecNormalize(agent.velocity) : { x: 1, y: 0 };
  const side: Vector2 = { x: -heading.y, y: heading.x };

  let nearestObstacle: Obstacle | null = null;
  let nearestDist = Infinity;

  for (const obs of obstacles) {
    const localOffset = vecSub(obs.position, agent.position);
    const ahead = localOffset.x * heading.x + localOffset.y * heading.y;
    if (ahead < 0 || ahead > detectionLength) continue;

    const lateral = Math.abs(localOffset.x * side.x + localOffset.y * side.y);
    if (lateral < obs.radius + 1) {
      if (ahead < nearestDist) {
        nearestDist = ahead;
        nearestObstacle = obs;
      }
    }
  }

  if (!nearestObstacle) return { x: 0, y: 0 };

  const localOffset = vecSub(nearestObstacle.position, agent.position);
  const lateralDot = localOffset.x * side.x + localOffset.y * side.y;
  const avoidDir = lateralDot > 0 ? vecScale(side, -1) : side;
  const urgency = 1 - nearestDist / detectionLength;

  return vecScale(avoidDir, agent.maxForce * urgency);
}
Wander 的 angleJitter 参数控制运动的随机程度——小值产生平缓的曲线运动（如鱼群游弋），大值产生频繁转向（如蜜蜂飞行）。障碍避让的紧迫度与距离成反比，越近避让力越强。


群体行为与 Flocking
Craig Reynolds 最著名的贡献是 Boids 模型——仅用三条简单规则就能模拟鸟群、鱼群等复杂群体行为：分离（Separation）、对齐（Alignment）和聚合（Cohesion）。


function separation(agent: SteeringAgent, neighbors: SteeringAgent[], desiredSeparation: number): Vector2 {
  let force: Vector2 = { x: 0, y: 0 };
  let count = 0;

  for (const other of neighbors) {
    const dist = vecLength(vecSub(agent.position, other.position));
    if (dist > 0 && dist < desiredSeparation) {
      const away = vecNormalize(vecSub(agent.position, other.position));
      force = vecAdd(force, vecScale(away, 1 / dist));
      count++;
    }
  }
  if (count > 0) {
    force = vecScale(force, 1 / count);
    force = vecScale(vecNormalize(force), agent.maxSpeed);
    force = vecSub(force, agent.velocity);
  }
  return force;
}

function alignment(agent: SteeringAgent, neighbors: SteeringAgent[]): Vector2 {
  let avgVelocity: Vector2 = { x: 0, y: 0 };
  let count = 0;

  for (const other of neighbors) {
    avgVelocity = vecAdd(avgVelocity, other.velocity);
    count++;
  }
  if (count > 0) {
    avgVelocity = vecScale(avgVelocity, 1 / count);
    avgVelocity = vecScale(vecNormalize(avgVelocity), agent.maxSpeed);
    return vecSub(avgVelocity, agent.velocity);
  }
  return { x: 0, y: 0 };
}

function cohesion(agent: SteeringAgent, neighbors: SteeringAgent[]): Vector2 {
  let center: Vector2 = { x: 0, y: 0 };
  let count = 0;

  for (const other of neighbors) {
    center = vecAdd(center, other.position);
    count++;
  }
  if (count > 0) {
    center = vecScale(center, 1 / count);
    return seek(agent, center);
  }
  return { x: 0, y: 0 };
}

class FlockingSystem {
  private agents: SteeringAgent[];
  private separationWeight: number;
  private alignmentWeight: number;
  private cohesionWeight: number;
  private neighborRadius: number;

  constructor(agents: SteeringAgent[], weights: { sep: number; ali: number; coh: number }, neighborRadius: number) {
    this.agents = agents;
    this.separationWeight = weights.sep;
    this.alignmentWeight = weights.ali;
    this.cohesionWeight = weights.coh;
    this.neighborRadius = neighborRadius;
  }

  update(dt: number): void {
    for (const agent of this.agents) {
      const neighbors = this.getNeighbors(agent);
      const sep = vecScale(separation(agent, neighbors, this.neighborRadius * 0.5), this.separationWeight);
      const ali = vecScale(alignment(agent, neighbors), this.alignmentWeight);
      const coh = vecScale(cohesion(agent, neighbors), this.cohesionWeight);
      const totalForce = vecAdd(vecAdd(sep, ali), coh);
      agent.applyForce(totalForce, dt);
    }
  }

  private getNeighbors(agent: SteeringAgent): SteeringAgent[] {
    return this.agents.filter((other) => {
      if (other === agent) return false;
      return vecLength(vecSub(agent.position, other.position)) < this.neighborRadius;
    });
  }
}

三种力的权重比例决定了群体的视觉特征。分离权重过高会导致群体散开，聚合权重过高会导致所有个体聚成一团，对齐权重过高则产生整齐但僵硬的运动。


小结
    

  • 转向行为将复杂运动分解为简单力的叠加，核心公式是"转向力 = 期望速度 - 当前速度"
    • 

  • Seek/Flee 处理趋近/远离，Arrive 实现平滑到达，Pursue/Evade 做预测性截击
    • 

  • Wander 用前方投射圆产生自然随机运动，障碍避让用前方射线检测提前转向
    • 

  • Boids 三规则（分离、对齐、聚合）用极简参数生成逼真群体行为
    • 

  • 权重调节直接影响群体视觉效果，是设计师可调的核心参数
    • 

  • 大规模群体需要空间划分优化邻居查询，常用网格或四叉树
