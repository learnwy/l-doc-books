个性模型

核心概念

个性模型（Personality Model）为 AI 角色注入独特的行为倾向，使同类型的 NPC 表现出不同的"性格"。一个勇敢的战士会冲锋在前，一个谨慎的战士会优先找掩体，一个冲动的战士会不顾弹药直接冲上去——这些差异不是通过编写不同的行为树实现的，而是通过参数化的个性系统驱动同一套行为逻辑产生的变异行为。

个性模型的核心思想是"一套代码，多种表现"——通过参数空间的不同位置产生不同的行为模式，避免为每种性格编写独立的 AI 逻辑。


特质参数系统
特质（Trait）是角色固有的长期属性，不会在战斗中快速变化。常见的特质系统借鉴心理学的五大人格模型（OCEAN），但针对游戏需求做了简化和调整。


interface PersonalityTraits {
  aggression: number;
  courage: number;
  caution: number;
  teamwork: number;
  patience: number;
  curiosity: number;
}

interface TraitRange {
  min: number;
  max: number;
}

interface PersonalityArchetype {
  name: string;
  traits: PersonalityTraits;
  variance: Partial<Record<keyof PersonalityTraits, TraitRange>>;
}

const archetypes: PersonalityArchetype[] = [
  {
    name: 'berserker',
    traits: { aggression: 0.9, courage: 0.95, caution: 0.1, teamwork: 0.3, patience: 0.1, curiosity: 0.2 },
    variance: { aggression: { min: -0.1, max: 0.1 }, courage: { min: -0.05, max: 0.05 } },
  },
  {
    name: 'tactician',
    traits: { aggression: 0.5, courage: 0.6, caution: 0.8, teamwork: 0.9, patience: 0.7, curiosity: 0.4 },
    variance: { caution: { min: -0.1, max: 0.1 }, teamwork: { min: -0.1, max: 0.1 } },
  },
  {
    name: 'coward',
    traits: { aggression: 0.2, courage: 0.15, caution: 0.95, teamwork: 0.5, patience: 0.3, curiosity: 0.6 },
    variance: { courage: { min: -0.1, max: 0.15 }, caution: { min: -0.05, max: 0.05 } },
  },
  {
    name: 'scout',
    traits: { aggression: 0.3, courage: 0.7, caution: 0.6, teamwork: 0.4, patience: 0.8, curiosity: 0.9 },
    variance: { curiosity: { min: -0.1, max: 0.1 } },
  },
];

class PersonalityGenerator {
  generateFromArchetype(archetype: PersonalityArchetype): PersonalityTraits {
    const traits = { ...archetype.traits };
    for (const [key, range] of Object.entries(archetype.variance)) {
      if (range) {
        const traitKey = key as keyof PersonalityTraits;
        const variation = range.min + Math.random() * (range.max - range.min);
        traits[traitKey] = Math.max(0, Math.min(1, traits[traitKey] + variation));
      }
    }
    return traits;
  }

  generateRandom(): PersonalityTraits {
    return {
      aggression: Math.random(),
      courage: Math.random(),
      caution: Math.random(),
      teamwork: Math.random(),
      patience: Math.random(),
      curiosity: Math.random(),
    };
  }

  blend(a: PersonalityTraits, b: PersonalityTraits, factor: number): PersonalityTraits {
    const f = Math.max(0, Math.min(1, factor));
    return {
      aggression: a.aggression * (1 - f) + b.aggression * f,
      courage: a.courage * (1 - f) + b.courage * f,
      caution: a.caution * (1 - f) + b.caution * f,
      teamwork: a.teamwork * (1 - f) + b.teamwork * f,
      patience: a.patience * (1 - f) + b.patience * f,
      curiosity: a.curiosity * (1 - f) + b.curiosity * f,
    };
  }
}
原型系统通过"基线+方差"的方式生成个体。同一个"狂战士"原型生成的两个角色，攻击性可能分别是 0.82 和 0.96——都很好斗，但表现细节不同。blend 方法支持混合原型，如 0.7 的战术家 + 0.3 的狂战士，创造"偏激进的战术家"。


行为调制
个性参数不直接控制行为，而是"调制"（Modulate）已有决策系统的参数。这意味着同一套行为树或效用 AI 逻辑，在不同个性参数下产生不同的行为倾向。


interface BehaviorModifiers {
  attackWeight: number;
  defendWeight: number;
  retreatHealthThreshold: number;
  coverPreference: number;
  engageDistance: number;
  waitTolerance: number;
  flankingtendency: number;
  supportAllyRange: number;
}

class PersonalityModulator {
  computeModifiers(traits: PersonalityTraits): BehaviorModifiers {
    return {
      attackWeight: 0.5 + traits.aggression * 1.5,
      defendWeight: 0.5 + traits.caution * 1.5,
      retreatHealthThreshold: 0.1 + (1 - traits.courage) * 0.5,
      coverPreference: 0.2 + traits.caution * 0.8,
      engageDistance: 50 + (1 - traits.aggression) * 150,
      waitTolerance: 1 + traits.patience * 9,
      flankingtendency: traits.aggression * 0.4 + traits.courage * 0.3 + (1 - traits.caution) * 0.3,
      supportAllyRange: 30 + traits.teamwork * 70,
    };
  }

  modulateUtilityScore(baseScore: number, actionType: string, traits: PersonalityTraits): number {
    const modifiers: Record<string, (t: PersonalityTraits) => number> = {
      attack: (t) => 0.5 + t.aggression * 1.0,
      defend: (t) => 0.5 + t.caution * 1.0,
      retreat: (t) => 0.3 + (1 - t.courage) * 1.4,
      heal: (t) => 0.5 + t.caution * 0.8,
      explore: (t) => 0.3 + t.curiosity * 1.4,
      support_ally: (t) => 0.3 + t.teamwork * 1.4,
      ambush: (t) => 0.3 + t.patience * 0.7 + t.caution * 0.5,
      charge: (t) => 0.2 + t.aggression * 0.8 + t.courage * 0.6,
    };

    const modifier = modifiers[actionType];
    return modifier ? baseScore * modifier(traits) : baseScore;
  }

  shouldRetreat(health: number, maxHealth: number, traits: PersonalityTraits): boolean {
    const threshold = 0.1 + (1 - traits.courage) * 0.5;
    const panicModifier = traits.aggression > 0.8 ? 0.7 : 1.0;
    return health / maxHealth < threshold * panicModifier;
  }
}

class PersonalityDrivenAI {
  private traits: PersonalityTraits;
  private modulator: PersonalityModulator;
  private modifiers: BehaviorModifiers;

  constructor(traits: PersonalityTraits) {
    this.traits = traits;
    this.modulator = new PersonalityModulator();
    this.modifiers = this.modulator.computeModifiers(traits);
  }

  evaluateActions(context: {
    health: number;
    maxHealth: number;
    distToEnemy: number;
    allyNearby: boolean;
  }): { action: string; score: number }[] {
    const actions = [
      { action: 'attack', base: context.distToEnemy < this.modifiers.engageDistance ? 0.7 : 0.3 },
      { action: 'defend', base: context.health / context.maxHealth < 0.5 ? 0.8 : 0.3 },
      { action: 'retreat', base: context.health / context.maxHealth < 0.3 ? 0.9 : 0.1 },
      { action: 'support_ally', base: context.allyNearby ? 0.5 : 0.1 },
      { action: 'charge', base: context.distToEnemy < 100 ? 0.4 : 0.2 },
    ];

    return actions
      .map(({ action, base }) => ({
        action,
        score: this.modulator.modulateUtilityScore(base, action, this.traits),
      }))
      .sort((a, b) => b.score - a.score);
  }
}

调制器的关键设计是"基线+倍数"——即使是最胆小的 AI（courage=0.0），当血量低于 10% 时撤退阈值也只是 60% 而非 100%。这防止了极端参数产生不合理的行为。


情绪动态系统
特质是长期稳定的，但 AI 在战斗中还需要短期的情绪波动——恐惧、愤怒、信心等。情绪系统在特质基础上叠加动态变化，使 AI 的行为更加生动和不可预测。


interface EmotionState {
  fear: number;
  anger: number;
  confidence: number;
  excitement: number;
}

interface EmotionTrigger {
  event: string;
  effects: Partial<EmotionState>;
  decayRate: Partial<EmotionState>;
}

class EmotionSystem {
  private state: EmotionState;
  private baseTraits: PersonalityTraits;
  private triggers: EmotionTrigger[];

  constructor(traits: PersonalityTraits, triggers: EmotionTrigger[]) {
    this.baseTraits = traits;
    this.triggers = triggers;
    this.state = {
      fear: (1 - traits.courage) * 0.2,
      anger: traits.aggression * 0.1,
      confidence: traits.courage * 0.5,
      excitement: 0,
    };
  }

  onEvent(eventName: string): void {
    const trigger = this.triggers.find((t) => t.event === eventName);
    if (!trigger) return;

    for (const [key, delta] of Object.entries(trigger.effects)) {
      const emotionKey = key as keyof EmotionState;
      if (delta !== undefined) {
        const traitResistance = this.getResistance(emotionKey);
        this.state[emotionKey] = Math.max(0, Math.min(1, this.state[emotionKey] + delta * traitResistance));
      }
    }
  }

  update(dt: number): void {
    const baselines = {
      fear: (1 - this.baseTraits.courage) * 0.1,
      anger: this.baseTraits.aggression * 0.05,
      confidence: this.baseTraits.courage * 0.3,
      excitement: 0.05,
    };

    for (const key of Object.keys(this.state) as (keyof EmotionState)[]) {
      const diff = baselines[key] - this.state[key];
      this.state[key] += diff * 0.1 * dt;
    }
  }

  getEffectiveTraits(): PersonalityTraits {
    return {
      aggression: Math.min(1, this.baseTraits.aggression + this.state.anger * 0.3 - this.state.fear * 0.2),
      courage: Math.min(1, this.baseTraits.courage + this.state.confidence * 0.3 - this.state.fear * 0.4),
      caution: Math.min(1, this.baseTraits.caution - this.state.anger * 0.2 + this.state.fear * 0.3),
      teamwork: this.baseTraits.teamwork,
      patience: Math.min(1, this.baseTraits.patience - this.state.excitement * 0.3 - this.state.anger * 0.2),
      curiosity: Math.min(1, this.baseTraits.curiosity + this.state.excitement * 0.2 - this.state.fear * 0.3),
    };
  }

  private getResistance(emotion: keyof EmotionState): number {
    switch (emotion) {
      case 'fear':
        return 1 - this.baseTraits.courage * 0.5;
      case 'anger':
        return 0.5 + this.baseTraits.aggression * 0.5;
      case 'confidence':
        return 0.5 + this.baseTraits.courage * 0.5;
      case 'excitement':
        return 0.5 + this.baseTraits.curiosity * 0.3;
    }
  }
}

const combatTriggers: EmotionTrigger[] = [
  {
    event: 'ally_killed',
    effects: { fear: 0.3, anger: 0.4, confidence: -0.2 },
    decayRate: { fear: 0.05, anger: 0.03 },
  },
  { event: 'took_damage', effects: { fear: 0.15, anger: 0.1, confidence: -0.1 }, decayRate: { fear: 0.08 } },
  { event: 'killed_enemy', effects: { confidence: 0.3, excitement: 0.2, fear: -0.1 }, decayRate: { confidence: 0.02 } },
  { event: 'flanked', effects: { fear: 0.4, confidence: -0.3 }, decayRate: { fear: 0.04 } },
  { event: 'reinforcement_arrived', effects: { confidence: 0.4, fear: -0.2 }, decayRate: { confidence: 0.03 } },
];
情绪系统让特质"活"了起来：一个基线 courage=0.7 的老兵在战友阵亡后（fear+0.3）可能变得犹豫，但在击杀敌人后（confidence+0.3）又恢复自信。特质决定了情绪的"弹性"——高 courage 角色恐惧增加更慢、恢复更快。


小结
    

  • 个性模型通过参数化特质让同一套 AI 代码产生多样化行为
    • 

  • 原型+方差机制在保证类型一致性的同时引入个体差异
    • 

  • 行为调制器不替换决策逻辑，而是调整权重和阈值，实现"软性"个性化
    • 

  • 情绪系统在特质基础上叠加短期波动，特质决定情绪的敏感度和恢复速度
    • 

  • 所有参数都有安全边界（基线+倍数），防止极端值产生不合理行为
    • 

  • 个性系统是效用 AI 和行为树的理想伴侣，三者结合构成完整的角色行为框架
