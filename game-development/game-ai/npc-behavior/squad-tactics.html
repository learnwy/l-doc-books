小队战术

核心概念

小队战术 AI 将多个 NPC 组织为协同作战单元，由小队指挥官（Squad Leader）分配角色、协调行动，使团队整体表现出超越个体的战术智能。与每个 NPC 独立决策不同，小队战术引入了指挥层——队长评估战场态势后下达命令，队员在命令框架内自主执行细节。这种分层架构是《F.E.A.R.》《光环》《神秘海域》等 AAA 游戏 AI 的核心。

分层架构的优势是职责清晰：指挥官不关心个体如何到达掩体点（寻路细节），个体不关心为什么要去侧翼（战术意图）。这种解耦使得两层可以独立开发和调试。


战术位置评估
小队战术的基础是对战场空间的战术评估——哪些位置是好的掩体、哪些适合伏击、哪些有射击优势。战术位置评估系统为地图上的关键点打分，指挥官根据分数分配队员位置。


interface Vector2 {
  x: number;
  y: number;
}

interface TacticalPoint {
  position: Vector2;
  coverQuality: number;
  lineOfSight: boolean;
  flankingAngle: number;
  distanceToEnemy: number;
  distanceToSquad: number;
  elevationAdvantage: number;
}

interface TacticalScoreWeights {
  cover: number;
  lineOfSight: number;
  flanking: number;
  proximity: number;
  squadCohesion: number;
  elevation: number;
}

class TacticalPointSystem {
  private points: TacticalPoint[] = [];

  evaluatePoint(point: TacticalPoint, role: SquadRole, weights: TacticalScoreWeights): number {
    let score = 0;

    score += point.coverQuality * weights.cover;

    if (role === 'assault') {
      score += (point.lineOfSight ? 1 : 0) * weights.lineOfSight;
      score += (1 - point.distanceToEnemy / 200) * weights.proximity;
    } else if (role === 'support') {
      score += (point.lineOfSight ? 1 : 0) * weights.lineOfSight * 1.5;
      score += point.elevationAdvantage * weights.elevation;
    } else if (role === 'flanker') {
      const flankBonus = Math.sin((point.flankingAngle * Math.PI) / 180);
      score += flankBonus * weights.flanking;
      score += (1 - point.coverQuality) * 0.3;
    }

    score += (1 - point.distanceToSquad / 100) * weights.squadCohesion;

    return Math.max(0, Math.min(1, score));
  }

  findBestPositions(
    candidates: TacticalPoint[],
    role: SquadRole,
    weights: TacticalScoreWeights,
    count: number,
  ): TacticalPoint[] {
    return candidates
      .map((p) => ({ point: p, score: this.evaluatePoint(p, role, weights) }))
      .sort((a, b) => b.score - a.score)
      .slice(0, count)
      .map((entry) => entry.point);
  }

  generateCandidates(center: Vector2, radius: number, spacing: number): TacticalPoint[] {
    const candidates: TacticalPoint[] = [];
    for (let x = center.x - radius; x <= center.x + radius; x += spacing) {
      for (let y = center.y - radius; y <= center.y + radius; y += spacing) {
        const dist = Math.sqrt((x - center.x) ** 2 + (y - center.y) ** 2);
        if (dist <= radius) {
          candidates.push({
            position: { x, y },
            coverQuality: 0,
            lineOfSight: false,
            flankingAngle: 0,
            distanceToEnemy: 0,
            distanceToSquad: 0,
            elevationAdvantage: 0,
          });
        }
      }
    }
    return candidates;
  }
}
不同角色对战术位置有截然不同的偏好：突击手需要近距离+视线，支援手需要远距离+高地+掩体，侧翼手需要大侧翼角度，不一定需要掩体。这种差异化评分是协同战术的基础。


小队指挥官与角色分配
小队指挥官是战术层的核心决策者。它维护队员列表和当前战术状态，根据战场态势选择战术模板，并将角色分配给每个队员。


type SquadRole = 'assault' | 'support' | 'flanker' | 'scout' | 'medic';

interface SquadMember {
  id: string;
  position: Vector2;
  health: number;
  maxHealth: number;
  role: SquadRole;
  assignedPosition: Vector2 | null;
  isAlive: boolean;
  ammo: number;
}

interface TacticTemplate {
  name: string;
  roles: SquadRole[];
  condition: (situation: BattleSituation) => boolean;
  priority: number;
}

interface BattleSituation {
  aliveMembers: number;
  totalMembers: number;
  avgHealth: number;
  enemyCount: number;
  enemyDistance: number;
  hasFlankOpportunity: boolean;
  inCover: number;
  ammoLevel: number;
}

class SquadLeader {
  private members: SquadMember[];
  private tactics: TacticTemplate[];
  private currentTactic: TacticTemplate | null = null;
  private tacticalSystem: TacticalPointSystem;

  constructor(members: SquadMember[], tactics: TacticTemplate[]) {
    this.members = members;
    this.tactics = tactics;
    this.tacticalSystem = new TacticalPointSystem();
  }

  update(situation: BattleSituation, enemyPositions: Vector2[]): void {
    const newTactic = this.selectTactic(situation);
    if (newTactic && newTactic !== this.currentTactic) {
      this.currentTactic = newTactic;
      this.assignRoles(newTactic.roles);
      this.assignPositions(enemyPositions);
    }
  }

  private selectTactic(situation: BattleSituation): TacticTemplate | null {
    const valid = this.tactics.filter((t) => t.condition(situation)).sort((a, b) => b.priority - a.priority);
    return valid[0] ?? null;
  }

  private assignRoles(roles: SquadRole[]): void {
    const alive = this.members.filter((m) => m.isAlive);
    for (let i = 0; i < alive.length && i < roles.length; i++) {
      alive[i].role = roles[i];
    }
  }

  private assignPositions(enemyPositions: Vector2[]): void {
    const enemyCenter = this.computeCenter(enemyPositions);
    const squadCenter = this.computeCenter(this.members.filter((m) => m.isAlive).map((m) => m.position));

    const weights: TacticalScoreWeights = {
      cover: 0.3,
      lineOfSight: 0.25,
      flanking: 0.2,
      proximity: 0.1,
      squadCohesion: 0.1,
      elevation: 0.05,
    };

    const candidates = this.tacticalSystem.generateCandidates(squadCenter, 100, 10);
    for (const candidate of candidates) {
      candidate.distanceToEnemy = Math.sqrt(
        (candidate.position.x - enemyCenter.x) ** 2 + (candidate.position.y - enemyCenter.y) ** 2,
      );
      candidate.distanceToSquad = Math.sqrt(
        (candidate.position.x - squadCenter.x) ** 2 + (candidate.position.y - squadCenter.y) ** 2,
      );
    }

    for (const member of this.members.filter((m) => m.isAlive)) {
      const best = this.tacticalSystem.findBestPositions(candidates, member.role, weights, 1);
      if (best.length > 0) {
        member.assignedPosition = best[0].position;
      }
    }
  }

  private computeCenter(positions: Vector2[]): Vector2 {
    if (positions.length === 0) return { x: 0, y: 0 };
    const sum = positions.reduce((acc, p) => ({ x: acc.x + p.x, y: acc.y + p.y }), { x: 0, y: 0 });
    return { x: sum.x / positions.length, y: sum.y / positions.length };
  }
}

战术模板的优先级是关键——"撤退"优先级最高（10），确保危急情况下无论其他条件是否满足都会触发撤退。


交替掩护与协同射击
高级小队战术包括交替掩护（Leapfrog）和压制射击（Suppressive Fire）。交替掩护让队员轮流移动和掩护，确保任何时刻都有队员在射击。压制射击不以命中为目标，而是迫使敌人躲在掩体后无法还击，为队友创造移动窗口。


interface CoverMoveOrder {
  moverId: string;
  covererIds: string[];
  targetPosition: Vector2;
  phase: 'suppressing' | 'moving' | 'settled';
}

class LeapfrogCoordinator {
  private members: SquadMember[];
  private currentOrders: CoverMoveOrder[] = [];
  private moveQueue: string[] = [];

  constructor(members: SquadMember[]) {
    this.members = members;
  }

  planLeapfrog(destination: Vector2, waypointSpacing: number): void {
    const alive = this.members.filter((m) => m.isAlive);
    if (alive.length < 2) return;

    const squadCenter = {
      x: alive.reduce((s, m) => s + m.position.x, 0) / alive.length,
      y: alive.reduce((s, m) => s + m.position.y, 0) / alive.length,
    };

    const dirToTarget = {
      x: destination.x - squadCenter.x,
      y: destination.y - squadCenter.y,
    };
    const dist = Math.sqrt(dirToTarget.x ** 2 + dirToTarget.y ** 2);
    const normalized = dist > 0 ? { x: dirToTarget.x / dist, y: dirToTarget.y / dist } : { x: 0, y: 0 };

    this.moveQueue = alive.map((m) => m.id);
    const steps = Math.ceil(dist / waypointSpacing);

    for (let i = 0; i < steps; i++) {
      const moverIdx = i % this.moveQueue.length;
      const moverId = this.moveQueue[moverIdx];
      const covererIds = this.moveQueue.filter((id) => id !== moverId);

      this.currentOrders.push({
        moverId,
        covererIds,
        targetPosition: {
          x: squadCenter.x + normalized.x * waypointSpacing * (i + 1),
          y: squadCenter.y + normalized.y * waypointSpacing * (i + 1),
        },
        phase: 'suppressing',
      });
    }
  }

  getCurrentOrder(): CoverMoveOrder | null {
    return this.currentOrders[0] ?? null;
  }

  advancePhase(): void {
    if (this.currentOrders.length === 0) return;
    const current = this.currentOrders[0];
    switch (current.phase) {
      case 'suppressing':
        current.phase = 'moving';
        break;
      case 'moving':
        current.phase = 'settled';
        break;
      case 'settled':
        this.currentOrders.shift();
        break;
    }
  }
}

class SuppressionManager {
  private suppressionTargets: Map<string, { position: Vector2; assignedShooters: string[]; duration: number }> =
    new Map();

  addTarget(targetId: string, position: Vector2, shooterIds: string[], duration: number): void {
    this.suppressionTargets.set(targetId, { position, assignedShooters: shooterIds, duration });
  }

  update(dt: number): void {
    const toRemove: string[] = [];
    for (const [id, target] of this.suppressionTargets) {
      target.duration -= dt;
      if (target.duration <= 0) toRemove.push(id);
    }
    for (const id of toRemove) this.suppressionTargets.delete(id);
  }

  getShootingOrders(shooterId: string): Vector2[] {
    const targets: Vector2[] = [];
    for (const target of this.suppressionTargets.values()) {
      if (target.assignedShooters.includes(shooterId)) {
        targets.push(target.position);
      }
    }
    return targets;
  }
}
交替掩护的三阶段循环（压制→移动→就位）确保了火力的连续性。《F.E.A.R.》的 AI 之所以给玩家强烈的"对手很聪明"的感觉，很大程度上就是因为实现了可见的掩护移动和侧翼包抄行为。


小结
    

  • 小队战术将 AI 决策分为战略/战术/执行三层，指挥官负责战术选择和角色分配
    • 

  • 战术位置评估为不同角色（突击/支援/侧翼）计算差异化的位置偏好
    • 

  • 战术模板通过条件匹配和优先级实现态势响应的战术切换
    • 

  • 交替掩护和压制射击让玩家看到可理解的协同行为，极大提升 AI 可信度
    • 

  • 分层架构使战术层和执行层可独立开发调试，扩展性强
    • 

  • 实际游戏中通常结合脚本事件和动态战术，在可预测性和灵活性间取得平衡
