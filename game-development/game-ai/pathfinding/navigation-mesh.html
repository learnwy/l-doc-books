导航网格

核心概念

导航网格（Navigation Mesh，简称 NavMesh）是现代游戏中最主流的寻路数据结构，它将游戏世界的可行走区域分解为一组凸多边形，AI 角色在这些多边形之间寻找路径。与网格图相比，导航网格大幅减少了搜索节点数量，天然支持任意宽度的角色通行，且能精确表示复杂地形。

导航网格将寻路问题转化为"在凸多边形邻接图上搜索"，每个多边形内部的移动无需额外计算，只有跨越多边形边界时才需要路径规划。


网格生成流程
导航网格生成是将原始三维几何转化为可寻路数据结构的过程。工业级实现通常采用 Recast 库的流水线，包含体素化、区域划分、轮廓提取和多边形构建四个阶段。


interface Vector3 {
  x: number;
  y: number;
  z: number;
}

interface NavPoly {
  id: number;
  vertices: Vector3[];
  neighbors: number[];
  center: Vector3;
  normal: Vector3;
}

interface HeightField {
  width: number;
  depth: number;
  cellSize: number;
  cellHeight: number;
  spans: HeightSpan[][];
}

interface HeightSpan {
  minY: number;
  maxY: number;
  walkable: boolean;
}

class VoxelBuilder {
  private cellSize: number;
  private cellHeight: number;
  private walkableSlope: number;
  private walkableHeight: number;
  private walkableClimb: number;

  constructor(config: {
    cellSize: number;
    cellHeight: number;
    walkableSlope: number;
    walkableHeight: number;
    walkableClimb: number;
  }) {
    this.cellSize = config.cellSize;
    this.cellHeight = config.cellHeight;
    this.walkableSlope = config.walkableSlope;
    this.walkableHeight = config.walkableHeight;
    this.walkableClimb = config.walkableClimb;
  }

  rasterizeTriangle(v0: Vector3, v1: Vector3, v2: Vector3, heightField: HeightField): void {
    const minX = Math.floor(Math.min(v0.x, v1.x, v2.x) / this.cellSize);
    const maxX = Math.floor(Math.max(v0.x, v1.x, v2.x) / this.cellSize);
    const minZ = Math.floor(Math.min(v0.z, v1.z, v2.z) / this.cellSize);
    const maxZ = Math.floor(Math.max(v0.z, v1.z, v2.z) / this.cellSize);

    for (let z = minZ; z <= maxZ; z++) {
      for (let x = minX; x <= maxX; x++) {
        const span = this.computeSpan(x, z, v0, v1, v2);
        if (span) {
          heightField.spans[z][x] = heightField.spans[z]?.[x] ?? [];
          this.mergeSpan(heightField.spans[z], x, span);
        }
      }
    }
  }

  markWalkable(heightField: HeightField): void {
    for (let z = 0; z < heightField.depth; z++) {
      for (let x = 0; x < heightField.width; x++) {
        const spans = heightField.spans[z]?.[x];
        if (!spans) continue;
        for (const span of Object.values(spans) as HeightSpan[]) {
          const slopeAngle = this.computeSlopeAngle(heightField, x, z, span);
          const clearance = this.computeClearance(heightField, x, z, span);
          span.walkable = slopeAngle <= this.walkableSlope && clearance >= this.walkableHeight;
        }
      }
    }
  }

  private computeSpan(_x: number, _z: number, _v0: Vector3, _v1: Vector3, _v2: Vector3): HeightSpan | null {
    return null;
  }

  private mergeSpan(_spanColumn: unknown[], _x: number, _span: HeightSpan): void {}
  private computeSlopeAngle(_hf: HeightField, _x: number, _z: number, _s: HeightSpan): number {
    return 0;
  }
  private computeClearance(_hf: HeightField, _x: number, _z: number, _s: HeightSpan): number {
    return 0;
  }
}
体素化的核心参数直接影响导航网格的质量：cellSize 决定水平精度，cellHeight 决定垂直精度，walkableSlope 限制可通行坡度，walkableHeight 确保角色有足够空间通过。


多边形邻接与搜索
导航网格生成后，需要建立多边形之间的邻接关系，然后使用 A* 等算法在邻接图上搜索路径。搜索的结果是一系列相邻多边形（通道），后续需要进一步处理为精确路径点。


class NavMesh {
  private polygons: NavPoly[];
  private adjacency: Map<number, { neighborId: number; sharedEdge: [Vector3, Vector3] }[]>;

  constructor(polygons: NavPoly[]) {
    this.polygons = polygons;
    this.adjacency = new Map();
    this.buildAdjacency();
  }

  private buildAdjacency(): void {
    for (const poly of this.polygons) {
      this.adjacency.set(poly.id, []);
    }
    for (let i = 0; i < this.polygons.length; i++) {
      for (let j = i + 1; j < this.polygons.length; j++) {
        const shared = this.findSharedEdge(this.polygons[i], this.polygons[j]);
        if (shared) {
          this.adjacency.get(this.polygons[i].id)!.push({
            neighborId: this.polygons[j].id,
            sharedEdge: shared,
          });
          this.adjacency.get(this.polygons[j].id)!.push({
            neighborId: this.polygons[i].id,
            sharedEdge: shared,
          });
        }
      }
    }
  }

  private findSharedEdge(a: NavPoly, b: NavPoly): [Vector3, Vector3] | null {
    const sharedVerts: Vector3[] = [];
    for (const va of a.vertices) {
      for (const vb of b.vertices) {
        if (Math.abs(va.x - vb.x) < 0.01 && Math.abs(va.y - vb.y) < 0.01 && Math.abs(va.z - vb.z) < 0.01) {
          sharedVerts.push(va);
        }
      }
    }
    return sharedVerts.length >= 2 ? [sharedVerts[0], sharedVerts[1]] : null;
  }

  findPath(startPos: Vector3, endPos: Vector3): Vector3[] | null {
    const startPoly = this.findContainingPoly(startPos);
    const endPoly = this.findContainingPoly(endPos);
    if (!startPoly || !endPoly) return null;
    if (startPoly.id === endPoly.id) return [startPos, endPos];

    const corridor = this.aStarSearch(startPoly.id, endPoly.id);
    if (!corridor) return null;

    return this.funnelSmooth(corridor, startPos, endPos);
  }

  private findContainingPoly(pos: Vector3): NavPoly | null {
    for (const poly of this.polygons) {
      if (this.isPointInPoly(pos, poly)) return poly;
    }
    return null;
  }

  private isPointInPoly(point: Vector3, poly: NavPoly): boolean {
    const verts = poly.vertices;
    let inside = false;
    for (let i = 0, j = verts.length - 1; i < verts.length; j = i++) {
      if (
        verts[i].z > point.z !== verts[j].z > point.z &&
        point.x < ((verts[j].x - verts[i].x) * (point.z - verts[i].z)) / (verts[j].z - verts[i].z) + verts[i].x
      ) {
        inside = !inside;
      }
    }
    return inside;
  }

  private aStarSearch(startId: number, goalId: number): number[] | null {
    const openSet = new Map<number, number>();
    const gScore = new Map<number, number>();
    const parent = new Map<number, number>();
    gScore.set(startId, 0);
    openSet.set(startId, this.heuristic(startId, goalId));

    while (openSet.size > 0) {
      let current = -1;
      let minF = Infinity;
      for (const [id, f] of openSet) {
        if (f < minF) {
          minF = f;
          current = id;
        }
      }
      if (current === goalId) return this.buildCorridor(parent, startId, goalId);
      openSet.delete(current);

      for (const neighbor of this.adjacency.get(current) ?? []) {
        const edge = neighbor.sharedEdge;
        const edgeCenter = {
          x: (edge[0].x + edge[1].x) / 2,
          y: (edge[0].y + edge[1].y) / 2,
          z: (edge[0].z + edge[1].z) / 2,
        };
        const currentCenter = this.polygons[current].center;
        const dist = Math.sqrt((edgeCenter.x - currentCenter.x) ** 2 + (edgeCenter.z - currentCenter.z) ** 2);
        const tentativeG = (gScore.get(current) ?? Infinity) + dist;

        if (tentativeG < (gScore.get(neighbor.neighborId) ?? Infinity)) {
          gScore.set(neighbor.neighborId, tentativeG);
          parent.set(neighbor.neighborId, current);
          openSet.set(neighbor.neighborId, tentativeG + this.heuristic(neighbor.neighborId, goalId));
        }
      }
    }
    return null;
  }

  private heuristic(aId: number, bId: number): number {
    const a = this.polygons[aId].center;
    const b = this.polygons[bId].center;
    return Math.sqrt((a.x - b.x) ** 2 + (a.z - b.z) ** 2);
  }

  private buildCorridor(parent: Map<number, number>, start: number, goal: number): number[] {
    const path: number[] = [];
    let current = goal;
    while (current !== start) {
      path.unshift(current);
      current = parent.get(current)!;
    }
    path.unshift(start);
    return path;
  }

  private funnelSmooth(_corridor: number[], start: Vector3, end: Vector3): Vector3[] {
    return [start, end];
  }
}
邻接关系通过检测共享边来建立：如果两个多边形恰好共享两个顶点，说明它们之间有一条可通行的门户（Portal）。搜索在多边形中心点之间进行，用欧氏距离作为启发式估计。


漏斗算法路径平滑
A* 搜索返回的是多边形序列（通道），但角色需要精确的路径点。漏斗算法（Funnel Algorithm，也称 Simple Stupid Funnel Algorithm）通过逐步收紧左右边界来计算通道内的最短路径，避免不必要的绕行。


interface Portal {
  left: Vector3;
  right: Vector3;
}

function cross2D(a: Vector3, b: Vector3): number {
  return a.x * b.z - a.z * b.x;
}

function sub(a: Vector3, b: Vector3): Vector3 {
  return { x: a.x - b.x, y: a.y - b.y, z: a.z - b.z };
}

function funnelSmooth(portals: Portal[], start: Vector3, end: Vector3): Vector3[] {
  const path: Vector3[] = [start];
  let apex = start;
  let leftIdx = 0;
  let rightIdx = 0;
  let portalLeft = start;
  let portalRight = start;

  for (let i = 1; i <= portals.length; i++) {
    const left = i < portals.length ? portals[i].left : end;
    const right = i < portals.length ? portals[i].right : end;

    if (cross2D(sub(portalRight, apex), sub(right, apex)) >= 0) {
      if (apex === portalRight || cross2D(sub(portalLeft, apex), sub(right, apex)) < 0) {
        portalRight = right;
        rightIdx = i;
      } else {
        path.push(portalLeft);
        apex = portalLeft;
        leftIdx = leftIdx;
        rightIdx = leftIdx;
        portalRight = apex;
        i = leftIdx;
        continue;
      }
    }

    if (cross2D(sub(portalLeft, apex), sub(left, apex)) <= 0) {
      if (apex === portalLeft || cross2D(sub(portalRight, apex), sub(left, apex)) > 0) {
        portalLeft = left;
        leftIdx = i;
      } else {
        path.push(portalRight);
        apex = portalRight;
        leftIdx = rightIdx;
        rightIdx = rightIdx;
        portalLeft = apex;
        i = rightIdx;
        continue;
      }
    }
  }

  if (path[path.length - 1] !== end) {
    path.push(end);
  }
  return path;
}
漏斗算法的时间复杂度为 O(n)，其中 n 是通道中的门户数量。它产生的路径是通道内几何意义上的最短路径，角色行走时看起来自然流畅。


层级导航网格
大型开放世界游戏中，单层导航网格无法高效处理长距离寻路。层级导航网格（HPA* 或 Multi-Layer NavMesh）将导航网格划分为多个区块，先在高层做粗粒度搜索确定区块序列，再在低层进行精细路径规划。


interface NavChunk {
  id: number;
  polygons: NavPoly[];
  boundaryPortals: Map<number, Portal[]>;
}

class HierarchicalNavMesh {
  private chunks: NavChunk[];
  private highLevelGraph: Map<number, { chunkId: number; cost: number }[]>;

  constructor(chunks: NavChunk[]) {
    this.chunks = chunks;
    this.highLevelGraph = new Map();
    this.buildHighLevelGraph();
  }

  private buildHighLevelGraph(): void {
    for (const chunk of this.chunks) {
      const neighbors: { chunkId: number; cost: number }[] = [];
      for (const [neighborId, portals] of chunk.boundaryPortals) {
        const avgCost =
          portals.reduce((sum, p) => {
            const dx = p.left.x - p.right.x;
            const dz = p.left.z - p.right.z;
            return sum + Math.sqrt(dx * dx + dz * dz);
          }, 0) / portals.length;
        neighbors.push({ chunkId: neighborId, cost: avgCost });
      }
      this.highLevelGraph.set(chunk.id, neighbors);
    }
  }

  findPath(startPos: Vector3, endPos: Vector3): Vector3[] | null {
    const startChunk = this.findChunk(startPos);
    const endChunk = this.findChunk(endPos);
    if (!startChunk || !endChunk) return null;

    if (startChunk.id === endChunk.id) {
      const localMesh = new NavMesh(startChunk.polygons);
      return localMesh.findPath(startPos, endPos);
    }

    const chunkPath = this.highLevelSearch(startChunk.id, endChunk.id);
    if (!chunkPath) return null;

    return this.stitchLocalPaths(chunkPath, startPos, endPos);
  }

  private findChunk(_pos: Vector3): NavChunk | null {
    return this.chunks[0] ?? null;
  }

  private highLevelSearch(startId: number, goalId: number): number[] | null {
    const gScore = new Map<number, number>([[startId, 0]]);
    const parent = new Map<number, number>();
    const open = new Set<number>([startId]);

    while (open.size > 0) {
      let current = -1;
      let minG = Infinity;
      for (const id of open) {
        if ((gScore.get(id) ?? Infinity) < minG) {
          minG = gScore.get(id)!;
          current = id;
        }
      }
      if (current === goalId) {
        const path: number[] = [];
        let c = goalId;
        while (c !== startId) {
          path.unshift(c);
          c = parent.get(c)!;
        }
        path.unshift(startId);
        return path;
      }
      open.delete(current);
      for (const neighbor of this.highLevelGraph.get(current) ?? []) {
        const tentativeG = (gScore.get(current) ?? Infinity) + neighbor.cost;
        if (tentativeG < (gScore.get(neighbor.chunkId) ?? Infinity)) {
          gScore.set(neighbor.chunkId, tentativeG);
          parent.set(neighbor.chunkId, current);
          open.add(neighbor.chunkId);
        }
      }
    }
    return null;
  }

  private stitchLocalPaths(chunkPath: number[], start: Vector3, end: Vector3): Vector3[] {
    const fullPath: Vector3[] = [start];
    for (let i = 0; i < chunkPath.length - 1; i++) {
      const chunk = this.chunks[chunkPath[i]];
      const portals = chunk.boundaryPortals.get(chunkPath[i + 1]);
      if (portals && portals.length > 0) {
        const portal = portals[0];
        const midpoint: Vector3 = {
          x: (portal.left.x + portal.right.x) / 2,
          y: (portal.left.y + portal.right.y) / 2,
          z: (portal.left.z + portal.right.z) / 2,
        };
        fullPath.push(midpoint);
      }
    }
    fullPath.push(end);
    return fullPath;
  }
}

小结
    

  • 导航网格是三维游戏寻路的工业标准，将可行走区域表示为凸多边形集合
    • 

  • 生成流程包括体素化、区域划分、轮廓提取和多边形构建四个阶段
    • 

  • 多边形邻接图上使用 A* 搜索找到通道，漏斗算法将通道转化为平滑路径
    • 

  • 层级导航网格通过区块划分解决大型地图的寻路性能问题
    • 

  • 关键参数包括 cellSize、cellHeight、walkableSlope 和 walkableHeight
    • 

  • Recast/Detour 是最广泛使用的开源导航网格库，Unity 和 Unreal 均基于其实现
