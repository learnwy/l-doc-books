A* 寻路算法

核心概念

A*（A-Star）是游戏开发中最经典、应用最广泛的寻路算法。它结合了 Dijkstra 算法的最优性保证和贪心最佳优先搜索的效率，通过启发式函数 h(n) 引导搜索方向，在保证找到最短路径的同时显著减少搜索节点数量。

A* 算法实现
A* 的核心是维护两个集合：开放列表（待探索节点）和关闭列表（已探索节点）。每次从开放列表中取出 f 值最小的节点，对其邻居进行松弛操作。


type Heuristic = (a: number, b: number, width: number) => number;

const heuristics = {
  manhattan(a: number, b: number, width: number): number {
    const ax = a % width,
      ay = Math.floor(a / width);
    const bx = b % width,
      by = Math.floor(b / width);
    return Math.abs(ax - bx) + Math.abs(ay - by);
  },

  euclidean(a: number, b: number, width: number): number {
    const ax = a % width,
      ay = Math.floor(a / width);
    const bx = b % width,
      by = Math.floor(b / width);
    return Math.sqrt((ax - bx) ** 2 + (ay - by) ** 2);
  },

  octile(a: number, b: number, width: number): number {
    const ax = a % width,
      ay = Math.floor(a / width);
    const bx = b % width,
      by = Math.floor(b / width);
    const dx = Math.abs(ax - bx);
    const dy = Math.abs(ay - by);
    return dx + dy + (Math.SQRT2 - 2) * Math.min(dx, dy);
  },
};

class AStar {
  private graph: GridGraph;
  private width: number;
  private heuristic: Heuristic;

  constructor(graph: GridGraph, width: number, heuristic: Heuristic = heuristics.octile) {
    this.graph = graph;
    this.width = width;
    this.heuristic = heuristic;
  }

  search(start: number, goal: number): { path: number[]; cost: number; nodesExplored: number } | null {
    const openSet = new MinHeap<number>();
    const gScore = new Map<number, number>([[start, 0]]);
    const fScore = new Map<number, number>();
    const parent = new Map<number, number>();
    const closed = new Set<number>();
    let nodesExplored = 0;

    const h = this.heuristic(start, goal, this.width);
    fScore.set(start, h);
    openSet.push(start, h);

    while (openSet.size > 0) {
      const current = openSet.pop()!;
      nodesExplored++;

      if (current === goal) {
        return {
          path: this.reconstructPath(parent, start, goal),
          cost: gScore.get(goal)!,
          nodesExplored,
        };
      }

      if (closed.has(current)) continue;
      closed.add(current);

      for (const neighbor of this.graph.getNeighbors(current)) {
        if (closed.has(neighbor.id)) continue;

        const tentativeG = gScore.get(current)! + neighbor.cost;
        if (!gScore.has(neighbor.id) || tentativeG < gScore.get(neighbor.id)!) {
          gScore.set(neighbor.id, tentativeG);
          const f = tentativeG + this.heuristic(neighbor.id, goal, this.width);
          fScore.set(neighbor.id, f);
          parent.set(neighbor.id, current);
          openSet.push(neighbor.id, f);
        }
      }
    }
    return null;
  }

  private reconstructPath(parent: Map<number, number>, start: number, goal: number): number[] {
    const path: number[] = [];
    let current = goal;
    while (current !== start) {
      path.unshift(current);
      current = parent.get(current)!;
    }
    path.unshift(start);
    return path;
  }
}
启发式函数的选择对性能至关重要：当 h(n) 是可容纳的（admissible，即不超过真实代价）时，A* 保证找到最优路径。八向网格推荐使用 Octile 距离。


启发式函数与权重
加权 A*（Weighted A*）通过放大启发式函数的权重 w 来加速搜索，代价是路径质量的轻微下降。当 w > 1 时，算法优先向目标方向搜索，减少探索的节点数量，但路径长度最多为最优解的 w 倍。


class WeightedAStar extends AStar {
  private weight: number;

  constructor(graph: GridGraph, width: number, weight: number) {
    super(graph, width);
    this.weight = weight;
  }

  searchWeighted(start: number, goal: number): { path: number[]; cost: number } | null {
    const openSet = new MinHeap<number>();
    const gScore = new Map<number, number>([[start, 0]]);
    const parent = new Map<number, number>();
    const closed = new Set<number>();

    openSet.push(start, this.weight * heuristics.octile(start, goal, 0));

    while (openSet.size > 0) {
      const current = openSet.pop()!;
      if (current === goal) {
        return {
          path: this.reconstructPathW(parent, start, goal),
          cost: gScore.get(goal)!,
        };
      }

      if (closed.has(current)) continue;
      closed.add(current);

      for (const neighbor of this.graph.getNeighbors(current)) {
        if (closed.has(neighbor.id)) continue;
        const tentativeG = gScore.get(current)! + neighbor.cost;
        if (!gScore.has(neighbor.id) || tentativeG < gScore.get(neighbor.id)!) {
          gScore.set(neighbor.id, tentativeG);
          const f = tentativeG + this.weight * heuristics.octile(neighbor.id, goal, 0);
          parent.set(neighbor.id, current);
          openSet.push(neighbor.id, f);
        }
      }
    }
    return null;
  }

  private reconstructPathW(parent: Map<number, number>, start: number, goal: number): number[] {
    const path: number[] = [];
    let current = goal;
    while (current !== start) {
      path.unshift(current);
      current = parent.get(current)!;
    }
    path.unshift(start);
    return path;
  }
}
路径平滑
A* 在网格上找到的路径通常呈锯齿状，不适合直接用于角色移动。路径平滑通过视线检测（Line of Sight）移除不必要的中间节点，使路径更自然。


class PathSmoother {
  static smooth(path: number[], graph: GridGraph, width: number): number[] {
    if (path.length <= 2) return path;

    const smoothed: number[] = [path[0]];
    let current = 0;

    while (current < path.length - 1) {
      let furthest = current + 1;
      for (let i = path.length - 1; i > current + 1; i--) {
        if (PathSmoother.hasLineOfSight(path[current], path[i], graph, width)) {
          furthest = i;
          break;
        }
      }
      smoothed.push(path[furthest]);
      current = furthest;
    }

    return smoothed;
  }

  static hasLineOfSight(from: number, to: number, graph: GridGraph, width: number): boolean {
    let x0 = from % width,
      y0 = Math.floor(from / width);
    const x1 = to % width,
      y1 = Math.floor(to / width);
    const dx = Math.abs(x1 - x0),
      dy = Math.abs(y1 - y0);
    const sx = x0 < x1 ? 1 : -1,
      sy = y0 < y1 ? 1 : -1;
    let err = dx - dy;

    while (true) {
      const nodeId = y0 * width + x0;
      const neighbors = graph.getNeighbors(nodeId);
      if (neighbors.length === 0 && nodeId !== from && nodeId !== to) return false;

      if (x0 === x1 && y0 === y1) break;
      const e2 = 2 * err;
      if (e2 > -dy) {
        err -= dy;
        x0 += sx;
      }
      if (e2 < dx) {
        err += dx;
        y0 += sy;
      }
    }
    return true;
  }
}

class FunnelAlgorithm {
  static smooth(path: Vector2[], portals: [Vector2, Vector2][]): Vector2[] {
    const smoothed: Vector2[] = [path[0]];
    let apex = path[0];
    let left = portals[0][0];
    let right = portals[0][1];
    let apexIdx = 0,
      leftIdx = 0,
      rightIdx = 0;

    for (let i = 1; i < portals.length; i++) {
      const newLeft = portals[i][0];
      const newRight = portals[i][1];

      if (FunnelAlgorithm.cross(apex, right, newRight) <= 0) {
        if (apex.equals(right) || FunnelAlgorithm.cross(apex, left, newRight) > 0) {
          right = newRight;
          rightIdx = i;
        } else {
          smoothed.push(left);
          apex = left;
          apexIdx = leftIdx;
          left = apex;
          right = apex;
          leftIdx = apexIdx;
          rightIdx = apexIdx;
          i = apexIdx;
          continue;
        }
      }

      if (FunnelAlgorithm.cross(apex, left, newLeft) >= 0) {
        if (apex.equals(left) || FunnelAlgorithm.cross(apex, right, newLeft) < 0) {
          left = newLeft;
          leftIdx = i;
        } else {
          smoothed.push(right);
          apex = right;
          apexIdx = rightIdx;
          left = apex;
          right = apex;
          leftIdx = apexIdx;
          rightIdx = apexIdx;
          i = apexIdx;
          continue;
        }
      }
    }

    smoothed.push(path[path.length - 1]);
    return smoothed;
  }

  private static cross(o: Vector2, a: Vector2, b: Vector2): number {
    return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
  }
}

小结
    

  • A* 通过 f(n)=g(n)+h(n) 结合了 Dijkstra 的最优性和启发式搜索的高效性
    • 

  • 启发式函数必须满足可容纳性（不高估真实代价）才能保证最优解
    • 

  • 加权 A* 以路径质量为代价换取搜索速度，适合实时游戏
    • 

  • 路径平滑通过视线检测和 Funnel 算法将锯齿路径转化为自然运动轨迹
    • 

  • 优先队列的实现质量直接影响 A* 的实际性能
