屏幕空间效果

核心概念

屏幕空间效果（Screen Space Effects）是一类基于屏幕像素信息（颜色缓冲、深度缓冲、法线缓冲、运动向量）进行计算的渲染技术。由于仅使用二维屏幕数据而非完整的三维场景，屏幕空间效果具有与场景复杂度无关的固定性能开销，同时不需要额外的场景数据结构。其局限在于无法获取屏幕外或被遮挡的信息，需要通过回退策略（Fallback）来处理边界情况。

屏幕空间方法的共同特征是利用 GPU 的纹理采样能力在二维空间高效计算三维信息的近似，用局部信息推断全局效果。


屏幕空间反射
屏幕空间反射（SSR, Screen Space Reflections）通过在屏幕空间沿反射方向步进光线，寻找与场景深度相交的位置，从而获取反射颜色。SSR 能够提供精确的平面反射和光泽反射效果，是 PBR 管线中镜面间接光的重要来源。


基础 SSR 使用线性光线步进（Linear Ray Marching），但效率较低。Hi-Z 加速利用深度缓冲的 Mipmap 链实现分层步进，先在低分辨率层级大步跳跃，命中后在高分辨率层级精确求交：


#version 450

layout(set = 0, binding = 0) uniform sampler2D colorBuffer;
layout(set = 0, binding = 1) uniform sampler2D depthBuffer;
layout(set = 0, binding = 2) uniform sampler2D normalBuffer;

layout(set = 0, binding = 3) uniform SSRParams {
    mat4 projection;
    mat4 invProjection;
    mat4 viewMatrix;
    float maxDistance;
    float thickness;
    int maxSteps;
    int maxBinarySteps;
};

layout(location = 0) in vec2 texCoord;
layout(location = 0) out vec4 outReflection;

vec3 viewPosFromDepth(vec2 uv) {
    float depth = texture(depthBuffer, uv).r;
    vec4 clip = vec4(uv * 2.0 - 1.0, depth, 1.0);
    vec4 view = invProjection * clip;
    return view.xyz / view.w;
}

vec2 projectToScreen(vec3 viewPos) {
    vec4 clip = projection * vec4(viewPos, 1.0);
    return (clip.xy / clip.w) * 0.5 + 0.5;
}

bool traceRay(vec3 origin, vec3 direction, out vec2 hitUV, out float hitDepth) {
    vec3 step = direction * (maxDistance / float(maxSteps));
    vec3 current = origin;

    for (int i = 0; i < maxSteps; i++) {
        current += step;
        vec2 screenUV = projectToScreen(current);

        if (screenUV.x < 0.0 || screenUV.x > 1.0 ||
            screenUV.y < 0.0 || screenUV.y > 1.0) return false;

        float sceneDepth = viewPosFromDepth(screenUV).z;
        float diff = current.z - sceneDepth;

        if (diff > 0.0 && diff < thickness) {
            vec3 lo = current - step;
            vec3 hi = current;
            for (int j = 0; j < maxBinarySteps; j++) {
                vec3 mid = (lo + hi) * 0.5;
                vec2 midUV = projectToScreen(mid);
                float midSceneDepth = viewPosFromDepth(midUV).z;
                if (mid.z > midSceneDepth) { hi = mid; } else { lo = mid; }
            }
            hitUV = projectToScreen((lo + hi) * 0.5);
            hitDepth = length(current - origin);
            return true;
        }
    }
    return false;
}

void main() {
    vec3 viewPos = viewPosFromDepth(texCoord);
    vec3 viewNormal = mat3(viewMatrix) * (texture(normalBuffer, texCoord).rgb * 2.0 - 1.0);
    vec3 reflectDir = reflect(normalize(viewPos), normalize(viewNormal));

    vec2 hitUV;
    float hitDist;
    if (traceRay(viewPos, reflectDir, hitUV, hitDist)) {
        vec3 reflectedColor = texture(colorBuffer, hitUV).rgb;
        float fadeEdge = 1.0 - smoothstep(0.0, 0.1, max(
            max(abs(hitUV.x - 0.5) - 0.4, 0.0),
            max(abs(hitUV.y - 0.5) - 0.4, 0.0)
        ));
        float fadeDist = 1.0 - hitDist / maxDistance;
        outReflection = vec4(reflectedColor * fadeEdge * fadeDist, 1.0);
    } else {
        outReflection = vec4(0.0);
    }
}
SSR 中二分搜索（Binary Search）细化是提高精度的关键步骤。初始线性步进确定大致交点后，二分搜索在最后一步的区间内精确定位交点。边缘淡出（Edge Fade）在屏幕边缘降低反射强度，避免反射突然截断的视觉突变。


粗糙表面的反射需要多条光线的模糊效果。常用的方案是根据粗糙度调整反射方向的散射角度，使用重要性采样（Importance Sampling）生成多条扰动后的反射光线，或者对 SSR 结果进行粗糙度相关的后模糊。


运动模糊
运动模糊（Motion Blur）模拟相机快门曝光期间场景运动产生的拖影效果。运动模糊增强画面的动感和速度感，也能在一定程度上掩盖低帧率带来的顿挫感。


基于运动向量的运动模糊（Per-Object Motion Blur）是目前最主流的方案。运动向量（Motion Vector / Velocity Buffer）记录每个像素从当前帧到上一帧的屏幕空间位移，在片段着色器中通过当前帧和上一帧的 MVP 矩阵计算：


#version 450

layout(set = 0, binding = 0) uniform sampler2D colorBuffer;
layout(set = 0, binding = 1) uniform sampler2D motionVectorBuffer;
layout(set = 0, binding = 2) uniform sampler2D depthBuffer;

layout(set = 0, binding = 3) uniform MotionBlurParams {
    float intensity;
    int sampleCount;
    float maxVelocity;
};

layout(location = 0) in vec2 texCoord;
layout(location = 0) out vec4 outColor;

void main() {
    vec2 velocity = texture(motionVectorBuffer, texCoord).rg;
    float speed = length(velocity);

    if (speed < 0.001) {
        outColor = texture(colorBuffer, texCoord);
        return;
    }

    velocity = normalize(velocity) * min(speed, maxVelocity);
    vec2 step = velocity * intensity / float(sampleCount);

    vec3 color = vec3(0.0);
    float totalWeight = 0.0;
    vec2 currentUV = texCoord - step * float(sampleCount) * 0.5;

    for (int i = 0; i < sampleCount; i++) {
        float weight = 1.0 - abs(float(i) - float(sampleCount) * 0.5) / (float(sampleCount) * 0.5);
        color += texture(colorBuffer, clamp(currentUV, 0.0, 1.0)).rgb * weight;
        totalWeight += weight;
        currentUV += step;
    }

    outColor = vec4(color / totalWeight, 1.0);
}
运动向量的计算需要同时处理相机运动和物体运动。对于蒙皮动画的角色，运动向量需要逐顶点计算（当前帧蒙皮位置与上一帧蒙皮位置的差）。

屏幕空间接触阴影
屏幕空间接触阴影（Screen Space Contact Shadow）弥补了阴影贴图在小尺度细节上分辨率不足的问题。它从片段位置沿光源方向在屏幕空间短距离步进，检测是否有遮挡物，为手指缝隙、按钮边缘、角色脚底等提供精确的接触阴影。


#version 450

layout(set = 0, binding = 0) uniform sampler2D depthBuffer;

layout(set = 0, binding = 1) uniform ContactShadowParams {
    mat4 viewMatrix;
    mat4 projection;
    vec3 lightDirView;
    float maxDistance;
    float thickness;
    int stepCount;
};

layout(location = 0) in vec2 texCoord;
layout(location = 0) out float outShadow;

vec3 viewPosFromDepth(vec2 uv) {
    float depth = texture(depthBuffer, uv).r;
    vec4 clip = vec4(uv * 2.0 - 1.0, depth, 1.0);
    vec4 view = inverse(projection) * clip;
    return view.xyz / view.w;
}

void main() {
    vec3 viewPos = viewPosFromDepth(texCoord);
    vec3 rayDir = normalize(-lightDirView);
    vec3 step = rayDir * maxDistance / float(stepCount);
    vec3 current = viewPos + step;

    float shadow = 0.0;
    for (int i = 0; i < stepCount; i++) {
        vec4 proj = projection * vec4(current, 1.0);
        vec2 sampleUV = (proj.xy / proj.w) * 0.5 + 0.5;

        if (sampleUV.x < 0.0 || sampleUV.x > 1.0 ||
            sampleUV.y < 0.0 || sampleUV.y > 1.0) break;

        float sceneDepth = viewPosFromDepth(sampleUV).z;
        float diff = current.z - sceneDepth;

        if (diff > 0.0 && diff < thickness) {
            float fadeout = 1.0 - float(i) / float(stepCount);
            shadow = max(shadow, fadeout);
            break;
        }
        current += step;
    }

    outShadow = 1.0 - shadow;
}
接触阴影的步进距离很短（通常 0.10.5 米），步数也较少（832 步），因此性能开销很低。它与阴影贴图的结果相乘即可获得兼具远距离大范围阴影和近距离小尺度阴影的完整效果。


体积光与 God Rays
体积光（Volumetric Lighting）模拟光线穿过散射介质（雾、灰尘、烟雾）时产生的可见光束效果，也称为 God Rays 或丁达尔效应。屏幕空间的体积光实现通过从光源位置向每个像素方向进行径向采样来模拟光线散射。

通用体积效果





Froxel（Frustum + Voxel）方法将视锥体体素化为三维网格，在每个体素中累积散射和吸收，然后沿视线方向积分得到最终的体积光效果。这是 Unreal Engine 和 Unity HDRP 中体积雾的标准实现方案。


体积光的散射使用 Henyey-Greenstein 相函数来控制散射方向的分布，参数 g 控制前向散射（正值）和后向散射（负值）的比例。太阳光的前向散射效果（g ≈ 0.7~0.9）产生明显的光束可见效果。


小结
    

  • 屏幕空间效果利用二维屏幕缓冲信息近似三维效果，性能与场景复杂度无关
    • 

  • SSR 通过屏幕空间光线步进实现反射，二分搜索和 Hi-Z 加速是效率和精度的关键
    • 

  • 运动模糊基于运动向量沿速度方向采样，需要处理相机和物体两种运动来源
    • 

  • 接触阴影以极低开销弥补阴影贴图在小尺度细节上的不足
    • 

  • 体积光通过散射积分模拟可见光束，Froxel 方法是现代引擎的标准方案
    • 

  • 所有屏幕空间效果都面临信息缺失问题，需要边缘淡出和回退策略来处理边界情况
    •
