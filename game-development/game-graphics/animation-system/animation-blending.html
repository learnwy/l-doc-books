动画混合

核心概念

动画混合（Animation Blending）是将多个动画片段的骨骼姿态按权重组合的技术，使角色能够在不同动画状态之间平滑过渡、叠加上半身和下半身的独立动作、以及根据参数连续混合多个动画。动画混合是角色动画系统复杂行为的基础——没有混合技术，角色在切换动画时会出现突兀的跳变。

所有混合操作的数学基础都是逐骨骼的姿态插值：平移分量使用线性插值，旋转分量使用四元数球面线性插值（Slerp），缩放分量使用线性插值。


交叉淡入淡出
交叉淡入淡出（Crossfade）是最基础的动画过渡方式。当角色从动画 A 切换到动画 B 时，在设定的过渡时间内，动画 A 的权重从 1 线性下降到 0，动画 B 的权重从 0 上升到 1。


interface AnimationState {
  clip: AnimationClip;
  time: number;
  weight: number;
  speed: number;
}

interface CrossfadeTransition {
  from: AnimationState;
  to: AnimationState;
  duration: number;
  elapsed: number;
}

interface AnimationClip {
  name: string;
  duration: number;
  channels: AnimationChannel[];
}

interface AnimationChannel {
  jointIndex: number;
  sample(time: number): JointPose;
}

interface JointPose {
  translation: [number, number, number];
  rotation: [number, number, number, number];
  scale: [number, number, number];
}

function updateCrossfade(transition: CrossfadeTransition, deltaTime: number): JointPose[] {
  transition.elapsed += deltaTime;
  const t = Math.min(transition.elapsed / transition.duration, 1.0);

  transition.from.time += deltaTime * transition.from.speed;
  transition.to.time += deltaTime * transition.to.speed;

  const fromWeight = 1.0 - smoothStep(t);
  const toWeight = smoothStep(t);

  const fromPoses = sampleAnimation(transition.from.clip, transition.from.time);
  const toPoses = sampleAnimation(transition.to.clip, transition.to.time);

  return blendPoses(fromPoses, toPoses, toWeight);
}

function smoothStep(t: number): number {
  return t * t * (3 - 2 * t);
}

function blendPoses(a: JointPose[], b: JointPose[], weight: number): JointPose[] {
  return a.map((poseA, i) => ({
    translation: lerpVec3(poseA.translation, b[i].translation, weight),
    rotation: slerpQuat(poseA.rotation, b[i].rotation, weight),
    scale: lerpVec3(poseA.scale, b[i].scale, weight),
  }));
}

function sampleAnimation(clip: AnimationClip, time: number): JointPose[] {
  const wrappedTime = time % clip.duration;
  return clip.channels.map((ch) => ch.sample(wrappedTime));
}

function lerpVec3(a: [number, number, number], b: [number, number, number], t: number): [number, number, number] {
  return [a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t, a[2] + (b[2] - a[2]) * t];
}

function slerpQuat(
  a: [number, number, number, number],
  b: [number, number, number, number],
  t: number,
): [number, number, number, number] {
  let dot = a[0] * b[0] + a[1] * b[1] + a[2] * b[2] + a[3] * b[3];
  const nb = dot < 0 ? ([-b[0], -b[1], -b[2], -b[3]] as [number, number, number, number]) : b;
  dot = Math.abs(dot);
  if (dot > 0.9995) {
    const r: [number, number, number, number] = [
      a[0] + (nb[0] - a[0]) * t,
      a[1] + (nb[1] - a[1]) * t,
      a[2] + (nb[2] - a[2]) * t,
      a[3] + (nb[3] - a[3]) * t,
    ];
    const len = Math.sqrt(r[0] * r[0] + r[1] * r[1] + r[2] * r[2] + r[3] * r[3]);
    return [r[0] / len, r[1] / len, r[2] / len, r[3] / len];
  }
  const theta = Math.acos(dot);
  const wa = Math.sin((1 - t) * theta) / Math.sin(theta);
  const wb = Math.sin(t * theta) / Math.sin(theta);
  return [wa * a[0] + wb * nb[0], wa * a[1] + wb * nb[1], wa * a[2] + wb * nb[2], wa * a[3] + wb * nb[3]];
}
使用 smoothStep 而非线性函数作为权重曲线，可以使过渡的起始和结束更加平缓自然。过渡时间的选择取决于动画的风格和节奏——快节奏的动作游戏通常使用 0.10.2 秒的短过渡，角色扮演游戏可以使用 0.30.5 秒的长过渡。


分层混合与骨骼遮罩
分层混合（Layered Blending）允许不同骨骼区域播放不同的动画。典型场景是角色上半身播放射击动画，下半身播放跑步动画。骨骼遮罩（Bone Mask）定义每根骨骼在各层中的混合权重。

加法混合（Additive Blending）的原理是计算动画姿态与参考姿态的差异，然后将这个差异叠加到当前姿态上。呼吸动画的参考姿态是静止站立，呼吸动画记录了身体随呼吸起伏的微小变化，这个变化可以叠加到任何基础动画上。


interface AnimationLayer {
  clip: AnimationClip;
  time: number;
  weight: number;
  blendMode: 'override' | 'additive';
  boneMask: Float32Array;
}

function evaluateLayers(
  layers: AnimationLayer[],
  skeleton: { jointCount: number },
  referencePose: JointPose[],
): JointPose[] {
  let result = sampleAnimation(layers[0].clip, layers[0].time);

  for (let l = 1; l < layers.length; l++) {
    const layer = layers[l];
    const layerPose = sampleAnimation(layer.clip, layer.time);

    for (let j = 0; j < skeleton.jointCount; j++) {
      const maskWeight = layer.boneMask[j] * layer.weight;
      if (maskWeight < 0.001) continue;

      if (layer.blendMode === 'override') {
        result[j] = {
          translation: lerpVec3(result[j].translation, layerPose[j].translation, maskWeight),
          rotation: slerpQuat(result[j].rotation, layerPose[j].rotation, maskWeight),
          scale: lerpVec3(result[j].scale, layerPose[j].scale, maskWeight),
        };
      } else {
        const diff = computeAdditiveDiff(referencePose[j], layerPose[j]);
        result[j] = applyAdditive(result[j], diff, maskWeight);
      }
    }
  }
  return result;
}

function computeAdditiveDiff(ref: JointPose, current: JointPose): JointPose {
  return {
    translation: [
      current.translation[0] - ref.translation[0],
      current.translation[1] - ref.translation[1],
      current.translation[2] - ref.translation[2],
    ],
    rotation: quatMultiply(quatInverse(ref.rotation), current.rotation),
    scale: [current.scale[0] / ref.scale[0], current.scale[1] / ref.scale[1], current.scale[2] / ref.scale[2]],
  };
}

function applyAdditive(base: JointPose, diff: JointPose, weight: number): JointPose {
  return {
    translation: [
      base.translation[0] + diff.translation[0] * weight,
      base.translation[1] + diff.translation[1] * weight,
      base.translation[2] + diff.translation[2] * weight,
    ],
    rotation: slerpQuat(base.rotation, quatMultiply(base.rotation, diff.rotation), weight),
    scale: [
      base.scale[0] * (1 + (diff.scale[0] - 1) * weight),
      base.scale[1] * (1 + (diff.scale[1] - 1) * weight),
      base.scale[2] * (1 + (diff.scale[2] - 1) * weight),
    ],
  };
}

function quatMultiply(
  a: [number, number, number, number],
  b: [number, number, number, number],
): [number, number, number, number] {
  return [
    a[3] * b[0] + a[0] * b[3] + a[1] * b[2] - a[2] * b[1],
    a[3] * b[1] - a[0] * b[2] + a[1] * b[3] + a[2] * b[0],
    a[3] * b[2] + a[0] * b[1] - a[1] * b[0] + a[2] * b[3],
    a[3] * b[3] - a[0] * b[0] - a[1] * b[1] - a[2] * b[2],
  ];
}

function quatInverse(q: [number, number, number, number]): [number, number, number, number] {
  return [-q[0], -q[1], -q[2], q[3]];
}
混合空间
混合空间（Blend Space）根据连续的输入参数在多个动画之间自动插值。1D 混合空间使用单个参数（如速度）在一系列动画之间线性插值；2D 混合空间使用两个参数（如速度和方向）在二维平面上的采样点之间进行三角形重心插值。


1D 混合空间的典型应用是运动速度控制。在参数轴上放置步行（speed=1.5）、慢跑（speed=3.0）、奔跑（speed=6.0）三个采样点，当速度为 4.0 时自动在慢跑和奔跑之间以 (6.0-4.0)/(6.0-3.0) = 0.67 的权重混合。


2D 混合空间最常用于八方向运动。参数空间的横轴为运动方向角度，纵轴为速度。在关键方向（前、后、左、右、左前、右前、左后、右后）放置对应的运动动画，输入任意方向和速度时通过三角形插值产生连续的混合动画。

动画状态机
动画状态机（Animation State Machine, ASM）是组织角色动画逻辑的最高层抽象。每个状态对应一个动画行为（播放单个片段、混合空间、子状态机），状态之间的转换由条件触发，转换过程使用交叉淡入淡出。

状态机可以层级嵌套。例如"地面移动"是一个子状态机，内部包含 Idle、Walk、Run 状态和它们之间的转换。在父状态机层级，"地面移动"和"空中"是两个顶层状态，跳跃/落地触发它们之间的转换。


动画事件（Animation Event）是在动画时间轴特定帧上触发的回调，常用于脚步声音同步、伤害判定帧、特效生成时机等。事件与动画数据一起存储，由动画系统在播放到对应帧时触发。


小结
    

  • 动画混合通过逐骨骼姿态插值实现多个动画的组合，是角色动画系统的基础能力
    • 

  • 交叉淡入淡出使用时间权重在两个动画间平滑过渡，SmoothStep 曲线提升自然感
    • 

  • 分层混合通过骨骼遮罩实现身体不同部位播放不同动画，加法混合叠加微小变化
    • 

  • 混合空间根据输入参数自动在多个动画间连续插值，是运动系统的核心技术
    • 

  • 动画状态机提供了组织复杂动画逻辑的高层抽象，支持层级嵌套和条件转换
    • 

  • 动画事件在特定帧触发回调，实现音效、特效与动画的精确同步
