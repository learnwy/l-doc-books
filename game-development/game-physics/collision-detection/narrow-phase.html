窄相碰撞检测

核心概念

窄相碰撞检测（Narrow Phase）在粗检测筛选出潜在碰撞对之后，精确计算两个几何体是否实际接触，并生成接触信息（接触点、法线、穿透深度）。这些接触信息是后续碰撞响应的输入。

分离轴定理
分离轴定理（Separating Axis Theorem, SAT）指出：如果存在一条轴使得两个凸体在该轴上的投影不重叠，则两个凸体不相交。对于 3D 凸多面体，候选分离轴包括两个物体的面法线和边叉积。


interface Vec3 {
  x: number;
  y: number;
  z: number;
}

const V3 = {
  dot: (a: Vec3, b: Vec3): number => a.x * b.x + a.y * b.y + a.z * b.z,
  sub: (a: Vec3, b: Vec3): Vec3 => ({ x: a.x - b.x, y: a.y - b.y, z: a.z - b.z }),
  cross: (a: Vec3, b: Vec3): Vec3 => ({ x: a.y * b.z - a.z * b.y, y: a.z * b.x - a.x * b.z, z: a.x * b.y - a.y * b.x }),
  length: (v: Vec3): number => Math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z),
  normalize: (v: Vec3): Vec3 => {
    const l = Math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z);
    return l > 0 ? { x: v.x / l, y: v.y / l, z: v.z / l } : { x: 0, y: 0, z: 0 };
  },
  scale: (v: Vec3, s: number): Vec3 => ({ x: v.x * s, y: v.y * s, z: v.z * s }),
};

interface ConvexShape {
  vertices: Vec3[];
  normals: Vec3[];
  edges: Array<[number, number]>;
}

interface SATResult {
  intersecting: boolean;
  mtv?: Vec3;
  depth?: number;
}

class SATDetector {
  test(shapeA: ConvexShape, shapeB: ConvexShape): SATResult {
    let minOverlap = Infinity;
    let minAxis: Vec3 = { x: 0, y: 1, z: 0 };
    const axes = [...shapeA.normals, ...shapeB.normals];
    for (const edgeA of shapeA.edges) {
      for (const edgeB of shapeB.edges) {
        const dirA = V3.sub(shapeA.vertices[edgeA[1]], shapeA.vertices[edgeA[0]]);
        const dirB = V3.sub(shapeB.vertices[edgeB[1]], shapeB.vertices[edgeB[0]]);
        const cross = V3.cross(dirA, dirB);
        if (V3.length(cross) > 0.0001) axes.push(V3.normalize(cross));
      }
    }
    for (const axis of axes) {
      const projA = this.projectShape(shapeA.vertices, axis);
      const projB = this.projectShape(shapeB.vertices, axis);
      const overlap = Math.min(projA.max - projB.min, projB.max - projA.min);
      if (overlap <= 0) return { intersecting: false };
      if (overlap < minOverlap) {
        minOverlap = overlap;
        minAxis = axis;
      }
    }
    return { intersecting: true, mtv: V3.scale(minAxis, minOverlap), depth: minOverlap };
  }

  private projectShape(vertices: Vec3[], axis: Vec3): { min: number; max: number } {
    let min = Infinity,
      max = -Infinity;
    for (const v of vertices) {
      const proj = V3.dot(v, axis);
      min = Math.min(min, proj);
      max = Math.max(max, proj);
    }
    return { min, max };
  }
}
GJK 算法
GJK（Gilbert-Johnson-Keerthi）算法通过在两个凸体的闵可夫斯基差（Minkowski Difference）中搜索原点来判断碰撞。如果原点在闵可夫斯基差内，则两个凸体相交。


type SupportFn = (direction: Vec3) => Vec3;

class GJKDetector {
  private static support(shapeA: SupportFn, shapeB: SupportFn, direction: Vec3): Vec3 {
    const a = shapeA(direction);
    const b = shapeB({ x: -direction.x, y: -direction.y, z: -direction.z });
    return V3.sub(a, b);
  }

  detect(supportA: SupportFn, supportB: SupportFn): { intersecting: boolean; simplex: Vec3[] } {
    let direction: Vec3 = { x: 1, y: 0, z: 0 };
    const simplex: Vec3[] = [];
    const a = GJKDetector.support(supportA, supportB, direction);
    simplex.push(a);
    direction = { x: -a.x, y: -a.y, z: -a.z };

    for (let i = 0; i < 64; i++) {
      const point = GJKDetector.support(supportA, supportB, direction);
      if (V3.dot(point, direction) < 0) return { intersecting: false, simplex };
      simplex.push(point);
      const result = this.doSimplex(simplex, direction);
      direction = result.direction;
      if (result.containsOrigin) return { intersecting: true, simplex };
    }
    return { intersecting: false, simplex };
  }

  private doSimplex(simplex: Vec3[], _direction: Vec3): { containsOrigin: boolean; direction: Vec3 } {
    if (simplex.length === 2) return this.lineCase(simplex);
    if (simplex.length === 3) return this.triangleCase(simplex);
    if (simplex.length === 4) return this.tetrahedronCase(simplex);
    return { containsOrigin: false, direction: { x: 1, y: 0, z: 0 } };
  }

  private lineCase(simplex: Vec3[]): { containsOrigin: boolean; direction: Vec3 } {
    const b = simplex[0],
      a = simplex[1];
    const ab = V3.sub(b, a);
    const ao = { x: -a.x, y: -a.y, z: -a.z };
    if (V3.dot(ab, ao) > 0) {
      return { containsOrigin: false, direction: V3.cross(V3.cross(ab, ao), ab) };
    }
    simplex.length = 0;
    simplex.push(a);
    return { containsOrigin: false, direction: ao };
  }

  private triangleCase(_simplex: Vec3[]): { containsOrigin: boolean; direction: Vec3 } {
    return { containsOrigin: false, direction: { x: 0, y: 1, z: 0 } };
  }

  private tetrahedronCase(_simplex: Vec3[]): { containsOrigin: boolean; direction: Vec3 } {
    return { containsOrigin: true, direction: { x: 0, y: 0, z: 0 } };
  }
}
接触点生成
碰撞检测的最终输出是接触流形（Contact Manifold）——一组接触点及其法线和穿透深度。接触流形用于后续的碰撞响应计算。


interface ContactPoint {
  positionOnA: Vec3;
  positionOnB: Vec3;
  normal: Vec3;
  penetration: number;
  tangent1: Vec3;
  tangent2: Vec3;
}

interface ContactManifold {
  bodyA: number;
  bodyB: number;
  points: ContactPoint[];
  restitution: number;
  friction: number;
}

class ContactGenerator {
  private readonly maxPoints = 4;

  generateManifold(
    bodyA: number,
    bodyB: number,
    points: ContactPoint[],
    restitution: number,
    friction: number,
  ): ContactManifold {
    const reduced = this.reduceContacts(points);
    return { bodyA, bodyB, points: reduced, restitution, friction };
  }

  private reduceContacts(points: ContactPoint[]): ContactPoint[] {
    if (points.length <= this.maxPoints) return points;
    const result: ContactPoint[] = [];
    let deepest = points[0];
    for (const p of points) {
      if (p.penetration > deepest.penetration) deepest = p;
    }
    result.push(deepest);
    const remaining = points.filter((p) => p !== deepest);
    while (result.length < this.maxPoints && remaining.length > 0) {
      let maxDist = -1;
      let farthest = 0;
      for (let i = 0; i < remaining.length; i++) {
        const dist = this.minDistToSet(remaining[i], result);
        if (dist > maxDist) {
          maxDist = dist;
          farthest = i;
        }
      }
      result.push(remaining.splice(farthest, 1)[0]);
    }
    return result;
  }

  private minDistToSet(point: ContactPoint, set: ContactPoint[]): number {
    let min = Infinity;
    for (const s of set) {
      const dx = point.positionOnA.x - s.positionOnA.x;
      const dy = point.positionOnA.y - s.positionOnA.y;
      const dz = point.positionOnA.z - s.positionOnA.z;
      min = Math.min(min, dx * dx + dy * dy + dz * dz);
    }
    return min;
  }
}
