包围体

核心概念

包围体（Bounding Volume）是用简单几何形状包裹复杂物体的技术，用于快速排除不可能碰撞的物体对。在碰撞检测的粗检测阶段，引擎先检测包围体是否重叠，只有重叠的物体对才进入精确的细检测阶段。

AABB 轴对齐包围盒
AABB 由最小点和最大点定义，六个面分别平行于坐标轴。AABB 的相交测试只需比较各轴的区间是否重叠，是所有包围体中最快的测试。


interface Vec3 {
  x: number;
  y: number;
  z: number;
}

class AABB {
  min: Vec3;
  max: Vec3;

  constructor(min: Vec3, max: Vec3) {
    this.min = min;
    this.max = max;
  }

  static fromPoints(points: Vec3[]): AABB {
    const min = { x: Infinity, y: Infinity, z: Infinity };
    const max = { x: -Infinity, y: -Infinity, z: -Infinity };
    for (const p of points) {
      min.x = Math.min(min.x, p.x);
      min.y = Math.min(min.y, p.y);
      min.z = Math.min(min.z, p.z);
      max.x = Math.max(max.x, p.x);
      max.y = Math.max(max.y, p.y);
      max.z = Math.max(max.z, p.z);
    }
    return new AABB(min, max);
  }

  intersects(other: AABB): boolean {
    return (
      this.min.x <= other.max.x &&
      this.max.x >= other.min.x &&
      this.min.y <= other.max.y &&
      this.max.y >= other.min.y &&
      this.min.z <= other.max.z &&
      this.max.z >= other.min.z
    );
  }

  contains(point: Vec3): boolean {
    return (
      point.x >= this.min.x &&
      point.x <= this.max.x &&
      point.y >= this.min.y &&
      point.y <= this.max.y &&
      point.z >= this.min.z &&
      point.z <= this.max.z
    );
  }

  merge(other: AABB): AABB {
    return new AABB(
      {
        x: Math.min(this.min.x, other.min.x),
        y: Math.min(this.min.y, other.min.y),
        z: Math.min(this.min.z, other.min.z),
      },
      {
        x: Math.max(this.max.x, other.max.x),
        y: Math.max(this.max.y, other.max.y),
        z: Math.max(this.max.z, other.max.z),
      },
    );
  }

  expand(margin: number): AABB {
    return new AABB(
      { x: this.min.x - margin, y: this.min.y - margin, z: this.min.z - margin },
      { x: this.max.x + margin, y: this.max.y + margin, z: this.max.z + margin },
    );
  }

  getCenter(): Vec3 {
    return { x: (this.min.x + this.max.x) / 2, y: (this.min.y + this.max.y) / 2, z: (this.min.z + this.max.z) / 2 };
  }

  getSurfaceArea(): number {
    const dx = this.max.x - this.min.x;
    const dy = this.max.y - this.min.y;
    const dz = this.max.z - this.min.z;
    return 2 * (dx * dy + dy * dz + dz * dx);
  }
}
AABB 的主要缺点是旋转不变性——当物体旋转时，AABB 需要从变换后的顶点重新计算，可能变得比物体大很多，导致更多的误报（false positive）。


包围球
包围球由圆心和半径定义，是最简单的包围体。它具有旋转不变性——物体旋转后包围球不变，这是它相对于 AABB 的优势。


class BoundingSphere {
  center: Vec3;
  radius: number;

  constructor(center: Vec3, radius: number) {
    this.center = center;
    this.radius = radius;
  }

  static fromPoints(points: Vec3[]): BoundingSphere {
    let cx = 0,
      cy = 0,
      cz = 0;
    for (const p of points) {
      cx += p.x;
      cy += p.y;
      cz += p.z;
    }
    const n = points.length;
    const center = { x: cx / n, y: cy / n, z: cz / n };
    let maxRadiusSq = 0;
    for (const p of points) {
      const dx = p.x - center.x,
        dy = p.y - center.y,
        dz = p.z - center.z;
      maxRadiusSq = Math.max(maxRadiusSq, dx * dx + dy * dy + dz * dz);
    }
    return new BoundingSphere(center, Math.sqrt(maxRadiusSq));
  }

  intersects(other: BoundingSphere): boolean {
    const dx = this.center.x - other.center.x;
    const dy = this.center.y - other.center.y;
    const dz = this.center.z - other.center.z;
    const distSq = dx * dx + dy * dy + dz * dz;
    const radiusSum = this.radius + other.radius;
    return distSq <= radiusSum * radiusSum;
  }

  intersectsAABB(aabb: AABB): boolean {
    let distSq = 0;
    for (const axis of ['x', 'y', 'z'] as const) {
      const v = this.center[axis];
      if (v < aabb.min[axis]) distSq += (aabb.min[axis] - v) ** 2;
      if (v > aabb.max[axis]) distSq += (v - aabb.max[axis]) ** 2;
    }
    return distSq <= this.radius * this.radius;
  }

  containsPoint(point: Vec3): boolean {
    const dx = point.x - this.center.x;
    const dy = point.y - this.center.y;
    const dz = point.z - this.center.z;
    return dx * dx + dy * dy + dz * dz <= this.radius * this.radius;
  }

  merge(other: BoundingSphere): BoundingSphere {
    const dx = other.center.x - this.center.x;
    const dy = other.center.y - this.center.y;
    const dz = other.center.z - this.center.z;
    const dist = Math.sqrt(dx * dx + dy * dy + dz * dz);
    if (dist + other.radius <= this.radius) return this;
    if (dist + this.radius <= other.radius) return other;
    const newRadius = (dist + this.radius + other.radius) / 2;
    const t = (newRadius - this.radius) / dist;
    return new BoundingSphere(
      { x: this.center.x + dx * t, y: this.center.y + dy * t, z: this.center.z + dz * t },
      newRadius,
    );
  }
}
射线与包围体相交测试
射线-AABB 相交测试使用 slab 方法，将射线在每个轴上的参数区间取交集。如果最终区间有效（tMin <= tMax 且 tMax >= 0），则射线与 AABB 相交。


interface Ray {
  origin: Vec3;
  direction: Vec3;
}

class RayTests {
  static rayAABB(ray: Ray, aabb: AABB): { hit: boolean; tMin: number; tMax: number } {
    let tMin = -Infinity;
    let tMax = Infinity;
    for (const axis of ['x', 'y', 'z'] as const) {
      const invD = 1 / ray.direction[axis];
      let t0 = (aabb.min[axis] - ray.origin[axis]) * invD;
      let t1 = (aabb.max[axis] - ray.origin[axis]) * invD;
      if (invD < 0) {
        const tmp = t0;
        t0 = t1;
        t1 = tmp;
      }
      tMin = Math.max(tMin, t0);
      tMax = Math.min(tMax, t1);
      if (tMax < tMin) return { hit: false, tMin: 0, tMax: 0 };
    }
    return { hit: tMax >= 0, tMin: Math.max(0, tMin), tMax };
  }

  static raySphere(ray: Ray, sphere: BoundingSphere): { hit: boolean; t: number } {
    const ox = ray.origin.x - sphere.center.x;
    const oy = ray.origin.y - sphere.center.y;
    const oz = ray.origin.z - sphere.center.z;
    const a = ray.direction.x ** 2 + ray.direction.y ** 2 + ray.direction.z ** 2;
    const b = 2 * (ox * ray.direction.x + oy * ray.direction.y + oz * ray.direction.z);
    const c = ox ** 2 + oy ** 2 + oz ** 2 - sphere.radius ** 2;
    const discriminant = b * b - 4 * a * c;
    if (discriminant < 0) return { hit: false, t: 0 };
    const t = (-b - Math.sqrt(discriminant)) / (2 * a);
    return { hit: t >= 0, t: Math.max(0, t) };
  }
}
包围体的动态更新
动态物体的包围体需要随物体运动实时更新。"膨胀 AABB"（Fat AABB）策略在包围体周围添加边距，使物体在小范围移动时无需重新计算包围体和重新插入空间数据结构。


class DynamicAABBTracker {
  private fatMargin = 0.1;
  private fatAABBs = new Map<number, AABB>();
  private tightAABBs = new Map<number, AABB>();

  updateBody(id: number, tightAABB: AABB): boolean {
    this.tightAABBs.set(id, tightAABB);
    const fatAABB = this.fatAABBs.get(id);
    if (fatAABB && this.containsAABB(fatAABB, tightAABB)) {
      return false;
    }
    const newFat = tightAABB.expand(this.fatMargin);
    this.fatAABBs.set(id, newFat);
    return true;
  }

  getFatAABB(id: number): AABB | undefined {
    return this.fatAABBs.get(id);
  }

  private containsAABB(outer: AABB, inner: AABB): boolean {
    return (
      inner.min.x >= outer.min.x &&
      inner.max.x <= outer.max.x &&
      inner.min.y >= outer.min.y &&
      inner.max.y <= outer.max.y &&
      inner.min.z >= outer.min.z &&
      inner.max.z <= outer.max.z
    );
  }
}
小结
    

  • 包围体用简单形状快速排除不碰撞的物体对
    • 

  • AABB 相交测试最快，但旋转后可能过度膨胀
    • 

  • 包围球具有旋转不变性，适合动态物体
    • 

  • 射线-AABB 测试使用 slab 方法在三个轴上求参数交集
    • 

  • 膨胀 AABB 策略减少动态物体的空间数据结构更新频率
