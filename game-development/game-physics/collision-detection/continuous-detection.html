连续碰撞检测

核心概念

连续碰撞检测（Continuous Collision Detection, CCD）解决离散碰撞检测中高速物体"穿透"其他物体的问题。当物体在一帧内的移动距离超过自身尺寸时，离散检测可能完全错过碰撞。CCD 通过计算物体运动轨迹上的首次碰撞时间来避免穿透。

扫掠体积法
扫掠体积法（Swept Volume）将物体从起始位置到结束位置的运动轨迹视为一个扫掠体，然后检测扫掠体与其他物体的相交。


interface Vec3 {
  x: number;
  y: number;
  z: number;
}
interface Sphere {
  center: Vec3;
  radius: number;
}
interface AABB {
  min: Vec3;
  max: Vec3;
}

class SweptVolumeDetector {
  static sphereVsSphere(a: Sphere, velocityA: Vec3, b: Sphere, velocityB: Vec3): { hit: boolean; t: number } {
    const relVel = { x: velocityA.x - velocityB.x, y: velocityA.y - velocityB.y, z: velocityA.z - velocityB.z };
    const relPos = { x: a.center.x - b.center.x, y: a.center.y - b.center.y, z: a.center.z - b.center.z };
    const r = a.radius + b.radius;
    const aa = relVel.x ** 2 + relVel.y ** 2 + relVel.z ** 2;
    const bb = 2 * (relPos.x * relVel.x + relPos.y * relVel.y + relPos.z * relVel.z);
    const cc = relPos.x ** 2 + relPos.y ** 2 + relPos.z ** 2 - r * r;
    if (cc < 0) return { hit: true, t: 0 };
    const discriminant = bb * bb - 4 * aa * cc;
    if (discriminant < 0) return { hit: false, t: 1 };
    const t = (-bb - Math.sqrt(discriminant)) / (2 * aa);
    if (t >= 0 && t <= 1) return { hit: true, t };
    return { hit: false, t: 1 };
  }

  static sweptAABB(a: AABB, vel: Vec3, b: AABB): { hit: boolean; tEntry: number; normal: Vec3 } {
    const invVel = {
      x: vel.x !== 0 ? 1 / vel.x : Infinity,
      y: vel.y !== 0 ? 1 / vel.y : Infinity,
      z: vel.z !== 0 ? 1 / vel.z : Infinity,
    };
    const axes: Array<'x' | 'y' | 'z'> = ['x', 'y', 'z'];
    let tEntry = -Infinity,
      tExit = Infinity;
    let entryAxis: 'x' | 'y' | 'z' = 'x';
    for (const axis of axes) {
      if (vel[axis] === 0) {
        if (a.min[axis] >= b.max[axis] || a.max[axis] <= b.min[axis])
          return { hit: false, tEntry: 1, normal: { x: 0, y: 0, z: 0 } };
        continue;
      }
      const t1 = (b.min[axis] - a.max[axis]) * invVel[axis];
      const t2 = (b.max[axis] - a.min[axis]) * invVel[axis];
      const tMin = Math.min(t1, t2),
        tMax = Math.max(t1, t2);
      if (tMin > tEntry) {
        tEntry = tMin;
        entryAxis = axis;
      }
      tExit = Math.min(tExit, tMax);
    }
    if (tEntry > tExit || tEntry > 1 || tEntry < 0) return { hit: false, tEntry: 1, normal: { x: 0, y: 0, z: 0 } };
    const normal = { x: 0, y: 0, z: 0 };
    normal[entryAxis] = vel[entryAxis] > 0 ? -1 : 1;
    return { hit: true, tEntry, normal };
  }
}
保守推进法
保守推进（Conservative Advancement）是一种迭代方法：每一步计算两个物体之间的最短距离，然后沿运动方向安全前进这个距离（保证不会错过碰撞）。重复直到距离小于阈值或时间耗尽。


class ConservativeAdvancement {
  private maxIterations = 32;
  private tolerance = 0.001;

  findTimeOfImpact(
    shapeA: ConvexShape,
    posA: Vec3,
    velA: Vec3,
    shapeB: ConvexShape,
    posB: Vec3,
    velB: Vec3,
  ): { hit: boolean; t: number } {
    let t = 0;
    const relVel = { x: velA.x - velB.x, y: velA.y - velB.y, z: velA.z - velB.z };
    const relSpeed = Math.sqrt(relVel.x ** 2 + relVel.y ** 2 + relVel.z ** 2);
    if (relSpeed < this.tolerance) return { hit: false, t: 1 };

    for (let iter = 0; iter < this.maxIterations; iter++) {
      const currentPosA = { x: posA.x + velA.x * t, y: posA.y + velA.y * t, z: posA.z + velA.z * t };
      const currentPosB = { x: posB.x + velB.x * t, y: posB.y + velB.y * t, z: posB.z + velB.z * t };
      const distance = this.computeDistance(shapeA, currentPosA, shapeB, currentPosB);
      if (distance < this.tolerance) return { hit: true, t };
      const advance = distance / relSpeed;
      t += advance;
      if (t > 1) return { hit: false, t: 1 };
    }
    return { hit: false, t: 1 };
  }

  private computeDistance(_a: ConvexShape, _posA: Vec3, _b: ConvexShape, _posB: Vec3): number {
    return 1.0;
  }
}

interface ConvexShape {
  vertices: Vec3[];
}
投机接触法
投机接触（Speculative Contacts）是一种性能友好的近似 CCD 方法。它通过扩展物体的包围体来覆盖其运动轨迹，然后在约束求解阶段限制物体不能穿透障碍。


class SpeculativeContacts {
  private readonly margin: number;

  constructor(margin = 0.01) {
    this.margin = margin;
  }

  expandAABB(aabb: AABB, velocity: Vec3): AABB {
    return {
      min: {
        x: Math.min(aabb.min.x, aabb.min.x + velocity.x) - this.margin,
        y: Math.min(aabb.min.y, aabb.min.y + velocity.y) - this.margin,
        z: Math.min(aabb.min.z, aabb.min.z + velocity.z) - this.margin,
      },
      max: {
        x: Math.max(aabb.max.x, aabb.max.x + velocity.x) + this.margin,
        y: Math.max(aabb.max.y, aabb.max.y + velocity.y) + this.margin,
        z: Math.max(aabb.max.z, aabb.max.z + velocity.z) + this.margin,
      },
    };
  }

  generateSpeculativeConstraint(distance: number, normal: Vec3, relVelNormal: number, dt: number): number {
    const targetVel = distance / dt;
    const deltaVel = targetVel - relVelNormal;
    return Math.max(0, deltaVel);
  }
}
子步进策略
子步进（Sub-stepping）是最简单的 CCD 替代方案：将一个大时间步分成多个小步，每步执行完整的离散碰撞检测。虽然不如真正的 CCD 精确，但实现简单且能显著改善穿透问题。


class SubSteppingCCD {
  private maxSubSteps: number;
  private minStepSize: number;

  constructor(maxSubSteps = 4, minStepSize = 1 / 240) {
    this.maxSubSteps = maxSubSteps;
    this.minStepSize = minStepSize;
  }

  determineSubSteps(velocity: Vec3, objectSize: number, dt: number): number {
    const speed = Math.sqrt(velocity.x ** 2 + velocity.y ** 2 + velocity.z ** 2);
    const moveDistance = speed * dt;
    if (moveDistance < objectSize * 0.5) return 1;
    const needed = Math.ceil(moveDistance / (objectSize * 0.5));
    return Math.min(needed, this.maxSubSteps);
  }

  stepWithSubStepping(
    dt: number,
    bodies: Array<{ velocity: Vec3; size: number }>,
    stepFn: (subDt: number) => void,
  ): void {
    let maxSubSteps = 1;
    for (const body of bodies) {
      const steps = this.determineSubSteps(body.velocity, body.size, dt);
      maxSubSteps = Math.max(maxSubSteps, steps);
    }
    const subDt = dt / maxSubSteps;
    if (subDt < this.minStepSize) {
      maxSubSteps = Math.floor(dt / this.minStepSize);
    }
    for (let i = 0; i < maxSubSteps; i++) {
      stepFn(dt / maxSubSteps);
    }
  }
}
小结
    

  • CCD 解决高速物体在离散检测中的穿透问题
    • 

  • 扫掠体积法精确计算首次碰撞时间，适用于关键碰撞
    • 

  • 保守推进通过迭代安全步进逼近碰撞时间
    • 

  • 投机接触是低成本的近似方案，在约束求解中阻止穿透
    • 

  • 子步进是最简单的替代方案，通过减小时间步来降低穿透风险
