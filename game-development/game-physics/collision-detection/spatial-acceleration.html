空间加速结构

核心概念

空间加速结构用于减少碰撞检测中的物体对比较次数。朴素的碰撞检测需要检查所有 n(n-1)/2 个物体对，而空间加速结构通过将物体按空间位置分组，只检查邻近物体之间的碰撞，将复杂度降至接近 O(n log n) 或 O(n)。

扫描剪枝算法
扫描剪枝（Sweep and Prune, SAP）是一种高效的粗检测算法。它在每个轴上维护物体 AABB 端点的排序列表，利用帧间一致性（Temporal Coherence）快速更新排序。


interface SAP_Entry {
  bodyId: number;
  value: number;
  isMin: boolean;
}

class SweepAndPrune {
  private axes: SAP_Entry[][] = [[], [], []];
  private overlaps = new Set<string>();
  private bodyBounds = new Map<number, { min: number[]; max: number[] }>();

  addBody(id: number, min: number[], max: number[]): void {
    this.bodyBounds.set(id, { min, max });
    for (let axis = 0; axis < 3; axis++) {
      this.axes[axis].push({ bodyId: id, value: min[axis], isMin: true });
      this.axes[axis].push({ bodyId: id, value: max[axis], isMin: false });
    }
    this.sortAllAxes();
  }

  updateBody(id: number, min: number[], max: number[]): void {
    this.bodyBounds.set(id, { min, max });
    for (let axis = 0; axis < 3; axis++) {
      for (const entry of this.axes[axis]) {
        if (entry.bodyId === id) {
          entry.value = entry.isMin ? min[axis] : max[axis];
        }
      }
    }
  }

  findOverlaps(): Array<[number, number]> {
    for (let axis = 0; axis < 3; axis++) {
      this.insertionSort(this.axes[axis]);
    }
    const pairs: Array<[number, number]> = [];
    this.overlaps.clear();
    const activeList = this.axes[0];
    const active = new Set<number>();
    for (const entry of activeList) {
      if (entry.isMin) {
        for (const otherId of active) {
          if (this.overlapsOnAllAxes(entry.bodyId, otherId)) {
            const key = entry.bodyId < otherId ? `${entry.bodyId}:${otherId}` : `${otherId}:${entry.bodyId}`;
            if (!this.overlaps.has(key)) {
              this.overlaps.add(key);
              pairs.push([entry.bodyId, otherId]);
            }
          }
        }
        active.add(entry.bodyId);
      } else {
        active.delete(entry.bodyId);
      }
    }
    return pairs;
  }

  private overlapsOnAllAxes(a: number, b: number): boolean {
    const boundsA = this.bodyBounds.get(a)!;
    const boundsB = this.bodyBounds.get(b)!;
    for (let axis = 0; axis < 3; axis++) {
      if (boundsA.max[axis] < boundsB.min[axis] || boundsB.max[axis] < boundsA.min[axis]) return false;
    }
    return true;
  }

  private insertionSort(entries: SAP_Entry[]): void {
    for (let i = 1; i < entries.length; i++) {
      const key = entries[i];
      let j = i - 1;
      while (j >= 0 && entries[j].value > key.value) {
        entries[j + 1] = entries[j];
        j--;
      }
      entries[j + 1] = key;
    }
  }

  private sortAllAxes(): void {
    for (const axis of this.axes) {
      axis.sort((a, b) => a.value - b.value);
    }
  }
}
SAP 的核心优势在于帧间一致性——相邻帧之间物体位置变化很小，排序列表几乎已有序，插入排序的实际复杂度接近 O(n)。


动态 BVH 树
动态 BVH（Bounding Volume Hierarchy）将物体的包围体组织为二叉树。当物体移动时，如果包围体仍在膨胀（Fat）AABB 内则不需要更新；否则重新插入。


interface AABB {
  min: number[];
  max: number[];
}

interface BVHNode {
  aabb: AABB;
  left: BVHNode | null;
  right: BVHNode | null;
  bodyId: number | null;
  height: number;
  parent: BVHNode | null;
}

class DynamicBVH {
  root: BVHNode | null = null;
  private fatMargin = 0.1;
  private nodeMap = new Map<number, BVHNode>();

  insert(bodyId: number, aabb: AABB): void {
    const fatAABB = this.fatten(aabb);
    const leaf: BVHNode = {
      aabb: fatAABB,
      left: null,
      right: null,
      bodyId,
      height: 0,
      parent: null,
    };
    this.nodeMap.set(bodyId, leaf);
    if (!this.root) {
      this.root = leaf;
      return;
    }
    this.insertLeaf(leaf);
  }

  update(bodyId: number, newAABB: AABB): boolean {
    const node = this.nodeMap.get(bodyId);
    if (!node) return false;
    if (this.contains(node.aabb, newAABB)) return false;
    this.removeLeaf(node);
    node.aabb = this.fatten(newAABB);
    this.insertLeaf(node);
    return true;
  }

  queryPairs(): Array<[number, number]> {
    const pairs: Array<[number, number]> = [];
    if (!this.root) return pairs;
    const leaves: BVHNode[] = [];
    this.collectLeaves(this.root, leaves);
    for (let i = 0; i < leaves.length; i++) {
      for (let j = i + 1; j < leaves.length; j++) {
        if (this.aabbOverlap(leaves[i].aabb, leaves[j].aabb)) {
          pairs.push([leaves[i].bodyId!, leaves[j].bodyId!]);
        }
      }
    }
    return pairs;
  }

  private insertLeaf(leaf: BVHNode): void {
    if (!this.root) {
      this.root = leaf;
      return;
    }
    this.root = this.mergeNodes(this.root, leaf);
  }

  private removeLeaf(_node: BVHNode): void {}
  private mergeNodes(a: BVHNode, b: BVHNode): BVHNode {
    const merged: BVHNode = {
      aabb: this.mergeAABB(a.aabb, b.aabb),
      left: a,
      right: b,
      bodyId: null,
      height: Math.max(a.height, b.height) + 1,
      parent: null,
    };
    a.parent = merged;
    b.parent = merged;
    return merged;
  }

  private fatten(aabb: AABB): AABB {
    const m = this.fatMargin;
    return {
      min: [aabb.min[0] - m, aabb.min[1] - m, aabb.min[2] - m],
      max: [aabb.max[0] + m, aabb.max[1] + m, aabb.max[2] + m],
    };
  }

  private contains(outer: AABB, inner: AABB): boolean {
    for (let i = 0; i < 3; i++) {
      if (inner.min[i] < outer.min[i] || inner.max[i] > outer.max[i]) return false;
    }
    return true;
  }

  private aabbOverlap(a: AABB, b: AABB): boolean {
    for (let i = 0; i < 3; i++) {
      if (a.max[i] < b.min[i] || b.max[i] < a.min[i]) return false;
    }
    return true;
  }

  private mergeAABB(a: AABB, b: AABB): AABB {
    return {
      min: [Math.min(a.min[0], b.min[0]), Math.min(a.min[1], b.min[1]), Math.min(a.min[2], b.min[2])],
      max: [Math.max(a.max[0], b.max[0]), Math.max(a.max[1], b.max[1]), Math.max(a.max[2], b.max[2])],
    };
  }

  private collectLeaves(node: BVHNode, leaves: BVHNode[]): void {
    if (node.bodyId !== null) {
      leaves.push(node);
      return;
    }
    if (node.left) this.collectLeaves(node.left, leaves);
    if (node.right) this.collectLeaves(node.right, leaves);
  }
}
多层网格
当场景中物体大小差异很大时（子弹 vs 建筑），单一网格的格子大小难以适配所有物体。多层网格使用不同粒度的网格层级，大物体放入粗网格，小物体放入细网格。


class MultiLevelGrid {
  private levels: Map<string, Set<number>>[] = [];
  private cellSizes: number[];
  private bodyLevels = new Map<number, number>();

  constructor(cellSizes: number[]) {
    this.cellSizes = cellSizes.sort((a, b) => a - b);
    for (let i = 0; i < cellSizes.length; i++) {
      this.levels.push(new Map());
    }
  }

  insert(id: number, size: number, x: number, y: number): void {
    const level = this.selectLevel(size);
    this.bodyLevels.set(id, level);
    const cellSize = this.cellSizes[level];
    const key = `${Math.floor(x / cellSize)},${Math.floor(y / cellSize)}`;
    if (!this.levels[level].has(key)) {
      this.levels[level].set(key, new Set());
    }
    this.levels[level].get(key)!.add(id);
  }

  private selectLevel(objectSize: number): number {
    for (let i = 0; i < this.cellSizes.length; i++) {
      if (objectSize <= this.cellSizes[i]) return i;
    }
    return this.cellSizes.length - 1;
  }

  queryNearby(x: number, y: number, radius: number): number[] {
    const results: number[] = [];
    for (let level = 0; level < this.levels.length; level++) {
      const cellSize = this.cellSizes[level];
      const minCx = Math.floor((x - radius) / cellSize);
      const maxCx = Math.floor((x + radius) / cellSize);
      const minCy = Math.floor((y - radius) / cellSize);
      const maxCy = Math.floor((y + radius) / cellSize);
      for (let cx = minCx; cx <= maxCx; cx++) {
        for (let cy = minCy; cy <= maxCy; cy++) {
          const cell = this.levels[level].get(`${cx},${cy}`);
          if (cell) results.push(...cell);
        }
      }
    }
    return results;
  }
}
