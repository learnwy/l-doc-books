堆叠稳定性

核心概念

堆叠稳定性是物理引擎面临的最严苛挑战之一。当多个物体层层堆叠时，底部物体必须同时承受上方所有物体的重力，任何微小的求解误差都会被放大，导致堆叠坍塌、物体抖动或漂移。解决堆叠稳定性需要从求解器迭代策略、约束排序、休眠机制等多个维度协同优化。

约束图排序
在堆叠场景中，约束的求解顺序对收敛速度有显著影响。如果先求解顶部约束再求解底部约束，底部约束的修正会被后续迭代推翻，需要更多迭代才能达到稳定。自底向上的排序使力的传播方向与求解顺序一致，大幅减少所需迭代次数。


interface ConstraintNode {
  constraint: Constraint;
  depth: number;
  bodyA: RigidBody;
  bodyB: RigidBody;
}

class ConstraintGraphSorter {
  sort(constraints: Constraint[], gravity: Vector3): Constraint[] {
    const nodes: ConstraintNode[] = constraints.map((c) => ({
      constraint: c,
      depth: 0,
      bodyA: c.bodyA,
      bodyB: c.bodyB,
    }));

    for (const node of nodes) {
      const centerA = node.bodyA.position;
      const centerB = node.bodyB.position;
      const avgHeight = (centerA.dot(gravity.negate()) + centerB.dot(gravity.negate())) / 2;
      node.depth = avgHeight;
    }

    nodes.sort((a, b) => a.depth - b.depth);
    return nodes.map((n) => n.constraint);
  }

  buildDependencyOrder(constraints: Constraint[]): Constraint[][] {
    const bodyToConstraints = new Map<number, Constraint[]>();
    for (const c of constraints) {
      if (!bodyToConstraints.has(c.bodyA.id)) bodyToConstraints.set(c.bodyA.id, []);
      if (!bodyToConstraints.has(c.bodyB.id)) bodyToConstraints.set(c.bodyB.id, []);
      bodyToConstraints.get(c.bodyA.id)!.push(c);
      bodyToConstraints.get(c.bodyB.id)!.push(c);
    }

    const visited = new Set<Constraint>();
    const levels: Constraint[][] = [];
    const staticBodies = new Set<number>();

    for (const c of constraints) {
      if (c.bodyA.isStatic) staticBodies.add(c.bodyA.id);
      if (c.bodyB.isStatic) staticBodies.add(c.bodyB.id);
    }

    let frontier = constraints.filter((c) => staticBodies.has(c.bodyA.id) || staticBodies.has(c.bodyB.id));

    while (frontier.length > 0) {
      const level: Constraint[] = [];
      const nextFrontier: Constraint[] = [];

      for (const c of frontier) {
        if (visited.has(c)) continue;
        visited.add(c);
        level.push(c);

        const otherBody = c.bodyA.isStatic ? c.bodyB : c.bodyA;
        const neighbors = bodyToConstraints.get(otherBody.id) ?? [];
        for (const neighbor of neighbors) {
          if (!visited.has(neighbor)) nextFrontier.push(neighbor);
        }
      }

      if (level.length > 0) levels.push(level);
      frontier = nextFrontier;
    }

    return levels;
  }
}
依赖图排序从静态物体（地面）开始，逐层向上扩展，形成类似 BFS 的层次结构。同一层的约束可以并行求解，不同层按顺序执行，保证力的传播方向正确。


休眠与唤醒机制
休眠机制是维持堆叠稳定性的终极手段。当一组物体的线速度和角速度在连续多帧内低于阈值时，物理引擎将它们标记为休眠状态，跳过所有模拟计算。休眠物体参与碰撞检测但不参与动力学求解。


interface SleepConfig {
  linearThreshold: number;
  angularThreshold: number;
  timeThreshold: number;
}

class SleepManager {
  private config: SleepConfig = {
    linearThreshold: 0.01,
    angularThreshold: 0.01,
    timeThreshold: 0.5,
  };
  private sleepTimers: Map<number, number> = new Map();

  update(bodies: RigidBody[], dt: number): void {
    for (const body of bodies) {
      if (body.isStatic) continue;

      const linearSpeed = body.linearVelocity.length();
      const angularSpeed = body.angularVelocity.length();

      const isBelowThreshold = linearSpeed < this.config.linearThreshold && angularSpeed < this.config.angularThreshold;

      const timer = this.sleepTimers.get(body.id) ?? 0;

      if (isBelowThreshold) {
        const newTimer = timer + dt;
        this.sleepTimers.set(body.id, newTimer);
        if (newTimer >= this.config.timeThreshold) {
          body.isSleeping = true;
          body.linearVelocity = Vector3.zero();
          body.angularVelocity = Vector3.zero();
        }
      } else {
        this.sleepTimers.set(body.id, 0);
        body.isSleeping = false;
      }
    }
  }

  wakeBody(body: RigidBody): void {
    if (!body.isSleeping) return;
    body.isSleeping = false;
    this.sleepTimers.set(body.id, 0);
  }

  wakeIsland(bodies: RigidBody[]): void {
    for (const body of bodies) {
      this.wakeBody(body);
    }
  }

  onCollision(bodyA: RigidBody, bodyB: RigidBody, impulse: number): void {
    const wakeThreshold = 0.1;
    if (impulse > wakeThreshold) {
      if (bodyA.isSleeping) this.wakeBody(bodyA);
      if (bodyB.isSleeping) this.wakeBody(bodyB);
    }
  }
}
休眠的粒度应该是整个岛屿而非单个物体。如果岛屿中任一物体需要唤醒，整个岛屿都必须唤醒，否则堆叠中的中间物体可能保持休眠而导致上方物体悬空。


稳定化技术组合
单一技术难以完全解决堆叠稳定性问题，实际物理引擎通常组合使用多种稳定化技术。


class StabilizedPhysicsWorld {
  private solver: SequentialImpulseSolver;
  private sleepManager: SleepManager;
  private sorter: ConstraintGraphSorter;
  private positionIterations = 3;
  private velocityIterations = 8;

  step(dt: number): void {
    this.integrateForces(dt);
    this.broadPhase();
    this.narrowPhase();

    const constraints = this.collectConstraints();
    const sorted = this.sorter.sort(constraints, new Vector3(0, -9.81, 0));

    this.solver.setConstraints(sorted);
    this.solver.warmStart();

    for (let i = 0; i < this.velocityIterations; i++) {
      this.solver.solveVelocities();
    }

    this.integratePositions(dt);

    for (let i = 0; i < this.positionIterations; i++) {
      this.solver.solvePositions();
    }

    this.sleepManager.update(this.bodies, dt);
    this.solver.cacheImpulses();
  }

  private integrateForces(dt: number): void {
    for (const body of this.bodies) {
      if (body.isStatic || body.isSleeping) continue;
      body.linearVelocity = body.linearVelocity.add(this.gravity.scale(dt));
      body.linearVelocity = body.linearVelocity.scale(1 / (1 + body.linearDamping * dt));
      body.angularVelocity = body.angularVelocity.scale(1 / (1 + body.angularDamping * dt));
    }
  }

  private integratePositions(dt: number): void {
    for (const body of this.bodies) {
      if (body.isStatic || body.isSleeping) continue;
      body.position = body.position.add(body.linearVelocity.scale(dt));
      const angle = body.angularVelocity.length() * dt;
      if (angle > 1e-8) {
        const axis = body.angularVelocity.normalize();
        body.rotation = Quaternion.fromAxisAngle(axis, angle).multiply(body.rotation);
      }
    }
  }
}

小结
    

  • 堆叠稳定性需要从约束排序、迭代策略、休眠机制等多维度协同解决
    • 

  • 自底向上的约束排序使力的传播方向与求解顺序一致，显著加速收敛
    • 

  • 岛屿粒度的休眠机制是消除静止堆叠抖动的终极手段
    • 

  • 暖启动、累积钳位、位置修正、阻尼等技术的组合使用是商业引擎的标准做法
    • 

  • 休眠唤醒必须以岛屿为单位，防止堆叠中间物体意外保持休眠
