约束求解器

核心概念

约束求解器是物理引擎的核心组件，负责同时处理所有碰撞约束和关节约束，使物理系统在满足所有约束条件的同时产生合理的运动。现代游戏物理引擎普遍采用 Sequential Impulse（SI）求解器，它通过迭代的方式逐个处理约束，每次迭代都使系统更接近全局最优解。

Sequential Impulse 算法
Sequential Impulse（SI）是 Erin Catto 在 Box2D 中提出并推广的约束求解方法。它的核心思想是将多约束系统分解为单约束问题序列，每次只处理一个约束，并利用其他约束的最新状态。


interface Constraint {
  bodyA: RigidBody;
  bodyB: RigidBody;
  jacobian: JacobianRow;
  bias: number;
  lowerLimit: number;
  upperLimit: number;
  accumulatedImpulse: number;
}

interface JacobianRow {
  linearA: Vector3;
  angularA: Vector3;
  linearB: Vector3;
  angularB: Vector3;
}

class SequentialImpulseSolver {
  private iterations = 10;
  private constraints: Constraint[] = [];

  preStep(dt: number): void {
    for (const c of this.constraints) {
      this.computeEffectiveMass(c);
      this.computeBias(c, dt);
      this.warmStart(c);
    }
  }

  private computeEffectiveMass(c: Constraint): void {
    const { jacobian: j, bodyA: a, bodyB: b } = c;
    const kLinear = j.linearA.scale(a.inverseMass).dot(j.linearA) + j.linearB.scale(b.inverseMass).dot(j.linearB);
    const kAngular =
      a.inverseInertia.mulVector(j.angularA).dot(j.angularA) + b.inverseInertia.mulVector(j.angularB).dot(j.angularB);
    (c as Constraint & { effectiveMass: number }).effectiveMass = 1 / (kLinear + kAngular);
  }

  private computeBias(c: Constraint, dt: number): void {
    const baumgarte = 0.2;
    const slop = 0.005;
    const error = this.computeConstraintError(c);
    c.bias = (baumgarte / dt) * Math.max(error - slop, 0);
  }

  private warmStart(c: Constraint): void {
    const impulse = c.accumulatedImpulse;
    if (Math.abs(impulse) > 1e-8) {
      this.applyConstraintImpulse(c, impulse * 0.85);
    }
  }

  solve(): void {
    for (let iter = 0; iter < this.iterations; iter++) {
      for (const c of this.constraints) {
        this.solveConstraint(c);
      }
    }
  }

  private solveConstraint(c: Constraint): void {
    const jv = this.computeJV(c);
    const em = (c as Constraint & { effectiveMass: number }).effectiveMass;
    const lambda = em * -(jv + c.bias);

    const oldImpulse = c.accumulatedImpulse;
    c.accumulatedImpulse = Math.max(c.lowerLimit, Math.min(oldImpulse + lambda, c.upperLimit));
    const delta = c.accumulatedImpulse - oldImpulse;

    if (Math.abs(delta) > 1e-8) {
      this.applyConstraintImpulse(c, delta);
    }
  }

  private computeJV(c: Constraint): number {
    const { jacobian: j, bodyA: a, bodyB: b } = c;
    return (
      j.linearA.dot(a.linearVelocity) +
      j.angularA.dot(a.angularVelocity) +
      j.linearB.dot(b.linearVelocity) +
      j.angularB.dot(b.angularVelocity)
    );
  }

  private applyConstraintImpulse(c: Constraint, lambda: number): void {
    const { jacobian: j, bodyA: a, bodyB: b } = c;
    a.linearVelocity = a.linearVelocity.add(j.linearA.scale(lambda * a.inverseMass));
    a.angularVelocity = a.angularVelocity.add(a.inverseInertia.mulVector(j.angularA.scale(lambda)));
    b.linearVelocity = b.linearVelocity.add(j.linearB.scale(lambda * b.inverseMass));
    b.angularVelocity = b.angularVelocity.add(b.inverseInertia.mulVector(j.angularB.scale(lambda)));
  }

  private computeConstraintError(_c: Constraint): number {
    return 0;
  }
}
SI 求解器的收敛速度取决于迭代次数和约束网络的拓扑结构。对于松散耦合的约束（如散落的物体），4-8 次迭代通常足够；对于紧密耦合的堆叠场景，可能需要 20-50 次迭代。


约束的雅可比矩阵
每种类型的约束都可以用雅可比矩阵来统一表达。雅可比矩阵描述了约束违反量对物体速度的敏感度，是求解器将不同约束类型统一处理的关键。


class ContactConstraintBuilder {
  buildNormalConstraint(manifold: ContactManifold, contact: ContactPoint): Constraint {
    const { bodyA, bodyB } = manifold;
    const rA = contact.pointOnA.sub(bodyA.position);
    const rB = contact.pointOnB.sub(bodyB.position);
    const n = contact.normal;

    return {
      bodyA,
      bodyB,
      jacobian: {
        linearA: n.negate(),
        angularA: rA.cross(n).negate(),
        linearB: n,
        angularB: rB.cross(n),
      },
      bias: 0,
      lowerLimit: 0,
      upperLimit: Infinity,
      accumulatedImpulse: 0,
    };
  }

  buildFrictionConstraint(manifold: ContactManifold, contact: ContactPoint, tangent: Vector3): Constraint {
    const { bodyA, bodyB } = manifold;
    const rA = contact.pointOnA.sub(bodyA.position);
    const rB = contact.pointOnB.sub(bodyB.position);

    return {
      bodyA,
      bodyB,
      jacobian: {
        linearA: tangent.negate(),
        angularA: rA.cross(tangent).negate(),
        linearB: tangent,
        angularB: rB.cross(tangent),
      },
      bias: 0,
      lowerLimit: -Infinity,
      upperLimit: Infinity,
      accumulatedImpulse: 0,
    };
  }
}

class DistanceConstraintBuilder {
  build(bodyA: RigidBody, anchorA: Vector3, bodyB: RigidBody, anchorB: Vector3, targetDistance: number): Constraint {
    const worldA = bodyA.position.add(anchorA);
    const worldB = bodyB.position.add(anchorB);
    const diff = worldB.sub(worldA);
    const currentDistance = diff.length();
    const n = diff.scale(1 / currentDistance);
    const rA = worldA.sub(bodyA.position);
    const rB = worldB.sub(bodyB.position);

    return {
      bodyA,
      bodyB,
      jacobian: {
        linearA: n.negate(),
        angularA: rA.cross(n).negate(),
        linearB: n,
        angularB: rB.cross(n),
      },
      bias: 0,
      lowerLimit: -Infinity,
      upperLimit: Infinity,
      accumulatedImpulse: 0,
    };
  }
}
法线约束的下限为 0（只推不拉），摩擦约束的上下限与法线冲量耦合（−μjn≤jt≤μjn-\mu j_n \leq j_t \leq \mu j_n−μjn​≤jt​≤μjn​），距离约束的上下限为无穷大（双向约束）。


约束排序与岛屿
将约束网络分解为互不相关的岛屿（Island）是物理引擎的重要优化。每个岛屿可以独立求解，甚至可以并行处理。当岛屿中所有物体的动能低于阈值时，整个岛屿进入休眠状态，完全跳过求解。


class IslandManager {
  private parent: Map<number, number> = new Map();
  private rank: Map<number, number> = new Map();

  find(id: number): number {
    if (!this.parent.has(id)) {
      this.parent.set(id, id);
      this.rank.set(id, 0);
    }
    let root = id;
    while (this.parent.get(root) !== root) {
      root = this.parent.get(root)!;
    }
    let current = id;
    while (current !== root) {
      const next = this.parent.get(current)!;
      this.parent.set(current, root);
      current = next;
    }
    return root;
  }

  union(a: number, b: number): void {
    const rootA = this.find(a);
    const rootB = this.find(b);
    if (rootA === rootB) return;

    const rankA = this.rank.get(rootA)!;
    const rankB = this.rank.get(rootB)!;
    if (rankA < rankB) {
      this.parent.set(rootA, rootB);
    } else if (rankA > rankB) {
      this.parent.set(rootB, rootA);
    } else {
      this.parent.set(rootB, rootA);
      this.rank.set(rootA, rankA + 1);
    }
  }

  buildIslands(constraints: Constraint[]): Map<number, Constraint[]> {
    for (const c of constraints) {
      this.union(c.bodyA.id, c.bodyB.id);
    }

    const islands = new Map<number, Constraint[]>();
    for (const c of constraints) {
      const island = this.find(c.bodyA.id);
      if (!islands.has(island)) islands.set(island, []);
      islands.get(island)!.push(c);
    }
    return islands;
  }
}

小结
    

  • Sequential Impulse 通过逐约束迭代求解，暖启动显著加速收敛
    • 

  • 雅可比矩阵将不同类型约束统一为速度级线性方程
    • 

  • 累积冲量钳位保证单调收敛和物理正确性
    •

    •
