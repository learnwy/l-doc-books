刚体模拟

核心概念

刚体模拟是物理引擎中最基础也是最重要的功能。刚体（Rigid Body）是形状和体积不发生变化的理想化物体，其运动由质心的平动和绕质心的转动完全描述。物理引擎在每个时间步中计算所有刚体的力、加速度、速度和位置。

刚体状态与属性
刚体的完整状态包括位置、朝向（四元数）、线速度和角速度。此外还有导出属性如变换矩阵、逆惯性张量等。


interface Vec3 {
  x: number;
  y: number;
  z: number;
}
interface Quat {
  x: number;
  y: number;
  z: number;
  w: number;
}
interface Mat3 {
  m: number[];
}

class RigidBody {
  id: number;
  position: Vec3 = { x: 0, y: 0, z: 0 };
  orientation: Quat = { x: 0, y: 0, z: 0, w: 1 };
  linearVelocity: Vec3 = { x: 0, y: 0, z: 0 };
  angularVelocity: Vec3 = { x: 0, y: 0, z: 0 };
  mass: number;
  inverseMass: number;
  localInertia: Mat3;
  inverseInertia: Mat3;
  restitution: number;
  friction: number;
  linearDamping: number;
  angularDamping: number;
  isStatic: boolean;
  isSleeping: boolean;
  sleepTimer: number;
  forceAccum: Vec3 = { x: 0, y: 0, z: 0 };
  torqueAccum: Vec3 = { x: 0, y: 0, z: 0 };
  collisionShape: CollisionShape;

  constructor(id: number, config: RigidBodyConfig) {
    this.id = id;
    this.mass = config.mass;
    this.inverseMass = config.mass > 0 ? 1 / config.mass : 0;
    this.isStatic = config.mass === 0;
    this.restitution = config.restitution ?? 0.3;
    this.friction = config.friction ?? 0.5;
    this.linearDamping = config.linearDamping ?? 0.01;
    this.angularDamping = config.angularDamping ?? 0.05;
    this.localInertia = config.inertia ?? this.computeInertia(config.shape);
    this.inverseInertia = this.invertMatrix(this.localInertia);
    this.isSleeping = false;
    this.sleepTimer = 0;
    this.collisionShape = config.shape;
  }

  applyForce(force: Vec3): void {
    if (this.isStatic) return;
    this.forceAccum.x += force.x;
    this.forceAccum.y += force.y;
    this.forceAccum.z += force.z;
    this.wakeUp();
  }

  applyImpulse(impulse: Vec3): void {
    if (this.isStatic) return;
    this.linearVelocity.x += impulse.x * this.inverseMass;
    this.linearVelocity.y += impulse.y * this.inverseMass;
    this.linearVelocity.z += impulse.z * this.inverseMass;
    this.wakeUp();
  }

  wakeUp(): void {
    this.isSleeping = false;
    this.sleepTimer = 0;
  }

  private computeInertia(_shape: CollisionShape): Mat3 {
    return { m: [1, 0, 0, 0, 1, 0, 0, 0, 1] };
  }
  private invertMatrix(m: Mat3): Mat3 {
    return m;
  }
}

interface RigidBodyConfig {
  mass: number;
  restitution?: number;
  friction?: number;
  linearDamping?: number;
  angularDamping?: number;
  inertia?: Mat3;
  shape: CollisionShape;
}

interface CollisionShape {
  type: string;
}
模拟管线
物理模拟的核心管线在每个固定时间步中依次执行：力的施加、碰撞检测、约束求解和状态积分。这个管线的执行顺序和每步的精确实现直接影响模拟的真实感和稳定性。


class PhysicsWorld {
  private bodies: RigidBody[] = [];
  private gravity: Vec3 = { x: 0, y: -9.81, z: 0 };
  private fixedDt = 1 / 60;

  addBody(body: RigidBody): void {
    this.bodies.push(body);
  }

  step(): void {
    this.applyGravity();
    const pairs = this.broadphase();
    const contacts = this.narrowphase(pairs);
    this.solveConstraints(contacts);
    this.integrate();
    this.updateSleepStates();
  }

  private applyGravity(): void {
    for (const body of this.bodies) {
      if (body.isStatic || body.isSleeping) continue;
      const gravityForce = {
        x: this.gravity.x / body.inverseMass,
        y: this.gravity.y / body.inverseMass,
        z: this.gravity.z / body.inverseMass,
      };
      body.applyForce(gravityForce);
    }
  }

  private integrate(): void {
    for (const body of this.bodies) {
      if (body.isStatic || body.isSleeping) continue;
      const accel = {
        x: body.forceAccum.x * body.inverseMass,
        y: body.forceAccum.y * body.inverseMass,
        z: body.forceAccum.z * body.inverseMass,
      };
      body.linearVelocity.x += accel.x * this.fixedDt;
      body.linearVelocity.y += accel.y * this.fixedDt;
      body.linearVelocity.z += accel.z * this.fixedDt;
      body.linearVelocity.x *= Math.pow(1 - body.linearDamping, this.fixedDt);
      body.linearVelocity.y *= Math.pow(1 - body.linearDamping, this.fixedDt);
      body.linearVelocity.z *= Math.pow(1 - body.linearDamping, this.fixedDt);
      body.position.x += body.linearVelocity.x * this.fixedDt;
      body.position.y += body.linearVelocity.y * this.fixedDt;
      body.position.z += body.linearVelocity.z * this.fixedDt;
      body.forceAccum = { x: 0, y: 0, z: 0 };
      body.torqueAccum = { x: 0, y: 0, z: 0 };
    }
  }

  private updateSleepStates(): void {
    const sleepThreshold = 0.01;
    const sleepTimeRequired = 1.0;
    for (const body of this.bodies) {
      if (body.isStatic) continue;
      const speed = Math.sqrt(body.linearVelocity.x ** 2 + body.linearVelocity.y ** 2 + body.linearVelocity.z ** 2);
      if (speed < sleepThreshold) {
        body.sleepTimer += this.fixedDt;
        if (body.sleepTimer >= sleepTimeRequired) body.isSleeping = true;
      } else {
        body.sleepTimer = 0;
      }
    }
  }

  private broadphase(): Array<[RigidBody, RigidBody]> {
    return [];
  }
  private narrowphase(_pairs: Array<[RigidBody, RigidBody]>): unknown[] {
    return [];
  }
  private solveConstraints(_contacts: unknown[]): void {}
}
休眠与唤醒机制
当刚体的线速度和角速度持续低于阈值一段时间后，物理引擎将其标记为"休眠"（Sleeping）。休眠的物体不参与积分和约束求解，显著降低计算开销。当外力作用或其他物体碰撞时，休眠物体被"唤醒"。


class SleepManager {
  private readonly velocityThreshold = 0.01;
  private readonly angularThreshold = 0.01;
  private readonly timeThreshold = 2.0;

  updateSleep(body: RigidBody, dt: number): void {
    if (body.isStatic) return;
    const linearSpeed = Math.sqrt(body.linearVelocity.x ** 2 + body.linearVelocity.y ** 2 + body.linearVelocity.z ** 2);
    const angularSpeed = Math.sqrt(
      body.angularVelocity.x ** 2 + body.angularVelocity.y ** 2 + body.angularVelocity.z ** 2,
    );
    if (linearSpeed < this.velocityThreshold && angularSpeed < this.angularThreshold) {
      body.sleepTimer += dt;
      if (body.sleepTimer >= this.timeThreshold) {
        body.isSleeping = true;
        body.linearVelocity = { x: 0, y: 0, z: 0 };
        body.angularVelocity = { x: 0, y: 0, z: 0 };
      }
    } else {
      body.sleepTimer = 0;
      body.isSleeping = false;
    }
  }

  wakeIsland(bodies: RigidBody[]): void {
    for (const body of bodies) {
      body.wakeUp();
    }
  }
}
大规模场景中可能有数千个休眠物体（堆叠的箱子、散落的碎片），休眠机制可以将物理计算开销降低一到两个数量级。


小结
    

  • 刚体状态由位置、朝向、线速度和角速度完整描述
    • 

  • 物理模拟管线依次执行力施加、碰撞检测、约束求解和积分
    • 

  • 休眠机制通过跳过静止物体的计算显著降低开销
    • 

  • 惯性张量决定物体的旋转响应，不同形状有不同的惯性分布
    • 

  • 正确的管线执行顺序是模拟稳定性的基础
