牛顿力学基础

核心概念

牛顿力学是游戏物理引擎的数学基础。三大运动定律描述了力、质量和加速度之间的关系，为刚体模拟提供了理论框架。游戏物理引擎通过数值积分将这些连续方程离散化，在每个时间步中更新物体状态。

力的累加与合力
物体在一个时间步内可能受到多个力的作用——重力、弹力、摩擦力、推力等。引擎在每帧收集所有作用力并累加为合力，再根据牛顿第二定律计算加速度。


interface Vec3 {
  x: number;
  y: number;
  z: number;
}

const Vec3Math = {
  add(a: Vec3, b: Vec3): Vec3 {
    return { x: a.x + b.x, y: a.y + b.y, z: a.z + b.z };
  },
  scale(v: Vec3, s: number): Vec3 {
    return { x: v.x * s, y: v.y * s, z: v.z * s };
  },
  length(v: Vec3): number {
    return Math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z);
  },
  zero(): Vec3 {
    return { x: 0, y: 0, z: 0 };
  },
  cross(a: Vec3, b: Vec3): Vec3 {
    return { x: a.y * b.z - a.z * b.y, y: a.z * b.x - a.x * b.z, z: a.x * b.y - a.y * b.x };
  },
};

interface ForceEntry {
  force: Vec3;
  point?: Vec3;
  type: string;
}

class ForceAccumulator {
  private forces: ForceEntry[] = [];
  private torques: Vec3[] = [];

  addForce(force: Vec3, type = 'generic'): void {
    this.forces.push({ force, type });
  }

  addForceAtPoint(force: Vec3, point: Vec3, centerOfMass: Vec3, type = 'generic'): void {
    this.forces.push({ force, point, type });
    const r = Vec3Math.add(point, Vec3Math.scale(centerOfMass, -1));
    this.torques.push(Vec3Math.cross(r, force));
  }

  addTorque(torque: Vec3): void {
    this.torques.push(torque);
  }

  getNetForce(): Vec3 {
    return this.forces.reduce((sum, entry) => Vec3Math.add(sum, entry.force), Vec3Math.zero());
  }

  getNetTorque(): Vec3 {
    return this.torques.reduce((sum, t) => Vec3Math.add(sum, t), Vec3Math.zero());
  }

  clear(): void {
    this.forces = [];
    this.torques = [];
  }
}
在施力点不在质心时，力还会产生力矩（Torque），导致物体旋转。力矩 τ = r × F，其中 r 是施力点到质心的向量。


数值积分方法
物理引擎的核心任务是求解运动方程的数值解。不同的积分方法在精度、稳定性和计算开销之间有不同的权衡。


interface PhysicsState {
  position: Vec3;
  velocity: Vec3;
  acceleration: Vec3;
}

class Integrators {
  static euler(state: PhysicsState, dt: number): PhysicsState {
    return {
      position: Vec3Math.add(state.position, Vec3Math.scale(state.velocity, dt)),
      velocity: Vec3Math.add(state.velocity, Vec3Math.scale(state.acceleration, dt)),
      acceleration: state.acceleration,
    };
  }

  static semiImplicitEuler(state: PhysicsState, dt: number): PhysicsState {
    const newVelocity = Vec3Math.add(state.velocity, Vec3Math.scale(state.acceleration, dt));
    const newPosition = Vec3Math.add(state.position, Vec3Math.scale(newVelocity, dt));
    return { position: newPosition, velocity: newVelocity, acceleration: state.acceleration };
  }

  static verlet(position: Vec3, previousPosition: Vec3, acceleration: Vec3, dt: number): Vec3 {
    return Vec3Math.add(
      Vec3Math.add(Vec3Math.scale(position, 2), Vec3Math.scale(previousPosition, -1)),
      Vec3Math.scale(acceleration, dt * dt),
    );
  }

  static rk4(state: PhysicsState, dt: number, accelFn: (pos: Vec3, vel: Vec3) => Vec3): PhysicsState {
    const evaluate = (s: PhysicsState, dt2: number, d: { dx: Vec3; dv: Vec3 }): { dx: Vec3; dv: Vec3 } => {
      const pos = Vec3Math.add(s.position, Vec3Math.scale(d.dx, dt2));
      const vel = Vec3Math.add(s.velocity, Vec3Math.scale(d.dv, dt2));
      return { dx: vel, dv: accelFn(pos, vel) };
    };
    const zero = { dx: Vec3Math.zero(), dv: Vec3Math.zero() };
    const k1 = evaluate(state, 0, zero);
    const k2 = evaluate(state, dt * 0.5, k1);
    const k3 = evaluate(state, dt * 0.5, k2);
    const k4 = evaluate(state, dt, k3);
    const dxdt = Vec3Math.scale(
      Vec3Math.add(Vec3Math.add(k1.dx, Vec3Math.scale(k2.dx, 2)), Vec3Math.add(Vec3Math.scale(k3.dx, 2), k4.dx)),
      1 / 6,
    );
    const dvdt = Vec3Math.scale(
      Vec3Math.add(Vec3Math.add(k1.dv, Vec3Math.scale(k2.dv, 2)), Vec3Math.add(Vec3Math.scale(k3.dv, 2), k4.dv)),
      1 / 6,
    );
    return {
      position: Vec3Math.add(state.position, Vec3Math.scale(dxdt, dt)),
      velocity: Vec3Math.add(state.velocity, Vec3Math.scale(dvdt, dt)),
      acceleration: accelFn(state.position, state.velocity),
    };
  }
}

阻尼与能量守恒
真实物理世界中，能量因摩擦和空气阻力而耗散。游戏物理引擎通过线性阻尼和角阻尼来模拟这种能量衰减，防止物体无限运动。


interface RigidBodyConfig {
  mass: number;
  linearDamping: number;
  angularDamping: number;
  restitution: number;
  friction: number;
}

class SimpleRigidBody {
  position: Vec3;
  velocity: Vec3;
  angularVelocity: Vec3;
  inverseMass: number;
  linearDamping: number;
  angularDamping: number;
  private forceAccum = new ForceAccumulator();

  constructor(config: RigidBodyConfig) {
    this.position = Vec3Math.zero();
    this.velocity = Vec3Math.zero();
    this.angularVelocity = Vec3Math.zero();
    this.inverseMass = config.mass > 0 ? 1 / config.mass : 0;
    this.linearDamping = config.linearDamping;
    this.angularDamping = config.angularDamping;
  }

  applyForce(force: Vec3): void {
    this.forceAccum.addForce(force);
  }

  integrate(dt: number): void {
    if (this.inverseMass === 0) return;

    const acceleration = Vec3Math.scale(this.forceAccum.getNetForce(), this.inverseMass);
    this.velocity = Vec3Math.add(this.velocity, Vec3Math.scale(acceleration, dt));
    this.velocity = Vec3Math.scale(this.velocity, Math.pow(this.linearDamping, dt));
    this.position = Vec3Math.add(this.position, Vec3Math.scale(this.velocity, dt));

    this.angularVelocity = Vec3Math.scale(this.angularVelocity, Math.pow(this.angularDamping, dt));

    this.forceAccum.clear();
  }

  getKineticEnergy(): number {
    const mass = this.inverseMass > 0 ? 1 / this.inverseMass : 0;
    const speed = Vec3Math.length(this.velocity);
    return 0.5 * mass * speed * speed;
  }

  isSleeping(): boolean {
    const threshold = 0.01;
    return Vec3Math.length(this.velocity) < threshold && Vec3Math.length(this.angularVelocity) < threshold;
  }
}
小结
    

  • 牛顿三定律是物理引擎的数学基础，F=ma 将力转化为运动
    • 

  • 力累加器在每帧收集所有作用力，施力点偏离质心产生力矩
    • 

  • 半隐式欧拉是游戏物理的最佳积分方法，兼顾稳定性和性能
    • 

  • 阻尼模拟能量耗散，休眠检测避免对静止物体的无意义计算
    • 

  • 选择积分方法需权衡精度、稳定性和计算成本
