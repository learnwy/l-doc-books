物理引擎集成

核心概念

将物理引擎集成到游戏引擎中涉及数据同步、时间步管理、碰撞回调和调试可视化等多个方面。物理引擎通常作为独立子系统运行，游戏引擎需要在每帧将物理状态同步到渲染状态。

物理代理组件
物理代理（Physics Proxy）是游戏对象和物理刚体之间的桥梁。它管理两者的生命周期同步和数据映射。


interface Vec3 {
  x: number;
  y: number;
  z: number;
}
interface Quat {
  x: number;
  y: number;
  z: number;
  w: number;
}

interface PhysicsBodyHandle {
  id: number;
}
interface PhysicsWorldRef {
  createBody(config: PhysicsBodyConfig): PhysicsBodyHandle;
  removeBody(handle: PhysicsBodyHandle): void;
  getPosition(handle: PhysicsBodyHandle): Vec3;
  getRotation(handle: PhysicsBodyHandle): Quat;
  getLinearVelocity(handle: PhysicsBodyHandle): Vec3;
  setPosition(handle: PhysicsBodyHandle, pos: Vec3): void;
  applyForce(handle: PhysicsBodyHandle, force: Vec3): void;
  applyImpulse(handle: PhysicsBodyHandle, impulse: Vec3): void;
}

interface PhysicsBodyConfig {
  mass: number;
  shape: { type: string; params: Record<string, number> };
  position: Vec3;
  rotation: Quat;
  material: { restitution: number; friction: number };
}

class PhysicsProxy {
  private handle: PhysicsBodyHandle;
  private world: PhysicsWorldRef;
  private entityId: number;
  private interpolatedPosition: Vec3 = { x: 0, y: 0, z: 0 };
  private previousPosition: Vec3 = { x: 0, y: 0, z: 0 };
  private interpolatedRotation: Quat = { x: 0, y: 0, z: 0, w: 1 };
  private previousRotation: Quat = { x: 0, y: 0, z: 0, w: 1 };
  private kinematic = false;

  constructor(entityId: number, world: PhysicsWorldRef, config: PhysicsBodyConfig) {
    this.entityId = entityId;
    this.world = world;
    this.handle = world.createBody(config);
    this.previousPosition = { ...config.position };
    this.interpolatedPosition = { ...config.position };
  }

  prePhysicsSync(gamePosition: Vec3, gameRotation: Quat): void {
    if (this.kinematic) {
      this.world.setPosition(this.handle, gamePosition);
    }
  }

  postPhysicsSync(): void {
    this.previousPosition = { ...this.interpolatedPosition };
    this.previousRotation = { ...this.interpolatedRotation };
    this.interpolatedPosition = this.world.getPosition(this.handle);
    this.interpolatedRotation = this.world.getRotation(this.handle);
  }

  getInterpolatedPosition(alpha: number): Vec3 {
    return {
      x: this.previousPosition.x + (this.interpolatedPosition.x - this.previousPosition.x) * alpha,
      y: this.previousPosition.y + (this.interpolatedPosition.y - this.previousPosition.y) * alpha,
      z: this.previousPosition.z + (this.interpolatedPosition.z - this.previousPosition.z) * alpha,
    };
  }

  applyForce(force: Vec3): void {
    this.world.applyForce(this.handle, force);
  }
  applyImpulse(impulse: Vec3): void {
    this.world.applyImpulse(this.handle, impulse);
  }
  getVelocity(): Vec3 {
    return this.world.getLinearVelocity(this.handle);
  }
  getEntityId(): number {
    return this.entityId;
  }

  destroy(): void {
    this.world.removeBody(this.handle);
  }
}
碰撞事件系统
碰撞事件通知游戏逻辑哪些物体发生了接触、持续接触或分离。事件系统需要追踪碰撞对的状态变化，区分进入、持续和退出三种事件。


type CollisionEventType = 'enter' | 'stay' | 'exit';

interface CollisionEvent {
  type: CollisionEventType;
  entityA: number;
  entityB: number;
  contactPoints: Vec3[];
  normal: Vec3;
  impulse: number;
}

class CollisionEventManager {
  private currentPairs = new Set<string>();
  private previousPairs = new Set<string>();
  private eventQueue: CollisionEvent[] = [];

  private pairKey(a: number, b: number): string {
    return a < b ? `${a}:${b}` : `${b}:${a}`;
  }

  beginFrame(): void {
    this.previousPairs = new Set(this.currentPairs);
    this.currentPairs.clear();
    this.eventQueue = [];
  }

  reportContact(entityA: number, entityB: number, points: Vec3[], normal: Vec3, impulse: number): void {
    const key = this.pairKey(entityA, entityB);
    this.currentPairs.add(key);

    const base = { entityA, entityB, contactPoints: points, normal, impulse };
    if (!this.previousPairs.has(key)) {
      this.eventQueue.push({ ...base, type: 'enter' });
    } else {
      this.eventQueue.push({ ...base, type: 'stay' });
    }
  }

  endFrame(): void {
    for (const key of this.previousPairs) {
      if (!this.currentPairs.has(key)) {
        const [a, b] = key.split(':').map(Number);
        this.eventQueue.push({
          type: 'exit',
          entityA: a,
          entityB: b,
          contactPoints: [],
          normal: { x: 0, y: 1, z: 0 },
          impulse: 0,
        });
      }
    }
  }

  getEvents(): ReadonlyArray<CollisionEvent> {
    return this.eventQueue;
  }

  getEventsForEntity(entityId: number): CollisionEvent[] {
    return this.eventQueue.filter((e) => e.entityA === entityId || e.entityB === entityId);
  }
}
射线查询与形状查询
物理引擎提供射线投射（Raycast）和形状扫描（Shape Cast）等查询功能，供游戏逻辑用于视线检测、武器命中判定和地面检测等。


interface RaycastHit {
  entityId: number;
  point: Vec3;
  normal: Vec3;
  distance: number;
  fraction: number;
}

interface PhysicsQuery {
  raycast(origin: Vec3, direction: Vec3, maxDistance: number, layerMask?: number): RaycastHit | null;
  raycastAll(origin: Vec3, direction: Vec3, maxDistance: number, layerMask?: number): RaycastHit[];
  overlapSphere(center: Vec3, radius: number, layerMask?: number): number[];
  overlapBox(center: Vec3, halfExtents: Vec3, layerMask?: number): number[];
}

class PhysicsQuerySystem implements PhysicsQuery {
  private bodies: Array<{ entityId: number; position: Vec3; radius: number; layer: number }> = [];

  raycast(origin: Vec3, direction: Vec3, maxDistance: number, layerMask = 0xffffffff): RaycastHit | null {
    let closest: RaycastHit | null = null;
    for (const body of this.bodies) {
      if ((body.layer & layerMask) === 0) continue;
      const hit = this.raySphereTest(origin, direction, body.position, body.radius);
      if (hit && hit.distance <= maxDistance) {
        if (!closest || hit.distance < closest.distance) {
          closest = { ...hit, entityId: body.entityId };
        }
      }
    }
    return closest;
  }

  raycastAll(origin: Vec3, direction: Vec3, maxDistance: number, layerMask = 0xffffffff): RaycastHit[] {
    const hits: RaycastHit[] = [];
    for (const body of this.bodies) {
      if ((body.layer & layerMask) === 0) continue;
      const hit = this.raySphereTest(origin, direction, body.position, body.radius);
      if (hit && hit.distance <= maxDistance) {
        hits.push({ ...hit, entityId: body.entityId });
      }
    }
    return hits.sort((a, b) => a.distance - b.distance);
  }

  overlapSphere(center: Vec3, radius: number, layerMask = 0xffffffff): number[] {
    return this.bodies
      .filter((b) => {
        if ((b.layer & layerMask) === 0) return false;
        const dx = b.position.x - center.x;
        const dy = b.position.y - center.y;
        const dz = b.position.z - center.z;
        return Math.sqrt(dx * dx + dy * dy + dz * dz) <= radius + b.radius;
      })
      .map((b) => b.entityId);
  }

  overlapBox(_center: Vec3, _halfExtents: Vec3, _layerMask = 0xffffffff): number[] {
    return [];
  }

  private raySphereTest(
    _origin: Vec3,
    _dir: Vec3,
    _center: Vec3,
    _radius: number,
  ): Omit<RaycastHit, 'entityId'> | null {
    return null;
  }
}
