粒子系统

核心概念

粒子系统是游戏中模拟大量微小物体集体行为的核心技术，用于实现火焰、烟雾、爆炸、雨雪、魔法特效等视觉效果。每个粒子拥有位置、速度、生命周期等属性，通过简单的物理规则驱动运动，数千个粒子的叠加效果创造出复杂而生动的视觉表现。

粒子生命周期管理
粒子系统的性能关键在于高效的内存管理。使用对象池和紧凑数组（Structure of Arrays）布局可以最大化缓存利用率，支撑数万粒子的实时模拟。


interface ParticleData {
  positions: Float32Array;
  velocities: Float32Array;
  colors: Float32Array;
  sizes: Float32Array;
  lifetimes: Float32Array;
  ages: Float32Array;
  alive: Uint8Array;
}

class ParticlePool {
  private data: ParticleData;
  private maxParticles: number;
  private aliveCount = 0;

  constructor(maxParticles: number) {
    this.maxParticles = maxParticles;
    this.data = {
      positions: new Float32Array(maxParticles * 3),
      velocities: new Float32Array(maxParticles * 3),
      colors: new Float32Array(maxParticles * 4),
      sizes: new Float32Array(maxParticles),
      lifetimes: new Float32Array(maxParticles),
      ages: new Float32Array(maxParticles),
      alive: new Uint8Array(maxParticles),
    };
  }

  emit(count: number, emitter: ParticleEmitter): void {
    for (let i = 0; i < count; i++) {
      const idx = this.findDeadSlot();
      if (idx === -1) return;

      emitter.initialize(this.data, idx);
      this.data.alive[idx] = 1;
      this.aliveCount++;
    }
  }

  update(dt: number): void {
    for (let i = 0; i < this.maxParticles; i++) {
      if (!this.data.alive[i]) continue;

      this.data.ages[i] += dt;
      if (this.data.ages[i] >= this.data.lifetimes[i]) {
        this.data.alive[i] = 0;
        this.aliveCount--;
        continue;
      }

      const idx3 = i * 3;
      this.data.positions[idx3] += this.data.velocities[idx3] * dt;
      this.data.positions[idx3 + 1] += this.data.velocities[idx3 + 1] * dt;
      this.data.positions[idx3 + 2] += this.data.velocities[idx3 + 2] * dt;
    }
  }

  private findDeadSlot(): number {
    for (let i = 0; i < this.maxParticles; i++) {
      if (!this.data.alive[i]) return i;
    }
    return -1;
  }

  compact(): void {
    let writeIdx = 0;
    for (let readIdx = 0; readIdx < this.maxParticles; readIdx++) {
      if (this.data.alive[readIdx]) {
        if (writeIdx !== readIdx) this.swapParticles(writeIdx, readIdx);
        writeIdx++;
      }
    }
  }

  private swapParticles(a: number, b: number): void {
    const swap3 = (arr: Float32Array, i: number, j: number) => {
      for (let k = 0; k < 3; k++) {
        const temp = arr[i * 3 + k];
        arr[i * 3 + k] = arr[j * 3 + k];
        arr[j * 3 + k] = temp;
      }
    };
    swap3(this.data.positions, a, b);
    swap3(this.data.velocities, a, b);

    [this.data.sizes[a], this.data.sizes[b]] = [this.data.sizes[b], this.data.sizes[a]];
    [this.data.ages[a], this.data.ages[b]] = [this.data.ages[b], this.data.ages[a]];
    [this.data.alive[a], this.data.alive[b]] = [this.data.alive[b], this.data.alive[a]];
  }
}
SoA（Structure of Arrays）布局比 AoS（Array of Structures）布局更适合粒子系统，因为更新循环通常只访问少数几个属性，SoA 布局确保这些属性在内存中连续，提高缓存命中率。


发射器与形状
发射器定义了粒子的初始属性分布，不同形状的发射器创造不同的视觉效果。


interface EmitterShape {
  samplePosition(): Vector3;
  sampleDirection(): Vector3;
}

class ConeEmitter implements EmitterShape {
  private angle: number;
  private radius: number;
  private height: number;

  constructor(angle: number, radius: number, height: number) {
    this.angle = angle;
    this.radius = radius;
    this.height = height;
  }

  samplePosition(): Vector3 {
    const r = Math.sqrt(Math.random()) * this.radius;
    const theta = Math.random() * Math.PI * 2;
    return new Vector3(r * Math.cos(theta), 0, r * Math.sin(theta));
  }

  sampleDirection(): Vector3 {
    const phi = Math.random() * this.angle;
    const theta = Math.random() * Math.PI * 2;
    return new Vector3(Math.sin(phi) * Math.cos(theta), Math.cos(phi), Math.sin(phi) * Math.sin(theta)).normalize();
  }
}

class ParticleEmitter {
  shape: EmitterShape;
  speedRange: [number, number] = [1, 5];
  lifetimeRange: [number, number] = [1, 3];
  sizeRange: [number, number] = [0.1, 0.5];
  startColor: [number, number, number, number] = [1, 1, 0, 1];
  endColor: [number, number, number, number] = [1, 0, 0, 0];

  constructor(shape: EmitterShape) {
    this.shape = shape;
  }

  initialize(data: ParticleData, idx: number): void {
    const pos = this.shape.samplePosition();
    const dir = this.shape.sampleDirection();
    const speed = this.lerp(this.speedRange[0], this.speedRange[1], Math.random());

    data.positions[idx * 3] = pos.x;
    data.positions[idx * 3 + 1] = pos.y;
    data.positions[idx * 3 + 2] = pos.z;
    data.velocities[idx * 3] = dir.x * speed;
    data.velocities[idx * 3 + 1] = dir.y * speed;
    data.velocities[idx * 3 + 2] = dir.z * speed;
    data.lifetimes[idx] = this.lerp(this.lifetimeRange[0], this.lifetimeRange[1], Math.random());
    data.ages[idx] = 0;
    data.sizes[idx] = this.lerp(this.sizeRange[0], this.sizeRange[1], Math.random());
  }

  private lerp(a: number, b: number, t: number): number {
    return a + (b - a) * t;
  }
}
力场与交互
粒子除了受初始速度驱动外，还可以受各种力场影响，创造更丰富的运动行为。


interface ForceField {
  apply(position: Vector3, velocity: Vector3, mass: number): Vector3;
}

class GravityField implements ForceField {
  private gravity: Vector3;
  constructor(gravity: Vector3) {
    this.gravity = gravity;
  }
  apply(_pos: Vector3, _vel: Vector3, mass: number): Vector3 {
    return this.gravity.scale(mass);
  }
}

class VortexField implements ForceField {
  private center: Vector3;
  private axis: Vector3;
  private strength: number;

  constructor(center: Vector3, axis: Vector3, strength: number) {
    this.center = center;
    this.axis = axis.normalize();
    this.strength = strength;
  }

  apply(position: Vector3, _vel: Vector3, _mass: number): Vector3 {
    const toCenter = this.center.sub(position);
    const projected = this.axis.scale(toCenter.dot(this.axis));
    const radial = toCenter.sub(projected);
    const tangent = this.axis.cross(radial).normalize();
    return tangent.scale(this.strength);
  }
}

class TurbulenceField implements ForceField {
  private strength: number;
  private frequency: number;
  private time = 0;

  constructor(strength: number, frequency: number) {
    this.strength = strength;
    this.frequency = frequency;
  }

  apply(position: Vector3, _vel: Vector3, _mass: number): Vector3 {
    const nx = Math.sin(position.x * this.frequency + this.time) * this.strength;
    const ny = Math.sin(position.y * this.frequency + this.time * 1.3) * this.strength;
    const nz = Math.sin(position.z * this.frequency + this.time * 0.7) * this.strength;
    return new Vector3(nx, ny, nz);
  }

  tick(dt: number): void {
    this.time += dt;
  }
}

GPU 粒子系统
当粒子数量达到数十万级别时，CPU 粒子系统的性能成为瓶颈。GPU 粒子系统将粒子的模拟和渲染全部放在 GPU 上执行，利用 Compute Shader 的并行性处理海量粒子。


class GPUParticleSystem {
  private particleBuffer: GPUBuffer;
  private emitBuffer: GPUBuffer;
  private deadListBuffer: GPUBuffer;
  private aliveListBuffer: GPUBuffer;
  private simulatePipeline: GPUComputePipeline;
  private emitPipeline: GPUComputePipeline;

  constructor(device: GPUDevice, maxParticles: number) {
    const particleStride = 48;
    this.particleBuffer = device.createBuffer({
      size: maxParticles * particleStride,
      usage: GPUBufferUsage.STORAGE | GPUBufferUsage.VERTEX,
    });
    this.deadListBuffer = device.createBuffer({
      size: maxParticles * 4 + 4,
      usage: GPUBufferUsage.STORAGE,
    });
    this.aliveListBuffer = device.createBuffer({
      size: maxParticles * 4 + 4,
      usage: GPUBufferUsage.STORAGE,
    });
  }

  emit(encoder: GPUCommandEncoder, count: number): void {
    const pass = encoder.beginComputePass();
    pass.setPipeline(this.emitPipeline);
    pass.setBindGroup(0, this.emitBindGroup);
    pass.dispatchWorkgroups(Math.ceil(count / 64));
    pass.end();
  }

  simulate(encoder: GPUCommandEncoder, dt: number): void {
    const pass = encoder.beginComputePass();
    pass.setPipeline(this.simulatePipeline);
    pass.setBindGroup(0, this.simulateBindGroup);
    pass.dispatchWorkgroups(Math.ceil(this.aliveCount / 64));
    pass.end();
  }
}
GPU 粒子系统的核心挑战是 Dead List 管理：当粒子死亡时将索引追加到 Dead List，发射新粒子时从 Dead List 弹出索引。这需要使用原子操作保证并发安全。


小结
    

  • 粒子系统通过大量简单个体的集体行为模拟复杂的视觉效果
    • 

  • SoA 内存布局和对象池是保证性能的关键数据结构设计
    • 

  • 发射器形状决定粒子初始分布，力场驱动运动行为的多样性
    • 

  • GPU 粒子系统利用 Compute Shader 支持数十万级粒子的实时模拟
    • 

  • 生命周期管理、碰撞检测、排序是粒子系统的三大性能瓶颈
