布料模拟

核心概念

布料模拟是游戏中实现旗帜飘扬、披风摆动、衣物褶皱等效果的核心技术。现代布料模拟基于质点-弹簧模型（Mass-Spring Model），将布料离散化为网格状的质点阵列，质点之间通过弹簧约束相连，每一步模拟都在重力、风力等外力与弹簧恢复力的共同作用下更新质点位置。

质点-弹簧模型
布料网格由 M×NM \times NM×N 个质点组成，每个质点与其邻居通过三种类型的弹簧相连：结构弹簧防止拉伸，剪切弹簧防止对角变形，弯曲弹簧防止过度折叠。


interface ClothParticle {
  position: Vector3;
  previousPosition: Vector3;
  velocity: Vector3;
  acceleration: Vector3;
  mass: number;
  inverseMass: number;
  pinned: boolean;
}

interface SpringConstraint {
  particleA: number;
  particleB: number;
  restLength: number;
  stiffness: number;
  type: 'structural' | 'shear' | 'bending';
}

class ClothMesh {
  particles: ClothParticle[] = [];
  springs: SpringConstraint[] = [];
  private width: number;
  private height: number;

  constructor(width: number, height: number, spacing: number) {
    this.width = width;
    this.height = height;

    for (let y = 0; y < height; y++) {
      for (let x = 0; x < width; x++) {
        this.particles.push({
          position: new Vector3(x * spacing, 0, y * spacing),
          previousPosition: new Vector3(x * spacing, 0, y * spacing),
          velocity: Vector3.zero(),
          acceleration: Vector3.zero(),
          mass: 1.0,
          inverseMass: 1.0,
          pinned: false,
        });
      }
    }

    this.createStructuralSprings(spacing);
    this.createShearSprings(spacing);
    this.createBendingSprings(spacing);
  }

  private idx(x: number, y: number): number {
    return y * this.width + x;
  }

  private createStructuralSprings(spacing: number): void {
    for (let y = 0; y < this.height; y++) {
      for (let x = 0; x < this.width; x++) {
        if (x < this.width - 1) {
          this.springs.push({
            particleA: this.idx(x, y),
            particleB: this.idx(x + 1, y),
            restLength: spacing,
            stiffness: 500,
            type: 'structural',
          });
        }
        if (y < this.height - 1) {
          this.springs.push({
            particleA: this.idx(x, y),
            particleB: this.idx(x, y + 1),
            restLength: spacing,
            stiffness: 500,
            type: 'structural',
          });
        }
      }
    }
  }

  private createShearSprings(spacing: number): void {
    const diagLength = spacing * Math.SQRT2;
    for (let y = 0; y < this.height - 1; y++) {
      for (let x = 0; x < this.width - 1; x++) {
        this.springs.push({
          particleA: this.idx(x, y),
          particleB: this.idx(x + 1, y + 1),
          restLength: diagLength,
          stiffness: 300,
          type: 'shear',
        });
        this.springs.push({
          particleA: this.idx(x + 1, y),
          particleB: this.idx(x, y + 1),
          restLength: diagLength,
          stiffness: 300,
          type: 'shear',
        });
      }
    }
  }

  private createBendingSprings(spacing: number): void {
    for (let y = 0; y < this.height; y++) {
      for (let x = 0; x < this.width - 2; x++) {
        this.springs.push({
          particleA: this.idx(x, y),
          particleB: this.idx(x + 2, y),
          restLength: spacing * 2,
          stiffness: 100,
          type: 'bending',
        });
      }
    }
    for (let y = 0; y < this.height - 2; y++) {
      for (let x = 0; x < this.width; x++) {
        this.springs.push({
          particleA: this.idx(x, y),
          particleB: this.idx(x, y + 2),
          restLength: spacing * 2,
          stiffness: 100,
          type: 'bending',
        });
      }
    }
  }
}
弹簧刚度的选择直接影响布料的视觉质感：高刚度模拟牛仔布等硬质面料，低刚度模拟丝绸等柔软面料。


Verlet 积分与约束投影
Verlet 积分是布料模拟中最常用的时间积分方法，它不显式存储速度，而是通过当前位置与上一帧位置的差值隐含速度信息。这种方法天然具有数值阻尼，有助于布料稳定。


class ClothSimulator {
  private cloth: ClothMesh;
  private gravity = new Vector3(0, -9.81, 0);
  private damping = 0.99;
  private constraintIterations = 15;

  constructor(cloth: ClothMesh) {
    this.cloth = cloth;
  }

  step(dt: number): void {
    this.applyExternalForces(dt);
    this.verletIntegrate(dt);
    this.solveConstraints();
  }

  private applyExternalForces(dt: number): void {
    for (const p of this.cloth.particles) {
      if (p.pinned) continue;
      p.acceleration = this.gravity;
    }
  }

  private verletIntegrate(dt: number): void {
    for (const p of this.cloth.particles) {
      if (p.pinned) continue;

      const temp = p.position.clone();
      const displacement = p.position.sub(p.previousPosition).scale(this.damping);
      p.position = p.position.add(displacement).add(p.acceleration.scale(dt * dt));
      p.previousPosition = temp;
    }
  }

  private solveConstraints(): void {
    for (let iter = 0; iter < this.constraintIterations; iter++) {
      for (const spring of this.cloth.springs) {
        this.solveDistanceConstraint(spring);
      }
    }
  }

  private solveDistanceConstraint(spring: SpringConstraint): void {
    const pA = this.cloth.particles[spring.particleA];
    const pB = this.cloth.particles[spring.particleB];

    const diff = pB.position.sub(pA.position);
    const currentLength = diff.length();
    if (currentLength < 1e-7) return;

    const error = currentLength - spring.restLength;
    const correction = diff.scale(error / currentLength);

    const totalInvMass = pA.inverseMass + pB.inverseMass;
    if (totalInvMass < 1e-7) return;

    if (!pA.pinned) {
      pA.position = pA.position.add(correction.scale(pA.inverseMass / totalInvMass));
    }
    if (!pB.pinned) {
      pB.position = pB.position.sub(correction.scale(pB.inverseMass / totalInvMass));
    }
  }
}
约束投影的迭代次数决定了布料的硬度表现：迭代次数越多，弹簧约束越接近刚性，布料越不会过度拉伸。


布料碰撞检测
布料与场景中的刚体碰撞需要特殊处理。对于简单碰撞体（球、胶囊、平面），可以直接对每个质点进行距离判断；对于复杂网格，需要借助空间分区加速。


interface CollisionPrimitive {
  resolveClothCollision(particle: ClothParticle): void;
}

class SphereCollider implements CollisionPrimitive {
  center: Vector3;
  radius: number;

  constructor(center: Vector3, radius: number) {
    this.center = center;
    this.radius = radius;
  }

  resolveClothCollision(particle: ClothParticle): void {
    const diff = particle.position.sub(this.center);
    const dist = diff.length();

    if (dist < this.radius) {
      const normal = diff.scale(1 / dist);
      particle.position = this.center.add(normal.scale(this.radius));
    }
  }
}

class PlaneCollider implements CollisionPrimitive {
  normal: Vector3;
  distance: number;

  constructor(normal: Vector3, distance: number) {
    this.normal = normal.normalize();
    this.distance = distance;
  }

  resolveClothCollision(particle: ClothParticle): void {
    const signedDist = particle.position.dot(this.normal) - this.distance;
    if (signedDist < 0) {
      particle.position = particle.position.sub(this.normal.scale(signedDist));
    }
  }
}

class ClothCollisionHandler {
  private colliders: CollisionPrimitive[] = [];

  addCollider(collider: CollisionPrimitive): void {
    this.colliders.push(collider);
  }

  resolve(particles: ClothParticle[]): void {
    for (const particle of particles) {
      if (particle.pinned) continue;
      for (const collider of this.colliders) {
        collider.resolveClothCollision(particle);
      }
    }
  }
}

风力与环境交互
风力模拟为布料添加了动态的环境交互效果。风力作用在布料三角面上，力的大小与面法线和风向的夹角相关。


class WindForce {
  direction: Vector3;
  strength: number;
  turbulence: number;
  private time = 0;

  constructor(direction: Vector3, strength: number, turbulence: number) {
    this.direction = direction.normalize();
    this.strength = strength;
    this.turbulence = turbulence;
  }

  applyToTriangle(p1: ClothParticle, p2: ClothParticle, p3: ClothParticle, dt: number): void {
    const edge1 = p2.position.sub(p1.position);
    const edge2 = p3.position.sub(p1.position);
    const faceNormal = edge1.cross(edge2);
    const area = faceNormal.length() * 0.5;

    if (area < 1e-7) return;

    const normalDir = faceNormal.normalize();
    const windVel = this.getWindVelocity(p1.position);
    const dot = windVel.dot(normalDir);
    const force = normalDir.scale(dot * area);

    const perParticleForce = force.scale(1 / 3);
    if (!p1.pinned) p1.position = p1.position.add(perParticleForce.scale(dt * dt));
    if (!p2.pinned) p2.position = p2.position.add(perParticleForce.scale(dt * dt));
    if (!p3.pinned) p3.position = p3.position.add(perParticleForce.scale(dt * dt));
  }

  private getWindVelocity(position: Vector3): Vector3 {
    const turb = new Vector3(
      Math.sin(position.x * 0.5 + this.time) * this.turbulence,
      Math.sin(position.y * 0.7 + this.time * 1.1) * this.turbulence,
      Math.cos(position.z * 0.3 + this.time * 0.9) * this.turbulence,
    );
    return this.direction.scale(this.strength).add(turb);
  }

  tick(dt: number): void {
    this.time += dt;
  }
}
