流体模拟

核心概念

流体模拟在游戏中用于实现水面波动、瀑布飞溅、岩浆流动等效果。与刚体物理不同，流体没有固定形状，需要求解 Navier-Stokes 方程来描述速度场和压力场的演化。游戏中的流体模拟通常采用粒子法（SPH）或网格法（欧拉法），在视觉质量和计算性能之间寻找平衡。

SPH 光滑粒子流体动力学
SPH（Smoothed Particle Hydrodynamics）是游戏中最常用的粒子流体方法。每个粒子携带质量、密度、压力等流体属性，通过核函数对邻居粒子的贡献进行加权求和来估算连续场量。


class SPHKernels {
  static poly6(r: number, h: number): number {
    if (r > h) return 0;
    const h2 = h * h;
    const r2 = r * r;
    const coeff = 315 / (64 * Math.PI * Math.pow(h, 9));
    return coeff * Math.pow(h2 - r2, 3);
  }

  static spikyGradient(r: number, h: number, direction: Vector3): Vector3 {
    if (r > h || r < 1e-7) return Vector3.zero();
    const coeff = -45 / (Math.PI * Math.pow(h, 6));
    const scale = coeff * Math.pow(h - r, 2);
    return direction.scale(scale / r);
  }

  static viscosityLaplacian(r: number, h: number): number {
    if (r > h) return 0;
    const coeff = 45 / (Math.PI * Math.pow(h, 6));
    return coeff * (h - r);
  }
}

interface SPHParticle {
  position: Vector3;
  velocity: Vector3;
  force: Vector3;
  density: number;
  pressure: number;
  mass: number;
}

class SPHSimulator {
  private particles: SPHParticle[] = [];
  private h = 0.1;
  private restDensity = 1000;
  private gasConstant = 2000;
  private viscosity = 250;
  private gravity = new Vector3(0, -9.81, 0);

  computeDensityPressure(): void {
    for (const pi of this.particles) {
      pi.density = 0;
      for (const pj of this.particles) {
        const r = pi.position.sub(pj.position).length();
        pi.density += pj.mass * SPHKernels.poly6(r, this.h);
      }
      pi.pressure = this.gasConstant * (pi.density - this.restDensity);
    }
  }

  computeForces(): void {
    for (const pi of this.particles) {
      let pressureForce = Vector3.zero();
      let viscosityForce = Vector3.zero();

      for (const pj of this.particles) {
        if (pi === pj) continue;
        const diff = pi.position.sub(pj.position);
        const r = diff.length();

        if (r < this.h && r > 1e-7) {
          const pressureGrad = SPHKernels.spikyGradient(r, this.h, diff);
          const pressureTerm = (pi.pressure + pj.pressure) / (2 * pj.density);
          pressureForce = pressureForce.sub(pressureGrad.scale(pj.mass * pressureTerm));

          const viscLap = SPHKernels.viscosityLaplacian(r, this.h);
          const velDiff = pj.velocity.sub(pi.velocity);
          viscosityForce = viscosityForce.add(velDiff.scale((this.viscosity * pj.mass * viscLap) / pj.density));
        }
      }

      pi.force = pressureForce.add(viscosityForce).add(this.gravity.scale(pi.density));
    }
  }

  integrate(dt: number): void {
    for (const p of this.particles) {
      p.velocity = p.velocity.add(p.force.scale(dt / p.density));
      p.position = p.position.add(p.velocity.scale(dt));
    }
  }
}
SPH 的核心挑战是邻居搜索的 O(n²) 复杂度。使用空间哈希或紧凑哈希将复杂度降低到 O(n)，是实时 SPH 的必要优化。


邻居搜索加速
class SpatialHash {
  private cellSize: number;
  private buckets: Map<number, number[]> = new Map();

  constructor(cellSize: number) {
    this.cellSize = cellSize;
  }

  clear(): void {
    this.buckets.clear();
  }

  insert(particles: SPHParticle[]): void {
    this.clear();
    for (let i = 0; i < particles.length; i++) {
      const hash = this.hashPosition(particles[i].position);
      if (!this.buckets.has(hash)) this.buckets.set(hash, []);
      this.buckets.get(hash)!.push(i);
    }
  }

  query(position: Vector3, radius: number, particles: SPHParticle[]): number[] {
    const neighbors: number[] = [];
    const minCell = this.toCell(position.sub(new Vector3(radius, radius, radius)));
    const maxCell = this.toCell(position.add(new Vector3(radius, radius, radius)));

    for (let z = minCell.z; z <= maxCell.z; z++) {
      for (let y = minCell.y; y <= maxCell.y; y++) {
        for (let x = minCell.x; x <= maxCell.x; x++) {
          const hash = this.hashCell(x, y, z);
          const bucket = this.buckets.get(hash);
          if (!bucket) continue;

          for (const idx of bucket) {
            const dist = particles[idx].position.sub(position).length();
            if (dist < radius) neighbors.push(idx);
          }
        }
      }
    }
    return neighbors;
  }

  private toCell(pos: Vector3): { x: number; y: number; z: number } {
    return {
      x: Math.floor(pos.x / this.cellSize),
      y: Math.floor(pos.y / this.cellSize),
      z: Math.floor(pos.z / this.cellSize),
    };
  }

  private hashPosition(pos: Vector3): number {
    const cell = this.toCell(pos);
    return this.hashCell(cell.x, cell.y, cell.z);
  }

  private hashCell(x: number, y: number, z: number): number {
    return ((x * 73856093) ^ (y * 19349663) ^ (z * 83492791)) & 0x7fffffff;
  }
}
网格尺寸应等于或略大于核函数的支撑半径 h，这样每个粒子只需查询周围 27 个网格（3×3×3），实现 O(1) 均摊查询。


高度场水面模拟
对于大范围水面效果，基于高度场的波动方程模拟比粒子法更加高效。它将水面简化为二维网格，每个网格点存储水面高度，通过离散波动方程传播波纹。


class HeightFieldWater {
  private heights: Float32Array;
  private velocities: Float32Array;
  private width: number;
  private depth: number;
  private damping = 0.995;
  private waveSpeed = 2.0;

  constructor(width: number, depth: number) {
    this.width = width;
    this.depth = depth;
    this.heights = new Float32Array(width * depth);
    this.velocities = new Float32Array(width * depth);
  }

  disturb(x: number, z: number, force: number, radius: number): void {
    for (let dz = -radius; dz <= radius; dz++) {
      for (let dx = -radius; dx <= radius; dx++) {
        const gx = Math.floor(x + dx);
        const gz = Math.floor(z + dz);
        if (gx < 0 || gx >= this.width || gz < 0 || gz >= this.depth) continue;
        const dist = Math.sqrt(dx * dx + dz * dz);
        if (dist <= radius) {
          const falloff = 1 - dist / radius;
          this.velocities[gz * this.width + gx] += force * falloff;
        }
      }
    }
  }

  step(dt: number): void {
    const c2 = this.waveSpeed * this.waveSpeed * dt * dt;

    for (let z = 1; z < this.depth - 1; z++) {
      for (let x = 1; x < this.width - 1; x++) {
        const idx = z * this.width + x;
        const laplacian =
          this.heights[idx - 1] +
          this.heights[idx + 1] +
          this.heights[idx - this.width] +
          this.heights[idx + this.width] -
          4 * this.heights[idx];

        this.velocities[idx] += c2 * laplacian;
        this.velocities[idx] *= this.damping;
      }
    }

    for (let i = 0; i < this.heights.length; i++) {
      this.heights[i] += this.velocities[i] * dt;
    }
  }

  getHeight(x: number, z: number): number {
    const gx = Math.floor(x);
    const gz = Math.floor(z);
    if (gx < 0 || gx >= this.width || gz < 0 || gz >= this.depth) return 0;
    return this.heights[gz * this.width + gx];
  }
}

小结
    

  • SPH 通过核函数对邻居粒子的贡献加权求和来近似连续流体场
    • 

  • 空间哈希将邻居搜索从 O(n²) 降低到 O(n)，网格尺寸应匹配核函数支撑半径
    • 

  • 高度场水面模拟通过二维波动方程实现大范围水面效果，性能远优于粒子法
    • 

  • 气体常数和粘性系数控制流体的可压缩性和粘滞度
    • 

  • 实际游戏中通常组合使用多种方法：高度场处理大水面，SPH 处理溅射和交互
