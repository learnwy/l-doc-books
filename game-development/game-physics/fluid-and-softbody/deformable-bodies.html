可变形体

核心概念

可变形体（Deformable Bodies）模拟是刚体物理和流体物理之间的中间地带，用于实现果冻状物体的弹性变形、软体角色的挤压碰撞、地形的塑性破坏等效果。与刚体不同，可变形体在受力时会改变形状；与流体不同，可变形体具有固有的静态形状并在外力消失后趋于恢复。

Shape Matching 方法
Shape Matching 是一种几何驱动的软体模拟方法，由 Müller 等人提出。它不使用弹簧，而是通过计算当前变形形状与原始形状之间的最优刚体变换来产生恢复力，然后将粒子向目标位置拉动。


interface DeformableParticle {
  position: Vector3;
  restPosition: Vector3;
  goalPosition: Vector3;
  velocity: Vector3;
  mass: number;
}

class ShapeMatchingSolver {
  private particles: DeformableParticle[];
  private stiffness = 0.5;
  private totalMass = 0;
  private restCenterOfMass: Vector3;

  constructor(particles: DeformableParticle[]) {
    this.particles = particles;
    this.totalMass = particles.reduce((sum, p) => sum + p.mass, 0);
    this.restCenterOfMass = this.computeCenterOfMass((p) => p.restPosition);
  }

  private computeCenterOfMass(getPos: (p: DeformableParticle) => Vector3): Vector3 {
    let com = Vector3.zero();
    for (const p of this.particles) {
      com = com.add(getPos(p).scale(p.mass));
    }
    return com.scale(1 / this.totalMass);
  }

  solve(): void {
    const currentCOM = this.computeCenterOfMass((p) => p.position);

    let Apq = Matrix3.zero();
    for (const p of this.particles) {
      const q = p.restPosition.sub(this.restCenterOfMass);
      const pi = p.position.sub(currentCOM);
      Apq = Apq.addOuterProduct(pi.scale(p.mass), q);
    }

    const { rotation } = this.polarDecomposition(Apq);

    for (const p of this.particles) {
      const relativeRest = p.restPosition.sub(this.restCenterOfMass);
      const target = rotation.mulVector(relativeRest).add(currentCOM);
      p.goalPosition = target;
      p.position = p.position.add(target.sub(p.position).scale(this.stiffness));
    }
  }

  private polarDecomposition(A: Matrix3): { rotation: Matrix3; stretch: Matrix3 } {
    let R = A.clone();
    for (let i = 0; i < 10; i++) {
      const Rit = R.inverse().transpose();
      R = R.add(Rit).scale(0.5);
    }
    const S = R.transpose().multiply(A);
    return { rotation: R, stretch: S };
  }
}
Shape Matching 的优势在于无条件稳定——无论时间步长多大都不会爆炸，因为它通过几何拟合而非力求解来确定目标位置。


基于位置的动力学（PBD）
PBD（Position Based Dynamics）直接在位置层面处理约束，避免了力和加速度的中间计算。它的迭代过程天然稳定，非常适合实时游戏中的软体模拟。


interface PBDConstraint {
  solve(particles: DeformableParticle[]): void;
}

class DistanceConstraint implements PBDConstraint {
  private idx1: number;
  private idx2: number;
  private restLength: number;
  private stiffness: number;

  constructor(idx1: number, idx2: number, restLength: number, stiffness: number) {
    this.idx1 = idx1;
    this.idx2 = idx2;
    this.restLength = restLength;
    this.stiffness = stiffness;
  }

  solve(particles: DeformableParticle[]): void {
    const p1 = particles[this.idx1];
    const p2 = particles[this.idx2];
    const diff = p2.position.sub(p1.position);
    const dist = diff.length();
    if (dist < 1e-7) return;

    const error = dist - this.restLength;
    const correction = diff.scale(error / dist);
    const w1 = 1 / p1.mass;
    const w2 = 1 / p2.mass;
    const totalW = w1 + w2;

    p1.position = p1.position.add(correction.scale((this.stiffness * w1) / totalW));
    p2.position = p2.position.sub(correction.scale((this.stiffness * w2) / totalW));
  }
}

class VolumeConstraint implements PBDConstraint {
  private indices: number[];
  private restVolume: number;
  private stiffness: number;

  constructor(indices: number[], restVolume: number, stiffness: number) {
    this.indices = indices;
    this.restVolume = restVolume;
    this.stiffness = stiffness;
  }

  solve(particles: DeformableParticle[]): void {
    const [i0, i1, i2, i3] = this.indices;
    const p0 = particles[i0].position;
    const p1 = particles[i1].position;
    const p2 = particles[i2].position;
    const p3 = particles[i3].position;

    const volume = p1.sub(p0).dot(p2.sub(p0).cross(p3.sub(p0))) / 6;
    const error = volume - this.restVolume;

    const grad0 = p1
      .sub(p2)
      .cross(p3.sub(p2))
      .scale(1 / 6);
    const grad1 = p2
      .sub(p0)
      .cross(p3.sub(p0))
      .scale(1 / 6);
    const grad2 = p0
      .sub(p1)
      .cross(p3.sub(p1))
      .scale(1 / 6);
    const grad3 = p1
      .sub(p0)
      .cross(p2.sub(p0))
      .scale(1 / 6);

    const sumGrad2 = grad0.lengthSq() + grad1.lengthSq() + grad2.lengthSq() + grad3.lengthSq();
    if (sumGrad2 < 1e-10) return;

    const lambda = (-error * this.stiffness) / sumGrad2;

    particles[i0].position = particles[i0].position.add(grad0.scale(lambda));
    particles[i1].position = particles[i1].position.add(grad1.scale(lambda));
    particles[i2].position = particles[i2].position.add(grad2.scale(lambda));
    particles[i3].position = particles[i3].position.add(grad3.scale(lambda));
  }
}

class PBDSolver {
  private constraints: PBDConstraint[] = [];
  private iterations = 10;
  private gravity = new Vector3(0, -9.81, 0);

  step(particles: DeformableParticle[], dt: number): void {
    for (const p of particles) {
      p.velocity = p.velocity.add(this.gravity.scale(dt));
      p.goalPosition = p.position.add(p.velocity.scale(dt));
    }

    const predicted = particles.map((p) => p.position.clone());
    for (const p of particles) {
      p.position = p.goalPosition;
    }

    for (let i = 0; i < this.iterations; i++) {
      for (const constraint of this.constraints) {
        constraint.solve(particles);
      }
    }

    for (let i = 0; i < particles.length; i++) {
      particles[i].velocity = particles[i].position.sub(predicted[i]).scale(1 / dt);
    }
  }
}
断裂与撕裂
可变形体的断裂模拟需要在运行时动态修改网格拓扑。当某条边上的应力超过断裂阈值时，沿该边将网格拆分为两个独立部分。


class FractureSystem {
  private fractureThreshold: number;

  constructor(threshold: number) {
    this.fractureThreshold = threshold;
  }

  checkFracture(
    particles: DeformableParticle[],
    constraints: DistanceConstraint[],
    getStress: (c: DistanceConstraint) => number,
  ): DistanceConstraint[] {
    const fractured: DistanceConstraint[] = [];

    for (const constraint of constraints) {
      const stress = getStress(constraint);
      if (stress > this.fractureThreshold) {
        fractured.push(constraint);
      }
    }

    return fractured;
  }

  applyFracture(
    particles: DeformableParticle[],
    constraints: DistanceConstraint[],
    fractured: DistanceConstraint[],
  ): DistanceConstraint[] {
    const fracturedSet = new Set(fractured);
    const remaining = constraints.filter((c) => !fracturedSet.has(c));

    for (const fc of fractured) {
      this.separateParticles(particles, fc);
    }

    return remaining;
  }

预碎片化（Pre-fracture）是游戏中最常用的断裂方法：在离线阶段用 Voronoi 图将物体预分割为碎块，运行时只需根据冲击力决定哪些碎块分离，避免了实时网格切割的高昂成本。


小结
    

  • Shape Matching 通过几何拟合产生恢复力，无条件稳定且实现简单
    • 

  • PBD 直接在位置层面处理约束，天然稳定，适合实时软体模拟
    • 

  • 体积约束保持四面体体积不变，防止软体过度压缩或膨胀
    • 

  • 断裂模拟通常使用预碎片化方法，运行时仅判断碎块分离条件
    • 

  • XPBD 通过引入合规矩阵扩展了 PBD，支持材料属性的物理正确参数化
