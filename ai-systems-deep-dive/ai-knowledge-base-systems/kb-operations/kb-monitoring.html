知识库监控与告警

核心概念

知识库监控与告警体系是保障知识库系统在生产环境中稳定运行的关键基础设施。一个完善的监控体系需要覆盖检索质量、知识覆盖率、系统性能和数据一致性四个核心维度。通过实时采集关键指标、设定智能告警规则和建立异常检测机制，运维团队能够在问题影响用户之前及时发现和响应。

检索质量监控
检索质量是知识库系统最核心的质量指标，直接影响下游 RAG 系统的生成质量。检索质量监控需要从在线和离线两个维度持续追踪检索效果的变化。


interface RetrievalQualityMetrics {
  readonly avgRelevanceScore: number;
  readonly emptyResultRate: number;
  readonly topKHitRate: number;
  readonly mrr: number;
  readonly queryLatencyP50: number;
  readonly queryLatencyP99: number;
  readonly totalQueries: number;
  readonly timestamp: number;
}

interface QualityMonitorConfig {
  readonly samplingRate: number;
  readonly windowSizeMinutes: number;
  readonly relevanceThreshold: number;
  readonly emptyResultAlertThreshold: number;
  readonly latencyAlertP99Ms: number;
}

const computeRetrievalQuality = (
  queryResults: ReadonlyArray<{
    readonly query: string;
    readonly results: ReadonlyArray<{ readonly docId: string; readonly score: number }>;
    readonly latencyMs: number;
    readonly hasRelevantResult: boolean;
  }>,
): RetrievalQualityMetrics => {
  const totalQueries = queryResults.length;
  if (totalQueries === 0) {
    return {
      avgRelevanceScore: 0,
      emptyResultRate: 0,
      topKHitRate: 0,
      mrr: 0,
      queryLatencyP50: 0,
      queryLatencyP99: 0,
      totalQueries: 0,
      timestamp: Date.now(),
    };
  }

  const emptyCount = queryResults.filter((q) => q.results.length === 0).length;
  const hitCount = queryResults.filter((q) => q.hasRelevantResult).length;

  const avgScore =
    queryResults.filter((q) => q.results.length > 0).reduce((sum, q) => sum + q.results[0].score, 0) /
    Math.max(totalQueries - emptyCount, 1);

  const sortedLatencies = [...queryResults].map((q) => q.latencyMs).sort((a, b) => a - b);
  const p50Index = Math.floor(sortedLatencies.length * 0.5);
  const p99Index = Math.floor(sortedLatencies.length * 0.99);

  const mrrSum = queryResults.reduce((sum, q) => {
    const firstRelevantIdx = q.results.findIndex((r) => r.score >= 0.7);
    return sum + (firstRelevantIdx >= 0 ? 1 / (firstRelevantIdx + 1) : 0);
  }, 0);

  return {
    avgRelevanceScore: avgScore,
    emptyResultRate: emptyCount / totalQueries,
    topKHitRate: hitCount / totalQueries,
    mrr: mrrSum / totalQueries,
    queryLatencyP50: sortedLatencies[p50Index] ?? 0,
    queryLatencyP99: sortedLatencies[p99Index] ?? 0,
    totalQueries,
    timestamp: Date.now(),
  };
};
在线检索质量监控通过采样线上查询请求，实时计算平均相关性分数、空结果率和 MRR。当这些指标出现突变（如空结果率突然上升 10%），通常意味着索引损坏、嵌入模型异常或知识源数据丢失。离线质量评估则通过定期运行标注好的评测集（Golden Set），检测检索效果的缓慢退化。


告警规则与异常检测
简单的阈值告警在监控指标波动较大时容易产生大量误报。智能告警系统结合统计方法和业务规则来提高告警的准确性。


interface AlertRule {
  readonly name: string;
  readonly metric: string;
  readonly condition: {
    readonly type: 'threshold' | 'rate-of-change' | 'anomaly-detection';
    readonly params: {
      readonly threshold?: number;
      readonly direction?: 'above' | 'below';
      readonly changeRatePercent?: number;
      readonly windowMinutes?: number;
      readonly sensitivity?: number;
    };
  };
  readonly severity: 'critical' | 'warning' | 'info';
  readonly cooldownMinutes: number;
  readonly notifyChannels: ReadonlyArray<string>;
}

interface AnomalyDetector {
  readonly windowSize: number;
  readonly history: ReadonlyArray<number>;
  readonly mean: number;
  readonly stddev: number;
}

const detectAnomaly = (
  detector: AnomalyDetector,
  currentValue: number,
  sensitivity: number,
): { readonly isAnomaly: boolean; readonly zScore: number; readonly expectedRange: readonly [number, number] } => {
  const zScore = detector.stddev > 0 ? (currentValue - detector.mean) / detector.stddev : 0;

  const threshold = sensitivity * detector.stddev;
  const expectedRange = [detector.mean - threshold, detector.mean + threshold] as const;

  return {
    isAnomaly: Math.abs(currentValue - detector.mean) > threshold,
    zScore,
    expectedRange,
  };
};

const updateDetector = (detector: AnomalyDetector, newValue: number): AnomalyDetector => {
  const newHistory = [...detector.history, newValue].slice(-detector.windowSize);
  const mean = newHistory.reduce((a, b) => a + b, 0) / newHistory.length;
  const variance = newHistory.reduce((sum, v) => sum + (v - mean) ** 2, 0) / newHistory.length;
  return {
    windowSize: detector.windowSize,
    history: newHistory,
    mean,
    stddev: Math.sqrt(variance),
  };
};

监控仪表盘设计
监控仪表盘是运维团队理解系统状态的核心工具。好的仪表盘设计遵循信息分层原则：最上层是全局健康度指标（红黄绿），中层是各维度的趋势图表，底层是详细的诊断数据。


interface DashboardConfig {
  readonly panels: ReadonlyArray<{
    readonly title: string;
    readonly type: 'stat' | 'timeseries' | 'table' | 'heatmap';
    readonly metrics: ReadonlyArray<string>;
    readonly refreshInterval: number;
    readonly timeRange: string;
  }>;
  readonly healthScore: {
    readonly weights: ReadonlyMap<string, number>;
    readonly thresholds: {
      readonly green: number;
      readonly yellow: number;
      readonly red: number;
    };
  };
}

interface HealthScoreResult {
  readonly overall: number;
  readonly status: 'healthy' | 'degraded' | 'critical';
  readonly dimensions: ReadonlyMap<
    string,
    {
      readonly score: number;
      readonly status: 'healthy' | 'degraded' | 'critical';
    }
  >;
}

const computeHealthScore = (
  metrics: ReadonlyMap<string, number>,
  config: DashboardConfig['healthScore'],
): HealthScoreResult => {
  let weightedSum = 0;
  let totalWeight = 0;
  const dimensions = new Map<string, { score: number; status: 'healthy' | 'degraded' | 'critical' }>();

  for (const [metric, weight] of config.weights) {
    const value = metrics.get(metric) ?? 0;
    const normalizedScore = Math.min(Math.max(value, 0), 1);
    weightedSum += normalizedScore * weight;
    totalWeight += weight;

    const status =
      normalizedScore >= config.thresholds.green
        ? 'healthy'
        : normalizedScore >= config.thresholds.yellow
          ? 'degraded'
          : 'critical';
    dimensions.set(metric, { score: normalizedScore, status });
  }

  const overall = totalWeight > 0 ? weightedSum / totalWeight : 0;
  const overallStatus =
    overall >= config.thresholds.green ? 'healthy' : overall >= config.thresholds.yellow ? 'degraded' : 'critical';

  return { overall, status: overallStatus, dimensions };
};
关键设计原则包括：将检索质量、系统性能、知识覆盖和数据一致性分为独立面板；使用色彩编码（绿/黄/红）标识状态；提供向下钻取（drill-down）能力从全局到具体查询。
