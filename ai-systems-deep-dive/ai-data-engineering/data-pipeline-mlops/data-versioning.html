数据版本控制

核心概念

数据版本控制是对训练数据集、特征定义和数据转换逻辑进行系统化版本管理的实践。与代码版本控制类似，数据版本控制实现数据的可追溯、可回滚和可复现，是 MLOps 中实验可复现性和模型审计的基础。

数据集版本管理
数据集通常体积庞大（GB-TB 级），不适合直接存储在 Git 中。专用的数据版本工具通过分离元数据和数据内容来高效管理版本。


DVC（Data Version Control） 的核心思路是在 Git 中存储数据的哈希指针，数据本身存储在远端存储（S3、GCS 等）：


interface DVCConfig {
  remoteStorage: {
    type: 's3' | 'gcs' | 'azure' | 'local';
    bucket: string;
    prefix: string;
    credentials: CredentialConfig;
  };
  cacheDir: string;
  autostage: boolean;
}

interface DVCFile {
  path: string;
  md5: string;
  size: number;
  nfiles?: number;
  hash: 'md5';
}

interface DatasetVersion {
  versionId: string;
  commitHash: string;
  timestamp: Date;
  files: DVCFile[];
  metrics: Record<string, number>;
  params: Record<string, unknown>;
  tags: string[];
}

function trackDataset(dataPath: string, config: DVCConfig): DVCFile {
  const hash = computeFileHash(dataPath, 'md5');
  const remotePath = `${config.remoteStorage.prefix}/${hash}`;

  uploadToRemote(dataPath, remotePath, config.remoteStorage);

  const dvcFile: DVCFile = {
    path: dataPath,
    md5: hash,
    size: getFileSize(dataPath),
    hash: 'md5',
  };

  writeDVCMetaFile(`${dataPath}.dvc`, dvcFile);
  return dvcFile;
}

变更追踪与差异分析
了解数据版本之间的差异对于调试模型性能变化至关重要。数据差异分析不仅比较行级变更，还需要分析统计分布的变化。


interface DataDiff {
  versionA: string;
  versionB: string;
  summary: {
    addedRows: number;
    removedRows: number;
    modifiedRows: number;
    unchangedRows: number;
  };
  schemaDiff: {
    addedColumns: string[];
    removedColumns: string[];
    typeChanges: { column: string; from: string; to: string }[];
  };
  statisticalDiff: {
    column: string;
    metricA: DescriptiveStats;
    metricB: DescriptiveStats;
    psiScore: number;
    significant: boolean;
  }[];
  labelDistributionDiff: {
    label: string;
    countA: number;
    countB: number;
    changePercent: number;
  }[];
}

function computeDataDiff(dataA: DataSnapshot, dataB: DataSnapshot, keyColumns: string[]): DataDiff {
  const keysA = new Set(dataA.rows.map((r) => extractKey(r, keyColumns)));
  const keysB = new Set(dataB.rows.map((r) => extractKey(r, keyColumns)));

  const added = [...keysB].filter((k) => !keysA.has(k));
  const removed = [...keysA].filter((k) => !keysB.has(k));
  const common = [...keysA].filter((k) => keysB.has(k));

  const modified = common.filter((key) => {
    const rowA = dataA.getByKey(key);
    const rowB = dataB.getByKey(key);
    return !deepEqual(rowA, rowB);
  });

  return {
    versionA: dataA.version,
    versionB: dataB.version,
    summary: {
      addedRows: added.length,
      removedRows: removed.length,
      modifiedRows: modified.length,
      unchangedRows: common.length - modified.length,
    },
    schemaDiff: compareSchemata(dataA.schema, dataB.schema),
    statisticalDiff: compareStatistics(dataA, dataB),
    labelDistributionDiff: compareLabelDistribution(dataA, dataB),
  };
}

管道版本与代码-数据绑定
数据版本需要与代码版本和模型版本建立关联，实现端到端的可复现性。


interface ExperimentSnapshot {
  experimentId: string;
  timestamp: Date;
  codeVersion: {
    gitCommit: string;
    gitBranch: string;
    gitTag?: string;
  };
  dataVersion: {
    trainingSet: DatasetVersion;
    validationSet: DatasetVersion;
    testSet: DatasetVersion;
    featureDefinitions: string;
  };
  modelVersion: {
    modelId: string;
    hyperparameters: Record<string, unknown>;
    metrics: Record<string, number>;
    artifacts: string[];
  };
  environment: {
    pythonVersion: string;
    dependencies: Record<string, string>;
    hardwareSpec: string;
  };
}

function ensureReproducibility(snapshot: ExperimentSnapshot): ReproducibilityCheck {
  const checks = [
    checkCodeVersion(snapshot.codeVersion),
    checkDataAvailability(snapshot.dataVersion),
    checkModelArtifacts(snapshot.modelVersion),
    checkEnvironmentMatch(snapshot.environment),
  ];

  return {
    reproducible: checks.every((c) => c.passed),
    details: checks,
    missingComponents: checks.filter((c) => !c.passed).map((c) => c.component),
  };
}
数据沿袭与合规
数据沿袭（Data Lineage）记录数据从产生到消费的完整生命周期，是合规审计和问题定位的核心能力。


interface DataLineage {
  datasetId: string;
  version: string;
  upstream: LineageNode[];
  downstream: LineageNode[];
  transformations: TransformationRecord[];
  accessLog: AccessRecord[];
}

interface LineageNode {
  nodeId: string;
  nodeType: 'source' | 'transform' | 'dataset' | 'model' | 'endpoint';
  name: string;
  version: string;
  metadata: Record<string, unknown>;
}

interface TransformationRecord {
  transformId: string;
  inputDatasets: string[];
  outputDataset: string;
  logic: string;
  executedAt: Date;
  executedBy: string;
  codeVersion: string;
}

小结
    

  • 数据版本控制是 MLOps 可复现性和审计的基础，不可或缺
    • 

  • DVC 通过哈希指针分离元数据和数据内容，适合与 Git 工作流集成
    • 

  • LakeFS 提供数据湖原生的分支和合并，适合大规模数据实验
    • 

  • 数据差异分析应涵盖行级变更和统计分布变化，PSI 是判断是否需要重训练的关键指标
    • 

  • 代码、数据和模型的版本绑定是端到端可复现的前提
    • 

  • 数据沿袭不仅服务于技术调试，更是合规审计的法律要求
