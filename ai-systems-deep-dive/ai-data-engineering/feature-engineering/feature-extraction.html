特征提取方法

核心概念

特征提取是将原始数据转化为机器学习模型可用的数值表示的过程。高质量的特征能够捕获数据中的关键信息，直接决定模型的上限。特征提取方法因数据类型不同而差异显著，但核心目标一致：在保留信息的同时降低数据的复杂度。

数值特征工程
数值特征是最常见的数据类型，工程化处理包括变换、分箱、交叉和聚合等操作。每种操作解决特定的问题，组合使用能显著提升模型效果。


数值变换 改变特征的分布形态，使其更适合模型学习：


interface NumericTransformer {
  transform(values: number[]): number[];
  inverseTransform(values: number[]): number[];
  fitTransform(values: number[]): number[];
}

function logTransform(values: number[]): number[] {
  return values.map((v) => Math.log1p(Math.max(v, 0)));
}

function boxCoxTransform(values: number[], lambda: number): number[] {
  if (Math.abs(lambda) < 1e-10) {
    return values.map((v) => Math.log(v));
  }
  return values.map((v) => (Math.pow(v, lambda) - 1) / lambda);
}

function standardize(values: number[]): number[] {
  const mean = values.reduce((s, v) => s + v, 0) / values.length;
  const std = Math.sqrt(values.reduce((s, v) => s + (v - mean) ** 2, 0) / values.length);
  return values.map((v) => (v - mean) / (std || 1));
}

function robustScale(values: number[]): number[] {
  const sorted = [...values].sort((a, b) => a - b);
  const median = sorted[Math.floor(sorted.length / 2)];
  const q1 = sorted[Math.floor(sorted.length * 0.25)];
  const q3 = sorted[Math.floor(sorted.length * 0.75)];
  const iqr = q3 - q1;
  return values.map((v) => (v - median) / (iqr || 1));
}

特征分箱（Binning） 将连续变量离散化，能够引入非线性并增强模型对异常值的鲁棒性：


interface BinningStrategy {
  method: 'equal_width' | 'equal_frequency' | 'kmeans' | 'decision_tree';
  nBins: number;
  encode: 'ordinal' | 'onehot' | 'target';
}

function equalFrequencyBinning(values: number[], nBins: number): { boundaries: number[]; binned: number[] } {
  const sorted = [...values].sort((a, b) => a - b);
  const binSize = Math.ceil(sorted.length / nBins);
  const boundaries: number[] = [];

  for (let i = 1; i < nBins; i++) {
    boundaries.push(sorted[Math.min(i * binSize, sorted.length - 1)]);
  }

  const binned = values.map((v) => {
    let bin = 0;
    for (const b of boundaries) {
      if (v > b) bin++;
    }
    return bin;
  });

  return { boundaries, binned };
}

文本特征提取
文本数据的特征提取经历了从统计方法到深度学习嵌入的演进。现代方法以预训练模型嵌入为主，但传统方法在特定场景仍有独特优势。


TF-IDF 特征 兼顾词频和逆文档频率，是最经典的文本特征化方法：


interface TFIDFConfig {
  maxFeatures: number;
  minDf: number;
  maxDf: number;
  ngramRange: [number, number];
  sublinearTf: boolean;
}

function calculateTFIDF(documents: string[][], config: TFIDFConfig): Map<string, number[]> {
  const df = new Map<string, number>();
  const N = documents.length;

  for (const doc of documents) {
    const uniqueTerms = new Set(doc);
    for (const term of uniqueTerms) {
      df.set(term, (df.get(term) ?? 0) + 1);
    }
  }

  const idf = new Map<string, number>();
  for (const [term, count] of df) {
    if (count >= config.minDf * N && count <= config.maxDf * N) {
      idf.set(term, Math.log((N + 1) / (count + 1)) + 1);
    }
  }

  const result = new Map<string, number[]>();
  for (const [i, doc] of documents.entries()) {
    const tf = new Map<string, number>();
    for (const term of doc) {
      tf.set(term, (tf.get(term) ?? 0) + 1);
    }

    for (const [term, idfVal] of idf) {
      const tfVal = tf.get(term) ?? 0;
      const normalizedTf = config.sublinearTf ? (tfVal > 0 ? 1 + Math.log(tfVal) : 0) : tfVal / doc.length;

      if (!result.has(term)) result.set(term, new Array(N).fill(0));
      result.get(term)![i] = normalizedTf * idfVal;
    }
  }

  return result;
}

时序特征提取
时序数据的特征提取需要同时考虑时域和频域的信息，以及趋势、周期性和突变等时间维度的特征。


滑动窗口统计 是最基础也最实用的时序特征提取方法：


interface TimeSeriesFeatureConfig {
  windows: number[];
  statistics: StatType[];
  lagFeatures: number[];
  rollingFeatures: boolean;
}

type StatType = 'mean' | 'std' | 'min' | 'max' | 'median' | 'skew' | 'kurtosis' | 'trend';

function extractWindowFeatures(series: number[], windowSize: number, stats: StatType[]): Record<string, number>[] {
  const features: Record<string, number>[] = [];

  for (let i = windowSize; i <= series.length; i++) {
    const window = series.slice(i - windowSize, i);
    const feat: Record<string, number> = {};

    for (const stat of stats) {
      feat[`${stat}_${windowSize}`] = computeStat(window, stat);
    }

    feat[`lag_1`] = series[i - 1];
    feat[`lag_${windowSize}`] = series[i - windowSize];
    feat[`diff_1`] = series[i - 1] - (series[i - 2] ?? 0);

    features.push(feat);
  }

  return features;
}

function computeStat(values: number[], stat: StatType): number {
  const n = values.length;
  const mean = values.reduce((s, v) => s + v, 0) / n;

  switch (stat) {
    case 'mean':
      return mean;
    case 'std':
      return Math.sqrt(values.reduce((s, v) => s + (v - mean) ** 2, 0) / n);
    case 'min':
      return Math.min(...values);
    case 'max':
      return Math.max(...values);
    case 'median': {
      const sorted = [...values].sort((a, b) => a - b);
      return sorted[Math.floor(n / 2)];
    }
    case 'trend': {
      const xMean = (n - 1) / 2;
      const num = values.reduce((s, v, i) => s + (i - xMean) * (v - mean), 0);
      const den = values.reduce((s, _, i) => s + (i - xMean) ** 2, 0);
      return den !== 0 ? num / den : 0;
    }
    default:
      return 0;
  }
}

小结
    

  • 特征提取是将原始数据转化为模型可用表示的关键步骤，直接决定模型效果上限
    • 

  • 数值特征通过变换、分箱和交叉操作引入非线性和交互信息
    • 

  • 文本特征从 TF-IDF 演进到预训练嵌入，语义理解能力显著提升
    • 

  • 时序特征需要同时提取时域统计量和频域频谱信息
    • 

  • 深度学习嵌入（BERT、CNN）逐步替代手工特征，但在小数据场景传统方法仍有优势
    • 

  • 特征提取方法的选择应基于数据规模、模型类型和可解释性需求综合判断
