深度推荐模型

核心概念

深度推荐模型（Deep Recommendation Models）利用深度学习自动学习特征交互和用户偏好，突破了传统方法依赖手工特征工程的瓶颈。从 Wide & Deep 开始，深度推荐模型经历了从简单 MLP 到注意力机制、图神经网络和序列建模的快速演进。

Wide & Deep 与特征交叉网络
Wide & Deep 模型由 Google 在 2016 年提出，将线性模型（Wide）的记忆能力和深度网络（Deep）的泛化能力结合。Wide 部分捕获高频的特征组合模式，Deep 部分学习低频和未见过的特征交互。


interface WideDeepConfig {
  wideFeatures: string[];
  deepFeatures: string[];
  deepLayers: number[];
  embeddingDim: number;
  dropout: number;
}

function wideDeepForward(
  wideInput: Float32Array,
  deepInput: Float32Array,
  wideWeight: Float32Array,
  deepLayers: MLPLayer[],
): number {
  const wideOutput = dotProduct(wideInput, wideWeight);

  let hidden = deepInput;
  for (const layer of deepLayers) {
    hidden = relu(addVec(matVecMul(layer.weight, hidden), layer.bias));
  }
  const deepOutput = hidden[0];

  return sigmoid(wideOutput + deepOutput);
}

interface DCNv2Config {
  inputDim: number;
  crossLayers: number;
  deepLayers: number[];
  mixType: 'parallel' | 'stacked';
}

function crossNetworkV2(x0: Float32Array, crossLayers: CrossLayerV2[]): Float32Array {
  let xi = new Float32Array(x0);

  for (const layer of crossLayers) {
    const projected = matVecMul(layer.weightMatrix, xi);
    const next = new Float32Array(x0.length);
    for (let j = 0; j < x0.length; j++) {
      next[j] = x0[j] * projected[j] + layer.bias[j] + xi[j];
    }
    xi = next;
  }

  return xi;
}
DCN-V2（Deep & Cross Network V2）用矩阵替代向量来参数化交叉层，显著提升了特征交叉的表达能力。相比 DeepFM 的 FM 二阶交叉，DCN-V2 可以建模任意阶的显式特征交叉。


用户行为序列建模
DIN（Deep Interest Network）是阿里巴巴提出的里程碑式工作，首次将注意力机制引入推荐模型。核心思想是：用户的兴趣是多样的，不同的候选物品应该激活用户历史中不同的兴趣点。


interface DINConfig {
  embeddingDim: number;
  attentionHiddenUnits: number[];
  maxSeqLength: number;
  dnnLayers: number[];
}

function dinAttention(
  queries: Float32Array,
  keys: Float32Array[],
  keyMask: boolean[],
  attentionMLP: MLPLayer[],
): Float32Array {
  const scores: number[] = [];
  for (let i = 0; i < keys.length; i++) {
    if (!keyMask[i]) {
      scores.push(-Infinity);
      continue;
    }
    const diff = subtractVec(queries, keys[i]);
    const product = hadamardProduct(queries, keys[i]);
    const features = concatVec([queries, keys[i], diff, product]);
    let hidden = features;
    for (const layer of attentionMLP) {
      hidden = relu(addVec(matVecMul(layer.weight, hidden), layer.bias));
    }
    scores.push(hidden[0]);
  }

  const weights = softmax(new Float32Array(scores));
  const dim = keys[0].length;
  const output = new Float32Array(dim);
  for (let i = 0; i < keys.length; i++) {
    for (let d = 0; d < dim; d++) {
      output[d] += weights[i] * keys[i][d];
    }
  }
  return output;
}
DIEN（Deep Interest Evolution Network）在 DIN 基础上引入兴趣演化建模。它使用 GRU 网络捕获用户兴趣随时间的变化轨迹，并通过 AUGRU（Attention Update Gate Recurrent Unit）让兴趣演化的方向受到候选物品的引导。


SIM（Search-based Interest Model）解决超长用户行为序列（千级以上）的建模问题。它采用两阶段策略：先用硬搜索（品类匹配）从超长序列中检索出与候选相关的子序列，再用软搜索（注意力机制）精细建模。

interface MMOEConfig {
  numExperts: number;
  expertDim: number;
  numTasks: number;
  gateType: 'softmax' | 'topk';
}

function mmoeForward(input: Float32Array, experts: MLPLayer[][], gates: MLPLayer[][]): Float32Array[] {
  const expertOutputs = experts.map((expert) => {
    let hidden = input;
    for (const layer of expert) {
      hidden = relu(addVec(matVecMul(layer.weight, hidden), layer.bias));
    }
    return hidden;
  });

  return gates.map((gate) => {
    let gateInput = input;
    for (const layer of gate) {
      gateInput = addVec(matVecMul(layer.weight, gateInput), layer.bias);
    }
    const gateWeights = softmax(gateInput);

    const dim = expertOutputs[0].length;
    const output = new Float32Array(dim);
    for (let e = 0; e < expertOutputs.length; e++) {
      for (let d = 0; d < dim; d++) {
        output[d] += gateWeights[e] * expertOutputs[e][d];
      }
    }
    return output;
  });
}
