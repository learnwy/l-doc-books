梯度下降

什么是梯度下降

梯度下降（Gradient Descent）是机器学习中最基础的优化算法。它通过计算损失函数关于参数的梯度（方向导数），沿着梯度的反方向更新参数，逐步逼近损失函数的最小值。

θt+1=θt−α∇θL(θt)\theta_{t+1} = \theta_t - \alpha \nabla_\theta L(\theta_t)θt+1​=θt​−α∇θ​L(θt​)

其中 α\alphaα 是学习率（Learning Rate），控制每次更新的步长。

import numpy as np

np.random.seed(42)
X = np.random.randn(1000, 2)
w_true = np.array([3.0, -2.0])
y = X @ w_true + np.random.randn(1000) * 0.1

def bgd(X, y, lr=0.01, epochs=100):
    w = np.zeros(2)
    for epoch in range(epochs):
        grad = -2 * X.T @ (y - X @ w) / len(y)
        w -= lr * grad
    return w

def sgd(X, y, lr=0.01, epochs=10):
    w = np.zeros(2)
    for epoch in range(epochs):
        for i in np.random.permutation(len(y)):
            grad = -2 * X[i] * (y[i] - X[i] @ w)
            w -= lr * grad
    return w

def mini_batch_gd(X, y, lr=0.01, epochs=50, batch_size=32):
    w = np.zeros(2)
    for epoch in range(epochs):
        indices = np.random.permutation(len(y))
        for start in range(0, len(y), batch_size):
            batch = indices[start:start + batch_size]
            grad = -2 * X[batch].T @ (y[batch] - X[batch] @ w) / len(batch)
            w -= lr * grad
    return w

for name, fn in [("BGD", bgd), ("SGD", sgd), ("Mini-batch", mini_batch_gd)]:
    w = fn(X, y)
    print(f"{name:12s} → w = [{w[0]:.4f}, {w[1]:.4f}]")
print(f"{'真实值':12s} → w = [{w_true[0]:.4f}, {w_true[1]:.4f}]")


动量法（Momentum）
SGD 在损失函数表面有"锯齿形"振荡。动量法引入历史梯度的指数加权平均，加速收敛并减少振荡。


vt=βvt−1+∇L(θt)v_t = \beta v_{t-1} + \nabla L(\theta_t)vt​=βvt−1​+∇L(θt​)
 


θt+1=θt−αvt\theta_{t+1} = \theta_t - \alpha v_tθt+1​=θt​−αvt​
import torch
import torch.nn as nn
import torch.optim as optim

torch.manual_seed(42)
model = nn.Linear(10, 1)
X = torch.randn(100, 10)
y = torch.randn(100, 1)

optimizers = {
    "SGD":          optim.SGD(model.parameters(), lr=0.01),
    "SGD+Momentum": optim.SGD(model.parameters(), lr=0.01, momentum=0.9),
}

for name, opt in optimizers.items():
    model_copy = nn.Linear(10, 1)
    model_copy.load_state_dict(model.state_dict())
    opt_copy = type(opt)(model_copy.parameters(), **opt.defaults)

    losses = []
    for epoch in range(50):
        pred = model_copy(X)
        loss = nn.MSELoss()(pred, y)
        opt_copy.zero_grad()
        loss.backward()
        opt_copy.step()
        losses.append(loss.item())

    print(f"{name:15s} 最终损失={losses[-1]:.4f}")


Adam 优化器
Adam（Adaptive Moment Estimation）结合了动量和自适应学习率，是目前最广泛使用的优化器。


mt=β1mt−1+(1−β1)∇L(一阶矩)m_t = \beta_1 m_{t-1} + (1-\beta_1)\nabla L \quad \text{(一阶矩)}mt​=β1​mt−1​+(1−β1​)∇L(一阶矩)
 


vt=β2vt−1+(1−β2)(∇L)2(二阶矩)v_t = \beta_2 v_{t-1} + (1-\beta_2)(\nabla L)^2 \quad \text{(二阶矩)}vt​=β2​vt−1​+(1−β2​)(∇L)2(二阶矩)
 


m^t=mt1−β1t,v^t=vt1−β2t(偏差修正)\hat{m}_t = \frac{m_t}{1-\beta_1^t}, \quad \hat{v}_t = \frac{v_t}{1-\beta_2^t} \quad \text{(偏差修正)}m^t​=1−β1t​mt​​,v^t​=1−β2t​vt​​(偏差修正)
 


θt+1=θt−αv^t+ϵm^t\theta_{t+1} = \theta_t - \frac{\alpha}{\sqrt{\hat{v}_t} + \epsilon}\hat{m}_tθt+1​=θt​−v^t​​+ϵα​m^t​
import torch
import torch.nn as nn
import torch.optim as optim

torch.manual_seed(42)
X = torch.randn(200, 20)
y = torch.randn(200, 1)

optimizers_config = {
    "SGD":      (optim.SGD, {"lr": 0.01}),
    "Momentum": (optim.SGD, {"lr": 0.01, "momentum": 0.9}),
    "Adam":     (optim.Adam, {"lr": 0.001}),
    "AdamW":    (optim.AdamW, {"lr": 0.001, "weight_decay": 0.01}),
}

for name, (OptClass, kwargs) in optimizers_config.items():
    model = nn.Sequential(nn.Linear(20, 50), nn.ReLU(), nn.Linear(50, 1))
    opt = OptClass(model.parameters(), **kwargs)

    for epoch in range(100):
        pred = model(X)
        loss = nn.MSELoss()(pred, y)
        opt.zero_grad()
        loss.backward()
        opt.step()

    print(f"{name:12s} 最终损失: {loss.item():.4f}")

学习率调度
学习率在训练过程中通常不应固定不变。学习率调度（Learning Rate Scheduling）可以在训练初期使用较大学习率快速收敛，后期降低学习率精细调整。


import torch
import torch.optim as optim
from torch.optim.lr_scheduler import StepLR, CosineAnnealingLR, OneCycleLR

model = torch.nn.Linear(10, 1)
optimizer = optim.Adam(model.parameters(), lr=0.01)

schedulers = {
    "StepLR":    StepLR(optimizer, step_size=10, gamma=0.5),
    "CosineAnnealing": CosineAnnealingLR(optimizer, T_max=50),
}

for name, scheduler in schedulers.items():
    optimizer.param_groups[0]['lr'] = 0.01
    lrs = []
    for epoch in range(50):
        lrs.append(optimizer.param_groups[0]['lr'])
        scheduler.step()
    print(f"{name:20s} 初始lr={lrs[0]:.4f} → 最终lr={lrs[-1]:.6f}")

提示： 在实际训练中，Adam 是最安全的默认选择，但在 CV 任务中 SGD+Momentum+余弦退火的组合通常能达到更好的最终性能。选择优化器后，学习率是最重要的超参数。






小结
    

  • 梯度下降是优化的基础，Mini-batch SGD 是实践中的标准形式。
    • 

  • 动量法通过累积梯度加速收敛，减少振荡。
    • 

  • Adam 结合动量和自适应学习率，是最通用的优化器。
    • 

  • 学习率调度在训练过程中动态调整学习率，帮助模型更好地收敛。
    • 

  • 优化器的选择和学习率设置对训练效果有重大影响，需要根据任务特点进行调优。
