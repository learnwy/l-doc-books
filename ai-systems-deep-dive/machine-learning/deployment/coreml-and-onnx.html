CoreML 与 ONNX

跨平台模型推理

在实际部署中，模型往往需要在不同的平台和设备上运行。CoreML 和 ONNX 是两种主流的跨平台模型格式，分别针对 Apple 生态和通用推理场景。



ONNX 格式详解
ONNX（Open Neural Network Exchange）定义了一套通用的模型表示格式和算子集，使得模型可以在不同框架和推理引擎之间无缝转换。


PyTorch → ONNX
import torch
import torch.nn as nn

class ImageClassifier(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 16, 3, padding=1),
            nn.ReLU(),
            nn.AdaptiveAvgPool2d(1),
        )
        self.classifier = nn.Linear(16, num_classes)

    def forward(self, x):
        x = self.features(x)
        x = x.view(x.size(0), -1)
        return self.classifier(x)

model = ImageClassifier()
model.eval()

dummy = torch.randn(1, 3, 224, 224)
torch.onnx.export(
    model, dummy, '/tmp/classifier.onnx',
    input_names=['image'],
    output_names=['logits'],
    dynamic_axes={
        'image': {0: 'batch_size'},
        'logits': {0: 'batch_size'},
    },
    opset_version=13,
)
print("ONNX 导出完成")
ONNX Runtime 推理
import numpy as np

def onnx_inference_example():
    try:
        import onnxruntime as ort

        session = ort.InferenceSession('/tmp/classifier.onnx')

        input_info = session.get_inputs()[0]
        output_info = session.get_outputs()[0]
        print(f"输入: name={input_info.name}, shape={input_info.shape}, type={input_info.type}")
        print(f"输出: name={output_info.name}, shape={output_info.shape}")

        image = np.random.randn(1, 3, 224, 224).astype(np.float32)
        results = session.run(None, {'image': image})
        print(f"推理结果形状: {results[0].shape}")
    except ImportError:
        print("需要安装 onnxruntime: pip install onnxruntime")

onnx_inference_example()

ONNX 模型优化
导出的 ONNX 模型可以进行多种优化以提升推理性能。


import numpy as np

optimization_techniques = {
    "算子融合": "合并连续的 Conv+BN+ReLU 为一个算子",
    "常量折叠": "预计算常量表达式",
    "冗余节点消除": "移除不影响输出的节点",
    "形状推断": "静态推断中间张量形状",
    "量化": "FP32 → INT8/FP16 降低精度",
}

print("ONNX 优化技术:")
for name, desc in optimization_techniques.items():
    print(f"  {name:15s}: {desc}")

print("\n量化类型对比:")
quant_comparison = [
    ("FP32", "32-bit 浮点", "1.0x", "基准精度"),
    ("FP16", "16-bit 浮点", "~2x", "精度损失极小"),
    ("INT8", "8-bit 整数", "~4x", "需要校准数据"),
    ("INT4", "4-bit 整数", "~8x", "精度下降明显"),
]
print(f"{'格式':8s} {'说明':15s} {'速度':8s} {'精度':15s}")
for fmt, desc, speed, acc in quant_comparison:
    print(f"{fmt:8s} {desc:15s} {speed:8s} {acc:15s}")


CoreML 转换
CoreML 是 Apple 的机器学习框架，专为 iOS、macOS、watchOS 和 tvOS 优化。模型可以利用 Neural Engine、GPU 和 CPU 进行硬件加速推理。


def coreml_conversion_example():
    try:
        import coremltools as ct
        import torch
        import torch.nn as nn

        class MobileNet(nn.Module):
            def __init__(self):
                super().__init__()
                self.features = nn.Sequential(
                    nn.Conv2d(3, 16, 3, padding=1),
                    nn.ReLU(),
                    nn.AdaptiveAvgPool2d(1),
                )
                self.classifier = nn.Linear(16, 10)

            def forward(self, x):
                x = self.features(x)
                x = x.view(x.size(0), -1)
                return self.classifier(x)

        model = MobileNet().eval()
        traced = torch.jit.trace(model, torch.randn(1, 3, 224, 224))

        mlmodel = ct.convert(
            traced,
            inputs=[ct.ImageType(name="image", shape=(1, 3, 224, 224))],
            convert_to="mlprogram",
        )

        mlmodel.save('/tmp/classifier.mlpackage')
        print("CoreML 模型已保存")
    except ImportError:
        print("需要安装 coremltools: pip install coremltools")

coreml_conversion_example()

提示： 如果目标是 Apple 设备，CoreML 是最优选择（充分利用 Neural Engine）。如果需要跨平台部署，ONNX + ONNX Runtime 是最通用的方案。在 NVIDIA GPU 上，TensorRT 提供最极致的推理性能。






小结
    

  • ONNX 是跨框架模型交换的开放标准，配合 ONNX Runtime 实现高效推理。
    • 

  • ONNX 模型可通过算子融合、量化等技术进一步优化性能。
    • 

  • CoreML 是 Apple 生态的首选，能充分利用 Neural Engine 实现硬件加速。
    • 

  • 从 PyTorch 到 CoreML 的转换路径：PyTorch → TorchScript → coremltools → .mlpackage。
    • 

  • 部署平台的选择取决于目标硬件：Apple 用 CoreML，跨平台用 ONNX，NVIDIA 用 TensorRT。
