降维

为什么需要降维

降维（Dimensionality Reduction）将高维数据映射到低维空间，同时尽量保留数据的关键信息。在机器学习中，降维可以缓解维度灾难、加速训练、辅助可视化，并在一定程度上减少过拟合。

维度灾难（Curse of Dimensionality）指随着特征维度增加，样本在高维空间中变得极其稀疏，距离度量失效，模型难以有效学习。




PCA（主成分分析）
PCA（Principal Component Analysis）是最经典的线性降维方法。它寻找数据方差最大的方向（主成分），将数据投影到这些方向上。


import numpy as np
from sklearn.decomposition import PCA
from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression

digits = load_digits()
X, y = digits.data, digits.target

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

lr_full = LogisticRegression(max_iter=5000)
lr_full.fit(X_train, y_train)
print(f"原始维度 ({X.shape[1]}): 准确率={lr_full.score(X_test, y_test):.4f}")

for n in [10, 20, 30]:
    pca = PCA(n_components=n)
    X_train_pca = pca.fit_transform(X_train)
    X_test_pca = pca.transform(X_test)

    lr_pca = LogisticRegression(max_iter=5000)
    lr_pca.fit(X_train_pca, y_train)

    var_ratio = pca.explained_variance_ratio_.sum()
    print(f"PCA 降至 {n:2d} 维: 准确率={lr_pca.score(X_test_pca, y_test):.4f}  "
          f"解释方差={var_ratio:.2%}")

t-SNE
t-SNE（t-distributed Stochastic Neighbor Embedding）是一种非线性降维方法，擅长将高维数据嵌入到 2D 或 3D 空间进行可视化。它保留数据的局部邻域结构。


from sklearn.manifold import TSNE
from sklearn.datasets import load_digits
import numpy as np

digits = load_digits()
X, y = digits.data, digits.target

tsne = TSNE(n_components=2, perplexity=30, random_state=42)
X_2d = tsne.fit_transform(X)

print(f"原始维度: {X.shape[1]} → t-SNE 维度: {X_2d.shape[1]}")

for digit in range(10):
    mask = y == digit
    center = X_2d[mask].mean(axis=0)
    print(f"  数字 {digit}: 中心=({center[0]:7.2f}, {center[1]:7.2f})  样本数={mask.sum()}")

UMAP
UMAP（Uniform Manifold Approximation and Projection）是 t-SNE 的改进版，速度更快，且支持对新数据的变换。


from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn.decomposition import PCA
import numpy as np

digits = load_digits()
X, y = digits.data, digits.target
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

knn = KNeighborsClassifier(n_neighbors=5)
knn.fit(X_train, y_train)
print(f"原始维度 ({X.shape[1]}): KNN 准确率={knn.score(X_test, y_test):.4f}")

pca = PCA(n_components=20)
X_train_pca = pca.fit_transform(X_train)
X_test_pca = pca.transform(X_test)
knn_pca = KNeighborsClassifier(n_neighbors=5).fit(X_train_pca, y_train)
print(f"PCA 20 维:         KNN 准确率={knn_pca.score(X_test_pca, y_test):.4f}")

自编码器降维
自编码器（Autoencoder）是一种基于神经网络的非线性降维方法。它学习将输入压缩到低维瓶颈层，再重构回原始维度。


import torch
import torch.nn as nn
from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

digits = load_digits()
X = StandardScaler().fit_transform(digits.data)
X_train, X_test = train_test_split(X, test_size=0.3, random_state=42)

X_train_t = torch.tensor(X_train, dtype=torch.float32)
X_test_t = torch.tensor(X_test, dtype=torch.float32)

class Autoencoder(nn.Module):
    def __init__(self, input_dim, latent_dim):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, 32),
            nn.ReLU(),
            nn.Linear(32, latent_dim),
        )
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 32),
            nn.ReLU(),
            nn.Linear(32, input_dim),
        )

    def forward(self, x):
        z = self.encoder(x)
        return self.decoder(z), z

ae = Autoencoder(64, 10)
optimizer = torch.optim.Adam(ae.parameters(), lr=0.001)

for epoch in range(200):
    recon, _ = ae(X_train_t)
    loss = nn.MSELoss()(recon, X_train_t)
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

ae.eval()
with torch.no_grad():
    recon_test, z_test = ae(X_test_t)
    test_loss = nn.MSELoss()(recon_test, X_test_t)
    print(f"重构损失: {test_loss.item():.4f}")
    print(f"潜在空间维度: {z_test.shape[1]}")


流形学习
流形学习（Manifold Learning）假设高维数据实际位于一个低维流形上，通过发现这个流形结构来进行降维。

from sklearn.manifold import Isomap, LocallyLinearEmbedding
from sklearn.datasets import load_digits
import numpy as np

digits = load_digits()
X, y = digits.data[:500], digits.target[:500]

methods = {
    "Isomap":  Isomap(n_components=2, n_neighbors=10),
    "LLE":     LocallyLinearEmbedding(n_components=2, n_neighbors=10),
}

for name, method in methods.items():
    X_2d = method.fit_transform(X)
    print(f"{name:10s} 输出形状: {X_2d.shape}")


提示： PCA 应作为降维的第一选择：速度快、可解释性好、可变换新数据。只有当 PCA 效果不足以满足需求时，再考虑非线性方法（t-SNE/UMAP 用于可视化，自编码器用于特征学习）。






小结
    

  • PCA 是线性降维的首选，通过方差最大化提取主成分，速度快且可解释。
    • 

  • t-SNE 擅长高维数据可视化，但不支持变换新数据且速度较慢。
    • 

  • UMAP 在 t-SNE 基础上改进了速度和新数据支持，是可视化的更好选择。
    • 

  • 自编码器利用神经网络实现非线性降维，可以学习更复杂的特征表示。
    • 

  • 流形学习假设数据位于低维流形上，通过发现流形结构进行降维。
