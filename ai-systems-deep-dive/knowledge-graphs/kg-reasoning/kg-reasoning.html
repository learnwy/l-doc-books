知识图谱推理

核心概念

知识图谱推理（KG Reasoning）是基于已有知识推导出新知识的过程。由于知识图谱的不完整性（实际存在的三元组远多于已记录的三元组），推理能力是知识图谱从简单存储升级为智能知识系统的关键。推理方法从符号逻辑推理演进到基于嵌入的统计推理，再到结合两者优势的神经符号推理。

基于规则的推理
基于规则的推理利用逻辑规则从已有事实推导出新事实。Horn 规则是最常用的规则形式，形如 "如果 A 且 B，则 C"。自动规则挖掘方法（如 AMIE）从知识图谱中自动学习高置信度的规则。


interface HornRule {
  head: RuleAtom;
  body: RuleAtom[];
  confidence: number;
  support: number;
  headCoverage: number;
}

interface RuleAtom {
  predicate: string;
  subject: string;
  object: string;
}

function applyRule(rule: HornRule, facts: Set<string>): Triple[] {
  const newFacts: Triple[] = [];
  const bindings = findBindings(rule.body, facts);

  for (const binding of bindings) {
    const headSubject = applyBinding(rule.head.subject, binding);
    const headObject = applyBinding(rule.head.object, binding);
    const newTriple = `${headSubject},${rule.head.predicate},${headObject}`;

    if (!facts.has(newTriple)) {
      newFacts.push({
        subject: headSubject,
        predicate: rule.head.predicate,
        object: headObject,
      });
    }
  }

  return newFacts;
}

interface AMIEConfig {
  minSupport: number;
  minConfidence: number;
  maxRuleLength: number;
  topK: number;
}

function mineRules(kg: KnowledgeGraph, config: AMIEConfig): HornRule[] {
  const rules: HornRule[] = [];
  const relations = kg.getRelations();

  for (const headRel of relations) {
    const candidates = generateRuleCandidates(headRel, relations, config.maxRuleLength);

    for (const candidate of candidates) {
      const support = countSupport(candidate, kg);
      const headCoverage = support / kg.getRelationCount(headRel);

      if (support >= config.minSupport) {
        const confidence = computePCAConfidence(candidate, kg);
        if (confidence >= config.minConfidence) {
          rules.push({ ...candidate, confidence, support, headCoverage });
        }
      }
    }
  }

  return rules.sort((a, b) => b.confidence - a.confidence).slice(0, config.topK);
}
AMIE（Association Rule Mining under Incomplete Evidence）是知识图谱规则挖掘的标准工具。它的 PCA（Partial Completeness Assumption）置信度度量考虑了知识图谱的不完整性，避免了将未观测到的三元组简单地视为负例。


链接预测与图谱补全
链接预测（Link Prediction）预测知识图谱中缺失的三元组，是知识图谱推理中最活跃的研究方向。给定头实体和关系 (h, r, ?)，预测尾实体；或给定尾实体和关系 (?, r, t)，预测头实体。


interface LinkPredictor {
  predict(head: string, relation: string, topK: number): PredictionResult[];
  predictHead(tail: string, relation: string, topK: number): PredictionResult[];
  score(head: string, relation: string, tail: string): number;
}

interface PredictionResult {
  entity: string;
  score: number;
  rank: number;
}

function evaluateLinkPrediction(
  predictor: LinkPredictor,
  testTriples: Triple[],
  allEntities: string[],
): LinkPredictionMetrics {
  let mrr = 0;
  let hits1 = 0;
  let hits3 = 0;
  let hits10 = 0;

  for (const triple of testTriples) {
    const predictions = predictor.predict(triple.subject, triple.predicate, allEntities.length);
    const rank = predictions.findIndex((p) => p.entity === triple.object) + 1;

    if (rank > 0) {
      mrr += 1 / rank;
      if (rank <= 1) hits1++;
      if (rank <= 3) hits3++;
      if (rank <= 10) hits10++;
    }
  }

  const n = testTriples.length;
  return {
    mrr: mrr / n,
    hits1: hits1 / n,
    hits3: hits3 / n,
    hits10: hits10 / n,
  };
}

图神经网络推理
图神经网络（GNN）通过消息传递机制聚合邻居节点的信息来学习节点表示，天然适合知识图谱的图结构数据。R-GCN（Relational Graph Convolutional Network）是知识图谱上 GNN 的开创性工作。


interface RGCNLayer {
  relationWeights: Map<string, Float32Array[]>;
  selfWeight: Float32Array[];
  bias: Float32Array;
  numBases?: number;
}

function rgcnForward(
  nodeFeatures: Map<string, Float32Array>,
  edges: Triple[],
  layer: RGCNLayer,
): Map<string, Float32Array> {
  const newFeatures = new Map<string, Float32Array>();
  const neighborMessages = new Map<string, Float32Array[]>();

  for (const edge of edges) {
    const weight = layer.relationWeights.get(edge.predicate);
    if (!weight) continue;

    const sourceFeature = nodeFeatures.get(edge.subject);
    if (!sourceFeature) continue;

    const message = matVecMul(weight, sourceFeature);
    const existing = neighborMessages.get(edge.object) ?? [];
    existing.push(message);
    neighborMessages.set(edge.object, existing);
  }

  for (const [nodeId, features] of nodeFeatures) {
    const selfTransform = matVecMul(layer.selfWeight, features);
    const messages = neighborMessages.get(nodeId) ?? [];
    const aggregated = meanAggregate(messages);

    const combined = addVectors(selfTransform, aggregated, layer.bias);
    newFeatures.set(nodeId, relu(combined));
  }

  return newFeatures;
}
CompGCN 将关系也作为可学习的嵌入，在消息传递过程中同时更新实体和关系的表示。这种设计使得模型能够捕获实体-关系之间的复杂交互模式。


小结
    

  • 知识图谱推理通过已有知识推导新知识，是应对图谱不完整性的核心技术
    • 

  • 基于规则的推理（AMIE）可解释性强，适合发现领域知识中的逻辑模式
    • 

  • 链接预测（TransE、RotatE）通过向量空间中的几何运算预测缺失三元组，高效可扩展
    • 

  • 图神经网络（R-GCN、CompGCN）通过消息传递聚合图结构信息，适合多跳推理
    • 

  • 神经符号方法结合逻辑推理的可解释性和神经网络的泛化能力，是推理研究的前沿方向
    •
