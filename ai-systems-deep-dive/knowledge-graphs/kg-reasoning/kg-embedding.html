知识图谱嵌入

核心概念

知识图谱嵌入（Knowledge Graph Embedding, KGE）将知识图谱中的实体和关系映射到低维连续向量空间，通过定义评分函数来衡量三元组的合理性。KGE 是链接预测、实体分类和关系预测的基础技术，也是知识图谱与深度学习结合的桥梁。

平移与旋转模型
平移模型将关系建模为实体空间中的平移操作。TransE 的核心思想是：对于一个正确的三元组 (h, r, t)，头实体向量加上关系向量应该近似等于尾实体向量。


interface TransEModel {
  entityEmbeddings: Map<string, Float32Array>;
  relationEmbeddings: Map<string, Float32Array>;
  dimension: number;
  margin: number;
  norm: 1 | 2;
}

function transEScoreFn(head: Float32Array, relation: Float32Array, tail: Float32Array, norm: 1 | 2): number {
  let distance = 0;
  for (let i = 0; i < head.length; i++) {
    const diff = head[i] + relation[i] - tail[i];
    distance += norm === 1 ? Math.abs(diff) : diff * diff;
  }
  return norm === 2 ? Math.sqrt(distance) : distance;
}

function rotatEScoreFn(head: Float32Array, relation: Float32Array, tail: Float32Array): number {
  const halfDim = head.length / 2;
  let distance = 0;

  for (let i = 0; i < halfDim; i++) {
    const hRe = head[i];
    const hIm = head[i + halfDim];
    const rRe = Math.cos(relation[i]);
    const rIm = Math.sin(relation[i]);
    const tRe = tail[i];
    const tIm = tail[i + halfDim];

    const rotRe = hRe * rRe - hIm * rIm;
    const rotIm = hRe * rIm + hIm * rRe;
    distance += (rotRe - tRe) ** 2 + (rotIm - tIm) ** 2;
  }

  return -Math.sqrt(distance);
}

function marginRankingLoss(posScore: number, negScore: number, margin: number): number {
  return Math.max(0, margin + posScore - negScore);
}
RotatE 将关系建模为复数空间中的旋转操作，天然能够建模对称、反对称、组合和逆关系等多种关系模式。HAKE（Hierarchy-Aware Knowledge Graph Embedding）进一步引入极坐标表示，用模长建模语义层次，用相位建模同层区分。


双线性与张量分解模型
双线性模型通过关系特定的矩阵捕获头尾实体之间的交互。DistMult 使用对角矩阵，ComplEx 将 DistMult 扩展到复数空间以建模非对称关系。


interface DistMultModel {
  entityEmbeddings: Map<string, Float32Array>;
  relationEmbeddings: Map<string, Float32Array>;
  dimension: number;
}

function distMultScore(head: Float32Array, relation: Float32Array, tail: Float32Array): number {
  let score = 0;
  for (let i = 0; i < head.length; i++) {
    score += head[i] * relation[i] * tail[i];
  }
  return score;
}

function complExScore(
  headRe: Float32Array,
  headIm: Float32Array,
  relRe: Float32Array,
  relIm: Float32Array,
  tailRe: Float32Array,
  tailIm: Float32Array,
): number {
  let score = 0;
  for (let i = 0; i < headRe.length; i++) {
    score += headRe[i] * relRe[i] * tailRe[i];
    score += headIm[i] * relRe[i] * tailIm[i];
    score += headRe[i] * relIm[i] * tailIm[i];
    score -= headIm[i] * relIm[i] * tailRe[i];
  }
  return score;
}

interface TuckERModel {
  entityEmbeddings: Float32Array[];
  relationEmbeddings: Float32Array[];
  coreTensor: Float32Array[][];
  entityDim: number;
  relationDim: number;
}
TuckER 基于 Tucker 张量分解，将三元组评分函数建模为核心张量与实体/关系向量的多模态乘积。它的表达能力最强，理论上可以表示所有双线性模型作为其特例。


训练策略与负采样
KGE 的训练核心是负采样策略。由于知识图谱只包含正例三元组（真实存在的事实），需要通过替换头实体或尾实体来生成负例。

interface NegativeSampler {
  sampleNegatives(triple: Triple, k: number): Triple[];
}

function uniformNegativeSampling(
  triple: Triple,
  entities: string[],
  k: number,
  corruprHead: boolean = Math.random() > 0.5,
): Triple[] {
  const negatives: Triple[] = [];
  const used = new Set<string>();

  while (negatives.length < k) {
    const randEntity = entities[Math.floor(Math.random() * entities.length)];
    const key = corruprHead ? `${randEntity}-${triple.object}` : `${triple.subject}-${randEntity}`;
    if (used.has(key)) continue;
    used.add(key);

    negatives.push({
      subject: corruprHead ? randEntity : triple.subject,
      predicate: triple.predicate,
      object: corruprHead ? triple.object : randEntity,
    });
  }

  return negatives;
}

function selfAdversarialSampling(
  triple: Triple,
  entities: string[],
  model: KGEModel,
  k: number,
  temperature: number,
): Triple[] {
  const candidates = uniformNegativeSampling(triple, entities, k * 3);
  const scores = candidates.map((c) => model.score(c.subject, c.predicate, c.object));
  const weights = softmax(scores.map((s) => s / temperature));

  const selected: Triple[] = [];
  for (let i = 0; i < k; i++) {
    const idx = weightedRandomChoice(weights);
    selected.push(candidates[idx]);
  }

  return selected;
}
小结
    

  • KGE 将实体和关系映射到连续向量空间，通过评分函数衡量三元组合理性
    • 

  • 平移模型（TransE）简单高效，旋转模型（RotatE）能建模多种关系模式
    • 

  • 双线性模型（DistMult、ComplEx）捕获实体间的二阶交互，张量分解（TuckER）表达能力最强
    • 

  • 负采样质量对训练效果至关重要，自对抗采样在大多数场景下效果最优
    • 

  • KGE 的应用包括链接预测、实体分类、关系预测以及作为下游任务（搜索、推荐）的特征输入
    •
