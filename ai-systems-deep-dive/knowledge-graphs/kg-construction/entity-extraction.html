实体抽取

核心概念

实体抽取（Entity Extraction）也称命名实体识别（Named Entity Recognition, NER），是从非结构化文本中自动识别和分类命名实体的任务。它是知识图谱自动构建的第一步，为后续的关系抽取和知识融合提供基础输入。

序列标注方法
序列标注（Sequence Labeling）是实体抽取的核心范式。它将文本中的每个 token 标注为 BIO/BIOES 标签序列，其中 B 表示实体开始，I 表示实体内部，O 表示非实体。


type BIOTag = 'B' | 'I' | 'O';
type BIOESTag = 'B' | 'I' | 'O' | 'E' | 'S';

interface NERLabel {
  tag: BIOTag | BIOESTag;
  entityType: string;
}

interface NERResult {
  entities: ExtractedEntity[];
  tokens: string[];
  labels: NERLabel[];
}

interface ExtractedEntity {
  text: string;
  type: string;
  start: number;
  end: number;
  confidence: number;
}

function decodeBIO(tokens: string[], labels: NERLabel[]): ExtractedEntity[] {
  const entities: ExtractedEntity[] = [];
  let current: Partial<ExtractedEntity> | null = null;

  for (let i = 0; i < tokens.length; i++) {
    const { tag, entityType } = labels[i];

    if (tag === 'B') {
      if (current) entities.push(current as ExtractedEntity);
      current = { text: tokens[i], type: entityType, start: i, end: i + 1, confidence: 0 };
    } else if (tag === 'I' && current && current.type === entityType) {
      current.text += tokens[i];
      current.end = i + 1;
    } else {
      if (current) entities.push(current as ExtractedEntity);
      current = null;
    }
  }

  if (current) entities.push(current as ExtractedEntity);
  return entities;
}

interface CRFLayer {
  transitionMatrix: Float32Array[];
  startTransitions: Float32Array;
  endTransitions: Float32Array;
}

function viterbiDecode(emissions: Float32Array[], crf: CRFLayer): number[] {
  const seqLen = emissions.length;
  const numTags = emissions[0].length;
  const viterbi: Float32Array[] = [];
  const backpointers: number[][] = [];

  viterbi.push(new Float32Array(numTags));
  for (let t = 0; t < numTags; t++) {
    viterbi[0][t] = crf.startTransitions[t] + emissions[0][t];
  }

  for (let i = 1; i < seqLen; i++) {
    const scores = new Float32Array(numTags);
    const bp = new Array(numTags);
    for (let t = 0; t < numTags; t++) {
      let bestScore = -Infinity;
      let bestPrev = 0;
      for (let p = 0; p < numTags; p++) {
        const score = viterbi[i - 1][p] + crf.transitionMatrix[p][t] + emissions[i][t];
        if (score > bestScore) {
          bestScore = score;
          bestPrev = p;
        }
      }
      scores[t] = bestScore;
      bp[t] = bestPrev;
    }
    viterbi.push(scores);
    backpointers.push(bp);
  }

  const path: number[] = [0];
  let bestLast = -Infinity;
  for (let t = 0; t < numTags; t++) {
    const score = viterbi[seqLen - 1][t] + crf.endTransitions[t];
    if (score > bestLast) {
      bestLast = score;
      path[0] = t;
    }
  }

  for (let i = backpointers.length - 1; i >= 0; i--) {
    path.unshift(backpointers[i][path[0]]);
  }

  return path;
}
BERT-CRF 是当前最主流的 NER 架构。BERT 作为编码器提取上下文化的 token 表示，CRF 层建模标签之间的转移约束（如 I-PER 不能出现在 B-LOC 之后），通过 Viterbi 算法解码出全局最优的标签序列。


基于 Span 的抽取方法
Span-based 方法将实体抽取建模为枚举和分类文本片段（Span）的问题，直接预测每个可能 span 的实体类型，避免了 BIO 序列标注中的错误传播问题。


interface SpanCandidate {
  start: number;
  end: number;
  text: string;
  score: number;
  type: string;
}

interface SpanConfig {
  maxSpanLength: number;
  negativeRatio: number;
  threshold: number;
}

function generateSpanCandidates(tokens: string[], maxLength: number): Array<[number, number]> {
  const spans: Array<[number, number]> = [];
  for (let i = 0; i < tokens.length; i++) {
    for (let j = i + 1; j <= Math.min(i + maxLength, tokens.length); j++) {
      spans.push([i, j]);
    }
  }
  return spans;
}

function spanClassification(
  tokenEmbeddings: Float32Array[],
  spans: Array<[number, number]>,
  classifier: SpanClassifier,
): SpanCandidate[] {
  const results: SpanCandidate[] = [];

  for (const [start, end] of spans) {
    const spanRepr = computeSpanRepresentation(tokenEmbeddings, start, end);
    const { type, score } = classifier.predict(spanRepr);
    if (type !== 'O' && score > classifier.threshold) {
      results.push({ start, end, text: '', score, type });
    }
  }

  return resolveOverlappingSpans(results);
}

少样本与零样本实体抽取
在领域知识图谱构建中，标注数据通常稀缺。少样本和零样本实体抽取方法能够在极少甚至没有标注数据的情况下进行实体识别。


基于提示学习（Prompt-based）的方法将 NER 转化为完形填空或文本生成任务，利用预训练语言模型的知识进行实体抽取。例如，将 "张三在北京工作" 转化为 "张三是一个[MASK]" 的完形填空问题，如果模型预测 [MASK] 为 "人名"，则 "张三" 被标注为 PER 实体。


LLM 驱动的实体抽取通过精心设计的 prompt 指示大语言模型从文本中提取指定类型的实体。这种方法无需训练，但需要控制输出格式和处理幻觉问题。


小结
    

  • 实体抽取是知识图谱构建的基础任务，将非结构化文本中的实体识别并分类
    • 

  • BERT-CRF 是序列标注方法的主流架构，CRF 层建模标签转移约束确保全局最优
    • 

  • Span-based 方法天然支持嵌套实体识别，但计算复杂度为 O(n²)
    • 

  • 少样本/零样本方法（提示学习、LLM 抽取）适合标注数据稀缺的领域场景
    • 

  • 实际系统通常组合规则抽取（高精度核心实体）和模型抽取（长尾实体）以平衡精度和召回
    •
