风险度量指标

核心概念

风险度量指标是评估投资策略和资产组合表现的量化工具。仅看收益率是远远不够的——年化收益 20% 的策略可能隐含巨大的尾部风险，而年化收益 8% 的策略可能在任何市场环境下都表现稳健。风险调整后收益指标帮助我们在"赚多少"和"承担多少风险"之间做出公正的比较。

对程序员而言，风险度量指标是构建量化回测系统和投资绩效评估平台的核心模块，每一个指标都对应着一段精确的计算逻辑。

夏普比率与信息比率
夏普比率（Sharpe Ratio）是最广泛使用的风险调整收益指标，衡量每承受一单位总风险所获得的超额收益：


SR=E(Rp)−RfσpSR = \frac{E(R_p) - R_f}{\sigma_p}SR=σp​E(Rp​)−Rf​​
其中 RpR_pRp​ 是组合收益率，RfR_fRf​ 是无风险利率，σp\sigma_pσp​ 是组合收益率的标准差。


信息比率（Information Ratio）则衡量主动管理的效率——每承受一单位跟踪误差所获得的超额收益：


IR=E(Rp)−E(Rb)σp−bIR = \frac{E(R_p) - E(R_b)}{\sigma_{p-b}}IR=σp−b​E(Rp​)−E(Rb​)​
interface PerformanceMetrics {
  annualizedReturn: number;
  annualizedVolatility: number;
  sharpeRatio: number;
  informationRatio: number;
  trackingError: number;
}

const calculateSharpe = (returns: number[], riskFreeRate: number = 0.03): number => {
  const n = returns.length;
  const meanReturn = returns.reduce((s, r) => s + r, 0) / n;
  const annualizedReturn = meanReturn * 252;
  const variance = returns.reduce((s, r) => s + (r - meanReturn) ** 2, 0) / (n - 1);
  const annualizedVol = Math.sqrt(variance * 252);
  return (annualizedReturn - riskFreeRate) / annualizedVol;
};

const calculateInformationRatio = (
  portfolioReturns: number[],
  benchmarkReturns: number[],
): { ir: number; trackingError: number } => {
  const activeReturns = portfolioReturns.map((r, i) => r - benchmarkReturns[i]);
  const n = activeReturns.length;
  const meanActive = activeReturns.reduce((s, r) => s + r, 0) / n;
  const variance = activeReturns.reduce((s, r) => s + (r - meanActive) ** 2, 0) / (n - 1);
  const trackingError = Math.sqrt(variance * 252);
  const annualizedActive = meanActive * 252;

  return {
    ir: annualizedActive / trackingError,
    trackingError,
  };
};

const portfolioReturns = Array.from({ length: 252 }, () => (Math.random() - 0.48) * 0.02);
const benchmarkReturns = Array.from({ length: 252 }, () => (Math.random() - 0.49) * 0.015);

const sharpe = calculateSharpe(portfolioReturns);
const { ir, trackingError } = calculateInformationRatio(portfolioReturns, benchmarkReturns);

console.log(`夏普比率: ${sharpe.toFixed(3)}`);
console.log(`信息比率: ${ir.toFixed(3)}`);
console.log(`跟踪误差: ${(trackingError * 100).toFixed(2)}%`);

最大回撤与回撤分析
最大回撤（Maximum Drawdown）衡量从历史最高点到最低点的最大跌幅，是投资者最直观感受到的风险指标。一个策略的夏普比率可能很高，但如果最大回撤达到 50%，大多数投资者都无法坚持持有。


interface DrawdownAnalysis {
  maxDrawdown: number;
  maxDrawdownStart: number;
  maxDrawdownEnd: number;
  avgDrawdown: number;
  drawdownDuration: number;
  calmarRatio: number;
}

const analyzeDrawdowns = (returns: number[], annualizedReturn: number): DrawdownAnalysis => {
  const cumReturns: number[] = [1];
  for (const r of returns) {
    cumReturns.push(cumReturns[cumReturns.length - 1] * (1 + r));
  }

  let peak = cumReturns[0];
  let maxDD = 0;
  let ddStart = 0;
  let ddEnd = 0;
  let currentStart = 0;
  const drawdowns: number[] = [];

  for (let i = 1; i < cumReturns.length; i++) {
    if (cumReturns[i] > peak) {
      peak = cumReturns[i];
      currentStart = i;
    }
    const dd = (peak - cumReturns[i]) / peak;
    drawdowns.push(dd);

    if (dd > maxDD) {
      maxDD = dd;
      ddStart = currentStart;
      ddEnd = i;
    }
  }

  const avgDD =
    drawdowns.filter((d) => d > 0).reduce((s, d) => s + d, 0) / Math.max(drawdowns.filter((d) => d > 0).length, 1);

  return {
    maxDrawdown: maxDD,
    maxDrawdownStart: ddStart,
    maxDrawdownEnd: ddEnd,
    avgDrawdown: avgDD,
    drawdownDuration: ddEnd - ddStart,
    calmarRatio: annualizedReturn / Math.max(maxDD, 0.001),
  };
};

const dailyReturns = Array.from({ length: 504 }, () => (Math.random() - 0.48) * 0.02);
const annReturn = (dailyReturns.reduce((s, r) => s + r, 0) / dailyReturns.length) * 252;
const dd = analyzeDrawdowns(dailyReturns, annReturn);

console.log(`最大回撤: ${(dd.maxDrawdown * 100).toFixed(2)}%`);
console.log(`平均回撤: ${(dd.avgDrawdown * 100).toFixed(2)}%`);
console.log(`回撤持续: ${dd.drawdownDuration} 天`);
console.log(`卡尔玛比率: ${dd.calmarRatio.toFixed(3)}`);
下行风险与索提诺比率
夏普比率的一个缺陷是：它将上行波动和下行波动同等对待。但投资者真正关心的是下行风险——亏损的可能性和幅度。索提诺比率（Sortino Ratio）通过只计算下行偏差来修正这一问题。


interface DownsideMetrics {
  downsideDeviation: number;
  sortinoRatio: number;
  omegaRatio: number;
  upsideCapture: number;
  downsideCapture: number;
}

const calculateDownsideMetrics = (
  returns: number[],
  benchmarkReturns: number[],
  riskFreeRate: number = 0.03,
  mar: number = 0,
): DownsideMetrics => {
  const dailyMAR = mar / 252;
  const dailyRf = riskFreeRate / 252;

  const downsideReturns = returns.filter((r) => r < dailyMAR);
  const downsideSum = downsideReturns.reduce((s, r) => s + (r - dailyMAR) ** 2, 0);
  const downsideDeviation = Math.sqrt((downsideSum / returns.length) * 252);

  const meanReturn = returns.reduce((s, r) => s + r, 0) / returns.length;
  const annualizedReturn = meanReturn * 252;
  const sortinoRatio = (annualizedReturn - riskFreeRate) / downsideDeviation;

  const gains = returns.filter((r) => r > dailyMAR).reduce((s, r) => s + (r - dailyMAR), 0);
  const losses = Math.abs(returns.filter((r) => r < dailyMAR).reduce((s, r) => s + (r - dailyMAR), 0));
  const omegaRatio = losses > 0 ? gains / losses : Infinity;

  const upMonths = returns.filter((_, i) => benchmarkReturns[i] > 0);
  const upBench = benchmarkReturns.filter((r) => r > 0);
  const upsideCapture =
    upMonths.length > 0
      ? upMonths.reduce((s, r) => s + r, 0) / upMonths.length / (upBench.reduce((s, r) => s + r, 0) / upBench.length)
      : 0;

  const downMonths = returns.filter((_, i) => benchmarkReturns[i] < 0);
  const downBench = benchmarkReturns.filter((r) => r < 0);
  const downsideCapture =
    downMonths.length > 0
      ? downMonths.reduce((s, r) => s + r, 0) /
        downMonths.length /
        (downBench.reduce((s, r) => s + r, 0) / downBench.length)
      : 0;

  return { downsideDeviation, sortinoRatio, omegaRatio, upsideCapture, downsideCapture };
};

const pReturns = Array.from({ length: 252 }, () => (Math.random() - 0.47) * 0.02);
const bReturns = Array.from({ length: 252 }, () => (Math.random() - 0.49) * 0.015);
const metrics = calculateDownsideMetrics(pReturns, bReturns);

console.log(`下行偏差: ${(metrics.downsideDeviation * 100).toFixed(2)}%`);
console.log(`索提诺比率: ${metrics.sortinoRatio.toFixed(3)}`);
console.log(`Omega比率: ${metrics.omegaRatio.toFixed(3)}`);

综合绩效评估框架
单一指标无法完整评价一个策略，实践中需要构建多维度的评估框架。一个完善的绩效评估报告通常包含收益、风险、风险调整收益和稳定性四个维度。


interface ComprehensiveReport {
  totalReturn: number;
  annualizedReturn: number;
  volatility: number;
  maxDrawdown: number;
  sharpeRatio: number;
  sortinoRatio: number;
  calmarRatio: number;
  winRate: number;
  profitFactor: number;
  grade: string;
}

const generateReport = (returns: number[], riskFreeRate: number = 0.03): ComprehensiveReport => {
  const n = returns.length;
  const totalReturn = returns.reduce((cum, r) => cum * (1 + r), 1) - 1;
  const years = n / 252;
  const annualizedReturn = (1 + totalReturn) ** (1 / years) - 1;

  const mean = returns.reduce((s, r) => s + r, 0) / n;
  const vol = Math.sqrt((returns.reduce((s, r) => s + (r - mean) ** 2, 0) / (n - 1)) * 252);

  const downsideReturns = returns.filter((r) => r < 0);
  const downsideDev = Math.sqrt((downsideReturns.reduce((s, r) => s + r ** 2, 0) / n) * 252);

  let peak = 1,
    maxDD = 0,
    cumValue = 1;
  for (const r of returns) {
    cumValue *= 1 + r;
    if (cumValue > peak) peak = cumValue;
    const dd = (peak - cumValue) / peak;
    if (dd > maxDD) maxDD = dd;
  }

  const sharpe = (annualizedReturn - riskFreeRate) / vol;
  const sortino = (annualizedReturn - riskFreeRate) / downsideDev;
  const calmar = annualizedReturn / Math.max(maxDD, 0.001);

  const wins = returns.filter((r) => r > 0);
  const losses = returns.filter((r) => r < 0);
  const winRate = wins.length / n;
  const grossProfit = wins.reduce((s, r) => s + r, 0);
  const grossLoss = Math.abs(losses.reduce((s, r) => s + r, 0));
  const profitFactor = grossLoss > 0 ? grossProfit / grossLoss : Infinity;

  let grade = 'D';
  if (sharpe > 1.5 && maxDD < 0.15) grade = 'A';
  else if (sharpe > 1.0 && maxDD < 0.2) grade = 'B';
  else if (sharpe > 0.5 && maxDD < 0.3) grade = 'C';

  return {
    totalReturn,
    annualizedReturn,
    volatility: vol,
    maxDrawdown: maxDD,
    sharpeRatio: sharpe,
    sortinoRatio: sortino,
    calmarRatio: calmar,
    winRate,
    profitFactor,
    grade,
  };
};

const strategyReturns = Array.from({ length: 756 }, () => (Math.random() - 0.47) * 0.018);
const report = generateReport(strategyReturns);

console.log(`=== 策略绩效报告 ===`);
console.log(`累计收益: ${(report.totalReturn * 100).toFixed(1)}%`);
console.log(`年化收益: ${(report.annualizedReturn * 100).toFixed(1)}%`);
console.log(`年化波动: ${(report.volatility * 100).toFixed(1)}%`);
console.log(`最大回撤: ${(report.maxDrawdown * 100).toFixed(1)}%`);
console.log(`夏普比率: ${report.sharpeRatio.toFixed(2)}`);
console.log(`索提诺比率: ${report.sortinoRatio.toFixed(2)}`);
console.log(`卡尔玛比率: ${report.calmarRatio.toFixed(2)}`);
console.log(`胜率: ${(report.winRate * 100).toFixed(1)}%`);
console.log(`盈亏比: ${report.profitFactor.toFixed(2)}`);
console.log(`综合评级: ${report.grade}`);
小结
    

  • 夏普比率是最通用的风险调整收益指标，但将上下行波动等同对待
    • 

  • 最大回撤是投资者体感最强的风险指标，直接影响持有信心
    • 

  • 索提诺比率通过只计算下行偏差修正了夏普比率的缺陷
    • 

  • Omega 比率和上下行捕获率提供了更细粒度的风险收益分析
    • 

  • 实践中应构建多维度评估框架，单一指标无法完整评价策略质量
    • 

  • 所有指标的计算逻辑明确，非常适合程序化实现和自动化回测
