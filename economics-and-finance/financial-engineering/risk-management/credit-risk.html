信用风险

核心概念

信用风险是交易对手无法履行合约义务（违约）而导致损失的风险。在所有金融风险中，信用风险造成的累计损失最大——2008 年金融危机的核心就是信用风险的系统性爆发。对程序员而言，信用风险模型是金融科技中最重要的应用场景之一，从信用评分到贷款审批都依赖信用风险分析。

信用评分模型
信用评分模型是信用风险管理的核心工具，它将借款人的特征转化为违约概率的预测。


interface BorrowerFeatures {
  income: number;
  debtToIncome: number;
  creditHistory: number;
  employmentYears: number;
  existingLoans: number;
  age: number;
}

interface CreditScore {
  score: number;
  pd: number;
  rating: string;
  decision: string;
}

const logisticCreditScore = (features: BorrowerFeatures): CreditScore => {
  const weights = {
    income: -0.00002,
    debtToIncome: 3.5,
    creditHistory: -0.5,
    employmentYears: -0.15,
    existingLoans: 0.8,
    age: -0.02,
  };
  const intercept = 1.5;

  const logit =
    intercept +
    weights.income * features.income +
    weights.debtToIncome * features.debtToIncome +
    weights.creditHistory * features.creditHistory +
    weights.employmentYears * features.employmentYears +
    weights.existingLoans * features.existingLoans +
    weights.age * features.age;

  const pd = 1 / (1 + Math.exp(-logit));
  const score = Math.round(300 + 550 * (1 - pd));

  let rating: string;
  let decision: string;
  if (score >= 750) {
    rating = 'A';
    decision = '批准，优惠利率';
  } else if (score >= 650) {
    rating = 'B';
    decision = '批准，标准利率';
  } else if (score >= 550) {
    rating = 'C';
    decision = '批准，上浮利率';
  } else {
    rating = 'D';
    decision = '拒绝';
  }

  return { score, pd: Math.round(pd * 10000) / 100, rating, decision };
};

const applicants: BorrowerFeatures[] = [
  { income: 300000, debtToIncome: 0.2, creditHistory: 8, employmentYears: 10, existingLoans: 1, age: 35 },
  { income: 150000, debtToIncome: 0.5, creditHistory: 3, employmentYears: 2, existingLoans: 3, age: 25 },
  { income: 80000, debtToIncome: 0.7, creditHistory: 1, employmentYears: 1, existingLoans: 5, age: 22 },
];

applicants.forEach((a, i) => {
  const result = logisticCreditScore(a);
  console.log(`申请人${i + 1}: 评分=${result.score}, PD=${result.pd}%, 评级=${result.rating}, ${result.decision}`);
});
预期损失与经济资本
interface LoanPortfolio {
  loans: Array<{ amount: number; pd: number; lgd: number }>;
}

interface PortfolioRisk {
  totalExposure: number;
  expectedLoss: number;
  expectedLossRate: number;
  unexpectedLoss: number;
  economicCapital: number;
}

const analyzePortfolioCredit = (portfolio: LoanPortfolio): PortfolioRisk => {
  const totalExposure = portfolio.loans.reduce((sum, l) => sum + l.amount, 0);
  const expectedLoss = portfolio.loans.reduce((sum, l) => sum + l.amount * l.pd * l.lgd, 0);

  const varianceLoss = portfolio.loans.reduce((sum, l) => {
    const elLoan = l.amount * l.pd * l.lgd;
    const varLoan = l.amount * l.amount * l.pd * (1 - l.pd) * l.lgd * l.lgd;
    return sum + varLoan;
  }, 0);

  const unexpectedLoss = Math.sqrt(varianceLoss);
  const economicCapital = unexpectedLoss * 2.326;

  return {
    totalExposure: Math.round(totalExposure),
    expectedLoss: Math.round(expectedLoss),
    expectedLossRate: Math.round((expectedLoss / totalExposure) * 10000) / 100,
    unexpectedLoss: Math.round(unexpectedLoss),
    economicCapital: Math.round(economicCapital),
  };
};

const portfolio: LoanPortfolio = {
  loans: [
    { amount: 1000000, pd: 0.01, lgd: 0.4 },
    { amount: 500000, pd: 0.03, lgd: 0.5 },
    { amount: 2000000, pd: 0.005, lgd: 0.35 },
    { amount: 300000, pd: 0.08, lgd: 0.6 },
  ],
};

const risk = analyzePortfolioCredit(portfolio);
console.log(`总敞口: ¥${risk.totalExposure}`);
console.log(`预期损失: ¥${risk.expectedLoss} (${risk.expectedLossRate}%)`);
console.log(`非预期损失: ¥${risk.unexpectedLoss}`);
console.log(`经济资本(99%): ¥${risk.economicCapital}`);
信用衍生品
信用违约互换（CDS）是最重要的信用衍生品，允许投资者在不持有债券的情况下交易信用风险。CDS 价差是市场对违约风险的实时定价，是监测信用状况的重要指标。

小结
    

  • 信用风险三要素（PD、LGD、EAD）是风险量化的基础
    • 

  • 信用评分模型将借款人特征映射为违约概率，是金融科技的核心应用
    • 

  • 预期损失通过定价和拨备覆盖，非预期损失需要经济资本缓冲
    • 

  • 信用衍生品（CDS）实现了信用风险的市场化交易和转移
    • 

  • 信用风险模型的开发是金融科技领域对程序员需求最大的方向之一
