监管合规

核心概念

监管合规是量化交易系统不可或缺的组成部分。无论策略多优秀，违反监管规则都可能导致巨额罚款、交易禁止甚至刑事责任。量化交易面临的监管环境日趋复杂，涵盖市场操纵禁令、交易报告义务、资本充足率要求和投资者保护规定等多个维度。

对程序化交易者而言，合规不仅是法律要求，更是一种系统设计约束。合规规则需要内嵌到交易系统中，通过自动化的预交易检查和实时监控来确保每笔交易都在合规边界内。

预交易合规引擎
预交易合规引擎在订单提交到交易所之前对每笔订单进行合规性检查。它是防止违规交易的第一道也是最重要的一道防线。任何未通过合规检查的订单都将被拦截。


type ComplianceCheckResult = 'pass' | 'block' | 'warn';

interface ComplianceCheck {
  name: string;
  check(order: OrderRequest, context: ComplianceContext): ComplianceCheckResult;
  message: string;
}

interface OrderRequest {
  symbol: string;
  side: 'buy' | 'sell';
  quantity: number;
  price: number;
  orderType: string;
  strategyId: string;
  timestamp: number;
}

interface ComplianceContext {
  positions: Map<string, number>;
  dailyTrades: Map<string, number>;
  dailyVolume: Map<string, number>;
  capitalUtilization: number;
  pendingOrders: OrderRequest[];
}

interface ComplianceLimits {
  maxPositionSize: number;
  maxDailyTrades: number;
  maxDailyVolume: number;
  maxOrderSize: number;
  maxPriceDeviation: number;
  maxCapitalUtilization: number;
  restrictedSymbols: Set<string>;
}

const createComplianceEngine = (
  limits: ComplianceLimits,
): {
  checks: ComplianceCheck[];
  validate: (
    order: OrderRequest,
    ctx: ComplianceContext,
  ) => {
    allowed: boolean;
    results: { check: string; result: ComplianceCheckResult; message: string }[];
  };
} => {
  const checks: ComplianceCheck[] = [
    {
      name: 'position_limit',
      check: (order, ctx) => {
        const current = ctx.positions.get(order.symbol) ?? 0;
        const projected = order.side === 'buy' ? current + order.quantity : current - order.quantity;
        return Math.abs(projected) > limits.maxPositionSize ? 'block' : 'pass';
      },
      message: '持仓超过单品种限额',
    },
    {
      name: 'order_size',
      check: (order) => {
        return order.quantity > limits.maxOrderSize ? 'block' : 'pass';
      },
      message: '单笔订单数量超限',
    },
    {
      name: 'daily_trade_count',
      check: (order, ctx) => {
        const count = ctx.dailyTrades.get(order.symbol) ?? 0;
        return count >= limits.maxDailyTrades ? 'warn' : 'pass';
      },
      message: '日内交易次数接近上限',
    },
    {
      name: 'restricted_symbol',
      check: (order) => {
        return limits.restrictedSymbols.has(order.symbol) ? 'block' : 'pass';
      },
      message: '标的在禁止交易名单中',
    },
    {
      name: 'capital_utilization',
      check: (_, ctx) => {
        return ctx.capitalUtilization > limits.maxCapitalUtilization ? 'block' : 'pass';
      },
      message: '资金使用率超过上限',
    },
    {
      name: 'price_reasonability',
      check: (order, ctx) => {
        const pendingSame = ctx.pendingOrders.filter((o) => o.symbol === order.symbol);
        if (pendingSame.length > 0) {
          const avgPrice = pendingSame.reduce((s, o) => s + o.price, 0) / pendingSame.length;
          const deviation = Math.abs(order.price - avgPrice) / avgPrice;
          if (deviation > limits.maxPriceDeviation) return 'block';
        }
        return 'pass';
      },
      message: '订单价格偏离市场价格过大',
    },
  ];

  return {
    checks,
    validate: (order, ctx) => {
      const results = checks.map((c) => ({
        check: c.name,
        result: c.check(order, ctx),
        message: c.message,
      }));
      const blocked = results.some((r) => r.result === 'block');
      return { allowed: !blocked, results };
    },
  };
};
市场操纵检测
量化交易系统需要内置市场操纵行为的检测逻辑，避免策略无意中产生操纵市场的交易模式。常见的需要监控的操纵行为包括欺骗下单（Spoofing）、分层下单（Layering）和尾市操纵。


interface ManipulationAlert {
  type: 'spoofing' | 'layering' | 'wash_trading' | 'closing_manipulation';
  severity: 'low' | 'medium' | 'high';
  symbol: string;
  timestamp: number;
  details: string;
  orders: string[];
}

interface TradePattern {
  symbol: string;
  orders: {
    orderId: string;
    side: string;
    price: number;
    quantity: number;
    created: number;
    cancelled: number | null;
    filled: number;
    status: string;
  }[];
}

const detectSpoofing = (
  pattern: TradePattern,
  cancelRatioThreshold = 0.9,
  timeWindowMs = 60000,
): ManipulationAlert | null => {
  const recentOrders = pattern.orders.filter((o) => Date.now() - o.created < timeWindowMs);

  if (recentOrders.length < 5) return null;

  const cancelledCount = recentOrders.filter((o) => o.status === 'cancelled').length;
  const cancelRatio = cancelledCount / recentOrders.length;

  if (cancelRatio >= cancelRatioThreshold) {
    return {
      type: 'spoofing',
      severity: cancelRatio > 0.95 ? 'high' : 'medium',
      symbol: pattern.symbol,
      timestamp: Date.now(),
      details: `撤单率 ${(cancelRatio * 100).toFixed(1)}%，${timeWindowMs / 1000}秒内 ${recentOrders.length} 笔订单`,
      orders: recentOrders.map((o) => o.orderId),
    };
  }

  return null;
};

const detectWashTrading = (
  trades: {
    buyAccount: string;
    sellAccount: string;
    symbol: string;
    quantity: number;
    price: number;
    timestamp: number;
  }[],
  sameEntityAccounts: Set<string>,
): ManipulationAlert | null => {
  const selfTrades = trades.filter(
    (t) => sameEntityAccounts.has(t.buyAccount) && sameEntityAccounts.has(t.sellAccount),
  );

  if (selfTrades.length > 0) {
    return {
      type: 'wash_trading',
      severity: 'high',
      symbol: selfTrades[0].symbol,
      timestamp: Date.now(),
      details: `检测到 ${selfTrades.length} 笔自成交`,
      orders: [],
    };
  }

  return null;
};
审计日志与交易记录
完整的审计日志是合规管理的基础设施。监管机构可能要求追溯任何一笔交易的完整决策链——从策略信号生成到订单提交到最终成交。日志系统需要保证不可篡改性和长期可查询性。


interface AuditLogEntry {
  eventId: string;
  timestamp: number;
  eventType:
    | 'signal'
    | 'order_created'
    | 'order_submitted'
    | 'order_filled'
    | 'order_cancelled'
    | 'compliance_check'
    | 'risk_alert';
  strategyId: string;
  symbol: string;
  details: Record<string, unknown>;
  userId: string;
  systemVersion: string;
}

interface AuditTrail {
  orderId: string;
  events: AuditLogEntry[];
  decisionChain: string[];
  complianceChecks: { check: string; result: string; timestamp: number }[];
}

interface AuditLogger {
  log(entry: Omit<AuditLogEntry, 'eventId' | 'timestamp'>): void;
  getTrail(orderId: string): AuditTrail;
  query(filter: AuditQueryFilter): AuditLogEntry[];
  exportForRegulator(startDate: Date, endDate: Date): AuditExport;
}

interface AuditQueryFilter {
  startTime?: number;
  endTime?: number;
  eventType?: string;
  symbol?: string;
  strategyId?: string;
}

interface AuditExport {
  period: { start: Date; end: Date };
  totalOrders: number;
  totalTrades: number;
  complianceBlockedOrders: number;
  alerts: ManipulationAlert[];
  entries: AuditLogEntry[];
}

const createAuditLogger = (storage: AuditStorage): AuditLogger => {
  let sequence = 0;

  return {
    log: (entry) => {
      const full: AuditLogEntry = {
        ...entry,
        eventId: `AUDIT-${Date.now()}-${++sequence}`,
        timestamp: Date.now(),
      };
      storage.append(full);
    },
    getTrail: (orderId) => {
      const events = storage.queryByOrder(orderId);
      return {
        orderId,
        events,
        decisionChain: events.map((e) => `${e.eventType}@${new Date(e.timestamp).toISOString()}`),
        complianceChecks: events
          .filter((e) => e.eventType === 'compliance_check')
          .map((e) => ({
            check: String(e.details['check'] ?? ''),
            result: String(e.details['result'] ?? ''),
            timestamp: e.timestamp,
          })),
      };
    },
    query: (filter) => storage.query(filter),
    exportForRegulator: (startDate, endDate) => {
      const entries = storage.query({
        startTime: startDate.getTime(),
        endTime: endDate.getTime(),
      });
      return {
        period: { start: startDate, end: endDate },
        totalOrders: entries.filter((e) => e.eventType === 'order_created').length,
        totalTrades: entries.filter((e) => e.eventType === 'order_filled').length,
        complianceBlockedOrders: entries.filter(
          (e) => e.eventType === 'compliance_check' && e.details['result'] === 'block',
        ).length,
        alerts: [],
        entries,
      };
    },
  };
};

interface AuditStorage {
  append(entry: AuditLogEntry): void;
  queryByOrder(orderId: string): AuditLogEntry[];
  query(filter: AuditQueryFilter): AuditLogEntry[];
}

小结
    

  • 预交易合规引擎是防止违规交易的第一道防线，所有订单必须通过合规检查才能提交
    • 

  • 市场操纵检测需要监控撤单率、自成交和尾市异常交易模式
    • 

  • 审计日志必须记录完整的交易决策链，保证不可篡改且长期可查
    • 

  • 合规规则应内嵌到系统设计中，通过自动化而非人工审核来保证执行
    • 

  • 监管要求随市场发展不断演变，合规系统需要具备灵活的规则配置能力
    • 

  • 良好的合规实践不仅是法律要求，也是建立机构信任和长期可持续经营的基础
