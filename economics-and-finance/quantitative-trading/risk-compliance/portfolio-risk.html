组合风险管理

核心概念

组合风险管理关注的是多个策略、多个资产构成的投资组合的整体风险特征。单个策略的风险指标无法反映组合层面的风险——策略间的相关性、因子暴露的集中度和极端市场下的尾部相关性才是真正的风险来源。

组合风险管理的目标是在给定风险预算下最大化风险调整后收益，同时确保在极端市场环境下组合不会遭受不可承受的损失。这需要从因子暴露、相关性结构和压力测试三个维度进行全面分析。

协方差矩阵估计
协方差矩阵是组合风险计算的核心输入。准确估计协方差矩阵是一个经典的统计难题——资产数量多时，样本协方差矩阵往往不稳定，需要采用收缩估计或因子模型方法。


interface CovarianceEstimate {
  matrix: number[][];
  method: string;
  condition_number: number;
  shrinkageIntensity?: number;
}

const sampleCovariance = (returns: number[][]): number[][] => {
  const n = returns.length;
  const t = returns[0].length;
  const means = returns.map((r) => r.reduce((a, b) => a + b, 0) / t);

  return returns.map((ri, i) =>
    returns.map((rj, j) => {
      let cov = 0;
      for (let k = 0; k < t; k++) {
        cov += (ri[k] - means[i]) * (rj[k] - means[j]);
      }
      return cov / (t - 1);
    }),
  );
};

const ledoitWolfShrinkage = (returns: number[][]): CovarianceEstimate => {
  const S = sampleCovariance(returns);
  const n = S.length;
  const t = returns[0].length;

  let trS = 0;
  for (let i = 0; i < n; i++) trS += S[i][i];
  const mu = trS / n;

  const F: number[][] = Array.from({ length: n }, (_, i) => Array.from({ length: n }, (_, j) => (i === j ? mu : 0)));

  let numerator = 0;
  let denominator = 0;
  for (let i = 0; i < n; i++) {
    for (let j = 0; j < n; j++) {
      const diff = S[i][j] - F[i][j];
      denominator += diff ** 2;
      let piij = 0;
      for (let k = 0; k < t; k++) {
        const means = returns.map((r) => r.reduce((a, b) => a + b, 0) / t);
        piij += ((returns[i][k] - means[i]) * (returns[j][k] - means[j]) - S[i][j]) ** 2;
      }
      numerator += piij / t;
    }
  }

  const delta = Math.min(1, Math.max(0, numerator / (t * denominator)));

  const shrunk = S.map((row, i) => row.map((val, j) => delta * F[i][j] + (1 - delta) * val));

  return {
    matrix: shrunk,
    method: 'ledoit_wolf',
    condition_number: 0,
    shrinkageIntensity: delta,
  };
};

const portfolioVolatility = (weights: number[], covMatrix: number[][]): number => {
  let variance = 0;
  for (let i = 0; i < weights.length; i++) {
    for (let j = 0; j < weights.length; j++) {
      variance += weights[i] * weights[j] * covMatrix[i][j];
    }
  }
  return Math.sqrt(variance * 252);
};
因子风险分析
因子风险分析将组合的风险分解为系统性因子风险和特质风险。通过分析组合对各因子的暴露程度，可以识别风险集中的来源并采取对冲措施。


interface FactorExposure {
  factorName: string;
  exposure: number;
  marginalRisk: number;
  riskContribution: number;
  riskContributionPct: number;
}

interface PortfolioRiskDecomposition {
  totalRisk: number;
  factorRisk: number;
  specificRisk: number;
  factorExposures: FactorExposure[];
  topRiskContributors: { asset: string; contribution: number }[];
}

const decomposePortfolioRisk = (
  weights: number[],
  factorLoadings: number[][],
  factorCovMatrix: number[][],
  specificVariances: number[],
): PortfolioRiskDecomposition => {
  const n = weights.length;
  const k = factorCovMatrix.length;

  const portfolioFactorExposure: number[] = Array(k).fill(0);
  for (let f = 0; f < k; f++) {
    for (let i = 0; i < n; i++) {
      portfolioFactorExposure[f] += weights[i] * factorLoadings[i][f];
    }
  }

  let factorVariance = 0;
  for (let f1 = 0; f1 < k; f1++) {
    for (let f2 = 0; f2 < k; f2++) {
      factorVariance += portfolioFactorExposure[f1] * portfolioFactorExposure[f2] * factorCovMatrix[f1][f2];
    }
  }

  let specificVariance = 0;
  for (let i = 0; i < n; i++) {
    specificVariance += weights[i] ** 2 * specificVariances[i];
  }

  const totalVariance = factorVariance + specificVariance;
  const totalRisk = Math.sqrt(totalVariance * 252);

  const exposures: FactorExposure[] = portfolioFactorExposure.map((exp, f) => {
    let margContrib = 0;
    for (let f2 = 0; f2 < k; f2++) {
      margContrib += portfolioFactorExposure[f2] * factorCovMatrix[f][f2];
    }
    const riskContrib = exp * margContrib;
    return {
      factorName: `Factor_${f}`,
      exposure: exp,
      marginalRisk: margContrib,
      riskContribution: riskContrib,
      riskContributionPct: totalVariance > 0 ? riskContrib / totalVariance : 0,
    };
  });

  return {
    totalRisk,
    factorRisk: Math.sqrt(factorVariance * 252),
    specificRisk: Math.sqrt(specificVariance * 252),
    factorExposures: exposures,
    topRiskContributors: [],
  };
};
压力测试与情景分析
压力测试评估组合在极端市场环境下的潜在损失。它补充了基于正常市场条件的 VaR/CVaR 分析的不足，帮助识别尾部风险和非线性风险。


历史情景法使用历史上的重大市场事件（如2008年金融危机、2020年新冠崩盘）的实际市场变动来估算组合损失。


假设情景法构建人为设计的极端场景（如利率上升300bp、股市下跌40%），评估组合在这些假设下的表现。


interface StressScenario {
  name: string;
  description: string;
  factorShocks: Map<string, number>;
  probability: number;
}

interface StressTestResult {
  scenario: string;
  portfolioLoss: number;
  portfolioLossPct: number;
  worstAsset: { symbol: string; loss: number };
  recoveryEstimateDays: number;
}

const historicalScenarios: StressScenario[] = [

  {
    name: '利率急升',
    description: '央行激进加息，债市暴跌',
    factorShocks: new Map([
      ['equity', -0.1],
      ['credit', -0.05],
      ['volatility', 0.2],
      ['rates', 0.03],
    ]),
    probability: 0.05,
  },
];

const runStressTest = (
  weights: number[],
  factorLoadings: number[][],
  assetPrices: number[],
  scenario: StressScenario,
): StressTestResult => {
  const n = weights.length;
  const factorNames = [...scenario.factorShocks.keys()];
  let portfolioLoss = 0;
  let worstLoss = 0;
  let worstSymbol = '';

  for (let i = 0; i < n; i++) {
    let assetShock = 0;
    for (let f = 0; f < factorNames.length; f++) {
      const shock = scenario.factorShocks.get(factorNames[f]) ?? 0;
      assetShock += factorLoadings[i][f] * shock;
    }
    const assetLoss = weights[i] * assetShock;
    portfolioLoss += assetLoss;
    if (Math.abs(assetLoss) > Math.abs(worstLoss)) {
      worstLoss = assetLoss;
      worstSymbol = `Asset_${i}`;
    }
  }

  return {
    scenario: scenario.name,
    portfolioLoss,
    portfolioLossPct: portfolioLoss * 100,
    worstAsset: { symbol: worstSymbol, loss: worstLoss },
    recoveryEstimateDays: Math.round(Math.abs(portfolioLoss) * 500),
  };
};
小结
    

  • 组合风险管理从因子暴露、相关性和尾部风险三个维度全面分析组合特征
    • 

  • Ledoit-Wolf 收缩估计比样本协方差矩阵更稳定，是实战中的标准选择
    • 

  • 因子风险分解识别系统性风险来源，帮助针对性地进行因子对冲
    • 

  • 压力测试补充了正常市场条件下 VaR 分析的不足，揭示极端场景下的潜在损失
    • 

  • 历史情景和假设情景结合使用，既覆盖已知风险又预防未知风险
    • 

  • 组合风险管理应建立日常化的监控和报告机制，而非仅在危机时才启动
    •
