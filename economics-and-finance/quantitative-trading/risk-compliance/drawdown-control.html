回撤控制

核心概念

回撤控制是量化策略风险管理中最实用的技术。回撤（Drawdown）衡量投资组合从历史最高点到当前值的跌幅，是投资者和基金管理人最直观感受到的风险指标。有效的回撤控制机制能在策略表现不佳时自动降低风险敞口，防止小亏损演变为大灾难。

回撤控制的设计需要在"过早减仓错过反弹"和"过晚减仓损失扩大"之间找到平衡。过于敏感的回撤控制会频繁止损导致策略被"锯齿"消耗，过于迟钝的控制则无法在真正的危机中保护资本。

分层回撤控制引擎

分层回撤控制将回撤深度映射到不同的风险响应级别，每个级别对应不同的仓位缩减比例和恢复条件。这种阶梯式设计避免了二元（全仓或空仓）决策的极端性。

interface DrawdownLevel {
  name: string;
  threshold: number;
  targetExposure: number;
  cooldownDays: number;
  recoveryCondition: 'time' | 'recovery_pct' | 'manual';
  recoveryParam: number;
}

interface DrawdownController {
  levels: DrawdownLevel[];
  currentLevel: number;
  highWaterMark: number;
  currentDrawdown: number;
  cooldownEndDate: number | null;
  history: DrawdownEvent[];
}

interface DrawdownEvent {
  timestamp: number;
  drawdown: number;
  level: number;
  action: string;
  exposure: number;
}

const createDrawdownController = (levels: DrawdownLevel[], initialEquity: number): DrawdownController => ({
  levels: levels.sort((a, b) => a.threshold - b.threshold),
  currentLevel: 0,
  highWaterMark: initialEquity,
  currentDrawdown: 0,
  cooldownEndDate: null,
  history: [],
});

const updateDrawdownState = (
  controller: DrawdownController,
  currentEquity: number,
  timestamp: number,
): { targetExposure: number; levelChanged: boolean; action: string } => {
  if (currentEquity > controller.highWaterMark) {
    controller.highWaterMark = currentEquity;
  }

  controller.currentDrawdown = (controller.highWaterMark - currentEquity) / controller.highWaterMark;

  if (controller.cooldownEndDate && timestamp < controller.cooldownEndDate) {
    return { targetExposure: 0, levelChanged: false, action: 'cooldown_active' };
  }

  let newLevel = 0;
  for (let i = controller.levels.length - 1; i >= 0; i--) {
    if (controller.currentDrawdown >= controller.levels[i].threshold) {
      newLevel = i + 1;
      break;
    }
  }

  const levelChanged = newLevel !== controller.currentLevel;
  controller.currentLevel = newLevel;

  let targetExposure = 1.0;
  let action = 'normal';

  if (newLevel > 0) {
    const level = controller.levels[newLevel - 1];
    targetExposure = level.targetExposure;
    action = `reduce_to_${(targetExposure * 100).toFixed(0)}pct`;

    if (level.cooldownDays > 0 && levelChanged) {
      controller.cooldownEndDate = timestamp + level.cooldownDays * 86400000;
      targetExposure = 0;
      action = `forced_exit_cooldown_${level.cooldownDays}d`;
    }
  }

  if (levelChanged) {
    controller.history.push({
      timestamp,
      drawdown: controller.currentDrawdown,
      level: newLevel,
      action,
      exposure: targetExposure,
    });
  }

  return { targetExposure, levelChanged, action };
};

动态止损策略

止损是单笔交易层面的回撤控制。合理的止损策略在保护资本和给策略足够运行空间之间取得平衡。常见的止损方法包括固定比例止损、波动率止损和追踪止损。

type StopLossType = 'fixed' | 'atr' | 'trailing' | 'time' | 'breakeven';

interface StopLossConfig {
  type: StopLossType;
  fixedPct?: number;
  atrMultiplier?: number;
  trailingActivation?: number;
  trailingDistance?: number;
  maxHoldingDays?: number;
  breakevenBuffer?: number;
}

interface StopLossState {
  entryPrice: number;
  currentStop: number;
  highestPrice: number;
  holdingDays: number;
  isActive: boolean;
  type: StopLossType;
}

const initStopLoss = (
  config: StopLossConfig,
  entryPrice: number,
  atr: number,
  side: 'long' | 'short',
): StopLossState => {
  let stopPrice: number;
  const direction = side === 'long' ? -1 : 1;

  switch (config.type) {
    case 'fixed':
      stopPrice = entryPrice * (1 + direction * (config.fixedPct ?? 0.05));
      break;
    case 'atr':
      stopPrice = entryPrice + direction * atr * (config.atrMultiplier ?? 2);
      break;
    case 'trailing':
      stopPrice = entryPrice * (1 + direction * (config.trailingDistance ?? 0.03));
      break;
    default:
      stopPrice = entryPrice * (1 + direction * 0.05);
  }

  return {
    entryPrice,
    currentStop: stopPrice,
    highestPrice: entryPrice,
    holdingDays: 0,
    isActive: true,
    type: config.type,
  };
};

const updateStopLoss = (
  state: StopLossState,
  currentPrice: number,
  config: StopLossConfig,
  side: 'long' | 'short',
): { triggered: boolean; newStop: number } => {
  state.holdingDays++;

  if (config.type === 'time' && state.holdingDays >= (config.maxHoldingDays ?? 10)) {
    return { triggered: true, newStop: state.currentStop };
  }

  if (side === 'long') {
    if (currentPrice > state.highestPrice) {
      state.highestPrice = currentPrice;
    }

    if (config.type === 'trailing') {
      const activation = config.trailingActivation ?? 0.02;
      const distance = config.trailingDistance ?? 0.03;
      if (currentPrice > state.entryPrice * (1 + activation)) {
        const newStop = state.highestPrice * (1 - distance);
        state.currentStop = Math.max(state.currentStop, newStop);
      }
    }

    if (config.type === 'breakeven') {
      const buffer = config.breakevenBuffer ?? 0.01;
      if (currentPrice > state.entryPrice * (1 + buffer * 2)) {
        state.currentStop = Math.max(state.currentStop, state.entryPrice * (1 + buffer));
      }
    }

    return { triggered: currentPrice <= state.currentStop, newStop: state.currentStop };
  }

  return { triggered: currentPrice >= state.currentStop, newStop: state.currentStop };
};

回撤恢复机制

从回撤控制状态恢复到正常运行同样重要。过于激进的恢复可能导致二次回撤，过于保守的恢复则错过反弹收益。系统化的恢复机制确保资金逐步回到正常配置。

interface RecoveryConfig {
  method: 'linear' | 'step' | 'conditional';
  recoverySteps: number;
  minRecoveryDays: number;
  volatilityCheck: boolean;
  volatilityThreshold: number;
  newHighRequired: boolean;
}

interface RecoveryState {
  isRecovering: boolean;
  currentStep: number;
  totalSteps: number;
  recoveryStartDate: number;
  targetExposure: number;
  lastIncrease: number;
}

const calculateRecoveryExposure = (
  config: RecoveryConfig,
  state: RecoveryState,
  currentEquity: number,
  highWaterMark: number,
  currentVolatility: number,
  timestamp: number,
): number => {
  if (!state.isRecovering) return 1.0;

  const daysSinceStart = (timestamp - state.recoveryStartDate) / 86400000;
  if (daysSinceStart < config.minRecoveryDays) {
    return state.targetExposure;
  }

  if (config.volatilityCheck && currentVolatility > config.volatilityThreshold) {
    return state.targetExposure;
  }

  if (config.newHighRequired && currentEquity < highWaterMark) {
    const recoveryPct = currentEquity / highWaterMark;
    return Math.min(1.0, state.targetExposure + (1 - state.targetExposure) * recoveryPct);
  }

  switch (config.method) {
    case 'linear': {
      const progress = Math.min(1, state.currentStep / state.totalSteps);
      return state.targetExposure + (1 - state.targetExposure) * progress;
    }
    case 'step': {
      const stepSize = (1 - state.targetExposure) / state.totalSteps;
      return Math.min(1.0, state.targetExposure + stepSize * state.currentStep);
    }
    case 'conditional':
      return currentEquity >= highWaterMark * 0.98 ? 1.0 : state.targetExposure;
    default:
      return state.targetExposure;
  }
};
恢复策略 速度 风险 适用场景
线性恢复 均匀 中 策略回撤为正常波动
阶梯恢复 离散 中低 需要观察每步效果
条件恢复 自适应 低 需确认环境改善
净值恢复 慢 最低 保守型资金管理

  • 分层回撤控制将回撤深度映射到不同风险响应级别，避免二元决策的极端性
  • 追踪止损在保护利润的同时给策略足够的运行空间，是最常用的止损方式
  • 保本止损在盈利达到一定水平后将止损位移至成本价上方，锁定零风险状态
  • 恢复机制应逐步增加仓位而非一次性恢复，防止二次回撤
  • 冷静期设计防止在高波动市场中频繁触发止损和恢复，减少"锯齿"损耗
  • 回撤控制参数应与策略特征匹配——趋势策略需要更宽的止损，均值回归策略需要更紧的控制
