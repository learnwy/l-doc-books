支付基础设施

核心概念

支付基础设施是金融科技的底层骨架，连接着消费者、商户、银行和清算机构。现代支付系统已经从单纯的资金转移演进为一个多层次、多协议的复杂技术栈，涵盖前端支付体验、交易处理引擎、风控系统、清算网络和最终结算等环节。

金融科技公司正在重塑支付基础设施的每一层。从 Stripe 的开发者友好支付 API，到支付宝/微信的二维码支付生态，再到区块链支付的去中介化探索，技术创新不断降低支付的摩擦和成本。

支付协议与消息标准
支付系统之间的互联互通依赖标准化的消息协议。ISO 8583 是传统银行卡交易的核心消息标准，而 ISO 20022 正在成为全球支付消息的统一标准，支持更丰富的数据格式和更灵活的业务场景。


interface ISO8583Message {
  mti: string;
  bitmap: string;
  fields: Map<number, string>;
}

const buildAuthorizationRequest = (
  cardNumber: string,
  amount: number,
  currency: string,
  merchantId: string,
): ISO8583Message => {
  const fields = new Map<number, string>();

  fields.set(2, cardNumber);
  fields.set(3, '000000');
  fields.set(4, amount.toFixed(2).replace('.', '').padStart(12, '0'));
  fields.set(11, generateSTAN());
  fields.set(12, formatLocalTime());
  fields.set(13, formatLocalDate());
  fields.set(22, '051');
  fields.set(25, '00');
  fields.set(41, merchantId.padEnd(8));
  fields.set(42, merchantId.padEnd(15));
  fields.set(49, currency);

  return {
    mti: '0100',
    bitmap: calculateBitmap(fields),
    fields,
  };
};

const generateSTAN = (): string => {
  return Math.floor(Math.random() * 999999)
    .toString()
    .padStart(6, '0');
};

const formatLocalTime = (): string => {
  const now = new Date();
  return [
    now.getHours().toString().padStart(2, '0'),
    now.getMinutes().toString().padStart(2, '0'),
    now.getSeconds().toString().padStart(2, '0'),
  ].join('');
};

const formatLocalDate = (): string => {
  const now = new Date();
  return [(now.getMonth() + 1).toString().padStart(2, '0'), now.getDate().toString().padStart(2, '0')].join('');
};

const calculateBitmap = (fields: Map<number, string>): string => {
  const bitmap = new Uint8Array(8);
  for (const fieldNum of fields.keys()) {
    if (fieldNum > 0 && fieldNum <= 64) {
      const byteIndex = Math.floor((fieldNum - 1) / 8);
      const bitIndex = 7 - ((fieldNum - 1) % 8);
      bitmap[byteIndex] |= 1 << bitIndex;
    }
  }
  return Array.from(bitmap)
    .map((b) => b.toString(16).padStart(2, '0'))
    .join('');
};
ISO 20022 使用 XML/JSON 格式，支持结构化的汇款信息和更丰富的交易元数据，是 SWIFT gpi、SEPA 和各国实时支付系统迁移的目标标准。


实时支付系统

interface RealTimePayment {
  paymentId: string;
  debtor: PartyInfo;
  creditor: PartyInfo;
  amount: MonetaryAmount;
  purpose: string;
  timestamp: number;
  status: PaymentStatus;
}

interface PartyInfo {
  name: string;
  accountId: string;
  institutionId: string;
}

interface MonetaryAmount {
  value: bigint;
  currency: string;
  decimals: number;
}

type PaymentStatus = 'initiated' | 'validated' | 'authorized' | 'cleared' | 'settled' | 'rejected';

interface PaymentProcessor {
  validate(payment: RealTimePayment): ValidationResult;
  authorize(payment: RealTimePayment): Promise<AuthorizationResult>;
  clear(payment: RealTimePayment): Promise<ClearingResult>;
  settle(payment: RealTimePayment): Promise<SettlementResult>;
}

interface ValidationResult {
  valid: boolean;
  errors: string[];
}

interface AuthorizationResult {
  authorized: boolean;
  authCode: string;
  reason?: string;
}

interface ClearingResult {
  cleared: boolean;
  clearingReference: string;
  timestamp: number;
}

interface SettlementResult {
  settled: boolean;
  settlementReference: string;
  finalTimestamp: number;
}

const processRealTimePayment = async (
  payment: RealTimePayment,
  processor: PaymentProcessor,
): Promise<PaymentStatus> => {
  const validation = processor.validate(payment);
  if (!validation.valid) return 'rejected';

  const authorization = await processor.authorize(payment);
  if (!authorization.authorized) return 'rejected';

  const clearing = await processor.clear(payment);
  if (!clearing.cleared) return 'rejected';

  const settlement = await processor.settle(payment);
  return settlement.settled ? 'settled' : 'cleared';
};
支付安全与风控
支付安全是金融基础设施的生命线。现代支付风控系统采用多层防御策略，结合规则引擎和机器学习模型来识别和阻止欺诈交易。


interface TransactionRiskAssessment {
  transactionId: string;
  riskScore: number;
  signals: RiskSignal[];
  decision: 'approve' | 'review' | 'decline';
  processingTimeMs: number;
}

interface RiskSignal {
  type: string;
  severity: 'low' | 'medium' | 'high';
  description: string;
  weight: number;
}

const RISK_RULES: Array<{
  name: string;
  evaluate: (tx: PaymentTransaction) => RiskSignal | null;
}> = [
  {
    name: 'high_amount',
    evaluate: (tx) =>
      tx.amount > 50000n ? { type: 'high_amount', severity: 'medium', description: '单笔大额交易', weight: 25 } : null,
  },
  {
    name: 'unusual_time',
    evaluate: (tx) => {
      const hour = new Date(tx.timestamp).getHours();
      return hour >= 1 && hour <= 5
        ? { type: 'unusual_time', severity: 'low', description: '异常时段交易', weight: 10 }
        : null;
    },
  },
  {
    name: 'cross_border',
    evaluate: (tx) =>
      tx.merchantCountry !== tx.cardCountry
        ? { type: 'cross_border', severity: 'medium', description: '跨境交易', weight: 20 }
        : null,
  },
  {
    name: 'velocity_check',
    evaluate: (tx) =>
      tx.recentTransactionCount > 10
        ? { type: 'velocity', severity: 'high', description: '高频交易', weight: 35 }
        : null,
  },
];

interface PaymentTransaction {
  id: string;
  amount: bigint;
  timestamp: number;
  merchantCountry: string;
  cardCountry: string;
  recentTransactionCount: number;
}

const assessTransactionRisk = (tx: PaymentTransaction): TransactionRiskAssessment => {
  const startTime = performance.now();
  const signals: RiskSignal[] = [];

  for (const rule of RISK_RULES) {
    const signal = rule.evaluate(tx);
    if (signal) signals.push(signal);
  }

  const riskScore = signals.reduce((sum, s) => sum + s.weight, 0);

  let decision: 'approve' | 'review' | 'decline';
  if (riskScore < 30) decision = 'approve';
  else if (riskScore < 70) decision = 'review';
  else decision = 'decline';

  return {
    transactionId: tx.id,
    riskScore,
    signals,
    decision,
    processingTimeMs: performance.now() - startTime,
  };
};
