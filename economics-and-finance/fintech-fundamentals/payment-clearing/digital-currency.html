数字货币与 CBDC

核心概念

数字货币是以数字形式存在的货币，涵盖加密货币（Bitcoin、Ethereum）、稳定币（USDC、USDT）和央行数字货币（CBDC）等多种形态。其中，CBDC（Central Bank Digital Currency）是由中央银行直接发行和背书的数字化法定货币，代表了货币数字化的最高层级——具备法定货币的全部属性，同时兼具数字化的效率优势。

各国央行正在积极探索 CBDC，截至目前全球超过 130 个国家和地区在研究或试点 CBDC。中国的数字人民币（e-CNY）是全球最大规模的 CBDC 试点项目，而欧洲央行的数字欧元和美联储的数字美元研究也在推进中。

CBDC 架构设计
CBDC 的技术架构通常分为两种模式：直接模式（央行直接管理所有用户账户）和间接/双层模式（央行发行给商业银行，商业银行面向公众分发）。大多数国家选择双层运营模式，以保留现有金融中介体系的作用。


interface CBDCArchitecture {
  model: 'direct' | 'indirect' | 'hybrid';
  ledgerType: 'centralized' | 'permissioned_dlt' | 'hybrid';
  privacyModel: PrivacyModel;
  offlineCapability: boolean;
  programmability: ProgrammabilityLevel;
  interoperability: InteropConfig;
}

interface PrivacyModel {
  type: 'full_anonymity' | 'controlled_anonymity' | 'pseudonymous' | 'identified';
  smallTxAnonymous: boolean;
  largeTxIdentified: boolean;
  thresholdAmount: bigint;
}

type ProgrammabilityLevel = 'none' | 'basic_conditions' | 'smart_contracts';

interface InteropConfig {
  domesticPaymentSystems: string[];
  crossBorderProtocols: string[];
  privateStablecoinBridge: boolean;
}

interface CBDCWallet {
  walletId: string;
  walletType: 'software' | 'hardware' | 'card';
  tier: WalletTier;
  balance: bigint;
  holdLimit: bigint;
  dailyTxLimit: bigint;
}

interface WalletTier {
  level: 1 | 2 | 3 | 4;
  kycRequirement: string;
  dailyLimit: bigint;
  holdingLimit: bigint;
  anonymityLevel: string;
}

    holdingLimit: 500000_00n,

const selectWalletTier = (dailyUsage: bigint, holdingNeed: bigint): WalletTier => {
  for (const tier of ECNY_WALLET_TIERS) {
    if (tier.level === 1) return tier;
    if (dailyUsage <= tier.dailyLimit && holdingNeed <= tier.holdingLimit) {
      return tier;
    }
  }
  return ECNY_WALLET_TIERS[ECNY_WALLET_TIERS.length - 1];
};
数字人民币（e-CNY）采用"一币、两库、三中心"架构：一种数字货币、央行数字货币发行库和商业银行数字货币库、认证中心/登记中心/大数据分析中心。


可编程货币与智能合约
CBDC 的一个革命性潜力是可编程性（Programmability）——通过智能合约为货币赋予自动执行的条件逻辑。这使得补贴精准发放、贸易融资自动化、合规自动检查等场景成为可能。


interface ProgrammablePayment {
  id: string;
  sender: string;
  recipient: string;
  amount: bigint;
  conditions: PaymentCondition[];
  expiresAt: number;
  status: 'pending' | 'executed' | 'expired' | 'cancelled';
}

type PaymentCondition =
  | { type: 'time_lock'; unlockTimestamp: number }
  | { type: 'delivery_confirmation'; trackingId: string; confirmed: boolean }
  | { type: 'milestone'; milestoneId: string; achieved: boolean }
  | { type: 'spending_category'; allowedCategories: string[] };

const evaluateConditions = (conditions: PaymentCondition[]): boolean => {
  return conditions.every((condition) => {
    switch (condition.type) {
      case 'time_lock':
        return Date.now() >= condition.unlockTimestamp * 1000;
      case 'delivery_confirmation':
        return condition.confirmed;
      case 'milestone':
        return condition.achieved;
      case 'spending_category':
        return condition.allowedCategories.length > 0;
      default:
        return false;
    }
  });
};

interface SubsidyVoucher {
  voucherId: string;
  beneficiary: string;
  amount: bigint;
  category: string;
  validFrom: number;
  validUntil: number;
  merchantWhitelist: string[];
  usedAmount: bigint;
}

const validateSubsidyUsage = (
  voucher: SubsidyVoucher,
  merchantId: string,
  amount: bigint,
  timestamp: number,
): { valid: boolean; reason?: string } => {
  if (timestamp < voucher.validFrom) return { valid: false, reason: '券未生效' };
  if (timestamp > voucher.validUntil) return { valid: false, reason: '券已过期' };
  if (voucher.usedAmount + amount > voucher.amount) return { valid: false, reason: '余额不足' };
  if (!voucher.merchantWhitelist.includes(merchantId)) return { valid: false, reason: '商户不在白名单' };
  return { valid: true };
};

隐私保护与监管平衡
CBDC 设计中最核心的争议是隐私与监管的平衡。完全匿名会为洗钱和恐怖融资提供便利，完全透明则侵犯公民隐私权。大多数 CBDC 设计采用"可控匿名"（Controlled Anonymity）策略。


interface PrivacyTier {
  transactionThreshold: bigint;
  identityVerification: 'none' | 'phone' | 'id_card' | 'face_recognition';
  dataAccessLevel: 'anonymous' | 'pseudonymous' | 'identified';
  auditTrail: boolean;
}

const PRIVACY_TIERS: PrivacyTier[] = [
  {
    transactionThreshold: 200_00n,
    identityVerification: 'none',
    dataAccessLevel: 'anonymous',
    auditTrail: false,
  },
  {
    transactionThreshold: 5000_00n,
    identityVerification: 'phone',
    dataAccessLevel: 'pseudonymous',
    auditTrail: true,
  },
  {
    transactionThreshold: 50000_00n,
    identityVerification: 'id_card',
    dataAccessLevel: 'identified',
    auditTrail: true,
  },
];

const determinePrivacyLevel = (amount: bigint): PrivacyTier => {
  for (const tier of PRIVACY_TIERS) {
    if (amount <= tier.transactionThreshold) return tier;
  }
  return PRIVACY_TIERS[PRIVACY_TIERS.length - 1];
};

interface AMLScreeningResult {
  transactionId: string;
  riskLevel: 'low' | 'medium' | 'high' | 'blocked';
  triggeredRules: string[];
  requiresReporting: boolean;
}

const screenTransaction = (
  sender: string,
  recipient: string,
  amount: bigint,
  _watchlist: Set<string>,
): AMLScreeningResult => {
  const triggeredRules: string[] = [];

  if (_watchlist.has(sender) || _watchlist.has(recipient)) {
    triggeredRules.push('watchlist_match');
  }

  if (amount > 50000_00n) {
    triggeredRules.push('large_transaction');
  }

  const riskLevel = triggeredRules.includes('watchlist_match')
    ? ('blocked' as const)
    : triggeredRules.length > 0
      ? ('medium' as const)
      : ('low' as const);

  return {
    transactionId: `tx_${Date.now()}`,
    riskLevel,
    triggeredRules,
    requiresReporting: amount > 50000_00n,
  };
};
