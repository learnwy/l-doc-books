Solidity 语言基础

核心概念

Solidity 是以太坊智能合约的主要编程语言，一种面向合约的、静态类型的高级语言，编译后运行在 EVM（以太坊虚拟机）上。它的语法受 JavaScript、Python 和 C++ 影响，但其执行模型与传统编程语言有本质区别——每一步计算都需要消耗 Gas，且所有状态变更都是全局可见且不可逆的。

对于有传统编程经验的开发者来说，Solidity 最大的范式转换在于：代码即法律（Code is Law），一旦部署就无法修改；每个操作都有经济成本；合约间的交互涉及复杂的安全考量。

数据类型与存储位置
Solidity 的数据类型分为值类型和引用类型。值类型在赋值时复制，引用类型则需要显式指定存储位置（storage、memory 或 calldata）。理解存储位置对于优化 Gas 消耗至关重要。


Storage 是合约的永久存储，数据写入区块链状态树，成本最高（写入 20,000 Gas，修改 5,000 Gas）。Memory 是函数执行期间的临时内存，调用结束后释放。Calldata 是只读的调用数据区域，仅用于外部函数参数，Gas 消耗最低。


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract DataTypes {
    uint256 public totalSupply;
    mapping(address => uint256) public balances;
    address[] public holders;

    struct TokenInfo {
        string name;
        string symbol;
        uint8 decimals;
        uint256 maxSupply;
    }

    TokenInfo public tokenInfo;

    enum OrderStatus { Pending, Filled, Cancelled }

    event Transfer(address indexed from, address indexed to, uint256 value);

    function transfer(address to, uint256 amount) external {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        require(to != address(0), "Invalid recipient");

        unchecked {
            balances[msg.sender] -= amount;
            balances[to] += amount;
        }

        emit Transfer(msg.sender, to, amount);
    }

    function batchTransfer(
        address[] calldata recipients,
        uint256[] calldata amounts
    ) external {
        require(recipients.length == amounts.length, "Length mismatch");

        uint256 totalAmount;
        for (uint256 i; i < amounts.length; ) {
            totalAmount += amounts[i];
            unchecked { ++i; }
        }
        require(balances[msg.sender] >= totalAmount, "Insufficient balance");

        for (uint256 i; i < recipients.length; ) {
            balances[msg.sender] -= amounts[i];
            balances[recipients[i]] += amounts[i];
            emit Transfer(msg.sender, recipients[i], amounts[i]);
            unchecked { ++i; }
        }
    }
}

函数修饰符与访问控制
Solidity 通过可见性修饰符（public、external、internal、private）和自定义 modifier 实现细粒度的访问控制。modifier 是 Solidity 独有的特性，允许在函数执行前后插入检查逻辑。


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

abstract contract AccessControl {
    mapping(bytes32 => mapping(address => bool)) private _roles;

    bytes32 public constant ADMIN_ROLE = keccak256("ADMIN_ROLE");
    bytes32 public constant MINTER_ROLE = keccak256("MINTER_ROLE");

    event RoleGranted(bytes32 indexed role, address indexed account);
    event RoleRevoked(bytes32 indexed role, address indexed account);

    modifier onlyRole(bytes32 role) {
        require(_roles[role][msg.sender], "AccessControl: unauthorized");
        _;
    }

    modifier nonReentrant() {
        require(_status != 2, "ReentrancyGuard: reentrant call");
        _status = 2;
        _;
        _status = 1;
    }

    uint256 private _status = 1;

    function grantRole(bytes32 role, address account) external onlyRole(ADMIN_ROLE) {
        _roles[role][account] = true;
        emit RoleGranted(role, account);
    }

    function revokeRole(bytes32 role, address account) external onlyRole(ADMIN_ROLE) {
        _roles[role][account] = false;
        emit RoleRevoked(role, account);
    }

    function hasRole(bytes32 role, address account) public view returns (bool) {
        return _roles[role][account];
    }
}

contract Token is AccessControl {
    string public name;
    string public symbol;
    uint256 public totalSupply;
    mapping(address => uint256) public balanceOf;

    constructor(string memory _name, string memory _symbol) {
        name = _name;
        symbol = _symbol;
        _grantInitialRole(ADMIN_ROLE, msg.sender);
    }

    function mint(address to, uint256 amount) external onlyRole(MINTER_ROLE) nonReentrant {
        totalSupply += amount;
        balanceOf[to] += amount;
    }

    function _grantInitialRole(bytes32 role, address account) internal {
        _roles[role][account] = true;
        emit RoleGranted(role, account);
    }

    mapping(bytes32 => mapping(address => bool)) private _roles;
}
事件与日志系统
事件（Event）是 Solidity 合约与外部世界通信的主要机制。事件数据存储在交易收据的日志中，不占用合约 Storage，Gas 成本相对较低。前端应用通过监听事件来响应链上状态变化。


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract EventDrivenDEX {
    event OrderCreated(
        uint256 indexed orderId,
        address indexed maker,
        address tokenIn,
        address tokenOut,
        uint256 amountIn,
        uint256 amountOut
    );

    event OrderFilled(
        uint256 indexed orderId,
        address indexed taker,
        uint256 filledAmount
    );

    event OrderCancelled(uint256 indexed orderId);

    struct Order {
        address maker;
        address tokenIn;
        address tokenOut;
        uint256 amountIn;
        uint256 amountOut;
        uint256 filled;
        bool cancelled;
    }

    uint256 private _nextOrderId;
    mapping(uint256 => Order) public orders;

    function createOrder(
        address tokenIn,
        address tokenOut,
        uint256 amountIn,
        uint256 amountOut
    ) external returns (uint256 orderId) {
        orderId = _nextOrderId++;

        orders[orderId] = Order({
            maker: msg.sender,
            tokenIn: tokenIn,
            tokenOut: tokenOut,
            amountIn: amountIn,
            amountOut: amountOut,
            filled: 0,
            cancelled: false
        });

        emit OrderCreated(orderId, msg.sender, tokenIn, tokenOut, amountIn, amountOut);
    }

    function cancelOrder(uint256 orderId) external {
        Order storage order = orders[orderId];
        require(order.maker == msg.sender, "Not order maker");
        require(!order.cancelled, "Already cancelled");

        order.cancelled = true;
        emit OrderCancelled(orderId);
    }
}
前端使用 ethers.js 或 viem 监听合约事件：


import { createPublicClient, http, parseAbiItem } from 'viem';
import { mainnet } from 'viem/chains';

const client = createPublicClient({
  chain: mainnet,
  transport: http(),
});

const watchTransfers = async (contractAddress: `0x${string}`) => {
  const unwatch = client.watchEvent({
    address: contractAddress,
    event: parseAbiItem('event Transfer(address indexed from, address indexed to, uint256 value)'),
    onLogs: (logs) => {
      for (const log of logs) {
        const { from, to, value } = log.args;
        console.log(`Transfer: ${from} -> ${to}: ${value}`);
      }
    },
  });

  return unwatch;
};

const getHistoricalEvents = async (contractAddress: `0x${string}`, fromBlock: bigint, toBlock: bigint) => {
  const logs = await client.getLogs({
    address: contractAddress,
    event: parseAbiItem(
      'event OrderCreated(uint256 indexed orderId, address indexed maker, address tokenIn, address tokenOut, uint256 amountIn, uint256 amountOut)',
    ),
    fromBlock,
    toBlock,
  });

  return logs.map((log) => ({
    orderId: log.args.orderId,
    maker: log.args.maker,
    blockNumber: log.blockNumber,
  }));
};

  • Solidity 是面向合约的静态类型语言，每一步执行都有 Gas 成本
    • 

  • 理解 storage/memory/calldata 三种存储位置是 Gas 优化的基础
    • 

  • modifier 机制实现了声明式的访问控制和前置/后置检查
    • 

  • 事件系统是合约与外部应用通信的低成本通道
    • 

  • unchecked 块和 calldata 参数是常见的 Gas 优化手段
    •
