DeFi 协议概览

核心概念

去中心化金融（Decentralized Finance，DeFi）是建立在区块链和智能合约之上的开放金融系统，旨在用代码替代传统金融中介（银行、交易所、保险公司等）的角色。DeFi 的核心理念是"无需许可、无需信任、透明可组合"——任何人都可以访问金融服务，协议之间可以像乐高积木一样自由组合。

2020 年的"DeFi 夏天"标志着这个领域的爆发式增长，总锁仓价值（TVL）从不到 10 亿美元飙升到超过 1000 亿美元。尽管经历了多次市场波动，DeFi 协议已经证明了去中心化金融的可行性，并持续创新。

可组合性与货币乐高
DeFi 最革命性的特征是可组合性（Composability）——协议之间可以无需许可地相互调用和集成，形成复杂的金融产品链。这种特性被形象地称为"货币乐高"（Money Legos）。


一个典型的 DeFi 组合策略：用户将 ETH 存入 Lido 获得 stETH，将 stETH 存入 Aave 作为抵押品借出 USDC，再将 USDC 投入 Curve 的稳定币池赚取交易手续费和 CRV 代币奖励。


interface DeFiProtocol {
  name: string;
  type: 'dex' | 'lending' | 'staking' | 'yield' | 'bridge';
  chainId: number;
  tvl: bigint;
  tokens: string[];
}

interface ComposableAction {
  protocol: string;
  action: string;
  tokenIn: string;
  tokenOut: string;
  amount: bigint;
}

const buildYieldStrategy = (
  principal: bigint,
  actions: ComposableAction[],
): {
  steps: string[];
  estimatedAPY: number;
  risks: string[];
} => {
  const steps = actions.map((a, i) => `${i + 1}. ${a.protocol}: ${a.action} ${a.tokenIn} → ${a.tokenOut}`);

  const estimatedAPY = actions.reduce((apy, action) => {
    const protocolAPY = getProtocolBaseAPY(action.protocol, action.action);
    return apy + protocolAPY;
  }, 0);

  const risks = [
    '智能合约风险：任一协议的漏洞可能影响整个策略',
    '清算风险：抵押品价值下跌可能触发强制清算',
    '协议依赖风险：上游协议变更可能中断策略',
    '预言机风险：价格数据异常可能导致异常操作',
  ];

  return { steps, estimatedAPY, risks };
};

const getProtocolBaseAPY = (protocol: string, action: string): number => {
  const apyMap: Record<string, Record<string, number>> = {
    lido: { stake: 3.5 },
    aave: { supply: 2.0, borrow: -4.5 },
    curve: { provide_liquidity: 5.0 },
    convex: { stake_lp: 8.0 },
  };

  return apyMap[protocol.toLowerCase()]?.[action] ?? 0;
};

interface ProtocolInteraction {
  target: string;
  calldata: string;
  value: bigint;
}

const encodeMulticall = (interactions: ProtocolInteraction[]): string => {
  return interactions.map((i) => `${i.target}:${i.calldata}:${i.value}`).join('|');
};
可组合性带来了巨大的创新空间，但也引入了系统性风险——当底层协议出问题时，所有依赖它的上层协议都会受到影响，这就是所谓的"DeFi 传染效应"。

MakerDAO 的 DAI 是最经典的去中心化稳定币：用户将 ETH 等加密资产超额抵押（至少 150%）到智能合约中，铸造出 DAI。当抵押率低于清算线时，系统自动清算抵押品以维持 DAI 的价值锚定。

pragma solidity ^0.8.20;

interface IPriceOracle {
    function getPrice(address token) external view returns (uint256);
}

contract SimpleCDP {
    IPriceOracle public immutable oracle;
    address public immutable collateralToken;

    uint256 public constant LIQUIDATION_RATIO = 150;
    uint256 public constant LIQUIDATION_PENALTY = 13;
    uint256 public constant STABILITY_FEE_BPS = 50;

    struct Vault {
        uint256 collateral;
        uint256 debt;
        uint256 lastUpdate;
    }

    mapping(address => Vault) public vaults;
    uint256 public totalDebt;

    event VaultOpened(address indexed owner, uint256 collateral, uint256 debt);
    event VaultLiquidated(address indexed owner, address indexed liquidator, uint256 debt);

    constructor(address _oracle, address _collateralToken) {
        oracle = IPriceOracle(_oracle);
        collateralToken = _collateralToken;
    }

    function getCollateralRatio(address owner) public view returns (uint256) {
        Vault storage v = vaults[owner];
        if (v.debt == 0) return type(uint256).max;

        uint256 price = oracle.getPrice(collateralToken);
        uint256 collateralValue = v.collateral * price / 1e18;

        return collateralValue * 100 / v.debt;
    }

    function isLiquidatable(address owner) public view returns (bool) {
        return getCollateralRatio(owner) < LIQUIDATION_RATIO;
    }
}
预言机与外部数据
DeFi 协议需要获取链外的价格数据来执行清算、交易和风控等操作，预言机（Oracle）是连接链上合约与链外世界的桥梁。Chainlink 是最广泛使用的去中心化预言机网络，通过多个独立数据源和节点运营商的聚合来保证数据质量。

  source: string;

interface TWAPConfig {
  windowSize: number;
  maxDeviation: number;
}

const calculateTWAP = (observations: Array<{ price: bigint; timestamp: number }>, config: TWAPConfig): bigint => {
  const now = Math.floor(Date.now() / 1000);
  const windowStart = now - config.windowSize;

  const relevant = observations.filter((o) => o.timestamp >= windowStart);
  if (relevant.length === 0) throw new Error('No observations in window');

  let weightedSum = 0n;
  let totalWeight = 0n;

  for (let i = 1; i < relevant.length; i++) {
    const duration = BigInt(relevant[i].timestamp - relevant[i - 1].timestamp);
    weightedSum += relevant[i - 1].price * duration;
    totalWeight += duration;
  }

  if (totalWeight === 0n) return relevant[0].price;
  return weightedSum / totalWeight;
};

const validatePriceDeviation = (spotPrice: bigint, twapPrice: bigint, maxDeviationBps: number): boolean => {
  const diff = spotPrice > twapPrice ? spotPrice - twapPrice : twapPrice - spotPrice;

  const maxDiff = (twapPrice * BigInt(maxDeviationBps)) / 10000n;
  return diff <= maxDiff;
};
