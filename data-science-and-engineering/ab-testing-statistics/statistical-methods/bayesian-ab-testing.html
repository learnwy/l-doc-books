贝叶斯 A/B 测试

核心概念

贝叶斯 A/B 测试从根本上改变了实验推断的框架：不再问"数据是否足以拒绝零假设"，而是直接回答"B 方案优于 A 的概率是多少"。这种直觉性的概率表述让产品决策更加自然。贝叶斯方法通过先验分布编码已有知识，结合数据更新为后验分布，实现了对参数不确定性的完整描述。

共轭先验与后验计算
在二项比率（转化率）的场景下，Beta 分布是二项分布的共轭先验，后验更新极其简洁。如果先验为 Beta(α, β)，观测到 s 次成功和 f 次失败后，后验为 Beta(α+s, β+f)。


import numpy as np
from scipy import stats
import matplotlib
matplotlib.use('Agg')

class BayesianABTest:
    def __init__(self, alpha_prior=1, beta_prior=1):
        self.alpha_prior = alpha_prior
        self.beta_prior = beta_prior

    def update(self, successes, failures):
        alpha_post = self.alpha_prior + successes
        beta_post = self.beta_prior + failures
        return alpha_post, beta_post

    def win_probability(self, alpha_a, beta_a, alpha_b, beta_b, n_samples=100000):
        samples_a = np.random.beta(alpha_a, beta_a, n_samples)
        samples_b = np.random.beta(alpha_b, beta_b, n_samples)
        return np.mean(samples_b > samples_a)

    def expected_loss(self, alpha_a, beta_a, alpha_b, beta_b, n_samples=100000):
        samples_a = np.random.beta(alpha_a, beta_a, n_samples)
        samples_b = np.random.beta(alpha_b, beta_b, n_samples)
        loss_choose_b = np.mean(np.maximum(samples_a - samples_b, 0))
        loss_choose_a = np.mean(np.maximum(samples_b - samples_a, 0))
        return loss_choose_a, loss_choose_b

    def credible_interval(self, alpha_post, beta_post, level=0.95):
        lower = stats.beta.ppf((1 - level) / 2, alpha_post, beta_post)
        upper = stats.beta.ppf(1 - (1 - level) / 2, alpha_post, beta_post)
        return lower, upper

test = BayesianABTest(alpha_prior=1, beta_prior=1)
alpha_a, beta_a = test.update(successes=2400, failures=47600)
alpha_b, beta_b = test.update(successes=2600, failures=47400)

prob_b_wins = test.win_probability(alpha_a, beta_a, alpha_b, beta_b)
loss_a, loss_b = test.expected_loss(alpha_a, beta_a, alpha_b, beta_b)
ci_a = test.credible_interval(alpha_a, beta_a)
ci_b = test.credible_interval(alpha_b, beta_b)

print(f"P(B > A) = {prob_b_wins:.4f}")
print(f"Expected loss (choose A): {loss_a:.6f}")
print(f"Expected loss (choose B): {loss_b:.6f}")
print(f"Control 95% CI: ({ci_a[0]:.4f}, {ci_a[1]:.4f})")
print(f"Treatment 95% CI: ({ci_b[0]:.4f}, {ci_b[1]:.4f})")
先验选择对结果有直接影响。弱信息先验 Beta(1,1) 等价于均匀分布，对所有转化率一视同仁。经验贝叶斯（Empirical Bayes）利用历史实验数据估计先验参数，在实验早期提供更合理的收缩效应，防止过度反应于少量数据。

连续指标的贝叶斯检验
对于人均消费、停留时长等连续指标，Normal-Normal 共轭或 Normal-InverseGamma 共轭模型是标准选择。当样本量较大时，可以直接使用正态近似。


import numpy as np
from scipy import stats

class BayesianContinuousTest:
    def __init__(self, mu_prior=0, sigma_prior=1000, known_sigma=None):
        self.mu_prior = mu_prior
        self.sigma_prior = sigma_prior
        self.known_sigma = known_sigma

    def normal_update(self, data):
        n = len(data)
        x_bar = np.mean(data)
        sigma = self.known_sigma if self.known_sigma else np.std(data, ddof=1)

        prior_precision = 1 / self.sigma_prior**2
        data_precision = n / sigma**2

        post_precision = prior_precision + data_precision
        post_mu = (prior_precision * self.mu_prior + data_precision * x_bar) / post_precision
        post_sigma = np.sqrt(1 / post_precision)

        return post_mu, post_sigma

    def diff_win_probability(self, mu_a, sigma_a, mu_b, sigma_b, n_samples=100000):
        samples_a = np.random.normal(mu_a, sigma_a, n_samples)
        samples_b = np.random.normal(mu_b, sigma_b, n_samples)
        return np.mean(samples_b > samples_a)

    def rope_probability(self, mu_a, sigma_a, mu_b, sigma_b, rope=(-0.5, 0.5), n_samples=100000):
        samples_a = np.random.normal(mu_a, sigma_a, n_samples)
        samples_b = np.random.normal(mu_b, sigma_b, n_samples)
        diff = samples_b - samples_a
        p_rope = np.mean((diff >= rope[0]) & (diff <= rope[1]))
        p_positive = np.mean(diff > rope[1])
        p_negative = np.mean(diff < rope[0])
        return p_negative, p_rope, p_positive

np.random.seed(42)
control_data = np.random.normal(50, 20, 10000)
treatment_data = np.random.normal(52, 20, 10000)

test = BayesianContinuousTest(mu_prior=50, sigma_prior=100)
mu_a, sigma_a = test.normal_update(control_data)
mu_b, sigma_b = test.normal_update(treatment_data)

prob = test.diff_win_probability(mu_a, sigma_a, mu_b, sigma_b)
neg, rope, pos = test.rope_probability(mu_a, sigma_a, mu_b, sigma_b, rope=(-0.5, 0.5))
print(f"P(B > A) = {prob:.4f}")
print(f"P(negative) = {neg:.4f}, P(ROPE) = {rope:.4f}, P(positive) = {pos:.4f}")
ROPE（Region of Practical Equivalence） 是贝叶斯框架中处理"实际等价性"的优雅方法。定义一个效应可忽略的区间（如 [-0.5%, +0.5%]），如果后验差异大部分落在 ROPE 内，则认为两组实际等价，即使点估计不为零。


决策规则与停止准则
贝叶斯 A/B 测试的一个核心优势是支持持续监控和灵活停止，不像频率派方法需要严格遵守预设样本量。但"随时查看"并不意味着"随时决策"——需要设计合理的停止规则来控制决策质量。


三种主要的决策准则：


胜出概率阈值：当 P(B > A) > 0.95 时宣布 B 获胜。简单直观，但在效应很小时可能需要很长时间才能达到阈值，且没有控制假阳性率。


期望损失阈值：当选择 B 的期望损失 < ε（如 0.01%）时停止。期望损失 = E[max(θ_A − θ_B, 0)]，表示选错的平均代价。这种方法更关注决策的经济后果而非统计证据。


import numpy as np

def bayesian_stopping_simulation(true_p_a, true_p_b, max_n=100000,
                                  batch_size=1000, win_threshold=0.95,
                                  loss_threshold=0.0001):
    alpha_a, beta_a = 1, 1
    alpha_b, beta_b = 1, 1

    history = []

    for day in range(1, max_n // batch_size + 1):
        successes_a = np.random.binomial(batch_size, true_p_a)
        successes_b = np.random.binomial(batch_size, true_p_b)

        alpha_a += successes_a
        beta_a += batch_size - successes_a
        alpha_b += successes_b
        beta_b += batch_size - successes_b

        samples_a = np.random.beta(alpha_a, beta_a, 50000)
        samples_b = np.random.beta(alpha_b, beta_b, 50000)

        prob_b_wins = np.mean(samples_b > samples_a)
        expected_loss_b = np.mean(np.maximum(samples_a - samples_b, 0))

        total_n = day * batch_size * 2
        history.append({
            'day': day,
            'total_n': total_n,
            'prob_b_wins': prob_b_wins,
            'expected_loss_b': expected_loss_b,
        })

        if prob_b_wins > win_threshold and expected_loss_b < loss_threshold:
            return history, 'B wins'
        if (1 - prob_b_wins) > win_threshold and \
           np.mean(np.maximum(samples_b - samples_a, 0)) < loss_threshold:
            return history, 'A wins'

    return history, 'Inconclusive'

np.random.seed(42)
history, decision = bayesian_stopping_simulation(0.048, 0.052)
print(f"Decision: {decision}")
print(f"Stopped at day {history[-1]['day']}, total N = {history[-1]['total_n']}")
print(f"Final P(B>A) = {history[-1]['prob_b_wins']:.4f}")

贝叶斯与频率派方法的对比
在实际应用中，贝叶斯方法和频率派方法各有适用场景。选择取决于团队的统计素养、实验平台的技术能力和业务需求。


在高频低风险的实验（如按钮颜色、文案优化）中，贝叶斯方法的灵活停止能力显著缩短实验周期。在高风险的核心功能变更中，频率派方法的严格错误率控制提供了更强的统计保证。

许多成熟的实验平台采用混合策略：使用频率派方法进行功效分析和样本量计算，使用贝叶斯方法进行持续监控和结果报告。


小结
    

  • 贝叶斯 A/B 测试直接回答"B 优于 A 的概率是多少"，解释性比 p 值更直观
    • 

  • Beta-Binomial 共轭使得比率型指标的后验更新极其高效，无需数值积分
    • 

  • 期望损失是比胜出概率更优的决策指标，因为它量化了做错决策的平均代价
    • 

  • ROPE 方法区分了"无效应"和"等价效应"，避免了频率派二分法的局限
    • 

  • 贝叶斯方法天然支持持续监控，但仍需设计停止规则以控制决策质量
    • 

  • 实际生产中推荐频率派和贝叶斯混合使用：频率派做功效分析，贝叶斯做持续监控
