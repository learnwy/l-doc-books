实验平台架构

核心概念

实验平台（Experimentation Platform）是支撑大规模 A/B 测试的基础设施，负责实验的全生命周期管理：从实验创建、流量分配、数据收集到统计分析和决策支持。一个成熟的实验平台需要同时满足工程可靠性（百万级 QPS 的流量分配）、统计严谨性（正确的假设检验和方差缩减）和产品易用性（非技术人员也能创建和分析实验）。

流量分配系统
流量分配是实验平台的核心引擎。在高并发场景下，需要在毫秒级别内确定用户属于哪个实验的哪个分组，同时保证分配的随机性、一致性和正交性。


一致性哈希分桶：使用用户 ID 和实验 Salt 的哈希值对桶数取模，确保同一用户在同一实验中始终被分配到相同的组。不同实验使用不同的 Salt，保证实验间的独立性。


import hashlib
import numpy as np

class TrafficAllocator:
    def __init__(self, n_buckets=10000):
        self.n_buckets = n_buckets
        self.experiments = {}

    def hash_user(self, user_id, salt):
        key = f"{user_id}_{salt}"
        hash_val = int(hashlib.md5(key.encode()).hexdigest(), 16)
        return hash_val % self.n_buckets

    def create_experiment(self, exp_id, layer, variants, traffic_pct=100):
        bucket_count = int(self.n_buckets * traffic_pct / 100)
        buckets_per_variant = bucket_count // len(variants)

        variant_ranges = {}
        start = 0
        for v in variants:
            variant_ranges[v] = (start, start + buckets_per_variant)
            start += buckets_per_variant

        self.experiments[exp_id] = {
            'layer': layer,
            'salt': f"{layer}_{exp_id}",
            'variants': variant_ranges,
            'traffic_pct': traffic_pct,
        }

    def get_variant(self, user_id, exp_id):
        exp = self.experiments[exp_id]
        bucket = self.hash_user(user_id, exp['salt'])

        for variant, (start, end) in exp['variants'].items():
            if start <= bucket < end:
                return variant
        return None

allocator = TrafficAllocator()
allocator.create_experiment('search_ranking_v2', layer='search',
                           variants=['control', 'treatment_a', 'treatment_b'])

results = {}
for uid in range(100000):
    variant = allocator.get_variant(str(uid), 'search_ranking_v2')
    results[variant] = results.get(variant, 0) + 1

for v, count in sorted(results.items()):
    print(f"{v}: {count} ({count/1000:.1f}%)")
分层正交流量：将全量流量划分为多个正交层（layer），每一层独立分桶。同一层内的实验互斥（用户只能参与一个），不同层的实验正交（用户可同时参与多个）。这解决了在同一用户群上并行运行多个实验的需求。


class OrthogonalLayerSystem:
    def __init__(self, n_buckets=10000):
        self.n_buckets = n_buckets
        self.layers = {}

    def create_layer(self, layer_id):
        self.layers[layer_id] = {
            'experiments': {},
            'allocated_buckets': set(),
        }

    def add_experiment(self, layer_id, exp_id, traffic_pct, n_variants):
        layer = self.layers[layer_id]
        n_buckets = int(self.n_buckets * traffic_pct / 100)

        available = set(range(self.n_buckets)) - layer['allocated_buckets']
        if len(available) < n_buckets:
            raise ValueError(f"Not enough buckets in layer {layer_id}")

        exp_buckets = sorted(available)[:n_buckets]
        layer['allocated_buckets'].update(exp_buckets)

        buckets_per_variant = n_buckets // n_variants
        variants = {}
        for i in range(n_variants):
            start = i * buckets_per_variant
            end = start + buckets_per_variant
            variants[f"variant_{i}"] = set(exp_buckets[start:end])

        layer['experiments'][exp_id] = {
            'variants': variants,
            'salt': f"{layer_id}_{exp_id}",
        }

    def get_all_variants(self, user_id):
        assignments = {}
        for layer_id, layer in self.layers.items():
            for exp_id, exp in layer['experiments'].items():
                key = f"{user_id}_{exp['salt']}"
                bucket = int(hashlib.md5(key.encode()).hexdigest(), 16) % self.n_buckets
                for variant_name, bucket_set in exp['variants'].items():
                    if bucket in bucket_set:
                        assignments[exp_id] = variant_name
                        break
        return assignments

system = OrthogonalLayerSystem()
system.create_layer('search')
system.create_layer('recommendation')
system.add_experiment('search', 'ranking_v2', traffic_pct=50, n_variants=2)
system.add_experiment('recommendation', 'recall_strategy', traffic_pct=30, n_variants=3)

指标体系与数据管道
实验平台的价值最终体现在指标分析上。一个完整的指标体系包含北极星指标、主要指标、辅助指标和护栏指标四个层次。


from dataclasses import dataclass, field
from enum import Enum

class MetricType(Enum):
    NORTH_STAR = "north_star"
    PRIMARY = "primary"
    SECONDARY = "secondary"
    GUARDRAIL = "guardrail"

class MetricAggregation(Enum):
    MEAN = "mean"
    SUM = "sum"
    RATIO = "ratio"
    PERCENTILE = "percentile"
    COUNT_DISTINCT = "count_distinct"

@dataclass
class MetricDefinition:
    name: str
    metric_type: MetricType
    aggregation: MetricAggregation
    numerator_event: str
    denominator_event: str = ""
    direction: str = "increase"
    mde: float = 0.0

    def compute(self, events):
        if self.aggregation == MetricAggregation.RATIO:
            num = events[events['event'] == self.numerator_event].shape[0]
            den = events[events['event'] == self.denominator_event].shape[0]
            return num / den if den > 0 else 0
        elif self.aggregation == MetricAggregation.MEAN:
            relevant = events[events['event'] == self.numerator_event]
            return relevant['value'].mean() if len(relevant) > 0 else 0
        return 0

@dataclass
class ExperimentMetrics:
    metrics: list = field(default_factory=list)

    def add_metric(self, metric):
        self.metrics.append(metric)

    def validate_guardrails(self, results):
        guardrails = [m for m in self.metrics if m.metric_type == MetricType.GUARDRAIL]
        violations = []
        for g in guardrails:
            if g.direction == "increase" and results.get(g.name, {}).get('diff', 0) < -g.mde:
                violations.append(g.name)
            elif g.direction == "decrease" and results.get(g.name, {}).get('diff', 0) > g.mde:
                violations.append(g.name)
        return violations

experiment_metrics = ExperimentMetrics()
experiment_metrics.add_metric(MetricDefinition(
    name="conversion_rate", metric_type=MetricType.PRIMARY,
    aggregation=MetricAggregation.RATIO,
    numerator_event="purchase", denominator_event="page_view",
    mde=0.005
))
experiment_metrics.add_metric(MetricDefinition(
    name="page_load_p99", metric_type=MetricType.GUARDRAIL,
    aggregation=MetricAggregation.PERCENTILE,
    numerator_event="page_load", direction="decrease",
    mde=0.5
))

自动化分析与报告
成熟的实验平台将统计分析自动化，减少人工计算错误和选择性报告偏差。自动化分析管线包含数据质量检查、统计检验、多重校正和可视化报告。


import numpy as np
from scipy import stats
from dataclasses import dataclass

@dataclass
class ExperimentResult:
    metric_name: str
    control_mean: float
    treatment_mean: float
    absolute_diff: float
    relative_diff: float
    ci_lower: float
    ci_upper: float
    p_value: float
    significant: bool
    sample_size_control: int
    sample_size_treatment: int

class AutomatedAnalyzer:
    def __init__(self, alpha=0.05, use_cuped=True, sequential=True):
        self.alpha = alpha
        self.use_cuped = use_cuped
        self.sequential = sequential

    def run_health_checks(self, data):
        checks = {}

        n_control = len(data[data['treatment'] == 0])
        n_treatment = len(data[data['treatment'] == 1])
        expected_ratio = 0.5
        chi2, p_srm = stats.chisquare([n_control, n_treatment])
        checks['srm'] = {'pass': p_srm > 0.001, 'p_value': p_srm, 'ratio': n_treatment/(n_control+n_treatment)}

        pre_metric = data.groupby('treatment')['pre_metric'].mean()
        t_stat, p_aa = stats.ttest_ind(
            data[data['treatment']==0]['pre_metric'],
            data[data['treatment']==1]['pre_metric']
        )
        checks['aa_test'] = {'pass': p_aa > 0.05, 'p_value': p_aa}

        return checks

    def analyze_metric(self, control, treatment, metric_name, pre_control=None, pre_treatment=None):
        if self.use_cuped and pre_control is not None:
            control, treatment = self._apply_cuped(control, treatment, pre_control, pre_treatment)

        diff = np.mean(treatment) - np.mean(control)
        se = np.sqrt(np.var(treatment, ddof=1)/len(treatment) + np.var(control, ddof=1)/len(control))
        z = diff / se if se > 0 else 0
        p_val = 2 * (1 - stats.norm.cdf(abs(z)))

        ci_lower = diff - 1.96 * se
        ci_upper = diff + 1.96 * se

        control_mean = np.mean(control)
        relative = diff / control_mean if control_mean != 0 else 0

        return ExperimentResult(
            metric_name=metric_name, control_mean=control_mean,
            treatment_mean=np.mean(treatment), absolute_diff=diff,
            relative_diff=relative, ci_lower=ci_lower, ci_upper=ci_upper,
            p_value=p_val, significant=p_val < self.alpha,
            sample_size_control=len(control), sample_size_treatment=len(treatment),
        )

    def _apply_cuped(self, y_ctrl, y_trt, x_ctrl, x_trt):
        y = np.concatenate([y_ctrl, y_trt])
        x = np.concatenate([x_ctrl, x_trt])
        theta = np.cov(y, x)[0, 1] / np.var(x) if np.var(x) > 0 else 0
        y_ctrl_adj = y_ctrl - theta * (x_ctrl - np.mean(x))
        y_trt_adj = y_trt - theta * (x_trt - np.mean(x))
        return y_ctrl_adj, y_trt_adj

analyzer = AutomatedAnalyzer(use_cuped=True)

np.random.seed(42)
n = 50000
pre = np.random.normal(100, 30, 2*n)
post_ctrl = pre[:n] * 0.6 + np.random.normal(0, 20, n)
post_trt = pre[n:] * 0.6 + np.random.normal(0, 20, n) + 1.5

result = analyzer.analyze_metric(
    post_ctrl, post_trt, 'revenue_per_user',
    pre_control=pre[:n], pre_treatment=pre[n:]
)
print(f"Metric: {result.metric_name}")
print(f"Diff: {result.absolute_diff:.3f} (relative: {result.relative_diff:.2%})")
print(f"95% CI: ({result.ci_lower:.3f}, {result.ci_upper:.3f})")
print(f"p-value: {result.p_value:.6f}, significant: {result.significant}")

小结
    

  • 实验平台架构分为配置、分配、采集、处理、分析和展示六个层次，每层有独立的技术挑战
    • 

  • 一致性哈希分桶确保用户在同一实验中被稳定分配，分层正交机制支持多实验并行
    • 

  • 指标体系应预先定义北极星、主要、辅助和护栏四类指标，避免事后挖掘
    • 

  • 数据管道的关键挑战是曝光与结果的精确关联和延迟转化的归因
    • 

  • 自动化分析包含 SRM 检测、A/A 检验、CUPED 方差缩减和多重校正，减少人为偏差
    • 

  • 生产级平台需要支持百万 QPS 的流量分配和天级刷新的统计分析
