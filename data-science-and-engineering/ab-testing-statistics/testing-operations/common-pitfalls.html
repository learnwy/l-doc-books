常见陷阱

核心概念

A/B 测试看似简单——分两组、比指标、看 p 值——但实际上充满了容易犯的统计和工程错误。这些陷阱可能导致假阳性泛滥、真实效应被掩盖、甚至做出方向性错误的产品决策。本文系统梳理 A/B 测试中最常见的陷阱，分析其根因和危害，并给出可操作的解决方案。

过早停止与持续窥探
过早停止（Peeking Problem）是最经典也最常见的 A/B 测试陷阱。实验运营者在实验进行中反复查看 p 值，一旦 p < 0.05 就宣布结论。问题在于：如果在固定样本量检验框架下多次查看，每次查看都有一定概率偶然达到显著性阈值，导致整体假阳性率远超 5%。


import numpy as np
from scipy import stats

def simulate_peeking(n_simulations=10000, max_n=10000, check_interval=100, alpha=0.05):
    false_positives = 0

    for _ in range(n_simulations):
        control = np.random.normal(0, 1, max_n)
        treatment = np.random.normal(0, 1, max_n)

        for n in range(check_interval, max_n + 1, check_interval):
            t_stat, p_value = stats.ttest_ind(control[:n], treatment[:n])
            if p_value < alpha:
                false_positives += 1
                break

    return false_positives / n_simulations

fp_no_peek = simulate_peeking(n_simulations=5000, max_n=5000, check_interval=5000)
fp_daily_peek = simulate_peeking(n_simulations=5000, max_n=5000, check_interval=500)
fp_hourly_peek = simulate_peeking(n_simulations=5000, max_n=5000, check_interval=100)

print(f"No peeking (1 check): {fp_no_peek:.1%}")
print(f"Daily peek (10 checks): {fp_daily_peek:.1%}")
print(f"Hourly peek (50 checks): {fp_hourly_peek:.1%}")

样本比率不匹配
样本比率不匹配（Sample Ratio Mismatch, SRM）是指实验组和对照组的实际样本量比例与预期比例显著不同。例如预期 50:50 分流，实际观测到 48:52。SRM 通常暗示实验实现中存在系统性错误，可能导致有偏的效应估计。


import numpy as np
from scipy import stats

def check_srm(n_control, n_treatment, expected_ratio=0.5, threshold=0.001):
    total = n_control + n_treatment
    observed = [n_control, n_treatment]
    expected = [total * expected_ratio, total * (1 - expected_ratio)]

    chi2, p_value = stats.chisquare(observed, expected)

    actual_ratio = n_treatment / total

    return {
        'chi2': chi2,
        'p_value': p_value,
        'actual_ratio': actual_ratio,
        'expected_ratio': expected_ratio,
        'srm_detected': p_value < threshold,
        'severity': 'critical' if p_value < 0.0001 else 'warning' if p_value < threshold else 'ok',
    }

result = check_srm(n_control=498213, n_treatment=501787)
print(f"SRM p-value: {result['p_value']:.6f}")
print(f"Detected: {result['srm_detected']}, Severity: {result['severity']}")

result2 = check_srm(n_control=475000, n_treatment=525000)
print(f"\nSRM p-value: {result2['p_value']:.10f}")
print(f"Detected: {result2['srm_detected']}, Severity: {result2['severity']}")

关键原则：一旦检测到 SRM，实验结果不可信，必须先排查并修复 SRM 的根因，然后重新实验。任何试图"校正" SRM 偏差的统计方法都不可靠，因为导致 SRM 的同一个 Bug 很可能也影响了结果指标。


辛普森悖论与混杂因素
辛普森悖论是指在每个子群中实验组都优于对照组，但合并后反而对照组更优（或反之）。根因是子群之间的组成比例不同，且组成比例与结果相关。


import pandas as pd
import numpy as np

def demonstrate_simpson():
    data = {
        'segment': ['mobile'] * 4 + ['desktop'] * 4,
        'group': ['control', 'treatment'] * 4,
        'users': [80000, 20000, 20000, 80000],
        'conversions': [2400, 700, 1000, 5600],
    }
    df = pd.DataFrame(data)
    df['rate'] = df['conversions'] / df['users']

    print("Per-segment analysis:")
    for seg in ['mobile', 'desktop']:
        seg_data = df[df['segment'] == seg]
        ctrl_rate = seg_data[seg_data['group'] == 'control']['rate'].values[0]
        trt_rate = seg_data[seg_data['group'] == 'treatment']['rate'].values[0]
        print(f"  {seg}: control={ctrl_rate:.2%}, treatment={trt_rate:.2%}, "
              f"diff={trt_rate - ctrl_rate:+.2%}")

    overall = df.groupby('group').agg({'conversions': 'sum', 'users': 'sum'})
    overall['rate'] = overall['conversions'] / overall['users']
    print(f"\nOverall: control={overall.loc['control', 'rate']:.2%}, "
          f"treatment={overall.loc['treatment', 'rate']:.2%}")
    print(f"Overall diff: {overall.loc['treatment', 'rate'] - overall.loc['control', 'rate']:+.2%}")

demonstrate_simpson()
在上面的例子中，移动端和桌面端分别看，实验组都比对照组转化率高。但由于实验组中桌面端（高转化率平台）用户比例更高，合并后的"优势"实际上是混杂因素（平台分布不均）导致的假象。


预防辛普森悖论的方法：


    

  • 检查随机化平衡：在实验开始时检查关键维度（设备、地区、新老用户）的分布是否平衡
    • 

  • 分层分析：按关键维度分别分析实验效应，检查方向是否一致
    • 

  • 分层随机化：在随机化阶段按关键维度分层，确保每层内的实验/对照比例一致
    •

新奇效应与首因偏差
新奇效应（Novelty Effect）是指用户对新功能的好奇心驱动的短期行为变化，这种变化会随时间衰减。反之，首因偏差（Primacy Effect）是指用户对变化的抵触导致的短期负面反应，长期适应后消失。两者都会使短期实验结果偏离长期真实效应。


import numpy as np

def simulate_novelty_effect(n_days=30, novelty_peak=0.03, decay_rate=0.15,
                            true_effect=0.01, n_users_per_day=10000):
    np.random.seed(42)

    daily_results = []
    for day in range(n_days):
        novelty = novelty_peak * np.exp(-decay_rate * day)
        observed_effect = true_effect + novelty

        control_rate = 0.05
        treatment_rate = control_rate + observed_effect

        control = np.random.binomial(n_users_per_day, control_rate)
        treatment = np.random.binomial(n_users_per_day, treatment_rate)

        daily_results.append({
            'day': day + 1,
            'novelty_component': novelty,
            'true_effect': true_effect,
            'observed_effect': treatment/n_users_per_day - control/n_users_per_day,
        })

    week1_effect = np.mean([r['observed_effect'] for r in daily_results[:7]])
    week4_effect = np.mean([r['observed_effect'] for r in daily_results[21:]])

    print(f"Week 1 observed effect: {week1_effect:.4f}")
    print(f"Week 4 observed effect: {week4_effect:.4f}")
    print(f"True long-term effect: {true_effect:.4f}")
    print(f"Week 1 overestimate: {(week1_effect - true_effect) / true_effect:.0%}")

    return daily_results

results = simulate_novelty_effect()

指标定义与计算陷阱
指标定义的微小差异可能导致实验结论的根本改变。常见的指标陷阱包括分母选择、极端值处理、以及比率型指标的 Delta Method 近似误差。


import numpy as np
from scipy import stats

def demonstrate_metric_pitfalls():
    np.random.seed(42)
    n = 100000

    revenue_ctrl = np.concatenate([
        np.zeros(int(n * 0.95)),
        np.random.lognormal(mean=3, sigma=1.5, size=int(n * 0.05))
    ])
    revenue_trt = np.concatenate([
        np.zeros(int(n * 0.945)),
        np.random.lognormal(mean=3, sigma=1.5, size=int(n * 0.055))
    ])

    arpu_ctrl = np.mean(revenue_ctrl)
    arpu_trt = np.mean(revenue_trt)
    print(f"ARPU (all users): ctrl={arpu_ctrl:.2f}, trt={arpu_trt:.2f}, diff={arpu_trt-arpu_ctrl:.2f}")

    arppu_ctrl = np.mean(revenue_ctrl[revenue_ctrl > 0])
    arppu_trt = np.mean(revenue_trt[revenue_trt > 0])
    print(f"ARPPU (payers only): ctrl={arppu_ctrl:.2f}, trt={arppu_trt:.2f}, diff={arppu_trt-arppu_ctrl:.2f}")

    revenue_ctrl_trimmed = np.clip(revenue_ctrl, 0, np.percentile(revenue_ctrl[revenue_ctrl>0], 99))
    revenue_trt_trimmed = np.clip(revenue_trt, 0, np.percentile(revenue_trt[revenue_trt>0], 99))

    arpu_ctrl_trim = np.mean(revenue_ctrl_trimmed)
    arpu_trt_trim = np.mean(revenue_trt_trimmed)
    print(f"ARPU (trimmed 99%): ctrl={arpu_ctrl_trim:.2f}, trt={arpu_trt_trim:.2f}, "
          f"diff={arpu_trt_trim-arpu_ctrl_trim:.2f}")

demonstrate_metric_pitfalls()

def demonstrate_outlier_impact():
    np.random.seed(42)
    n = 50000

    ctrl = np.random.normal(100, 30, n)
    trt = np.random.normal(101, 30, n)

    t_clean, p_clean = stats.ttest_ind(ctrl, trt)

    ctrl_dirty = ctrl.copy()
    ctrl_dirty[0] = 100000
    t_dirty, p_dirty = stats.ttest_ind(ctrl_dirty, trt)

    print(f"\nClean data: t={t_clean:.3f}, p={p_clean:.4f}")
    print(f"With 1 outlier: t={t_dirty:.3f}, p={p_dirty:.4f}")

demonstrate_outlier_impact()

小结
    

  • 过早停止是最常见的陷阱，在固定样本量框架下频繁查看可使假阳性率从 5% 膨胀至 30%
    • 

  • SRM 暗示实验实现存在系统性错误，检测到 SRM 时实验结果不可信，必须排查根因后重做
    • 

  • 辛普森悖论来自子群组成比例不均，分层随机化和分层分析是有效的预防和检测手段
    • 

  • 新奇效应导致短期实验高估真实效应，实验应至少运行 2-4 周并进行时间分段分析
    • 

  • 指标定义的微小差异（分母选择、极端值处理、归因窗口）可能根本改变实验结论
    • 

  • 防范陷阱的最佳实践是：预注册假设和指标、自动化 SRM 和护栏检测、强制实验审查制度
