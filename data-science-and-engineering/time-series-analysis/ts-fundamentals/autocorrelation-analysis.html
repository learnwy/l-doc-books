自相关分析

核心概念

自相关分析（Autocorrelation Analysis）是时间序列建模的核心诊断工具，它量化序列与自身不同滞后版本之间的线性依赖关系。自相关函数（ACF）和偏自相关函数（PACF）的图形模式直接指导 ARIMA 模型的阶数选择：ACF 的截尾行为对应 MA 阶数，PACF 的截尾行为对应 AR 阶数。掌握 ACF 和 PACF 的解读是时间序列分析的基本功。

自相关分析不仅用于模型识别，还是模型诊断的重要工具。拟合模型后的残差应当表现为白噪声——ACF 和 PACF 的所有滞后值都应落在置信区间内。如果残差的 ACF/PACF 显示显著的自相关，说明模型未能充分捕获序列中的信息，需要调整模型阶数或结构。


自相关函数与偏自相关函数
自相关函数（ACF）衡量序列在不同滞后阶数上的线性相关性。滞后 kkk 的 ACF 定义为：


ρk=γkγ0=Cov(Xt,Xt−k)Var(Xt)\rho_k = \frac{\gamma_k}{\gamma_0} = \frac{\text{Cov}(X_t, X_{t-k})}{\text{Var}(X_t)}ρk​=γ0​γk​​=Var(Xt​)Cov(Xt​,Xt−k​)​
ACF 同时包含了直接相关和间接相关（通过中间滞后传递的相关性）。偏自相关函数（PACF）去除了中间滞后的影响，只衡量 XtX_tXt​ 和 Xt−kX_{t-k}Xt−k​ 之间的"净"相关性。PACF 在数学上等价于在控制了 Xt−1,Xt−2,…,Xt−k+1X_{t-1}, X_{t-2}, \ldots, X_{t-k+1}Xt−1​,Xt−2​,…,Xt−k+1​ 后 XtX_tXt​ 和 Xt−kX_{t-k}Xt−k​ 的条件相关性。


from statsmodels.tsa.stattools import acf, pacf
from statsmodels.stats.diagnostic import acorr_ljungbox
import numpy as np
import pandas as pd

def compute_acf_pacf(series, nlags=40, alpha=0.05):
    acf_values, acf_ci = acf(series, nlags=nlags, alpha=alpha)
    pacf_values, pacf_ci = pacf(series, nlags=nlags, alpha=alpha)

    n = len(series)
    confidence_bound = 1.96 / np.sqrt(n)

    significant_acf = [k for k in range(1, nlags + 1)
                       if abs(acf_values[k]) > confidence_bound]
    significant_pacf = [k for k in range(1, nlags + 1)
                        if abs(pacf_values[k]) > confidence_bound]

    return {
        'acf_values': acf_values,
        'pacf_values': pacf_values,
        'acf_ci': acf_ci,
        'pacf_ci': pacf_ci,
        'confidence_bound': confidence_bound,
        'significant_acf_lags': significant_acf,
        'significant_pacf_lags': significant_pacf
    }

def identify_arma_order(acf_result, max_order=5):
    acf_vals = acf_result['acf_values']
    pacf_vals = acf_result['pacf_values']
    cb = acf_result['confidence_bound']

    p_candidate = 0
    for k in range(1, max_order + 1):
        if abs(pacf_vals[k]) > cb:
            p_candidate = k
        else:
            break

    q_candidate = 0
    for k in range(1, max_order + 1):
        if abs(acf_vals[k]) > cb:
            q_candidate = k
        else:
            break

    return {
        'suggested_p': p_candidate,
        'suggested_q': q_candidate,
        'model': f'ARMA({p_candidate},{q_candidate})'
    }
95% 置信区间的近似边界为 ±1.96/n\pm 1.96 / \sqrt{n}±1.96/n​（nnn 为序列长度），超出此边界的 ACF/PACF 值被认为是统计显著的。这个近似基于白噪声序列的 ACF 渐近服从 N(0,1/n)N(0, 1/n)N(0,1/n) 的假设。对于非白噪声序列（如 MA(q) 过程），Bartlett 公式提供了更精确的 ACF 标准误估计。


Ljung-Box 白噪声检验
Ljung-Box 检验是判断序列是否为白噪声的联合检验。与逐个滞后检验 ACF 是否显著不同，Ljung-Box 检验同时考虑前 mmm 个滞后的 ACF 是否联合为零。检验统计量为：


Q(m)=n(n+2)∑k=1mρ^k2n−kQ(m) = n(n+2) \sum_{k=1}^{m} \frac{\hat{\rho}_k^2}{n-k}Q(m)=n(n+2)k=1∑m​n−kρ^​k2​​
在白噪声假设下，Q(m)Q(m)Q(m) 近似服从自由度为 mmm 的卡方分布。


def ljung_box_test(series, lags=None, significance=0.05):
    if lags is None:
        n = len(series)
        lags = min(20, n // 5)

    lb_result = acorr_ljungbox(series, lags=lags, return_df=True)

    any_significant = (lb_result['lb_pvalue'] < significance).any()
    first_significant = None
    for lag, row in lb_result.iterrows():
        if row['lb_pvalue'] < significance:
            first_significant = lag
            break

    return {
        'test_results': lb_result,
        'is_white_noise': not any_significant,
        'first_significant_lag': first_significant,
        'conclusion': 'White noise' if not any_significant else 'Autocorrelated'
    }

def residual_diagnostics(residuals, lags=20, significance=0.05):
    acf_result = compute_acf_pacf(residuals, nlags=lags)
    lb_result = ljung_box_test(residuals, lags=lags)

    from scipy.stats import jarque_bera, shapiro
    jb_stat, jb_pvalue = jarque_bera(residuals.dropna())

    mean_test = abs(np.mean(residuals)) < 2 * np.std(residuals) / np.sqrt(len(residuals))

    return {
        'ljung_box': lb_result,
        'significant_acf': acf_result['significant_acf_lags'],
        'significant_pacf': acf_result['significant_pacf_lags'],
        'normality': {
            'jarque_bera_stat': jb_stat,
            'jarque_bera_pvalue': jb_pvalue,
            'is_normal': jb_pvalue > significance
        },
        'zero_mean': mean_test,
        'residual_mean': float(np.mean(residuals)),
        'residual_std': float(np.std(residuals)),
        'model_adequate': lb_result['is_white_noise'] and mean_test
    }
Ljung-Box 检验的滞后阶数 mmm 的选择影响检验的功效。mmm 太小可能漏检高阶自相关，mmm 太大会降低检验功效。经验法则是取 m=min⁡(20,n/5)m = \min(20, n/5)m=min(20,n/5)。在 ARIMA 模型诊断中，检验残差时需要调整自由度：Q(m)Q(m)Q(m) 的自由度变为 m−p−qm - p - qm−p−q（ppp 和 qqq 为 ARIMA 模型的 AR 和 MA 阶数）。

交叉相关分析
交叉相关函数（CCF）衡量两个时间序列在不同滞后阶数上的线性相关性，是分析序列间领先-滞后关系（Lead-Lag Relationship）的基本工具。CCF 在因果分析和预测模型的外生变量选择中发挥重要作用。


from statsmodels.tsa.stattools import ccf as compute_ccf

def cross_correlation_analysis(series_x, series_y, max_lags=20):
    ccf_values = compute_ccf(series_x, series_y, adjusted=True)

    n = min(len(series_x), len(series_y))
    confidence_bound = 1.96 / np.sqrt(n)

    ccf_df = pd.DataFrame({
        'lag': range(-max_lags, max_lags + 1),
        'ccf': ccf_values[len(ccf_values)//2 - max_lags: len(ccf_values)//2 + max_lags + 1]
    })

    ccf_df['significant'] = ccf_df['ccf'].abs() > confidence_bound

    peak_lag = ccf_df.loc[ccf_df['ccf'].abs().idxmax(), 'lag']
    peak_ccf = ccf_df.loc[ccf_df['ccf'].abs().idxmax(), 'ccf']

    return {
        'ccf_values': ccf_df,
        'confidence_bound': confidence_bound,
        'peak_lag': int(peak_lag),
        'peak_ccf': float(peak_ccf),
        'lead_lag_interpretation': (
            f'X leads Y by {abs(peak_lag)} periods' if peak_lag < 0
            else f'Y leads X by {peak_lag} periods' if peak_lag > 0
            else 'Contemporaneous correlation'
        )
    }

def granger_causality_test(series_x, series_y, max_lag=10, significance=0.05):
    from statsmodels.tsa.stattools import grangercausalitytests

    data = pd.DataFrame({'y': series_y, 'x': series_x}).dropna()

    results = {}
    for lag in range(1, max_lag + 1):
        try:
            gc_result = grangercausalitytests(data[['y', 'x']], maxlag=[lag], verbose=False)
            f_stat = gc_result[lag][0]['ssr_ftest'][0]
            p_value = gc_result[lag][0]['ssr_ftest'][1]
            results[lag] = {'f_statistic': f_stat, 'p_value': p_value,
                           'significant': p_value < significance}
        except Exception:
            results[lag] = {'f_statistic': None, 'p_value': None, 'significant': False}

    return results
Granger 因果检验是 CCF 分析的统计推断版本，它检验一个序列的历史值是否能够帮助预测另一个序列。如果在包含 XXX 的历史值后模型对 YYY 的预测误差显著减小，则称 XXX Granger 因果于 YYY。需要注意的是，Granger 因果是统计意义上的预测关系，不等于真实因果关系。


频域分析
频域分析（Frequency Domain Analysis）从另一个视角理解时间序列的依赖结构。时域中的自相关函数和频域中的功率谱密度是 Fourier 变换对——它们包含了关于序列依赖结构的完全等价的信息，但频域表示更适合识别周期性成分。


from scipy.signal import periodogram, welch

def spectral_analysis(series, sampling_freq=1.0):
    frequencies, psd_periodogram = periodogram(series, fs=sampling_freq)
    frequencies_welch, psd_welch = welch(series, fs=sampling_freq, nperseg=min(256, len(series)//4))

    top_k = 5
    top_indices = np.argsort(psd_periodogram[1:])[-top_k:] + 1
    dominant_frequencies = frequencies[top_indices]
    dominant_periods = 1.0 / dominant_frequencies
    dominant_powers = psd_periodogram[top_indices]

    return {
        'frequencies': frequencies,
        'periodogram_psd': psd_periodogram,
        'welch_psd': psd_welch,
        'welch_frequencies': frequencies_welch,
        'dominant_periods': sorted(dominant_periods),
        'dominant_powers': dominant_powers[np.argsort(dominant_periods)],
        'total_power': np.sum(psd_periodogram)
    }
功率谱密度（Power Spectral Density, PSD）描述了序列的方差在不同频率上的分布。PSD 中的尖峰对应序列中的主要周期成分。周期图（Periodogram）是 PSD 的最简单估计器，但它的方差很大。Welch 方法通过分段平均减小估计方差，是更常用的 PSD 估计方法。

小结
    

  • ACF 衡量序列与自身滞后版本的总体相关性，PACF 去除中间滞后的间接效应，衡量净相关性
    • 

  • ACF 截尾指示 MA 阶数，PACF 截尾指示 AR 阶数，两者结合指导 ARIMA 模型阶数选择
    • 

  • 95% 置信区间 ±1.96/n\pm 1.96/\sqrt{n}±1.96/n​ 是判断 ACF/PACF 是否显著的标准阈值
    • 

  • Ljung-Box 检验联合检验前 m 个滞后的 ACF 是否同时为零，是白噪声检验和模型诊断的核心工具
    • 

  • 交叉相关函数揭示两个序列之间的领先-滞后关系，Granger 因果检验提供统计推断
    • 

  • 频域分析通过功率谱密度识别主要周期成分，Welch 方法比原始周期图更稳健
    • 

  • 模型诊断的标准流程：拟合模型 → 提取残差 → ACF/PACF 检查 → Ljung-Box 检验 → 正态性检验
    •
