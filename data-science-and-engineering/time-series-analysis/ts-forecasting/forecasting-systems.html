预测系统工程

核心概念

预测系统工程（Forecasting System Engineering）关注的是如何将时间序列预测模型从研究环境部署到生产环境，并持续可靠地运行。一个工业级的预测系统不仅仅是一个模型，而是包含数据管道、特征计算、模型训练、预测生成、预测监控和反馈循环的完整闭环系统。系统的设计需要考虑可扩展性（同时预测数万条序列）、自动化（最小化人工干预）和鲁棒性（优雅处理异常数据和模型失效）。

预测系统的核心设计原则是"自动化优先，人工兜底"——系统应当能够自动处理绝大多数情况，只在异常场景下升级给人工干预。这要求系统具备自动模型选择、自动异常检测和自动降级三个自动化能力。


大规模预测架构
当需要同时预测数千到数百万条时间序列时（如零售商的 SKU 级销量预测），系统架构面临吞吐量和延迟的双重挑战。常见的架构模式是"分组建模"——将具有相似模式的序列分组，每组使用共享的模型参数或模型选择策略。


import pandas as pd
import numpy as np
from concurrent.futures import ProcessPoolExecutor
from typing import Dict, List, Callable

class ScalableForecaster:
    def __init__(self, model_pool, selection_criterion='aic'):
        self.model_pool = model_pool
        self.selection_criterion = selection_criterion
        self.fitted_models: Dict[str, dict] = {}

    def forecast_single(self, series_id: str, series: pd.Series,
                          horizon: int) -> dict:
        best_score = np.inf
        best_model = None
        best_forecast = None

        for model_name, model_fn in self.model_pool.items():
            try:
                model, forecast, metrics = model_fn(series, horizon)
                score = metrics.get(self.selection_criterion, np.inf)
                if score < best_score:
                    best_score = score
                    best_model = model_name
                    best_forecast = forecast
            except Exception:
                continue

        if best_forecast is None:
            best_forecast = np.full(horizon, series.mean())
            best_model = 'fallback_mean'

        self.fitted_models[series_id] = {
            'model': best_model,
            'score': best_score
        }

        return {
            'series_id': series_id,
            'model': best_model,
            'forecast': best_forecast,
            'score': best_score
        }

    def forecast_batch(self, series_dict: Dict[str, pd.Series],
                        horizon: int, n_workers: int = 4) -> List[dict]:
        results = []

        with ProcessPoolExecutor(max_workers=n_workers) as executor:
            futures = {}
            for sid, series in series_dict.items():
                future = executor.submit(
                    self.forecast_single, sid, series, horizon
                )
                futures[future] = sid

            for future in futures:
                try:
                    result = future.result(timeout=60)
                    results.append(result)
                except Exception as e:
                    sid = futures[future]
                    results.append({
                        'series_id': sid,
                        'model': 'timeout_fallback',
                        'forecast': np.full(horizon, 0),
                        'error': str(e)
                    })

        return results
全局模型（Global Model）是大规模预测的另一种范式。与为每条序列独立建模不同，全局模型使用所有序列的数据共同训练一个模型（如 LightGBM 或深度学习模型），通过序列标识符和序列特征区分不同序列。全局模型的优势在于能够从跨序列的模式中学习（如所有商品的节假日效应），且在短序列和新序列（冷启动）场景下表现更好。

预测监控与漂移检测
预测系统上线后必须持续监控预测质量。预测性能可能因数据分布漂移、外部事件冲击或上游数据质量问题而退化。有效的监控系统需要能够快速检测性能退化并触发应对措施。


from collections import deque
from datetime import datetime

class ForecastMonitor:
    def __init__(self, window_size=30, alert_threshold=1.5):
        self.window_size = window_size
        self.alert_threshold = alert_threshold
        self.error_history: Dict[str, deque] = {}
        self.baseline_mase: Dict[str, float] = {}

    def set_baseline(self, series_id: str, baseline_mase: float):
        self.baseline_mase[series_id] = baseline_mase
        self.error_history[series_id] = deque(maxlen=self.window_size)

    def record_error(self, series_id: str, actual: float,
                      predicted: float, naive: float):
        ae = abs(actual - predicted)
        naive_ae = abs(actual - naive)
        mase = ae / max(naive_ae, 1e-8)

        if series_id not in self.error_history:
            self.error_history[series_id] = deque(maxlen=self.window_size)
        self.error_history[series_id].append(mase)

        return self._check_alert(series_id)

    def _check_alert(self, series_id: str) -> dict:
        history = self.error_history.get(series_id)
        baseline = self.baseline_mase.get(series_id, 1.0)

        if not history or len(history) < 5:
            return {'status': 'insufficient_data'}

        current_mase = np.mean(list(history)[-7:])
        ratio = current_mase / max(baseline, 1e-8)

        if ratio > self.alert_threshold * 2:
            level = 'critical'
        elif ratio > self.alert_threshold:
            level = 'warning'
        else:
            level = 'normal'

        return {
            'series_id': series_id,
            'current_mase': current_mase,
            'baseline_mase': baseline,
            'degradation_ratio': ratio,
            'alert_level': level,
            'timestamp': datetime.now()
        }

    def get_system_health(self) -> dict:
        alerts = {}
        for sid in self.error_history:
            result = self._check_alert(sid)
            if result.get('alert_level') in ['warning', 'critical']:
                alerts[sid] = result

        total = len(self.error_history)
        n_alerts = len(alerts)

        return {
            'total_series': total,
            'alerting_series': n_alerts,
            'alert_rate': n_alerts / max(total, 1),
            'system_status': 'healthy' if n_alerts / max(total, 1) < 0.1 else 'degraded',
            'alerts': alerts
        }
预测回测（Backtesting）是在历史数据上模拟预测系统的实际运行，评估系统在不同时间段上的预测性能。回测结果揭示系统的稳定性、在特殊时期（如节假日、促销、疫情）的表现和长期性能趋势。

特殊事件与外生变量处理
真实世界的时间序列受到各种特殊事件的影响：节假日、促销活动、天气异常、政策变化和黑天鹅事件。预测系统需要系统化地整合这些外生信息以提升预测精度。


节假日效应的建模通常使用虚拟变量（Dummy Variables）。每个节假日创建一个二元指示变量，标记该节假日发生的时间步。节假日的影响可能持续多天（如春节假期前后），需要使用窗口化的虚拟变量来捕获前置效应和后续效应。


促销活动的建模比节假日更复杂，因为促销的幅度（折扣力度）、品类（参与商品范围）和渠道（线上/线下）都会影响效果。将促销信息编码为数值特征（如折扣率、是否参与活动）并输入模型是常见做法。


天气和宏观经济指标等外生变量在预测未来时面临一个悖论：预测模型需要这些变量的未来值作为输入，但这些变量本身也需要预测。解决方案包括：使用天气预报数据（未来 7-14 天）、使用滞后值代替当期值、或者使用情景分析（乐观/中性/悲观三种天气假设下的预测）。

小结
    

  • 工业级预测系统包含数据管道、特征计算、模型训练、预测生成、协调和监控的完整闭环
    • 

  • 大规模预测需要在局部模型和全局模型之间选择，全局模型在短序列和冷启动场景下优势明显
    • 

  • 预测监控通过滚动 MASE、预测偏差和覆盖率等指标持续追踪系统健康状态
    • 

  • 预测回测在历史数据上模拟系统运行，揭示性能稳定性和特殊时期的表现
    • 

  • 节假日和促销等确定性事件通过虚拟变量建模，天气等不确定外生变量需要情景分析
    • 

  • 降级策略（如回退到简单模型或历史均值）是高可用预测系统的必备能力
    • 

  • 预测系统设计的核心原则："自动化优先，人工兜底"
