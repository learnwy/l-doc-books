特征监控与治理

核心概念

特征监控（Feature Monitoring）是确保机器学习系统在生产环境中持续稳定运行的关键环节。模型上线后，输入特征的分布可能因数据源变更、用户行为变化、季节性波动或上游系统故障而发生偏移，导致模型性能默默退化。特征监控通过持续检测特征分布的变化，在模型性能受到影响之前发出预警，使团队能够及时响应。

特征监控的核心理念是"特征即数据合约（Feature as Contract）"——每个特征在注册时就定义了其期望的分布范围、数据类型、更新频率和质量标准，监控系统持续验证生产数据是否满足这些合约。


分布偏移检测
特征分布偏移（Distribution Drift）是指生产环境中的特征值分布相对于训练时的基线分布发生了显著变化。分布偏移是模型性能退化的最常见原因之一，但它的隐蔽性在于模型可能不会立即报错，只是预测质量逐渐下降。


PSI（Population Stability Index）是检测分布偏移最广泛使用的指标。它将特征值分为若干个分箱（通常基于训练集的分位数），然后比较基线分布和当前分布在每个分箱中的占比差异。


import numpy as np
from scipy.stats import ks_2samp, chi2_contingency
from typing import Dict, List, Tuple

def compute_psi(baseline: np.ndarray, current: np.ndarray,
                n_bins: int = 10, eps: float = 1e-4) -> float:
    bins = np.quantile(baseline, np.linspace(0, 1, n_bins + 1))
    bins[0] = -np.inf
    bins[-1] = np.inf

    baseline_counts = np.histogram(baseline, bins=bins)[0]
    current_counts = np.histogram(current, bins=bins)[0]

    baseline_pct = baseline_counts / len(baseline) + eps
    current_pct = current_counts / len(current) + eps

    psi = np.sum((current_pct - baseline_pct) * np.log(current_pct / baseline_pct))
    return psi

def compute_drift_metrics(baseline: np.ndarray, current: np.ndarray) -> Dict:
    psi = compute_psi(baseline, current)
    ks_stat, ks_pvalue = ks_2samp(baseline, current)

    js_divergence = compute_js_divergence(baseline, current)

    return {
        'psi': psi,
        'psi_status': 'stable' if psi < 0.1 else ('warning' if psi < 0.2 else 'alert'),
        'ks_statistic': ks_stat,
        'ks_pvalue': ks_pvalue,
        'js_divergence': js_divergence,
        'baseline_mean': np.mean(baseline),
        'current_mean': np.mean(current),
        'mean_shift': (np.mean(current) - np.mean(baseline)) / (np.std(baseline) + 1e-8)
    }

def compute_js_divergence(p: np.ndarray, q: np.ndarray, n_bins: int = 50) -> float:
    bins = np.histogram_bin_edges(np.concatenate([p, q]), bins=n_bins)
    p_hist = np.histogram(p, bins=bins, density=True)[0] + 1e-8
    q_hist = np.histogram(q, bins=bins, density=True)[0] + 1e-8

    m = 0.5 * (p_hist + q_hist)
    js = 0.5 * np.sum(p_hist * np.log(p_hist / m)) + 0.5 * np.sum(q_hist * np.log(q_hist / m))
    return js
PSI 的经验阈值为：PSI < 0.1 表示分布稳定，0.1-0.2 表示轻微偏移需关注，PSI > 0.2 表示显著偏移需要采取行动。分箱数量的选择影响 PSI 的敏感度：分箱太少会漏检细微偏移，分箱太多会对噪声过度敏感。10-20 个分箱是大多数场景的合理选择。

数据质量监控
数据质量监控关注的是特征数据本身的完整性、准确性和时效性，与分布偏移检测互补。一个特征可能分布没有偏移，但空值率突然上升或出现越界值，同样会导致模型行为异常。


from dataclasses import dataclass
from datetime import datetime
from typing import Optional

@dataclass
class FeatureQualityRule:
    feature_name: str
    data_type: str
    nullable: bool = False
    min_value: Optional[float] = None
    max_value: Optional[float] = None
    allowed_values: Optional[set] = None
    max_null_rate: float = 0.01
    max_cardinality: Optional[int] = None
    freshness_hours: Optional[float] = None

class FeatureQualityMonitor:
    def __init__(self):
        self.rules: Dict[str, FeatureQualityRule] = {}
        self.history: List[Dict] = []

    def register_rule(self, rule: FeatureQualityRule):
        self.rules[rule.feature_name] = rule

    def check_quality(self, feature_name: str, values: np.ndarray,
                       timestamp: datetime = None) -> Dict:
        rule = self.rules.get(feature_name)
        if not rule:
            return {'status': 'no_rule', 'feature': feature_name}

        issues = []

        null_count = np.sum(np.isnan(values)) if np.issubdtype(values.dtype, np.floating) else 0
        null_rate = null_count / len(values)
        if null_rate > rule.max_null_rate:
            issues.append({
                'type': 'high_null_rate',
                'value': null_rate,
                'threshold': rule.max_null_rate
            })

        if rule.min_value is not None:
            below_min = np.sum(values[~np.isnan(values)] < rule.min_value)
            if below_min > 0:
                issues.append({
                    'type': 'below_minimum',
                    'count': int(below_min),
                    'min_observed': float(np.nanmin(values))
                })

        if rule.max_value is not None:
            above_max = np.sum(values[~np.isnan(values)] > rule.max_value)
            if above_max > 0:
                issues.append({
                    'type': 'above_maximum',
                    'count': int(above_max),
                    'max_observed': float(np.nanmax(values))
                })

        if rule.max_cardinality is not None:
            cardinality = len(np.unique(values[~np.isnan(values)]))
            if cardinality > rule.max_cardinality:
                issues.append({
                    'type': 'high_cardinality',
                    'value': cardinality,
                    'threshold': rule.max_cardinality
                })

        result = {
            'feature': feature_name,
            'timestamp': timestamp or datetime.now(),
            'n_samples': len(values),
            'null_rate': null_rate,
            'issues': issues,
            'status': 'fail' if issues else 'pass'
        }

        self.history.append(result)
        return result
特征新鲜度监控确保在线特征数据及时更新。每个特征有预期的更新周期（如用户画像每天更新，实时行为特征每分钟更新），当实际更新延迟超过预期的 2 倍时触发告警。新鲜度问题通常指向上游数据管道的故障或延迟。

训练-推理一致性验证
训练-推理偏差（Training-Serving Skew）是模型上线后性能退化的另一个重要原因。即使特征名称和数据管道完全相同，训练和推理阶段的特征值也可能因为以下原因产生差异：数据预处理顺序不一致、特征工程代码版本不同、聚合窗口边界对齐方式不同、或者在线系统使用了缓存而非实时计算的值。


class TrainingServingConsistencyChecker:
    def __init__(self, feature_names: List[str]):
        self.feature_names = feature_names
        self.training_stats: Dict[str, Dict] = {}
        self.serving_stats: Dict[str, Dict] = {}

    def record_training_stats(self, X: np.ndarray):
        for i, name in enumerate(self.feature_names):
            col = X[:, i]
            self.training_stats[name] = {
                'mean': float(np.nanmean(col)),
                'std': float(np.nanstd(col)),
                'min': float(np.nanmin(col)),
                'max': float(np.nanmax(col)),
                'null_rate': float(np.sum(np.isnan(col)) / len(col)),
                'quantiles': {
                    'q25': float(np.nanpercentile(col, 25)),
                    'q50': float(np.nanpercentile(col, 50)),
                    'q75': float(np.nanpercentile(col, 75))
                }
            }

    def check_serving_batch(self, X: np.ndarray) -> Dict:
        discrepancies = []

        for i, name in enumerate(self.feature_names):
            col = X[:, i]
            train = self.training_stats.get(name)
            if not train:
                continue

            serve_mean = float(np.nanmean(col))
            serve_std = float(np.nanstd(col))

            mean_shift = abs(serve_mean - train['mean']) / (train['std'] + 1e-8)
            if mean_shift > 2.0:
                discrepancies.append({
                    'feature': name,
                    'type': 'mean_shift',
                    'training_mean': train['mean'],
                    'serving_mean': serve_mean,
                    'shift_sigma': mean_shift
                })

            serve_null = float(np.sum(np.isnan(col)) / len(col))
            if serve_null > train['null_rate'] * 3 + 0.01:
                discrepancies.append({
                    'feature': name,
                    'type': 'null_rate_increase',
                    'training_null': train['null_rate'],
                    'serving_null': serve_null
                })

        return {
            'n_features': len(self.feature_names),
            'n_discrepancies': len(discrepancies),
            'discrepancies': discrepancies,
            'status': 'pass' if len(discrepancies) == 0 else 'fail'
        }
影子模式（Shadow Mode）是验证训练-推理一致性的金标准方法。在影子模式下，将在线推理请求的特征值记录下来，然后在离线环境中使用相同的数据重新运行特征工程管道，比较两者的输出差异。任何非零差异都表明存在训练-推理偏差。


告警策略与自动响应
特征监控产生的告警需要一套分级处理策略，避免告警疲劳（Alert Fatigue）同时确保真正的问题得到及时处理。


from enum import Enum

class AlertLevel(Enum):
    INFO = 'info'
    WARNING = 'warning'
    CRITICAL = 'critical'

@dataclass
class AlertPolicy:
    feature_name: str
    metric: str
    warning_threshold: float
    critical_threshold: float
    auto_action: Optional[str] = None
    cooldown_minutes: int = 60

class AlertManager:
    def __init__(self):
        self.policies: Dict[str, AlertPolicy] = {}
        self.alert_history: List[Dict] = []
        self.last_alert_time: Dict[str, datetime] = {}

    def register_policy(self, policy: AlertPolicy):
        key = f"{policy.feature_name}:{policy.metric}"
        self.policies[key] = policy

    def evaluate(self, feature_name: str, metric: str,
                  value: float, timestamp: datetime = None) -> Optional[Dict]:
        key = f"{feature_name}:{metric}"
        policy = self.policies.get(key)
        if not policy:
            return None

        ts = timestamp or datetime.now()
        last = self.last_alert_time.get(key)
        if last and (ts - last).total_seconds() < policy.cooldown_minutes * 60:
            return None

        if value >= policy.critical_threshold:
            level = AlertLevel.CRITICAL
        elif value >= policy.warning_threshold:
            level = AlertLevel.WARNING
        else:
            return None

        alert = {
            'feature': feature_name,
            'metric': metric,
            'value': value,
            'level': level.value,
            'timestamp': ts,
            'auto_action': policy.auto_action if level == AlertLevel.CRITICAL else None
        }

        self.alert_history.append(alert)
        self.last_alert_time[key] = ts
        return alert
自动降级策略在特征质量严重恶化时自动触发，避免低质量特征影响在线预测。常见的降级策略包括：使用历史缓存值替代实时特征值、回退到简化模型（不依赖问题特征的备用模型）、返回默认预测值。

小结
    

  • 特征监控是 ML 系统生产环境稳定性的保障，覆盖分布偏移、数据质量和训练-推理一致性三个维度
    • 

  • PSI 是最常用的分布偏移指标，阈值 0.1/0.2 对应关注/告警两个级别
    • 

  • KS 检验和 JS 散度从不同角度补充 PSI，多指标联合使用提高检测可靠性
    • 

  • 数据质量监控关注空值率、越界值、基数异常和新鲜度，每个特征应定义质量合约
    • 

  • 训练-推理偏差的检测通过影子模式和统计量比对实现，是模型上线后必须持续进行的验证
    • 

  • 告警策略需要分级处理，配合冷却期机制避免告警疲劳
    • 

  • 自动降级策略在特征严重异常时保护在线服务，是高可用 ML 系统的必备能力
    • 

  • 特征治理需要组织级别的制度保障，包括特征所有者制度、变更审批流程和定期审计机制
    •
