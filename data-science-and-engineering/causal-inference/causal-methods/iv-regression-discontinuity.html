工具变量与断点回归

核心概念

工具变量（Instrumental Variables, IV）和断点回归（Regression Discontinuity, RD）是处理未观测混杂的两种强大的准实验方法。当无法进行随机实验且存在不可观测的混杂因素时，传统的匹配和回归调整方法都无法给出无偏的因果估计。工具变量利用外生变异源来"绕过"混杂，断点回归利用政策或规则中的断点创造局部随机化。

工具变量方法
工具变量 Z 必须满足三个条件：（1）相关性——Z 与处理 X 相关；（2）排斥性——Z 仅通过 X 影响 Y；（3）外生性——Z 与未观测混杂 U 不相关。满足这三个条件的变量可以作为"自然实验"的来源。


两阶段最小二乘（2SLS） 是最常用的 IV 估计方法：第一阶段用 Z 预测 X 的变异，第二阶段用预测的 X 解释 Y。


import numpy as np
from scipy import stats

class TwoStageLeastSquares:
    def __init__(self):
        self.first_stage = None
        self.second_stage = None

    def fit(self, Z, X, Y, covariates=None):
        n = len(Y)

        if covariates is not None:
            Z_full = np.column_stack([np.ones(n), Z, covariates])
        else:
            Z_full = np.column_stack([np.ones(n), Z])

        beta_first = np.linalg.lstsq(Z_full, X, rcond=None)[0]
        X_hat = Z_full @ beta_first

        residuals_first = X - X_hat
        ss_res = np.sum(residuals_first**2)
        ss_tot = np.sum((X - np.mean(X))**2)
        r2_first = 1 - ss_res / ss_tot

        f_stat = (ss_tot - ss_res) / (ss_res / (n - Z_full.shape[1]))

        if covariates is not None:
            X_second = np.column_stack([np.ones(n), X_hat, covariates])
        else:
            X_second = np.column_stack([np.ones(n), X_hat])

        beta_second = np.linalg.lstsq(X_second, Y, rcond=None)[0]

        if covariates is not None:
            X_actual = np.column_stack([np.ones(n), X, covariates])
        else:
            X_actual = np.column_stack([np.ones(n), X])

        residuals = Y - X_actual @ beta_second
        sigma2 = np.sum(residuals**2) / (n - X_actual.shape[1])
        var_beta = sigma2 * np.linalg.pinv(X_second.T @ X_second)
        se = np.sqrt(np.abs(np.diag(var_beta)))

        iv_estimate = beta_second[1]
        iv_se = se[1]

        self.first_stage = {
            'coefficients': beta_first,
            'r_squared': r2_first,
            'f_statistic': f_stat,
            'weak_instrument': f_stat < 10,
        }

        self.second_stage = {
            'iv_estimate': iv_estimate,
            'se': iv_se,
            't_stat': iv_estimate / iv_se,
            'p_value': 2 * (1 - stats.t.cdf(abs(iv_estimate / iv_se), n - X_second.shape[1])),
            'ci_95': (iv_estimate - 1.96 * iv_se, iv_estimate + 1.96 * iv_se),
        }

        return self

np.random.seed(42)
n = 5000
u = np.random.normal(0, 1, n)
z = np.random.normal(0, 1, n)
x = 0.5 * z + 0.8 * u + np.random.normal(0, 0.5, n)
true_effect = 3.0
y = true_effect * x + 1.5 * u + np.random.normal(0, 1, n)

ols_beta = np.linalg.lstsq(
    np.column_stack([np.ones(n), x]), y, rcond=None
)[0][1]

iv = TwoStageLeastSquares()
iv.fit(z, x, y)

print(f"True effect: {true_effect:.3f}")
print(f"OLS (biased): {ols_beta:.3f}")
print(f"IV estimate: {iv.second_stage['iv_estimate']:.3f}")
print(f"IV SE: {iv.second_stage['se']:.3f}")
print(f"First stage F: {iv.first_stage['f_statistic']:.1f}")

断点回归设计
断点回归（RD）利用某个"运行变量"（running variable）在断点处的不连续变化来识别因果效应。当运行变量超过某个阈值时，单位接受处理；低于阈值时不接受。如果运行变量在断点附近是"随机分配"的，断点两侧的比较就类似于随机实验。


import numpy as np
from scipy import stats

class RegressionDiscontinuity:
    def __init__(self, cutoff=0):
        self.cutoff = cutoff

    def fit_sharp(self, running_var, outcome, bandwidth=None, kernel='triangular'):
        if bandwidth is None:
            bandwidth = self._ik_bandwidth(running_var, outcome)

        mask = np.abs(running_var - self.cutoff) <= bandwidth
        r_local = running_var[mask] - self.cutoff
        y_local = outcome[mask]
        treated = (r_local >= 0).astype(float)

        if kernel == 'triangular':
            weights = 1 - np.abs(r_local) / bandwidth
        elif kernel == 'uniform':
            weights = np.ones_like(r_local)
        else:
            weights = np.exp(-0.5 * (r_local / (bandwidth * 0.5))**2)

        X = np.column_stack([
            np.ones_like(r_local),
            treated,
            r_local,
            r_local * treated,
        ])

        W = np.diag(weights)
        beta = np.linalg.lstsq(W @ X, W @ y_local, rcond=None)[0]

        residuals = y_local - X @ beta
        sigma2 = np.average(residuals**2, weights=weights)
        XtWX_inv = np.linalg.pinv(X.T @ W @ X)
        var_beta = sigma2 * XtWX_inv
        se = np.sqrt(np.abs(np.diag(var_beta)))

        return {
            'rd_estimate': beta[1],
            'se': se[1],
            'ci_95': (beta[1] - 1.96 * se[1], beta[1] + 1.96 * se[1]),
            'bandwidth': bandwidth,
            'n_local': np.sum(mask),
            'n_left': np.sum(r_local < 0),
            'n_right': np.sum(r_local >= 0),
        }

    def _ik_bandwidth(self, running_var, outcome):
        sd_r = np.std(running_var)
        return 1.06 * sd_r * len(running_var) ** (-0.2)

    def manipulation_test(self, running_var, n_bins=50):
        hist, edges = np.histogram(running_var, bins=n_bins)
        bin_centers = (edges[:-1] + edges[1:]) / 2

        cutoff_idx = np.argmin(np.abs(bin_centers - self.cutoff))
        left_density = hist[max(0, cutoff_idx-3):cutoff_idx].mean()
        right_density = hist[cutoff_idx:min(len(hist), cutoff_idx+3)].mean()

        ratio = right_density / left_density if left_density > 0 else float('inf')

        return {
            'density_ratio': ratio,
            'manipulation_suspected': abs(ratio - 1) > 0.5,
            'left_density': left_density,
            'right_density': right_density,
        }

np.random.seed(42)
n = 5000
running = np.random.uniform(-5, 5, n)
treated = (running >= 0).astype(float)
true_rd_effect = 4.0
outcome = 2 + 0.5 * running + true_rd_effect * treated + np.random.normal(0, 1, n)

rd = RegressionDiscontinuity(cutoff=0)
result = rd.fit_sharp(running, outcome, bandwidth=2.0)
manip = rd.manipulation_test(running)

print(f"True RD effect: {true_rd_effect:.3f}")
print(f"RD estimate: {result['rd_estimate']:.3f} (SE: {result['se']:.3f})")
print(f"95% CI: ({result['ci_95'][0]:.3f}, {result['ci_95'][1]:.3f})")
print(f"Bandwidth: {result['bandwidth']:.2f}")
print(f"Local sample: {result['n_local']} (left: {result['n_left']}, right: {result['n_right']})")
print(f"Manipulation test: ratio = {manip['density_ratio']:.2f}, suspected = {manip['manipulation_suspected']}")

RD 设计的有效性检验
RD 设计的有效性依赖于运行变量在断点处是否"连续"——即个体不能精确操纵自己的运行变量值以跳过断点。多种检验方法用于验证这一假设。


import numpy as np

def rd_validity_checks(running_var, covariates, outcome, cutoff=0, bandwidth=2.0):
    results = {}

    mask = np.abs(running_var - cutoff) <= bandwidth
    r_local = running_var[mask]
    treated = (r_local >= cutoff).astype(float)

    for i, cov in enumerate(covariates.T[mask] if len(covariates.shape) > 1 else [covariates[mask]]):
        from scipy.stats import ttest_ind
        left = cov[treated == 0]
        right = cov[treated == 1]
        t_stat, p_val = ttest_ind(left, right)
        results[f'covariate_{i}_balance'] = {
            'mean_left': np.mean(left),
            'mean_right': np.mean(right),
            'p_value': p_val,
            'balanced': p_val > 0.05,
        }

    bandwidths = [bandwidth * f for f in [0.5, 0.75, 1.0, 1.25, 1.5]]
    rd_results = []
    rd_obj = RegressionDiscontinuity(cutoff=cutoff)
    for bw in bandwidths:
        r = rd_obj.fit_sharp(running_var, outcome, bandwidth=bw)
        rd_results.append({'bandwidth': bw, 'estimate': r['rd_estimate'], 'se': r['se']})
    results['bandwidth_sensitivity'] = rd_results

    placebo_cutoffs = [cutoff - 2, cutoff - 1, cutoff + 1, cutoff + 2]
    placebo_results = []
    for pc in placebo_cutoffs:
        rd_p = RegressionDiscontinuity(cutoff=pc)
        r = rd_p.fit_sharp(running_var, outcome, bandwidth=bandwidth)
        placebo_results.append({'cutoff': pc, 'estimate': r['rd_estimate'], 'se': r['se']})
    results['placebo_cutoffs'] = placebo_results

    return results

np.random.seed(42)
n = 5000
running = np.random.uniform(-5, 5, n)
covariates = np.random.normal(0, 1, (n, 2))
treated = (running >= 0).astype(float)
outcome = 2 + 0.5 * running + 4.0 * treated + np.random.normal(0, 1, n)

checks = rd_validity_checks(running, covariates, outcome)
for key, val in checks.items():
    if 'balance' in key:
        print(f"{key}: balanced={val['balanced']}, p={val['p_value']:.3f}")

小结
    

  • 工具变量利用外生变异源绕过未观测混杂，2SLS 是标准估计方法，但需注意弱工具变量问题
    • 

  • 第一阶段 F 统计量 > 10 是工具变量强度的基本标准，弱工具变量导致严重偏差
    • 

  • 断点回归利用运行变量在断点处的不连续性，创造局部随机化条件
    • 

  • Sharp RD 在断点处处理从 0 跳到 1，Fuzzy RD 处理概率跳变，类似于局部 IV
    • 

  • RD 有效性依赖于运行变量不可操纵，需通过密度检验、协变量平衡和带宽敏感性验证
    • 

  • 互联网场景中搜索排名、补贴阈值、平台规则等天然产生断点，是 RD 分析的理想应用
    •
