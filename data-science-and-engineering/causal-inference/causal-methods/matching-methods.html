匹配方法

核心概念

匹配方法（Matching Methods）是因果推断中最直观的观测数据分析技术：对每个处理组个体，在对照组中找到"最相似"的个体进行比较，从而模拟随机实验的效果。匹配的核心思想是"同类比较"——如果两个人在所有可观测特征上都相似，那么他们结果的差异可以归因于是否接受处理。匹配方法的优势在于直观性强、不需要参数化假设，且匹配质量可以通过协变量平衡检验来验证。

近邻匹配与倾向得分匹配
近邻匹配（Nearest Neighbor Matching）是最基础的匹配策略：对每个处理组个体，在对照组中找到协变量距离最近的 k 个个体作为匹配对照。当协变量维度较高时，直接使用多维协变量匹配面临"维度灾难"——高维空间中距离变得无意义。倾向得分匹配通过将多维协变量压缩为一维倾向得分来解决这个问题。


import numpy as np
from sklearn.neighbors import NearestNeighbors
from sklearn.linear_model import LogisticRegression
from scipy import stats

class NearestNeighborMatcher:
    def __init__(self, n_neighbors=1, caliper=None):
        self.n_neighbors = n_neighbors
        self.caliper = caliper

    def fit_match(self, X_treated, X_control):
        nn = NearestNeighbors(n_neighbors=self.n_neighbors, metric='mahalanobis',
                              metric_params={'VI': np.linalg.inv(np.cov(
                                  np.vstack([X_treated, X_control]).T))})
        nn.fit(X_control)
        distances, indices = nn.kneighbors(X_treated)

        if self.caliper is not None:
            ps_std = np.std(np.concatenate([
                X_treated[:, 0], X_control[:, 0]
            ])) if X_treated.shape[1] == 1 else 1.0
            caliper_threshold = self.caliper * ps_std
            valid = distances[:, 0] <= caliper_threshold
        else:
            valid = np.ones(len(X_treated), dtype=bool)

        return indices, valid

class PropensityScoreMatcher:
    def __init__(self, n_neighbors=1, caliper=0.2):
        self.n_neighbors = n_neighbors
        self.caliper = caliper
        self.ps_model = LogisticRegression(max_iter=1000)

    def fit(self, X, W):
        self.ps_model.fit(X, W)
        self.ps = self.ps_model.predict_proba(X)[:, 1]
        return self

    def match(self, W):
        treated_idx = np.where(W == 1)[0]
        control_idx = np.where(W == 0)[0]

        ps_treated = self.ps[treated_idx].reshape(-1, 1)
        ps_control = self.ps[control_idx].reshape(-1, 1)

        nn = NearestNeighbors(n_neighbors=self.n_neighbors)
        nn.fit(ps_control)
        distances, indices = nn.kneighbors(ps_treated)

        ps_std = np.std(self.ps)
        valid = distances[:, 0] <= self.caliper * ps_std

        matched_treated = treated_idx[valid]
        matched_control = control_idx[indices[valid, :self.n_neighbors].flatten()]

        return matched_treated, matched_control

    def estimate_att(self, Y, W):
        matched_treated, matched_control = self.match(W)
        att = np.mean(Y[matched_treated]) - np.mean(Y[matched_control])

        n_matched = len(matched_treated)
        se = np.sqrt(
            np.var(Y[matched_treated], ddof=1) / n_matched +
            np.var(Y[matched_control], ddof=1) / len(matched_control)
        )

        return {'att': att, 'se': se, 'n_matched': n_matched, 'n_dropped': np.sum(W) - n_matched}

np.random.seed(42)
n = 5000
X = np.random.normal(0, 1, (n, 4))
ps_true = 1 / (1 + np.exp(-(0.5 * X[:, 0] + 0.3 * X[:, 1] - 0.2 * X[:, 2])))
W = np.random.binomial(1, ps_true)
true_effect = 2.0
Y = 1 + X[:, 0] + 0.5 * X[:, 1] + true_effect * W + np.random.normal(0, 1, n)

matcher = PropensityScoreMatcher(n_neighbors=1, caliper=0.2)
matcher.fit(X, W)
result = matcher.estimate_att(Y, W)
print(f"True ATT: ~{true_effect:.1f}")
print(f"PS Matched ATT: {result['att']:.3f} (SE: {result['se']:.3f})")
print(f"Matched: {result['n_matched']}, Dropped: {result['n_dropped']}")

粗化精确匹配
粗化精确匹配（Coarsened Exact Matching, CEM）是一种模型无关的匹配方法。它先将连续变量"粗化"为区间，然后在粗化后的协变量上进行精确匹配。CEM 的核心优势是它不依赖任何模型假设，且匹配后的不平衡程度有已知的上界。


import numpy as np
import pandas as pd

class CoarsenedExactMatcher:
    def __init__(self, n_bins=5):
        self.n_bins = n_bins

    def coarsen(self, X, var_names=None):
        n, p = X.shape
        coarsened = np.zeros_like(X, dtype=int)
        self.bin_edges = {}

        for j in range(p):
            edges = np.percentile(X[:, j], np.linspace(0, 100, self.n_bins + 1))
            edges[-1] += 1e-10
            coarsened[:, j] = np.digitize(X[:, j], edges[1:])
            name = var_names[j] if var_names else f'var_{j}'
            self.bin_edges[name] = edges

        return coarsened

    def match(self, X, W, var_names=None):
        coarsened = self.coarsen(X, var_names)

        strata = {}
        for i in range(len(W)):
            key = tuple(coarsened[i])
            if key not in strata:
                strata[key] = {'treated': [], 'control': []}
            if W[i] == 1:
                strata[key]['treated'].append(i)
            else:
                strata[key]['control'].append(i)

        matched_treated = []
        matched_control = []
        weights_treated = []
        weights_control = []

        for key, group in strata.items():
            n_t = len(group['treated'])
            n_c = len(group['control'])
            if n_t > 0 and n_c > 0:
                matched_treated.extend(group['treated'])
                matched_control.extend(group['control'])
                for _ in group['treated']:
                    weights_treated.append(1.0)
                for _ in group['control']:
                    weights_control.append(n_t / n_c)

        return {
            'treated_idx': np.array(matched_treated),
            'control_idx': np.array(matched_control),
            'weights_treated': np.array(weights_treated),
            'weights_control': np.array(weights_control),
            'n_strata': sum(1 for g in strata.values() if len(g['treated']) > 0 and len(g['control']) > 0),
            'n_dropped': sum(len(g['treated']) for g in strata.values() if len(g['control']) == 0),
        }

np.random.seed(42)
n = 5000
X = np.random.normal(0, 1, (n, 3))
ps = 1 / (1 + np.exp(-(0.8 * X[:, 0] + 0.5 * X[:, 1])))
W = np.random.binomial(1, ps)
Y = 1 + X[:, 0] + 0.5 * X[:, 1] + 2.0 * W + np.random.normal(0, 1, n)

cem = CoarsenedExactMatcher(n_bins=5)
result = cem.match(X, W, var_names=['age', 'income', 'usage'])
print(f"Matched strata: {result['n_strata']}")
print(f"Dropped treated: {result['n_dropped']}")

att = (
    np.average(Y[result['treated_idx']], weights=result['weights_treated']) -
    np.average(Y[result['control_idx']], weights=result['weights_control'])
)
print(f"CEM ATT estimate: {att:.3f}")

遗传匹配与最优匹配
遗传匹配（Genetic Matching）使用进化算法自动搜索最优的协变量权重矩阵，使得匹配后的协变量平衡最大化。最优匹配（Optimal Matching）通过全局优化最小化总体匹配距离，而非贪心地逐个匹配。


import numpy as np
from scipy.optimize import linear_sum_assignment

def optimal_matching(X_treated, X_control, metric='mahalanobis'):
    n_t = len(X_treated)
    n_c = len(X_control)

    if metric == 'mahalanobis':
        cov_inv = np.linalg.inv(np.cov(np.vstack([X_treated, X_control]).T))
        cost_matrix = np.zeros((n_t, n_c))
        for i in range(n_t):
            diff = X_control - X_treated[i]
            cost_matrix[i] = np.sqrt(np.sum(diff @ cov_inv * diff, axis=1))
    else:
        from sklearn.metrics import pairwise_distances
        cost_matrix = pairwise_distances(X_treated, X_control)

    row_ind, col_ind = linear_sum_assignment(cost_matrix)

    total_distance = cost_matrix[row_ind, col_ind].sum()
    mean_distance = total_distance / len(row_ind)

    return {
        'treated_matches': row_ind,
        'control_matches': col_ind,
        'distances': cost_matrix[row_ind, col_ind],
        'mean_distance': mean_distance,
        'max_distance': cost_matrix[row_ind, col_ind].max(),
    }

np.random.seed(42)
X_t = np.random.normal(0.5, 1, (100, 3))
X_c = np.random.normal(0, 1, (200, 3))

result = optimal_matching(X_t, X_c)
print(f"Mean matching distance: {result['mean_distance']:.3f}")
print(f"Max matching distance: {result['max_distance']:.3f}")
遗传匹配的优势在于可以处理非线性平衡需求，不需要人为指定距离度量。最优匹配则保证了全局最优的匹配质量，避免了贪心匹配中先匹配的个体"抢走"好的对照而导致后续匹配质量下降的问题。

匹配方法的局限与替代
匹配方法尽管直观有效，但存在固有局限。最根本的局限是它只能控制可观测的混杂因素，无法处理未观测混杂。此外，匹配在样本重叠区域有限时可能丢弃大量数据，降低外部效度。


匹配方法的现代替代和补充方案包括加权方法（IPW、平衡权重）、双重稳健估计和机器学习因果推断方法。这些方法在不同维度上弥补了匹配的不足。

小结
    

  • 匹配方法通过寻找相似的对照组个体来模拟随机实验，核心假设是在匹配变量上条件独立
    • 

  • 倾向得分匹配将多维协变量压缩为一维，解决了高维匹配的维度灾难问题
    • 

  • CEM 不依赖任何模型假设，匹配后的不平衡程度有已知上界，适用于中低维场景
    • 

  • 匹配质量通过协变量平衡检验（SMD < 0.1）验证，良好的平衡是可信推断的前提
    • 

  • 最优匹配避免了贪心匹配的次优问题，但计算复杂度为 O(n³)
    • 

  • 匹配方法的根本局限是无法处理未观测混杂，应结合敏感性分析评估结论的稳健性
