双重差分法

核心概念

双重差分法（Difference-in-Differences, DID/DiD）是因果推断中最经典的准实验方法之一。它利用"自然实验"——政策变化、产品上线、外部冲击等事件，通过比较处理组和对照组在事件前后的变化差异来估计因果效应。名称"双重差分"来自两次差分操作：第一次差分消除组间固定差异（组别效应），第二次差分消除共同时间趋势（时间效应），剩余的差异即为处理效应。

经典 DID 模型
经典 DID 可以通过简单的 2×2 表格计算，也可以通过回归模型实现。回归模型的优势在于可以方便地加入协变量控制和进行统计推断。


import numpy as np
import pandas as pd
from scipy import stats

def classic_did(data, outcome_col, group_col, time_col):
    groups = data.groupby([group_col, time_col])[outcome_col].mean()

    y_t_post = groups.loc[(1, 1)]
    y_t_pre = groups.loc[(1, 0)]
    y_c_post = groups.loc[(0, 1)]
    y_c_pre = groups.loc[(0, 0)]

    did_estimate = (y_t_post - y_t_pre) - (y_c_post - y_c_pre)

    return {
        'did_estimate': did_estimate,
        'treated_change': y_t_post - y_t_pre,
        'control_change': y_c_post - y_c_pre,
        'counterfactual': y_t_pre + (y_c_post - y_c_pre),
    }

def did_regression(data, outcome_col, group_col, time_col, covariates=None):
    y = data[outcome_col].values
    treatment = data[group_col].values
    post = data[time_col].values
    interaction = treatment * post

    if covariates:
        X_cov = data[covariates].values
        X = np.column_stack([np.ones(len(y)), treatment, post, interaction, X_cov])
    else:
        X = np.column_stack([np.ones(len(y)), treatment, post, interaction])

    beta = np.linalg.lstsq(X, y, rcond=None)[0]

    residuals = y - X @ beta
    sigma2 = np.sum(residuals**2) / (len(y) - len(beta))
    var_beta = sigma2 * np.linalg.inv(X.T @ X)
    se = np.sqrt(np.diag(var_beta))

    did_coef = beta[3]
    did_se = se[3]
    t_stat = did_coef / did_se
    p_value = 2 * (1 - stats.t.cdf(abs(t_stat), len(y) - len(beta)))

    return {
        'did_estimate': did_coef,
        'se': did_se,
        't_stat': t_stat,
        'p_value': p_value,
        'ci_95': (did_coef - 1.96 * did_se, did_coef + 1.96 * did_se),
    }

np.random.seed(42)
n_units = 200
n_periods = 2

data_rows = []
for i in range(n_units):
    treated = 1 if i < n_units // 2 else 0
    unit_fe = np.random.normal(0, 2)
    for t in range(n_periods):
        time_fe = 3 * t
        effect = 5.0 * treated * t
        y = 10 + unit_fe + time_fe + effect + np.random.normal(0, 1)
        data_rows.append({'unit': i, 'treated': treated, 'post': t, 'outcome': y})

df = pd.DataFrame(data_rows)
simple_result = classic_did(df, 'outcome', 'treated', 'post')
reg_result = did_regression(df, 'outcome', 'treated', 'post')

print(f"Simple DID: {simple_result['did_estimate']:.3f}")
print(f"Regression DID: {reg_result['did_estimate']:.3f} (SE: {reg_result['se']:.3f})")
print(f"95% CI: ({reg_result['ci_95'][0]:.3f}, {reg_result['ci_95'][1]:.3f})")
print(f"p-value: {reg_result['p_value']:.6f}")

平行趋势假设
平行趋势假设（Parallel Trends Assumption）是 DID 有效性的核心。它要求：在没有处理的情况下，处理组和对照组的结果变量会遵循相同的时间趋势。这是一个关于反事实的假设，无法直接检验，但可以通过事前数据提供间接证据。


import numpy as np
import pandas as pd

def parallel_trends_test(data, outcome_col, group_col, time_col, pre_periods):
    pre_data = data[data[time_col].isin(pre_periods)]

    results = []
    for t in pre_periods:
        t_data = pre_data[pre_data[time_col] == t]
        treated_mean = t_data[t_data[group_col] == 1][outcome_col].mean()
        control_mean = t_data[t_data[group_col] == 0][outcome_col].mean()
        diff = treated_mean - control_mean
        results.append({'period': t, 'diff': diff})

    diffs = [r['diff'] for r in results]
    changes = [diffs[i+1] - diffs[i] for i in range(len(diffs)-1)]

    from scipy.stats import ttest_1samp
    if len(changes) > 1:
        t_stat, p_value = ttest_1samp(changes, 0)
    else:
        t_stat, p_value = 0, 1.0

    return {
        'period_diffs': results,
        'trend_changes': changes,
        't_stat': t_stat,
        'p_value': p_value,
        'parallel_trends_plausible': p_value > 0.05,
    }

def event_study_did(data, outcome_col, unit_col, time_col, group_col,
                     treatment_time, pre_periods=4, post_periods=4):
    data = data.copy()
    data['relative_time'] = data[time_col] - treatment_time
    data = data[(data['relative_time'] >= -pre_periods) & (data['relative_time'] <= post_periods)]

    ref_period = -1

    dummies = []
    for t in range(-pre_periods, post_periods + 1):
        if t == ref_period:
            continue
        col_name = f'D_{t}'
        data[col_name] = ((data['relative_time'] == t) & (data[group_col] == 1)).astype(int)
        dummies.append(col_name)

    y = data[outcome_col].values
    X_dummy = data[dummies].values

    unit_dummies = pd.get_dummies(data[unit_col], drop_first=True).values
    time_dummies = pd.get_dummies(data['relative_time'], drop_first=True).values

    X = np.column_stack([X_dummy, unit_dummies, time_dummies])
    X = np.column_stack([np.ones(len(y)), X])

    beta = np.linalg.lstsq(X, y, rcond=None)[0]

    residuals = y - X @ beta
    sigma2 = np.sum(residuals**2) / (len(y) - X.shape[1])
    var_beta = sigma2 * np.linalg.pinv(X.T @ X)
    se = np.sqrt(np.abs(np.diag(var_beta)))

    results = []
    for i, col in enumerate(dummies):
        t = int(col.split('_')[1])
        results.append({
            'relative_time': t,
            'coefficient': beta[1 + i],
            'se': se[1 + i],
        })

    results.append({'relative_time': ref_period, 'coefficient': 0, 'se': 0})
    results.sort(key=lambda x: x['relative_time'])

    return results

多期 DID 与交错处理
当不同单位在不同时间点接受处理（staggered treatment adoption）时，传统的双向固定效应（TWFE）DID 回归可能产生严重偏差。近年来的因果推断文献对此问题进行了深入研究，提出了多种改进方法。


import numpy as np
import pandas as pd

def simulate_staggered_did(n_units=100, n_periods=10, treatment_periods=None):
    np.random.seed(42)

    if treatment_periods is None:
        treatment_periods = {i: np.random.choice([5, 6, 7, None], p=[0.2, 0.2, 0.2, 0.4])
                           for i in range(n_units)}

    rows = []
    for i in range(n_units):
        unit_fe = np.random.normal(0, 2)
        treat_time = treatment_periods.get(i)

        for t in range(n_periods):
            time_fe = 0.5 * t
            treated = 1 if (treat_time is not None and t >= treat_time) else 0

            dynamic_effect = 0
            if treated and treat_time is not None:
                periods_since = t - treat_time
                dynamic_effect = 2.0 + 0.5 * periods_since

            y = 10 + unit_fe + time_fe + dynamic_effect + np.random.normal(0, 1)

            rows.append({
                'unit': i, 'time': t, 'treated': treated,
                'treatment_time': treat_time if treat_time else -1,
                'outcome': y, 'true_effect': dynamic_effect,
            })

    return pd.DataFrame(rows)

def callaway_santanna_simple(data, outcome_col, unit_col, time_col, treatment_time_col):
    cohorts = data[data[treatment_time_col] > 0].groupby(treatment_time_col)[unit_col].unique()
    never_treated = data[data[treatment_time_col] == -1][unit_col].unique()

    att_gt = {}
    for g_time, g_units in cohorts.items():
        for t in data[time_col].unique():
            if t < g_time:
                continue

            g_data = data[(data[unit_col].isin(g_units)) & (data[time_col] == t)]
            c_data = data[(data[unit_col].isin(never_treated)) & (data[time_col] == t)]

            g_pre = data[(data[unit_col].isin(g_units)) & (data[time_col] == g_time - 1)]
            c_pre = data[(data[unit_col].isin(never_treated)) & (data[time_col] == g_time - 1)]

            did = (g_data[outcome_col].mean() - g_pre[outcome_col].mean()) - \
                  (c_data[outcome_col].mean() - c_pre[outcome_col].mean())

            att_gt[(g_time, t)] = did

    if att_gt:
        att_overall = np.mean(list(att_gt.values()))
    else:
        att_overall = 0

    return {'att_gt': att_gt, 'att_overall': att_overall}

df = simulate_staggered_did()
result = callaway_santanna_simple(df, 'outcome', 'unit', 'time', 'treatment_time')
print(f"Overall ATT: {result['att_overall']:.3f}")
print(f"Group-time effects (first 5):")
for (g, t), att in list(result['att_gt'].items())[:5]:
    print(f"  Cohort {g}, Time {t}: ATT = {att:.3f}")

合成 DID
合成双重差分（Synthetic DID, SDID）结合了合成控制法和 DID 的优势。它既使用单位权重使对照组加权平均匹配处理组的事前水平，又使用时间权重强调事件前后的关键时期。SDID 在平行趋势假设不完全成立时比传统 DID 更稳健。


import numpy as np

def synthetic_did(Y, W, T0):
    N, T = Y.shape
    treated = np.where(W[:, -1] == 1)[0]
    control = np.where(W[:, -1] == 0)[0]

    Y_control_pre = Y[np.ix_(control, range(T0))]
    Y_treated_pre = Y[np.ix_(treated, range(T0))].mean(axis=0)

    from scipy.optimize import minimize

    def unit_weight_obj(w):
        pred = Y_control_pre.T @ w
        return np.sum((pred - Y_treated_pre)**2) + 0.01 * np.sum(w**2)

    n_control = len(control)
    w0 = np.ones(n_control) / n_control
    bounds = [(0, 1)] * n_control
    constraints = {'type': 'eq', 'fun': lambda w: np.sum(w) - 1}

    result = minimize(unit_weight_obj, w0, bounds=bounds, constraints=constraints)
    omega = result.x

    Y_treated_post = Y[np.ix_(treated, range(T0, T))].mean(axis=0)
    Y_treated_pre_mean = Y[np.ix_(treated, range(T0))].mean(axis=0)
    Y_control_post = Y[np.ix_(control, range(T0, T))]
    Y_control_pre_agg = Y[np.ix_(control, range(T0))]

    did_treated = Y_treated_post.mean() - Y_treated_pre_mean.mean()
    did_control = (omega @ Y_control_post).mean() - (omega @ Y_control_pre_agg).mean()

    tau_sdid = did_treated - did_control

    return {
        'tau': tau_sdid,
        'unit_weights': omega,
        'n_effective_controls': 1 / np.sum(omega**2),
    }

np.random.seed(42)
N, T = 50, 20
T0 = 15
unit_fe = np.random.normal(0, 2, N)
time_fe = np.cumsum(np.random.normal(0, 0.5, T))
Y = unit_fe[:, None] + time_fe[None, :] + np.random.normal(0, 0.5, (N, T))

W = np.zeros((N, T))
W[:5, T0:] = 1
Y[:5, T0:] += 3.0

result = synthetic_did(Y, W, T0)
print(f"SDID estimate: {result['tau']:.3f} (true: 3.0)")
print(f"Effective controls: {result['n_effective_controls']:.1f}")

小结
    

  • DID 通过两次差分消除组别固定效应和时间趋势，估计处理的因果效应
    • 

  • 平行趋势假设是 DID 有效性的核心，应通过事件研究和安慰剂检验提供间接证据
    • 

  • 交错处理（staggered DID）下传统 TWFE 回归可能产生偏差，需使用 Callaway-Sant'Anna 等新方法
    • 

  • 事件研究设计可以估计动态处理效应，同时检验事前平行趋势
    • 

  • 合成 DID 结合了合成控制法和 DID 的优势，在平行趋势假设部分不满足时更稳健
    • 

  • DID 是互联网产品评估政策效果、分析自然实验的重要工具
