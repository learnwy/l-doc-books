因果问题形式化
从业务问题到因果推断问题的转化是最关键的第一步。一个好的因果问题需要明确：处理变量是什么（谁被干预了）、结果变量是什么（衡量什么效果）、目标因果量是什么（ATE/ATT/CATE）、以及决策场景是什么（推断结果用来做什么决策）。


from dataclasses import dataclass, field
from enum import Enum

class CausalEstimand(Enum):
    ATE = "average_treatment_effect"
    ATT = "average_treatment_on_treated"
    CATE = "conditional_average_treatment_effect"
    LATE = "local_average_treatment_effect"

class IdentificationStrategy(Enum):
    RCT = "randomized_controlled_trial"
    DID = "difference_in_differences"
    RD = "regression_discontinuity"
    IV = "instrumental_variables"
    MATCHING = "matching"
    DML = "double_machine_learning"
    SYNTH = "synthetic_control"

@dataclass
class CausalAnalysisPlan:
    business_question: str
    treatment: str
    outcome: str
    estimand: CausalEstimand
    strategy: IdentificationStrategy
    key_assumptions: list = field(default_factory=list)
    potential_confounders: list = field(default_factory=list)
    robustness_checks: list = field(default_factory=list)

    def validate(self):
        issues = []
        if not self.key_assumptions:
            issues.append("No assumptions specified")
        if not self.potential_confounders:
            issues.append("No confounders identified")
        if not self.robustness_checks:
            issues.append("No robustness checks planned")
        return issues

    def summary(self):
        issues = self.validate()
        return {
            'question': self.business_question,
            'treatment': self.treatment,
            'outcome': self.outcome,
            'estimand': self.estimand.value,
            'strategy': self.strategy.value,
            'n_assumptions': len(self.key_assumptions),
            'n_confounders': len(self.potential_confounders),
            'n_robustness': len(self.robustness_checks),
            'validation_issues': issues,
        }

plan = CausalAnalysisPlan(
    business_question="新搜索排序算法对购买转化率的因果效应是什么？",
    treatment="新搜索排序算法（vs 旧算法）",
    outcome="7 天购买转化率",
    estimand=CausalEstimand.ATE,
    strategy=IdentificationStrategy.DID,
    key_assumptions=[
        "平行趋势：实验组和对照组在上线前趋势一致",
        "无溢出效应：组间用户互不影响",
        "无同期干预：期间无其他功能上线",
    ],
    potential_confounders=[
        "用户活跃度（可能影响是否使用搜索和是否购买）",
        "季节性（节假日效应）",
        "用户设备类型",
    ],
    robustness_checks=[
        "事前平行趋势检验",
        "安慰剂时间检验",
        "不同时间窗口的效应一致性",
        "按用户分层的异质效应",
    ],
)
summary = plan.summary()
for k, v in summary.items():
    print(f"{k}: {v}")

数据质量与预处理
因果推断对数据质量的要求比预测建模更高——数据中的系统性偏差会直接转化为因果估计的偏差。


import numpy as np
import pandas as pd

class CausalDataQuality:
    def __init__(self, data, treatment_col, outcome_col, covariate_cols):
        self.data = data
        self.treatment_col = treatment_col
        self.outcome_col = outcome_col
        self.covariate_cols = covariate_cols

    def check_missing(self):
        missing_rates = self.data.isnull().mean()
        issues = []
        for col in [self.treatment_col, self.outcome_col] + self.covariate_cols:
            rate = missing_rates.get(col, 0)
            if rate > 0:
                issues.append({'column': col, 'missing_rate': rate,
                             'severity': 'high' if rate > 0.1 else 'medium' if rate > 0.01 else 'low'})
        return issues

    def check_overlap(self):
        from sklearn.linear_model import LogisticRegression

        X = self.data[self.covariate_cols].values
        W = self.data[self.treatment_col].values

        model = LogisticRegression(max_iter=1000)
        model.fit(X, W)
        ps = model.predict_proba(X)[:, 1]

        return {
            'ps_min': np.min(ps),
            'ps_max': np.max(ps),
            'ps_mean': np.mean(ps),
            'pct_extreme': np.mean((ps < 0.01) | (ps > 0.99)),
            'effective_sample_pct': np.mean((ps > 0.05) & (ps < 0.95)),
        }

    def check_balance(self):
        results = []
        for col in self.covariate_cols:
            treated = self.data[self.data[self.treatment_col] == 1][col]
            control = self.data[self.data[self.treatment_col] == 0][col]

            smd = (treated.mean() - control.mean()) / np.sqrt((treated.var() + control.var()) / 2)

            results.append({
                'covariate': col,
                'smd': abs(smd),
                'balanced': abs(smd) < 0.1,
                'treated_mean': treated.mean(),
                'control_mean': control.mean(),
            })
        return results

    def full_check(self):
        return {
            'missing': self.check_missing(),
            'overlap': self.check_overlap(),
            'balance': self.check_balance(),
        }

np.random.seed(42)
n = 5000
data = pd.DataFrame({
    'treatment': np.random.binomial(1, 0.4, n),
    'outcome': np.random.normal(10, 3, n),
    'age': np.random.normal(30, 10, n),
    'income': np.random.lognormal(10, 1, n),
    'tenure': np.random.exponential(365, n),
})
data.loc[np.random.choice(n, 50), 'income'] = np.nan

quality = CausalDataQuality(data, 'treatment', 'outcome', ['age', 'income', 'tenure'])
report = quality.full_check()

print("Missing data:")
for m in report['missing']:
    print(f"  {m['column']}: {m['missing_rate']:.1%} ({m['severity']})")
print(f"\nOverlap: effective sample = {report['overlap']['effective_sample_pct']:.1%}")
print("\nBalance:")
for b in report['balance']:
    status = "PASS" if b['balanced'] else "FAIL"
    print(f"  {b['covariate']}: SMD = {b['smd']:.3f} [{status}]")

多方法三角验证
因果推断中最强大的验证策略是"三角验证"（Triangulation）：使用多种不同假设的方法分析同一因果问题。如果不同方法在不同假设下给出一致的结论，因果关系的可信度大大增强。


import numpy as np
from scipy import stats

class TriangulationFramework:
    def __init__(self):
        self.results = {}

    def add_result(self, method, estimate, se, assumptions):
        self.results[method] = {
            'estimate': estimate,
            'se': se,
            'ci_95': (estimate - 1.96 * se, estimate + 1.96 * se),
            'assumptions': assumptions,
        }

    def compare(self):
        estimates = [r['estimate'] for r in self.results.values()]
        ses = [r['se'] for r in self.results.values()]

        range_est = max(estimates) - min(estimates)
        mean_se = np.mean(ses)

        consistent = range_est < 2 * mean_se

        weights = [1/se**2 for se in ses]
        total_weight = sum(weights)
        pooled = sum(e * w for e, w in zip(estimates, weights)) / total_weight
        pooled_se = np.sqrt(1 / total_weight)

        return {
            'consistent': consistent,
            'pooled_estimate': pooled,
            'pooled_se': pooled_se,
            'pooled_ci': (pooled - 1.96 * pooled_se, pooled + 1.96 * pooled_se),
            'range': range_est,
            'methods': list(self.results.keys()),
        }

    def report(self):
        comparison = self.compare()
        lines = []
        lines.append(f"Triangulation Report ({len(self.results)} methods)")
        lines.append(f"Consistency: {'PASS' if comparison['consistent'] else 'FAIL'}")
        lines.append(f"Pooled estimate: {comparison['pooled_estimate']:.3f} (SE: {comparison['pooled_se']:.3f})")
        lines.append(f"Pooled 95% CI: ({comparison['pooled_ci'][0]:.3f}, {comparison['pooled_ci'][1]:.3f})")
        lines.append(f"Range across methods: {comparison['range']:.3f}")
        lines.append("")

        for method, result in self.results.items():
            lines.append(f"  {method}: {result['estimate']:.3f} ({result['ci_95'][0]:.3f}, {result['ci_95'][1]:.3f})")
            for a in result['assumptions']:
                lines.append(f"    Assumption: {a}")

        return "\n".join(lines)

tri = TriangulationFramework()
tri.add_result('OLS with controls', 2.1, 0.3, ['No unmeasured confounders', 'Linear model correct'])
tri.add_result('IPW', 1.9, 0.4, ['Propensity score model correct', 'Positivity'])
tri.add_result('Doubly Robust', 2.0, 0.35, ['At least one model correct', 'Positivity'])
tri.add_result('DID', 2.3, 0.5, ['Parallel trends', 'No spillover'])

print(tri.report())

因果推断报告规范
因果推断结果的报告应遵循透明性原则，帮助读者评估结论的可信度。


from dataclasses import dataclass

@dataclass
class CausalReport:
    question: str
    estimand: str
    strategy: str
    main_estimate: float
    confidence_interval: tuple
    key_assumptions: list
    robustness_results: dict
    limitations: list

    def generate(self):
        report = f"""
因果分析报告
{'=' * 50}

研究问题: {self.question}
因果量: {self.estimand}
识别策略: {self.strategy}

主要结果
--------
效应估计: {self.main_estimate:.3f}
95% 置信区间: ({self.confidence_interval[0]:.3f}, {self.confidence_interval[1]:.3f})

核心假设
--------"""
        for i, a in enumerate(self.key_assumptions, 1):
            report += f"\n{i}. {a}"

        report += "\n\n稳健性检查\n--------"
        for method, result in self.robustness_results.items():
            report += f"\n{method}: {result:.3f}"

        report += "\n\n局限性\n--------"
        for i, l in enumerate(self.limitations, 1):
            report += f"\n{i}. {l}"

        return report

report = CausalReport(
    question="新搜索排序算法对 7 天购买转化率的因果效应",
    estimand="ATE",
    strategy="DID + 事件研究",
    main_estimate=0.023,
    confidence_interval=(0.015, 0.031),
    key_assumptions=[
        "平行趋势：通过事前 4 期趋势检验验证（所有系数 p > 0.1）",
        "无溢出效应：搜索结果个性化，用户间无直接干扰",
        "无同期干预：期间无其他功能上线（已与产品团队确认）",
    ],
    robustness_results={
        'OLS with controls': 0.021,
        'IPW': 0.024,
        'Doubly Robust': 0.022,
        'Different bandwidth': 0.025,
    },
    limitations=[
        "效应基于短期数据（上线后 4 周），长期效果可能不同",
        "仅针对搜索场景，其他入口的溢出效应未评估",
        "用户设备分布在实验期间有轻微变化",
    ],
)

print(report.generate())

小结
    

  • 因果分析的第一步是将业务问题形式化为清晰的因果问题，明确处理、结果和目标因果量
    • 

  • 画因果图（DAG）是不可跳过的步骤，它将因果假设显式化，指导变量选择和方法选择
    • 

  • 数据质量检查（缺失、重叠、平衡）在因果推断中比预测建模更关键
    • 

  • 三角验证（多方法、多数据源、多度量）是增强因果结论可信度的最有效策略
    • 

  • 因果推断报告必须包含假设、假设检验、稳健性分析和局限性，遵循透明性原则
    • 

  • 记住没有完美的因果推断——目标是做出最可信的推断并诚实地报告不确定性
