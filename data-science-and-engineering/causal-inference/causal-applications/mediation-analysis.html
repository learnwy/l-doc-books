中介分析

核心概念

中介分析（Mediation Analysis）研究因果效应的传递机制：处理 X 如何通过中介变量 M 影响结果 Y？不同于只关心"X 是否影响 Y"，中介分析回答"X 为什么影响 Y"以及"多大程度上通过 M 传递"。在产品分析中，理解因果机制对于优化策略至关重要。例如，一个新的搜索算法提升了购买转化率，中介分析可以揭示这个提升是通过改善搜索结果相关性（M₁）还是通过增加展示多样性（M₂）实现的。

Baron-Kenny 方法
Baron-Kenny 方法（又称逐步回归法）是最经典的中介分析方法，通过三个回归模型分解总效应为直接效应和间接效应。


import numpy as np
from scipy import stats

class BaronKennyMediation:
    def __init__(self):
        self.results = {}

    def fit(self, X, M, Y, covariates=None):
        n = len(Y)

        if covariates is not None:
            X_cov = np.column_stack([np.ones(n), X, covariates])
            XM_cov = np.column_stack([np.ones(n), X, M, covariates])
        else:
            X_cov = np.column_stack([np.ones(n), X])
            XM_cov = np.column_stack([np.ones(n), X, M])

        beta_total = np.linalg.lstsq(X_cov, Y, rcond=None)[0]
        c = beta_total[1]

        beta_mediator = np.linalg.lstsq(X_cov, M, rcond=None)[0]
        a = beta_mediator[1]

        beta_outcome = np.linalg.lstsq(XM_cov, Y, rcond=None)[0]
        c_prime = beta_outcome[1]
        b = beta_outcome[2]

        indirect = a * b
        direct = c_prime
        total = c

        se_a = self._compute_se(X_cov, M, 1)
        se_b = self._compute_se(XM_cov, Y, 2)
        se_indirect = np.sqrt(a**2 * se_b**2 + b**2 * se_a**2)

        z_indirect = indirect / se_indirect
        p_indirect = 2 * (1 - stats.norm.cdf(abs(z_indirect)))

        self.results = {
            'total_effect': total,
            'direct_effect': direct,
            'indirect_effect': indirect,
            'a_path': a,
            'b_path': b,
            'se_indirect': se_indirect,
            'z_indirect': z_indirect,
            'p_indirect': p_indirect,
            'mediation_proportion': indirect / total if total != 0 else 0,
        }
        return self

    def _compute_se(self, X, Y, coef_idx):
        beta = np.linalg.lstsq(X, Y, rcond=None)[0]
        residuals = Y - X @ beta
        sigma2 = np.sum(residuals**2) / (len(Y) - X.shape[1])
        var_beta = sigma2 * np.linalg.pinv(X.T @ X)
        return np.sqrt(np.abs(var_beta[coef_idx, coef_idx]))

    def bootstrap_ci(self, X, M, Y, n_boot=1000, alpha=0.05):
        n = len(Y)
        indirect_effects = []

        for _ in range(n_boot):
            idx = np.random.choice(n, n, replace=True)
            model = BaronKennyMediation()
            model.fit(X[idx], M[idx], Y[idx])
            indirect_effects.append(model.results['indirect_effect'])

        ci_lower = np.percentile(indirect_effects, 100 * alpha / 2)
        ci_upper = np.percentile(indirect_effects, 100 * (1 - alpha / 2))

        return {
            'bootstrap_ci': (ci_lower, ci_upper),
            'bootstrap_se': np.std(indirect_effects),
            'significant': ci_lower > 0 or ci_upper < 0,
        }

np.random.seed(42)
n = 2000
X = np.random.binomial(1, 0.5, n).astype(float)
M = 0.5 * X + np.random.normal(0, 1, n)
Y = 0.3 * X + 0.4 * M + np.random.normal(0, 1, n)

mediation = BaronKennyMediation()
mediation.fit(X, M, Y)
boot = mediation.bootstrap_ci(X, M, Y)

r = mediation.results
print(f"Total effect (c): {r['total_effect']:.3f}")
print(f"Direct effect (c'): {r['direct_effect']:.3f}")
print(f"Indirect effect (a×b): {r['indirect_effect']:.3f}")
print(f"a path: {r['a_path']:.3f}, b path: {r['b_path']:.3f}")
print(f"Mediation proportion: {r['mediation_proportion']:.1%}")
print(f"Bootstrap 95% CI: ({boot['bootstrap_ci'][0]:.3f}, {boot['bootstrap_ci'][1]:.3f})")

因果中介分析
Baron-Kenny 方法依赖线性模型和无交互效应的假设。因果中介分析（Causal Mediation Analysis）基于潜在结果框架，提供了更严格的定义和更灵活的估计方法，允许非线性关系和处理-中介交互效应。


import numpy as np

class CausalMediation:
    def __init__(self, outcome_model=None, mediator_model=None):
        from sklearn.ensemble import GradientBoostingRegressor
        self.outcome_model = outcome_model or GradientBoostingRegressor(n_estimators=100)
        self.mediator_model = mediator_model or GradientBoostingRegressor(n_estimators=100)

    def fit(self, X, T, M, Y, n_simulations=1000):
        from sklearn.base import clone

        covariates_m = np.column_stack([T, X])
        m_model = clone(self.mediator_model)
        m_model.fit(covariates_m, M)

        covariates_y = np.column_stack([T, M, X])
        y_model = clone(self.outcome_model)
        y_model.fit(covariates_y, Y)

        n = len(Y)

        m_pred_t1 = []
        m_pred_t0 = []
        for _ in range(n_simulations):
            X_t1 = np.column_stack([np.ones(n), X])
            X_t0 = np.column_stack([np.zeros(n), X])

            m1 = m_model.predict(X_t1)
            m0 = m_model.predict(X_t0)

            noise = np.random.normal(0, np.std(M - m_model.predict(covariates_m)), n)
            m_pred_t1.append(m1 + noise)
            m_pred_t0.append(m0 + noise)

        nde_samples = []
        nie_samples = []

        for i in range(n_simulations):
            y_1_m0 = y_model.predict(np.column_stack([np.ones(n), m_pred_t0[i], X]))
            y_0_m0 = y_model.predict(np.column_stack([np.zeros(n), m_pred_t0[i], X]))
            nde = np.mean(y_1_m0 - y_0_m0)
            nde_samples.append(nde)

            y_1_m1 = y_model.predict(np.column_stack([np.ones(n), m_pred_t1[i], X]))
            y_1_m0_2 = y_model.predict(np.column_stack([np.ones(n), m_pred_t0[i], X]))
            nie = np.mean(y_1_m1 - y_1_m0_2)
            nie_samples.append(nie)

        nde = np.mean(nde_samples)
        nie = np.mean(nie_samples)
        te = nde + nie

        return {
            'total_effect': te,
            'nde': nde,
            'nie': nie,
            'mediation_proportion': nie / te if te != 0 else 0,
            'nde_ci': (np.percentile(nde_samples, 2.5), np.percentile(nde_samples, 97.5)),
            'nie_ci': (np.percentile(nie_samples, 2.5), np.percentile(nie_samples, 97.5)),
        }

np.random.seed(42)
n = 3000
X = np.random.normal(0, 1, (n, 3))
T = np.random.binomial(1, 0.5, n).astype(float)
M = 0.5 * T + 0.3 * X[:, 0] + np.random.normal(0, 0.5, n)
Y = 0.4 * T + 0.6 * M + 0.3 * T * M + 0.2 * X[:, 0] + np.random.normal(0, 0.5, n)

cm = CausalMediation()
result = cm.fit(X, T, M, Y, n_simulations=200)
print(f"Total effect: {result['total_effect']:.3f}")
print(f"NDE: {result['nde']:.3f} (CI: {result['nde_ci'][0]:.3f}, {result['nde_ci'][1]:.3f})")
print(f"NIE: {result['nie']:.3f} (CI: {result['nie_ci'][0]:.3f}, {result['nie_ci'][1]:.3f})")
print(f"Mediation proportion: {result['mediation_proportion']:.1%}")

多中介与路径分析
实际问题中往往存在多个中介变量，形成复杂的因果路径网络。多中介分析需要区分不同路径的贡献，并处理中介之间的因果关系。


import numpy as np

def multi_mediator_analysis(X, T, M_list, Y, m_names=None):
    n = len(Y)
    if m_names is None:
        m_names = [f'M{i}' for i in range(len(M_list))]

    X_full = np.column_stack([np.ones(n), T])
    beta_total = np.linalg.lstsq(X_full, Y, rcond=None)[0]
    total_effect = beta_total[1]

    M_all = np.column_stack(M_list)
    X_full_all = np.column_stack([np.ones(n), T, M_all])
    beta_all = np.linalg.lstsq(X_full_all, Y, rcond=None)[0]
    direct_effect = beta_all[1]

    path_effects = {}
    for i, (m, name) in enumerate(zip(M_list, m_names)):
        X_m = np.column_stack([np.ones(n), T])
        beta_a = np.linalg.lstsq(X_m, m, rcond=None)[0]
        a_i = beta_a[1]
        b_i = beta_all[2 + i]

        path_effects[name] = {
            'a_path': a_i,
            'b_path': b_i,
            'indirect': a_i * b_i,
            'proportion': (a_i * b_i) / total_effect if total_effect != 0 else 0,
        }

    total_indirect = sum(pe['indirect'] for pe in path_effects.values())

    return {
        'total_effect': total_effect,
        'direct_effect': direct_effect,
        'total_indirect': total_indirect,
        'path_effects': path_effects,
    }

np.random.seed(42)
n = 3000
T = np.random.binomial(1, 0.5, n).astype(float)
M1 = 0.6 * T + np.random.normal(0, 1, n)
M2 = 0.3 * T + 0.2 * M1 + np.random.normal(0, 1, n)
M3 = 0.1 * T + np.random.normal(0, 1, n)
Y = 0.2 * T + 0.5 * M1 + 0.3 * M2 + 0.1 * M3 + np.random.normal(0, 1, n)

result = multi_mediator_analysis(T, T, [M1, M2, M3], Y, ['relevance', 'diversity', 'freshness'])
print(f"Total effect: {result['total_effect']:.3f}")
print(f"Direct effect: {result['direct_effect']:.3f}")
print(f"Total indirect: {result['total_indirect']:.3f}")
for name, pe in result['path_effects'].items():
    print(f"  {name}: indirect = {pe['indirect']:.3f} ({pe['proportion']:.1%})")

小结
    

  • 中介分析回答"为什么"和"通过什么机制"，将总效应分解为直接效应和间接效应
    • 

  • Baron-Kenny 方法是最经典的中介分析框架，Bootstrap 置信区间是推荐的推断方法
    • 

  • 因果中介分析基于潜在结果框架，允许非线性关系和处理-中介交互效应
    • 

  • 序贯可忽略性是因果中介分析的核心假设，中介路径上的未观测混杂是主要威胁
    • 

  • 多中介分析需要区分并行和序列中介结构，不同结构的路径分解方法不同
    • 

  • 产品分析中中介分析帮助理解变更的传递机制，指导精细化的优化方向
