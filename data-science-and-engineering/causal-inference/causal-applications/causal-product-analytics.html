因果推断与产品分析

核心概念

因果推断在产品分析中的应用远超 A/B 测试。当随机实验不可行（伦理限制、成本过高、技术不可能）、或需要利用历史数据进行回顾性分析时，观测数据的因果推断方法成为核心工具。产品分析师需要回答的问题本质上都是因果问题：这个功能是否提升了用户留存？价格调整对收入的影响是什么？竞争对手的新产品对我们的市场份额有什么影响？

用户留存的因果分析
用户留存分析是产品分析中最常见的因果问题之一。观察到"使用某功能的用户留存率更高"并不意味着该功能促进了留存——可能只是本身就活跃的用户更倾向于使用该功能。准确估计功能对留存的因果效应需要系统的因果推断方法。


import numpy as np
from sklearn.linear_model import LogisticRegression

class RetentionCausalAnalysis:
    def __init__(self):
        self.ps_model = None

    def estimate_feature_impact(self, X, feature_used, retained, method='ipw'):
        n = len(retained)

        self.ps_model = LogisticRegression(max_iter=1000)
        self.ps_model.fit(X, feature_used)
        ps = self.ps_model.predict_proba(X)[:, 1]
        ps = np.clip(ps, 0.01, 0.99)

        if method == 'ipw':
            w_treated = feature_used / ps
            w_control = (1 - feature_used) / (1 - ps)

            ate = (np.sum(w_treated * retained) / np.sum(w_treated) -
                   np.sum(w_control * retained) / np.sum(w_control))

            return {'ate': ate, 'method': 'IPW'}

        elif method == 'dr':
            from sklearn.ensemble import GradientBoostingClassifier
            from sklearn.base import clone

            m1 = GradientBoostingClassifier(n_estimators=50)
            m0 = GradientBoostingClassifier(n_estimators=50)

            if np.sum(feature_used == 1) > 10:
                m1.fit(X[feature_used == 1], retained[feature_used == 1])
            if np.sum(feature_used == 0) > 10:
                m0.fit(X[feature_used == 0], retained[feature_used == 0])

            mu1 = m1.predict_proba(X)[:, 1] if np.sum(feature_used == 1) > 10 else np.full(n, 0.5)
            mu0 = m0.predict_proba(X)[:, 1] if np.sum(feature_used == 0) > 10 else np.full(n, 0.5)

            dr_ate = np.mean(
                mu1 - mu0
                + feature_used * (retained - mu1) / ps
                - (1 - feature_used) * (retained - mu0) / (1 - ps)
            )

            return {'ate': dr_ate, 'method': 'Doubly Robust'}

    def stratified_analysis(self, X, feature_used, retained, strata_col):
        unique_strata = np.unique(strata_col)
        results = []

        for s in unique_strata:
            mask = strata_col == s
            n_s = np.sum(mask)

            if np.sum(feature_used[mask] == 1) < 5 or np.sum(feature_used[mask] == 0) < 5:
                continue

            ret_used = np.mean(retained[mask & (feature_used == 1)])
            ret_not = np.mean(retained[mask & (feature_used == 0)])

            results.append({
                'stratum': s,
                'n': n_s,
                'retention_used': ret_used,
                'retention_not_used': ret_not,
                'diff': ret_used - ret_not,
            })

        return results

np.random.seed(42)
n = 10000
user_activity = np.random.normal(5, 2, n)
user_tenure = np.random.exponential(30, n)
X = np.column_stack([user_activity, user_tenure, np.random.normal(0, 1, (n, 3))])

p_use = 1 / (1 + np.exp(-(0.5 * user_activity - 2)))
feature_used = np.random.binomial(1, p_use)

true_effect = 0.05
p_retain = 1 / (1 + np.exp(-(0.3 * user_activity + 0.01 * user_tenure - 3 + true_effect * feature_used)))
retained = np.random.binomial(1, p_retain)

naive = np.mean(retained[feature_used == 1]) - np.mean(retained[feature_used == 0])

analysis = RetentionCausalAnalysis()
ipw_result = analysis.estimate_feature_impact(X, feature_used, retained, method='ipw')
dr_result = analysis.estimate_feature_impact(X, feature_used, retained, method='dr')

print(f"Naive estimate: {naive:.4f} (confounded)")
print(f"IPW estimate: {ipw_result['ate']:.4f}")
print(f"DR estimate: {dr_result['ate']:.4f}")
print(f"True effect: {true_effect:.4f}")

定价因果分析
价格变动对需求、收入和利润的因果效应是电商和平台经济中的核心问题。由于价格通常由供需关系决定而非随机设定，简单的价格-需求回归会产生严重的内生性偏差。


import numpy as np

class PricingCausalAnalysis:
    def __init__(self):
        pass

    def iv_price_elasticity(self, price, quantity, instrument, covariates=None):
        n = len(price)

        if covariates is not None:
            Z = np.column_stack([np.ones(n), instrument, covariates])
            X_full = np.column_stack([np.ones(n), price, covariates])
        else:
            Z = np.column_stack([np.ones(n), instrument])
            X_full = np.column_stack([np.ones(n), price])

        beta_first = np.linalg.lstsq(Z, price, rcond=None)[0]
        price_hat = Z @ beta_first

        first_stage_resid = price - price_hat
        ss_res = np.sum(first_stage_resid**2)
        ss_tot = np.sum((price - np.mean(price))**2)
        f_stat = ((ss_tot - ss_res) / 1) / (ss_res / (n - Z.shape[1]))

        if covariates is not None:
            X_second = np.column_stack([np.ones(n), price_hat, covariates])
        else:
            X_second = np.column_stack([np.ones(n), price_hat])

        beta_second = np.linalg.lstsq(X_second, quantity, rcond=None)[0]

        price_effect = beta_second[1]
        mean_price = np.mean(price)
        mean_quantity = np.mean(quantity)
        elasticity = price_effect * mean_price / mean_quantity

        return {
            'price_coefficient': price_effect,
            'elasticity': elasticity,
            'first_stage_f': f_stat,
            'mean_price': mean_price,
            'mean_quantity': mean_quantity,
        }

    def did_price_change(self, data_before, data_after, price_change_group, control_group):
        y_t_pre = np.mean(data_before[price_change_group])
        y_t_post = np.mean(data_after[price_change_group])
        y_c_pre = np.mean(data_before[control_group])
        y_c_post = np.mean(data_after[control_group])

        did = (y_t_post - y_t_pre) - (y_c_post - y_c_pre)

        return {
            'did_estimate': did,
            'treatment_change': y_t_post - y_t_pre,
            'control_change': y_c_post - y_c_pre,
        }

np.random.seed(42)
n = 3000
demand_shock = np.random.normal(0, 1, n)
cost_instrument = np.random.normal(10, 3, n)
price = 20 + 0.5 * cost_instrument + 0.3 * demand_shock + np.random.normal(0, 1, n)
quantity = 100 - 2.0 * price + 5 * demand_shock + np.random.normal(0, 5, n)

ols_beta = np.linalg.lstsq(
    np.column_stack([np.ones(n), price]), quantity, rcond=None
)[0][1]

analysis = PricingCausalAnalysis()
iv_result = analysis.iv_price_elasticity(price, quantity, cost_instrument)

print(f"OLS price effect: {ols_beta:.3f} (biased due to demand shock)")
print(f"IV price effect: {iv_result['price_coefficient']:.3f} (true: -2.0)")
print(f"Price elasticity: {iv_result['elasticity']:.3f}")
print(f"First stage F: {iv_result['first_stage_f']:.1f}")

归因分析
多触点归因（Multi-Touch Attribution）是数字营销中的核心因果问题：用户在购买前接触了多个广告渠道（搜索、社交、邮件、展示广告等），每个渠道对最终转化的贡献是多少？


import numpy as np

class CausalAttribution:
    def __init__(self):
        pass

    def shapley_attribution(self, channels, conversion, n_permutations=1000):
        unique_channels = list(set(c for path in channels for c in path))
        n_channels = len(unique_channels)
        channel_idx = {c: i for i, c in enumerate(unique_channels)}

        channel_present = np.zeros((len(channels), n_channels))
        for i, path in enumerate(channels):
            for c in path:
                channel_present[i, channel_idx[c]] = 1

        def coalition_value(coalition_mask):
            matching = np.all(channel_present[:, coalition_mask] >= 1, axis=1) if np.any(coalition_mask) else np.ones(len(conversion), dtype=bool)
            if not np.any(matching):
                return 0
            return np.mean(conversion[matching])

        shapley_values = np.zeros(n_channels)

        for _ in range(n_permutations):
            perm = np.random.permutation(n_channels)
            coalition = np.zeros(n_channels, dtype=bool)
            prev_value = 0

            for ch in perm:
                coalition[ch] = True
                new_value = coalition_value(coalition)
                shapley_values[ch] += new_value - prev_value
                prev_value = new_value

        shapley_values /= n_permutations

        total = np.sum(shapley_values)
        if total > 0:
            shapley_values = shapley_values / total * np.mean(conversion)

        return {c: shapley_values[i] for c, i in channel_idx.items()}

    def incremental_attribution(self, X, channel_exposure, conversion):
        from sklearn.linear_model import LogisticRegression

        results = {}
        for ch_name, exposure in channel_exposure.items():
            model = LogisticRegression(max_iter=1000)
            model.fit(X, exposure)
            ps = np.clip(model.predict_proba(X)[:, 1], 0.01, 0.99)

            w_t = exposure / ps
            w_c = (1 - exposure) / (1 - ps)

            att = (np.sum(w_t * conversion) / np.sum(w_t) -
                   np.sum(w_c * conversion) / np.sum(w_c))

            results[ch_name] = {
                'incremental_effect': att,
                'exposed_rate': np.mean(conversion[exposure == 1]),
                'unexposed_rate': np.mean(conversion[exposure == 0]),
            }

        return results

np.random.seed(42)
n = 5000
X = np.random.normal(0, 1, (n, 5))
channel_exposure = {
    'search': np.random.binomial(1, 1/(1+np.exp(-X[:, 0]))),
    'social': np.random.binomial(1, 1/(1+np.exp(-0.5*X[:, 1]))),
    'email': np.random.binomial(1, 1/(1+np.exp(-0.3*X[:, 2]))),
}

p_convert = 1/(1+np.exp(-(
    -3 + 0.3*X[:, 0] + 0.2*X[:, 1] +
    0.5*channel_exposure['search'] +
    0.3*channel_exposure['social'] +
    0.2*channel_exposure['email']
)))
conversion = np.random.binomial(1, p_convert)

inc_attr = CausalAttribution().incremental_attribution(X, channel_exposure, conversion)
for ch, r in inc_attr.items():
    print(f"{ch}: incremental = {r['incremental_effect']:.4f}, "
          f"exposed_rate = {r['exposed_rate']:.3f}, unexposed = {r['unexposed_rate']:.3f}")

小结
    

  • 产品分析中的核心问题本质上都是因果问题，不同场景需要不同的因果推断方法
    • 

  • 用户留存的因果分析需要处理选择偏差和反向因果，IPW 和双重稳健方法是推荐工具
    • 

  • 定价因果分析面临内生性问题，工具变量（成本、汇率）和断点回归是主要解决方案
    • 

  • 多触点归因从规则方法到 Shapley 值再到增量因果分析，因果性逐步增强
    • 

  • 在选择方法时应遵循"能实验就实验，不能实验找自然实验，都没有则用协变量调整"的优先级
    • 

  • 任何观测数据的因果分析都应配合敏感性分析评估结论的稳健性
    •
