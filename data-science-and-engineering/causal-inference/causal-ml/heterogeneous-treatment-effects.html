异质处理效应

核心概念

异质处理效应（Heterogeneous Treatment Effects, HTE）研究的是：同一个干预对不同人群的效果有多大差异？传统因果推断聚焦于平均处理效应（ATE），但在精准医疗、个性化推荐、定向营销等场景中，理解"谁获益最多"和"谁可能受害"才是真正有价值的问题。条件平均处理效应（CATE） τ(x)=E[Y(1)−Y(0)∣X=x]\tau(x) = E[Y(1) - Y(0) \mid X = x]τ(x)=E[Y(1)−Y(0)∣X=x] 是 HTE 研究的核心目标。

元学习器
元学习器（Meta-Learners）是估计 CATE 的一类方法，通过将因果问题拆解为标准的监督学习子问题来利用任意机器学习算法的预测能力。


S-Learner：训练一个模型 μ^(x,w)\hat{\mu}(x, w)μ^​(x,w) 同时包含协变量和处理指示，CATE 估计为 τ^(x)=μ^(x,1)−μ^(x,0)\hat{\tau}(x) = \hat{\mu}(x, 1) - \hat{\mu}(x, 0)τ^(x)=μ^​(x,1)−μ^​(x,0)。简单但在处理效应较小时容易被正则化消除。


T-Learner：分别为处理组和对照组训练两个模型 μ^1(x)\hat{\mu}_1(x)μ^​1​(x) 和 μ^0(x)\hat{\mu}_0(x)μ^​0​(x)，CATE 估计为差值。充分建模了每组的结果函数，但在数据不平衡时效果不佳。


X-Learner：在 T-Learner 基础上利用"虚拟处理效应"进行交叉拟合，特别适合处理组远少于对照组的场景。


import numpy as np
from sklearn.ensemble import GradientBoostingRegressor

class SLearner:
    def __init__(self, base_model=None):
        self.model = base_model or GradientBoostingRegressor(n_estimators=100)

    def fit(self, X, W, Y):
        XW = np.column_stack([X, W])
        self.model.fit(XW, Y)
        return self

    def predict_cate(self, X):
        X1 = np.column_stack([X, np.ones(len(X))])
        X0 = np.column_stack([X, np.zeros(len(X))])
        return self.model.predict(X1) - self.model.predict(X0)

class TLearner:
    def __init__(self, base_model_class=None):
        model_cls = base_model_class or GradientBoostingRegressor
        self.model_1 = model_cls(n_estimators=100)
        self.model_0 = model_cls(n_estimators=100)

    def fit(self, X, W, Y):
        self.model_1.fit(X[W == 1], Y[W == 1])
        self.model_0.fit(X[W == 0], Y[W == 0])
        return self

    def predict_cate(self, X):
        return self.model_1.predict(X) - self.model_0.predict(X)

class XLearner:
    def __init__(self, base_model_class=None):
        model_cls = base_model_class or GradientBoostingRegressor
        self.mu_0 = model_cls(n_estimators=100)
        self.mu_1 = model_cls(n_estimators=100)
        self.tau_0 = model_cls(n_estimators=100)
        self.tau_1 = model_cls(n_estimators=100)
        self.ps_model = None

    def fit(self, X, W, Y):
        self.mu_0.fit(X[W == 0], Y[W == 0])
        self.mu_1.fit(X[W == 1], Y[W == 1])

        d1 = Y[W == 1] - self.mu_0.predict(X[W == 1])
        d0 = self.mu_1.predict(X[W == 0]) - Y[W == 0]

        self.tau_1.fit(X[W == 1], d1)
        self.tau_0.fit(X[W == 0], d0)

        from sklearn.linear_model import LogisticRegression
        self.ps_model = LogisticRegression(max_iter=1000)
        self.ps_model.fit(X, W)

        return self

    def predict_cate(self, X):
        ps = self.ps_model.predict_proba(X)[:, 1]
        tau_1_pred = self.tau_1.predict(X)
        tau_0_pred = self.tau_0.predict(X)
        return ps * tau_0_pred + (1 - ps) * tau_1_pred

np.random.seed(42)
n = 5000
X = np.random.normal(0, 1, (n, 5))
true_cate = 2.0 + 1.5 * X[:, 0] - 0.5 * X[:, 1]
W = np.random.binomial(1, 0.5, n)
Y = 1 + X[:, 0] + 0.5 * X[:, 1] + true_cate * W + np.random.normal(0, 1, n)

s_learner = SLearner().fit(X, W, Y)
t_learner = TLearner().fit(X, W, Y)
x_learner = XLearner().fit(X, W, Y)

X_test = np.random.normal(0, 1, (1000, 5))
true_test = 2.0 + 1.5 * X_test[:, 0] - 0.5 * X_test[:, 1]

for name, model in [('S-Learner', s_learner), ('T-Learner', t_learner), ('X-Learner', x_learner)]:
    pred = model.predict_cate(X_test)
    mse = np.mean((pred - true_test)**2)
    corr = np.corrcoef(pred, true_test)[0, 1]
    print(f"{name}: MSE={mse:.3f}, Correlation={corr:.3f}")

因果森林
因果森林（Causal Forest）是 Athey 和 Imbens 提出的专门用于 CATE 估计的非参数方法。它基于随机森林，但修改了分裂准则：不是最大化预测准确度，而是最大化处理效应的异质性。


import numpy as np

class SimpleCausalTree:
    def __init__(self, max_depth=3, min_samples_leaf=50):
        self.max_depth = max_depth
        self.min_samples_leaf = min_samples_leaf
        self.tree = None

    def _treatment_effect(self, Y, W):
        treated = Y[W == 1]
        control = Y[W == 0]
        if len(treated) < 2 or len(control) < 2:
            return 0.0
        return np.mean(treated) - np.mean(control)

    def _causal_criterion(self, Y_left, W_left, Y_right, W_right):
        n_left = len(Y_left)
        n_right = len(Y_right)
        n_total = n_left + n_right

        tau_left = self._treatment_effect(Y_left, W_left)
        tau_right = self._treatment_effect(Y_right, W_right)

        criterion = (n_left / n_total) * tau_left**2 + (n_right / n_total) * tau_right**2
        return criterion

    def _find_best_split(self, X, Y, W):
        best_score = -np.inf
        best_feature = None
        best_threshold = None

        for feature in range(X.shape[1]):
            thresholds = np.percentile(X[:, feature], np.linspace(10, 90, 20))
            for threshold in thresholds:
                left_mask = X[:, feature] <= threshold
                right_mask = ~left_mask

                if (np.sum(left_mask) < self.min_samples_leaf or
                    np.sum(right_mask) < self.min_samples_leaf):
                    continue

                if (np.sum(W[left_mask]) < 5 or np.sum(1-W[left_mask]) < 5 or
                    np.sum(W[right_mask]) < 5 or np.sum(1-W[right_mask]) < 5):
                    continue

                score = self._causal_criterion(
                    Y[left_mask], W[left_mask], Y[right_mask], W[right_mask]
                )

                if score > best_score:
                    best_score = score
                    best_feature = feature
                    best_threshold = threshold

        return best_feature, best_threshold

    def _build_tree(self, X, Y, W, depth=0):
        tau = self._treatment_effect(Y, W)

        if depth >= self.max_depth or len(Y) < 2 * self.min_samples_leaf:
            return {'tau': tau, 'n': len(Y), 'n_treated': np.sum(W)}

        feature, threshold = self._find_best_split(X, Y, W)

        if feature is None:
            return {'tau': tau, 'n': len(Y), 'n_treated': np.sum(W)}

        left_mask = X[:, feature] <= threshold

        return {
            'feature': feature,
            'threshold': threshold,
            'left': self._build_tree(X[left_mask], Y[left_mask], W[left_mask], depth + 1),
            'right': self._build_tree(X[~left_mask], Y[~left_mask], W[~left_mask], depth + 1),
        }

    def fit(self, X, Y, W):
        self.tree = self._build_tree(X, Y, W)
        return self

    def predict(self, X):
        return np.array([self._predict_single(x, self.tree) for x in X])

    def _predict_single(self, x, node):
        if 'feature' not in node:
            return node['tau']
        if x[node['feature']] <= node['threshold']:
            return self._predict_single(x, node['left'])
        return self._predict_single(x, node['right'])

np.random.seed(42)
n = 3000
X = np.random.normal(0, 1, (n, 5))
true_cate = 2.0 + 1.5 * X[:, 0] - 0.5 * X[:, 1]
W = np.random.binomial(1, 0.5, n)
Y = X[:, 0] + true_cate * W + np.random.normal(0, 1, n)

ct = SimpleCausalTree(max_depth=3, min_samples_leaf=100)
ct.fit(X, Y, W)
pred = ct.predict(X[:100])
print(f"Causal Tree CATE correlation: {np.corrcoef(pred, true_cate[:100])[0, 1]:.3f}")

策略评估与最优处理规则
CATE 估计的最终目标是制定最优处理规则（Optimal Treatment Rule）：d∗(x)=1[τ^(x)>0]d^*(x) = \mathbb{1}[\hat{\tau}(x) > 0]d∗(x)=1[τ^(x)>0]，对 CATE > 0 的个体实施处理，对 CATE < 0 的个体不处理。策略评估（Policy Evaluation）用于估计实施特定规则的预期总价值。


import numpy as np

class PolicyEvaluator:
    def __init__(self):
        pass

    def evaluate_policy(self, Y, W, ps, policy):
        ipw_value = np.mean(
            policy * W * Y / ps + (1 - policy) * (1 - W) * Y / (1 - ps)
        )
        return ipw_value

    def optimal_policy_value(self, cate_predictions, Y, W, ps):
        policy = (cate_predictions > 0).astype(float)
        value = self.evaluate_policy(Y, W, ps, policy)

        treat_all_value = self.evaluate_policy(Y, W, ps, np.ones(len(Y)))
        treat_none_value = self.evaluate_policy(Y, W, ps, np.zeros(len(Y)))

        return {
            'optimal_policy_value': value,
            'treat_all_value': treat_all_value,
            'treat_none_value': treat_none_value,
            'lift_over_treat_all': value - treat_all_value,
            'pct_treated': np.mean(policy),
        }

np.random.seed(42)
n = 5000
X = np.random.normal(0, 1, (n, 3))
true_cate = 1.0 + 2.0 * X[:, 0]
W = np.random.binomial(1, 0.5, n)
Y = X[:, 0] + true_cate * W + np.random.normal(0, 1, n)

evaluator = PolicyEvaluator()
ps = np.full(n, 0.5)
result = evaluator.optimal_policy_value(true_cate, Y, W, ps)

print(f"Optimal policy value: {result['optimal_policy_value']:.3f}")
print(f"Treat all value: {result['treat_all_value']:.3f}")
print(f"Treat none value: {result['treat_none_value']:.3f}")
print(f"Lift over treat all: {result['lift_over_treat_all']:.3f}")
print(f"Pct treated: {result['pct_treated']:.1%}")

小结
    

  • CATE 估计回答"谁获益最多"的问题，是精准化干预的基础
    • 

  • 元学习器（S/T/X-Learner）将因果问题转化为标准监督学习，可利用任意 ML 算法
    • 

  • 因果森林通过修改分裂准则直接优化效应异质性，Honesty 保证统计推断有效
    • 

  • 最优处理规则基于 CATE 估计实现个性化决策，策略评估量化规则的预期价值
    • 

  • 在实际应用中，应比较多种 CATE 估计方法并通过交叉验证选择最优模型
    • 

  • 异质效应分析在个性化推荐、定向营销、精准医疗等场景有直接的商业价值
