合成控制法

核心概念

合成控制法（Synthetic Control Method, SCM）是评估宏观干预效应的强大工具。当处理单位只有一个（如某个国家实施了政策、某个城市推出了新项目）时，传统的 DID 方法因对照组选择困难而不可靠。SCM 通过对多个未受干预的对照单位进行加权组合，构造一个"合成版"的处理单位作为反事实。处理效应就是实际观测值与合成对照之间的差距。

经典合成控制法
经典 SCM 通过最小化事前期间处理单位和合成对照在结果变量和预测变量上的差距来求解权重。权重非负且和为 1，使得合成对照是供体池单位的凸组合。


import numpy as np
from scipy.optimize import minimize

class SyntheticControl:
    def __init__(self):
        self.weights = None
        self.treated_pre = None
        self.treated_post = None
        self.synthetic_pre = None
        self.synthetic_post = None

    def fit(self, Y_donors, Y_treated, T0, X_donors=None, X_treated=None):
        J = Y_donors.shape[0]

        Y_donors_pre = Y_donors[:, :T0]
        Y_treated_pre = Y_treated[:T0]

        if X_donors is not None and X_treated is not None:
            Z_donors = np.column_stack([Y_donors_pre.T, X_donors.T]).T
            Z_treated = np.concatenate([Y_treated_pre, X_treated])
        else:
            Z_donors = Y_donors_pre
            Z_treated = Y_treated_pre

        def objective(w):
            synthetic = Z_donors.T @ w
            return np.sum((Z_treated - synthetic)**2)

        constraints = {'type': 'eq', 'fun': lambda w: np.sum(w) - 1}
        bounds = [(0, 1)] * J
        w0 = np.ones(J) / J

        result = minimize(objective, w0, bounds=bounds, constraints=constraints, method='SLSQP')
        self.weights = result.x

        self.treated_pre = Y_treated_pre
        self.treated_post = Y_treated[T0:]
        self.synthetic_pre = Y_donors_pre.T @ self.weights
        self.synthetic_post = Y_donors[:, T0:].T @ self.weights

        return self

    def treatment_effect(self):
        return self.treated_post - self.synthetic_post

    def pre_fit_quality(self):
        rmspe = np.sqrt(np.mean((self.treated_pre - self.synthetic_pre)**2))
        mae = np.mean(np.abs(self.treated_pre - self.synthetic_pre))
        r2 = 1 - np.sum((self.treated_pre - self.synthetic_pre)**2) / \
             np.sum((self.treated_pre - np.mean(self.treated_pre))**2)
        return {'rmspe': rmspe, 'mae': mae, 'r2': r2}

    def summary(self):
        effects = self.treatment_effect()
        quality = self.pre_fit_quality()
        top_donors = np.argsort(-self.weights)[:5]

        return {
            'avg_effect': np.mean(effects),
            'effect_by_period': effects,
            'pre_rmspe': quality['rmspe'],
            'pre_r2': quality['r2'],
            'top_donors': [(i, self.weights[i]) for i in top_donors if self.weights[i] > 0.01],
        }

np.random.seed(42)
T = 30
T0 = 20
J = 15

unit_effects = np.random.normal(0, 3, J + 1)
common_trend = np.cumsum(np.random.normal(0.5, 1, T))
Y_all = unit_effects[:, None] + common_trend[None, :] + np.random.normal(0, 1, (J + 1, T))

true_effect = 5.0
Y_all[0, T0:] += true_effect + np.arange(T - T0) * 0.5

Y_treated = Y_all[0]
Y_donors = Y_all[1:]

sc = SyntheticControl()
sc.fit(Y_donors, Y_treated, T0)
summary = sc.summary()

print(f"True average effect: {true_effect + (T-T0-1)*0.5/2:.1f}")
print(f"SC average effect: {summary['avg_effect']:.3f}")
print(f"Pre-treatment R²: {summary['pre_r2']:.3f}")
print(f"Pre-treatment RMSPE: {summary['pre_rmspe']:.3f}")

推断方法
合成控制法的统计推断不依赖传统的渐近理论，而是使用置换检验（Permutation Inference）。核心思想是：如果处理效应不存在，处理单位与每个供体单位的行为应该没有系统差异。


import numpy as np

def placebo_test(Y_all, T0, treated_idx=0):
    J_plus_1, T = Y_all.shape

    sc_treated = SyntheticControl()
    Y_treated = Y_all[treated_idx]
    Y_donors = np.delete(Y_all, treated_idx, axis=0)
    sc_treated.fit(Y_donors, Y_treated, T0)
    treated_effect = sc_treated.treatment_effect()
    treated_rmspe_pre = sc_treated.pre_fit_quality()['rmspe']
    treated_rmspe_post = np.sqrt(np.mean(treated_effect**2))
    treated_ratio = treated_rmspe_post / treated_rmspe_pre if treated_rmspe_pre > 0 else float('inf')

    placebo_ratios = []
    for j in range(J_plus_1):
        if j == treated_idx:
            continue
        sc_placebo = SyntheticControl()
        Y_placebo = Y_all[j]
        Y_placebo_donors = np.delete(Y_all, j, axis=0)
        sc_placebo.fit(Y_placebo_donors, Y_placebo, T0)

        placebo_effect = sc_placebo.treatment_effect()
        rmspe_pre = sc_placebo.pre_fit_quality()['rmspe']
        rmspe_post = np.sqrt(np.mean(placebo_effect**2))
        ratio = rmspe_post / rmspe_pre if rmspe_pre > 0 else 0
        placebo_ratios.append(ratio)

    p_value = np.mean([r >= treated_ratio for r in placebo_ratios])

    return {
        'treated_ratio': treated_ratio,
        'placebo_ratios': placebo_ratios,
        'p_value': p_value,
        'rank': sum(1 for r in placebo_ratios if r >= treated_ratio) + 1,
        'total': len(placebo_ratios) + 1,
    }

T = 30
T0 = 20
J = 15

Y_all = unit_effects[:, None] + common_trend[None, :] + np.random.normal(0, 1, (J + 1, T))
Y_all[0, T0:] += 5.0

result = placebo_test(Y_all, T0, treated_idx=0)
print(f"Treated RMSPE ratio: {result['treated_ratio']:.3f}")
print(f"Rank: {result['rank']} / {result['total']}")
print(f"p-value: {result['p_value']:.3f}")

增广合成控制法
传统 SCM 要求合成对照在事前期间精确匹配处理单位，但当供体池不足以精确匹配时（外推问题），估计可能严重有偏。增广合成控制法（Augmented SCM, ASCM）通过结合回归调整和合成控制来修正偏差。


import numpy as np
from sklearn.linear_model import Ridge

class AugmentedSyntheticControl:
    def __init__(self, ridge_alpha=1.0):
        self.ridge_alpha = ridge_alpha
        self.sc_weights = None
        self.ridge_model = None

    def fit(self, Y_donors, Y_treated, T0):
        J = Y_donors.shape[0]

        sc = SyntheticControl()
        sc.fit(Y_donors, Y_treated, T0)
        self.sc_weights = sc.weights

        Y_donors_pre = Y_donors[:, :T0]
        Y_donors_post = Y_donors[:, T0:]

        self.ridge_model = Ridge(alpha=self.ridge_alpha)
        self.ridge_model.fit(Y_donors_pre.T, Y_donors_post.T)

        sc_pre_residual = Y_treated[:T0] - Y_donors_pre.T @ self.sc_weights

        bias_correction = self.ridge_model.predict(sc_pre_residual.reshape(1, -1)).flatten()

        sc_post = Y_donors_post.T @ self.sc_weights
        ascm_post = sc_post + bias_correction

        self.treated_post = Y_treated[T0:]
        self.ascm_post = ascm_post
        self.sc_post = sc_post
        self.bias_correction = bias_correction

        return self

    def treatment_effect(self):
        return self.treated_post - self.ascm_post

    def compare_with_sc(self):
        sc_effect = self.treated_post - self.sc_post
        ascm_effect = self.treatment_effect()
        return {
            'sc_avg_effect': np.mean(sc_effect),
            'ascm_avg_effect': np.mean(ascm_effect),
            'avg_bias_correction': np.mean(self.bias_correction),
        }

np.random.seed(42)
T = 30
T0 = 20
J = 10
unit_effects = np.random.normal(0, 5, J + 1)
common_trend = np.cumsum(np.random.normal(0.3, 0.8, T))
Y_all = unit_effects[:, None] + common_trend[None, :] + np.random.normal(0, 1, (J + 1, T))
Y_all[0, T0:] += 4.0

ascm = AugmentedSyntheticControl(ridge_alpha=1.0)
ascm.fit(Y_all[1:], Y_all[0], T0)
comparison = ascm.compare_with_sc()
print(f"SC average effect: {comparison['sc_avg_effect']:.3f}")
print(f"ASCM average effect: {comparison['ascm_avg_effect']:.3f}")
print(f"Average bias correction: {comparison['avg_bias_correction']:.3f}")

小结
    

  • 合成控制法通过加权组合供体单位构造反事实，适用于只有一个或少量处理单位的宏观因果推断
    • 

  • 权重非负且和为 1，使合成对照是供体池的凸组合，避免了外推
    • 

  • 事前拟合质量（RMSPE、R²）是评估 SCM 可信度的关键指标
    • 

  • 安慰剂检验通过对每个供体做 SCM 来构造置换分布，提供统计推断
    • 

  • 增广 SCM 通过回归偏差校正改善事前匹配不完美时的估计质量
    • 

  • 互联网产品中城市级策略评估、国家级产品上线等场景是 SCM 的理想应用
