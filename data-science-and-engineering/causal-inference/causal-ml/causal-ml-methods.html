因果推断与机器学习

核心概念

因果推断与机器学习的融合是数据科学领域最活跃的前沿之一。传统因果推断依赖参数化假设和低维模型，传统机器学习聚焦预测而非因果。二者的结合——因果机器学习（Causal ML）——利用 ML 的灵活性处理高维数据和复杂非线性关系，同时保持因果推断的统计严谨性。这一融合催生了一系列新方法，如双重机器学习（Double ML）、因果森林、深度因果模型等。

双重机器学习
双重机器学习（Double Machine Learning, DML）由 Chernozhukov 等人提出，是因果 ML 的基石方法。核心思想是：将因果效应的估计分解为两个"nuisance"预测问题（结果建模和处理建模），用任意 ML 方法估计这两个nuisance函数，然后通过 Neyman 正交化保证最终的因果效应估计具有 n\sqrt{n}n​ 一致性和渐近正态性。


import numpy as np
from sklearn.ensemble import GradientBoostingRegressor, GradientBoostingClassifier
from sklearn.model_selection import KFold
from scipy import stats

class DoubleML:
    def __init__(self, outcome_model=None, treatment_model=None, n_folds=5):
        self.outcome_model = outcome_model or GradientBoostingRegressor(n_estimators=100)
        self.treatment_model = treatment_model or GradientBoostingRegressor(n_estimators=100)
        self.n_folds = n_folds

    def fit(self, X, W, Y):
        n = len(Y)
        kf = KFold(n_splits=self.n_folds, shuffle=True, random_state=42)

        m_hat = np.zeros(n)
        e_hat = np.zeros(n)

        for train_idx, test_idx in kf.split(X):
            from sklearn.base import clone

            outcome_m = clone(self.outcome_model)
            outcome_m.fit(X[train_idx], Y[train_idx])
            m_hat[test_idx] = outcome_m.predict(X[test_idx])

            treatment_m = clone(self.treatment_model)
            treatment_m.fit(X[train_idx], W[train_idx])
            e_hat[test_idx] = treatment_m.predict(X[test_idx])

        V_hat = W - e_hat
        Y_tilde = Y - m_hat

        theta = np.sum(V_hat * Y_tilde) / np.sum(V_hat * W)

        psi = V_hat * (Y_tilde - theta * W)
        J = -np.mean(V_hat * W)

        sigma2 = np.mean(psi**2) / J**2
        se = np.sqrt(sigma2 / n)

        t_stat = theta / se
        p_value = 2 * (1 - stats.norm.cdf(abs(t_stat)))

        return {
            'theta': theta,
            'se': se,
            't_stat': t_stat,
            'p_value': p_value,
            'ci_95': (theta - 1.96 * se, theta + 1.96 * se),
        }

np.random.seed(42)
n = 5000
p = 20
X = np.random.normal(0, 1, (n, p))
W = (0.5 * X[:, 0] + 0.3 * X[:, 1] + np.random.normal(0, 1, n) > 0).astype(float)
true_effect = 3.0
Y = (np.sin(X[:, 0]) + X[:, 1]**2 + 0.5 * X[:, 2] +
     true_effect * W + np.random.normal(0, 1, n))

dml = DoubleML()
result = dml.fit(X, W, Y)
print(f"True effect: {true_effect:.3f}")
print(f"DML estimate: {result['theta']:.3f} (SE: {result['se']:.3f})")
print(f"95% CI: ({result['ci_95'][0]:.3f}, {result['ci_95'][1]:.3f})")
print(f"p-value: {result['p_value']:.6f}")

深度学习因果模型
深度学习在因果推断中的应用主要集中在两个方向：学习复杂的处理效应函数（representation learning for treatment effects）和处理高维/非结构化数据（如文本、图像作为混杂变量或结果变量）。


CEVAE（Causal Effect Variational Autoencoder） 使用变分自编码器学习潜在混杂因素的表征，即使混杂因素不可直接观测，也能从代理变量中恢复。


TARNet / CFRNet 使用共享表征层学习协变量的深度表征，然后分别为处理组和对照组训练独立的预测头。IPM（Integral Probability Metric）正则化确保两组表征的分布接近。


import numpy as np

class SimpleTARNet:
    def __init__(self, hidden_dim=64, repr_dim=32, lr=0.001):
        self.hidden_dim = hidden_dim
        self.repr_dim = repr_dim
        self.lr = lr
        self.params = {}

    def _init_params(self, input_dim):
        np.random.seed(42)
        scale = 0.1
        self.params = {
            'W_repr': np.random.randn(input_dim, self.repr_dim) * scale,
            'b_repr': np.zeros(self.repr_dim),
            'W_head0': np.random.randn(self.repr_dim, 1) * scale,
            'b_head0': np.zeros(1),
            'W_head1': np.random.randn(self.repr_dim, 1) * scale,
            'b_head1': np.zeros(1),
        }

    def _relu(self, x):
        return np.maximum(0, x)

    def _forward(self, X, W):
        repr_layer = self._relu(X @ self.params['W_repr'] + self.params['b_repr'])

        y0_pred = repr_layer @ self.params['W_head0'] + self.params['b_head0']
        y1_pred = repr_layer @ self.params['W_head1'] + self.params['b_head1']

        y_pred = W[:, None] * y1_pred + (1 - W[:, None]) * y0_pred

        return y_pred.flatten(), y0_pred.flatten(), y1_pred.flatten(), repr_layer

    def fit(self, X, W, Y, n_epochs=100):
        self._init_params(X.shape[1])

        for epoch in range(n_epochs):
            y_pred, y0, y1, repr_layer = self._forward(X, W)
            loss = np.mean((Y - y_pred)**2)

            error = y_pred - Y

            for key in self.params:
                if 'head0' in key and 'W' in key:
                    grad = repr_layer[W == 0].T @ error[W == 0, None] / max(np.sum(W == 0), 1)
                    self.params[key] -= self.lr * grad
                elif 'head1' in key and 'W' in key:
                    grad = repr_layer[W == 1].T @ error[W == 1, None] / max(np.sum(W == 1), 1)
                    self.params[key] -= self.lr * grad

        return self

    def predict_cate(self, X):
        W_dummy = np.zeros(len(X))
        _, y0, y1, _ = self._forward(X, W_dummy)
        return y1 - y0

np.random.seed(42)
n = 3000
X = np.random.normal(0, 1, (n, 10))
true_cate = 2.0 + 1.5 * np.sin(X[:, 0]) + 0.5 * X[:, 1]
W = np.random.binomial(1, 0.5, n)
Y = X[:, 0] + 0.5 * X[:, 1]**2 + true_cate * W + np.random.normal(0, 0.5, n)

model = SimpleTARNet(repr_dim=16, lr=0.001)
model.fit(X, W, Y, n_epochs=200)
pred_cate = model.predict_cate(X[:500])
corr = np.corrcoef(pred_cate, true_cate[:500])[0, 1]
print(f"TARNet CATE correlation: {corr:.3f}")

因果表征学习
因果表征学习（Causal Representation Learning）是因果 ML 最前沿的方向，目标是从高维观测数据中学习因果变量的低维表征。传统因果推断假设变量已经被正确定义和测量，但在实际应用中（如从图像、文本中提取因果变量），这一假设并不成立。


import numpy as np

class CausalVAESimple:
    def __init__(self, latent_dim=5, observed_dim=50):
        self.latent_dim = latent_dim
        self.observed_dim = observed_dim

    def simulate_causal_generative(self, n=5000):
        np.random.seed(42)

        z1 = np.random.normal(0, 1, n)
        z2 = 0.5 * z1 + np.random.normal(0, 0.5, n)
        z3 = np.random.normal(0, 1, n)
        z4 = 0.3 * z2 + 0.7 * z3 + np.random.normal(0, 0.3, n)
        z5 = 0.5 * z4 + np.random.normal(0, 0.5, n)

        Z = np.column_stack([z1, z2, z3, z4, z5])

        mixing_matrix = np.random.randn(self.observed_dim, self.latent_dim)
        X = np.tanh(Z @ mixing_matrix.T) + np.random.normal(0, 0.1, (n, self.observed_dim))

        return X, Z

    def evaluate_disentanglement(self, Z_true, Z_learned):
        from scipy.stats import pearsonr

        n_true = Z_true.shape[1]
        n_learned = Z_learned.shape[1]

        correlation_matrix = np.zeros((n_true, n_learned))
        for i in range(n_true):
            for j in range(n_learned):
                r, _ = pearsonr(Z_true[:, i], Z_learned[:, j])
                correlation_matrix[i, j] = abs(r)

        max_corr_per_true = np.max(correlation_matrix, axis=1)

        return {
            'correlation_matrix': correlation_matrix,
            'avg_max_correlation': np.mean(max_corr_per_true),
            'min_max_correlation': np.min(max_corr_per_true),
        }

model = CausalVAESimple()
X, Z_true = model.simulate_causal_generative()
print(f"Observed data shape: {X.shape}")
print(f"True latent shape: {Z_true.shape}")

from sklearn.decomposition import PCA
pca = PCA(n_components=5)
Z_pca = pca.fit_transform(X)

result = model.evaluate_disentanglement(Z_true, Z_pca)
print(f"PCA avg max correlation: {result['avg_max_correlation']:.3f}")
print(f"PCA min max correlation: {result['min_max_correlation']:.3f}")
因果表征学习的核心挑战在于可识别性：从观测数据中恢复因果变量在一般条件下是不可能的（需要额外的归纳偏置）。最近的理论进展表明，在特定条件下（如非线性 ICA、时间序列、多视角数据），因果变量可以被唯一识别。

小结
    

  • 因果 ML 结合了 ML 的预测能力和因果推断的统计严谨性，是处理高维因果问题的现代方法
    • 

  • 双重机器学习通过 cross-fitting 和 Neyman 正交化，使用任意 ML 算法估计因果效应并保持 n\sqrt{n}n​ 一致性
    • 

  • 深度因果模型（TARNet、CFRNet、CEVAE）利用神经网络学习复杂的表征和处理效应函数
    • 

  • 因果表征学习致力于从高维观测数据中恢复因果变量，是最前沿但也最困难的方向
    • 

  • EconML、CausalML、DoWhy 等开源工具降低了因果 ML 的应用门槛
    • 

  • 选择方法时应根据数据特点、假设强度和应用需求权衡，没有单一最优方法
    •
