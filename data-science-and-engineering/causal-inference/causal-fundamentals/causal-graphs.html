因果图

核心概念

因果图（Causal Graph），特别是有向无环图（DAG, Directed Acyclic Graph），是 Pearl 因果推断框架的核心工具。DAG 用节点表示变量，有向边表示因果关系（A→B 表示 A 是 B 的直接原因）。因果图不仅是一种可视化工具，更是一套严格的数学语言，用于判断哪些变量需要控制、哪些不能控制、以及因果效应是否可从观测数据中识别。

d-分离与条件独立性
d-分离（d-separation）是因果图中判断条件独立性的核心算法。给定一组条件化变量 S，如果 X 和 Y 之间的所有路径都被 S 阻断，则 X 和 Y 关于 S d-分离，即 X⊥Y∣SX \perp Y \mid SX⊥Y∣S。


路径阻断规则：


    

  1. 链路径 X → M → Y：条件化 M 阻断路径
    2. 

  2. 叉路径 X ← Z → Y：条件化 Z 阻断路径
    3. 

  3. 碰撞路径 X → Z ← Y：默认阻断；条件化 Z 或 Z 的后代打开路径
    4. 



from itertools import product

class CausalDAG:
    def __init__(self):
        self.edges = {}
        self.nodes = set()

    def add_edge(self, parent, child):
        self.nodes.add(parent)
        self.nodes.add(child)
        if parent not in self.edges:
            self.edges[parent] = set()
        self.edges[parent].add(child)

    def parents(self, node):
        return {p for p, children in self.edges.items() if node in children}

    def children(self, node):
        return self.edges.get(node, set())

    def descendants(self, node):
        desc = set()
        queue = list(self.children(node))
        while queue:
            current = queue.pop(0)
            if current not in desc:
                desc.add(current)
                queue.extend(self.children(current))
        return desc

    def is_collider(self, node, path):
        idx = path.index(node)
        if idx == 0 or idx == len(path) - 1:
            return False
        prev_node = path[idx - 1]
        next_node = path[idx + 1]
        return node in self.children(prev_node) or prev_node in self.parents(node) and \
               node in self.children(next_node) or next_node in self.parents(node)

d-分离的直觉是：信息可以沿着因果路径"流动"，条件化某些变量会阻断或打开这些路径。关键的反直觉点是碰撞节点：不应该条件化碰撞节点，否则会在原本独立的变量之间创造虚假的关联。


import numpy as np
from scipy import stats

def demonstrate_collider_bias(n=10000):
    np.random.seed(42)

    talent = np.random.normal(0, 1, n)
    effort = np.random.normal(0, 1, n)

    score = talent + effort + np.random.normal(0, 0.5, n)

    r_no_condition, _ = stats.pearsonr(talent, effort)
    print(f"Unconditional r(talent, effort) = {r_no_condition:.3f}")
    print("(Should be ~0, as they are independent)")

    admitted = score > np.percentile(score, 70)
    r_conditioned, _ = stats.pearsonr(talent[admitted], effort[admitted])
    print(f"\nConditional r(talent, effort | admitted) = {r_conditioned:.3f}")
    print("(Negative! Conditioning on collider creates spurious association)")

后门准则与调整公式
后门准则（Backdoor Criterion）是 Pearl 提出的识别因果效应的核心定理。给定处理变量 X 和结果变量 Y，一组变量 S 满足后门准则当且仅当：


    

  1. S 中没有 X 的后代
    2. 

  2. S 阻断了所有从 X 到 Y 的后门路径（即经过指向 X 的箭头的路径）
    3. 



class BackdoorCriterion:
    def __init__(self, dag):
        self.dag = dag

    def find_backdoor_paths(self, treatment, outcome):
        paths = []
        self._find_paths(treatment, outcome, [treatment], paths, direction='backward')
        return [p for p in paths if len(p) > 1 and self._is_backdoor(p, treatment)]

    def _find_paths(self, current, target, path, all_paths, direction='any'):
        if current == target and len(path) > 1:
            all_paths.append(list(path))
            return

        neighbors = set()
        neighbors.update(self.dag.parents(current))
        neighbors.update(self.dag.children(current))

        for neighbor in neighbors:
            if neighbor not in path:
                path.append(neighbor)
                self._find_paths(neighbor, target, path, all_paths)
                path.pop()

    def _is_backdoor(self, path, treatment):
        if len(path) < 2:
            return False
        second_node = path[1]
        return treatment in self.dag.children(second_node) or second_node in self.dag.parents(treatment)

    def check_adjustment_set(self, treatment, outcome, adjustment_set):
        descendants_x = self.dag.descendants(treatment)
        for var in adjustment_set:
            if var in descendants_x:
                return False, f"{var} is a descendant of {treatment}"

        return True, "Valid adjustment set"

因果图的构建与验证
在实际应用中，因果图的构建依赖领域知识、数据探索和假设检验的结合。正确的因果图是可信因果推断的前提。


import numpy as np
from itertools import combinations

class CausalDiscovery:
    def __init__(self, data, var_names):
        self.data = data
        self.var_names = var_names
        self.n_vars = len(var_names)

    def pc_skeleton(self, alpha=0.05):
        from scipy import stats

        n = self.data.shape[0]
        adjacency = np.ones((self.n_vars, self.n_vars), dtype=bool)
        np.fill_diagonal(adjacency, False)

        for i, j in combinations(range(self.n_vars), 2):
            r, p = stats.pearsonr(self.data[:, i], self.data[:, j])
            if p > alpha:
                adjacency[i, j] = False
                adjacency[j, i] = False

        separating_sets = {}
        for i, j in combinations(range(self.n_vars), 2):
            if not adjacency[i, j]:
                continue

            possible_separators = [k for k in range(self.n_vars) if k != i and k != j]
            for k in possible_separators:
                r_partial = self._partial_correlation(i, j, [k])
                n_eff = n - 1 - 1
                t_stat = r_partial * np.sqrt(n_eff / (1 - r_partial**2 + 1e-10))
                p_val = 2 * (1 - stats.t.cdf(abs(t_stat), n_eff))
                if p_val > alpha:
                    adjacency[i, j] = False
                    adjacency[j, i] = False
                    separating_sets[(i, j)] = {k}
                    break

        return adjacency, separating_sets

    def _partial_correlation(self, i, j, conditioning):
        from numpy.linalg import lstsq

        X_cond = self.data[:, conditioning]
        if X_cond.ndim == 1:
            X_cond = X_cond.reshape(-1, 1)
        X_cond = np.column_stack([np.ones(len(X_cond)), X_cond])

        res_i = self.data[:, i] - X_cond @ lstsq(X_cond, self.data[:, i], rcond=None)[0]
        res_j = self.data[:, j] - X_cond @ lstsq(X_cond, self.data[:, j], rcond=None)[0]

        from scipy.stats import pearsonr
        r, _ = pearsonr(res_i, res_j)
        return r

np.random.seed(42)
n = 5000
Z = np.random.normal(0, 1, n)
X = 0.7 * Z + np.random.normal(0, 0.5, n)
M = 0.5 * X + np.random.normal(0, 0.5, n)
Y = 0.3 * X + 0.6 * M + 0.4 * Z + np.random.normal(0, 0.5, n)

data = np.column_stack([Z, X, M, Y])
discovery = CausalDiscovery(data, ['Z', 'X', 'M', 'Y'])
adj, sep_sets = discovery.pc_skeleton(alpha=0.01)

print("Discovered skeleton (adjacency):")
for i in range(4):
    for j in range(i+1, 4):
        if adj[i, j]:
            print(f"  {discovery.var_names[i]} --- {discovery.var_names[j]}")

do 算子与干预分布
Pearl 的 do 算子 do(X=x)do(X=x)do(X=x) 形式化了"干预"与"观察"的区别。P(Y∣X=x)P(Y|X=x)P(Y∣X=x) 是观察到 X=x 时 Y 的条件分布，可能包含混杂；P(Y∣do(X=x))P(Y|do(X=x))P(Y∣do(X=x)) 是主动将 X 设定为 x（切断所有指向 X 的箭头）时 Y 的分布，排除了混杂的影响。


图手术（Graph Surgery）的操作是：在因果图中删除所有指向 X 的边，保留 X 发出的边。这在图上对应了物理干预的效果——外部强制 X 取某个值，打断了 X 的自然产生机制，但保留了 X 对下游变量的因果影响。


do-calculus 的三条规则提供了将任意干预分布转化为观测分布的完整算法。当三条规则都无法简化时，因果效应在给定因果图下不可识别。

(Y⊥Z∣X,W)GXˉ(Y \perp Z \mid X, W)_{G_{\bar{X}}}(Y⊥Z∣X,W)GXˉ​​

规则 3（插入/删除干预）
(Y⊥Z∣X,W)GXˉZ(S)ˉ(Y \perp Z \mid X, W)_{G_{\bar{X}\bar{Z(S)}}}(Y⊥Z∣X,W)GXˉZ(S)ˉ​​​

小结
    

  • 因果图（DAG）是因果推断的结构化语言，用有向边表示因果关系，d-分离判断条件独立性
    • 

  • 四种基本结构（链、叉、碰撞、直接因果）各有不同的条件化规则，碰撞节点不应被条件化
    • 

  • 后门准则判断哪些变量需要控制来估计因果效应，调整公式将 do 算子转化为可计算的条件分布
    • 

  • 因果图的构建依赖领域知识和数据验证的结合，PC 算法等发现算法可辅助但不能替代专家判断
    • 

  • do 算子形式化了"干预"与"观察"的区别，图手术通过删除指向处理变量的箭头实现干预
    • 

  • 因果图是连接统计方法与因果问题的桥梁，正确的因果图是可信因果推断的前提
    •
