潜在结果框架

核心概念

潜在结果框架（Potential Outcomes Framework），也称 Rubin 因果模型（RCM），是现代因果推断的基石之一。其核心思想是：每个个体在每种处理条件下都有一个"潜在结果"，因果效应定义为不同处理下潜在结果的差异。对于二元处理（处理/不处理），个体 i 的因果效应为 τi=Yi(1)−Yi(0)\tau_i = Y_i(1) - Y_i(0)τi​=Yi​(1)−Yi​(0)。问题的根本挑战在于反事实（counterfactual）不可观测——我们永远无法同时观察到同一个体接受和不接受处理的结果。

平均处理效应与变体
由于个体因果效应不可观测，因果推断的实际目标是估计群体层面的平均因果效应（Average Treatment Effect, ATE）及其变体。


ATE=E[τi]=E[Yi(1)−Yi(0)]=E[Yi(1)]−E[Yi(0)]ATE = E[\tau_i] = E[Y_i(1) - Y_i(0)] = E[Y_i(1)] - E[Y_i(0)]ATE=E[τi​]=E[Yi​(1)−Yi​(0)]=E[Yi​(1)]−E[Yi​(0)]
import numpy as np
from scipy import stats

def simulate_potential_outcomes(n=10000, ate=2.0, treatment_effect_heterogeneity=1.0):
    np.random.seed(42)

    x = np.random.normal(0, 1, n)

    individual_te = ate + treatment_effect_heterogeneity * x

    y0 = 10 + 3 * x + np.random.normal(0, 2, n)
    y1 = y0 + individual_te

    propensity = 1 / (1 + np.exp(-x))
    w = np.random.binomial(1, propensity)
    y_obs = w * y1 + (1 - w) * y0

    true_ate = np.mean(y1 - y0)
    true_att = np.mean((y1 - y0)[w == 1])
    true_atc = np.mean((y1 - y0)[w == 0])

    naive_ate = np.mean(y_obs[w == 1]) - np.mean(y_obs[w == 0])

    return {
        'true_ate': true_ate,
        'true_att': true_att,
        'true_atc': true_atc,
        'naive_ate': naive_ate,
        'selection_bias': naive_ate - true_ate,
        'n_treated': np.sum(w),
        'n_control': np.sum(1 - w),
    }

result = simulate_potential_outcomes()
print(f"True ATE: {result['true_ate']:.3f}")
print(f"True ATT: {result['true_att']:.3f}")
print(f"True ATC: {result['true_atc']:.3f}")
print(f"Naive estimate: {result['naive_ate']:.3f}")
print(f"Selection bias: {result['selection_bias']:+.3f}")

可识别性假设
从观测数据中识别因果效应需要三个关键假设，统称为"强可忽略处理分配"假设（Strong Ignorability）。


假设 1：条件独立性（Unconfoundedness / Ignorability）：


{Y(1),Y(0)}⊥W∣X\{Y(1), Y(0)\} \perp W \mid X{Y(1),Y(0)}⊥W∣X
给定协变量 X，处理分配独立于潜在结果。这意味着在相同 X 值的子群中，处理组和对照组是可比较的。


假设 2：正向性（Positivity / Overlap）：


0<P(W=1∣X=x)<1∀x0 < P(W=1|X=x) < 1 \quad \forall x0<P(W=1∣X=x)<1∀x
对于每个协变量值，都有非零概率被分配到处理组或对照组。如果某些子群完全没有对照（或处理），则无法估计该子群的因果效应。


假设 3：一致性（Consistency / SUTVA）：


Yiobs=Yi(Wi)Y_i^{obs} = Y_i(W_i)Yiobs​=Yi​(Wi​)
实际观察的结果等于对应处理水平下的潜在结果，且不存在个体间的干扰效应。


import numpy as np

def check_positivity(propensity_scores, min_threshold=0.01, max_threshold=0.99):
    n_total = len(propensity_scores)
    n_violations = np.sum(
        (propensity_scores < min_threshold) | (propensity_scores > max_threshold)
    )

    return {
        'n_violations': n_violations,
        'violation_rate': n_violations / n_total,
        'min_ps': np.min(propensity_scores),
        'max_ps': np.max(propensity_scores),
        'p5': np.percentile(propensity_scores, 5),
        'p95': np.percentile(propensity_scores, 95),
        'positivity_ok': n_violations / n_total < 0.05,
    }

def check_overlap(ps_treated, ps_control):
    common_min = max(np.min(ps_treated), np.min(ps_control))
    common_max = min(np.max(ps_treated), np.max(ps_control))

    n_treated_in = np.sum((ps_treated >= common_min) & (ps_treated <= common_max))
    n_control_in = np.sum((ps_control >= common_min) & (ps_control <= common_max))

    return {
        'common_support': (common_min, common_max),
        'treated_in_support': n_treated_in / len(ps_treated),
        'control_in_support': n_control_in / len(ps_control),
    }

np.random.seed(42)
x = np.random.normal(0, 1, 10000)
ps = 1 / (1 + np.exp(-1.5 * x))
w = np.random.binomial(1, ps)

result = check_positivity(ps)
print(f"Positivity violations: {result['n_violations']} ({result['violation_rate']:.1%})")
print(f"PS range: [{result['min_ps']:.4f}, {result['max_ps']:.4f}]")

overlap = check_overlap(ps[w == 1], ps[w == 0])
print(f"Common support: [{overlap['common_support'][0]:.4f}, {overlap['common_support'][1]:.4f}]")
print(f"Treated in support: {overlap['treated_in_support']:.1%}")

倾向得分方法
倾向得分（Propensity Score）e(X)=P(W=1∣X)e(X) = P(W=1|X)e(X)=P(W=1∣X) 将多维协变量压缩为一维，极大简化了混杂控制。Rosenbaum 和 Rubin 证明：如果潜在结果在给定 X 时独立于处理分配，那么在给定 e(X) 时也独立。


import numpy as np
from sklearn.linear_model import LogisticRegression

class PropensityScoreEstimator:
    def __init__(self):
        self.model = LogisticRegression(max_iter=1000)

    def fit(self, X, W):
        self.model.fit(X, W)
        self.ps = self.model.predict_proba(X)[:, 1]
        return self

    def ipw_estimate(self, Y, W, trimming=(0.01, 0.99)):
        ps = np.clip(self.ps, trimming[0], trimming[1])

        weight_treated = 1 / ps
        weight_control = 1 / (1 - ps)

        ate_ipw = (
            np.sum(W * Y * weight_treated) / np.sum(W * weight_treated) -
            np.sum((1 - W) * Y * weight_control) / np.sum((1 - W) * weight_control)
        )
        return ate_ipw

    def doubly_robust_estimate(self, X, Y, W, outcome_model_cls=None):
        from sklearn.linear_model import LinearRegression

        ps = np.clip(self.ps, 0.01, 0.99)

        model_1 = LinearRegression().fit(X[W == 1], Y[W == 1])
        model_0 = LinearRegression().fit(X[W == 0], Y[W == 0])

        mu1_hat = model_1.predict(X)
        mu0_hat = model_0.predict(X)

        dr_ate = np.mean(
            mu1_hat - mu0_hat
            + W * (Y - mu1_hat) / ps
            - (1 - W) * (Y - mu0_hat) / (1 - ps)
        )
        return dr_ate

np.random.seed(42)
n = 5000
X = np.random.normal(0, 1, (n, 3))
true_ps = 1 / (1 + np.exp(-(0.5 * X[:, 0] + 0.3 * X[:, 1])))
W = np.random.binomial(1, true_ps)
true_effect = 2.0
Y = 1 + X[:, 0] + 0.5 * X[:, 1] + true_effect * W + np.random.normal(0, 1, n)

estimator = PropensityScoreEstimator()
estimator.fit(X, W)

naive = np.mean(Y[W == 1]) - np.mean(Y[W == 0])
ipw = estimator.ipw_estimate(Y, W)
dr = estimator.doubly_robust_estimate(X, Y, W)

print(f"True ATE: {true_effect:.3f}")
print(f"Naive: {naive:.3f}")
print(f"IPW: {ipw:.3f}")
print(f"Doubly Robust: {dr:.3f}")

敏感性分析
由于条件独立性假设不可检验，敏感性分析用来评估未观测混杂因素对因果估计的影响程度。如果估计结果对未观测混杂非常敏感，则因果结论不可靠。


import numpy as np

def rosenbaum_sensitivity(gamma_values, observed_effect, se):
    results = []
    for gamma in gamma_values:
        p_upper = 1 / (1 + gamma)
        p_lower = gamma / (1 + gamma)

        adjusted_upper = observed_effect + np.log(gamma) * se
        adjusted_lower = observed_effect - np.log(gamma) * se

        results.append({
            'gamma': gamma,
            'ci_lower': adjusted_lower,
            'ci_upper': adjusted_upper,
            'still_significant': adjusted_lower > 0,
        })
    return results

gammas = [1.0, 1.1, 1.2, 1.5, 2.0, 3.0]
results = rosenbaum_sensitivity(gammas, observed_effect=2.0, se=0.5)
for r in results:
    print(f"Γ={r['gamma']:.1f}: CI=({r['ci_lower']:.3f}, {r['ci_upper']:.3f}), "
          f"significant={r['still_significant']}")
Rosenbaum 敏感性分析的参数 Γ 表示：在最坏情况下，未观测混杂因素最多使得两个协变量相同的个体被分配到处理组的几率比相差 Γ 倍。如果 Γ 需要非常大才能推翻结论，说明结论对未观测混杂是稳健的。

小结
    

  • 潜在结果框架将因果效应定义为同一个体在不同处理下的结果差异，反事实不可观测是核心挑战
    • 

  • ATE、ATT、ATC 是不同的因果量，选择取决于具体的政策问题和决策场景
    • 

  • 条件独立性、正向性和一致性是从观测数据识别因果效应的三个关键假设
    • 

  • 倾向得分将多维协变量压缩为一维，IPW 和双重稳健估计是两种主要的因果效应估计方法
    • 

  • 双重稳健估计只要求倾向得分模型或结果模型之一正确即可得到一致估计，是实践中的推荐方法
    • 

  • 敏感性分析（Rosenbaum Bounds、E-value）评估因果结论对未观测混杂的稳健程度
