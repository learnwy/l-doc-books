RFM 分层与用户画像

核心概念

RFM 模型是电商用户分层最经典也最实用的分析框架。它通过三个维度对用户进行价值评分：Recency（最近一次购买距今多久）、Frequency（购买频率有多高）、Monetary（累计消费金额多大）。三个维度的组合将用户划分为不同的价值群体，每个群体对应差异化的运营策略。RFM 的核心优势在于简洁直观、数据要求低、业务可解释性强。

RFM 模型的底层逻辑基于三个经验法则：最近有购买行为的客户更可能再次购买（时间衰减效应）、购买频率高的客户忠诚度更高、消费金额大的客户贡献价值更高。

RFM 评分与用户分层
RFM 分层的第一步是对每个维度进行评分。常见的评分方式有两种：等分法（按百分位将用户均匀分为 5 等份）和业务阈值法（根据业务经验设定分界点）。等分法更客观，业务阈值法更贴合实际运营需求。


interface RfmScore {
  userId: string;
  recencyDays: number;
  frequencyCount: number;
  monetaryAmount: number;
  rScore: number;
  fScore: number;
  mScore: number;
  rfmSegment: string;
  rfmLabel: string;
}

interface RfmThresholds {
  recency: number[];
  frequency: number[];
  monetary: number[];
}

function calculateRfmScores(
  users: Array<{ userId: string; lastPurchaseDate: Date; purchaseCount: number; totalSpend: number }>,
  asOfDate: Date,
  method: 'quantile' | 'threshold',
  thresholds?: RfmThresholds,
): RfmScore[] {
  const usersWithRecency = users.map((u) => ({
    ...u,
    recencyDays: Math.floor((asOfDate.getTime() - u.lastPurchaseDate.getTime()) / 86400000),
  }));

  if (method === 'quantile') {
    const recencyValues = usersWithRecency.map((u) => u.recencyDays).sort((a, b) => a - b);
    const frequencyValues = users.map((u) => u.purchaseCount).sort((a, b) => a - b);
    const monetaryValues = users.map((u) => u.totalSpend).sort((a, b) => a - b);

    return usersWithRecency.map((u) => {
      const rScore = 5 - Math.floor(getPercentile(u.recencyDays, recencyValues) / 20);
      const fScore = Math.ceil(getPercentile(u.purchaseCount, frequencyValues) / 20);
      const mScore = Math.ceil(getPercentile(u.totalSpend, monetaryValues) / 20);
      return buildRfmScore(u.userId, u.recencyDays, u.purchaseCount, u.totalSpend, rScore, fScore, mScore);
    });
  }

  return usersWithRecency.map((u) => {
    const rScore = scoreByThreshold(u.recencyDays, thresholds!.recency, true);
    const fScore = scoreByThreshold(u.purchaseCount, thresholds!.frequency, false);
    const mScore = scoreByThreshold(u.totalSpend, thresholds!.monetary, false);
    return buildRfmScore(u.userId, u.recencyDays, u.purchaseCount, u.totalSpend, rScore, fScore, mScore);
  });
}

function buildRfmScore(
  userId: string,
  recencyDays: number,
  frequency: number,
  monetary: number,
  rScore: number,
  fScore: number,
  mScore: number,
): RfmScore {
  const segment = `${rScore}${fScore}${mScore}`;
  return {
    userId,
    recencyDays,
    frequencyCount: frequency,
    monetaryAmount: monetary,
    rScore,
    fScore,
    mScore,
    rfmSegment: segment,
    rfmLabel: getRfmLabel(rScore, fScore, mScore),
  };
}

function getRfmLabel(r: number, f: number, m: number): string {
  if (r >= 4 && f >= 4 && m >= 4) return '重要价值客户';
  if (r >= 4 && f <= 2 && m >= 4) return '重要发展客户';
  if (r >= 4 && f >= 4 && m <= 2) return '一般忠诚客户';
  if (r <= 2 && f >= 4 && m >= 4) return '重要保持客户';
  if (r >= 4 && f <= 2 && m <= 2) return '新客户';
  if (r <= 2 && f <= 2 && m <= 2) return '流失客户';
  if (r <= 2 && f >= 3) return '沉默忠诚客户';
  return '一般客户';
}

function getPercentile(value: number, sortedArray: number[]): number {
  const index = sortedArray.findIndex((v) => v >= value);
  return (index / sortedArray.length) * 100;
}

function scoreByThreshold(value: number, thresholds: number[], reverse: boolean): number {
  const sorted = reverse ? [...thresholds].reverse() : thresholds;
  for (let i = 0; i < sorted.length; i++) {
    if (reverse ? value <= sorted[i] : value >= sorted[i]) return 5 - i;
  }
  return 1;
}
用户画像构建
用户画像（User Persona）在 RFM 基础上扩展更丰富的维度，构建对用户的全方位认知。电商用户画像通常包含人口统计属性、消费偏好、行为特征和社交属性四个层面。


interface UserPersona {
  userId: string;
  demographics: {
    ageRange: string;
    gender: string;
    city: string;
    incomeLevel: string;
  };
  consumption: {
    rfmSegment: string;
    preferredCategories: string[];
    preferredBrands: string[];
    priceRange: { min: number; max: number; median: number };
    promotionSensitivity: number;
    averageCartSize: number;
  };
  behavior: {
    preferredDevice: string;
    peakActivityHours: number[];
    averageSessionDuration: number;
    searchQueryCategories: string[];
    browseToCartRatio: number;
    returnRate: number;
  };
  lifecycle: {
    stage: string;
    daysSinceRegistration: number;
    churnRisk: number;
    predictedLtv: number;
    nextPurchasePrediction: Date;
  };
  tags: string[];
}

function buildUserPersona(
  rfm: RfmScore,
  behaviorData: UserBehaviorSummary,
  transactionData: UserTransactionSummary,
): UserPersona {
  const preferredCategories = transactionData.categoryPurchases
    .sort((a, b) => b.count - a.count)
    .slice(0, 5)
    .map((c) => c.category);

  const pricePoints = transactionData.orderAmounts;
  const priceRange = {
    min: Math.min(...pricePoints),
    max: Math.max(...pricePoints),
    median: pricePoints.sort((a, b) => a - b)[Math.floor(pricePoints.length / 2)],
  };

  return {
    userId: rfm.userId,
    demographics: behaviorData.demographics,
    consumption: {
      rfmSegment: rfm.rfmLabel,
      preferredCategories,
      preferredBrands: transactionData.topBrands,
      priceRange,
      promotionSensitivity: transactionData.couponUsageRate,
      averageCartSize: transactionData.avgItemsPerOrder,
    },
    behavior: {
      preferredDevice: behaviorData.primaryDevice,
      peakActivityHours: behaviorData.activeHours,
      averageSessionDuration: behaviorData.avgSessionMinutes,
      searchQueryCategories: behaviorData.searchCategories,
      browseToCartRatio: behaviorData.browseToCartRate,
      returnRate: transactionData.returnRate,
    },
    lifecycle: {
      stage: rfm.rfmLabel.includes('新') ? 'new' : rfm.rfmLabel.includes('流失') ? 'lost' : 'active',
      daysSinceRegistration: behaviorData.accountAgeDays,
      churnRisk: rfm.rScore <= 2 ? 0.8 : rfm.rScore <= 3 ? 0.4 : 0.1,
      predictedLtv: rfm.monetaryAmount * 2.5,
      nextPurchasePrediction: new Date(Date.now() + rfm.recencyDays * 0.8 * 86400000),
    },
    tags: generateTags(rfm),
  };
}

function generateTags(rfm: RfmScore): string[] {
  const tags: string[] = [rfm.rfmLabel];
  if (rfm.mScore >= 4) tags.push('高消费');
  if (rfm.fScore >= 4) tags.push('高频用户');
  if (rfm.rScore <= 2) tags.push('沉默用户');
  return tags;
}

interface UserBehaviorSummary {
  demographics: { ageRange: string; gender: string; city: string; incomeLevel: string };
  primaryDevice: string;
  activeHours: number[];
  avgSessionMinutes: number;
  searchCategories: string[];
  browseToCartRate: number;
  accountAgeDays: number;
}
interface UserTransactionSummary {
  categoryPurchases: Array<{ category: string; count: number }>;
  topBrands: string[];
  orderAmounts: number[];
  couponUsageRate: number;
  avgItemsPerOrder: number;
  returnRate: number;
}
分群运营与精准营销
基于 RFM 分层和用户画像的分群运营是电商精准营销的核心方法。每个用户群体的营销策略、触达渠道、激励力度和沟通调性都需要差异化定制。

精准营销的效果通过对照实验验证。将同一分群的用户随机分为实验组和对照组，实验组接收营销触达，对照组不接收，比较两组在目标指标（购买率、GMV、LTV）上的差异。增量效果（Incrementality）才是营销的真实贡献。


小结
    

  • RFM 模型通过最近购买、购买频率和消费金额三个维度将用户分为八大价值群体
    • 

  • 评分方式分为等分法和业务阈值法，等分法更客观，阈值法更贴合运营实际
    • 

  • 用户画像在 RFM 基础上扩展人口统计、消费偏好和行为特征，构建全方位用户认知
    • 

  • 分群运营为每个群体定制差异化策略，重要价值客户维持忠诚，流失客户控制投入
    • 

  • 精准营销效果通过增量实验验证，只有对照组差异才是营销的真实贡献
