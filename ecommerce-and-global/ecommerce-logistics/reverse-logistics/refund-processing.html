退款处理体系

核心概念

退款处理是退货流程的最终环节，直接影响用户满意度和平台资金安全。退款涉及金额计算（扣除运费、优惠分摊）、多渠道资金原路返回、对账核销和财务记账等复杂流程。电商平台需要在退款速度和风险控制之间取得平衡——退款越快用户体验越好，但过快的退款可能被恶意利用。

退款金额计算
退款金额计算是整个退款流程中最复杂的环节，需要处理优惠券分摊、满减分摊、积分抵扣和多商品退货等场景：


interface RefundCalculation {
  orderId: string;
  originalAmount: number;
  returnItems: ReturnLineItem[];
  discounts: DiscountAllocation[];
  shippingFee: number;
  pointsUsed: number;
}

interface ReturnLineItem {
  skuId: string;
  quantity: number;
  unitPrice: number;
  returnQuantity: number;
}

interface DiscountAllocation {
  type: 'coupon' | 'promotion' | 'points' | 'shipping';
  totalDiscount: number;
  allocatedToItems: Map<string, number>;
}

const calculateRefundAmount = (calc: RefundCalculation): RefundResult => {
  let refundAmount = 0;
  let refundPoints = 0;

  for (const item of calc.returnItems) {
    const itemTotal = item.unitPrice * item.returnQuantity;
    let itemDiscount = 0;

    for (const discount of calc.discounts) {
      if (discount.type !== 'shipping') {
        const skuDiscount = discount.allocatedToItems.get(item.skuId) ?? 0;
        const ratio = item.returnQuantity / item.quantity;
        itemDiscount += skuDiscount * ratio;
      }
    }

    refundAmount += itemTotal - itemDiscount;
  }

  const isFullReturn = calc.returnItems.every((item) => item.returnQuantity === item.quantity);

  if (isFullReturn) {
    refundAmount += calc.shippingFee;
  }

  if (calc.pointsUsed > 0) {
    const pointsRatio = refundAmount / calc.originalAmount;
    refundPoints = Math.floor(calc.pointsUsed * pointsRatio);
    refundAmount -= refundPoints * 0.01;
  }

  return {
    cashRefund: Math.round(refundAmount * 100) / 100,
    pointsRefund: refundPoints,
    shippingRefund: isFullReturn ? calc.shippingFee : 0,
    totalRefund: Math.round(refundAmount * 100) / 100,
  };
};

interface RefundResult {
  cashRefund: number;
  pointsRefund: number;
  shippingRefund: number;
  totalRefund: number;
}

interface RefundRequest {
  refundId: string;
  orderId: string;
  channel: string;

  currency: string;

  notifyUrl: string;
}

const executeRefund = async (request: RefundRequest): Promise<RefundResponse> => {
  const maxRetries = 3;
  let lastError: Error | null = null;

  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      const response = await callRefundApi(request);
      if (response.status === 'SUCCESS') {
        return response;
      }
      if (response.status === 'PROCESSING') {
        return response;
      }
      lastError = new Error(response.errorMessage);
    } catch (error) {
      lastError = error as Error;
      await sleep(attempt * 1000);
    }
  }

  return {
    status: 'FAILED',
    errorMessage: lastError?.message ?? '退款失败',
    refundId: request.refundId,
  };
};
对账与异常处理
退款对账确保平台账目与支付渠道记录一致，异常处理覆盖退款失败、重复退款和金额不一致等场景：


interface ReconciliationRecord {
  refundId: string;
  platformAmount: number;
  channelAmount: number;
  status: 'matched' | 'mismatch' | 'missing_platform' | 'missing_channel';
  resolvedAt: Date | null;
}

const reconcileRefunds = (
  platformRecords: Map<string, number>,
  channelRecords: Map<string, number>,
): ReconciliationRecord[] => {
  const results: ReconciliationRecord[] = [];
  const allIds = new Set([...platformRecords.keys(), ...channelRecords.keys()]);

  for (const refundId of allIds) {
    const platformAmt = platformRecords.get(refundId);
    const channelAmt = channelRecords.get(refundId);

    if (platformAmt !== undefined && channelAmt !== undefined) {
      results.push({
        refundId,
        platformAmount: platformAmt,
        channelAmount: channelAmt,
        status: Math.abs(platformAmt - channelAmt) < 0.01 ? 'matched' : 'mismatch',
        resolvedAt: null,
      });
    } else if (platformAmt !== undefined) {
      results.push({
        refundId,
        platformAmount: platformAmt,
        channelAmount: 0,
        status: 'missing_channel',
        resolvedAt: null,
      });
    } else {
      results.push({
        refundId,
        platformAmount: 0,
        channelAmount: channelAmt!,
        status: 'missing_platform',
        resolvedAt: null,
      });
    }
  }

  return results;
};
